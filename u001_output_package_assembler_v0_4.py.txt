class PackageError(RuntimeError):
    """Fail-closed package error with bounded machine evidence."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = machine_token(reason, 64)
        self.detail = machine_token(detail, 180)
        super().__init__(self.reason)


def machine_token(value: object, limit: int = 100) -> str:
    """Return one bounded low-bandwidth token."""

    text = re.sub(r"[^A-Z0-9_.:+-]+", "_", str(value).upper())
    return text.strip("_")[:limit] or "NA"


def canonical_json_bytes(value: object) -> bytes:
    """Serialize one deterministic JSON value."""

    return (
        json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"), allow_nan=False)
        + "\n"
    ).encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash a file in bounded blocks."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def write_text(path: Path, text: str) -> None:
    """Write UTF-8 text atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8", newline="\n") as stream:
        stream.write(text)
    os.replace(temporary, path)


def write_json(path: Path, value: Mapping[str, object]) -> None:
    """Write deterministic pretty JSON atomically."""

    write_text(path, json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n")


def write_csv(path: Path, header: Sequence[str], rows: Iterable[Sequence[object]]) -> None:
    """Write one deterministic UTF-8 CSV."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.writer(stream, lineterminator="\n")
        writer.writerow(header)
        writer.writerows(rows)
    os.replace(temporary, path)


def read_json(path: Path, label: str) -> Dict[str, object]:
    """Read one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise PackageError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise PackageError(label + "_INVALID", type(exc).__name__) from exc
    if not isinstance(value, dict):
        raise PackageError(label + "_NOT_OBJECT", path.name)
    return value


def read_header(path: Path) -> List[str]:
    """Read one CSV header."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        return [str(value) for value in next(csv.reader(stream), [])]


def read_control(path: Path, required: Sequence[str], label: str) -> List[Dict[str, str]]:
    """Read one bounded policy table and require declared columns."""

    if not path.is_file() or path.is_symlink():
        raise PackageError(label + "_MISSING", path.name)
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        header = list(reader.fieldnames or [])
        missing = [name for name in required if name not in header]
        if missing:
            raise PackageError(label + "_COLUMN", missing[0])
        return [
            {str(key): "" if value is None else str(value) for key, value in row.items()}
            for row in reader
        ]


def truthy(value: object) -> bool:
    """Interpret controlled CSV booleans."""

    return str(value).strip().upper() in {"1", "TRUE", "T", "YES", "Y"}


def count_csv_records(path: Path) -> int:
    """Count CSV records without loading the complete file."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        try:
            next(reader)
        except StopIteration:
            return 0
        return sum(1 for _ in reader)


def discover_inputs(root: Path) -> Dict[str, Path]:
    """Resolve exactly one of every required Gate artifact."""

    if not root.is_dir() or root.is_symlink():
        raise PackageError("DISCOVERY_ROOT", root.name)
    discovered: Dict[str, Path] = {}
    for basename in REQUIRED_INPUTS:
        matches = [
            path.resolve()
            for path in root.rglob(basename)
            if path.is_file() and not path.is_symlink()
        ]
        if len(matches) != 1:
            raise PackageError("INPUT_IDENTITY", f"{basename}:{len(matches)}")
        discovered[basename] = matches[0]
    return discovered


def require_columns(header: Sequence[str], required: Sequence[str], label: str) -> None:
    """Require a subset of columns."""

    missing = [name for name in required if name not in header]
    if missing:
        raise PackageError(label + "_COLUMN", missing[0])


def frame_groups(header: Sequence[str]) -> List[str]:
    """Validate D2 group structure and return dynamic canonical order."""

    if list(header[: len(D2_LEADING)]) != list(D2_LEADING):
        raise PackageError("FRAME_LEADING_COLUMNS", ".".join(header[:4]))
    tail = list(header[len(D2_LEADING) :])
    if not tail or len(tail) % len(D2_SUFFIXES) != 0:
        raise PackageError("FRAME_GROUP_WIDTH", len(tail))
    groups: List[str] = []
    for start in range(0, len(tail), len(D2_SUFFIXES)):
        block = tail[start : start + len(D2_SUFFIXES)]
        first = block[0]
        if not first.endswith("__value"):
            raise PackageError("FRAME_GROUP_VALUE", first)
        canonical = first[: -len("__value")]
        expected = [canonical + suffix for suffix in D2_SUFFIXES]
        if block != expected:
            raise PackageError("FRAME_GROUP_ORDER", canonical)
        groups.append(canonical)
    if len(groups) != len(set(groups)):
        raise PackageError("FRAME_GROUP_DUPLICATE", len(groups))
    return groups


def inspect_updates(path: Path) -> Dict[str, object]:
    """Stream D1 source and validate required schema and SourceRowIndex order."""

    header = read_header(path)
    require_columns(header, D1_REQUIRED, "UPDATE")
    index_position = header.index("SourceRowIndex")
    rows = 0
    previous: Optional[int] = None
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        next(reader, None)
        for row in reader:
            if index_position >= len(row):
                raise PackageError("UPDATE_ROW_WIDTH", rows)
            try:
                current = int(row[index_position].strip())
            except Exception as exc:
                raise PackageError("SOURCE_ROW_INDEX", rows) from exc
            if previous is not None and current < previous:
                raise PackageError("SOURCE_ORDER_DECREASE", current)
            previous = current
            rows += 1
    if rows == 0:
        raise PackageError("UPDATE_EMPTY", path.name)
    return {"Rows": rows, "Header": header, "LastSourceRowIndex": previous}


def parse_frame(value: object) -> int:
    """Parse an integer-valued FrameCount."""

    try:
        number = float(str(value).strip())
    except Exception as exc:
        raise PackageError("FRAME_NON_NUMERIC", value) from exc
    integer = int(number)
    if number != integer:
        raise PackageError("FRAME_NON_INTEGER", value)
    return integer


def inspect_frames(path: Path) -> Dict[str, object]:
    """Stream D2 source and validate canonical FrameCount and group structure."""

    header = read_header(path)
    groups = frame_groups(header)
    rows = 0
    previous: Optional[int] = None
    frame_min: Optional[int] = None
    frame_max: Optional[int] = None
    gap_count = 0
    largest_gap = 0
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        for row in reader:
            current = parse_frame(row.get("FrameCount", ""))
            if previous is not None:
                if current <= previous:
                    raise PackageError("FRAME_NOT_STRICT", current)
                if current > previous + 1:
                    gap_count += 1
                    largest_gap = max(largest_gap, current - previous - 1)
            if frame_min is None:
                frame_min = current
            frame_max = current
            previous = current
            rows += 1
    if rows == 0:
        raise PackageError("FRAME_EMPTY", path.name)
    return {
        "Rows": rows,
        "Header": header,
        "Groups": groups,
        "FrameMin": frame_min,
        "FrameMax": frame_max,
        "GapCount": gap_count,
        "LargestGap": largest_gap,
    }


def canonical_map(rows: Sequence[Mapping[str, str]], label: str) -> Dict[str, Mapping[str, str]]:
    """Create one exact CanonicalName map."""

    result: Dict[str, Mapping[str, str]] = {}
    for row in rows:
        name = str(row.get("CanonicalName", "")).strip()
        if not name or name in result:
            raise PackageError(label + "_CANONICAL_IDENTITY", name or "BLANK")
        result[name] = row
    return result


def control_tables(inputs: Mapping[str, Path]) -> Dict[str, object]:
    """Load and cross-check G2/G4/G5/G6 policy identity."""

    g2_all = read_control(
        inputs["u001_g2_selected_signals.csv"],
        ("Selected", "Channel", "CAN_ID_Hex", "MessageName", "SignalName", "CanonicalName", "Required"),
        "G2",
    )
    g2 = [row for row in g2_all if truthy(row.get("Selected", ""))]
    if not g2:
        raise PackageError("G2_NO_SELECTED", 0)
    g4 = read_control(
        inputs["u001_g4_signal_policy.csv"],
        ("CanonicalName", "ValueKind", "SignalRole", "UpdateMode", "StorageDtype", "ValidityRule"),
        "G4",
    )
    g5 = read_control(
        inputs["u001_g5_frame_policy.csv"],
        ("CanonicalName", "IncludeInFrameView", "WithinFramePolicy"),
        "G5",
    )
    g6 = read_control(
        inputs["u001_g6_hold_policy.csv"],
        ("SignalIndex", "CanonicalName", "HoldPolicy", "MaxAgeFrames"),
        "G6",
    )
    maps = {"G2": canonical_map(g2, "G2"), "G4": canonical_map(g4, "G4"), "G5": canonical_map(g5, "G5"), "G6": canonical_map(g6, "G6")}
    selected_order = [row["CanonicalName"].strip() for row in g2]
    if set(maps["G4"]) != set(selected_order) or set(maps["G5"]) != set(selected_order):
        raise PackageError("POLICY_MEMBERSHIP", f"G2{len(g2)}G4{len(g4)}G5{len(g5)}")
    included_order = [row["CanonicalName"].strip() for row in g5 if truthy(row.get("IncludeInFrameView", ""))]
    g6_order = [row["CanonicalName"].strip() for row in g6]
    if included_order != g6_order:
        raise PackageError("G5_G6_ORDER", f"G5{len(included_order)}G6{len(g6_order)}")
    for name in selected_order:
        g4_row = maps["G4"][name]
        g5_row = maps["G5"][name]
        for field in ("ValueKind", "SignalRole"):
            if g5_row.get(field, "").strip() and g4_row.get(field, "").strip() != g5_row.get(field, "").strip():
                raise PackageError("G4_G5_SEMANTIC", f"{name}:{field}")
    for index, row in enumerate(g6, start=1):
        signal_index = row.get("SignalIndex", "").strip()
        if signal_index != f"S{index}":
            raise PackageError("G6_SIGNAL_INDEX", signal_index)
        name = row["CanonicalName"].strip()
        for field in ("ValueKind", "SignalRole", "UpdateMode"):
            expected = maps["G4"][name].get(field, "").strip()
            observed = row.get(field, "").strip()
            if observed and expected != observed:
                raise PackageError("G4_G6_SEMANTIC", f"{name}:{field}")
    return {
        "G2Rows": g2,
        "G4Rows": g4,
        "G5Rows": g5,
        "G6Rows": g6,
        "Maps": maps,
        "SelectedOrder": selected_order,
        "IncludedOrder": included_order,
    }


def file_record(path: Path, include_rows: bool = False) -> Dict[str, object]:
    """Create one local-config input identity record."""

    record: Dict[str, object] = {
        "Path": str(path.resolve()),
        "Basename": path.name,
        "Bytes": path.stat().st_size,
        "Sha256": sha256_file(path),
    }
    if path.suffix.lower() == ".csv":
        record["Header"] = read_header(path)
        if include_rows:
            record["Rows"] = count_csv_records(path)
    return record


def safe_config_projection(config: Mapping[str, object]) -> Dict[str, object]:
    """Remove local paths and fingerprint field before canonical hashing."""

    projected = json.loads(json.dumps(config))
    projected.pop("ConfigFingerprint", None)
    inputs = projected.get("Inputs")
    if isinstance(inputs, dict):
        for record in inputs.values():
            if isinstance(record, dict):
                record.pop("Path", None)
    return projected


def config_fingerprint(config: Mapping[str, object]) -> str:
    """Fingerprint nonlocal configuration semantics."""

    return hashlib.sha256(canonical_json_bytes(safe_config_projection(config))).hexdigest()


def make_config(
    root: Path,
    out: Path,
    dataset_id: str,
    run_id: str,
    segment_id: str,
    source_id: str,
    identity_status: str,
) -> Dict[str, object]:
    """Discover accepted inputs and write a dynamic local config."""

    inputs = discover_inputs(root)
    controls = control_tables(inputs)
    frame_info = inspect_frames(inputs["u001_g6_frame_values.csv"])
    if frame_info["Groups"] != controls["IncludedOrder"]:
        raise PackageError("FRAME_POLICY_GROUP_ORDER", f"D2{len(frame_info['Groups'])}G6{len(controls['IncludedOrder'])}")
    records = {
        basename: file_record(
            path,
            basename in {"u001_g4_semantic_validity_updates.csv", "u001_g6_frame_values.csv"},
        )
        for basename, path in inputs.items()
    }
    review_columns = ["FrameCount"] + [name + "__value" for name in controls["IncludedOrder"]]
    config: Dict[str, object] = {
        "Program": {"ProgramId": PROGRAM_ID, "ProgramVersion": PROGRAM_VERSION},
        "Contract": {
            "ContractVersion": CONTRACT_VERSION,
            "ImplementationProfile": IMPLEMENTATION_PROFILE,
            "ProcessingMode": PROCESSING_MODE,
            "SchemaVersion": SCHEMA_VERSION,
        },
        "Identity": {
            "DatasetId": dataset_id,
            "RunId": run_id,
            "SegmentId": segment_id,
            "SourceId": source_id,
            "IdentityStatus": identity_status,
        },
        "Inputs": records,
        "ReviewView": {"Columns": review_columns},
        "Serialization": {"CSVEncoding": "UTF-8", "JSONEncoding": "UTF-8", "MissingValue": "", "XLSX": False},
        "Safety": {
            "RequireExactSixFiles": True,
            "RejectAbsolutePathsInManifest": True,
            "PreserveAcceptedUpdateBytes": True,
            "PreserveAcceptedFrameBytes": True,
            "RejectOverwrite": True,
        },
        "DynamicEvidence": {
            "SelectedSignalCount": len(controls["SelectedOrder"]),
            "FrameSignalCount": len(controls["IncludedOrder"]),
            "FrameRows": frame_info["Rows"],
        },
    }
    config["ConfigFingerprint"] = config_fingerprint(config)
    if out.exists():
        raise PackageError("CONFIG_OUTPUT_EXISTS", out.name)
    write_json(out, config)
    return config


def validate_config(config: Mapping[str, object]) -> Dict[str, Path]:
    """Validate config identity and all source file fingerprints."""

    program = config.get("Program")
    if not isinstance(program, dict) or str(program.get("ProgramVersion")) != PROGRAM_VERSION:
        raise PackageError("CONFIG_PROGRAM_VERSION", program)
    observed_fingerprint = str(config.get("ConfigFingerprint", ""))
    if observed_fingerprint != config_fingerprint(config):
        raise PackageError("CONFIG_FINGERPRINT", observed_fingerprint[:12])
    inputs = config.get("Inputs")
    if not isinstance(inputs, dict) or set(inputs) != set(REQUIRED_INPUTS):
        raise PackageError("CONFIG_INPUT_SET", 0 if not isinstance(inputs, dict) else len(inputs))
    paths: Dict[str, Path] = {}
    for basename in REQUIRED_INPUTS:
        record = inputs.get(basename)
        if not isinstance(record, dict):
            raise PackageError("CONFIG_INPUT_RECORD", basename)
        path = Path(str(record.get("Path", "")))
        if not path.is_file() or path.is_symlink() or path.name != basename:
            raise PackageError("CONFIG_INPUT_PATH", basename)
        if path.stat().st_size != record.get("Bytes") or sha256_file(path) != record.get("Sha256"):
            raise PackageError("INPUT_MUTATED", basename)
        paths[basename] = path.resolve()
    return paths


def pick(row: Mapping[str, str], *names: str) -> str:
    """Return the first present policy field without inference."""

    for name in names:
        if name in row and row.get(name) is not None:
            return str(row.get(name, "")).strip()
    return ""


def build_catalog(controls: Mapping[str, object], review_columns: Sequence[str]) -> List[Dict[str, str]]:
    """Merge G2/G4/G5/G6 by exact CanonicalName into D3 rows."""

    maps = controls["Maps"]
    rows: List[Dict[str, str]] = []
    for name in controls["SelectedOrder"]:
        g2 = maps["G2"][name]
        g4 = maps["G4"][name]
        g5 = maps["G5"][name]
        g6 = maps["G6"].get(name, {})
        unit = pick(g4, "Unit") or pick(g2, "Unit")
        notes = ";".join(
            part for part in (pick(g2, "Notes"), pick(g4, "Notes"), pick(g5, "Notes"), pick(g6, "Notes")) if part
        )
        record = {column: "" for column in D3_REQUIRED + D3_RESERVED}
        record.update(
            {
                "CanonicalName": name,
                "CAN_ID": pick(g2, "CAN_ID", "CAN_ID_Hex") or pick(g4, "CAN_ID", "CAN_ID_Hex"),
                "Channel": pick(g2, "Channel"),
                "MessageName": pick(g2, "MessageName"),
                "SignalName": pick(g2, "SignalName"),
                "Unit": unit,
                "ValueKind": pick(g4, "ValueKind"),
                "SignalRole": pick(g4, "SignalRole"),
                "UpdateMode": pick(g4, "UpdateMode"),
                "StorageDtype": pick(g4, "StorageDtype"),
                "StartBit": pick(g2, "StartBit"),
                "BitLength": pick(g2, "BitLength"),
                "ByteOrder": pick(g2, "ByteOrder"),
                "Signed": pick(g2, "Signed"),
                "Scale": pick(g2, "Scale"),
                "Offset": pick(g2, "Offset"),
                "DBCMinimum": pick(g4, "DBCMinimum"),
                "DBCMaximum": pick(g4, "DBCMaximum"),
                "EnumerationMap": pick(g2, "EnumerationMap"),
                "ValidityRule": pick(g4, "ValidityRule"),
                "ValiditySourceSignal": pick(g4, "ValiditySourceSignal"),
                "ValidityExpression": pick(g4, "ValidityExpression"),
                "WithinFramePolicy": pick(g5, "WithinFramePolicy"),
                "HoldPolicy": pick(g6, "HoldPolicy") or "NONE",
                "MaxAgeFrames": pick(g6, "MaxAgeFrames"),
                "NominalRateHz": pick(g6, "NominalRateHz"),
                "RequiredForAnalysis": pick(g2, "Required"),
                "IncludeInFrameView": pick(g5, "IncludeInFrameView"),
                "IncludeInReviewView": "1" if name + "__value" in review_columns else "0",
                "Notes": notes,
                "SourceType": "CAN",
                "SourceAdapter": "CAN_CSV",
                "SourceFormat": "MATLAB_CAN_CSV",
                "ClockDomain": "FRAMECOUNT_VIDEO",
                "OriginalUnit": unit,
                "CanonicalUnit": unit,
                "UnitConversionRule": "IDENTITY" if unit else "",
            }
        )
        rows.append(record)
    return rows


def flatten_json(value: object, path: Tuple[str, ...] = ()) -> Iterator[Tuple[Tuple[str, ...], object]]:
    """Yield scalar JSON leaves with their paths."""

    if isinstance(value, dict):
        for key, child in value.items():
            yield from flatten_json(child, path + (str(key),))
    elif isinstance(value, list):
        for index, child in enumerate(value):
            yield from flatten_json(child, path + (str(index),))
    else:
        yield path, value


def normalized_key(value: object) -> str:
    """Normalize a metric/path token for conservative lookup."""

    return re.sub(r"[^A-Z0-9]+", "", str(value).upper())


def quality_sources(inputs: Mapping[str, Path]) -> List[Tuple[str, List[Tuple[Tuple[str, ...], object]]]]:
    """Load accepted quality JSON leaves with Gate provenance."""

    result = []
    for basename in REQUIRED_INPUTS:
        if "quality.json" not in basename:
            continue
        gate_match = re.search(r"_g(\d)_", basename)
        gate = "G" + gate_match.group(1) if gate_match else "UNKNOWN"
        result.append((gate, list(flatten_json(read_json(inputs[basename], basename)))))
    return result


def find_quality_value(
    sources: Sequence[Tuple[str, Sequence[Tuple[Tuple[str, ...], object]]]],
    metric: str,
    canonical: str = "",
) -> Tuple[str, str, str, str]:
    """Find one unambiguous accepted quality leaf or report unavailable."""

    metric_key = normalized_key(metric)
    canonical_key = normalized_key(canonical)
    candidates: List[Tuple[str, object]] = []
    for gate, leaves in sources:
        for path, value in leaves:
            if not path or normalized_key(path[-1]) != metric_key:
                continue
            if canonical and canonical_key not in {normalized_key(part) for part in path}:
                continue
            candidates.append((gate, value))
    distinct = {(gate, json.dumps(value, ensure_ascii=False, sort_keys=True)) for gate, value in candidates}
    values = {serialized for _, serialized in distinct}
    if len(values) == 1 and candidates:
        gate, value = candidates[0]
        return str(value), gate, "SOURCE", "Accepted quality JSON leaf"
    if candidates:
        return "", "MULTI", "NOT_AVAILABLE", "Ambiguous quality JSON leaves; not guessed"
    return "", "NA", "NOT_AVAILABLE", "Metric not exposed by accepted Gate quality JSON"


def build_quality_rows(
    inputs: Mapping[str, Path],
    catalog: Sequence[Mapping[str, str]],
    update_info: Mapping[str, object],
    frame_info: Mapping[str, object],
    review_column_count: int,
) -> List[List[object]]:
    """Create tidy D4 rows while keeping zero distinct from unavailable."""

    sources = quality_sources(inputs)
    computed = {
        "selected_signal_count": len(catalog),
        "decoded_update_count": update_info["Rows"],
        "frame_row_count": frame_info["Rows"],
        "frame_min": frame_info["FrameMin"],
        "frame_max": frame_info["FrameMax"],
        "frame_non_numeric_count": 0,
        "missing_frame_gap_count": frame_info["GapCount"],
        "largest_frame_gap": frame_info["LargestGap"],
    }
    rows: List[List[object]] = []
    for metric in GLOBAL_METRICS:
        if metric in computed:
            value, gate, status, notes = computed[metric], "G7", "PASS", "Validated during package assembly"
        else:
            value, gate, status, notes = find_quality_value(sources, metric)
        rows.append(["GLOBAL", "", metric, value, "", gate, status, notes])
    package_values = {
        "package_file_count": 6,
        "update_source_hash_match": 1,
        "frame_source_hash_match": 1,
        "catalog_signal_count": len(catalog),
        "frame_view_signal_count": len(frame_info["Groups"]),
        "review_view_column_count": review_column_count,
        "absolute_path_leak_count": 0,
        "package_error_count": 0,
    }
    for metric in PACKAGE_GLOBAL_METRICS:
        rows.append(["GLOBAL", "", metric, package_values[metric], "", "G7", "PASS", "Package invariant"])
    for record in catalog:
        canonical = record["CanonicalName"]
        for metric in SIGNAL_METRICS:
            if metric == "HoldPolicy":
                value, gate, status, notes = record["HoldPolicy"], "G6", "SOURCE", "Exact policy"
            elif metric == "MaxAgeFrames":
                value, gate, status, notes = record["MaxAgeFrames"], "G6", "SOURCE", "Exact policy; blank means not applicable"
            else:
                value, gate, status, notes = find_quality_value(sources, metric, canonical)
            rows.append(["SIGNAL", canonical, metric, value, "", gate, status, notes])
    return rows


def copy_binary_verified(source: Path, destination: Path, expected_hash: str) -> None:
    """Binary-copy one accepted data artifact and verify identity."""

    shutil.copyfile(source, destination)
    if destination.stat().st_size != source.stat().st_size:
        raise PackageError("BINARY_COPY_SIZE", destination.name)
    if sha256_file(destination) != expected_hash or sha256_file(source) != expected_hash:
        raise PackageError("BINARY_COPY_HASH", destination.name)


def build_review_view(frame_path: Path, out: Path, columns: Sequence[str]) -> int:
    """Stream an exact FrameCount-led subset of D2 into D6."""

    header = read_header(frame_path)
    if not columns or columns[0] != "FrameCount" or len(columns) != len(set(columns)):
        raise PackageError("REVIEW_CONFIG_COLUMNS", len(columns))
    missing = [name for name in columns if name not in header]
    if missing:
        raise PackageError("REVIEW_COLUMN_MISSING", missing[0])
    temporary = out.with_name(out.name + ".tmp")
    rows = 0
    with frame_path.open("r", encoding="utf-8-sig", newline="") as source, temporary.open(
        "w", encoding="utf-8", newline=""
    ) as destination:
        reader = csv.DictReader(source)
        writer = csv.DictWriter(destination, fieldnames=list(columns), extrasaction="ignore", lineterminator="\n")
        writer.writeheader()
        for row in reader:
            writer.writerow({name: row.get(name, "") for name in columns})
            rows += 1
    os.replace(temporary, out)
    return rows


def contains_absolute_path(value: object) -> bool:
    """Detect workstation paths in a portable manifest."""

    text = json.dumps(value, ensure_ascii=False)
    patterns = (r"[A-Za-z]:\\", r"/Users/", r"/home/", r"\\\\")
    return any(re.search(pattern, text) for pattern in patterns)


def unique_child(root: Path, stem: str) -> Path:
    """Return the first deterministic nonexisting child path."""

    candidate = root / stem
    if not candidate.exists():
        return candidate
    for index in range(1, 1000):
        candidate = root / f"{stem}_{index:03d}"
        if not candidate.exists():
            return candidate
    raise PackageError("OUTPUT_SUFFIX_EXHAUSTED", stem)


def build_package(config_path: Path, out_root: Path) -> Tuple[str, Dict[str, object]]:
    """Build and validate one dynamic six-file package."""

    started = time.monotonic()
    config = read_json(config_path, "CONFIG")
    inputs = validate_config(config)
    controls = control_tables(inputs)
    update_info = inspect_updates(inputs["u001_g4_semantic_validity_updates.csv"])
    frame_info = inspect_frames(inputs["u001_g6_frame_values.csv"])
    if frame_info["Groups"] != controls["IncludedOrder"]:
        raise PackageError("FRAME_G6_GROUP_ORDER", f"F{len(frame_info['Groups'])}G{len(controls['IncludedOrder'])}")
    review_section = config.get("ReviewView")
    if not isinstance(review_section, dict) or not isinstance(review_section.get("Columns"), list):
        raise PackageError("REVIEW_CONFIG", "COLUMNS")
    review_columns = [str(value) for value in review_section["Columns"]]
    expected_review = ["FrameCount"] + [name + "__value" for name in controls["IncludedOrder"]]
    if review_columns != expected_review:
        raise PackageError("REVIEW_POLICY_ORDER", len(review_columns))

    input_records = config["Inputs"]
    expected_update_rows = input_records["u001_g4_semantic_validity_updates.csv"].get("Rows")
    expected_frame_rows = input_records["u001_g6_frame_values.csv"].get("Rows")
    if update_info["Rows"] != expected_update_rows or frame_info["Rows"] != expected_frame_rows:
        raise PackageError("DYNAMIC_ROW_COUNT", f"U{update_info['Rows']}R{frame_info['Rows']}")

    out_root.mkdir(parents=True, exist_ok=True)
    package_path = unique_child(out_root, "U001_PACKAGE")
    temporary = unique_child(out_root, ".U001_PACKAGE_INCOMPLETE")
    temporary.mkdir()
    try:
        d1 = temporary / FINAL_PACKAGE_FILES[0]
        d2 = temporary / FINAL_PACKAGE_FILES[1]
        copy_binary_verified(
            inputs["u001_g4_semantic_validity_updates.csv"],
            d1,
            str(input_records["u001_g4_semantic_validity_updates.csv"]["Sha256"]),
        )
        copy_binary_verified(
            inputs["u001_g6_frame_values.csv"],
            d2,
            str(input_records["u001_g6_frame_values.csv"]["Sha256"]),
        )

        catalog = build_catalog(controls, review_columns)
        write_csv(
            temporary / FINAL_PACKAGE_FILES[2],
            D3_REQUIRED + D3_RESERVED,
            ([row[column] for column in D3_REQUIRED + D3_RESERVED] for row in catalog),
        )
        quality_rows = build_quality_rows(inputs, catalog, update_info, frame_info, len(review_columns))
        write_csv(temporary / FINAL_PACKAGE_FILES[3], D4_COLUMNS, quality_rows)
        review_rows = build_review_view(d2, temporary / FINAL_PACKAGE_FILES[5], review_columns)
        if review_rows != frame_info["Rows"]:
            raise PackageError("REVIEW_ROW_COUNT", review_rows)

        output_records = []
        for name in (FINAL_PACKAGE_FILES[0], FINAL_PACKAGE_FILES[1], FINAL_PACKAGE_FILES[2], FINAL_PACKAGE_FILES[3], FINAL_PACKAGE_FILES[5]):
            path = temporary / name
            output_records.append({"Basename": name, "Bytes": path.stat().st_size, "Sha256": sha256_file(path)})
        identity = config.get("Identity")
        contract = config.get("Contract")
        manifest = {
            "ContractVersion": contract.get("ContractVersion") if isinstance(contract, dict) else CONTRACT_VERSION,
            "ImplementationProfile": IMPLEMENTATION_PROFILE,
            "ProcessingMode": PROCESSING_MODE,
            "SchemaVersion": SCHEMA_VERSION,
            "DatasetId": identity.get("DatasetId") if isinstance(identity, dict) else "",
            "RunId": identity.get("RunId") if isinstance(identity, dict) else "",
            "SegmentId": identity.get("SegmentId") if isinstance(identity, dict) else "",
            "SourceId": identity.get("SourceId") if isinstance(identity, dict) else "",
            "IdentityStatus": identity.get("IdentityStatus") if isinstance(identity, dict) else "",
            "ProgramName": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "RunTimestampUTC": datetime.now(timezone.utc).isoformat(),
            "InputFiles": [
                {
                    "Basename": basename,
                    "Bytes": record["Bytes"],
                    "Sha256": record["Sha256"],
                }
                for basename, record in sorted(input_records.items())
            ],
            "SourceAdapter": {"Name": "CAN_CSV", "ClockDomain": "FRAMECOUNT_VIDEO"},
            "SelectedSignalCount": len(catalog),
            "SelectedSignals": [row["CanonicalName"] for row in catalog],
            "FrameCountDefinition": {
                "CanonicalKey": "FrameCount",
                "AssignmentOwner": "U001",
                "StartFrameCount": frame_info["FrameMin"],
                "EndFrameCount": frame_info["FrameMax"],
            },
            "Policies": [
                {
                    "CanonicalName": row["CanonicalName"],
                    "ValueKind": row["ValueKind"],
                    "SignalRole": row["SignalRole"],
                    "ValidityRule": row["ValidityRule"],
                    "WithinFramePolicy": row["WithinFramePolicy"],
                    "HoldPolicy": row["HoldPolicy"],
                    "MaxAgeFrames": row["MaxAgeFrames"],
                }
                for row in catalog
            ],
            "ConfigFingerprint": config["ConfigFingerprint"],
            "Serialization": {"CSV": "UTF-8", "JSON": "UTF-8", "MissingValue": ""},
            "OutputFiles": output_records,
            "Warnings": [
                "Radar is a provisional engineering reference, not ground truth.",
                "FrameCount is the authoritative video-domain key, not a high-resolution timestamp.",
            ],
            "ExecutionDurationSeconds": round(time.monotonic() - started, 6),
            "Status": "SUCCESS",
        }
        if contains_absolute_path(manifest):
            raise PackageError("MANIFEST_ABSOLUTE_PATH", 1)
        write_json(temporary / FINAL_PACKAGE_FILES[4], manifest)

        actual = {path.name for path in temporary.iterdir() if path.is_file()}
        if actual != set(FINAL_PACKAGE_FILES):
            raise PackageError("PACKAGE_FILE_SET", len(actual))
        if any((temporary / name).stat().st_size == 0 for name in FINAL_PACKAGE_FILES):
            raise PackageError("PACKAGE_EMPTY_FILE", "SIX")
        os.replace(temporary, package_path)
    except Exception:
        shutil.rmtree(temporary, ignore_errors=True)
        raise

    capsule = (
        f"UDP001G7V04 N-U{update_info['Rows']}-R{frame_info['Rows']}"
        f"-S{len(catalog)}-K{len(frame_info['Groups'])}-V{frame_info['Rows']}"
        f"-W{len(review_columns)}-F6-X0 C-0"
    )
    capsule_path = unique_child(out_root, "udp001_g7v04_return_capsule.txt")
    write_text(capsule_path, capsule + "\n")
    report = {
        "CompletionCode": 0,
        "PackagePath": str(package_path.resolve()),
        "CapsulePath": str(capsule_path.resolve()),
        "UpdateRows": update_info["Rows"],
        "FrameRows": frame_info["Rows"],
        "CatalogSignals": len(catalog),
        "FrameSignals": len(frame_info["Groups"]),
        "ReviewColumns": len(review_columns),
        "Capsule": capsule,
    }
    return capsule, report


def snapshot_small_tree(root: Path) -> Dict[str, Tuple[int, str]]:
    """Snapshot accepted U001 program/control files for recovery isolation."""

    result: Dict[str, Tuple[int, str]] = {}
    if not root.is_dir() or root.is_symlink():
        raise PackageError("PROTECTED_ROOT", root.name)
    for path in sorted(root.rglob("*")):
        if path.is_symlink():
            raise PackageError("PROTECTED_SYMLINK", path.name)
        if path.is_file():
            result[path.relative_to(root).as_posix()] = (path.stat().st_size, sha256_file(path))
    return result


def recover_b14(
    udp001_root: Path,
    failed_run_root: Path,
    out_root: Path,
    dataset_id: str,
    run_id: str,
    segment_id: str,
    source_id: str,
) -> Tuple[str, str]:
    """Rebuild B14 from accepted G1-G6 artifacts without using legacy G7."""

    if out_root.exists():
        raise PackageError("RECOVERY_OUTPUT_EXISTS", out_root.name)
    manifest_path = failed_run_root / "03_RUN" / "CONTROL_OUTPUT" / "u001_g8a2_orchestration_manifest.json"
    old_manifest = read_json(manifest_path, "B14_MANIFEST")
    stages = old_manifest.get("Stages")
    passed = sum(isinstance(item, dict) and item.get("Success") is True for item in stages) if isinstance(stages, list) else 0
    if old_manifest.get("Status") != "FAILURE" or passed != 8:
        raise PackageError("B14_FAILURE_IDENTITY", f"{old_manifest.get('Status')}:{passed}")
    protected_roots = (udp001_root / "01_PROGRAMS", udp001_root / "02_CONTROL")
    protected_before = [snapshot_small_tree(root) for root in protected_roots]

    out_root.mkdir(parents=True)
    config_path = out_root / "u001_g7v04_package_config.json"
    make_config(
        failed_run_root / "03_RUN",
        config_path,
        dataset_id,
        run_id,
        segment_id,
        source_id,
        "CONTROLLED_PROFILE_SMOKE",
    )
    capsule, report = build_package(config_path, out_root / "G7V04_BUILD")
    protected_after = [snapshot_small_tree(root) for root in protected_roots]
    if protected_before != protected_after:
        raise PackageError("ACCEPTED_U001_CHANGED", "PROGRAM_OR_CONTROL")

    line1 = (
        f"B14R3 N-G7V04-F6-C{report['CatalogSignals']}-V{report['FrameSignals']}"
        f"-U{report['UpdateRows']}-R{report['FrameRows']}-H1-X0 C-0"
    )
    line2 = "B14R3I N-NEW1-OLD0-A0-E1-X0 C-0"
    write_json(
        out_root / "u22b1b14r3_recovery_manifest_v0_1.json",
        {
            "CompletionCode": 0,
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "ExecutionMode": "INDEPENDENT_G7V04_FROM_ACCEPTED_G1_G6",
            "LegacyG7Executed": False,
            "AcceptedU001Changed": False,
            "OldSuccessfulStageCount": passed,
            "G7V04Capsule": capsule,
            "ReturnCapsule": [line1, line2],
        },
    )
    write_text(out_root / "u22b1b14r3_return_capsule.txt", line1 + "\n" + line2 + "\n")
    return line1, line2

        for index, _ in enumerate(signals):

        frame_rows.append(row)
    write_csv(root / "u001_g6_frame_values.csv", frame_header, frame_rows)


def self_test() -> int:
    """Run independent non-five-signal package tests."""

    checks: List[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            raise AssertionError(name)
        checks.append(name)

    with tempfile.TemporaryDirectory() as temp_name:
        root = Path(temp_name)
        source = root / "source"
        source.mkdir()
        make_synthetic_gate_inputs(source, included_count=3)
        check("T01_REQUIRED_INPUTS", len(discover_inputs(source)) == len(REQUIRED_INPUTS))
        controls = control_tables(discover_inputs(source))
        check("T02_SELECTED_DYNAMIC", len(controls["SelectedOrder"]) == 4)
        check("T03_INCLUDED_DYNAMIC", len(controls["IncludedOrder"]) == 3)
        check("T04_NOT_FIVE", len(controls["IncludedOrder"]) != 5)
        info = inspect_frames(source / "u001_g6_frame_values.csv")
        check("T05_FRAME_ROWS", info["Rows"] == 3)
        check("T06_FRAME_GROUPS", info["Groups"] == ["SIG_1", "SIG_2", "SIG_3"])
        check("T07_FRAME_MIN", info["FrameMin"] == 10)
        check("T08_FRAME_MAX", info["FrameMax"] == 12)
        updates = inspect_updates(source / "u001_g4_semantic_validity_updates.csv")
        check("T09_UPDATE_ROWS", updates["Rows"] == 3)

        config_path = root / "config.json"
        config = make_config(source, config_path, "D", "R", "S", "SRC", "SYNTHETIC")
        check("T10_CONFIG", config_path.is_file())
        check("T11_FINGERPRINT", config["ConfigFingerprint"] == config_fingerprint(config))
        check("T12_REVIEW_WIDTH", len(config["ReviewView"]["Columns"]) == 4)
        capsule, report = build_package(config_path, root / "build")
        check("T13_CAPSULE", capsule.endswith("C-0"))
        check("T14_PACKAGE", Path(report["PackagePath"]).is_dir())
        check("T15_FILE_COUNT", len(list(Path(report["PackagePath"]).iterdir())) == 6)

        check("T21_D2_HASH", sha256_file(Path(report["PackagePath"]) / "u001_frame_values.csv") == sha256_file(source / "u001_g6_frame_values.csv"))
        catalog_rows = read_control(Path(report["PackagePath"]) / "u001_signal_catalog.csv", D3_REQUIRED, "D3")
        check("T22_D3_ROWS", len(catalog_rows) == 4)
        check("T23_D3_INCLUDED", sum(truthy(row["IncludeInFrameView"]) for row in catalog_rows) == 3)
        quality_rows = read_control(Path(report["PackagePath"]) / "u001_quality_report.csv", D4_COLUMNS, "D4")

        duplicate = root / "duplicate"
        shutil.copytree(source, duplicate)
        nested = duplicate / "nested"
        nested.mkdir()

            duplicate_failed = True

            mutation_failed = exc.reason == "INPUT_MUTATED"
        check("T32_MUTATION_FAIL", mutation_failed)

    own_source = Path(__file__).read_text(encoding="utf-8")
    legacy_failure_token = "FRAME_SIGNAL_COUNT_NOT_" + str(5)
    check("T33_NO_LEGACY_GUARD", legacy_failure_token not in own_source)
    import ast

    own_tree = ast.parse(own_source)
    imported_roots = {
        alias.name.split(".")[0]
        for node in ast.walk(own_tree)
        if isinstance(node, (ast.Import, ast.ImportFrom))
        for alias in node.names
    }
    check("T34_NO_PANDAS", "pandas" not in imported_roots)
    check("T35_SIX_FILES", len(FINAL_PACKAGE_FILES) == 6)

    check("T40_DYNAMIC_POLICY", "IncludedOrder" in own_source)
    if len(checks) != 40:

def build_parser() -> argparse.ArgumentParser:
    """Build command-line modes."""

    parser = argparse.ArgumentParser(description="Dynamic U001 G7 v0.4 package assembler")
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--make-config", action="store_true")
    mode.add_argument("--build", action="store_true")
    mode.add_argument("--recover-b14", action="store_true")
    parser.add_argument("--root")
    parser.add_argument("--config")
    parser.add_argument("--out")
    parser.add_argument("--out-root")
    parser.add_argument("--udp001-root")
    parser.add_argument("--failed-run-root")
    parser.add_argument("--dataset-id", default="UDP001_SMOKE_DATASET")
    parser.add_argument("--run-id", default="UDP001_G7V04_RUN")
    parser.add_argument("--segment-id", default="SEGMENT_001")
    parser.add_argument("--source-id", default="CAN_SOURCE_001")
    parser.add_argument("--identity-status", default="PROVISIONAL_SMOKE")
    return parser


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Program entry point with integrated failure evidence."""

    args = build_parser().parse_args(argv)
    try:
        if args.self_test:
            return self_test()
        if args.make_config:
            if not args.root or not args.out:
                raise PackageError("ARGUMENT_MISSING", "ROOT_OR_OUT")
            config = make_config(
                Path(args.root), Path(args.out), args.dataset_id, args.run_id,
                args.segment_id, args.source_id, args.identity_status,
            )
            print(
                f"UDP001G7V04C N-S{config['DynamicEvidence']['SelectedSignalCount']}"
                f"-K{config['DynamicEvidence']['FrameSignalCount']}"
                f"-R{config['DynamicEvidence']['FrameRows']}-X0 C-0"
            )
            return 0
        if args.build:
            if not args.config or not args.out_root:
                raise PackageError("ARGUMENT_MISSING", "CONFIG_OR_OUT_ROOT")
            capsule, _ = build_package(Path(args.config), Path(args.out_root))
            print(capsule)
            return 0
        if not args.udp001_root or not args.failed_run_root or not args.out_root:
            raise PackageError("ARGUMENT_MISSING", "RECOVERY_PATH")
        lines = recover_b14(
            Path(args.udp001_root), Path(args.failed_run_root), Path(args.out_root),
            args.dataset_id, args.run_id, args.segment_id, args.source_id,
        )
        print(lines[0])
        print(lines[1])
        return 0
    except Exception as exc:
        if isinstance(exc, PackageError):
            reason, detail = exc.reason, exc.detail
        else:
            reason, detail = machine_token(type(exc).__name__), machine_token(str(exc), 180)
        line1 = f"B14R3E K-{reason}-X1 C-7"
        line2 = f"B14R3D D-{detail} C-0"
        if args.out_root:
            try:
                failure_root = Path(args.out_root)
                failure_root.mkdir(parents=True, exist_ok=True)
                write_json(
                    failure_root / "u001_g7v04_failure_diagnostic.json",
                    {
                        "CompletionCode": 7,
                        "ProgramId": PROGRAM_ID,
                        "ProgramVersion": PROGRAM_VERSION,
                        "Reason": reason,
                        "Detail": detail,
                        "Mode": "RECOVER_B14" if args.recover_b14 else "BUILD",
                        "Traceback": traceback.format_exc(),
                        "ReturnCapsule": [line1, line2],
                    },
                )
                write_text(failure_root / "u001_g7v04_failure_capsule.txt", line1 + "\n" + line2 + "\n")
            except Exception:
                pass
        print(line1)
        print(line2)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
