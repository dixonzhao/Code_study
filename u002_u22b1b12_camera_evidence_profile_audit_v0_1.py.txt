class AuditError(RuntimeError):
    """A deterministic, fail-closed audit error."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = token(reason, 64)
        self.detail = token(detail, 160)
        super().__init__(self.reason)


def token(value: object, limit: int = 80) -> str:
    """Return one bounded machine-code token."""

    cleaned = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper())
    return cleaned.strip("_")[:limit] or "NA"


def normalized_name(value: object) -> str:
    """Normalize a signal name for exact identity matching."""

    return re.sub(r"[^A-Z0-9]+", "", str(value).upper())


def normalized_can_id(value: object) -> str:
    """Normalize one CAN identifier to uppercase 0x hexadecimal."""

    text = str(value).strip()
    if not text:
        raise AuditError("CAN_ID_BLANK")
    try:
        number = int(text, 16)
    except ValueError as exc:
        raise AuditError("CAN_ID_INVALID", text) from exc
    if number < 0:
        raise AuditError("CAN_ID_NEGATIVE", text)
    return f"0x{number:X}"


def sha256_file(path: Path) -> str:
    """Hash one file without loading it into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def read_csv(path: Path, required: Sequence[str], label: str) -> List[Dict[str, str]]:
    """Read one CSV and enforce its required schema."""

    if not path.is_file() or path.is_symlink():
        raise AuditError(label + "_MISSING", path.name)
    try:
        with path.open("r", encoding="utf-8-sig", newline="") as stream:
            reader = csv.DictReader(stream)
            header = reader.fieldnames or []
            missing = [name for name in required if name not in header]
            if missing:
                raise AuditError(label + "_COLUMN", missing[0])
            return [dict(row) for row in reader]
    except AuditError:
        raise
    except Exception as exc:
        raise AuditError(label + "_READ", type(exc).__name__) from exc


def read_json(path: Path, label: str) -> Dict[str, object]:
    """Read one JSON object."""

    if not path.is_file() or path.is_symlink():
        raise AuditError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise AuditError(label + "_JSON", type(exc).__name__) from exc
    if not isinstance(value, dict):
        raise AuditError(label + "_TYPE", path.name)
    return value


def ci_get(mapping: Mapping[str, object], key: str, default: object = None) -> object:
    """Read a top-level JSON field case-insensitively."""

    wanted = key.casefold()
    for actual, value in mapping.items():
        if str(actual).casefold() == wanted:
            return value
    return default


def completion_is_zero(manifest: Mapping[str, object]) -> bool:
    """Accept only an explicit successful U001 profile manifest."""

    value = ci_get(manifest, "completion_code", None)
    if isinstance(value, bool):
        return not value
    try:
        return int(str(value).strip()) == 0
    except (TypeError, ValueError):
        return False


def expected_requests() -> List[Dict[str, object]]:
    """Return the approved B12 evidence request in deterministic order."""

    camera_features = (
        ("D", "Distance", "CORE", 1, "longitudinal geometry and continuity"),
        ("L", "LateralPosition", "CORE", 1, "target bearing geometry"),
        ("R", "RelativeSpeed", "CORE", 1, "relative-motion continuity"),
        ("O", "ObjectDetectionType", "SEMANTIC", 0, "tracking/detection state evidence"),
        ("T", "ObjectType", "SEMANTIC", 0, "object-class consistency evidence"),
        ("S", "SwitchingFlag", "SEMANTIC", 0, "declared target transition evidence"),
        ("M", "MessageCounter", "INTEGRITY", 0, "message freshness/integrity only"),
        ("Y", "LateralSpeed", "OPTIONAL", 0, "lateral-motion continuity"),
        ("W", "Width", "OPTIONAL", 0, "object-shape consistency"),
    )
    requests: List[Dict[str, object]] = []
    order = 1
    for code, signal, requirement, required, use in camera_features:
        for can_id, role in (("0x221", "CAMERA_1"), ("0x251", "CAMERA_3")):
            requests.append(
                {
                    "RequestOrder": order,
                    "Group": "CAMERA_PAIR",
                    "PairCode": code,
                    "SourceRole": role,
                    "Channel": "2",
                    "CAN_ID_Hex": can_id,
                    "SignalName": signal,
                    "RequirementClass": requirement,
                    "Required": required,
                    "EngineeringUse": use,
                }
            )
            order += 1

    radar_features = (
        ("D", "DistanceToObject", 1, "comparison distance; not ground truth"),
        ("F", "PrecedingVehicleDetectionFlag", 1, "radar measurement availability"),
        ("R", "RelativeSpeed", 0, "supporting relative-motion evidence"),
    )
    for code, signal, required, use in radar_features:
        requests.append(
            {
                "RequestOrder": order,
                "Group": "RADAR_SUPPORT",
                "PairCode": code,
                "SourceRole": "RADAR",
                "Channel": "1",
                "CAN_ID_Hex": "0x380",
                "SignalName": signal,
                "RequirementClass": "SUPPORT",
                "Required": required,
                "EngineeringUse": use,
            }
        )
        order += 1

    requests.append(
        {
            "RequestOrder": order,
            "Group": "FRAME_DOMAIN",
            "PairCode": "F",
            "SourceRole": "FRAME_KEY",
            "Channel": "2",
            "CAN_ID_Hex": "0x620",
            "SignalName": "FrameCount",
            "RequirementClass": "FRAME_KEY",
            "Required": 1,
            "EngineeringUse": "authoritative video/CAN alignment key",
        }
    )
    return requests


def profile_candidates(case_root: Path) -> Tuple[List[Dict[str, object]], int]:
    """Find complete U001 discovery profiles and collapse byte-identical copies."""

    if not case_root.is_dir():
        raise AuditError("U001_CASE_ROOT_MISSING", case_root.name)

    candidates: List[Dict[str, object]] = []
    for manifest_path in sorted(case_root.rglob(PROFILE_MANIFEST_NAME)):
        root = manifest_path.parent
        inventory_path = root / INVENTORY_NAME
        batch_path = root / BATCH_MAP_NAME
        if not inventory_path.is_file() or not batch_path.is_file():
            continue
        manifest = read_json(manifest_path, "PROFILE_MANIFEST")
        if not completion_is_zero(manifest):
            continue
        try:
            relative_root = root.relative_to(case_root).as_posix() or "."
        except ValueError:
            relative_root = root.name
        candidates.append(
            {
                "root": root,
                "relative_root": relative_root,
                "manifest": manifest,
                "inventory_hash": sha256_file(inventory_path),
                "batch_hash": sha256_file(batch_path),
            }
        )

    if not candidates:
        raise AuditError("COMPLETE_DISCOVERY_PROFILE_NOT_FOUND", case_root.name)

    by_identity: Dict[Tuple[str, str], List[Dict[str, object]]] = {}
    for item in candidates:
        identity = (str(item["inventory_hash"]), str(item["batch_hash"]))
        by_identity.setdefault(identity, []).append(item)

    if len(by_identity) != 1:
        raise AuditError("MULTIPLE_DISTINCT_DISCOVERY_PROFILES", len(by_identity))

    copies = next(iter(by_identity.values()))
    copies.sort(key=lambda item: (len(str(item["relative_root"])), str(item["relative_root"])))
    return copies, len(candidates)


def selected_inventory_rows(rows: Sequence[Mapping[str, str]]) -> List[Dict[str, str]]:
    """Return selected rows after validating basic identity fields."""

    selected: List[Dict[str, str]] = []
    seen: set[str] = set()
    for source in rows:
        if str(source.get("Selected", "")).strip() not in {"1", "true", "TRUE"}:
            continue
        row = dict(source)
        canonical = row["CanonicalName"].strip()
        if not canonical:
            raise AuditError("SELECTED_CANONICAL_BLANK")
        if canonical in seen:
            raise AuditError("SELECTED_CANONICAL_DUPLICATE", canonical)
        seen.add(canonical)
        row["Channel"] = str(int(str(row["Channel"]).strip()))
        row["CAN_ID_Hex"] = normalized_can_id(row["CAN_ID_Hex"])
        selected.append(row)
    if not selected:
        raise AuditError("SELECTED_INVENTORY_EMPTY")
    return selected


def batch_lookup(rows: Sequence[Mapping[str, str]]) -> Dict[str, str]:
    """Create CanonicalName-to-BatchId lookup with duplicate protection."""

    result: Dict[str, str] = {}
    for row in rows:
        canonical = str(row["CanonicalName"]).strip()
        if not canonical:
            raise AuditError("BATCH_CANONICAL_BLANK")
        if canonical in result:
            raise AuditError("BATCH_CANONICAL_DUPLICATE", canonical)
        result[canonical] = str(row["BatchId"]).strip()
    return result


def match_request(
    request: Mapping[str, object],
    rows: Sequence[Mapping[str, str]],
    batches: Mapping[str, str],
) -> Dict[str, object]:
    """Match one explicit Channel/CAN-ID/Signal identity."""

    channel = str(request["Channel"])
    can_id = normalized_can_id(request["CAN_ID_Hex"])
    wanted_name = normalized_name(request["SignalName"])
    same_message = [
        row
        for row in rows
        if str(row["Channel"]) == channel and row["CAN_ID_Hex"] == can_id
    ]
    exact = [row for row in same_message if normalized_name(row["SignalName"]) == wanted_name]
    fuzzy = [
        row
        for row in same_message
        if wanted_name in normalized_name(row["SignalName"])
        or normalized_name(row["SignalName"]) in wanted_name
    ]

    if len(exact) == 1:
        status = "EXACT"
        chosen: Optional[Mapping[str, str]] = exact[0]
    elif len(exact) > 1:
        status = "AMBIGUOUS"
        chosen = None
    elif fuzzy:
        status = "FUZZY_ONLY"
        chosen = None
    else:
        status = "MISSING"
        chosen = None

    suggestions = "|".join(
        f"{row['SignalName']}@{row['CanonicalName']}" for row in fuzzy[:5]
    )
    output: Dict[str, object] = {
        "RequestOrder": request["RequestOrder"],
        "Group": request["Group"],
        "PairCode": request["PairCode"],
        "SourceRole": request["SourceRole"],
        "ExpectedChannel": channel,
        "ExpectedCAN_ID_Hex": can_id,
        "ExpectedSignalName": request["SignalName"],
        "RequirementClass": request["RequirementClass"],
        "MatchStatus": status,
        "MatchCount": len(exact),
        "CanonicalName": "",
        "ActualMessageName": "",
        "ActualSignalName": "",
        "Unit": "",
        "ProposedValueKind": "",
        "ProposedSignalRole": "",
        "ProposedUpdateMode": "",
        "BatchId": "",
        "SuggestedRequired": request["Required"],
        "Suggestions": suggestions,
        "EngineeringUse": request["EngineeringUse"],
    }
    if chosen is not None:
        canonical = str(chosen["CanonicalName"])
        output.update(
            {
                "CanonicalName": canonical,
                "ActualMessageName": chosen["MessageName"],
                "ActualSignalName": chosen["SignalName"],
                "Unit": chosen["Unit"],
                "ProposedValueKind": chosen["ProposedValueKind"],
                "ProposedSignalRole": chosen["ProposedSignalRole"],
                "ProposedUpdateMode": chosen["ProposedUpdateMode"],
                "BatchId": batches.get(canonical, ""),
            }
        )
    return output


def camera_pair_codes(rows: Sequence[Mapping[str, object]]) -> str:
    """Encode nine Camera pair completeness states in fixed feature order."""

    result = ""
    for code in ("D", "L", "R", "O", "T", "S", "M", "Y", "W"):
        pair = [row for row in rows if row["Group"] == "CAMERA_PAIR" and row["PairCode"] == code]
        if any(row["MatchStatus"] == "AMBIGUOUS" for row in pair):
            result += "3"
        else:
            result += str(sum(row["MatchStatus"] == "EXACT" for row in pair))
    return result


def radar_codes(rows: Sequence[Mapping[str, object]]) -> str:
    """Encode three Radar support identities as exact/not-exact states."""

    result = ""
    for code in ("D", "F", "R"):
        item = next(row for row in rows if row["Group"] == "RADAR_SUPPORT" and row["PairCode"] == code)
        if item["MatchStatus"] == "AMBIGUOUS":
            result += "3"
        else:
            result += "1" if item["MatchStatus"] == "EXACT" else "0"
    return result


def write_csv(path: Path, columns: Sequence[str], rows: Iterable[Mapping[str, object]]) -> None:
    """Write deterministic UTF-8 CSV."""

    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(columns), lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({name: row.get(name, "") for name in columns})


def controlled_request_rows(audit_rows: Sequence[Mapping[str, object]]) -> List[Dict[str, object]]:
    """Convert exact matches into a non-executable controlled-profile request."""

    rows: List[Dict[str, object]] = []
    for row in audit_rows:
        if row["MatchStatus"] != "EXACT":
            continue
        rows.append(
            {
                "RequestOrder": row["RequestOrder"],
                "RequestedProfileId": "CAM13_RADAR_EVIDENCE_V0_1",
                "Group": row["Group"],
                "PairCode": row["PairCode"],
                "SourceRole": row["SourceRole"],
                "Selected": 1,
                "Required": row["SuggestedRequired"],
                "Channel": row["ExpectedChannel"],
                "CAN_ID_Hex": row["ExpectedCAN_ID_Hex"],
                "MessageName": row["ActualMessageName"],
                "SignalName": row["ActualSignalName"],
                "CanonicalName": row["CanonicalName"],
                "Unit": row["Unit"],
                "ProposedValueKind": row["ProposedValueKind"],
                "ProposedSignalRole": row["ProposedSignalRole"],
                "ProposedUpdateMode": row["ProposedUpdateMode"],
                "DiscoveryBatchId": row["BatchId"],
                "PolicyStatus": "REQUEST_ONLY_NOT_APPROVED",
                "EngineeringUse": row["EngineeringUse"],
            }
        )
    return rows


def audit(case_root: Path, output_root: Path) -> Tuple[str, str]:
    """Run the metadata audit and return its two short machine-code lines."""

    copies, physical_profile_count = profile_candidates(case_root)
    selected_profile = copies[0]
    profile_root = Path(selected_profile["root"])
    inventory = read_csv(profile_root / INVENTORY_NAME, INVENTORY_COLUMNS, "INVENTORY")
    batches = read_csv(profile_root / BATCH_MAP_NAME, BATCH_COLUMNS, "BATCH_MAP")
    selected = selected_inventory_rows(inventory)
    batch_by_canonical = batch_lookup(batches)
    audit_rows = [
        match_request(request, selected, batch_by_canonical)
        for request in expected_requests()
    ]

    exact_count = sum(row["MatchStatus"] == "EXACT" for row in audit_rows)
    fuzzy_count = sum(row["MatchStatus"] == "FUZZY_ONLY" for row in audit_rows)
    ambiguous_count = sum(row["MatchStatus"] == "AMBIGUOUS" for row in audit_rows)
    missing_count = sum(row["MatchStatus"] == "MISSING" for row in audit_rows)
    unique_batches = len({row["BatchId"] for row in audit_rows if row["BatchId"]})
    eligible_count = len(selected)
    catalog_count = len(inventory)
    frame_row = next(row for row in audit_rows if row["Group"] == "FRAME_DOMAIN")
    frame_code = 1 if frame_row["MatchStatus"] == "EXACT" else 0
    camera_codes = camera_pair_codes(audit_rows)
    radar = radar_codes(audit_rows)

    if output_root.exists() and any(output_root.iterdir()):
        raise AuditError("OUTPUT_NOT_EMPTY", output_root.name)
    output_root.mkdir(parents=True, exist_ok=True)

    request_rows = controlled_request_rows(audit_rows)
    temp_root = output_root / ".partial_b12"
    if temp_root.exists():
        shutil.rmtree(temp_root)
    temp_root.mkdir()
    try:
        write_csv(temp_root / AUDIT_NAME, AUDIT_COLUMNS, audit_rows)
        write_csv(temp_root / REQUEST_NAME, REQUEST_COLUMNS, request_rows)
        manifest = {
            "CompletionCode": 0,
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": VERSION,
            "Purpose": "METADATA_ONLY_CAMERA_EVIDENCE_PROFILE_AUDIT",
            "RequestedProfileId": "CAM13_RADAR_EVIDENCE_V0_1",
            "SourceProfileId": str(ci_get(selected_profile["manifest"], "profile_id", "UNKNOWN")),
            "SourceProfileRelativeRoot": str(selected_profile["relative_root"]),
            "SourceInventorySha256": str(selected_profile["inventory_hash"]),
            "SourceBatchMapSha256": str(selected_profile["batch_hash"]),
            "PhysicalProfileCopies": physical_profile_count,
            "EquivalentProfileCopies": len(copies),
            "CatalogRows": catalog_count,
            "EligibleRows": eligible_count,
            "RequestedIdentities": len(audit_rows),
            "ExactMatches": exact_count,
            "FuzzyOnlyMatches": fuzzy_count,
            "AmbiguousMatches": ambiguous_count,
            "MissingMatches": missing_count,
            "UniqueDiscoveryBatches": unique_batches,
            "CameraPairCodes": camera_codes,
            "RadarCodes": radar,
            "FrameKeyCode": frame_code,
            "DecisionBoundary": (
                "REQUEST_ONLY; does not approve a U001 Controlled Profile and does not change U002 S1"
            ),
            "Outputs": {},
        }
        for name in (AUDIT_NAME, REQUEST_NAME):
            path = temp_root / name
            manifest["Outputs"][name] = {
                "Sha256": sha256_file(path),
                "Bytes": path.stat().st_size,
            }
        (temp_root / MANIFEST_NAME).write_text(
            json.dumps(manifest, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
            encoding="utf-8",
        )
        line1 = (
            f"B12 N-P{physical_profile_count}-D{len(copies)}-C{catalog_count}"
            f"-E{eligible_count}-M{exact_count}-B{unique_batches}-F{fuzzy_count}"
            f"-A{ambiguous_count}-X{missing_count} C-0"
        )
        line2 = f"B12P C-{camera_codes}-R{radar}-F{frame_code} C-0"
        (temp_root / CAPSULE_NAME).write_text(line1 + "\n" + line2 + "\n", encoding="utf-8")
        for source in sorted(temp_root.iterdir()):
            source.replace(output_root / source.name)
    finally:
        if temp_root.exists():
            shutil.rmtree(temp_root)
    return line1, line2


def write_diagnostic(path: Path, error: BaseException, context: Mapping[str, object]) -> None:
    """Write detailed machine-readable failure evidence without another probe."""

    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        profile_evidence: List[Dict[str, object]] = []
        case_text = str(context.get("u001_case_root", ""))
        case_root = Path(case_text)
        if case_root.is_dir():
            for manifest_path in sorted(case_root.rglob(PROFILE_MANIFEST_NAME))[:50]:
                root = manifest_path.parent
                profile_evidence.append(
                    {
                        "RelativeRoot": root.relative_to(case_root).as_posix(),
                        "ManifestBytes": manifest_path.stat().st_size,
                        "InventoryPresent": (root / INVENTORY_NAME).is_file(),
                        "BatchMapPresent": (root / BATCH_MAP_NAME).is_file(),
                    }
                )
        payload = {
            "CompletionCode": 7,
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": VERSION,
            "ErrorType": type(error).__name__,
            "Reason": getattr(error, "reason", token(type(error).__name__)),
            "Detail": getattr(error, "detail", token(str(error), 160)),
            "Context": {key: str(value) for key, value in context.items()},
            "DiscoveryProfileEvidence": profile_evidence,
            "Traceback": traceback.format_exception(type(error), error, error.__traceback__)[-8:],
        }
        path.write_text(
            json.dumps(payload, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
            encoding="utf-8",
        )
    except Exception:
        pass

                    "PolicyStatus": "AUTO_DISCOVERY_PROVISIONAL",
                }
            )
            batch_rows.append(batch)
    excluded = {name: "" for name in INVENTORY_COLUMNS}
    excluded.update(

        )
        + "\n",

    )


def self_test() -> None:
    """Run deterministic tests without touching company data."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise AssertionError(f"self_test_{len(checks) + 1}")
        checks.append(True)

    check(normalized_can_id("0x221") == "0x221")
    check(normalized_can_id("251") == "0x251")
    check(normalized_name("Lateral_Position") == "LATERALPOSITION")
    requests = expected_requests()
    check(len(requests) == 22)
    check(sum(row["Group"] == "CAMERA_PAIR" for row in requests) == 18)
    check(sum(row["Group"] == "RADAR_SUPPORT" for row in requests) == 3)
    check(sum(row["Group"] == "FRAME_DOMAIN" for row in requests) == 1)
    check(sum(int(row["Required"]) for row in requests) == 9)

    with tempfile.TemporaryDirectory(prefix="u22b1b12_") as temporary:
        temporary_root = Path(temporary)
        case_root = temporary_root / "CASE"
        profile_root = case_root / "RUN" / "02_PROFILE_CONFIG" / "AUTO_DISCOVERY_ALL_V0_2"
        synthetic_profile(profile_root)
        duplicate_root = case_root / "COPY"
        shutil.copytree(profile_root, duplicate_root)
        copies, physical = profile_candidates(case_root)
        check(physical == 2)
        check(len(copies) == 2)
        check(Path(copies[0]["root"]).is_dir())

        inventory = read_csv(profile_root / INVENTORY_NAME, INVENTORY_COLUMNS, "I")
        selected = selected_inventory_rows(inventory)
        check(len(inventory) == 23)
        check(len(selected) == 22)
        batch_rows = read_csv(profile_root / BATCH_MAP_NAME, BATCH_COLUMNS, "B")
        batches = batch_lookup(batch_rows)
        check(len(batches) == 21)

        audit_rows = [match_request(item, selected, batches) for item in requests]
        check(len(audit_rows) == 22)
        check(all(row["MatchStatus"] == "EXACT" for row in audit_rows))
        check(camera_pair_codes(audit_rows) == "222222222")
        check(radar_codes(audit_rows) == "111")
        check(len(controlled_request_rows(audit_rows)) == 22)

        output_root = temporary_root / "OUT"
        line1, line2 = audit(case_root, output_root)
        check(line1.startswith("B12 N-P2-D2-C23-E22-M22-B1-F0-A0-X0 C-0"))
        check(line2 == "B12P C-222222222-R111-F1 C-0")
        check((output_root / AUDIT_NAME).is_file())
        check((output_root / REQUEST_NAME).is_file())
        check((output_root / MANIFEST_NAME).is_file())
        check((output_root / CAPSULE_NAME).is_file())
        manifest = read_json(output_root / MANIFEST_NAME, "O")
        check(manifest["CompletionCode"] == 0)
        check(manifest["ExactMatches"] == 22)
        check(manifest["MissingMatches"] == 0)
        check(manifest["CameraPairCodes"] == "222222222")
        check(manifest["DecisionBoundary"].startswith("REQUEST_ONLY"))
        capsule = (output_root / CAPSULE_NAME).read_text(encoding="utf-8").splitlines()
        check(capsule == [line1, line2])

        selected_without_last = selected[:-1]
        missing = match_request(requests[-1], selected_without_last, batches)
        check(missing["MatchStatus"] == "MISSING")

    check(len(checks) == 32)
    print("B12-ST N-T32-X0 C-0")


def parser() -> argparse.ArgumentParser:
    """Build the command-line interface."""

    result = argparse.ArgumentParser(description=PROGRAM_ID)
    mode = result.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--audit", action="store_true")
    result.add_argument("--u001-case-root", type=Path)
    result.add_argument("--out", type=Path)
    result.add_argument("--diagnostic", type=Path, default=DEFAULT_DIAGNOSTIC)
    return result


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test or one read-only company profile audit."""

    arguments = parser().parse_args(argv)
    if arguments.self_test:
        self_test()
        return 0

    context = {
        "u001_case_root": arguments.u001_case_root or "MISSING",
        "out": arguments.out or "MISSING",
    }
    try:
        if arguments.u001_case_root is None:
            raise AuditError("U001_CASE_ROOT_ARGUMENT_MISSING")
        if arguments.out is None:
            raise AuditError("OUTPUT_ARGUMENT_MISSING")
        line1, line2 = audit(arguments.u001_case_root, arguments.out)
        print(line1)
        print(line2)
        return 0
    except Exception as exc:
        write_diagnostic(arguments.diagnostic, exc, context)
        reason = getattr(exc, "reason", token(type(exc).__name__))
        detail = getattr(exc, "detail", token(str(exc), 72))
        print(f"B12E K-{reason} D-{detail} C-7")
        return 7


if __name__ == "__main__":
    sys.exit(main())
