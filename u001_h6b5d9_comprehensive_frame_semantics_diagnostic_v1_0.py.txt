DOMINANCE_THRESHOLD_PERMILLE = 900


class ProbeError(RuntimeError):
    """Fail-closed hypothesis-probe error."""


@dataclass
class FrameTracker:
    """Streaming metrics for one candidate FrameCount interpretation."""

    name: str
    row_count: int = 0
    observation_count: int = 0
    change_count: int = 0
    decrease_count: int = 0
    delta_one_count: int = 0
    delta_positive_count: int = 0
    delta_large_count: int = 0
    minimum: Optional[int] = None
    maximum: Optional[int] = None
    previous_row_value: Optional[int] = None
    previous_changed_value: Optional[int] = None
    unique_values: set[int] = field(default_factory=set)

    def observe(self, value: int) -> None:
        self.row_count += 1
        self.observation_count += 1
        self.unique_values.add(value)
        self.minimum = value if self.minimum is None else min(self.minimum, value)
        self.maximum = value if self.maximum is None else max(self.maximum, value)
        if self.previous_row_value is None or value != self.previous_row_value:
            self.change_count += 1
            if self.previous_changed_value is not None:
                delta = value - self.previous_changed_value
                if delta < 0:
                    self.decrease_count += 1
                elif delta > 0:
                    self.delta_positive_count += 1
                    if delta == 1:
                        self.delta_one_count += 1
                    elif delta > 1:
                        self.delta_large_count += 1
            self.previous_changed_value = value
        self.previous_row_value = value

    def summary(self) -> Dict[str, Any]:
        delta_one_permille = (
            int(round(1000 * self.delta_one_count / self.delta_positive_count))
            if self.delta_positive_count else 0
        )
        exact = self.change_count == HISTORICAL_FRAME_ROWS
        near = (
            abs(self.change_count - HISTORICAL_FRAME_ROWS) <= max(10, HISTORICAL_FRAME_ROWS // 100)
            and self.decrease_count == 0
            and delta_one_permille >= 950
        )
        return {
            "Policy": self.name,
            "RowsDecoded": self.row_count,
            "ObservationCount": self.observation_count,
            "ChangedFrameObservationCount": self.change_count,
            "UniqueFrameCount": len(self.unique_values),
            "MinimumFrameCount": self.minimum,
            "MaximumFrameCount": self.maximum,
            "FrameDecreaseCount": self.decrease_count,
            "PositiveDeltaCount": self.delta_positive_count,
            "DeltaOneCount": self.delta_one_count,
            "DeltaGreaterThanOneCount": self.delta_large_count,
            "DeltaOnePermille": delta_one_permille,
            "HistoricalExactMatch": exact,
            "HistoricalNearMatch": near,
        }


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def write_json(path: Path, value: Dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        raise ProbeError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise ProbeError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise ProbeError(label + "_ROOT_NOT_OBJECT")
    return value


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise ProbeError("U_UDP001_ROOT_NOT_FOUND")
    return root


def version_key(path: Path) -> Tuple[int, ...]:
    values = re.findall(r"\d+", path.parent.name)
    return tuple(int(value) for value in values) if values else (0,)


def locate_adapter_core(root: Path) -> Path:
    base = root / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc"
    candidates = list(base.glob("*/u001_matlab_can_csv_dbc_core_v1_0.py"))
    if not candidates:
        raise ProbeError("ADAPTER_CORE_NOT_FOUND")
    return max(candidates, key=version_key)


def load_module(path: Path) -> Any:
    name = "u001_h6b5d9_adapter_core"
    spec = importlib.util.spec_from_file_location(name, path)
    if spec is None or spec.loader is None:
        raise ProbeError("ADAPTER_CORE_SPEC")
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

def configured_inputs(request: Dict[str, Any]) -> Tuple[Path, Dict[int, Path]]:
    source: Optional[Path] = None
    dbcs: Dict[int, Path] = {}
    inputs = request.get("Inputs")
    if not isinstance(inputs, list):
        raise ProbeError("REQUEST_INPUTS")
    for item in inputs:
        if not isinstance(item, dict):
            continue
        raw = item.get("Path", item.get("FilePath"))
        if not isinstance(raw, str) or not raw.strip():
            continue
        path = Path(raw)
        kind = str(item.get("Kind", "")).upper()
        if kind == "MATLAB_CAN_CSV":
            source = path
        elif kind == "DBC_CHANNEL":
            dbcs[int(item["Channel"])] = path
    if source is None or not source.is_file():
        raise ProbeError("RAW_SOURCE_NOT_FOUND")
    if len(dbcs) < 2 or any(not path.is_file() for path in dbcs.values()):
        raise ProbeError("DBC_INPUTS")
    return source, dbcs


def frame_definition(core: Any, source: Path, dbcs: Dict[int, Path]) -> Tuple[Any, Any, Tuple[str, ...]]:
    text_format = core.detect_text_format(source)
    groups = {(group.channel, group.can_id): group for group in core.discover_source_message_groups(text_format.columns)}
    matches: List[Tuple[Any, Any]] = []
    for channel, dbc_path in sorted(dbcs.items()):
        parsed = core.parse_dbc(dbc_path, channel)
        for message in parsed.messages:
            group = groups.get((channel, message.can_id))
            if group is None or not group.complete:
                continue
            for signal in message.signals:
                token = re.sub(r"[^a-z0-9]", "", signal.signal_name.casefold())
                if "frame" in token and ("count" in token or "counter" in token):
                    matches.append((signal, group))
    if len(matches) != 1:
        raise ProbeError(f"FRAME_DEFINITION_COUNT_{len(matches)}")
    signal, group = matches[0]
    names = tuple(group.byte_columns_by_index[index] for index in range(group.dlc))
    return text_format, signal, names


def derive_sentinels(d8: Dict[str, Any]) -> Tuple[List[str], List[int], List[int]]:
    audit = d8.get("CurrentRawAudit")
    if not isinstance(audit, dict):
        raise ProbeError("D8_RAW_AUDIT")
    rows = audit.get("PerByte")
    if not isinstance(rows, list) or len(rows) != 8:
        raise ProbeError("D8_PER_BYTE_COUNT")
    sentinels: List[str] = []
    counts: List[int] = []
    dominance: List[int] = []
    for expected_index, row in enumerate(rows):
        if not isinstance(row, dict) or int(row.get("ByteIndex", -1)) != expected_index:
            raise ProbeError("D8_BYTE_ORDER")
        invalid_count = int(row.get("InvalidByteCellCount", 0))
        top = row.get("TopInvalidTokens")
        if invalid_count <= 0:
            sentinels.append("")
            counts.append(0)
            dominance.append(0)
            continue
        if not isinstance(top, list) or not top or not isinstance(top[0], dict):
            raise ProbeError(f"D8_BYTE_{expected_index}_INVALID_TOKEN_EVIDENCE")
        token = str(top[0].get("Token", ""))
        count = int(top[0].get("Count", 0))
        permille = int(round(1000 * count / invalid_count)) if invalid_count else 0
        if not token:
            raise ProbeError(f"D8_BYTE_{expected_index}_EMPTY_TOP_TOKEN")
        sentinels.append(token)
        counts.append(count)
        dominance.append(permille)
    return sentinels, counts, dominance


def header_positions(source: Path, text_format: Any, names: Sequence[str]) -> Tuple[List[str], List[int]]:
    with source.open("r", encoding=text_format.encoding, errors="replace", newline="") as stream:
        for _ in range(text_format.header_row_index):
            next(stream, None)
        reader = csv.reader(stream, delimiter=text_format.delimiter)
        header = next(reader, None)
    if header is None:
        raise ProbeError("SOURCE_HEADER")
    normalized = [str(value).strip() for value in header]
    positions: List[int] = []
    for name in names:
        matches = [index for index, value in enumerate(normalized) if value == name]
        if len(matches) != 1:
            raise ProbeError(f"BYTE_HEADER_COUNT_{name}_{len(matches)}")
        positions.append(matches[0])
    return normalized, positions


def iter_rows(source: Path, text_format: Any, expected_header: Sequence[str]) -> Iterator[List[str]]:
    with source.open("r", encoding=text_format.encoding, errors="replace", newline="") as stream:
        for _ in range(text_format.header_row_index):
            next(stream, None)
        reader = csv.reader(stream, delimiter=text_format.delimiter)
        header = next(reader, None)
        if header is None or [str(value).strip() for value in header] != list(expected_header):
            raise ProbeError("SOURCE_HEADER_CHANGED")
        for row in reader:
            if len(row) < len(expected_header):
                row.extend([""] * (len(expected_header) - len(row)))
            yield row


def decode_frame(core: Any, payload: Sequence[int], signal: Any) -> Optional[int]:
    try:
        _, physical = core.decode_physical_value(payload, signal)
        if not math.isfinite(physical) or not float(physical).is_integer():
            return None
        return int(physical)
    except Exception:
        return None


def normalized_header_key(value: str) -> str:
    return re.sub(r"[^a-z0-9]", "", value.casefold())


def inspect_h2_frame_rows(path: Path) -> Dict[str, Any]:
    tracker = FrameTracker("CURRENT_H2_FRAME_EVENTS")
    row_count = 0
    with path.open("r", encoding="utf-8-sig", errors="replace", newline="") as stream:
        reader = csv.DictReader(stream)
        if reader.fieldnames is None:
            raise ProbeError("H2_HEADER")
        required = {"SourceChannel", "SourceMessageId", "SourceSignalName", "PhysicalValue"}
        if not required.issubset(set(reader.fieldnames)):
            raise ProbeError("H2_FRAME_COLUMNS")
        for row in reader:
            if (
                str(row.get("SourceChannel", "")).strip() == "2"
                and str(row.get("SourceMessageId", "")).strip().casefold() == "0x620"
                and str(row.get("SourceSignalName", "")).strip() == "FrameCount"
            ):
                row_count += 1
                try:
                    numeric = float(str(row.get("PhysicalValue", "")).strip())
                    if math.isfinite(numeric) and numeric.is_integer():
                        tracker.observe(int(numeric))
                except ValueError:
                    continue
    summary = tracker.summary()
    summary["MatchedH2Rows"] = row_count
    return summary


def comprehensive_scan(
    core: Any,
    source: Path,
    text_format: Any,
    signal: Any,
    byte_names: Sequence[str],
    sentinels: Sequence[str],
    sentinel_evidence_reliable: bool,
) -> Dict[str, Any]:
    """Evaluate strict, dominant-sentinel carry and any-missing carry in one pass."""

    header, positions = header_positions(source, text_format, byte_names)
    byte_position_set = set(positions)
    direct_candidates = [
        (index, name, f"{index}:{name}")
        for index, name in enumerate(header)
        if index not in byte_position_set and "framecount" in normalized_header_key(name)
    ]
    time_candidates = [
        index for index, name in enumerate(header)
        if normalized_header_key(name) in {"timeus", "sourcetimeus", "microseconds"}
    ]
    time_position = time_candidates[0] if time_candidates else None

    strict = FrameTracker("STRICT_ALL_8_BYTES_VALID")
    dominant = FrameTracker("DOMINANT_SENTINEL_OR_BLANK_CARRY")
    any_carry = FrameTracker("ANY_INVALID_OR_BLANK_CARRY")
    direct_trackers = {
        key: FrameTracker("DIRECT_COLUMN:" + key)
        for _, _, key in direct_candidates
    }
    dominant_state: List[Optional[int]] = [None] * len(positions)
    any_state: List[Optional[int]] = [None] * len(positions)
    stats: Counter[str] = Counter()
    valid_mask_counts: Counter[str] = Counter()
    invalid_mask_counts: Counter[str] = Counter()
    observed_invalid_tokens: List[Counter[str]] = [Counter() for _ in positions]
    previous_time: Optional[float] = None

    for source_row, row in enumerate(iter_rows(source, text_format, header)):
        stats["SourceRowCount"] += 1
        strict_payload: List[int] = []
        strict_valid = True
        dominant_row_blocked = False
        dominant_pending: Dict[int, int] = {}
        valid_mask = 0
        invalid_mask = 0

        for byte_index, position in enumerate(positions):
            token = str(row[position]).strip()
            if not token:
                stats["BlankCellCount"] += 1
                strict_valid = False
                continue
            try:
                value = core.parse_byte_cell(token)
            except Exception:
                strict_valid = False
                invalid_mask |= 1 << byte_index
                observed_invalid_tokens[byte_index][token] += 1
                stats["InvalidCellCount"] += 1
                if sentinels[byte_index] and token == sentinels[byte_index]:
                    stats["DominantSentinelCellCount"] += 1
                else:
                    stats["UnapprovedInvalidCellCount"] += 1
                    dominant_row_blocked = True
                continue

            valid_mask |= 1 << byte_index
            stats["ValidByteCellCount"] += 1
            strict_payload.append(value)
            dominant_pending[byte_index] = value
            any_state[byte_index] = value

        valid_mask_counts[f"0x{valid_mask:02X}"] += 1
        invalid_mask_counts[f"0x{invalid_mask:02X}"] += 1

        if strict_valid and len(strict_payload) == len(positions):
            frame = decode_frame(core, strict_payload, signal)
            if frame is not None:
                strict.observe(frame)
            else:
                stats["StrictFrameDecodeInvalid"] += 1

        if not dominant_row_blocked:
            for byte_index, value in dominant_pending.items():
                dominant_state[byte_index] = value
            if all(value is not None for value in dominant_state):
                frame = decode_frame(core, [int(value) for value in dominant_state if value is not None], signal)
                if frame is not None:
                    dominant.observe(frame)
                else:
                    stats["DominantCarryFrameDecodeInvalid"] += 1
            else:
                stats["DominantCarryUninitializedRows"] += 1
        else:
            stats["DominantCarryBlockedRows"] += 1

        if all(value is not None for value in any_state):
            frame = decode_frame(core, [int(value) for value in any_state if value is not None], signal)
            if frame is not None:
                any_carry.observe(frame)
            else:
                stats["AnyCarryFrameDecodeInvalid"] += 1
        else:
            stats["AnyCarryUninitializedRows"] += 1

        for position, _, key in direct_candidates:
            token = str(row[position]).strip()
            if not token:
                continue
            try:
                numeric = float(token)
            except ValueError:
                continue
            if math.isfinite(numeric) and numeric.is_integer():
                direct_trackers[key].observe(int(numeric))

        if time_position is not None:
            token = str(row[time_position]).strip()
            if token:
                try:
                    current_time = float(token)
                    if math.isfinite(current_time):
                        stats["TimeUsNumericRows"] += 1
                        if previous_time is not None and current_time < previous_time:
                            stats["TimeUsDecreaseCount"] += 1
                        previous_time = current_time
                except ValueError:
                    stats["TimeUsInvalidRows"] += 1

        if (source_row + 1) % 50_000 == 0:
            print(f"[PROGRESS] raw_rows={source_row + 1}", flush=True)

    token_details = []
    for byte_index, counter in enumerate(observed_invalid_tokens):
        token_details.append({
            "ByteIndex": byte_index,
            "TopInvalidTokens": [
                {"Token": token, "Count": count}
                for token, count in counter.most_common(12)
            ],
        })

    policies = [strict.summary(), dominant.summary(), any_carry.summary()]
    direct_summaries = [tracker.summary() for tracker in direct_trackers.values()]
    candidates = policies + direct_summaries
    best = min(
        candidates,
        key=lambda item: (
            0 if item["FrameDecreaseCount"] == 0 else 1,
            abs(item["ChangedFrameObservationCount"] - HISTORICAL_FRAME_ROWS),
            -item["DeltaOnePermille"],
        ),
    ) if candidates else {}

    strict_summary = strict.summary()
    dominant_summary = dominant.summary()
    any_summary = any_carry.summary()

    def strong_exact(item: Dict[str, Any]) -> bool:
        return bool(
            item["HistoricalExactMatch"]
            and item["FrameDecreaseCount"] == 0
            and item["DeltaOnePermille"] >= 950
        )

    if strong_exact(strict_summary):
        root_cause = "STRICT_DECODE_MATCHES_HISTORY"
    elif any(strong_exact(item) for item in direct_summaries):
        root_cause = "DIRECT_FRAME_COLUMN_CONFIRMED"
    elif (
        sentinel_evidence_reliable
        and strong_exact(dominant_summary)
    ):
        root_cause = "DOMINANT_SENTINEL_BYTE_CARRY_CONFIRMED"
    elif strong_exact(any_summary):
        root_cause = "ANY_INVALID_BYTE_CARRY_CONFIRMED"
    elif strict_summary["HistoricalNearMatch"]:
        root_cause = "STRICT_DECODE_NEAR_HISTORY"
    elif any(item["HistoricalNearMatch"] for item in direct_summaries):
        root_cause = "DIRECT_FRAME_COLUMN_NEAR_MATCH"
    elif sentinel_evidence_reliable and dominant_summary["HistoricalNearMatch"]:
        root_cause = "DOMINANT_SENTINEL_BYTE_CARRY_NEAR_MATCH"
    elif any_summary["HistoricalNearMatch"]:
        root_cause = "ANY_INVALID_BYTE_CARRY_NEAR_MATCH"
    else:
        root_cause = "NO_RECONSTRUCTION_MATCH_REQUIRES_INPUT_IDENTITY_OR_DBC_AUDIT"

    return {
        "Counters": dict(stats),
        "ValidByteMaskTop": [
            {"Mask": mask, "Rows": count} for mask, count in valid_mask_counts.most_common(16)
        ],
        "InvalidByteMaskTop": [
            {"Mask": mask, "Rows": count} for mask, count in invalid_mask_counts.most_common(16)
        ],
        "ObservedInvalidTokens": token_details,
        "DominantSentinelEvidenceReliable": sentinel_evidence_reliable,
        "Policies": policies,
        "DirectFrameColumns": direct_summaries,
        "BestPolicy": best,
        "RootCauseClassification": root_cause,
    }


def token_signature(tokens: Sequence[str]) -> str:
    cleaned = [re.sub(r"[^A-Za-z0-9.+-]", "?", token)[:16] or "EMPTY" for token in tokens]
    if len(set(cleaned)) == 1:
        return cleaned[0][:12]
    digest = hashlib.sha256("\x1f".join(cleaned).encode("utf-8")).hexdigest()[:8].upper()
    return "M" + digest


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        temp = root / "04_TEMP"
        prepared = temp / PREPARED_FOLDER
        output = temp / OUTPUT_FOLDER
        archive = root / "A-Archive" / ("U1H6B5D9_HYGIENE_" + datetime.now().strftime("%Y%m%dT%H%M%S_%f"))
        print("[PREPARE] H6B5D9 byte-sentinel reconstruction hypothesis")
        print("[PREPARE] Why: test retained-byte semantics without modifying H2 or promoting inferred data.")
        if prepared.exists():
            shutil.rmtree(prepared)
            print(f"[CLEAN] 04_TEMP/{PREPARED_FOLDER} | REMOVED | Why: invalidate stale receipt.")
        if output.exists():
            archive.mkdir(parents=True, exist_ok=True)
            shutil.move(str(output), str(archive / output.name))
            print(f"[CLEAN] 04_TEMP/{OUTPUT_FOLDER} | ARCHIVED | Why: prevent stale hypothesis evidence reuse.")
        paths = required_paths(root)
        for label, path in paths.items():
            print(f"[CHECK] {label} | {'FOUND' if path.is_file() else 'MISSING'} | Why: required controlled evidence.")
        missing = [label for label, path in paths.items() if not path.is_file()]
        if missing:
            raise ProbeError("PREPARE_MISSING_" + ".".join(missing))
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        source, dbcs = configured_inputs(request)
        print(f"[CHECK] RawSource | FOUND | {source.name} | Why: reconstruction must replay the exact D8 source.")
        print(f"[CHECK] DBCChannels | {len(dbcs)} | Why: FrameCount remains DBC-derived, not guessed.")
        print("[KEEP] D8 detailed JSON | Why: it owns the observed invalid-token evidence.")
        prepared.mkdir(parents=True)
        write_json(prepared / "u001_h6b5d9_prepare_receipt.json", {
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "RawSource": str(source),
            "PreparedUtc": utc_now(),
            "Status": "READY",
            "CompletionCode": 0,
        })
        print("H6B5D9P-N3-D2-X0-C0")
        return 0
    except Exception as exc:
        print(f"H6B5D9P-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)[:300]}")
        return 7


def require_prepare(root: Path) -> Dict[str, Any]:
    receipt = read_json(root / "04_TEMP" / PREPARED_FOLDER / "u001_h6b5d9_prepare_receipt.json", "PREPARE_RECEIPT")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise ProbeError("PREPARE_NOT_READY")
    return receipt


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        require_prepare(root)
        paths = required_paths(root)
        core = load_module(paths["AdapterCore"])
        d8 = read_json(paths["D8Diagnostic"], "D8_DIAGNOSTIC")
        sentinels, _, dominance = derive_sentinels(d8)
        if len(sentinels) != 8 or len(dominance) != 8:
            raise ProbeError("SENTINEL_DERIVATION")
        if not callable(getattr(core, "parse_byte_cell", None)):
            raise ProbeError("BYTE_PARSER_MISSING")
        tracker = FrameTracker("SYNTHETIC")
        for value in (100, 100, 101, 102):
            tracker.observe(value)
        summary = tracker.summary()
        if summary["ChangedFrameObservationCount"] != 3 or summary["DeltaOnePermille"] != 1000:
            raise ProbeError("TRACKER_SYNTHETIC")
        print("[SELFTEST] D8 token evidence | Why: reconstruction may use only observed dominant rejected tokens.")
        print("[SELFTEST] Three parallel policies | Why: strict, dominant-sentinel and any-invalid carry must be compared in one scan.")
        print("[SELFTEST] Frame continuity metrics | Why: count alone cannot validate a video synchronization domain.")
        print("[SELFTEST] H2 comparison input | Why: current adapter behavior remains an explicit control branch.")
        print("[SELFTEST] No H2 writes | Why: this node is hypothesis evidence, not a production repair.")
        print(f"H6B5D9ST-N16-P{min(dominance)}-X0-C0")
        return 0
    except Exception as exc:
        print(f"H6B5D9ST-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)[:300]}")
        return 7


def execute(root_argument: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        receipt = require_prepare(root)
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        source, dbcs = configured_inputs(request)
        if str(source) != str(receipt.get("RawSource")):
            raise ProbeError("PREPARED_SOURCE_CHANGED")
        d8 = read_json(paths["D8Diagnostic"], "D8_DIAGNOSTIC")
        sentinels, sentinel_counts, dominance = derive_sentinels(d8)
        evidence_dominance = [
            dominance[index]
            for index, count in enumerate(sentinel_counts)
            if count > 0
        ]
        sentinel_evidence_reliable = bool(evidence_dominance) and all(
            value >= DOMINANCE_THRESHOLD_PERMILLE for value in evidence_dominance
        )
        core = load_module(paths["AdapterCore"])
        text_format, signal, byte_names = frame_definition(core, source, dbcs)
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            raise ProbeError("OUTPUT_EXISTS_RUN_PREPARE")
        output.mkdir(parents=True)
        comprehensive = comprehensive_scan(
            core,
            source,
            text_format,
            signal,
            byte_names,
            sentinels,
            sentinel_evidence_reliable,
        )
        h2_summary = inspect_h2_frame_rows(paths["H2Updates"])
        root_cause = comprehensive["RootCauseClassification"]
        result = {
            "DiagnosticSchema": "U001_H6B5D9_COMPREHENSIVE_FRAME_SEMANTICS_DIFFERENTIAL_V1_0",
            "GeneratedUtc": utc_now(),
            "Purpose": "COMPARE_STRICT_DOMINANT_SENTINEL_ANY_INVALID_DIRECT_COLUMN_AND_CURRENT_H2_FRAME_DOMAINS",
            "PerByteSentinel": [
                {
                    "ByteIndex": index,
                    "Token": sentinels[index],
                    "ObservedCount": sentinel_counts[index],
                    "DominancePermille": dominance[index],
                }
                for index in range(8)
            ],
            "ComprehensiveSourceAudit": comprehensive,
            "CurrentH2FrameEvents": h2_summary,
            "HistoricalAcceptedFrameRows": HISTORICAL_FRAME_ROWS,
            "RootCauseClassification": root_cause,
            "Status": "COMPREHENSIVE_DIAGNOSTIC_COMPLETE",
            "CompletionCode": 0,
        }
        write_json(output / "u001_h6b5d9_comprehensive_frame_semantics_differential.json", result)
        signature = token_signature(sentinels)
        root_codes = {
            "DOMINANT_SENTINEL_BYTE_CARRY_CONFIRMED": "DSC",
            "DOMINANT_SENTINEL_BYTE_CARRY_NEAR_MATCH": "DSN",
            "ANY_INVALID_BYTE_CARRY_CONFIRMED": "AIC",
            "ANY_INVALID_BYTE_CARRY_NEAR_MATCH": "AIN",
            "DIRECT_FRAME_COLUMN_CONFIRMED": "DFC",
            "DIRECT_FRAME_COLUMN_NEAR_MATCH": "DFN",
            "STRICT_DECODE_MATCHES_HISTORY": "SDC",
            "STRICT_DECODE_NEAR_HISTORY": "SDN",
            "NO_RECONSTRUCTION_MATCH_REQUIRES_INPUT_IDENTITY_OR_DBC_AUDIT": "NOM",
        }
        policy_codes = {
            "STRICT_ALL_8_BYTES_VALID": "ST",
            "DOMINANT_SENTINEL_OR_BLANK_CARRY": "DS",
            "ANY_INVALID_OR_BLANK_CARRY": "AI",
        }
        best = comprehensive.get("BestPolicy", {})
        best_policy = str(best.get("Policy", "NA"))
        if best_policy.startswith("DIRECT_COLUMN:"):
            best_code = "DC"
        else:
            best_code = policy_codes.get(best_policy, "NA")
        counters = comprehensive.get("Counters", {})
        policy_summaries = comprehensive.get("Policies", [])
        policy_change_counts = [
            int(item.get("ChangedFrameObservationCount", 0))
            for item in policy_summaries
        ]
        while len(policy_change_counts) < 3:
            policy_change_counts.append(0)
        capsule = (
            f"H6B5D9-K{root_codes.get(root_cause, 'UNK')}-Q{best_code}-T{signature}-"
            f"P{min(evidence_dominance) if evidence_dominance else 0}-"
            f"S{counters.get('SourceRowCount', 0)}-H{h2_summary.get('UniqueFrameCount', 0)}-"
            f"V{policy_change_counts[0]}.{policy_change_counts[1]}.{policy_change_counts[2]}-"
            f"F{best.get('ChangedFrameObservationCount', 0)}-U{best.get('UniqueFrameCount', 0)}-"
            f"R{best.get('MinimumFrameCount')}.{best.get('MaximumFrameCount')}-"
            f"D{best.get('FrameDecreaseCount', 0)}-O{best.get('DeltaOnePermille', 0)}-"
            f"I{counters.get('InvalidCellCount', 0)}-"
            f"A{counters.get('UnapprovedInvalidCellCount', 0)}-"
            f"M{2 if best.get('HistoricalExactMatch') else 1 if best.get('HistoricalNearMatch') else 0}-X0-C0"
        )
        (output / "u001_h6b5d9_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        write_json(output / "u001_h6b5d9_error.json", {
            "Type": type(exc).__name__,
            "Detail": str(exc)[:3000],
            "CompletionCode": 7,
        })
        print(f"H6B5D9-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)[:300]}")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H6B5D9 comprehensive FrameCount semantics diagnostic")
    value.add_argument("--root")
    mode = value.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    arguments = parser().parse_args(argv)
    if arguments.prepare:
        return prepare(arguments.root)
    if arguments.self_test:
        return self_test(arguments.root)
    return execute(arguments.root)


if __name__ == "__main__":
    raise SystemExit(main())
