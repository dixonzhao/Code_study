"""Independent acceptance for the U22A4 Module Qualification responsibility."""

PROGRAM_ID = "UDP002-U22A4-MODULE-QUALIFICATION-ACCEPTANCE"
VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"
CANDIDATE = "u002_u22_module_qualification_v0_1.py.txt"
INSTALLED = "u002_u22_module_qualification_v0_1.py"
VALIDATOR = "u002_u22_contract_validator_v0_1.py"
REGISTRY = "u002_u22_module_registry_v0_1.py"
ARTIFACT_STORE = "u002_u22_artifact_store_v0_1.py"
STAGES = ("S0", "S1", "S2", "S3", "S4", "S5", "S6", "S7")


class AcceptanceError(RuntimeError):
    """Represent one deterministic independent acceptance failure."""


def regular_file(path: Path) -> bool:
    """Accept ordinary files only during dependency checks and installation."""

    return path.is_file() and not path.is_symlink()


def canonical_text(value: object) -> str:
    """Serialize synthetic control documents deterministically."""

    return json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    ) + "\n"


def run(command: List[str], timeout: int = 30) -> subprocess.CompletedProcess:
    """Run one bounded child process and preserve its evidence streams."""

    return subprocess.run(
        command,
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )


def check(value: bool, reason: str, checks: List[bool]) -> None:
    """Append one visible acceptance check or fail immediately."""

    if not value:
        raise AcceptanceError(reason)
    checks.append(True)


def contains_semicolon_operator(source: str) -> bool:
    """Ignore prose punctuation and reject executable semicolon operators."""

    tokens = tokenize.generate_tokens(io.StringIO(source).readline)
    return any(
        token.type == tokenize.OP and token.string == ";"
        for token in tokens
    )


def algorithm_card(module_id: str, version: str) -> Dict[str, object]:
    """Create one valid synthetic Algorithm Card."""

    return {
        "ContractVersion": CONTRACT_VERSION,
        "AlgorithmId": module_id,
        "AlgorithmVersion": version,
        "Status": "PROVISIONAL",
        "Purpose": "Exercise controlled Module qualification",
        "ProblemSolved": "Prove self-test evidence is bound to package hashes",
        "MathematicalDefinition": ["y = x"],
        "Inputs": ["x"],
        "Outputs": ["y"],
        "Assumptions": ["synthetic input"],
        "FailureConditions": ["fixture failure"],
        "Limitations": ["acceptance only"],
        "Alternatives": ["none"],
        "EvaluationPlan": ["qualify, audit and mutate"],
        "DecisionReference": "D-U22A4",
    }


def module_manifest(module_id: str, version: str) -> Dict[str, object]:
    """Create one valid synthetic Module Manifest."""

    return {
        "ContractVersion": CONTRACT_VERSION,
        "ModuleId": module_id,
        "ModuleVersion": version,
        "Stage": "S1",
        "AlgorithmStatus": "PROVISIONAL",
        "ScopeSelector": {},
        "RequiredInputs": ["Signal.A"],
        "OptionalInputs": [],
        "RequiresArtifacts": ["INPUT_VIEW"],
        "ProvidesArtifacts": ["OUTPUT_VIEW." + module_id],
        "OutputNamespace": module_id.lower(),
        "ConfigurationSchema": "configuration_schema.json",
        "FailurePolicy": "REQUIRED",
        "ConflictPolicy": "EXCLUSIVE",
        "EntryPoint": "implementation.py",
        "SelfTestEntryPoint": "selftest.py",
        "AlgorithmCard": "algorithm_card.json",
    }


def write_module(
    root: Path,
    module_id: str,
    version: str,
    mode: str,
) -> Path:
    """Write a passing, failing or self-mutating synthetic Module package."""

    directory = root / "01_PROGRAMS/modules/S1" / module_id / version
    directory.mkdir(parents=True)
    (directory / "module_manifest.json").write_text(
        canonical_text(module_manifest(module_id, version)),
        encoding="utf-8",
    )
    (directory / "algorithm_card.json").write_text(
        canonical_text(algorithm_card(module_id, version)),
        encoding="utf-8",
    )
    (directory / "configuration_schema.json").write_text(
        canonical_text({"type": "object"}),
        encoding="utf-8",
    )
    sentinel = root / "implementation_executed.txt"
    implementation_source = (
        "from pathlib import Path\n"
        f"Path({str(sentinel)!r}).write_text('BAD', encoding='utf-8')\n"
    )
    (directory / "implementation.py").write_text(
        implementation_source,
        encoding="utf-8",
    )
    if mode == "PASS":
        selftest_source = build_selftest_source(
            "SYNTHETIC-MODULE-SELFTEST C-0",
            mutate=False,
            return_code=0,
        )
    elif mode == "FAIL":
        selftest_source = build_selftest_source(
            "SYNTHETIC-MODULE-SELFTEST C-7",
            mutate=False,
            return_code=7,
        )
    else:
        selftest_source = build_selftest_source(
            "SYNTHETIC-MODULE-SELFTEST C-0",
            mutate=True,
            return_code=0,
        )
    (directory / "selftest.py").write_text(
        selftest_source,
        encoding="utf-8",
    )
    return directory


def build_selftest_source(capsule: str, mutate: bool, return_code: int) -> str:
    """Create a self-test that follows the new explicit CLI protocol."""

    lines = [
        "import argparse",
        "from pathlib import Path",
        "parser = argparse.ArgumentParser()",
        "parser.add_argument('--module-root', required=True)",
        "parser.add_argument('--workspace-root', required=True)",
        "args = parser.parse_args()",
    ]
    if mutate:
        lines.extend(
            [
                "path = Path(args.module_root) / 'implementation.py'",
                "path.write_text(path.read_text(encoding='utf-8') + '# changed\\n', encoding='utf-8')",
            ]
        )
    lines.append(f"print({capsule!r})")
    lines.append(f"raise SystemExit({return_code})")
    return "\n".join(lines) + "\n"


def copy_environment(source: Path, target: Path) -> None:
    """Build one isolated U22 workspace without copying company data."""

    target.mkdir(parents=True)
    for stage in STAGES:
        (target / "01_PROGRAMS/modules" / stage).mkdir(parents=True)
    (target / "00_CONTROL/contracts").mkdir(parents=True)
    (target / "00_CONTROL/registry").mkdir(parents=True)
    (target / "01_PROGRAMS/core").mkdir(parents=True)
    contract_names = (
        "u002_module_manifest_schema_v0_1.json",
        "u002_artifact_manifest_schema_v0_1.json",
        "u002_run_configuration_schema_v0_1.json",
        "u002_algorithm_card_schema_v0_1.json",
    )
    for name in contract_names:
        source_file = source / "00_CONTROL/contracts" / name
        if not regular_file(source_file):
            raise AcceptanceError("CONTRACT_MISSING_" + name)
        shutil.copy2(
            source_file,
            target / "00_CONTROL/contracts" / name,
        )
    for name in (VALIDATOR, REGISTRY):
        source_file = source / "01_PROGRAMS/core" / name
        if not regular_file(source_file):
            raise AcceptanceError("CORE_DEPENDENCY_MISSING_" + name)
        shutil.copy2(
            source_file,
            target / "01_PROGRAMS/core" / name,
        )


def install_candidate(root: Path, candidate: Path, checks: List[bool]) -> Path:
    """Install the byte-identical accepted Core program atomically."""

    installed = root / "01_PROGRAMS/core" / INSTALLED
    if installed.exists():
        check(regular_file(installed), "INSTALL_NOT_REGULAR", checks)
        check(
            installed.read_bytes() == candidate.read_bytes(),
            "INSTALL_CONFLICT",
            checks,
        )
        return installed
    temporary = installed.with_name(installed.name + ".installing")
    check(not temporary.exists(), "INSTALL_TEMP_EXISTS", checks)
    temporary.write_bytes(candidate.read_bytes())
    os.replace(temporary, installed)
    check(
        installed.read_bytes() == candidate.read_bytes(),
        "INSTALL_IDENTITY",
        checks,
    )
    return installed


def archive_previous(root: Path) -> int:
    """Move completed U22A3 transfer files out of active TEMP."""

    archive = root / "A-Archive/U22_COMPLETED/U22A3/04_TEMP"
    names = (
        "u002_u22_artifact_store_v0_1.py.txt",
        "udp002_u22a3_artifact_store_acceptance_v0_1.py.txt",
    )
    moved = 0
    for name in names:
        source = root / "04_TEMP" / name
        if not regular_file(source):
            continue
        archive.mkdir(parents=True, exist_ok=True)
        destination = archive / name
        if destination.exists():
            if destination.read_bytes() != source.read_bytes():
                raise AcceptanceError("ARCHIVE_CONFLICT")
            source.unlink()
        else:
            os.replace(source, destination)
        moved += 1
    return moved


def main(argv: Sequence[str] = None) -> int:
    """Run isolated qualification tests and install only on full success."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    checks: List[bool] = []
    try:
        root = Path(args.workspace_root).resolve()
        check(
            root.is_dir() and root.name == "U-UDP002",
            "WORKSPACE_ROOT",
            checks,
        )
        candidate = root / "04_TEMP" / CANDIDATE
        check(regular_file(candidate), "CANDIDATE_MISSING", checks)
        artifact_store = root / "01_PROGRAMS/core" / ARTIFACT_STORE
        check(regular_file(artifact_store), "ARTIFACT_STORE_MISSING", checks)
        store_selftest = run(
            [
                sys.executable,
                str(artifact_store),
                "--self-test",
                "--workspace-root",
                str(root),
            ]
        )
        check(
            store_selftest.returncode == 0 and "N-T24" in store_selftest.stdout,
            "ARTIFACT_STORE_SELFTEST",
            checks,
        )
        source = candidate.read_text(encoding="utf-8")
        check(
            not contains_semicolon_operator(source),
            "SEMICOLON_STYLE",
            checks,
        )
        check("def qualify_module(" in source, "QUALIFY_INTERFACE", checks)
        check("def audit_module(" in source, "AUDIT_INTERFACE", checks)
        with tempfile.TemporaryDirectory(prefix="u22a4_acceptance_") as temporary:
            test_root = Path(temporary) / "U-UDP002"
            copy_environment(root, test_root)
            test_candidate = test_root / "01_PROGRAMS/core" / INSTALLED
            shutil.copy2(candidate, test_candidate)
            passing_module = write_module(
                test_root,
                "Synthetic.Pass",
                "0.1.0",
                "PASS",
            )
            write_module(
                test_root,
                "Synthetic.Fail",
                "0.1.0",
                "FAIL",
            )
            write_module(
                test_root,
                "Synthetic.Mutate",
                "0.1.0",
                "MUTATE",
            )
            selftest = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--self-test",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                selftest.returncode == 0 and "N-T24" in selftest.stdout,
                "CANDIDATE_SELFTEST",
                checks,
            )
            qualify = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--qualify-module",
                    "Synthetic.Pass@0.1.0",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                qualify.returncode == 0 and "N-Q1-A1-F0" in qualify.stdout,
                "PASS_QUALIFICATION",
                checks,
            )
            receipt = (
                test_root
                / "00_CONTROL/registry/qualifications"
                / "Synthetic.Pass/0.1.0/qualification_receipt.json"
            )
            check(regular_file(receipt), "RECEIPT_MISSING", checks)
            receipt_value = json.loads(receipt.read_text(encoding="utf-8"))
            check(
                receipt_value["QualificationStatus"] == "AVAILABLE",
                "RECEIPT_STATUS",
                checks,
            )
            check(
                len(receipt_value["EvidenceId"]) == 64,
                "EVIDENCE_ID",
                checks,
            )
            availability_path = (
                test_root
                / "00_CONTROL/registry/u002_module_availability_snapshot_v0_1.json"
            )
            check(regular_file(availability_path), "AVAILABILITY_MISSING", checks)
            availability = json.loads(
                availability_path.read_text(encoding="utf-8")
            )
            check(
                availability["AvailableCount"] == 1,
                "AVAILABLE_COUNT",
                checks,
            )
            check(
                not (test_root / "implementation_executed.txt").exists(),
                "IMPLEMENTATION_EXECUTED",
                checks,
            )
            audit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--audit-module",
                    "Synthetic.Pass@0.1.0",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                audit.returncode == 0 and "N-Q0-A1-F0" in audit.stdout,
                "PASS_AUDIT",
                checks,
            )
            implementation = passing_module / "implementation.py"
            original_implementation = implementation.read_bytes()
            implementation.write_bytes(original_implementation + b"# tampered\n")
            stale_audit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--audit-module",
                    "Synthetic.Pass@0.1.0",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                stale_audit.returncode == 7 and "MODULE_NOT_AVAILABLE" in stale_audit.stdout,
                "STALE_HASH_DETECTION",
                checks,
            )
            implementation.write_bytes(original_implementation)
            restored_audit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--audit-module",
                    "Synthetic.Pass@0.1.0",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(restored_audit.returncode == 0, "RESTORED_AUDIT", checks)
            failed = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--qualify-module",
                    "Synthetic.Fail@0.1.0",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                failed.returncode == 7 and "SELFTEST_FAILED" in failed.stdout,
                "FAILED_SELFTEST_REJECTED",
                checks,
            )
            failed_receipt = (
                test_root
                / "00_CONTROL/registry/qualifications"
                / "Synthetic.Fail/0.1.0/qualification_receipt.json"
            )
            check(not failed_receipt.exists(), "FAILED_RECEIPT_CREATED", checks)
            mutated = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--qualify-module",
                    "Synthetic.Mutate@0.1.0",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                mutated.returncode == 7
                and "MODULE_MUTATED_DURING_SELFTEST" in mutated.stdout,
                "MUTATION_REJECTED",
                checks,
            )
        installed = install_candidate(root, candidate, checks)
        qualifications = root / "00_CONTROL/registry/qualifications"
        qualifications.mkdir(parents=True, exist_ok=True)
        check(qualifications.is_dir(), "QUALIFICATION_DIRECTORY", checks)
        installed_selftest = run(
            [
                sys.executable,
                str(installed),
                "--self-test",
                "--workspace-root",
                str(root),
            ]
        )
        check(
            installed_selftest.returncode == 0 and "N-T24" in installed_selftest.stdout,
            "INSTALLED_SELFTEST",
            checks,
        )
        check(len(checks) == 25, "CHECK_COUNT_PRE", checks)
        if len(checks) != 26:
            raise AcceptanceError("CHECK_COUNT")
        archived = archive_previous(root)
        print(
            f"U22A4 N-Q1-A1-F2-M1-T26-P1-V{archived}-X0 C-0"
        )
        return 0
    except Exception as exc:
        reason = reformat_reason(exc)
        print(
            f"U22A4 N-Q0-A0-F0-M0-T0-P0-V0-X1 C-7 R-{reason}"
        )
        return 7


def reformat_reason(exc: Exception) -> str:
    """Keep the failure return short enough for manual transfer."""

    source = str(exc).upper() or exc.__class__.__name__.upper()
    normalized = "".join(
        character if character.isalnum() else "_"
        for character in source
    )
    return normalized.strip("_")[:80] or "INTERNAL"


if __name__ == "__main__":
    raise SystemExit(main())
