#!/usr/bin/env python3
"""Direct U001 H5 D1-D6 assembly and independent acceptance."""

from __future__ import annotations

import argparse
import csv
import json
import re
import shutil
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Optional, Sequence


PREPARED_FOLDER = "U1H5D_PREPARED_V12"
OUTPUT_FOLDER = "U1H5D_OUTPUT_V12"
EXPECTED_UPDATES = 1157704
EXPECTED_FRAMES = 56596
EXPECTED_FRAME_SIGNALS = 21
EXPECTED_REVIEW_SIGNALS = 6

REVIEW_TARGETS = [
    ("221", "distance"),
    ("221", "lateralposition"),
    ("251", "distance"),
    ("251", "lateralposition"),
    ("380", "distancetoobject"),
    ("380", "precedingvehicledetectionflag"),
]


class H5Error(RuntimeError):
    """Fail-closed H5 direct-acceptance error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def resolve_root(value: Optional[str]) -> Path:
    if value:
        root = Path(value).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise H5Error("U_UDP001_ROOT_NOT_FOUND")
    return root


def write_json(path: Path, value: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> dict[str, Any]:
    if not path.is_file():
        raise H5Error(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise H5Error(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise H5Error(label + "_ROOT_NOT_OBJECT")
    return value


def dependency_paths(root: Path) -> dict[str, Path]:
    core = root / "01_PROGRAMS" / "core"
    h4a_root = root / "04_TEMP" / "U1H4D_OUTPUT"
    h4b_root = root / "04_TEMP" / "U1H4BD_OUTPUT"
    return {
        "AssemblerPayload": root / "04_TEMP" / "u001_h5_package_assembler_v1_1.py.txt",
        "H3": root / "04_TEMP" / "U1H3D_OUTPUT" / "h3_output",
        "H4A": h4a_root / "h4a_package",
        "FramePolicy": h4a_root / "frame_policy" / "u001_h4_compiled_frame_policy.csv",
        "H4B": h4b_root / "h4b_package",
        "HoldPolicy": h4b_root / "hold_policy_package",
        "Assembler": core / "u001_h5_package_assembler_v1_1.py",
        "Validator": core / "u001_h5_package_validator_v1_0.py",
    }


def run(command: list[str], cwd: Path, label: str) -> None:
    print(f"[RUN] {label}")
    result = subprocess.run(command, cwd=str(cwd), check=False)
    if result.returncode != 0:
        raise H5Error(f"{label}_EXIT_{result.returncode}")


def normalized_message_id(value: str) -> str:
    text = value.strip().lower()
    if text.startswith("id"):
        text = text[2:]
    if text.startswith("0x"):
        text = text[2:]
    text = text.lstrip("0") or "0"
    if not re.fullmatch(r"[0-9a-f]+", text):
        return ""
    return text


def normalized_signal_name(value: str) -> str:
    return re.sub(r"[^a-z0-9]", "", value.casefold())


def frame_signal_names(frame_policy: Path) -> set[str]:
    with frame_policy.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    return {
        row.get("CanonicalName", "").strip()
        for row in rows
        if row.get("IncludeInFrameView", "").strip() == "1"
    }


def make_review_profile(h3: Path, frame_policy: Path, output: Path) -> list[str]:
    available = frame_signal_names(frame_policy)
    policy_path = h3 / "u001_h3_compiled_signal_policy.csv"
    with policy_path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))

    lookup: dict[tuple[str, str], str] = {}
    for row in rows:
        canonical = row.get("CanonicalName", "").strip()
        if canonical not in available:
            continue
        key = (
            normalized_message_id(row.get("SourceMessageId", "")),
            normalized_signal_name(row.get("SourceSignalName", "")),
        )
        if key in lookup:
            raise H5Error("REVIEW_SIGNAL_AMBIGUOUS:" + ".".join(key))
        lookup[key] = canonical

    selected: list[str] = []
    missing: list[str] = []
    for message_id, signal_name in REVIEW_TARGETS:
        canonical = lookup.get((message_id, signal_name))
        if canonical is None:
            missing.append(message_id + "." + signal_name)
        else:
            selected.append(canonical)
    if missing:
        raise H5Error("REVIEW_SIGNAL_MISSING:" + ".".join(missing))
    if len(selected) != EXPECTED_REVIEW_SIGNALS or len(set(selected)) != EXPECTED_REVIEW_SIGNALS:
        raise H5Error("REVIEW_SIGNAL_COUNT_OR_IDENTITY")

    fields = ["ReviewProfileSchema", "CanonicalName", "IncludeInReviewView", "ReviewOrder"]
    with output.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields, lineterminator="\n")
        writer.writeheader()
        for order, canonical in enumerate(selected, start=1):
            writer.writerow(
                {
                    "ReviewProfileSchema": "U001_H5_REVIEW_PROFILE_V1_0",
                    "CanonicalName": canonical,
                    "IncludeInReviewView": 1,
                    "ReviewOrder": order,
                }
            )
    return selected


def prepare(root_value: Optional[str]) -> int:
    try:
        root = resolve_root(root_value)
        prepared = root / "04_TEMP" / PREPARED_FOLDER
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if prepared.exists():
            shutil.rmtree(prepared)
        if output.exists():
            shutil.rmtree(output)

        required = dependency_paths(root)
        payload = required["AssemblerPayload"]
        if not payload.is_file():
            raise H5Error("ASSEMBLER_V1_1_PAYLOAD_MISSING")
        required["Assembler"].parent.mkdir(parents=True, exist_ok=True)
        shutil.copyfile(payload, required["Assembler"])
        if required["Assembler"].read_bytes() != payload.read_bytes():
            raise H5Error("ASSEMBLER_V1_1_INSTALL_IDENTITY")
        folder_labels = {"H3", "H4A", "H4B", "HoldPolicy"}
        reasons = {
            "AssemblerPayload": "Approved v1.1 source that tolerates delayed Windows staging-directory cleanup.",
            "H3": "Provides canonical update stream and complete semantic policy.",
            "H4A": "Provides the immutable raw one-row-per-observed-frame view.",
            "FramePolicy": "Proves the exact 21 signals included in the frame domain.",
            "H4B": "Provides the accepted no-cross-frame-hold view.",
            "HoldPolicy": "Preserves the explicit NONE decision for every frame signal.",
            "Assembler": "Installed v1.1 atomically creates D1-D6 without treating delayed temp cleanup as data failure.",
            "Validator": "Independently checks inventory, hashes, counts, identity and subsets.",
        }
        missing: list[str] = []
        for label, path in required.items():
            found = path.is_dir() if label in folder_labels else path.is_file()
            print(f"[CHECK] {label} | {'FOUND' if found else 'MISSING'} | Why: {reasons[label]}")
            if not found:
                missing.append(label)
        if missing:
            raise H5Error("DEPENDENCY_MISSING:" + ".".join(missing))

        h3_manifest = read_json(required["H3"] / "u001_h3_run_manifest.json", "H3_MANIFEST")
        h4a_quality = read_json(required["H4A"] / "u001_h4_raw_frame_quality.json", "H4A_QUALITY")
        h4b_quality = read_json(required["H4B"] / "u001_h4_bounded_frame_quality.json", "H4B_QUALITY")
        if int(h3_manifest.get("Counts", {}).get("InputUpdateRows", -1)) != EXPECTED_UPDATES:
            raise H5Error("H3_UPDATE_COUNT")
        if int(h4a_quality.get("OutputFrameRowCount", -1)) != EXPECTED_FRAMES:
            raise H5Error("H4A_FRAME_COUNT")
        if int(h4b_quality.get("OutputFrameCount", -1)) != EXPECTED_FRAMES:
            raise H5Error("H4B_FRAME_COUNT")
        if int(h4b_quality.get("SignalCount", -1)) != EXPECTED_FRAME_SIGNALS:
            raise H5Error("H4B_SIGNAL_COUNT")
        if any(int(h4b_quality.get(key, -1)) != 0 for key in ("CarriedCellCount", "StaleCellCount", "HoldBarrierCellCount")):
            raise H5Error("H4B_NO_HOLD_BASELINE_DRIFT")

        prepared.mkdir(parents=True)
        write_json(
            prepared / "receipt.json",
            {
                "Schema": "U001_H5_DIRECT_PREPARE_V1_0",
                "PreparedUtc": utc_now(),
                "Status": "READY",
                "CompletionCode": 0,
            },
        )
        print("H5P-P7-H3-X0-C0")
        return 0
    except Exception as exc:
        print(f"H5PE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def require_prepared(root: Path) -> None:
    receipt = read_json(root / "04_TEMP" / PREPARED_FOLDER / "receipt.json", "PREPARE_RECEIPT")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise H5Error("PREPARE_NOT_READY")


def self_test(root_value: Optional[str]) -> int:
    try:
        root = resolve_root(root_value)
        require_prepared(root)
        required = dependency_paths(root)
        print("[SELFTEST] Assembler | Why: D1-D6 must be created atomically from exact upstream identities.")
        run([sys.executable, str(required["Assembler"]), "--self-test"], required["Assembler"].parent, "ASSEMBLER_SELFTEST")
        print("[SELFTEST] Validator | Why: a separate reader must reject count, hash, inventory, schema or subset drift.")
        run([sys.executable, str(required["Validator"]), "--self-test"], required["Validator"].parent, "VALIDATOR_SELFTEST")
        print("H5S-A3-V2-X0-C0")
        return 0
    except Exception as exc:
        print(f"H5SE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def execute(root_value: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_value)
        require_prepared(root)
        required = dependency_paths(root)
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            shutil.rmtree(output)
        output.mkdir(parents=True)

        review_profile = output / "u001_h5_review_profile.csv"
        final_package = output / "u001_final_package"
        selected = make_review_profile(required["H3"], required["FramePolicy"], review_profile)
        print("[REVIEW PROFILE] " + " | ".join(selected))

        run(
            [
                sys.executable,
                str(required["Assembler"]),
                "--h3-package", str(required["H3"]),
                "--h4a-package", str(required["H4A"]),
                "--h4b-package", str(required["H4B"]),
                "--frame-policy", str(required["FramePolicy"]),
                "--hold-policy-package", str(required["HoldPolicy"]),
                "--review-profile", str(review_profile),
                "--out", str(final_package),
            ],
            root,
            "H5_ASSEMBLE",
        )
        run(
            [sys.executable, str(required["Validator"]), "--package", str(final_package)],
            root,
            "H5_VALIDATE",
        )

        manifest = read_json(final_package / "u001_run_manifest.json", "FINAL_MANIFEST")
        counts = manifest.get("Counts", {})
        expected = {
            "PackageFiles": 6,
            "Updates": EXPECTED_UPDATES,
            "Frames": EXPECTED_FRAMES,
            "FrameSignals": EXPECTED_FRAME_SIGNALS,
            "ReviewSignals": EXPECTED_REVIEW_SIGNALS,
        }
        for key, value in expected.items():
            actual = int(counts.get(key, -1))
            if actual != value:
                raise H5Error(f"FINAL_{key.upper()}_{actual}_EXPECTED_{value}")
        sources = int(counts.get("Sources", 0))
        if sources < 1:
            raise H5Error("FINAL_SOURCE_COUNT")
        if manifest.get("Status") != "COMPLETE" or manifest.get("CompletionCode") != 0:
            raise H5Error("FINAL_MANIFEST_STATUS")
        if len(selected) != EXPECTED_REVIEW_SIGNALS:
            raise H5Error("FINAL_REVIEW_SELECTION")

        result = {
            "Schema": "U001_H5_DIRECT_RESULT_V1_0",
            "GeneratedUtc": utc_now(),
            "PackageFiles": 6,
            "Sources": sources,
            "Updates": EXPECTED_UPDATES,
            "Frames": EXPECTED_FRAMES,
            "FrameSignals": EXPECTED_FRAME_SIGNALS,
            "ReviewSignals": EXPECTED_REVIEW_SIGNALS,
            "ReviewCanonicalNames": selected,
            "FinalPackageRelativePath": "04_TEMP/U1H5D_OUTPUT_V12/u001_final_package",
            "Status": "H5_ACCEPTED",
            "CompletionCode": 0,
        }
        write_json(output / "u001_h5_direct_result.json", result)
        capsule = (
            f"H5-P6-N{sources}-R{EXPECTED_UPDATES}-F{EXPECTED_FRAMES}"
            f"-S{EXPECTED_FRAME_SIGNALS}-V{EXPECTED_REVIEW_SIGNALS}-X0-C0"
        )
        (output / "u001_h5_direct_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        write_json(
            output / "u001_h5_direct_diagnostic.json",
            {
                "GeneratedUtc": utc_now(),
                "ExceptionType": type(exc).__name__,
                "Detail": str(exc)[:8000],
                "Recovery": "Return H5E only; do not alter any accepted upstream package.",
            },
        )
        print(f"H5E-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="Direct U001 H5 acceptance")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    raise SystemExit(main())
