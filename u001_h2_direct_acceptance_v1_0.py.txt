#!/usr/bin/env python3
"""Direct H2 production run and acceptance for Adapter v1.2.0."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import shutil
import stat
import subprocess
import sys
import time
import zipfile
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Optional, Sequence


PROGRAM_ID = "U001_H2_DIRECT_ACCEPTANCE"
PROGRAM_VERSION = "1.0"
PACKAGE_NAME = "UDP001_H2_HEX_SNAPSHOT_SPRINT_v1_0.zip"
PREPARED_FOLDER = "U1H2D_PREPARED"
OUTPUT_FOLDER = "U1H2D_OUTPUT"
ADAPTER_RELATIVE = Path("01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.2.0")
EXPECTED_SOURCE_ROWS = 274522
EXPECTED_SIGNALS = 22
EXPECTED_FRAMES = 56596


class DirectH2Error(RuntimeError):
    pass


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise DirectH2Error("U_UDP001_ROOT_NOT_FOUND")
    return root


def write_json(path: Path, value: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> dict[str, Any]:
    if not path.is_file():
        raise DirectH2Error(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise DirectH2Error(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise DirectH2Error(label + "_ROOT_NOT_OBJECT")
    return value


def make_writable(path: Path) -> None:
    try:
        os.chmod(path, stat.S_IWRITE | stat.S_IREAD)
    except OSError:
        pass


def remove_tree(path: Path) -> None:
    if not path.exists():
        return

    def onerror(function: Any, target: str, exc_info: Any) -> None:
        make_writable(Path(target))
        function(target)

    shutil.rmtree(path, onerror=onerror)


def locate_package(root: Path) -> Path:
    candidates = (root / "04_TEMP" / PACKAGE_NAME, Path(__file__).resolve().parent / PACKAGE_NAME)
    for path in candidates:
        if path.is_file():
            return path
    raise DirectH2Error("H2_FIX_PACKAGE_NOT_FOUND")


def ensure_adapter(root: Path) -> Path:
    target = root / ADAPTER_RELATIVE
    adapter = target / "adapter.py"
    card = target / "adapter_card.json"
    core = target / "u001_matlab_can_csv_dbc_core_v1_0.py"
    if adapter.is_file() and card.is_file() and core.is_file():
        description = read_json(card, "ADAPTER_CARD")
        if description.get("AdapterVersion") == "1.2.0":
            return adapter
    package = locate_package(root)
    temporary = root / "04_TEMP" / ".U1H2D_ADAPTER_INSTALL"
    remove_tree(temporary)
    with zipfile.ZipFile(package, "r") as archive:
        archive.extractall(temporary)
    source = temporary / "payload" / ADAPTER_RELATIVE
    if not (source / "adapter.py").is_file():
        raise DirectH2Error("PACKAGE_ADAPTER_NOT_FOUND")
    remove_tree(target)
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copytree(source, target)
    remove_tree(temporary)
    return target / "adapter.py"


def request_path(root: Path) -> Path:
    return root / "04_TEMP" / "U1H6B3_OUTPUT" / "compiled_profile" / "u001_h2_controlled_request.json"


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        prepared = root / "04_TEMP" / PREPARED_FOLDER
        output = root / "04_TEMP" / OUTPUT_FOLDER
        remove_tree(prepared)
        remove_tree(output)
        adapter = ensure_adapter(root)
        request = request_path(root)
        print("[CHECK] Adapter v1.2.0 | FOUND | Why: owns hexadecimal-byte and snapshot-change semantics.")
        print(f"[CHECK] Controlled 22-signal request | {'FOUND' if request.is_file() else 'MISSING'} | Why: freezes selected signals and CH2/0x620 FrameCount.")
        if not request.is_file():
            raise DirectH2Error("CONTROLLED_REQUEST_NOT_FOUND")
        value = read_json(request, "CONTROLLED_REQUEST")
        selected = value.get("Configuration", {}).get("SelectedSignals", [])
        if not isinstance(selected, list) or len(selected) != EXPECTED_SIGNALS:
            raise DirectH2Error("CONTROLLED_REQUEST_SIGNAL_COUNT")
        inputs = value.get("Inputs", [])
        if not isinstance(inputs, list) or any(not Path(str(item.get("Path", ""))).is_file() for item in inputs if isinstance(item, dict)):
            raise DirectH2Error("CONTROLLED_REQUEST_INPUT_NOT_FOUND")
        prepared.mkdir(parents=True)
        write_json(prepared / "prepare_receipt.json", {
            "Schema": "U001_H2_DIRECT_PREPARE_V1_0",
            "PreparedUtc": utc_now(),
            "AdapterPath": str(adapter),
            "RequestPath": str(request),
            "SelectedSignalCount": len(selected),
            "Status": "READY",
            "CompletionCode": 0,
        })
        print("H2DP-N22-A12-X0-C0")
        return 0
    except Exception as exc:
        print(f"H2DPE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def require_prepared(root: Path) -> dict[str, Any]:
    receipt = read_json(root / "04_TEMP" / PREPARED_FOLDER / "prepare_receipt.json", "PREPARE_RECEIPT")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise DirectH2Error("PREPARE_RECEIPT_NOT_READY")
    return receipt


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        receipt = require_prepared(root)
        adapter = Path(str(receipt["AdapterPath"]))
        core = adapter.parent / "u001_matlab_can_csv_dbc_core_v1_0.py"
        print("[SELFTEST] Hex parser | Why: unprefixed digit-only raw bytes remain base-16.")
        first = subprocess.run([sys.executable, str(core), "--self-test"], cwd=str(adapter.parent), check=False)
        if first.returncode != 0:
            raise DirectH2Error("HEX_CORE_SELFTEST")
        print("[SELFTEST] Snapshot adapter | Why: unchanged retained-buffer payloads do not create false updates.")
        second = subprocess.run([sys.executable, str(adapter), "--self-test"], cwd=str(adapter.parent), check=False)
        if second.returncode != 0:
            raise DirectH2Error("ADAPTER_SELFTEST")
        print("H2DS-H26-A38-X0-C0")
        return 0
    except Exception as exc:
        print(f"H2DSE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def run_adapter(adapter: Path, request: Path, output: Path) -> int:
    process = subprocess.Popen(
        [sys.executable, str(adapter), "--run", "--request", str(request), "--out", str(output)],
        cwd=str(adapter.parent),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
    )
    started = time.monotonic()
    update_file = output / "u001_adapter_source_updates.csv"
    while process.poll() is None:
        elapsed = int(time.monotonic() - started)
        size = update_file.stat().st_size if update_file.is_file() else 0
        print(f"[H2] elapsed={elapsed}s update_bytes={size}", flush=True)
        time.sleep(30)
    stdout, stderr = process.communicate()
    (output.parent / "h2_stdout.txt").write_text(stdout, encoding="utf-8")
    (output.parent / "h2_stderr.txt").write_text(stderr, encoding="utf-8")
    if stdout.strip():
        print(stdout.strip())
    if process.returncode != 0:
        detail = " ".join(stderr.split())[-1000:] or "NO_STDERR"
        raise DirectH2Error(f"ADAPTER_EXIT_{process.returncode}:{detail}")
    return elapsed


def execute(root_argument: Optional[str]) -> int:
    output_root: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        receipt = require_prepared(root)
        output_root = root / "04_TEMP" / OUTPUT_FOLDER
        remove_tree(output_root)
        output_root.mkdir(parents=True)
        h2 = output_root / "h2_package"
        adapter = Path(str(receipt["AdapterPath"]))
        request = Path(str(receipt["RequestPath"]))
        elapsed = run_adapter(adapter, request, h2)
        quality = read_json(h2 / "u001_adapter_quality.json", "H2_QUALITY")
        manifest = read_json(h2 / "u001_adapter_run_manifest.json", "H2_MANIFEST")
        expected_files = {
            "u001_adapter_source_updates.csv", "u001_adapter_signal_catalog.csv",
            "u001_adapter_quality.json", "u001_adapter_run_manifest.json",
            "u001_adapter_completion_capsule.txt",
        }
        if {path.name for path in h2.iterdir() if path.is_file()} != expected_files:
            raise DirectH2Error("H2_FILESET")
        facts = {
            "AdapterVersion": quality.get("AdapterVersion"),
            "RawByteEncoding": quality.get("RawByteEncoding"),
            "ObservationSemantics": quality.get("ObservationSemantics"),
            "SourceRows": int(quality.get("SourceRowCount", -1)),
            "Signals": int(quality.get("SelectedSignalCount", -1)),
            "Updates": int(quality.get("DecodedUpdateCount", -1)),
            "Frames": int(quality.get("FrameKeyUpdateCount", -1)),
            "Failures": int(quality.get("DecodeFailureCount", -1)),
            "RepeatedSnapshots": int(quality.get("RepeatedSnapshotObservationCount", -1)),
        }
        expected = {
            "AdapterVersion": "1.2.0", "RawByteEncoding": "HEX_TEXT",
            "ObservationSemantics": "SNAPSHOT_STATE_CHANGE",
            "SourceRows": EXPECTED_SOURCE_ROWS, "Signals": EXPECTED_SIGNALS,
            "Frames": EXPECTED_FRAMES, "Failures": 0,
        }
        for key, value in expected.items():
            if facts[key] != value:
                raise DirectH2Error(f"{key.upper()}_{facts[key]}_EXPECTED_{value}")
        if manifest.get("Status") != "COMPLETE" or manifest.get("CompletionCode") != 0:
            raise DirectH2Error("H2_MANIFEST_STATUS")
        result = {
            "ResultSchema": "U001_H2_DIRECT_ACCEPTANCE_RESULT_V1_0",
            "GeneratedUtc": utc_now(),
            **facts,
            "ElapsedSeconds": elapsed,
            "NextRequiredAction": "DIRECT_H3_ACCEPTANCE_FROM_U1H2D_OUTPUT",
            "Status": "H2_ACCEPTED",
            "CompletionCode": 0,
        }
        write_json(output_root / "u001_h2_direct_result.json", result)
        capsule = (
            f"H2D-A12-R{facts['SourceRows']}-U{facts['Updates']}-F{facts['Frames']}"
            f"-S{facts['Signals']}-E{facts['Failures']}-X0-C0"
        )
        (output_root / "u001_h2_direct_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        return 0
    except Exception as exc:
        if output_root is None:
            output_root = Path.cwd() / OUTPUT_FOLDER
        output_root.mkdir(parents=True, exist_ok=True)
        write_json(output_root / "u001_h2_direct_diagnostic.json", {
            "ProgramId": PROGRAM_ID, "GeneratedUtc": utc_now(),
            "ExceptionType": type(exc).__name__, "Detail": str(exc)[:8000],
            "Recovery": "Return H2DE only; do not transcribe log files.",
        })
        print(f"H2DE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="Direct H2 production acceptance")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    raise SystemExit(main())
