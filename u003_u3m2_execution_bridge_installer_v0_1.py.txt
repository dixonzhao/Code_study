#!/usr/bin/env python3
"""Install the governed U3M2 execution bridge from its adjacent ZIP file."""

from __future__ import annotations

import argparse
import csv
import hashlib
import shutil
import sys
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PACKAGE_NAME = "UDP003_U3M2_EXECUTION_BRIDGE_v0_1.zip"
FOUNDATION_RELATIVE_PATH = "01_PROGRAMS/u003_module_runtime_v0_1.py"
REQUIRED_MEMBERS = {
    "01_PROGRAMS/u003_module_executor_v0_1.py",
    "02_CONTROLS/contracts/u003_module_execution_contract_bundle_v0_1.json",
    "02_MODULES/REVIEW_ASSIST/CameraRadar.U003.SampleMemoDraft/0.1.0/module_manifest.json",
    "02_MODULES/REVIEW_ASSIST/CameraRadar.U003.SampleMemoDraft/0.1.0/algorithm_card.json",
    "02_MODULES/REVIEW_ASSIST/CameraRadar.U003.SampleMemoDraft/0.1.0/implementation.py",
    "PACKAGE_SHA256.csv",
}


class InstallError(RuntimeError):
    """Raised when U3M1 is absent or the U3M2 package is unsafe."""


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def safe_members(archive: zipfile.ZipFile) -> list[str]:
    members: list[str] = []
    for name in archive.namelist():
        pure = PurePosixPath(name)
        if pure.is_absolute() or ".." in pure.parts:
            raise InstallError(f"Unsafe ZIP member: {name}")
        if not name.endswith("/"):
            members.append(name)
    if set(members) != REQUIRED_MEMBERS:
        missing = sorted(REQUIRED_MEMBERS - set(members))
        extra = sorted(set(members) - REQUIRED_MEMBERS)
        raise InstallError(f"Package inventory mismatch. Missing={missing}; Extra={extra}")
    return members


def validate_payloads(package_root: Path) -> list[tuple[str, str, str]]:
    manifest_path = package_root / "PACKAGE_SHA256.csv"
    with manifest_path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    if len(rows) != 5:
        raise InstallError("PACKAGE_SHA256.csv must contain exactly five payload rows.")

    checked: list[tuple[str, str, str]] = []
    for row in rows:
        relative = row.get("RelativePath", "").strip()
        expected = row.get("Sha256", "").strip().lower()
        purpose = row.get("Purpose", "").strip()
        payload = (package_root / relative).resolve()
        try:
            payload.relative_to(package_root.resolve())
        except ValueError as error:
            raise InstallError(f"Manifest path escapes package root: {relative}") from error
        if not payload.is_file() or payload.is_symlink():
            raise InstallError(f"Payload is missing: {relative}")
        actual = sha256_file(payload)
        if actual != expected:
            raise InstallError(f"Payload hash mismatch: {relative}")
        checked.append((relative, actual, purpose))
    return checked


def archive_existing(root: Path, relative_paths: list[str]) -> Path | None:
    existing = [relative for relative in relative_paths if (root / relative).is_file()]
    if not existing:
        return None
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    archive_root = root / "A-Archive" / f"U3M2_REPLACED_{stamp}"
    for relative in existing:
        source = root / relative
        destination = archive_root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, destination)
    return archive_root


def install(package_path: Path, root: Path) -> int:
    package_path = package_path.resolve()
    root = root.resolve()
    if root.name.upper() != "U-UDP003":
        raise InstallError(f"Expected U-UDP003 root, got: {root}")
    foundation = root / FOUNDATION_RELATIVE_PATH
    if not foundation.is_file():
        raise InstallError(
            "U3M1 foundation is missing. Complete U3M1 before installing U3M2."
        )
    if not package_path.is_file():
        raise InstallError(f"Transfer ZIP is missing: {package_path}")

    temp_root = root / "04_TEMP" / "U3M2_PREPARED"
    if temp_root.exists():
        shutil.rmtree(temp_root)
    temp_root.mkdir(parents=True)
    with zipfile.ZipFile(package_path, "r") as archive:
        members = safe_members(archive)
        archive.extractall(temp_root)
    checked = validate_payloads(temp_root)
    relative_paths = [relative for relative, _digest, _purpose in checked]
    archive_root = archive_existing(root, relative_paths)

    for relative in relative_paths:
        source = temp_root / relative
        destination = root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        writing = destination.with_name(destination.name + ".writing")
        shutil.copy2(source, writing)
        writing.replace(destination)

    print("U3M2 prepare checks:")
    print(f"  [OK] U003 root: {root} - company topology is explicit")
    print(f"  [OK] U3M1 foundation: {foundation} - execution reuses the accepted scanner")
    print(f"  [OK] ZIP inventory: {len(members)} files - foreign payloads are rejected")
    for relative, digest, purpose in checked:
        print(f"  [OK] {relative} | {digest[:12]} - {purpose}")
    print("  [OK] U3M2_PREPARED refreshed - stale extraction cannot be reused")
    if archive_root is None:
        print("  [OK] No previous U3M2-owned file required archival")
    else:
        print(f"  [OK] Previous U3M2-owned files archived: {archive_root}")
    print("U3M2P N-R1-F5-M1-B{}-X0 C-0".format(1 if archive_root else 0))
    return 0


def parse_arguments(argv: list[str]) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--root", type=Path)
    parser.add_argument("--package", type=Path)
    return parser.parse_args(argv)


def main(argv: list[str] | None = None) -> int:
    arguments = parse_arguments(sys.argv[1:] if argv is None else argv)
    try:
        if not arguments.prepare:
            raise InstallError("Use --prepare.")
        script_directory = Path(__file__).resolve().parent
        root = arguments.root if arguments.root is not None else script_directory.parent
        package = arguments.package if arguments.package is not None else script_directory / PACKAGE_NAME
        return install(package, root)
    except Exception as error:
        print(f"U3M2PE K-{type(error).__name__.upper()} D-{error} X1 C-7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
