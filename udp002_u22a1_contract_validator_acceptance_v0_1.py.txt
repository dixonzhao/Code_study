class AcceptanceError(RuntimeError):
    pass


def stable(value: object) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")[:80] or "INTERNAL"


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def require_file(path: Path, reason: str) -> Path:
    if not path.is_file() or path.is_symlink():
        raise AcceptanceError(reason)
    return path


def write_json(path: Path, value: object) -> None:
    path.write_text(json.dumps(value, ensure_ascii=False, sort_keys=True,
                               indent=2, allow_nan=False) + "\n", encoding="utf-8")


def source_audit(path: Path) -> dict:
    text = path.read_text(encoding="utf-8-sig")
    try:
        tree = ast.parse(text)
    except SyntaxError as exc:
        raise AcceptanceError("CANDIDATE_SYNTAX") from exc
    imports = set()
    calls = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            imports.update(alias.name.split(".")[0] for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            imports.add(node.module.split(".")[0])
        elif isinstance(node, ast.Call):
            if isinstance(node.func, ast.Name):
                calls.add(node.func.id)
    if "jsonschema" in imports or "eval" in calls or "exec" in calls:
        raise AcceptanceError("CANDIDATE_FORBIDDEN_SURFACE")
    required_functions = {
        "load_json", "validate_schema", "validate_document", "validate_module",
        "validate_artifact", "validate_run", "validate_algorithm", "self_test",
    }
    functions = {node.name for node in tree.body if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))}
    if not required_functions.issubset(functions):
        raise AcceptanceError("CANDIDATE_FUNCTIONS")
    return {"Sha256": sha256_file(path), "Functions": sorted(required_functions),
            "UsesThirdPartyJsonSchema": False, "UsesEvalOrExec": False}


def verify_contract_manifest(root: Path) -> int:
    manifest = require_file(root / "00_CONTROL/u22a0_contract_manifest.csv", "A0_MANIFEST_MISSING")
    with manifest.open("r", encoding="utf-8-sig", newline="") as stream:
        records = {row["RelativePath"]: row for row in csv.DictReader(stream)}
    for relative in SCHEMA_FILES:
        path = require_file(root / relative, "SCHEMA_MISSING")
        if relative not in records or records[relative]["Sha256"] != sha256_file(path):
            raise AcceptanceError("SCHEMA_IDENTITY")
    return len(SCHEMA_FILES)

def run_child(candidate: Path, arguments: List[str], log: Path) -> subprocess.CompletedProcess:
    result = subprocess.run([sys.executable, str(candidate), *arguments],
                            stdin=subprocess.DEVNULL, capture_output=True,
                            timeout=60, check=False)
    log.with_suffix(".stdout.txt").write_bytes(result.stdout)
    log.with_suffix(".stderr.txt").write_bytes(result.stderr)
    return result


def lines(result: subprocess.CompletedProcess) -> List[str]:
    return [line.strip() for line in result.stdout.decode("utf-8", errors="replace").splitlines() if line.strip()]


def install_identical(candidate: Path, destination: Path) -> None:
    destination.parent.mkdir(parents=True, exist_ok=True)
    if destination.exists():
        if not destination.is_file() or destination.is_symlink() or sha256_file(destination) != sha256_file(candidate):
            raise AcceptanceError("DESTINATION_CONFLICT")
        return
    temporary = destination.with_name(destination.name + ".installing")
    if temporary.exists():
        raise AcceptanceError("INSTALL_TEMP_EXISTS")
    shutil.copyfile(candidate, temporary)
    os.replace(temporary, destination)
    if sha256_file(destination) != sha256_file(candidate):
        raise AcceptanceError("INSTALL_IDENTITY")


def archive_a0_installer(root: Path) -> int:
    source = root / "04_TEMP/udp002_u22a0_contract_scaffold_installer_v0_1.py.txt"
    if not source.exists():
        return 0
    require_file(source, "A0_INSTALLER_TYPE")
    destination = root / "A-Archive/U22_BUILD_HISTORY/U22A0" / source.name
    destination.parent.mkdir(parents=True, exist_ok=True)
    if destination.exists():
        if sha256_file(destination) != sha256_file(source):
            raise AcceptanceError("A0_ARCHIVE_CONFLICT")
        source.unlink()
    else:
        os.replace(source, destination)
    return 1


def accept(root: Path) -> str:
    root = root.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise AcceptanceError("WORKSPACE_ROOT")
    candidate = require_file(root / CANDIDATE, "CANDIDATE_MISSING")
    audit = source_audit(candidate)
    schema_count = verify_contract_manifest(root)
    out = root / OUT
    if out.exists() or out.with_name(out.name + ".staging").exists():
        raise AcceptanceError("OUTPUT_EXISTS")
    staging = out.with_name(out.name + ".staging")
    staging.mkdir(parents=True)
    try:
        self_result = run_child(candidate, ["--self-test", "--workspace-root", str(root)], staging / "selftest")
        if self_result.returncode != 0 or EXPECTED_SELFTEST not in lines(self_result):
            raise AcceptanceError("SELFTEST")
        valid_count = 0
        invalid_count = 0
        for name, (valid, invalid, option) in fixtures().items():
            valid_path = staging / f"valid_{name}.json"
            invalid_path = staging / f"invalid_{name}.json"
            write_json(valid_path, valid); write_json(invalid_path, invalid)
            positive = run_child(candidate, [option, str(valid_path), "--workspace-root", str(root)], staging / f"valid_{name}")
            if positive.returncode != 0 or not any(line.endswith("C-0") for line in lines(positive)):
                raise AcceptanceError("POSITIVE_" + name)
            valid_count += 1
            negative = run_child(candidate, [option, str(invalid_path), "--workspace-root", str(root)], staging / f"invalid_{name}")
            if negative.returncode != 7 or not any("C-7" in line for line in lines(negative)):
                raise AcceptanceError("NEGATIVE_" + name)
            invalid_count += 1
        destination = root / DESTINATION
        install_identical(candidate, destination)
        installed_selftest = run_child(destination, ["--self-test", "--workspace-root", str(root)], staging / "installed_selftest")
        if installed_selftest.returncode != 0 or EXPECTED_SELFTEST not in lines(installed_selftest):
            raise AcceptanceError("INSTALLED_SELFTEST")
        capsule = f"U22A1 N-S{schema_count}-V{valid_count}-I{invalid_count}-T24-P1-X0 C-0"
        write_json(staging / "u22a1_acceptance.json", {
            "AcceptanceId": PROGRAM_ID, "AcceptanceVersion": VERSION, "Status": "ACCEPTED",
            "CandidateAudit": audit, "SchemaCount": schema_count, "PositiveFixtures": valid_count,
            "NegativeFixtures": invalid_count, "InstalledPath": DESTINATION,
            "InstalledByteIdentical": sha256_file(destination) == sha256_file(candidate),
            "SuccessCapsule": capsule,
        })
        (staging / "u22a1_return_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        os.replace(staging, out)
        archive_a0_installer(root)
        return capsule
    except Exception:
        raise


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    try:
        print(accept(Path(args.workspace_root)))
        return 0
    except Exception as exc:
        print(f"U22A1 N-S0-V0-I0-T0-P0-X1 C-7 K-{stable(exc)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
