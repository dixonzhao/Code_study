VERSION = "0.1"

CANDIDATE = "u002_u22_engine_preflight_v0_1.py.txt"

class AcceptanceError(RuntimeError):
    """Represent one deterministic independent acceptance failure."""


def regular_file(path: Path) -> bool:
    """Accept ordinary files only during dependency checks and installation."""

    return path.is_file() and not path.is_symlink()


def canonical_text(value: object) -> str:
    """Serialize synthetic control documents deterministically."""

    return json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    ) + "\n"


def run(command: List[str], timeout: int = 30) -> subprocess.CompletedProcess:
    """Run one bounded child process and preserve its evidence streams."""

    return subprocess.run(
        command,
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )


def check(value: bool, reason: str, checks: List[bool]) -> None:
    """Append one visible acceptance check or fail immediately."""

    if not value:
        raise AcceptanceError(reason)
    checks.append(True)


def contains_semicolon_operator(source: str) -> bool:
    """Ignore prose punctuation and reject executable semicolon operators."""

    tokens = tokenize.generate_tokens(io.StringIO(source).readline)
    return any(
        token.type == tokenize.OP and token.string == ";"
        for token in tokens
    )

    """Create one valid synthetic Algorithm Card."""

    return {
        "ContractVersion": CONTRACT_VERSION,
        "AlgorithmId": module_id,
        "AlgorithmVersion": version,
        "Status": "PROVISIONAL",
        "Purpose": "Exercise non-executing Engine Preflight",
        "ProblemSolved": "Resolve an explicit qualified synthetic Run Plan",
        "MathematicalDefinition": ["y = x"],
        "Inputs": ["x"],
        "Outputs": ["y"],
        "Assumptions": ["synthetic input"],
        "FailureConditions": ["fixture failure"],
        "Limitations": ["acceptance only"],
        "Alternatives": ["none"],
        "EvaluationPlan": ["qualify and prepare without implementation execution"],
        "DecisionReference": "D-U22A5",
    }

    module_id: str,
    version: str,
    stage: str,
    requires_artifacts: List[str],
    provides_artifact: str,

    """Create one valid synthetic Module Manifest."""

    return {
        "ContractVersion": CONTRACT_VERSION,
        "ModuleId": module_id,

        "Stage": stage,
        "AlgorithmStatus": "PROVISIONAL",
        "ScopeSelector": {},
        "RequiredInputs": ["Signal.A"],

        "RequiresArtifacts": requires_artifacts,
        "ProvidesArtifacts": [provides_artifact],
        "OutputNamespace": module_id.lower(),

        "ConflictPolicy": "EXCLUSIVE",
        "EntryPoint": "implementation.py",
        "SelfTestEntryPoint": "selftest.py",
        "AlgorithmCard": "algorithm_card.json",
    }


def configuration_schema() -> Dict[str, object]:
    """Create a schema that exercises real Module configuration validation."""

    return {
        "$schema": "https://json-schema.org/draft/2020-12/schema",
        "type": "object",
        "additionalProperties": False,
        "required": ["Threshold"],
        "properties": {
            "Threshold": {
                "type": "number",
                "minimum": 0,
            }
        },
    }


def write_module(
    root: Path,
    module_id: str,
    stage: str,
    requires_artifacts: List[str],
    provides_artifact: str,
) -> Path:
    """Write one valid synthetic Module with visible execution side effect."""

    version = "0.1.0"
    directory = root / "01_PROGRAMS/modules" / stage / module_id / version
    directory.mkdir(parents=True)
    (directory / "module_manifest.json").write_text(
        canonical_text(
            module_manifest(
                module_id,
                version,
                stage,
                requires_artifacts,
                provides_artifact,
            )
        ),
        encoding="utf-8",
    )
    (directory / "algorithm_card.json").write_text(
        canonical_text(algorithm_card(module_id, version)),
        encoding="utf-8",
    )
    (directory / "configuration_schema.json").write_text(
        canonical_text(configuration_schema()),
        encoding="utf-8",
    )
    sentinel = root / ("implementation_executed_" + module_id + ".txt")
    implementation_source = (
        "from pathlib import Path\n"
        f"Path({str(sentinel)!r}).write_text('BAD', encoding='utf-8')\n"
    )
    (directory / "implementation.py").write_text(
        implementation_source,
        encoding="utf-8",
    )
    selftest_source = "\n".join(
        [
            "import argparse",
            "parser = argparse.ArgumentParser()",
            "parser.add_argument('--module-root', required=True)",
            "parser.add_argument('--workspace-root', required=True)",
            "parser.parse_args()",
            f"print('SYNTHETIC-{module_id}-SELFTEST C-0')",
            "raise SystemExit(0)",
            "",
        ]
    )
    (directory / "selftest.py").write_text(
        selftest_source,
        encoding="utf-8",
    )
    return directory

def write_run_configuration(
    root: Path,
    file_name: str,
    value: Dict[str, object],
) -> Path:
    """Write one explicit fixture under the controlled Run config directory."""

    path = root / "02_CONFIG/run_configs" / file_name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(canonical_text(value), encoding="utf-8")
    return path


def copy_environment(source: Path, target: Path) -> None:
    """Build one isolated U22 workspace without copying company data."""

    target.mkdir(parents=True)
    for stage in STAGES:
        (target / "01_PROGRAMS/modules" / stage).mkdir(parents=True)
    (target / "00_CONTROL/contracts").mkdir(parents=True)
    (target / "00_CONTROL/registry").mkdir(parents=True)
    (target / "01_PROGRAMS/core").mkdir(parents=True)
    (target / "02_CONFIG/run_configs").mkdir(parents=True)
    (target / "07_OUTPUT").mkdir(parents=True)
    contract_names = (
        "u002_module_manifest_schema_v0_1.json",
        "u002_artifact_manifest_schema_v0_1.json",
        "u002_run_configuration_schema_v0_1.json",
        "u002_algorithm_card_schema_v0_1.json",
    )
    for name in contract_names:
        source_file = source / "00_CONTROL/contracts" / name
        if not regular_file(source_file):
            raise AcceptanceError("CONTRACT_MISSING_" + name)
        shutil.copy2(
            source_file,
            target / "00_CONTROL/contracts" / name,
        )
    core_names = (
        VALIDATOR,
        REGISTRY,
        ARTIFACT_STORE,
        QUALIFICATION,
    )
    for name in core_names:
        source_file = source / "01_PROGRAMS/core" / name
        if not regular_file(source_file):
            raise AcceptanceError("CORE_DEPENDENCY_MISSING_" + name)
        shutil.copy2(
            source_file,
            target / "01_PROGRAMS/core" / name,
        )


def qualify(root: Path, module_id: str) -> subprocess.CompletedProcess:
    """Use the accepted U22A4 responsibility to qualify one fixture Module."""

    program = root / "01_PROGRAMS/core" / QUALIFICATION
    return run(
        [
            sys.executable,
            str(program),
            "--qualify-module",
            module_id + "@0.1.0",
            "--workspace-root",
            str(root),
        ]
    )


def prepare(
    program: Path,
    root: Path,
    config_path: Path,
) -> subprocess.CompletedProcess:
    """Invoke one non-executing Engine Preflight operation."""

    return run(
        [
            sys.executable,
            str(program),
            "--prepare-run",
            str(config_path),
            "--workspace-root",
            str(root),
        ]
    )


def install_candidate(root: Path, candidate: Path, checks: List[bool]) -> Path:
    """Install the byte-identical accepted Core program atomically."""

    installed = root / "01_PROGRAMS/core" / INSTALLED
    if installed.exists():
        check(regular_file(installed), "INSTALL_NOT_REGULAR", checks)
        check(
            installed.read_bytes() == candidate.read_bytes(),
            "INSTALL_CONFLICT",
            checks,
        )
        return installed
    temporary = installed.with_name(installed.name + ".installing")
    check(not temporary.exists(), "INSTALL_TEMP_EXISTS", checks)
    temporary.write_bytes(candidate.read_bytes())
    os.replace(temporary, installed)
    check(
        installed.read_bytes() == candidate.read_bytes(),
        "INSTALL_IDENTITY",
        checks,
    )
    return installed


def archive_previous(root: Path) -> int:
    """Move completed U22A4 transfer files out of active TEMP."""

    archive = root / "A-Archive/U22_COMPLETED/U22A4/04_TEMP"
    names = (
        "u002_u22_module_qualification_v0_1.py.txt",
        "udp002_u22a4_module_qualification_acceptance_v0_1.py.txt",
    )
    moved = 0
    for name in names:
        source = root / "04_TEMP" / name
        if not regular_file(source):
            continue
        archive.mkdir(parents=True, exist_ok=True)
        destination = archive / name
        if destination.exists():
            if destination.read_bytes() != source.read_bytes():
                raise AcceptanceError("ARCHIVE_CONFLICT")
            source.unlink()
        else:
            os.replace(source, destination)
        moved += 1
    return moved


def main(argv: Sequence[str] = None) -> int:
    """Run isolated Preflight tests and install only on full success."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    checks: List[bool] = []
    try:
        root = Path(args.workspace_root).resolve()
        check(
            root.is_dir() and root.name == "U-UDP002",
            "WORKSPACE_ROOT",
            checks,
        )
        candidate = root / "04_TEMP" / CANDIDATE
        check(regular_file(candidate), "CANDIDATE_MISSING", checks)
        qualification = root / "01_PROGRAMS/core" / QUALIFICATION
        check(regular_file(qualification), "QUALIFICATION_MISSING", checks)
        qualification_selftest = run(
            [
                sys.executable,
                str(qualification),
                "--self-test",
                "--workspace-root",
                str(root),
            ]
        )
        check(
            qualification_selftest.returncode == 0
            and "N-T24" in qualification_selftest.stdout,
            "QUALIFICATION_SELFTEST",
            checks,
        )
        source = candidate.read_text(encoding="utf-8")
        check(
            not contains_semicolon_operator(source),
            "SEMICOLON_STYLE",
            checks,
        )
        check("def prepare_run(" in source, "PREPARE_INTERFACE", checks)
        check("implementation.py" not in source, "IMPLEMENTATION_LITERAL", checks)
        with tempfile.TemporaryDirectory(prefix="u22a5_acceptance_") as temporary:
            test_root = Path(temporary) / "U-UDP002"
            copy_environment(root, test_root)
            test_candidate = test_root / "01_PROGRAMS/core" / INSTALLED
            shutil.copy2(candidate, test_candidate)
            input_package = test_root / "03_CASES/SYNTHETIC_U001_PACKAGE"
            input_package.mkdir(parents=True)
            (input_package / "package_identity.txt").write_text(
                "SYNTHETIC\n",
                encoding="utf-8",
            )
            write_module(
                test_root,
                "Synthetic.S0.Valid",
                "S0",
                [],
                "S0_OUTPUT",
            )
            write_module(
                test_root,
                "Synthetic.S0.Unqualified",
                "S0",
                [],
                "S0_OTHER_OUTPUT",
            )
            write_module(
                test_root,
                "Synthetic.S1.Dependency",
                "S1",
                ["UNKNOWN_INPUT"],
                "S1_OUTPUT",
            )
            good_qualification = qualify(test_root, "Synthetic.S0.Valid")
            check(
                good_qualification.returncode == 0,
                "GOOD_QUALIFICATION",
                checks,
            )
            dependency_qualification = qualify(
                test_root,
                "Synthetic.S1.Dependency",
            )
            check(
                dependency_qualification.returncode == 0,
                "DEPENDENCY_QUALIFICATION",
                checks,
            )
            selftest = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--self-test",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                selftest.returncode == 0 and "N-T24" in selftest.stdout,
                "CANDIDATE_SELFTEST",
                checks,
            )
            good_config = write_run_configuration(
                test_root,
                "run_good.json",
                run_configuration(
                    "RUN_GOOD",
                    "03_CASES/SYNTHETIC_U001_PACKAGE",
                    "Synthetic.S0.Valid",
                    "S0",
                    "S0_OUTPUT",
                    {"Threshold": 1.0},
                ),
            )
            prepared = prepare(test_candidate, test_root, good_config)
            if prepared.returncode != 0:
                child_reason = prepared.stdout.strip() or prepared.stderr.strip()
                raise AcceptanceError("GOOD_PREPARE_" + child_reason)
            check(
                prepared.returncode == 0 and "N-M1-S1-R0-P1" in prepared.stdout,
                "GOOD_PREPARE",
                checks,
            )
            run_root = test_root / "07_OUTPUT/RUN_GOOD"
            check(run_root.is_dir(), "RUN_ROOT_MISSING", checks)
            status_path = run_root / "run_status.json"
            check(regular_file(status_path), "RUN_STATUS_MISSING", checks)
            status = json.loads(status_path.read_text(encoding="utf-8"))
            check(status["Status"] == "PREPARED", "RUN_STATUS", checks)
            plan_path = run_root / "00_CONFIG/resolved_run_plan.json"
            check(regular_file(plan_path), "RESOLVED_PLAN_MISSING", checks)
            plan = json.loads(plan_path.read_text(encoding="utf-8"))
            check(len(plan["ModulePlan"]) == 1, "PLAN_MODULE_COUNT", checks)
            check(len(plan["PlanSha256"]) == 64, "PLAN_HASH", checks)
            check(
                plan["ModulePlan"][0]["Stage"] == "S0",
                "PLAN_STAGE",
                checks,
            )
            check(
                not (
                    test_root
                    / "implementation_executed_Synthetic.S0.Valid.txt"
                ).exists(),
                "IMPLEMENTATION_EXECUTED",
                checks,
            )
            replay = prepare(test_candidate, test_root, good_config)
            check(
                replay.returncode == 0 and "N-M1-S1-R1-P1" in replay.stdout,
                "IDEMPOTENT_REPLAY",
                checks,
            )
            bad_config = write_run_configuration(
                test_root,
                "run_bad_config.json",
                run_configuration(
                    "RUN_BAD_CONFIG",
                    "03_CASES/SYNTHETIC_U001_PACKAGE",
                    "Synthetic.S0.Valid",
                    "S0",
                    "S0_OUTPUT",
                    {"Unknown": 1},
                ),
            )
            bad_configuration_result = prepare(
                test_candidate,
                test_root,
                bad_config,
            )
            check(
                bad_configuration_result.returncode == 7
                and "MODULE_CONFIGURATION" in bad_configuration_result.stdout,
                "BAD_CONFIGURATION_REJECTED",
                checks,
            )
            unqualified_config = write_run_configuration(
                test_root,
                "run_unqualified.json",
                run_configuration(
                    "RUN_UNQUALIFIED",
                    "03_CASES/SYNTHETIC_U001_PACKAGE",
                    "Synthetic.S0.Unqualified",
                    "S0",
                    "S0_OTHER_OUTPUT",
                    {"Threshold": 1.0},
                ),
            )
            unqualified_result = prepare(
                test_candidate,
                test_root,
                unqualified_config,
            )
            check(
                unqualified_result.returncode == 7
                and "MODULE_NOT_AVAILABLE" in unqualified_result.stdout,
                "UNQUALIFIED_REJECTED",
                checks,
            )
            dependency_config = write_run_configuration(
                test_root,
                "run_dependency.json",
                run_configuration(
                    "RUN_DEPENDENCY",
                    "03_CASES/SYNTHETIC_U001_PACKAGE",
                    "Synthetic.S1.Dependency",
                    "S1",
                    "S1_OUTPUT",
                    {"Threshold": 1.0},
                ),
            )
            dependency_result = prepare(
                test_candidate,
                test_root,
                dependency_config,
            )
            check(
                dependency_result.returncode == 7
                and "PLAN_ARTIFACT_DEPENDENCY" in dependency_result.stdout,
                "DEPENDENCY_REJECTED",
                checks,
            )
            expected_directories = [
                "00_CONFIG",
                "01_LOGS",
                "02_DIAGNOSTICS",
                "03_ARTIFACTS/SOURCE",
                "03_ARTIFACTS/S0",
                "03_ARTIFACTS/S7",
                "04_REVIEW_PACKAGE",
                "05_EXPORTS",
                "00_STAGING",
            ]
            check(
                all((run_root / item).is_dir() for item in expected_directories),
                "RUN_DIRECTORIES",
                checks,
            )
        installed = install_candidate(root, candidate, checks)
        prepared_root = root / "07_OUTPUT"
        prepared_root.mkdir(parents=True, exist_ok=True)
        check(prepared_root.is_dir(), "OUTPUT_DIRECTORY", checks)
        installed_selftest = run(
            [
                sys.executable,
                str(installed),
                "--self-test",
                "--workspace-root",
                str(root),
            ]
        )
        check(
            installed_selftest.returncode == 0 and "N-T24" in installed_selftest.stdout,
            "INSTALLED_SELFTEST",
            checks,
        )
        if len(checks) != 28:
            raise AcceptanceError("CHECK_COUNT_PRE_" + str(len(checks)))
        checks.append(True)
        if len(checks) != 29:
            raise AcceptanceError("CHECK_COUNT")
        archived = archive_previous(root)
        print(
            f"U22A5 N-P1-R1-V3-F3-T29-I1-A{archived}-X0 C-0"
        )
        return 0
    except Exception as exc:
        reason = reformat_reason(exc)
        print(
            f"U22A5 N-P0-R0-V0-F0-T0-I0-A0-X1 C-7 R-{reason}"
        )
        return 7


def reformat_reason(exc: Exception) -> str:
    """Keep the failure return short enough for manual transfer."""

    source = str(exc).upper() or exc.__class__.__name__.upper()
    normalized = "".join(
        character if character.isalnum() else "_"
        for character in source
    )
    return normalized.strip("_")[:80] or "INTERNAL"


if __name__ == "__main__":
    raise SystemExit(main())
