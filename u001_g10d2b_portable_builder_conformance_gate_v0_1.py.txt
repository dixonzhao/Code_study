#!/usr/bin/env python3
"""Independent structural conformance gate for G10D2B builder v0.7."""

from __future__ import annotations

import argparse
import ast
import contextlib
import importlib.util
import io
import json
import subprocess
import sys
import tempfile
from pathlib import Path


GATE_NAME = "u001_g10d2b_portable_builder_conformance_gate_v0_1.py"
GATE_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2B-CG1-SELFTEST-PASS-16"
EXPECTED_BUILDER_TOKEN = "U001-PORTABLE-HOST-PACKAGE-BUILDER-V07-SELFTEST-PASS-40"
EXPECTED_PROGRAMS = (
    "u001_foundation_core_v0_1.py",
    "u001_selected_update_decoder_v0_1.py",
    "u001_framecount_assignment_v0_1.py",
    "u001_semantic_validity_classifier_v0_1.py",
    "u001_framecount_view_builder_v0_1.py",
    "u001_signal_cadence_bounded_hold_v0_1.py",
    "u001_output_package_assembler_v0_3.py",
    "u001_g8a_thin_orchestrator_v0_5.py",
    "u001_auto_discovery_profile_builder_v0_2.py",
    "u001_g10b_single_pass_discovery_engine_v0_5a.py",
    "u001_discovery_controller_v0_2.py",
    "u001_data_preparation_workbench_gui_v0_4a.py",
)
EXPECTED_CONTROLS = (
    "u001_g2_selected_signals.csv",
    "u001_g4_signal_policy.csv",
    "u001_g5_frame_policy.csv",
    "u001_g6_hold_policy.csv",
)
EXPECTED_HIDDEN = (
    "tkinter.filedialog",
    "tkinter.messagebox",
    "tkinter.ttk",
)
EXPECTED_EXCLUDES = ("pandas.tests", "numpy.tests")


class GateFailure(RuntimeError):
    pass


def fail(code: str) -> None:
    raise GateFailure(code)


def load_candidate(path: Path):
    spec = importlib.util.spec_from_file_location("g10d2b_candidate", path)
    if spec is None or spec.loader is None:
        fail("IMPORT_SPEC")
    module = importlib.util.module_from_spec(spec)
    with contextlib.redirect_stdout(io.StringIO()), contextlib.redirect_stderr(io.StringIO()):
        spec.loader.exec_module(module)
    return module


def run_source_selftest(path: Path) -> None:
    result = subprocess.run(
        [sys.executable, str(path), "--self-test"],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=120,
        shell=False,
    )
    if result.returncode != 0:
        fail("SELFTEST_RC")
    lines = [line.strip() for line in result.stdout.splitlines() if line.strip()]
    if lines != [EXPECTED_BUILDER_TOKEN]:
        fail("SELFTEST_TOKEN")


def require_contract(module) -> dict:
    fn = getattr(module, "get_portable_builder_contract", None)
    if not callable(fn):
        fail("CONTRACT_API")
    contract = fn()
    if not isinstance(contract, dict):
        fail("CONTRACT_TYPE")
    expected = {
        "ProgramName": "u001_portable_host_package_builder_v0_7.py",
        "ProgramVersion": "0.7",
        "PackageName": "U001_Portable_v0_2",
        "ManifestSchema": "UDP001_PORTABLE_HOST_PACKAGE_V0_2",
        "ExpectedRuntimeHostVersion": "0.5",
        "DefaultGUIName": "u001_data_preparation_workbench_gui_v0_4a.py",
        "Architecture": "PYINSTALLER_ONEDIR_RUNTIME_HOST_WITH_EXTERNAL_PROGRAMS_AND_CONTROLS",
        "ProgramCount": 12,
        "ControlCount": 4,
        "CompanyDataIncluded": False,
        "RealBuildRequiredForAcceptance": True,
    }
    for key, value in expected.items():
        if contract.get(key) != value:
            fail("CONTRACT_" + key.upper())
    if tuple(contract.get("ExpectedPrograms", ())) != EXPECTED_PROGRAMS:
        fail("CONTRACT_PROGRAMS")
    if tuple(contract.get("ExpectedControls", ())) != EXPECTED_CONTROLS:
        fail("CONTRACT_CONTROLS")
    if list(contract.get("ExpectedDependencies", [])) != ["pandas"]:
        fail("CONTRACT_DEPENDENCIES")
    return contract


def require_plan(module) -> dict:
    fn = getattr(module, "build_release_plan", None)
    if not callable(fn):
        fail("PLAN_API")
    root = Path("X:/C-CODE/UDP001")
    host = root / "04_TEMP" / "u001_portable_runtime_host_v0_5.py"
    out = Path("X:/C-CODE/UDP001_RELEASE")
    before = (root.exists(), host.exists(), out.exists())
    plan = fn(root, host, out)
    after = (root.exists(), host.exists(), out.exists())
    if before != after or not isinstance(plan, dict):
        fail("PLAN_NOT_PURE")
    encoded = json.dumps(plan, ensure_ascii=False, sort_keys=True, default=str)
    required = (
        "U001_Portable_v0_2",
        "01_PROGRAMS",
        "02_CONTROL",
        "03_CASES",
        "00_LAUNCHER",
        "u001_portable_host_package_manifest.json",
    )
    if not all(token in encoded for token in required):
        fail("PLAN_CONTENT")
    if "preflight" in encoded.lower() or "v0_3.py" in encoded:
        fail("PLAN_LEGACY")
    return plan


def require_pyinstaller_command(module) -> None:
    fn = getattr(module, "pyinstaller_host_command", None)
    if not callable(fn):
        fail("PYINSTALLER_API")
    command = fn(Path("host_v0_5.py"), Path("dist"), Path("work"), Path("spec"))
    if not isinstance(command, list) or not all(isinstance(item, str) for item in command):
        fail("PYINSTALLER_TYPE")
    if "--onedir" not in command or "--onefile" in command or "--windowed" not in command:
        fail("PYINSTALLER_MODE")
    if "--collect-all" in command:
        fail("PYINSTALLER_COLLECT_ALL")
    if not all(item in command for item in EXPECTED_HIDDEN):
        fail("PYINSTALLER_HIDDEN")
    if not all(item in command for item in EXPECTED_EXCLUDES):
        fail("PYINSTALLER_EXCLUDES")
    if "U001_Workbench" not in command or str(Path("host_v0_5.py")) not in command:
        fail("PYINSTALLER_TARGET")


def require_source(path: Path) -> None:
    text = path.read_text(encoding="utf-8-sig")
    try:
        ast.parse(text, filename=str(path))
    except SyntaxError:
        fail("SYNTAX")
    required = (
        'PROGRAM_VERSION = "0.7"',
        'PACKAGE_NAME = "U001_Portable_v0_2"',
        'MANIFEST_SCHEMA = "UDP001_PORTABLE_HOST_PACKAGE_V0_2"',
        "U001-PORTABLE-HOST-V05-SELFTEST-PASS-24",
        "U001PK04",
        "N-R1-D1-P12-C4-H1-B1-E1-S12-Z1-M1-X0",
        "02_CONTROL",
        "--onedir",
        "--windowed",
    )
    if not all(token in text for token in required):
        fail("SOURCE_REQUIRED")
    forbidden = (
        "U001-PORTABLE-RUNTIME-HOST-V04-SELFTEST-PASS-25",
        "U001_Portable_v0_1",
        "UDP001_PORTABLE_HOST_PACKAGE_V0_1",
        "--preflight-report",
        "PREFLIGHT_SCHEMA",
        "P9-H1",
    )
    if any(token in text for token in forbidden):
        fail("SOURCE_LEGACY")
    if text.count("u001_data_preparation_workbench_gui_v0_3.py") > 1:
        fail("SOURCE_GUI03_EXPANSION")


def validate(path: Path) -> None:
    if not path.is_file():
        fail("CANDIDATE_MISSING")
    require_source(path)
    run_source_selftest(path)
    module = load_candidate(path)
    if getattr(module, "PROGRAM_NAME", None) != "u001_portable_host_package_builder_v0_7.py":
        fail("IDENTITY_NAME")
    if getattr(module, "PROGRAM_VERSION", None) != "0.7":
        fail("IDENTITY_VERSION")
    if getattr(module, "PACKAGE_NAME", None) != "U001_Portable_v0_2":
        fail("IDENTITY_PACKAGE")
    if tuple(getattr(module, "EXPECTED_PROGRAMS", ())) != EXPECTED_PROGRAMS:
        fail("PROGRAM_INVENTORY")
    if tuple(getattr(module, "EXPECTED_CONTROLS", ())) != EXPECTED_CONTROLS:
        fail("CONTROL_INVENTORY")
    if "u001_data_preparation_workbench_gui_v0_3.py" in EXPECTED_PROGRAMS:
        fail("LEGACY_GUI_PACKAGED")
    require_contract(module)
    require_plan(module)
    require_pyinstaller_command(module)


def run_self_test() -> None:
    checks = (
        GATE_VERSION == "0.1",
        GATE_NAME.endswith("v0_1.py"),
        SELFTEST_TOKEN.endswith("PASS-16"),
        len(EXPECTED_PROGRAMS) == 12,
        len(set(EXPECTED_PROGRAMS)) == 12,
        EXPECTED_PROGRAMS[-1].endswith("gui_v0_4a.py"),
        len(EXPECTED_CONTROLS) == 4,
        len(set(EXPECTED_CONTROLS)) == 4,
        all(name.endswith(".csv") for name in EXPECTED_CONTROLS),
        len(EXPECTED_HIDDEN) == 3,
        all(name.startswith("tkinter.") for name in EXPECTED_HIDDEN),
        len(EXPECTED_EXCLUDES) == 2,
        "pandas.tests" in EXPECTED_EXCLUDES,
        "numpy.tests" in EXPECTED_EXCLUDES,
        EXPECTED_BUILDER_TOKEN.endswith("PASS-40"),
        "u001_data_preparation_workbench_gui_v0_3.py" not in EXPECTED_PROGRAMS,
    )
    if sum(bool(item) for item in checks) != 16:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="G10D2B independent builder v0.7 conformance gate")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--candidate")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.candidate:
        parser.error("--candidate is required")
    try:
        validate(Path(args.candidate))
    except (GateFailure, Exception) as exc:
        reason = exc.args[0] if isinstance(exc, GateFailure) and exc.args else type(exc).__name__
        print(f"G10D2BCG1-FAIL:{reason} C-7")
        return 7
    print("G10D2BCG1-PASS P12-C4-O1-M1-X0 C-0")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
