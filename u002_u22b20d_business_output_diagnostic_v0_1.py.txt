class DiagnosticError(RuntimeError):
    """Controlled diagnostic failure."""

    def __init__(self, reason: object, detail: object = "NA") -> None:
        self.reason = token(reason)
        self.detail = token(detail)[:160]
        super().__init__(f"{self.reason}: {self.detail}")


def token(value: object) -> str:
    """Return a compact manual-transfer-safe token."""

    text = str(value).strip().upper()
    result = "".join(character if character.isalnum() else "_" for character in text)
    return result.strip("_") or "NA"


def read_json(path: Path, label: str) -> Dict[str, Any]:
    """Read one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise DiagnosticError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise DiagnosticError(label + "_JSON", error) from error
    if not isinstance(value, dict):
        raise DiagnosticError(label + "_TYPE", type(value).__name__)
    return value


def write_json(path: Path, value: Mapping[str, object]) -> None:
    """Write deterministic diagnostic evidence."""

    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True, allow_nan=False) + "\n",
        encoding="utf-8",
    )


def sha256_file(path: Path) -> str:
    """Hash one payload without loading the complete file into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def selected_entry(index: Mapping[str, object], artifact_type: str, owner: str) -> Dict[str, Any]:
    """Select exactly one Artifact of the requested type."""

    artifacts = index.get("Artifacts")
    if not isinstance(artifacts, list):
        raise DiagnosticError(owner + "_ARTIFACTS")
    matches = [
        item
        for item in artifacts
        if isinstance(item, dict) and item.get("ArtifactType") == artifact_type
    ]
    if len(matches) != 1:
        raise DiagnosticError(owner + "_COUNT", f"{artifact_type}@{len(matches)}")
    return matches[0]


def package_root(run_root: Path, entry: Mapping[str, object]) -> Path:
    """Resolve the governed package folder from its manifest path."""

    relative = entry.get("ManifestRelativePath")
    if not isinstance(relative, str) or not relative:
        raise DiagnosticError("MANIFEST_PATH", entry.get("ArtifactType"))
    return (run_root / relative).parent


def payload_map(entry: Mapping[str, object]) -> Dict[str, Dict[str, Any]]:
    """Index declared payloads by stable relative path."""

    payloads = entry.get("Payloads")
    if not isinstance(payloads, list):
        raise DiagnosticError("PAYLOADS_TYPE", entry.get("ArtifactType"))
    result: Dict[str, Dict[str, Any]] = {}
    for payload in payloads:
        if not isinstance(payload, dict) or not isinstance(payload.get("RelativePath"), str):
            raise DiagnosticError("PAYLOAD_DECLARATION", entry.get("ArtifactType"))
        name = str(payload["RelativePath"])
        if name in result:
            raise DiagnosticError("PAYLOAD_DUPLICATE", name)
        result[name] = payload
    return result


def read_csv(path: Path) -> Tuple[List[str], List[List[str]]]:
    """Read one governed CSV for cell-level comparison."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.reader(stream))
    return (rows[0] if rows else []), (rows[1:] if rows else [])


def compare_csv(reference: Path, candidate: Path) -> Dict[str, Any]:
    """Compare CSV structure and every business cell."""

    ref_header, ref_rows = read_csv(reference)
    new_header, new_rows = read_csv(candidate)
    differing_cells = 0
    differing_rows = set()
    first: Dict[str, Any] | None = None
    maximum_rows = max(len(ref_rows), len(new_rows))
    maximum_columns = max(len(ref_header), len(new_header))
    for row_index in range(maximum_rows):
        left = ref_rows[row_index] if row_index < len(ref_rows) else []
        right = new_rows[row_index] if row_index < len(new_rows) else []
        for column_index in range(maximum_columns):
            left_value = left[column_index] if column_index < len(left) else None
            right_value = right[column_index] if column_index < len(right) else None
            if left_value != right_value:
                differing_cells += 1
                differing_rows.add(row_index + 1)
                if first is None:
                    name = (
                        ref_header[column_index]
                        if column_index < len(ref_header)
                        else new_header[column_index]
                        if column_index < len(new_header)
                        else "NA"
                    )
                    first = {
                        "DataRow": row_index + 1,
                        "ColumnIndex": column_index + 1,
                        "ColumnName": name,
                        "ReferenceValue": left_value,
                        "CandidateValue": right_value,
                    }
    return {
        "Format": "CSV",
        "HeaderEqual": ref_header == new_header,
        "ReferenceRowCount": len(ref_rows),
        "CandidateRowCount": len(new_rows),
        "DifferingRowCount": len(differing_rows),
        "DifferingCellCount": differing_cells,
        "FirstDifference": first,
    }


def flatten_json(value: object, prefix: str = "$") -> Dict[str, object]:
    """Flatten JSON scalars so the first semantic path is visible."""

    result: Dict[str, object] = {}
    if isinstance(value, dict):
        if not value:
            result[prefix] = {}
        for key in sorted(value):
            result.update(flatten_json(value[key], prefix + "." + str(key)))
    elif isinstance(value, list):
        if not value:
            result[prefix] = []
        for index, item in enumerate(value):
            result.update(flatten_json(item, f"{prefix}[{index}]"))
    else:
        result[prefix] = value
    return result


def compare_json(reference: Path, candidate: Path) -> Dict[str, Any]:
    """Compare JSON values by semantic path."""

    left = flatten_json(json.loads(reference.read_text(encoding="utf-8-sig")))
    right = flatten_json(json.loads(candidate.read_text(encoding="utf-8-sig")))
    paths = sorted(set(left) | set(right))
    differences = [path for path in paths if left.get(path) != right.get(path)]
    first = None
    if differences:
        path = differences[0]
        first = {
            "JsonPath": path,
            "ReferenceValue": left.get(path),
            "CandidateValue": right.get(path),
        }
    return {
        "Format": "JSON",
        "DifferingPathCount": len(differences),
        "FirstDifference": first,
        "FirstTwentyDifferingPaths": differences[:20],
    }


def compare_text(reference: Path, candidate: Path) -> Dict[str, Any]:
    """Compare JSONL or plain text by line."""

    left = reference.read_text(encoding="utf-8-sig").splitlines()
    right = candidate.read_text(encoding="utf-8-sig").splitlines()
    differing = []
    for index in range(max(len(left), len(right))):
        a = left[index] if index < len(left) else None
        b = right[index] if index < len(right) else None
        if a != b:
            differing.append(index + 1)
    return {
        "Format": "TEXT",
        "ReferenceLineCount": len(left),
        "CandidateLineCount": len(right),
        "DifferingLineCount": len(differing),
        "FirstDifference": {"Line": differing[0]} if differing else None,
    }


def compare_payload_files(reference: Path, candidate: Path) -> Dict[str, Any]:
    """Select an explainable comparison appropriate for the payload format."""

    if reference.suffix.lower() == ".csv" and candidate.suffix.lower() == ".csv":
        return compare_csv(reference, candidate)
    if reference.suffix.lower() == ".json" and candidate.suffix.lower() == ".json":
        return compare_json(reference, candidate)
    return compare_text(reference, candidate)


def compare_artifact(
    artifact_type: str,
    reference_root: Path,
    candidate_root: Path,
    reference_entry: Mapping[str, object],
    candidate_entry: Mapping[str, object],
) -> Dict[str, Any]:
    """Compare every declared payload and report primary versus auxiliary scope."""

    ref_payloads = payload_map(reference_entry)
    new_payloads = payload_map(candidate_entry)
    ref_package = package_root(reference_root, reference_entry)
    new_package = package_root(candidate_root, candidate_entry)
    details = []
    for name in sorted(set(ref_payloads) | set(new_payloads)):
        left = ref_payloads.get(name)
        right = new_payloads.get(name)
        kind = str((right or left or {}).get("PayloadKind", "UNKNOWN"))
        detail: Dict[str, Any] = {
            "RelativePath": name,
            "PayloadKind": kind,
            "ReferenceDeclared": left is not None,
            "CandidateDeclared": right is not None,
        }
        if left is None or right is None:
            detail.update({"Equal": False, "DifferenceClass": "DECLARATION"})
        else:
            ref_path = ref_package / name
            new_path = new_package / name
            if not ref_path.is_file() or not new_path.is_file():
                detail.update({"Equal": False, "DifferenceClass": "MISSING_FILE"})
            else:
                ref_hash = sha256_file(ref_path)
                new_hash = sha256_file(new_path)
                equal = ref_hash == new_hash
                detail.update(
                    {
                        "Equal": equal,
                        "ReferenceSha256": ref_hash,
                        "CandidateSha256": new_hash,
                        "ReferenceRowCount": left.get("RowCount"),
                        "CandidateRowCount": right.get("RowCount"),
                    }
                )
                if not equal:
                    detail["SemanticDifference"] = compare_payload_files(ref_path, new_path)
        details.append(detail)
    differing = [item for item in details if not item.get("Equal")]
    primary = [item for item in differing if item.get("PayloadKind") == "PRIMARY"]
    return {
        "ArtifactType": artifact_type,
        "ReferenceProducerModuleId": reference_entry.get("ProducerModuleId"),
        "ReferenceProducerModuleVersion": reference_entry.get("ProducerModuleVersion"),
        "CandidateProducerModuleId": candidate_entry.get("ProducerModuleId"),
        "CandidateProducerModuleVersion": candidate_entry.get("ProducerModuleVersion"),
        "PayloadCount": len(details),
        "DifferingPayloadCount": len(differing),
        "PrimaryDifferingPayloadCount": len(primary),
        "Payloads": details,
    }


def candidate_count(entry: Mapping[str, object], preferred_name: str) -> int:
    """Read the declared candidate-row count from one primary inventory."""

    payload = payload_map(entry).get(preferred_name)
    count = payload.get("RowCount") if payload else None
    return count if isinstance(count, int) and not isinstance(count, bool) else -1


def first_difference(reports: Sequence[Mapping[str, object]]) -> Tuple[str, int, int]:
    """Return filename, first row and first column index for the compact capsule."""

    for report in reports:
        payloads = report.get("Payloads")
        if not isinstance(payloads, list):
            continue
        for payload in payloads:
            if not isinstance(payload, dict) or payload.get("Equal") is not False:
                continue
            semantic = payload.get("SemanticDifference")
            first = semantic.get("FirstDifference") if isinstance(semantic, dict) else None
            row = first.get("DataRow", first.get("Line", 0)) if isinstance(first, dict) else 0
            column = first.get("ColumnIndex", 0) if isinstance(first, dict) else 0
            return token(payload.get("RelativePath"))[:24], int(row or 0), int(column or 0)
    return "NA", 0, 0


def diagnostic(root: Path) -> str:
    """Generate the complete read-only Source-versus-Candidate diagnosis."""

    profile = read_json(root / PROFILE_RELATIVE, "PROFILE")
    configuration = read_json(root / CONFIG_RELATIVE, "CONFIGURATION")
    source_id = profile.get("SourceGoldenRunId")
    candidate_id = configuration.get("RunId")
    if not isinstance(source_id, str) or not isinstance(candidate_id, str):
        raise DiagnosticError("RUN_ID")
    source_root = root / "07_OUTPUT" / source_id
    candidate_root = root / "07_OUTPUT" / candidate_id
    source_index = read_json(source_root / INDEX_NAME, "SOURCE_INDEX")
    candidate_index = read_json(candidate_root / INDEX_NAME, "CANDIDATE_INDEX")
    reports = []
    for artifact_type in TARGET_TYPES:
        reports.append(
            compare_artifact(
                artifact_type,
                source_root,
                candidate_root,
                selected_entry(source_index, artifact_type, "SOURCE"),
                selected_entry(candidate_index, artifact_type, "CANDIDATE"),
            )
        )
    different_artifacts = [item for item in reports if item["DifferingPayloadCount"]]
    primary_differences = sum(int(item["PrimaryDifferingPayloadCount"]) for item in reports)
    all_differences = sum(int(item["DifferingPayloadCount"]) for item in reports)
    auxiliary_differences = all_differences - primary_differences
    source_s6 = selected_entry(source_index, "ANOMALY_CANDIDATES", "SOURCE")
    candidate_s6 = selected_entry(candidate_index, "ANOMALY_CANDIDATES", "CANDIDATE")
    ref_count = candidate_count(source_s6, "candidate_inventory.csv")
    new_count = candidate_count(candidate_s6, "candidate_inventory.csv")
    first_file, first_row, first_column = first_difference(reports)
    difference_class = "EQUAL" if not different_artifacts else "PRIMARY" if primary_differences else "AUX"
    report = {
        "ContractId": "U002_B20D_BUSINESS_OUTPUT_DIAGNOSTIC",
        "ContractVersion": "0.1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "GeneratedAtUtc": dt.datetime.now(dt.timezone.utc).isoformat(),
        "ReadOnlyDiagnostic": True,
        "SourceGoldenRunId": source_id,
        "CandidateRunId": candidate_id,
        "DifferenceClass": difference_class,
        "DifferingArtifactCount": len(different_artifacts),
        "DifferingPayloadCount": all_differences,
        "PrimaryDifferingPayloadCount": primary_differences,
        "AuxiliaryDifferingPayloadCount": auxiliary_differences,
        "SourceCandidateCount": ref_count,
        "CandidateCandidateCount": new_count,
        "ArtifactComparisons": reports,
    }
    write_json(root / REPORT_RELATIVE, report)
    stage_counts = {str(item["ArtifactType"]): int(item["DifferingPayloadCount"]) for item in reports}
    capsule = (
        f"B20DD-N{len(different_artifacts)}-S6D{stage_counts['ANOMALY_CANDIDATES']}-"
        f"S7D{stage_counts['U003_REVIEW_PACKAGE']}-P{primary_differences}-A{auxiliary_differences}-"
        f"CR{ref_count}-CN{new_count}-F{first_file}-R{first_row}-J{first_column}-"
        f"K{difference_class}-X0-C0"
    )
    print(capsule)
    return capsule


def prepare(root: Path) -> str:
    """Install this diagnostic without changing analytical programs or outputs."""

    source = Path(__file__).resolve()
    destination = root / INSTALL_RELATIVE
    destination.parent.mkdir(parents=True, exist_ok=True)
    if source != destination:
        shutil.copy2(source, destination)
    print("PASS | B20C configuration and Source Golden Profile paths are fixed | WHY: diagnosis must compare the exact failed B20D pair")
    print("PASS | Diagnostic is read-only except for its own JSON report | WHY: failed Golden evidence must remain intact")
    print(f"B20DDP-I1-R1-X0-C0")
    return "B20DDP-I1-R1-X0-C0"


def self_test() -> str:
    """Exercise CSV and JSON semantic-difference localization."""

    import tempfile

    with tempfile.TemporaryDirectory(prefix="u22b20dd_") as temporary:
        root = Path(temporary)
        left_csv = root / "left.csv"
        right_csv = root / "right.csv"
        left_csv.write_text("A,B\n1,2\n", encoding="utf-8")
        right_csv.write_text("A,B\n1,3\n", encoding="utf-8")
        csv_result = compare_csv(left_csv, right_csv)
        if csv_result["DifferingCellCount"] != 1 or csv_result["FirstDifference"]["ColumnIndex"] != 2:
            raise DiagnosticError("SELFTEST_CSV")
        left_json = root / "left.json"
        right_json = root / "right.json"
        left_json.write_text('{"a":{"b":1}}', encoding="utf-8")
        right_json.write_text('{"a":{"b":2}}', encoding="utf-8")
        json_result = compare_json(left_json, right_json)
        if json_result["DifferingPathCount"] != 1:
            raise DiagnosticError("SELFTEST_JSON")
    print("PASS | CSV first-row and first-column localization works | WHY: Candidate changes must be distinguished from metadata drift")
    print("PASS | JSON semantic-path localization works | WHY: key order or formatting must not masquerade as business change")
    print("B20DDST-N2-X0-C0")
    return "B20DDST-N2-X0-C0"


def parse_arguments(argv: Sequence[str]) -> argparse.Namespace:
    """Parse the standard three-command interface."""

    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    return parser.parse_args(argv)


def main(argv: Sequence[str] | None = None) -> int:
    """Run one fail-closed operation."""

    try:
        arguments = parse_arguments(sys.argv[1:] if argv is None else argv)
        root = Path(arguments.workspace_root).resolve()
        if not root.is_dir() or root.is_symlink():
            raise DiagnosticError("WORKSPACE_ROOT", root)
        if arguments.prepare:
            prepare(root)
        elif arguments.self_test:
            self_test()
        else:
            diagnostic(root)
        return 0
    except DiagnosticError as error:
        print(f"B20DDE-K{error.reason}-D{error.detail}-X1-C7")
        return 7
    except Exception as error:
        print(f"B20DDE-K{token(type(error).__name__)}-D{token(error)[:160]}-X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
