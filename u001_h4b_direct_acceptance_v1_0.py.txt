#!/usr/bin/env python3
"""Direct U001 H4B cadence, explicit no-hold policy, and acceptance."""

from __future__ import annotations

import argparse
import csv
import json
import shutil
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Optional, Sequence


PREPARED_FOLDER = "U1H4BD_PREPARED"
OUTPUT_FOLDER = "U1H4BD_OUTPUT"
EXPECTED_FRAMES = 56596
EXPECTED_SIGNALS = 21
EXPECTED_MISSING_FRAMES = 2
EXPECTED_MAXIMUM_STEP = 2


class H4BError(RuntimeError):
    """Fail-closed H4B direct-acceptance error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def resolve_root(value: Optional[str]) -> Path:
    if value:
        root = Path(value).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise H4BError("U_UDP001_ROOT_NOT_FOUND")
    return root


def write_json(path: Path, value: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> dict[str, Any]:
    if not path.is_file():
        raise H4BError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise H4BError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise H4BError(label + "_ROOT_NOT_OBJECT")
    return value


def dependency_paths(root: Path) -> dict[str, Path]:
    core = root / "01_PROGRAMS" / "core"
    h4a_root = root / "04_TEMP" / "U1H4D_OUTPUT"
    return {
        "H4A": h4a_root / "h4a_package",
        "FramePolicy": h4a_root / "frame_policy" / "u001_h4_compiled_frame_policy.csv",
        "CadenceProfiler": core / "u001_h4b_cadence_profiler_v1_0.py",
        "HoldPolicyCompiler": core / "u001_h4b_hold_policy_compiler_v1_0.py",
        "HoldEngine": core / "u001_h4b_bounded_hold_engine_v1_0.py",
    }


def run(command: list[str], cwd: Path, label: str) -> None:
    print(f"[RUN] {label}")
    result = subprocess.run(command, cwd=str(cwd), check=False)
    if result.returncode != 0:
        raise H4BError(f"{label}_EXIT_{result.returncode}")


def prepare(root_value: Optional[str]) -> int:
    try:
        root = resolve_root(root_value)
        prepared = root / "04_TEMP" / PREPARED_FOLDER
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if prepared.exists():
            shutil.rmtree(prepared)
        if output.exists():
            shutil.rmtree(output)

        required = dependency_paths(root)
        reasons = {
            "H4A": "Accepted one-row-per-observed-FrameCount package.",
            "FramePolicy": "Exact identity and order of the 21 frame signals.",
            "CadenceProfiler": "Measures real update gaps without modifying values.",
            "HoldPolicyCompiler": "Requires explicit per-signal hold decisions.",
            "HoldEngine": "Applies NONE or finite bounded hold with provenance.",
        }
        missing: list[str] = []
        for label, path in required.items():
            found = path.is_dir() if label == "H4A" else path.is_file()
            print(f"[CHECK] {label} | {'FOUND' if found else 'MISSING'} | Why: {reasons[label]}")
            if not found:
                missing.append(label)
        if missing:
            raise H4BError("DEPENDENCY_MISSING:" + ".".join(missing))

        quality = read_json(required["H4A"] / "u001_h4_raw_frame_quality.json", "H4A_QUALITY")
        checks = {
            "OutputFrameRowCount": EXPECTED_FRAMES,
            "IncludedSignalCount": EXPECTED_SIGNALS,
            "MissingFrameGapCount": EXPECTED_MISSING_FRAMES,
            "MaximumAdjacentFrameStep": EXPECTED_MAXIMUM_STEP,
            "CrossFrameHeldCellCount": 0,
            "ZeroFillCount": 0,
        }
        for key, expected in checks.items():
            actual = int(quality.get(key, -1))
            if actual != expected:
                raise H4BError(f"H4A_{key.upper()}_{actual}_EXPECTED_{expected}")

        prepared.mkdir(parents=True)
        write_json(
            prepared / "receipt.json",
            {
                "Schema": "U001_H4B_DIRECT_PREPARE_V1_0",
                "PreparedUtc": utc_now(),
                "Status": "READY",
                "CompletionCode": 0,
            },
        )
        print("H4BP-P5-H1-X0-C0")
        return 0
    except Exception as exc:
        print(f"H4BPE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def require_prepared(root: Path) -> None:
    receipt = read_json(root / "04_TEMP" / PREPARED_FOLDER / "receipt.json", "PREPARE_RECEIPT")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise H4BError("PREPARE_NOT_READY")


def self_test(root_value: Optional[str]) -> int:
    try:
        root = resolve_root(root_value)
        require_prepared(root)
        required = dependency_paths(root)
        tests = [
            ("CadenceProfiler", "Cadence statistics must not alter measurements."),
            ("HoldPolicyCompiler", "No hold policy can enter without explicit approval and a finite bound."),
            ("HoldEngine", "Invalid barriers, age, true zero, and stale state stay distinguishable."),
        ]
        for label, reason in tests:
            print(f"[SELFTEST] {label} | Why: {reason}")
            program = required[label]
            run([sys.executable, str(program), "--self-test"], program.parent, label + "_SELFTEST")
        print("H4BS-C19-P21-E25-X0-C0")
        return 0
    except Exception as exc:
        print(f"H4BSE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def approve_no_hold_policy(template: Path, approved: Path) -> int:
    with template.open("r", encoding="utf-8-sig", newline="") as source:
        reader = csv.DictReader(source)
        fields = list(reader.fieldnames or [])
        rows = list(reader)
    required = {"HoldPolicy", "MaxAgeFrames", "PolicyStatus", "PolicyBasis"}
    if len(rows) != EXPECTED_SIGNALS or not required.issubset(fields):
        raise H4BError("HOLD_POLICY_TEMPLATE")
    for row in rows:
        row["HoldPolicy"] = "NONE"
        row["MaxAgeFrames"] = ""
        row["PolicyStatus"] = "APPROVED"
        row["PolicyBasis"] = "Approved U001 golden-run baseline: preserve only current-frame observations."
        row["Notes"] = "Cadence retained as evidence; it does not authorize imputation."
    with approved.open("w", encoding="utf-8", newline="") as target:
        writer = csv.DictWriter(target, fieldnames=fields, lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)
    return len(rows)


def execute(root_value: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_value)
        require_prepared(root)
        required = dependency_paths(root)
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            shutil.rmtree(output)
        output.mkdir(parents=True)

        cadence = output / "cadence_package"
        policy_candidate = output / "hold_policy_candidate.csv"
        policy_approved = output / "hold_policy_approved.csv"
        policy_package = output / "hold_policy_package"
        h4b = output / "h4b_package"

        run(
            [sys.executable, str(required["CadenceProfiler"]), "--profile",
             "--h4a-package", str(required["H4A"]),
             "--frame-policy", str(required["FramePolicy"]), "--out", str(cadence)],
            root,
            "CADENCE_PROFILE",
        )
        run(
            [sys.executable, str(required["HoldPolicyCompiler"]), "--make-template",
             "--cadence-package", str(cadence), "--out", str(policy_candidate)],
            root,
            "HOLD_POLICY_TEMPLATE",
        )
        approved_count = approve_no_hold_policy(policy_candidate, policy_approved)
        run(
            [sys.executable, str(required["HoldPolicyCompiler"]), "--compile",
             "--cadence-package", str(cadence), "--policy", str(policy_approved),
             "--out", str(policy_package)],
            root,
            "HOLD_POLICY_COMPILE",
        )
        run(
            [sys.executable, str(required["HoldEngine"]), "--apply",
             "--h4a-package", str(required["H4A"]),
             "--policy-package", str(policy_package), "--out", str(h4b)],
            root,
            "H4B_APPLY",
        )

        cadence_quality = read_json(cadence / "u001_h4b_cadence_quality.json", "CADENCE_QUALITY")
        policy_quality = read_json(policy_package / "u001_h4b_hold_policy_quality.json", "HOLD_POLICY_QUALITY")
        h4b_quality = read_json(h4b / "u001_h4_bounded_frame_quality.json", "H4B_QUALITY")

        cadence_expected = {
            "InputFrameCount": EXPECTED_FRAMES,
            "SignalCount": EXPECTED_SIGNALS,
            "MissingFrameCount": EXPECTED_MISSING_FRAMES,
            "MaximumAdjacentFrameStep": EXPECTED_MAXIMUM_STEP,
            "HeldCellCount": 0,
            "ZeroFillCount": 0,
            "InputMutationCount": 0,
        }
        for key, expected in cadence_expected.items():
            actual = int(cadence_quality.get(key, -1))
            if actual != expected:
                raise H4BError(f"CADENCE_{key.upper()}_{actual}_EXPECTED_{expected}")
        if cadence_quality.get("WholeFileMemoryLoadRequired") is not False:
            raise H4BError("CADENCE_WHOLE_FILE_MEMORY_LOAD")

        policy_expected = {
            "PolicyRowCount": EXPECTED_SIGNALS,
            "BoundedHoldSignalCount": 0,
            "NoHoldSignalCount": EXPECTED_SIGNALS,
            "UnapprovedPolicyCount": 0,
            "BlankCompiledMaxAgeCount": 0,
        }
        for key, expected in policy_expected.items():
            actual = int(policy_quality.get(key, -1))
            if actual != expected:
                raise H4BError(f"POLICY_{key.upper()}_{actual}_EXPECTED_{expected}")

        h4b_expected = {
            "InputFrameCount": EXPECTED_FRAMES,
            "OutputFrameCount": EXPECTED_FRAMES,
            "SignalCount": EXPECTED_SIGNALS,
            "CarriedCellCount": 0,
            "StaleCellCount": 0,
            "HoldBarrierCellCount": 0,
            "SyntheticFrameCount": 0,
            "SyntheticZeroFillCount": 0,
            "UnboundedHoldCount": 0,
            "InputMutationCount": 0,
        }
        for key, expected in h4b_expected.items():
            actual = int(h4b_quality.get(key, -1))
            if actual != expected:
                raise H4BError(f"H4B_{key.upper()}_{actual}_EXPECTED_{expected}")
        if h4b_quality.get("WholeFileMemoryLoadRequired") is not False:
            raise H4BError("H4B_WHOLE_FILE_MEMORY_LOAD")
        if approved_count != EXPECTED_SIGNALS:
            raise H4BError("APPROVED_POLICY_COUNT")

        result = {
            "Schema": "U001_H4B_DIRECT_RESULT_V1_0",
            "GeneratedUtc": utc_now(),
            "Frames": EXPECTED_FRAMES,
            "Signals": EXPECTED_SIGNALS,
            "MissingFrames": EXPECTED_MISSING_FRAMES,
            "MaximumFrameStep": EXPECTED_MAXIMUM_STEP,
            "HoldPolicy": "NONE",
            "CarriedCells": 0,
            "StaleCells": 0,
            "Status": "H4B_ACCEPTED",
            "CompletionCode": 0,
        }
        write_json(output / "u001_h4b_direct_result.json", result)
        capsule = (
            f"H4B-F{EXPECTED_FRAMES}-S{EXPECTED_SIGNALS}-G{EXPECTED_MISSING_FRAMES}"
            f"-L{EXPECTED_MAXIMUM_STEP}-N{EXPECTED_SIGNALS}-C0-T0-B0-X0-C0"
        )
        (output / "u001_h4b_direct_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        write_json(
            output / "u001_h4b_direct_diagnostic.json",
            {
                "GeneratedUtc": utc_now(),
                "ExceptionType": type(exc).__name__,
                "Detail": str(exc)[:8000],
                "Recovery": "Return H4BE only; do not edit the core programs.",
            },
        )
        print(f"H4BE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="Direct U001 H4B acceptance")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    raise SystemExit(main())
