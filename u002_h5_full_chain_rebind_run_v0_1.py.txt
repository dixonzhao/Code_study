class ControlledError(RuntimeError):
    """A fail-closed error with a bounded reason and detail."""

    def __init__(self, reason: object, detail: object = "NA") -> None:
        self.reason = token(reason, 72)
        self.detail = compact(detail, 170)
        super().__init__(self.reason)


def token(value: object, limit: int = 96) -> str:
    """Return a compact upper-case token suitable for manual transfer."""

    text = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (text or "NA")[:limit]


def compact(value: object, limit: int = 180) -> str:
    """Keep hyphens as capsule separators and bound one detail field."""

    text = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper()).strip("_.-")
    return (text or "NA")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize deterministic JSON."""

    return (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash a file without loading it into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    return path.is_file() and not path.is_symlink()


def load_json(path: Path, label: str) -> dict[str, Any]:
    """Load one required JSON object."""

    if not regular_file(path):
        raise ControlledError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise ControlledError(label + "_JSON", path.name) from error
    if not isinstance(value, dict):
        raise ControlledError(label + "_TYPE", path.name)
    return value


def write_json(path: Path, value: Mapping[str, Any]) -> None:
    """Write deterministic JSON atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(dict(value)))
    os.replace(temporary, path)


def write_bytes_identical_or_create(path: Path, payload: bytes) -> str:
    """Create one governed file or accept only a byte-identical replay."""

    if path.exists():
        if not regular_file(path) or path.read_bytes() != payload:
            raise ControlledError("PROMOTION_CONFLICT", path.name)
        return "IDENTICAL"
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(payload)
    os.replace(temporary, path)
    return "CREATED"


def require_workspace(value: str) -> Path:
    root = Path(value).resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise ControlledError("WORKSPACE_ROOT", root.name)
    return root


def safe_relative(value: object, label: str) -> Path:
    """Resolve a portable workspace-relative path without traversal."""

    text = str(value)
    if "\\" in text:
        raise ControlledError(label + "_SEPARATOR", text)
    pure = PurePosixPath(text)
    if pure.is_absolute() or not pure.parts or any(part in {"", ".", ".."} for part in pure.parts):
        raise ControlledError(label + "_PATH", text)
    return Path(*pure.parts)


def inside(path: Path, root: Path) -> bool:
    try:
        path.resolve().relative_to(root.resolve())
        return True
    except ValueError:
        return False


def archive_temp(root: Path) -> tuple[int, int]:
    """Archive previous TEMP material while retaining this delivery."""

    temp = root / "04_TEMP"
    temp.mkdir(parents=True, exist_ok=True)
    keep = {"A-Archive", PROGRAM_NAME, GUIDE_NAME}
    stale = [item for item in temp.iterdir() if item.name not in keep]
    if not stale:
        return 0, 0
    stamp = dt.datetime.now().strftime("%Y%m%d_%H%M%S_%f")
    destination = temp / "A-Archive" / ("H5_CHAIN_PREPARE_" + stamp)
    destination.mkdir(parents=True, exist_ok=False)
    moved = 0
    locked = 0
    for source in sorted(stale):
        try:
            shutil.move(str(source), str(destination / source.name))
            moved += 1
        except PermissionError:
            locked += 1
    if moved == 0:
        try:
            destination.rmdir()
        except OSError:
            pass
    return moved, locked


def traceback_location(error: BaseException) -> tuple[str, int, str]:
    frames = traceback.extract_tb(error.__traceback__)
    if not frames:
        return "NA", 0, "NA"
    frame = frames[-1]
    return Path(frame.filename).name, int(frame.lineno), frame.name


def module_items(plan: object) -> list[dict[str, Any]]:
    if not isinstance(plan, list):
        raise ControlledError("MODULE_PLAN_TYPE")
    output: list[dict[str, Any]] = []
    for item in plan:
        if not isinstance(item, dict):
            raise ControlledError("MODULE_PLAN_ITEM")
        output.append(copy.deepcopy(item))
    return output


def module_by_id(plan: object, module_id: str) -> dict[str, Any]:
    if not isinstance(plan, list):
        raise ControlledError("MODULE_PLAN_TYPE")
    selected = [
        item
        for item in plan
        if isinstance(item, dict) and item.get("ModuleId") == module_id
    ]
    if len(selected) != 1:
        raise ControlledError("MODULE_COUNT", f"{module_id}-{len(selected)}")
    return selected[0]


def latest_profile_failure_run(root: Path) -> Path:
    """Find the newest failed Run whose evidence contains PROFILE_INVENTORY_HASH."""

    candidates: list[tuple[float, Path]] = []
    output = root / "07_OUTPUT"
    if not output.is_dir():
        raise ControlledError("OUTPUT_ROOT_MISSING")
    for status_path in output.glob("*/run_status.json"):
        try:
            status = load_json(status_path, "RUN_STATUS")
        except ControlledError:
            continue
        if status.get("Status") != "FAILED":
            continue
        run_root = status_path.parent
        found = False
        for evidence in run_root.rglob("*"):
            if not regular_file(evidence) or evidence.stat().st_size > 2_000_000:
                continue
            try:
                if b"PROFILE_INVENTORY_HASH" in evidence.read_bytes():
                    found = True
                    break
            except OSError:
                continue
        if found:
            candidates.append((status_path.stat().st_mtime, run_root))
    if not candidates:
        raise ControlledError("SOURCE_RUN_AUTO_DISCOVERY", "NO_PROFILE_INVENTORY_HASH_FAILURE")
    return max(candidates, key=lambda item: item[0])[1]


def resolve_source_run(root: Path, requested: Optional[str]) -> Path:
    if requested and requested.upper() != "AUTO":
        run_root = root / "07_OUTPUT" / requested
        if not run_root.is_dir() or run_root.is_symlink():
            raise ControlledError("SOURCE_RUN_MISSING", requested)
        return run_root
    return latest_profile_failure_run(root)

def locate_s0_artifact(run_root: Path) -> tuple[Path, Path, dict[str, Any]]:
    """Locate the committed S0 inventory inside a partially successful Run."""

    matches: list[tuple[Path, Path, dict[str, Any]]] = []
    for manifest_path in (run_root / "03_ARTIFACTS" / "S0").rglob("artifact_manifest.json"):
        manifest = load_json(manifest_path, "S0_ARTIFACT_MANIFEST")
        if manifest.get("ArtifactType") != "U001_INTAKE_ACCEPTED":
            continue
        for payload in manifest.get("Payloads", []):
            if not isinstance(payload, dict) or payload.get("RelativePath") != "u001_signal_inventory.csv":
                continue
            inventory_path = manifest_path.parent / "u001_signal_inventory.csv"
            if not regular_file(inventory_path):
                raise ControlledError("S0_INVENTORY_MISSING")
            if sha256_file(inventory_path) != payload.get("Sha256"):
                raise ControlledError("S0_INVENTORY_PAYLOAD_HASH")
            matches.append((manifest_path, inventory_path, manifest))
    if len(matches) != 1:
        raise ControlledError("S0_ARTIFACT_COUNT", len(matches))
    return matches[0]


def read_inventory(path: Path) -> list[dict[str, str]]:
    """Load only the compact S0 catalog; no U001 data table is loaded."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {
            "CanonicalName",
            "CAN_ID",
            "Channel",
            "MessageName",
            "SignalName",
            "Unit",
            "ValueKind",
            "UpdateMode",
            "StorageDtype",
            "WithinFramePolicy",
            "HoldPolicy",
            "MaxAgeFrames",
            "RequiredForAnalysis",
            "IncludeInFrameView",
            "IncludeInReviewView",
        }
        missing = sorted(required - set(reader.fieldnames or []))
        if missing:
            raise ControlledError("INVENTORY_COLUMN", missing[0])
        rows = [{key: str(value or "") for key, value in row.items()} for row in reader]
    if not rows:
        raise ControlledError("INVENTORY_EMPTY")
    names = [row["CanonicalName"] for row in rows]
    if any(not name for name in names) or len(names) != len(set(names)):
        raise ControlledError("INVENTORY_CANONICAL_UNIQUE")
    return rows


def load_source_profile(root: Path, module: Mapping[str, Any]) -> tuple[Path, dict[str, Any]]:
    configuration = module.get("Configuration")
    if not isinstance(configuration, dict):
        raise ControlledError("SOURCE_MODULE_CONFIGURATION", module.get("ModuleId"))
    relative = safe_relative(configuration.get("BindingProfilePath"), "SOURCE_PROFILE")
    path = (root / relative).resolve()
    if not regular_file(path) or not inside(path, root):
        raise ControlledError("SOURCE_PROFILE_MISSING", relative.as_posix())
    expected_hash = configuration.get("BindingProfileSha256")
    if sha256_file(path) != expected_hash:
        raise ControlledError("SOURCE_PROFILE_HASH", module.get("ModuleId"))
    profile = load_json(path, "SOURCE_PROFILE")
    if profile.get("ContractVersion") != "U002_SIGNAL_BINDING_PROFILE_V0_1":
        raise ControlledError("SOURCE_PROFILE_CONTRACT")
    return path, profile


def one_old_binding(profile: Mapping[str, Any], role: str) -> dict[str, Any]:
    bindings = profile.get("Bindings")
    if not isinstance(bindings, dict):
        raise ControlledError("SOURCE_BINDINGS")
    selected = bindings.get(role)
    if not isinstance(selected, list) or len(selected) != 1 or not isinstance(selected[0], dict):
        raise ControlledError("SOURCE_ROLE_COUNT", role)
    return dict(selected[0])


def normalized_can_id(value: object) -> str:
    text = str(value or "").strip().lower()
    if not text:
        return ""
    try:
        return f"0x{int(text, 0):x}"
    except ValueError:
        return text


def matching_inventory_row(
    rows: Sequence[Mapping[str, str]],
    old: Mapping[str, Any],
    role: str,
) -> dict[str, str]:
    """Map one governed role to the H5 inventory without keyword guessing."""

    canonical = str(old.get("CanonicalName", "")).strip()
    exact = [row for row in rows if row.get("CanonicalName") == canonical]
    if len(exact) == 1:
        return dict(exact[0])
    old_can = normalized_can_id(old.get("CAN_ID"))
    old_channel = str(old.get("Channel", "")).strip()
    old_signal = str(old.get("SignalName", "")).strip().casefold()
    identity = [
        row
        for row in rows
        if normalized_can_id(row.get("CAN_ID")) == old_can
        and str(row.get("Channel", "")).strip() == old_channel
        and str(row.get("SignalName", "")).strip().casefold() == old_signal
    ]
    if len(identity) != 1:
        raise ControlledError("H5_ROLE_BINDING", f"{role}-{len(identity)}")
    return dict(identity[0])


def build_h5_profile(
    rows: Sequence[Mapping[str, str]],
    inventory_sha256: str,
    source_profiles: Mapping[str, Mapping[str, Any]],
    source_run_id: str,
) -> dict[str, Any]:
    bindings: dict[str, list[dict[str, str]]] = {}
    for role, owner in ROLE_OWNER.items():
        old = one_old_binding(source_profiles[owner], role)
        selected = matching_inventory_row(rows, old, role)
        selected["SignalRole"] = role
        if selected.get("IncludeInFrameView", "").strip().lower() not in {"1", "true"}:
            raise ControlledError("ROLE_NOT_IN_FRAME_VIEW", role)
        try:
            if int(selected.get("MaxAgeFrames", "")) < 0:
                raise ValueError
        except ValueError as error:
            raise ControlledError("ROLE_MAX_AGE", role) from error
        bindings[role] = [selected]
    return {
        "ContractVersion": "U002_SIGNAL_BINDING_PROFILE_V0_1",
        "BindingProfileId": "TC001.H5.CAMERA_RADAR.RADAR_LOSS.GOLDEN",
        "BindingProfileVersion": "0.5.0",
        "SourceInventorySha256": inventory_sha256,
        "SourceRunId": source_run_id,
        "DuplicateRoleCount": 0,
        "BlockingIssues": [],
        "Prohibitions": [
            "RADAR_NOT_USED_FOR_CAMERA_SELECTION",
            "WIDTH_IS_SAME_FRAME_EVIDENCE_ONLY",
            "WIDTH_MUST_NOT_SELECT_CAMERA_TARGET",
            "WIDTH_MUST_NOT_BE_HELD_ACROSS_FRAMES",
            "NO_ROLE_GUESSING_AFTER_PROFILE_FREEZE",
        ],
        "Bindings": bindings,
    }


def expected_manifest(root: Path, item: Mapping[str, Any]) -> tuple[Path, dict[str, Any], Path]:
    stage = str(item.get("Stage"))
    module_id = str(item.get("ModuleId"))
    version = str(item.get("ModuleVersion"))
    module_root = root / "01_PROGRAMS" / "modules" / stage / module_id / version
    manifest_path = module_root / "module_manifest.json"
    manifest = load_json(manifest_path, "MODULE_MANIFEST")
    if manifest.get("ModuleId") != module_id or manifest.get("ModuleVersion") != version:
        raise ControlledError("MODULE_MANIFEST_IDENTITY", module_id)
    receipt = (
        root
        / "00_CONTROL"
        / "registry"
        / "qualifications"
        / module_id
        / version
        / "qualification_receipt.json"
    )
    receipt_doc = load_json(receipt, "QUALIFICATION_RECEIPT")
    if receipt_doc.get("QualificationStatus") != "AVAILABLE":
        raise ControlledError("MODULE_NOT_AVAILABLE", module_id)
    if receipt_doc.get("ManifestSha256") != sha256_file(manifest_path):
        raise ControlledError("QUALIFICATION_MANIFEST_HASH", module_id)
    return module_root, manifest, receipt


def audit_topology(root: Path, plan: object) -> list[dict[str, Any]]:
    """Audit the exact selected five-Module graph and qualification evidence."""

    items = module_items(plan)
    if len(items) != len(EXPECTED_PLAN):
        raise ControlledError("PLAN_MODULE_COUNT", len(items))
    for item, expected in zip(items, EXPECTED_PLAN):
        sequence, stage, module_id, version, requires, provides = expected
        observed = (
            item.get("Sequence"),
            item.get("Stage"),
            item.get("ModuleId"),
            item.get("ModuleVersion"),
        )
        if observed != (sequence, stage, module_id, version):
            raise ControlledError("PLAN_TOPOLOGY", f"SEQ{sequence}-{observed}")
        _, manifest, _ = expected_manifest(root, item)
        if manifest.get("RequiresArtifacts") != requires:
            raise ControlledError("REQUIRES_ARTIFACTS", module_id)
        if manifest.get("ProvidesArtifacts") != provides:
            raise ControlledError("PROVIDES_ARTIFACTS", module_id)
        if manifest.get("FailurePolicy") != "REQUIRED":
            raise ControlledError("FAILURE_POLICY", module_id)
    return items


def source_context(root: Path, source_run: Path, requested_target: Optional[str]) -> dict[str, Any]:
    source_status = load_json(source_run / "run_status.json", "SOURCE_STATUS")
    if source_status.get("Status") != "FAILED":
        raise ControlledError("SOURCE_STATUS", source_status.get("Status"))
    source_config = load_json(source_run / "00_CONFIG" / "run_configuration.json", "SOURCE_CONFIG")
    source_id = str(source_config.get("RunId", ""))
    if source_id != source_run.name:
        raise ControlledError("SOURCE_RUN_ID", f"{source_id}-{source_run.name}")
    plan = audit_topology(root, source_config.get("ModulePlan"))
    manifest_path, inventory_path, s0_manifest = locate_s0_artifact(source_run)
    inventory_hash = sha256_file(inventory_path)
    inventory_rows = read_inventory(inventory_path)
    source_profiles: dict[str, Mapping[str, Any]] = {}
    source_profile_paths: dict[str, str] = {}
    for module_id in (CTA_ID, RMV_ID):
        item = module_by_id(plan, module_id)
        profile_path, profile = load_source_profile(root, item)
        source_profiles[module_id] = profile
        source_profile_paths[module_id] = profile_path.relative_to(root).as_posix()
    profile = build_h5_profile(inventory_rows, inventory_hash, source_profiles, source_id)
    target_id = target_run_id(source_id, requested_target)
    target = copy.deepcopy(source_config)
    target["RunId"] = target_id
    target["SignalProfile"] = profile["BindingProfileId"] + "@" + profile["BindingProfileVersion"]
    profile_relative = Path("02_CONFIG/signal_profiles") / PROFILE_NAME
    profile_sha = hashlib.sha256(canonical_bytes(profile)).hexdigest()
    for module_id in (CTA_ID, RMV_ID):
        module = module_by_id(target["ModulePlan"], module_id)
        configuration = module.get("Configuration")
        if not isinstance(configuration, dict):
            raise ControlledError("TARGET_CONFIGURATION", module_id)
        configuration["BindingProfilePath"] = profile_relative.as_posix()
        configuration["BindingProfileSha256"] = profile_sha
    return {
        "SourceRunId": source_id,
        "SourceRunRoot": source_run.relative_to(root).as_posix(),
        "SourceStatus": source_status,
        "SourceConfig": source_config,
        "SourceProfiles": source_profile_paths,
        "S0Manifest": manifest_path.relative_to(root).as_posix(),
        "S0ArtifactId": s0_manifest.get("ArtifactId"),
        "InventoryPath": inventory_path.relative_to(root).as_posix(),
        "InventorySha256": inventory_hash,
        "InventoryRowCount": len(inventory_rows),
        "TargetRunId": target_id,
        "TargetConfig": target,
        "Profile": profile,
        "ProfileRelativePath": profile_relative.as_posix(),
        "ProfileSha256": profile_sha,
        "ModuleCount": len(plan),
    }

def freeze_prepared(root: Path, context: Mapping[str, Any]) -> Path:
    destination = prepared_root(root)
    if destination.exists():
        shutil.rmtree(destination)
    destination.mkdir(parents=True)
    profile_path = destination / PROFILE_NAME
    run_path = destination / (str(context["TargetRunId"]) + ".json")
    saved_path = destination / SAVED_PROFILE_NAME
    profile_path.write_bytes(canonical_bytes(context["Profile"]))
    run_path.write_bytes(canonical_bytes(context["TargetConfig"]))
    saved_path.write_bytes(canonical_bytes(saved_gui_profile(context["TargetConfig"])))
    manifest = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "SourceRunId": context["SourceRunId"],
        "SourceRunRoot": context["SourceRunRoot"],
        "SourceProfiles": context["SourceProfiles"],
        "S0Manifest": context["S0Manifest"],
        "S0ArtifactId": context["S0ArtifactId"],
        "InventoryPath": context["InventoryPath"],
        "InventorySha256": context["InventorySha256"],
        "InventoryRowCount": context["InventoryRowCount"],
        "TargetRunId": context["TargetRunId"],
        "ModuleCount": context["ModuleCount"],
        "PreparedFiles": {
            PROFILE_NAME: sha256_file(profile_path),
            run_path.name: sha256_file(run_path),
            SAVED_PROFILE_NAME: sha256_file(saved_path),
        },
    }
    write_json(destination / PREPARED_MANIFEST, manifest)
    return destination


def load_prepared(root: Path) -> dict[str, Any]:
    destination = prepared_root(root)
    manifest = load_json(destination / PREPARED_MANIFEST, "PREPARED_MANIFEST")
    if manifest.get("ProgramId") != PROGRAM_ID or manifest.get("ProgramVersion") != PROGRAM_VERSION:
        raise ControlledError("PREPARED_PROGRAM_IDENTITY")
    prepared_files = manifest.get("PreparedFiles")
    if not isinstance(prepared_files, dict) or len(prepared_files) != 3:
        raise ControlledError("PREPARED_FILESET")
    for name, expected_hash in prepared_files.items():
        path = destination / str(name)
        if not regular_file(path) or sha256_file(path) != expected_hash:
            raise ControlledError("PREPARED_FILE_HASH", name)
    source_run = root / str(manifest["SourceRunRoot"])
    _, inventory_path, _ = locate_s0_artifact(source_run)
    if sha256_file(inventory_path) != manifest.get("InventorySha256"):
        raise ControlledError("PREPARED_SOURCE_INVENTORY_CHANGED")
    manifest["PreparedRoot"] = destination
    manifest["Profile"] = load_json(destination / PROFILE_NAME, "PREPARED_PROFILE")
    manifest["TargetConfig"] = load_json(
        destination / (str(manifest["TargetRunId"]) + ".json"),
        "PREPARED_RUN_CONFIG",
    )
    return manifest


def load_module(path: Path, name: str):
    spec = importlib.util.spec_from_file_location(name, path)
    if spec is None or spec.loader is None:
        raise ControlledError("MODULE_LOAD_SPEC", path.name)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def validate_run_contract(root: Path, run_config: Mapping[str, Any]) -> None:
    validator_path = root / CONTRACT_VALIDATOR
    validator = load_module(validator_path, "u002_h5_contract_validator")
    try:
        validator.validate_document(root, "RUN", dict(run_config))
    except Exception as error:
        reason = getattr(error, "reason", type(error).__name__)
        location = getattr(error, "path", "NA")
        raise ControlledError("RUN_CONTRACT", f"{reason}-{location}") from error


def run_module_selftests(root: Path, plan: object) -> list[str]:
    capsules: list[str] = []
    for index, item in enumerate(audit_topology(root, plan), start=1):
        module_root, _, _ = expected_manifest(root, item)
        selftest = module_root / "selftest.py"
        if not regular_file(selftest):
            raise ControlledError("MODULE_SELFTEST_MISSING", item.get("ModuleId"))
        result = subprocess.run(
            [
                sys.executable,
                str(selftest),
                "--module-root",
                str(module_root),
                "--workspace-root",
                str(root),
            ],
            text=True,
            capture_output=True,
            timeout=240,
            check=False,
        )
        if result.returncode != 0 or "C-0" not in result.stdout:
            detail = (result.stdout + " " + result.stderr)[-160:]
            raise ControlledError("MODULE_SELFTEST", f"M{index}-{detail}")
        capsules.append(result.stdout.strip().splitlines()[-1])
    return capsules


def validate_module_configurations(root: Path, run_config: Mapping[str, Any]) -> None:
    """Exercise each installed Module's own configuration validator where exposed."""

    for index, item in enumerate(audit_topology(root, run_config.get("ModulePlan")), start=1):
        module_root, _, _ = expected_manifest(root, item)
        implementation = load_module(module_root / "implementation.py", f"u002_h5_impl_{index}")
        configuration = item.get("Configuration")
        if item.get("ModuleId") == RMV_ID:
            required = {"BindingProfilePath", "BindingProfileSha256"}
            if not isinstance(configuration, dict) or set(configuration) != required:
                raise ControlledError("RMV_CONFIGURATION_SHAPE")
            continue
        validator = getattr(implementation, "validate_configuration", None)
        if callable(validator):
            try:
                validator(configuration)
            except Exception as error:
                raise ControlledError("MODULE_CONFIGURATION", f"M{index}-{error}") from error


def audit_profile_against_inventory(root: Path, prepared: Mapping[str, Any]) -> None:
    profile = prepared["Profile"]
    inventory_path = root / str(prepared["InventoryPath"])
    rows = read_inventory(inventory_path)
    if profile.get("SourceInventorySha256") != sha256_file(inventory_path):
        raise ControlledError("PROFILE_INVENTORY_HASH")
    bindings = profile.get("Bindings")
    if not isinstance(bindings, dict) or set(bindings) != set(ROLE_OWNER):
        raise ControlledError("PROFILE_ROLE_SET")
    for role in ROLE_OWNER:
        binding = one_old_binding(profile, role)
        selected = matching_inventory_row(rows, binding, role)
        if selected.get("CanonicalName") != binding.get("CanonicalName"):
            raise ControlledError("PROFILE_ROLE_IDENTITY", role)

def promote(root: Path, prepared: Mapping[str, Any]) -> tuple[Path, int]:
    prepared_root_path = Path(prepared["PreparedRoot"])
    target_id = str(prepared["TargetRunId"])
    files = [
        (
            prepared_root_path / PROFILE_NAME,
            root / "02_CONFIG" / "signal_profiles" / PROFILE_NAME,
        ),
        (
            prepared_root_path / (target_id + ".json"),
            root / "02_CONFIG" / "run_configs" / (target_id + ".json"),
        ),
        (
            prepared_root_path / SAVED_PROFILE_NAME,
            root / "02_CONFIG" / "gui_profiles" / "user" / SAVED_PROFILE_NAME,

        created += int(disposition == "CREATED")
    return files[1][1], created


def bounded_failure_evidence(run_root: Path) -> tuple[str, str, str]:
    """Extract the deepest useful cause without requiring another diagnostic tool."""

    status = load_json(run_root / "run_status.json", "FAILED_RUN_STATUS")
    module_id = "NA"
    stage = "NA"
    reason = status.get("FailureReason") or status.get("Reason") or status.get("Status")
    evidence_files = sorted(
        [path for path in (run_root / "02_DIAGNOSTICS").rglob("*") if regular_file(path)],
        key=lambda path: path.stat().st_mtime,
        reverse=True,
    )
    for path in evidence_files:
        if path.suffix.lower() == ".json":
            try:
                doc = load_json(path, "FAILURE_EVIDENCE")
            except ControlledError:
                continue
            module_id = str(doc.get("ModuleId", module_id))
            stage = str(doc.get("Stage", stage))
            for key in ("FailureReason", "Reason", "Error", "Detail", "Capsule"):
                if doc.get(key):
                    reason = doc[key]
                    break
        else:
            try:
                text = path.read_text(encoding="utf-8-sig", errors="replace").strip()
            except OSError:
                continue
            if text and text.upper() not in {"NONE", "EMPTY"}:
                reason = text.splitlines()[-1]
        if reason and str(reason).upper() not in {"FAILED", "NA"}:
            break
    return token(stage, 8), token(module_id.split(".")[-1], 36), compact(reason, 120)


def artifact_manifests(run_root: Path) -> list[dict[str, Any]]:
    output: list[dict[str, Any]] = []
    for path in sorted((run_root / "03_ARTIFACTS").rglob("artifact_manifest.json")):
        output.append(load_json(path, "RESULT_ARTIFACT"))
    return output


def payload_csv_rows(run_root: Path, artifact_type: str) -> Optional[int]:
    for manifest_path in (run_root / "03_ARTIFACTS").rglob("artifact_manifest.json"):
        manifest = load_json(manifest_path, "RESULT_ARTIFACT")
        if manifest.get("ArtifactType") != artifact_type:
            continue
        for payload in manifest.get("Payloads", []):
            if not isinstance(payload, dict):
                continue
            if payload.get("PayloadKind") != "PRIMARY" or payload.get("MediaType") != "text/csv":
                continue
            path = manifest_path.parent / str(payload.get("RelativePath"))
            if not regular_file(path) or sha256_file(path) != payload.get("Sha256"):
                raise ControlledError("RESULT_PAYLOAD_HASH", artifact_type)
            with path.open("r", encoding="utf-8-sig", newline="") as stream:
                return max(sum(1 for _ in stream) - 1, 0)
    return None


def review_data_row_count(run_root: Path) -> Optional[int]:
    """Read S7's governed aggregate review-row count from Artifact metadata."""

    for manifest_path in (run_root / "03_ARTIFACTS").rglob("artifact_manifest.json"):
        manifest = load_json(manifest_path, "RESULT_ARTIFACT")
        if manifest.get("ArtifactType") != "U003_REVIEW_PACKAGE":
            continue
        metadata = manifest.get("Metadata")
        if not isinstance(metadata, dict):
            raise ControlledError("S7_METADATA_TYPE")
        value = metadata.get("ReviewDataRowCount")
        if isinstance(value, bool) or not isinstance(value, int) or value < 0:
            raise ControlledError("S7_REVIEW_ROW_COUNT", value)
        return value
    return None


def validate_success(root: Path, run_root: Path, target: Mapping[str, Any]) -> tuple[int, int, int]:
    status = load_json(run_root / "run_status.json", "TARGET_STATUS")
    if status.get("Status") != "SUCCEEDED":
        stage, module, reason = bounded_failure_evidence(run_root)
        raise ControlledError("CHAIN_FAILED", f"S{stage}-M{module}-D{reason}")
    resolved = load_json(run_root / "00_CONFIG" / "resolved_run_plan.json", "TARGET_PLAN")
    audit_topology(root, resolved.get("ModulePlan"))
    manifests = artifact_manifests(run_root)
    expected_types = {
        "U001_INTAKE_ACCEPTED",
        "CAMERA_TARGET_RECONSTRUCTION",
        "RADAR_VALIDITY",
        "ANOMALY_CANDIDATES",
        "U003_REVIEW_PACKAGE",
    }
    observed_types = {str(item.get("ArtifactType")) for item in manifests}
    if observed_types != expected_types or len(manifests) != 5:
        raise ControlledError(
            "ARTIFACT_TYPE_SET",
            f"E5-O{len(observed_types)}-M{len(manifests)}",
        )
    candidates = payload_csv_rows(run_root, "ANOMALY_CANDIDATES")
    review_rows = review_data_row_count(run_root)
    if candidates is None or review_rows is None:
        raise ControlledError("BUSINESS_OUTPUT_MISSING")
    return len(manifests), candidates, review_rows


def execute(root: Path, timeout_seconds: int) -> str:
    prepared = load_prepared(root)
    audit_profile_against_inventory(root, prepared)
    validate_run_contract(root, prepared["TargetConfig"])
    config_path, promoted = promote(root, prepared)
    target_id = str(prepared["TargetRunId"])
    run_root = root / "07_OUTPUT" / target_id
    if run_root.exists():
        status = load_json(run_root / "run_status.json", "EXISTING_TARGET_STATUS")
        if status.get("Status") != "SUCCEEDED":
            raise ControlledError("TARGET_RUN_ALREADY_EXISTS", f"{target_id}-{status.get('Status')}")
    else:
        preflight = subprocess.run(
            [
                sys.executable,
                str(root / ENGINE_PREFLIGHT),
                "--prepare-run",
                str(config_path),
                "--workspace-root",
                str(root),
            ],
            text=True,
            capture_output=True,
            timeout=600,
            check=False,
        )
        if preflight.returncode != 0 or "C-0" not in preflight.stdout:
            detail = (preflight.stdout + " " + preflight.stderr)[-170:]
            raise ControlledError("ENGINE_PREFLIGHT", detail)
        result = subprocess.run(
            [
                sys.executable,
                str(root / STAGE_EXECUTOR),
                "--execute-run",
                str(run_root),
                "--workspace-root",
                str(root),
                "--module-timeout-seconds",
                str(timeout_seconds),
            ],
            text=True,
            capture_output=True,
            timeout=max(timeout_seconds * 5, 3600),
            check=False,
        )
        if result.returncode != 0 or "C-0" not in result.stdout:
            if run_root.is_dir():
                stage, module, reason = bounded_failure_evidence(run_root)
                raise ControlledError("CHAIN_FAILED", f"S{stage}-M{module}-D{reason}")
            detail = (result.stdout + " " + result.stderr)[-170:]
            raise ControlledError("EXECUTOR_NO_RUN_EVIDENCE", detail)
    artifacts, candidates, review_rows = validate_success(root, run_root, prepared["TargetConfig"])
    receipt = {
        "Status": "SUCCEEDED",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "SourceRunId": prepared["SourceRunId"],
        "TargetRunId": target_id,
        "ProfileSha256": prepared["PreparedFiles"][PROFILE_NAME],
        "SelectedModuleCount": 5,
        "AcceptedArtifactCount": artifacts,
        "CandidateRows": candidates,
        "ReviewRows": review_rows,
        "AllRemainingModulesAuditedBeforeExecution": True,
        "GoldenPromotionApproved": False,
    }
    diagnostics = run_root / "02_DIAGNOSTICS" / "H5_FULL_CHAIN_CONTROLLER"
    write_json(diagnostics / "h5_full_chain_acceptance.json", receipt)
    capsule = (
        f"H5GR-M5-A{artifacts}-C{candidates}-R{review_rows}"
        f"-P{promoted}-U1-X0-C0"
    )
    (diagnostics / "H5GR_RETURN_CAPSULE.txt").write_text(capsule + "\n", encoding="utf-8")
    print(capsule)
    print(f"OUTPUT-{run_root}")
    return capsule


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    parser.add_argument("--source-run-id", default="AUTO")
    parser.add_argument("--target-run-id")
    parser.add_argument("--module-timeout-seconds", type=int, default=1800)
    return parser.parse_args(argv)


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        root = require_workspace(args.workspace_root)
        if args.prepare:
            prepare(root, args.source_run_id, args.target_run_id)
        elif args.self_test:
            self_test(root)
        else:
            execute(root, args.module_timeout_seconds)
        return 0
    except ControlledError as error:
        print(f"H5E-K{error.reason}-D{error.detail}-X1-C7")
        return 7
    except PermissionError as error:
        filename, line, function = traceback_location(error)
        print(
            f"H5E-KPERMISSION-D{token(filename, 36)}-L{line}-F{token(function, 28)}"
            f"-E{getattr(error, 'errno', 0) or 0}-W{getattr(error, 'winerror', 0) or 0}-X1-C7"
        )
        return 7
    except Exception as error:
        filename, line, function = traceback_location(error)
        print(
            f"H5E-KUNEXPECTED-D{token(type(error).__name__, 32)}"
            f"-F{token(filename, 36)}-L{line}-N{token(function, 28)}-X1-C7"
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
