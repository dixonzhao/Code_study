#!/usr/bin/env python3
"""Read the existing B2D failed Run and retain the exact Module evidence."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import sys
import tempfile
from typing import Dict, List, Mapping, Optional, Sequence, Tuple


PROGRAM_ID = "UDP002-U22B2DR1-EXISTING-MODULE-FAILURE-DIAGNOSTIC"
PROGRAM_VERSION = "0.1"
RUN_ID = "U22B2C_TC001_F41480_41750_V01"
RUN_STATUS = Path("run_status.json")
ARTIFACT_INDEX = Path("run_artifact_index_v0_1.json")
REPORT = Path("02_DIAGNOSTICS/b2dr1_existing_failure_diagnostic_v0_1.json")


class DiagnosticError(RuntimeError):
    """Represent one controlled diagnostic failure."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = stable(reason, 60)
        self.detail = stable(detail, 100)
        super().__init__(self.reason)


def stable(value: object, limit: int = 80) -> str:
    text = re.sub(r"[^A-Z0-9_.:+-]+", "_", str(value).upper())
    return text.strip("_")[:limit] or "NA"


def canonical_bytes(value: object) -> bytes:
    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    return path.is_file() and not path.is_symlink()


def load_json(path: Path, reason: str) -> Dict[str, object]:
    if not regular_file(path):
        raise DiagnosticError(reason + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise DiagnosticError(reason + "_JSON", path.name) from exc
    if not isinstance(value, dict):
        raise DiagnosticError(reason + "_TYPE", path.name)
    return value


def write_json_atomic(path: Path, value: Mapping[str, object]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def require_run(path: Path) -> Tuple[Path, Path]:
    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise DiagnosticError("WORKSPACE_ROOT", root.name)
    run_root = root / "07_OUTPUT" / RUN_ID
    if not run_root.is_dir() or run_root.is_symlink():
        raise DiagnosticError("RUN_ROOT", RUN_ID)
    return root, run_root


def nonempty_lines(path: Path) -> List[str]:
    if not regular_file(path):
        return []
    try:
        text = path.read_text(encoding="utf-8-sig", errors="replace")
    except OSError:
        return []
    return [line.strip() for line in text.splitlines() if line.strip()]


def relative_regular(run_root: Path, relative: object) -> Optional[Path]:
    text = str(relative or "")
    if not text or "\\" in text:
        return None
    path = (run_root / text).resolve()
    try:
        path.relative_to(run_root.resolve())
    except ValueError:
        return None
    return path if regular_file(path) else None


def record_candidates(run_root: Path) -> List[Tuple[Path, Dict[str, object]]]:
    records: List[Tuple[Path, Dict[str, object]]] = []
    for path in sorted((run_root / "01_LOGS").glob("*.json")):
        try:
            value = load_json(path, "EXECUTION_RECORD")
            if value.get("ExecutionContract") == "U002_U22_MODULE_EXECUTION_RECORD":
                records.append((path, value))
        except Exception:
            continue
    return records


def artifact_count(run_root: Path) -> int:
    try:
        index = load_json(run_root / ARTIFACT_INDEX, "ARTIFACT_INDEX")
        artifacts = index.get("Artifacts")
        return len(artifacts) if isinstance(artifacts, list) else 0
    except Exception:
        return 0


def diagnose(root: Path, run_root: Path, emit_summary: bool = True) -> str:
    """Extract the failed sequence, Module capsule, traceback and request identity."""

    status = load_json(run_root / RUN_STATUS, "RUN_STATUS")
    if status.get("Status") != "FAILED":
        raise DiagnosticError("RUN_NOT_FAILED", status.get("Status"))
    records = record_candidates(run_root)
    failed = [(path, item) for path, item in records if item.get("Status") == "FAILED"]
    if len(failed) != 1:
        raise DiagnosticError("FAILED_RECORD_COUNT", len(failed))
    record_path, record = failed[0]
    diagnostic_root = run_root / str(record.get("DiagnosticRoot", ""))
    if not diagnostic_root.is_dir() or diagnostic_root.is_symlink():
        raise DiagnosticError("DIAGNOSTIC_ROOT", record.get("DiagnosticRoot"))

    stdout_path = diagnostic_root / "stdout.txt"
    stderr_path = diagnostic_root / "stderr.txt"
    result_path = diagnostic_root / "module_result.json"
    request_path = diagnostic_root / "module_request.json"
    stdout_lines = nonempty_lines(stdout_path)
    stderr_lines = nonempty_lines(stderr_path)
    stdout_last = stdout_lines[-1] if stdout_lines else "MISSING"
    stderr_last = stderr_lines[-1] if stderr_lines else "MISSING"
    request = load_json(request_path, "MODULE_REQUEST") if regular_file(request_path) else {}
    result = load_json(result_path, "MODULE_RESULT") if regular_file(result_path) else {}

    module_reason = "MISSING"
    reason_match = re.search(r"(?:^|\s)R-([A-Z0-9_.:+-]+)(?:\s|$)", stdout_last.upper())
    if reason_match is not None:
        module_reason = reason_match.group(1)
    elif stderr_last != "MISSING":
        module_reason = stderr_last

    sequence = int(record.get("Sequence", 0))
    exit_code = int(record.get("ExitCode", -1))
    artifacts = artifact_count(run_root)
    request_identity = {
        "Sequence": request.get("Sequence"),
        "Stage": request.get("Stage"),
        "ModuleId": request.get("ModuleId"),
        "ModuleVersion": request.get("ModuleVersion"),
        "ExpectedOutputTypes": request.get("ExpectedOutputTypes"),
        "InputArtifactTypes": [
            item.get("ArtifactType")
            for item in request.get("InputArtifacts", [])
            if isinstance(item, dict)
        ],
    }
    hashes = {
        "RunStatusSha256": sha256_file(run_root / RUN_STATUS),
        "ExecutionRecordSha256": sha256_file(record_path),
        "RequestSha256": sha256_file(request_path) if regular_file(request_path) else None,
        "StdoutSha256": sha256_file(stdout_path) if regular_file(stdout_path) else None,
        "StderrSha256": sha256_file(stderr_path) if regular_file(stderr_path) else None,
        "ResultSha256": sha256_file(result_path) if regular_file(result_path) else None,
    }
    report = {
        "DiagnosticContract": "UDP002_U22B2DR1_EXISTING_FAILURE_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "RunId": RUN_ID,
        "RunStatus": status,
        "ExecutionRecord": record,
        "RequestIdentity": request_identity,
        "StdoutLastLine": stdout_last,
        "StderrLastLine": stderr_last,
        "ModuleResult": result if result else None,
        "DerivedModuleReason": module_reason,
        "ArtifactCount": artifacts,
        "Hashes": hashes,
        "SourceFilesModified": False,
    }
    write_json_atomic(run_root / REPORT, report)
    if emit_summary:
        print(
            f"B2DR1 N-Q{sequence}-E{exit_code}-A{artifacts}"
            f"-O{int(bool(stdout_lines))}-S{int(bool(stderr_lines))}"
            f"-J{int(bool(result))}-X0 C-0"
        )
    return "B2DR1K K-" + stable(module_reason, 100) + "-X0 C-0"


def self_test() -> str:
    checks: List[bool] = []

    def ck(value: bool) -> None:
        if not value:
            raise AssertionError("SELFTEST")
        checks.append(True)

    ck(PROGRAM_ID.startswith("UDP002"))
    ck(PROGRAM_VERSION == "0.1")
    ck(RUN_ID.startswith("U22B2C_"))
    ck(stable("a b/c") == "A_B_C")
    with tempfile.TemporaryDirectory() as temporary:
        root = Path(temporary) / "U-UDP002"
        run_root = root / "07_OUTPUT" / RUN_ID
        diagnostic = run_root / "02_DIAGNOSTICS/modules/0002_Camera"
        logs = run_root / "01_LOGS"
        diagnostic.mkdir(parents=True)
        logs.mkdir(parents=True)
        write_json_atomic(run_root / RUN_STATUS, {"Status": "FAILED"})
        request = {
            "Sequence": 2,
            "Stage": "S1",
            "ModuleId": "Camera",
            "ModuleVersion": "0.1.0",
            "ExpectedOutputTypes": ["CAMERA_TARGET_RECONSTRUCTION"],
            "InputArtifacts": [{"ArtifactType": "U001_INTAKE_ACCEPTED"}],
        }
        write_json_atomic(diagnostic / "module_request.json", request)
        (diagnostic / "stdout.txt").write_text(
            "CAM N-R0-X1 C-7 R-FRAME_RANGE_EMPTY\n", encoding="utf-8"
        )
        (diagnostic / "stderr.txt").write_text("", encoding="utf-8")
        record = {
            "ExecutionContract": "U002_U22_MODULE_EXECUTION_RECORD",
            "Sequence": 2,
            "Status": "FAILED",
            "ExitCode": 7,
            "DiagnosticRoot": diagnostic.relative_to(run_root).as_posix(),
        }
        write_json_atomic(logs / "0002_Camera.json", record)
        write_json_atomic(run_root / ARTIFACT_INDEX, {"Artifacts": [{}]})
        ck(len(record_candidates(run_root)) == 1)
        ck(artifact_count(run_root) == 1)
        ck(nonempty_lines(diagnostic / "stdout.txt")[-1].endswith("FRAME_RANGE_EMPTY"))
        ck(relative_regular(run_root, "run_status.json") == (run_root / RUN_STATUS).resolve())
        capsule = diagnose(root, run_root, emit_summary=False)
        ck(capsule == "B2DR1K K-FRAME_RANGE_EMPTY-X0 C-0")
        ck(regular_file(run_root / REPORT))
        report = load_json(run_root / REPORT, "REPORT")
        ck(report["SourceFilesModified"] is False)
        ck(report["RequestIdentity"]["Sequence"] == 2)
        ck(report["ArtifactCount"] == 1)
        ck(report["DerivedModuleReason"] == "FRAME_RANGE_EMPTY")
        ck(report["Hashes"]["ResultSha256"] is None)
    ck(len(hashlib.sha256(b"x").hexdigest()) == 64)
    ck(REPORT.suffix == ".json")
    ck(ARTIFACT_INDEX.suffix == ".json")
    if len(checks) != 18:
        raise AssertionError("SELFTEST_COUNT")
    return "B2DR1-ST N-T18-X0 C-0"


def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--diagnose", action="store_true")
    parser.add_argument("--workspace-root")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            print(self_test())
            return 0
        if not args.workspace_root:
            raise DiagnosticError("WORKSPACE_ROOT_REQUIRED")
        root, run_root = require_run(Path(args.workspace_root))
        print(diagnose(root, run_root))
        return 0
    except Exception as exc:
        reason = getattr(exc, "reason", stable(exc.__class__.__name__))
        detail = getattr(exc, "detail", stable(exc))
        print(f"B2DR1E K-{reason}-D-{detail}-X1 C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
