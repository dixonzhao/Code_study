class ProfileError(RuntimeError):
    """A deterministic fail-closed profile error."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = machine_token(reason, 64)
        self.detail = machine_token(detail, 160)
        super().__init__(self.reason)


def machine_token(value: object, limit: int = 80) -> str:
    """Return a bounded token suitable for manual transfer."""

    cleaned = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper())
    return cleaned.strip("_")[:limit] or "NA"


def sha256_file(path: Path) -> str:
    """Hash one file in bounded blocks."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def read_csv_exact(path: Path, columns: Sequence[str], label: str) -> List[Dict[str, str]]:
    """Read a CSV whose full header is part of the contract."""

    if not path.is_file() or path.is_symlink():
        raise ProfileError(label + "_MISSING", path.name)
    try:
        with path.open("r", encoding="utf-8-sig", newline="") as stream:
            reader = csv.DictReader(stream)
            header = tuple(reader.fieldnames or ())
            if header != tuple(columns):
                raise ProfileError(label + "_SCHEMA", "|".join(header[:8]))
            return [dict(row) for row in reader]
    except ProfileError:
        raise
    except Exception as exc:
        raise ProfileError(label + "_READ", type(exc).__name__) from exc


def read_csv_required(path: Path, columns: Sequence[str], label: str) -> List[Dict[str, str]]:
    """Read a CSV and require at least the named columns."""

    if not path.is_file() or path.is_symlink():
        raise ProfileError(label + "_MISSING", path.name)
    try:
        with path.open("r", encoding="utf-8-sig", newline="") as stream:
            reader = csv.DictReader(stream)
            header = tuple(reader.fieldnames or ())
            missing = [name for name in columns if name not in header]
            if missing:
                raise ProfileError(label + "_COLUMN", missing[0])
            return [dict(row) for row in reader]
    except ProfileError:
        raise
    except Exception as exc:
        raise ProfileError(label + "_READ", type(exc).__name__) from exc


def read_json(path: Path, label: str) -> Dict[str, object]:
    """Read one JSON object."""

    if not path.is_file() or path.is_symlink():
        raise ProfileError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise ProfileError(label + "_JSON", type(exc).__name__) from exc
    if not isinstance(value, dict):
        raise ProfileError(label + "_TYPE", path.name)
    return value


def write_csv(path: Path, columns: Sequence[str], rows: Iterable[Mapping[str, object]]) -> None:
    """Write deterministic UTF-8 CSV with LF line endings."""

    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(columns), lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({name: row.get(name, "") for name in columns})


def write_json(path: Path, value: Mapping[str, object]) -> None:
    """Write deterministic JSON."""

    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
    )


def parse_int(value: object, label: str) -> int:
    """Parse one contract integer without accepting fractions."""

    text = str(value).strip()
    try:
        number = int(text)
    except ValueError as exc:
        raise ProfileError(label + "_INTEGER", text) from exc
    return number


def parse_zero_one(value: object, label: str) -> int:
    """Parse an exact zero-or-one field."""

    number = parse_int(value, label)
    if number not in (0, 1):
        raise ProfileError(label + "_FLAG", number)
    return number


def ensure_relative_child(root: Path, relative_text: object) -> Path:
    """Resolve a manifest path and reject escape from the declared Case."""

    relative = Path(str(relative_text))
    if relative.is_absolute():
        raise ProfileError("SOURCE_PROFILE_ABSOLUTE_PATH")
    resolved_root = root.resolve()
    resolved = (resolved_root / relative).resolve()
    try:
        resolved.relative_to(resolved_root)
    except ValueError as exc:
        raise ProfileError("SOURCE_PROFILE_PATH_ESCAPE", relative) from exc
    return resolved


def require_b12_success(b12_root: Path) -> Tuple[Dict[str, object], List[Dict[str, str]]]:
    """Validate the complete B12 handoff and return its request rows."""

    if not b12_root.is_dir():
        raise ProfileError("B12_ROOT_MISSING", b12_root.name)
    required_names = {B12_AUDIT, B12_REQUEST, B12_MANIFEST, B12_CAPSULE}
    actual_names = {path.name for path in b12_root.iterdir() if path.is_file()}
    if actual_names != required_names:
        missing = sorted(required_names - actual_names)
        extra = sorted(actual_names - required_names)
        raise ProfileError("B12_FILE_SET", f"M{len(missing)}E{len(extra)}")

    manifest = read_json(b12_root / B12_MANIFEST, "B12_MANIFEST")
    if parse_int(manifest.get("CompletionCode", -1), "B12_COMPLETION") != 0:
        raise ProfileError("B12_NOT_COMPLETE")
    expected_counts = {
        "RequestedIdentities": 22,
        "ExactMatches": 22,
        "FuzzyOnlyMatches": 0,
        "AmbiguousMatches": 0,
        "MissingMatches": 0,
    }
    for field, expected in expected_counts.items():
        if parse_int(manifest.get(field, -1), "B12_" + field) != expected:
            raise ProfileError("B12_COUNT", field)
    if manifest.get("CameraPairCodes") != "222222222":
        raise ProfileError("B12_CAMERA_PAIR_CODES", manifest.get("CameraPairCodes"))
    if manifest.get("RadarCodes") != "111" or manifest.get("FrameKeyCode") != 1:
        raise ProfileError("B12_SUPPORT_CODES")

    outputs = manifest.get("Outputs")
    if not isinstance(outputs, dict):
        raise ProfileError("B12_OUTPUT_RECORDS_TYPE")
    for name in (B12_AUDIT, B12_REQUEST):
        record = outputs.get(name)
        if not isinstance(record, dict):
            raise ProfileError("B12_OUTPUT_RECORD_MISSING", name)
        if record.get("Sha256") != sha256_file(b12_root / name):
            raise ProfileError("B12_OUTPUT_HASH", name)

    capsule_lines = (b12_root / B12_CAPSULE).read_text(
        encoding="utf-8-sig", errors="replace"
    ).splitlines()
    if len(capsule_lines) != 2 or not all(line.endswith("C-0") for line in capsule_lines):
        raise ProfileError("B12_CAPSULE_NOT_ACCEPTED")

    request_rows = read_csv_exact(b12_root / B12_REQUEST, B12_REQUEST_COLUMNS, "B12_REQUEST")
    if len(request_rows) != 22:
        raise ProfileError("B12_REQUEST_ROWS", len(request_rows))
    canonical_names = [row["CanonicalName"].strip() for row in request_rows]
    if any(not name for name in canonical_names) or len(set(canonical_names)) != 22:
        raise ProfileError("B12_REQUEST_CANONICAL_IDENTITY")
    if any(row["RequestedProfileId"] != PROFILE_ID for row in request_rows):
        raise ProfileError("B12_REQUEST_PROFILE_ID")
    if any(row["PolicyStatus"] != "REQUEST_ONLY_NOT_APPROVED" for row in request_rows):
        raise ProfileError("B12_REQUEST_STATUS")
    return manifest, request_rows


def load_source_inventory(
    u001_case_root: Path,
    b12_manifest: Mapping[str, object],
) -> Dict[str, Dict[str, str]]:
    """Load the exact U001 discovery inventory referenced by B12."""

    if not u001_case_root.is_dir():
        raise ProfileError("U001_CASE_ROOT_MISSING", u001_case_root.name)
    source_root = ensure_relative_child(
        u001_case_root,
        b12_manifest.get("SourceProfileRelativeRoot", ""),
    )
    inventory_path = source_root / SOURCE_INVENTORY
    batch_path = source_root / SOURCE_BATCH_MAP
    if sha256_file(inventory_path) != b12_manifest.get("SourceInventorySha256"):
        raise ProfileError("SOURCE_INVENTORY_HASH")
    if sha256_file(batch_path) != b12_manifest.get("SourceBatchMapSha256"):
        raise ProfileError("SOURCE_BATCH_MAP_HASH")
    rows = read_csv_required(inventory_path, SOURCE_INVENTORY_COLUMNS, "SOURCE_INVENTORY")
    selected: Dict[str, Dict[str, str]] = {}
    for row in rows:
        if str(row["Selected"]).strip() != "1":
            continue
        canonical = row["CanonicalName"].strip()
        if not canonical:
            raise ProfileError("SOURCE_CANONICAL_BLANK")
        if canonical in selected:
            raise ProfileError("SOURCE_CANONICAL_DUPLICATE", canonical)
        selected[canonical] = dict(row)
    return selected


def profile_order(row: Mapping[str, str]) -> Tuple[int, int, int]:
    """Order Frame key, Radar support, then paired Camera features."""

    group = row["Group"]
    pair_code = row["PairCode"]
    if group == "FRAME_DOMAIN":
        return (0, 0, 0)
    if group == "RADAR_SUPPORT":
        return (1, {"D": 0, "F": 1, "R": 2}.get(pair_code, 99), 0)
    feature_order = {code: index for index, code in enumerate("DLROTSMYW")}
    source_order = {"CAMERA_1": 0, "CAMERA_3": 1}
    return (
        2,
        feature_order.get(pair_code, 99),
        source_order.get(row["SourceRole"], 99),
    )


def semantics(row: Mapping[str, str]) -> Dict[str, str]:
    """Return the explicitly approved semantics for one profile identity."""

    group = row["Group"]
    code = row["PairCode"]
    role = row["SourceRole"]
    result = {
        "ValueKind": "FLOAT",
        "SignalRole": "MEASUREMENT",
        "UpdateMode": "PERIODIC",
        "StorageDtype": "Float64",
        "ValidityRule": "NONE",
        "InvalidExactValues": "",
        "InvalidReasonCode": "",
        "IncludeInFrameView": "1",
        "WithinFramePolicy": "LAST_UPDATE",
    }

    if group == "FRAME_DOMAIN":
        result.update(
            {
                "ValueKind": "INTEGER",
                "SignalRole": "FRAME_KEY",
                "StorageDtype": "Int64",
                "IncludeInFrameView": "0",
                "WithinFramePolicy": "KEEP_STREAM_ONLY",
            }
        )
    elif group == "CAMERA_PAIR" and code in {"O", "T"}:
        result.update(
            {
                "ValueKind": "ENUM",
                "SignalRole": "STATE",
                "StorageDtype": "Int64",
            }
        )
    elif group == "CAMERA_PAIR" and code == "S":
        result.update(
            {
                "ValueKind": "BOOLEAN",
                "SignalRole": "EVENT_FLAG",
                "StorageDtype": "boolean",
                "WithinFramePolicy": "ANY_TRUE",
            }
        )
    elif group == "CAMERA_PAIR" and code == "M":
        result.update(
            {
                "ValueKind": "INTEGER",
                "SignalRole": "COUNTER",
                "StorageDtype": "Int64",
            }
        )
    elif group == "RADAR_SUPPORT" and code == "F":
        result.update(
            {
                "ValueKind": "BOOLEAN",
                "SignalRole": "VALIDITY_FLAG",
                "StorageDtype": "boolean",
            }
        )

    if group == "CAMERA_PAIR" and code == "D":
        result.update(
            {
                "ValidityRule": "EXACT_VALUES",
                "InvalidExactValues": "327.66",
                "InvalidReasonCode": "USER_INVALID_RULE",
            }
        )
    elif group == "RADAR_SUPPORT" and code == "D":
        result.update(
            {
                "ValidityRule": "EXACT_VALUES",
                "InvalidExactValues": "0;327.66",
                "InvalidReasonCode": "USER_INVALID_RULE",
            }
        )

    if role == "FRAME_KEY" and code != "F":
        raise ProfileError("FRAME_SEMANTIC_IDENTITY")
    return result


def validate_request_against_source(
    request_rows: Sequence[Mapping[str, str]],
    source_inventory: Mapping[str, Mapping[str, str]],
) -> List[Dict[str, str]]:
    """Rebind every B12 identity to the exact accepted source inventory row."""

    validated: List[Dict[str, str]] = []
    for request in request_rows:
        canonical = request["CanonicalName"].strip()
        source = source_inventory.get(canonical)
        if source is None:
            raise ProfileError("REQUEST_NOT_IN_SOURCE", canonical)
        for field in ("Channel", "CAN_ID_Hex", "MessageName", "SignalName", "Unit"):
            if request[field].strip() != source[field].strip():
                raise ProfileError("REQUEST_SOURCE_MISMATCH", f"{canonical}:{field}")
        if source["SourceGroupComplete"].strip().lower() not in {"1", "true"}:
            raise ProfileError("SOURCE_GROUP_INCOMPLETE", canonical)
        if source["ReuseStatus"].strip().upper() != "MATCHED":
            raise ProfileError("SOURCE_REUSE_NOT_MATCHED", canonical)
        row = dict(request)
        row["Required"] = str(parse_zero_one(request["Required"], "REQUEST_REQUIRED"))
        validated.append(row)
    if len(validated) != 22:
        raise ProfileError("VALIDATED_REQUEST_COUNT", len(validated))
    return sorted(validated, key=profile_order)


def build_control_rows(
    rows: Sequence[Mapping[str, str]],
    source_inventory: Mapping[str, Mapping[str, str]],
) -> Tuple[List[Dict[str, object]], List[Dict[str, object]], List[Dict[str, object]], List[Dict[str, object]]]:
    """Materialize the four U001 control tables."""

    g2: List[Dict[str, object]] = []
    g4: List[Dict[str, object]] = []
    g5: List[Dict[str, object]] = []
    g6: List[Dict[str, object]] = []
    signal_index = 0
    for row in rows:
        canonical = row["CanonicalName"]
        source = source_inventory[canonical]
        sem = semantics(row)
        required = row["Required"]
        note = (
            f"{PROFILE_ID};ENGINEERING_APPROVED_SEMANTICS;"
            f"{row['Group']}:{row['SourceRole']}:{row['PairCode']}"
        )
        g2.append(
            {
                "Selected": 1,
                "Channel": source["Channel"],
                "CAN_ID_Hex": source["CAN_ID_Hex"],
                "MessageName": source["MessageName"],
                "SignalName": source["SignalName"],
                "CanonicalName": canonical,
                "Required": required,
                "SourceGroupComplete": 1,
                "ReuseStatus": "MATCHED",
            }
        )
        g4.append(
            {
                "CanonicalName": canonical,
                "Channel": source["Channel"],
                "CAN_ID_Hex": source["CAN_ID_Hex"],
                "MessageName": source["MessageName"],
                "SignalName": source["SignalName"],
                "Unit": source["Unit"],
                "ValueKind": sem["ValueKind"],
                "SignalRole": sem["SignalRole"],
                "UpdateMode": sem["UpdateMode"],
                "StorageDtype": sem["StorageDtype"],
                "ValidityRule": sem["ValidityRule"],
                "InvalidExactValues": sem["InvalidExactValues"],
                "DBCMinimum": "",
                "DBCMaximum": "",
                "UserMinimum": "",
                "UserMaximum": "",
                "InvalidReasonCode": sem["InvalidReasonCode"],
                "ValiditySourceSignal": "",
                "ValidityExpression": "",
                "Required": required,
                "Notes": note + ";NO_CROSS_SIGNAL_VALIDITY",
            }
        )
        g5.append(
            {
                "CanonicalName": canonical,
                "ValueKind": sem["ValueKind"],
                "SignalRole": sem["SignalRole"],
                "IncludeInFrameView": sem["IncludeInFrameView"],
                "WithinFramePolicy": sem["WithinFramePolicy"],
                "Required": required,
                "Notes": note,
            }
        )
        if sem["IncludeInFrameView"] == "1":
            signal_index += 1
            g6.append(
                {
                    "SignalIndex": f"S{signal_index}",
                    "CanonicalName": canonical,
                    "ValueKind": sem["ValueKind"],
                    "SignalRole": sem["SignalRole"],
                    "UpdateMode": sem["UpdateMode"],
                    "HoldPolicy": "NONE",
                    "MaxAgeFrames": "",
                    "Required": required,
                    "PolicyBasis": "B13_EXPLICIT_NO_CROSS_FRAME_HOLD",
                    "Notes": note + ";U002_OWNS_RECONSTRUCTION",
                }
            )
    return g2, g4, g5, g6


def validate_control_rows(
    g2: Sequence[Mapping[str, str]],
    g4: Sequence[Mapping[str, str]],
    g5: Sequence[Mapping[str, str]],
    g6: Sequence[Mapping[str, str]],
) -> Dict[str, int]:
    """Apply independent cross-table and semantic invariants."""

    if (len(g2), len(g4), len(g5), len(g6)) != (22, 22, 22, 21):
        raise ProfileError("CONTROL_ROW_COUNTS", f"{len(g2)}.{len(g4)}.{len(g5)}.{len(g6)}")
    names2 = [row["CanonicalName"] for row in g2]
    names4 = [row["CanonicalName"] for row in g4]
    names5 = [row["CanonicalName"] for row in g5]
    names6 = [row["CanonicalName"] for row in g6]
    if len(set(names2)) != 22 or names2 != names4 or names2 != names5:
        raise ProfileError("CONTROL_MEMBERSHIP_ORDER")
    frame = [row for row in g4 if row["SignalRole"] == "FRAME_KEY"]
    if len(frame) != 1:
        raise ProfileError("FRAME_KEY_COUNT", len(frame))
    if names6 != [name for name in names5 if name != frame[0]["CanonicalName"]]:
        raise ProfileError("G6_MEMBERSHIP_ORDER")
    if [row["SignalIndex"] for row in g6] != [f"S{i}" for i in range(1, 22)]:
        raise ProfileError("G6_SIGNAL_INDEX")
    if any(row["HoldPolicy"] != "NONE" or row["MaxAgeFrames"] for row in g6):
        raise ProfileError("G6_HOLD_POLICY")
    if any(row["Selected"] != "1" for row in g2):
        raise ProfileError("G2_SELECTED")
    required_count = sum(row["Required"] == "1" for row in g2)
    if required_count != 9:
        raise ProfileError("REQUIRED_COUNT", required_count)

    exact_rows = [row for row in g4 if row["ValidityRule"] == "EXACT_VALUES"]
    if len(exact_rows) != 3:
        raise ProfileError("EXACT_VALIDITY_COUNT", len(exact_rows))
    for row in exact_rows:
        identity = (row["Channel"], row["CAN_ID_Hex"], row["SignalName"])
        if identity in {
            ("2", "0x221", "Distance"),
            ("2", "0x251", "Distance"),
        }:
            if row["InvalidExactValues"] != "327.66":
                raise ProfileError("CAMERA_SENTINEL_POLICY", row["CanonicalName"])
        elif identity == ("1", "0x380", "DistanceToObject"):
            if row["InvalidExactValues"] != "0;327.66":
                raise ProfileError("RADAR_SENTINEL_POLICY", row["CanonicalName"])
        else:
            raise ProfileError("UNAPPROVED_EXACT_VALIDITY", row["CanonicalName"])
    if any(
        row["ValidityRule"] == "NONE"
        and any(
            row[field]
            for field in (
                "InvalidExactValues",
                "DBCMinimum",
                "DBCMaximum",
                "UserMinimum",
                "UserMaximum",
                "InvalidReasonCode",
                "ValiditySourceSignal",
                "ValidityExpression",
            )
        )
        for row in g4
    ):
        raise ProfileError("NONE_VALIDITY_PARAMETERS")
    if any(
        row["SignalRole"] == "FRAME_KEY"
        and (row["IncludeInFrameView"] != "0" or row["WithinFramePolicy"] != "KEEP_STREAM_ONLY")
        for row in g5
    ):
        raise ProfileError("FRAME_VIEW_POLICY")
    return {
        "SelectedSignals": 22,
        "RequiredSignals": required_count,
        "FrameViewSignals": 21,
        "ExactValidityRules": len(exact_rows),
        "NoHoldSignals": 21,
    }


def validate_output(output_root: Path) -> Tuple[str, str]:
    """Independently validate one completed candidate profile folder."""

    if not output_root.is_dir():
        raise ProfileError("OUTPUT_ROOT_MISSING", output_root.name)
    actual = {path.name for path in output_root.iterdir() if path.is_file()}
    if actual != set(OUTPUT_NAMES):
        raise ProfileError("OUTPUT_FILE_SET", f"N{len(actual)}")
    g2 = read_csv_exact(output_root / G2_NAME, G2_COLUMNS, "G2")
    g4 = read_csv_exact(output_root / G4_NAME, G4_COLUMNS, "G4")
    g5 = read_csv_exact(output_root / G5_NAME, G5_COLUMNS, "G5")
    g6 = read_csv_exact(output_root / G6_NAME, G6_COLUMNS, "G6")
    counts = validate_control_rows(g2, g4, g5, g6)
    manifest = read_json(output_root / PROFILE_MANIFEST, "PROFILE_MANIFEST")
    if manifest.get("CompletionCode") != 0:
        raise ProfileError("PROFILE_MANIFEST_COMPLETION")
    if manifest.get("ProfileId") != PROFILE_ID or manifest.get("ApprovalStatus") != "CANDIDATE_NOT_ACTIVATED":
        raise ProfileError("PROFILE_MANIFEST_IDENTITY")
    if manifest.get("ActivationAuthorized") is not False:
        raise ProfileError("PROFILE_ACTIVATION_STATE")
    outputs = manifest.get("ControlFiles")
    if not isinstance(outputs, dict) or set(outputs) != set(CONTROL_NAMES):
        raise ProfileError("PROFILE_CONTROL_RECORDS")
    for name in CONTROL_NAMES:
        record = outputs[name]
        if not isinstance(record, dict) or record.get("Sha256") != sha256_file(output_root / name):
            raise ProfileError("PROFILE_CONTROL_HASH", name)
    if any(str(output_root.resolve()) in value for value in json.dumps(manifest).splitlines()):
        raise ProfileError("MANIFEST_ABSOLUTE_PATH")
    capsule = (output_root / CAPSULE_NAME).read_text(encoding="utf-8-sig").splitlines()
    line1 = (
        f"B13 N-S{counts['SelectedSignals']}-R{counts['RequiredSignals']}"
        f"-F{counts['FrameViewSignals']}-E{counts['ExactValidityRules']}"
        f"-N{counts['NoHoldSignals']}-P4-H0-X0 C-0"
    )
    line2 = "B13I C32766-R0.32766-HNONE-A0 C-0"
    if capsule != [line1, line2]:
        raise ProfileError("PROFILE_CAPSULE")
    return line1, line2


def build_profile(b12_root: Path, u001_case_root: Path, output_root: Path) -> Tuple[str, str]:
    """Build a new immutable candidate folder and self-validate it."""

    if output_root.exists():
        raise ProfileError("OUTPUT_ALREADY_EXISTS", output_root.name)
    b12_manifest, request_rows = require_b12_success(b12_root)
    source_inventory = load_source_inventory(u001_case_root, b12_manifest)
    validated = validate_request_against_source(request_rows, source_inventory)
    g2, g4, g5, g6 = build_control_rows(validated, source_inventory)

    parent = output_root.parent
    parent.mkdir(parents=True, exist_ok=True)
    temporary = parent / ("." + output_root.name + ".partial")
    if temporary.exists():
        raise ProfileError("PARTIAL_OUTPUT_EXISTS", temporary.name)
    temporary.mkdir()
    try:
        write_csv(temporary / G2_NAME, G2_COLUMNS, g2)
        write_csv(temporary / G4_NAME, G4_COLUMNS, g4)
        write_csv(temporary / G5_NAME, G5_COLUMNS, g5)
        write_csv(temporary / G6_NAME, G6_COLUMNS, g6)
        normalized = {
            name: read_csv_exact(
                temporary / name,
                {G2_NAME: G2_COLUMNS, G4_NAME: G4_COLUMNS, G5_NAME: G5_COLUMNS, G6_NAME: G6_COLUMNS}[name],
                name,
            )
            for name in CONTROL_NAMES
        }
        counts = validate_control_rows(
            normalized[G2_NAME], normalized[G4_NAME], normalized[G5_NAME], normalized[G6_NAME]
        )
        control_records: Dict[str, object] = {}
        for name in CONTROL_NAMES:
            path = temporary / name
            control_records[name] = {
                "Sha256": sha256_file(path),
                "Bytes": path.stat().st_size,
                "Rows": len(normalized[name]),
            }
        manifest: Dict[str, object] = {
            "CompletionCode": 0,
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "ContractVersion": CONTRACT_VERSION,
            "ProfileId": PROFILE_ID,
            "ProfileVersion": PROFILE_VERSION,
            "ProcessingMode": "CONTROLLED_PROFILE",
            "ApprovalStatus": "CANDIDATE_NOT_ACTIVATED",
            "ActivationAuthorized": False,
            "SourceB12ManifestSha256": sha256_file(b12_root / B12_MANIFEST),
            "SourceB12RequestSha256": sha256_file(b12_root / B12_REQUEST),
            "SourceDiscoveryInventorySha256": b12_manifest["SourceInventorySha256"],
            "SourceDiscoveryBatchMapSha256": b12_manifest["SourceBatchMapSha256"],
            "SelectedSignals": counts["SelectedSignals"],
            "RequiredSignals": counts["RequiredSignals"],
            "FrameViewSignals": counts["FrameViewSignals"],
            "ExactValidityRules": counts["ExactValidityRules"],
            "NoHoldSignals": counts["NoHoldSignals"],
            "ValidityPolicies": {
                "CameraDistance": "EXACT_VALUES:327.66",
                "RadarDistanceToObject": "EXACT_VALUES:0;327.66",
                "AllOtherSignals": "NONE",
            },
            "HoldPolicy": "NONE_FOR_ALL_FRAME_VIEW_SIGNALS",
            "DecisionBoundary": (
                "Candidate profile only; no U001/02_CONTROL activation, Raw CAN decode, or U002 S1 change"
            ),
            "ControlFiles": control_records,
        }
        write_json(temporary / PROFILE_MANIFEST, manifest)
        line1 = (
            f"B13 N-S{counts['SelectedSignals']}-R{counts['RequiredSignals']}"
            f"-F{counts['FrameViewSignals']}-E{counts['ExactValidityRules']}"
            f"-N{counts['NoHoldSignals']}-P4-H0-X0 C-0"
        )
        line2 = "B13I C32766-R0.32766-HNONE-A0 C-0"
        (temporary / CAPSULE_NAME).write_text(line1 + "\n" + line2 + "\n", encoding="utf-8")
        os.replace(temporary, output_root)
        return validate_output(output_root)
    except Exception:
        if temporary.exists():
            shutil.rmtree(temporary)
        raise


def write_failure_diagnostic(
    path: Path,
    error: BaseException,
    context: Mapping[str, object],
) -> None:
    """Write rich diagnostics so a separate probe is normally unnecessary."""

    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        evidence: Dict[str, object] = {}
        for label, raw_path in context.items():
            candidate = Path(str(raw_path))
            evidence[label] = {
                "Exists": candidate.exists(),
                "IsDirectory": candidate.is_dir(),
                "FileCount": (
                    len([item for item in candidate.iterdir() if item.is_file()])
                    if candidate.is_dir()
                    else 0
                ),
            }
        payload = {
            "CompletionCode": 7,
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "Reason": getattr(error, "reason", machine_token(type(error).__name__)),
            "Detail": getattr(error, "detail", machine_token(str(error), 160)),
            "Context": {key: str(value) for key, value in context.items()},
            "PathEvidence": evidence,
            "Traceback": traceback.format_exception(type(error), error, error.__traceback__)[-10:],
        }
        write_json(path, payload)
    except Exception:
        pass


def make_synthetic_b12(b12_root: Path, case_root: Path) -> None:
    """Create a complete synthetic B12/U001 metadata handoff."""

    request_rows: List[Dict[str, object]] = []
    source_rows: List[Dict[str, object]] = []
    order = 1
    features = (
        ("D", "Distance"),
        ("L", "LateralPosition"),
        ("R", "RelativeSpeed"),
        ("O", "ObjectDetectionType"),
        ("T", "ObjectType"),
        ("S", "SwitchingFlag"),
        ("M", "MessageCounter"),
        ("Y", "LateralSpeed"),
        ("W", "Width"),
    )
    identities: List[Tuple[str, str, str, str, str, int, str]] = []
    for code, signal in features:
        for can_id, role in (("0x221", "CAMERA_1"), ("0x251", "CAMERA_3")):
            required = 1 if code in {"D", "L", "R"} else 0
            identities.append(("CAMERA_PAIR", code, role, "2", can_id, required, signal))
    identities.extend(
        [
            ("RADAR_SUPPORT", "D", "RADAR", "1", "0x380", 1, "DistanceToObject"),
            ("RADAR_SUPPORT", "F", "RADAR", "1", "0x380", 1, "PrecedingVehicleDetectionFlag"),
            ("RADAR_SUPPORT", "R", "RADAR", "1", "0x380", 0, "RelativeSpeed"),
            ("FRAME_DOMAIN", "F", "FRAME_KEY", "2", "0x620", 1, "FrameCount"),
        ]
    )
    for group, code, role, channel, can_id, required, signal in identities:
        canonical = f"CH{channel}_{can_id[2:]}_{signal}"
        request = {name: "" for name in B12_REQUEST_COLUMNS}
        request.update(
            {
                "RequestOrder": order,
                "RequestedProfileId": PROFILE_ID,
                "Group": group,
                "PairCode": code,
                "SourceRole": role,
                "Selected": 1,
                "Required": required,
                "Channel": channel,
                "CAN_ID_Hex": can_id,
                "MessageName": f"MSG_{can_id[2:]}",
                "SignalName": signal,
                "CanonicalName": canonical,
                "Unit": "m" if "Distance" in signal else "",
                "ProposedValueKind": "FLOAT",
                "ProposedSignalRole": "MEASUREMENT",
                "ProposedUpdateMode": "PERIODIC",
                "DiscoveryBatchId": "" if role == "FRAME_KEY" else "B001",
                "PolicyStatus": "REQUEST_ONLY_NOT_APPROVED",
                "EngineeringUse": "synthetic",
            }
        )
        request_rows.append(request)
        source_rows.append(
            {
                "Selected": 1,
                "Required": 1 if role == "FRAME_KEY" else 0,
                "CanonicalName": canonical,
                "Channel": channel,
                "CAN_ID_Hex": can_id,
                "MessageName": f"MSG_{can_id[2:]}",
                "SignalName": signal,
                "Unit": request["Unit"],
                "SourceGroupComplete": 1,
                "ReuseStatus": "MATCHED",
            }
        )
        order += 1

    profile_root = case_root / "DISCOVERY_PROFILE"
    profile_root.mkdir(parents=True)
    source_inventory_columns = SOURCE_INVENTORY_COLUMNS
    write_csv(profile_root / SOURCE_INVENTORY, source_inventory_columns, source_rows)
    (profile_root / SOURCE_BATCH_MAP).write_text("ProfileId,BatchId\nP,B001\n", encoding="utf-8")

    b12_root.mkdir(parents=True)
    write_csv(b12_root / B12_REQUEST, B12_REQUEST_COLUMNS, request_rows)
    (b12_root / B12_AUDIT).write_text("x\n1\n", encoding="utf-8")
    b12_manifest = {
        "CompletionCode": 0,
        "RequestedIdentities": 22,
        "ExactMatches": 22,
        "FuzzyOnlyMatches": 0,
        "AmbiguousMatches": 0,
        "MissingMatches": 0,
        "CameraPairCodes": "222222222",
        "RadarCodes": "111",
        "FrameKeyCode": 1,
        "SourceProfileRelativeRoot": "DISCOVERY_PROFILE",
        "SourceInventorySha256": sha256_file(profile_root / SOURCE_INVENTORY),
        "SourceBatchMapSha256": sha256_file(profile_root / SOURCE_BATCH_MAP),
        "Outputs": {
            B12_AUDIT: {"Sha256": sha256_file(b12_root / B12_AUDIT)},
            B12_REQUEST: {"Sha256": sha256_file(b12_root / B12_REQUEST)},
        },
    }
    write_json(b12_root / B12_MANIFEST, b12_manifest)
    (b12_root / B12_CAPSULE).write_text(
        "B12 N-P1-D1-C23-E22-M22-B1-F0-A0-X0 C-0\n"
        "B12P C-222222222-R111-F1 C-0\n",
        encoding="utf-8",
    )


def self_test() -> None:
    """Run deterministic configuration tests."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise AssertionError(f"self_test_{len(checks) + 1}")
        checks.append(True)

    with tempfile.TemporaryDirectory(prefix="u22b1b13_") as temporary:
        root = Path(temporary)
        b12_root = root / "B12"
        case_root = root / "CASE"
        output_root = root / PROFILE_ID
        make_synthetic_b12(b12_root, case_root)
        manifest, requests = require_b12_success(b12_root)
        check(len(requests) == 22)
        check(manifest["CameraPairCodes"] == "222222222")
        inventory = load_source_inventory(case_root, manifest)
        check(len(inventory) == 22)
        validated = validate_request_against_source(requests, inventory)
        check(validated[0]["SourceRole"] == "FRAME_KEY")
        check(validated[1]["Group"] == "RADAR_SUPPORT")
        g2, g4, g5, g6 = build_control_rows(validated, inventory)
        check(len(g2) == 22)
        check(len(g4) == 22)
        check(len(g5) == 22)
        check(len(g6) == 21)
        counts = validate_control_rows(
            [{k: str(v) for k, v in row.items()} for row in g2],
            [{k: str(v) for k, v in row.items()} for row in g4],
            [{k: str(v) for k, v in row.items()} for row in g5],
            [{k: str(v) for k, v in row.items()} for row in g6],
        )
        check(counts["RequiredSignals"] == 9)
        check(counts["ExactValidityRules"] == 3)
        check(all(row["HoldPolicy"] == "NONE" for row in g6))
        check(all(row["MaxAgeFrames"] == "" for row in g6))
        check(g6[0]["SignalIndex"] == "S1")
        check(g6[-1]["SignalIndex"] == "S21")
        frame4 = next(row for row in g4 if row["SignalRole"] == "FRAME_KEY")
        check(frame4["StorageDtype"] == "Int64")
        camera_distance = [
            row for row in g4
            if row["Channel"] == "2" and row["SignalName"] == "Distance"
        ]
        check(len(camera_distance) == 2)
        check(all(row["InvalidExactValues"] == "327.66" for row in camera_distance))
        radar_distance = next(row for row in g4 if row["SignalName"] == "DistanceToObject")
        check(radar_distance["InvalidExactValues"] == "0;327.66")
        switch_rows = [row for row in g5 if row["SignalRole"] == "EVENT_FLAG"]
        check(len(switch_rows) == 2)
        check(all(row["WithinFramePolicy"] == "ANY_TRUE" for row in switch_rows))

        line1, line2 = build_profile(b12_root, case_root, output_root)
        check(line1 == "B13 N-S22-R9-F21-E3-N21-P4-H0-X0 C-0")
        check(line2 == "B13I C32766-R0.32766-HNONE-A0 C-0")
        check(set(path.name for path in output_root.iterdir()) == set(OUTPUT_NAMES))
        v1, v2 = validate_output(output_root)
        check(v1 == line1)
        check(v2 == line2)
        profile_manifest = read_json(output_root / PROFILE_MANIFEST, "P")
        check(profile_manifest["ActivationAuthorized"] is False)
        check(profile_manifest["ApprovalStatus"] == "CANDIDATE_NOT_ACTIVATED")
        check(profile_manifest["HoldPolicy"] == "NONE_FOR_ALL_FRAME_VIEW_SIGNALS")
        check(len(profile_manifest["ControlFiles"]) == 4)
        check(all(len(record["Sha256"]) == 64 for record in profile_manifest["ControlFiles"].values()))
        check(not any("/private/" in line for line in (output_root / PROFILE_MANIFEST).read_text().splitlines()))
        check((output_root / CAPSULE_NAME).read_text().splitlines() == [line1, line2])

        try:
            build_profile(b12_root, case_root, output_root)
        except ProfileError as exc:
            check(exc.reason == "OUTPUT_ALREADY_EXISTS")
        else:
            check(False)
        check(tuple(profile_manifest["ControlFiles"].keys()) == tuple(sorted(CONTROL_NAMES)))
        check(profile_manifest["SelectedSignals"] == 22)
        if len(checks) != 36:
            raise AssertionError(f"self_test_count_{len(checks)}")
    print("B13-ST N-T36-X0 C-0")


def build_parser() -> argparse.ArgumentParser:
    """Build the command-line interface."""

    parser = argparse.ArgumentParser(description=PROGRAM_ID)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--build", action="store_true")
    mode.add_argument("--validate-output", action="store_true")
    parser.add_argument("--b12-root", type=Path)
    parser.add_argument("--u001-case-root", type=Path)
    parser.add_argument("--out", type=Path)
    parser.add_argument("--diagnostic", type=Path, default=DEFAULT_DIAGNOSTIC)
    return parser


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test, build, or read-only output validation."""

    args = build_parser().parse_args(argv)
    if args.self_test:
        self_test()
        return 0
    context = {
        "b12_root": args.b12_root or "MISSING",
        "u001_case_root": args.u001_case_root or "MISSING",
        "out": args.out or "MISSING",
    }
    try:
        if args.out is None:
            raise ProfileError("OUTPUT_ARGUMENT_MISSING")
        if args.validate_output:
            line1, line2 = validate_output(args.out)
        else:
            if args.b12_root is None or args.u001_case_root is None:
                raise ProfileError("BUILD_ARGUMENT_MISSING")
            line1, line2 = build_profile(args.b12_root, args.u001_case_root, args.out)
        print(line1)
        print(line2)
        return 0
    except Exception as exc:
        write_failure_diagnostic(args.diagnostic, exc, context)
        reason = getattr(exc, "reason", machine_token(type(exc).__name__))
        detail = getattr(exc, "detail", machine_token(str(exc), 72))
        print(f"B13E K-{reason} D-{detail} C-7")
        return 7


if __name__ == "__main__":
    sys.exit(main())
