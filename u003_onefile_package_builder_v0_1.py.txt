#!/usr/bin/env python3
"""Build and record the U003 Evidence Review Workbench onefile EXE."""

from __future__ import annotations

import argparse
import ast
import hashlib
import importlib
import importlib.util
import json
import os
import platform
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path
from typing import Any


BUILDER_VERSION = "0.1"
PRODUCT_EXE = "U003_Evidence_Review_Workbench_v0_1.exe"
EXPECTED_SOURCE_NAME = "framecount_video_reviewer_v0_5.py"
EXPECTED_ENTRY_NAME = "u003_onefile_entry_v0_1.py"
EXPECTED_SOURCE_SHA256 = "eb710fda7aeb59d0f59b251d9cd00aad4814ca4cd484e684a35eca8cf9a862e5"
EXPECTED_ENTRY_SHA256 = "5a6d1c2695c1d74aff158523971d959d607037bc9b49a106b8f32a106fbe595b"
OWNER_FILE = "UDP003PK01_BUILD_OWNER.json"


def canonical_bytes(path: Path) -> bytes:
    text = path.read_text(encoding="utf-8-sig")
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    return text.encode("utf-8")


def canonical_sha256(path: Path) -> str:
    return hashlib.sha256(canonical_bytes(path)).hexdigest()


def file_sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        while True:
            block = handle.read(1024 * 1024)
            if not block:
                break
            digest.update(block)
    return digest.hexdigest()


def ensure_owned_build_root(path: Path) -> None:
    path.mkdir(parents=True, exist_ok=True)
    owner = path / OWNER_FILE
    if owner.exists():
        data = json.loads(owner.read_text(encoding="utf-8"))
        if data.get("owner") != "UDP003PK01":
            raise RuntimeError("BUILD_ROOT_OWNER_MISMATCH")
    elif any(path.iterdir()):
        raise RuntimeError("BUILD_ROOT_NOT_EMPTY_OR_OWNED")
    owner.write_text(
        json.dumps(
            {"owner": "UDP003PK01", "builder_version": BUILDER_VERSION},
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    for name in ("work", "spec"):
        child = path / name
        if child.exists():
            shutil.rmtree(child)
        child.mkdir(parents=True)


def module_version(name: str) -> str:
    module = importlib.import_module(name)
    return str(getattr(module, "__version__", "available"))


def dependency_status() -> tuple[dict[str, Any], list[str]]:
    status: dict[str, Any] = {}
    failures: list[str] = []
    for name in ("PyInstaller", "cv2", "tkinter"):
        try:
            status[name] = module_version(name)
        except Exception as exc:
            status[name] = f"MISSING:{type(exc).__name__}"
            failures.append(f"DEPENDENCY_{name}")
    return status, failures


def write_evidence(evidence: Path, result: dict[str, Any], capsule: str, memos: list[str]) -> None:
    evidence.mkdir(parents=True, exist_ok=True)
    (evidence / "UDP003PK01_BUILD_RESULT.json").write_text(
        json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
    )
    lines = [capsule] + [f"MEMO{i + 1}-{memo}" for i, memo in enumerate(memos[:3])]
    (evidence / "UDP003PK01_RETURN_CAPSULE.txt").write_text(
        "\n".join(lines) + "\n", encoding="utf-8"
    )


def run_build(source: Path, entry: Path, dist: Path, evidence: Path, build_root: Path) -> int:
    failures: list[str] = []
    source_ok = (
        source.is_file()
        and source.name.lower() == EXPECTED_SOURCE_NAME.lower()
        and canonical_sha256(source) == EXPECTED_SOURCE_SHA256
    )
    entry_ok = (
        entry.is_file()
        and entry.name.lower() == EXPECTED_ENTRY_NAME.lower()
        and canonical_sha256(entry) == EXPECTED_ENTRY_SHA256
    )
    if not source_ok:
        failures.append("SOURCE_IDENTITY")
    if not entry_ok:
        failures.append("ENTRY_IDENTITY")

    dependency, dependency_failures = dependency_status()
    failures.extend(dependency_failures)
    pyinstaller_ok = "PyInstaller" in dependency and not str(dependency["PyInstaller"]).startswith("MISSING")
    tkinter_ok = "tkinter" in dependency and not str(dependency["tkinter"]).startswith("MISSING")
    cv2_ok = "cv2" in dependency and not str(dependency["cv2"]).startswith("MISSING")

    if failures:
        capsule = (
            f"UDP003PK01B N-S{int(source_ok)}-E{int(entry_ok)}-P{int(pyinstaller_ok)}-"
            f"T{int(tkinter_ok)}-C{int(cv2_ok)}-O0-X{len(failures)} C-7"
        )
        write_evidence(
            evidence,
            {"failures": failures, "dependency": dependency, "capsule": capsule},
            capsule,
            failures,
        )
        print(capsule)
        for index, failure in enumerate(failures[:3], start=1):
            print(f"MEMO{index}-{failure}")
        return 7

    if os.name != "nt":
        failures.append("BUILD_HOST_NOT_WINDOWS")
        capsule = "UDP003PK01B N-S1-E1-P1-T1-C1-O0-X1 C-7"
        write_evidence(evidence, {"failures": failures, "capsule": capsule}, capsule, failures)
        print(capsule)
        print("MEMO1-BUILD_HOST_NOT_WINDOWS")
        return 7

    ensure_owned_build_root(build_root)
    dist.mkdir(parents=True, exist_ok=True)
    exe = dist / PRODUCT_EXE
    if exe.exists():
        failures.append("RELEASE_EXE_ALREADY_EXISTS")
        capsule = "UDP003PK01B N-S1-E1-P1-T1-C1-O0-X1 C-7"
        write_evidence(evidence, {"failures": failures, "capsule": capsule}, capsule, failures)
        print(capsule)
        print("MEMO1-RELEASE_EXE_ALREADY_EXISTS")
        return 7

    work = build_root / "work"
    spec = build_root / "spec"
    command = [
        sys.executable,
        "-m",
        "PyInstaller",
        "--noconfirm",
        "--clean",
        "--onefile",
        "--windowed",
        "--noupx",
        "--log-level=WARN",
        "--name=U003_Evidence_Review_Workbench_v0_1",
        f"--distpath={dist.resolve()}",
        f"--workpath={work.resolve()}",
        f"--specpath={spec.resolve()}",
        f"--paths={source.parent.resolve()}",
        "--hidden-import=cv2",
        "--collect-all=cv2",
        "--hidden-import=tkinter",
        "--hidden-import=tkinter.ttk",
        "--hidden-import=tkinter.filedialog",
        "--hidden-import=tkinter.messagebox",
        "--hidden-import=tkinter.font",
        "--hidden-import=tkinter.scrolledtext",
        "--exclude-module=pandas",
        "--exclude-module=scipy",
        "--exclude-module=matplotlib",
        "--exclude-module=sklearn",
        "--exclude-module=statsmodels",
        "--exclude-module=pytest",
        str(entry.resolve()),
    ]
    process = subprocess.run(command, capture_output=True, text=True, errors="replace")
    evidence.mkdir(parents=True, exist_ok=True)
    (evidence / "UDP003PK01_BUILD_STDOUT.txt").write_text(process.stdout, encoding="utf-8")
    (evidence / "UDP003PK01_BUILD_STDERR.txt").write_text(process.stderr, encoding="utf-8")

    output_ok = process.returncode == 0 and exe.is_file() and exe.stat().st_size > 1024 * 1024
    if not output_ok:
        failures.append(f"PYINSTALLER_RC{process.returncode}")

    exe_size = exe.stat().st_size if exe.is_file() else 0
    exe_hash = file_sha256(exe) if exe.is_file() else ""
    capsule = (
        f"UDP003PK01B N-S1-E1-P1-T1-C1-O{int(output_ok)}-X{len(failures)} "
        f"Z-B{exe_size}-H{exe_hash[:12] or 'NONE'} C-{0 if output_ok else 7}"
    )
    result = {
        "builder_version": BUILDER_VERSION,
        "platform": platform.platform(),
        "python": sys.version,
        "source": str(source.resolve()),
        "source_canonical_sha256": canonical_sha256(source),
        "entry": str(entry.resolve()),
        "entry_canonical_sha256": canonical_sha256(entry),
        "dependency": dependency,
        "command": command,
        "process_returncode": process.returncode,
        "exe": str(exe.resolve()),
        "exe_size_bytes": exe_size,
        "exe_sha256": exe_hash,
        "failures": failures,
        "capsule": capsule,
    }
    write_evidence(evidence, result, capsule, failures)
    print(capsule)
    for index, failure in enumerate(failures[:3], start=1):
        print(f"MEMO{index}-{failure}")
    return 0 if output_ok else 7


def run_self_test() -> int:
    checks = 0

    def ok(condition: bool) -> None:
        nonlocal checks
        assert condition
        checks += 1

    with tempfile.TemporaryDirectory(prefix="udp003_pk01_selftest_") as temp_dir:
        root = Path(temp_dir)
        lf = root / "lf.py"
        crlf = root / "crlf.py"
        lf.write_bytes(b"x = 1\n")
        crlf.write_bytes(b"x = 1\r\n")
        ok(canonical_bytes(lf) == b"x = 1\n")
        ok(canonical_bytes(crlf) == b"x = 1\n")
        ok(canonical_sha256(lf) == canonical_sha256(crlf))
        ok(len(file_sha256(lf)) == 64)

        owned = root / "owned"
        ensure_owned_build_root(owned)
        ok((owned / OWNER_FILE).is_file())
        ok((owned / "work").is_dir())
        ok((owned / "spec").is_dir())
        ensure_owned_build_root(owned)
        ok((owned / "work").is_dir())

        foreign = root / "foreign"
        foreign.mkdir()
        (foreign / "data.txt").write_text("x", encoding="utf-8")
        try:
            ensure_owned_build_root(foreign)
            foreign_rejected = False
        except RuntimeError:
            foreign_rejected = True
        ok(foreign_rejected)

        evidence = root / "evidence"
        write_evidence(evidence, {"ok": True}, "TEST C-0", [])
        ok((evidence / "UDP003PK01_BUILD_RESULT.json").is_file())
        ok((evidence / "UDP003PK01_RETURN_CAPSULE.txt").is_file())
        ok("TEST C-0" in (evidence / "UDP003PK01_RETURN_CAPSULE.txt").read_text(encoding="utf-8"))

        tree = ast.parse("def main():\n    return 0\n")
        ok(isinstance(tree, ast.Module))
        ok(PRODUCT_EXE.endswith(".exe"))
        ok(EXPECTED_SOURCE_NAME.endswith(".py"))
        ok(EXPECTED_ENTRY_NAME.endswith(".py"))
        ok(len(EXPECTED_SOURCE_SHA256) == 64)
        ok(len(EXPECTED_ENTRY_SHA256) == 64)
        ok(BUILDER_VERSION == "0.1")

    assert checks == 19, checks
    print("UDP003-PK01-SELFTEST-PASS-19")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--source", type=Path)
    parser.add_argument("--entry", type=Path)
    parser.add_argument("--dist", type=Path)
    parser.add_argument("--evidence", type=Path)
    parser.add_argument("--build-root", type=Path)
    args = parser.parse_args()

    if args.self_test:
        return run_self_test()
    required = (args.source, args.entry, args.dist, args.evidence, args.build_root)
    if any(value is None for value in required):
        parser.error("build requires --source --entry --dist --evidence --build-root")
    return run_build(args.source, args.entry, args.dist, args.evidence, args.build_root)


if __name__ == "__main__":
    raise SystemExit(main())
