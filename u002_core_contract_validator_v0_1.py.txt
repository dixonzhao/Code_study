#!/usr/bin/env python3
"""U002 Core Contract validator v0.1.

Standard-library validator for the U2-0A Module, Artifact and Run manifests.
It intentionally validates the approved U002 contract subset rather than
attempting to implement the complete JSON Schema standard.
"""

from __future__ import annotations

import argparse
import copy
import hashlib
import json
import os
import platform
import re
import sys
from datetime import datetime
from pathlib import Path
from typing import Any, Callable


CONTRACT_VERSION = "0.1"
BUNDLE_ID = "urn:u002:core-contract-schema-bundle:0.1"
STAGES = tuple(f"S{i}" for i in range(8))
SHA_RE = re.compile(r"^[0-9a-f]{64}$")
VERSION_RE = re.compile(r"^[0-9]+\.[0-9]+\.[0-9]+$")
MODULE_ID_RE = re.compile(r"^U002\.[A-Za-z0-9][A-Za-z0-9_.-]{2,95}$")
ARTIFACT_ID_RE = re.compile(r"^ART_[A-Za-z0-9][A-Za-z0-9_.-]{5,127}$")
RUN_ID_RE = re.compile(r"^RUN_[A-Za-z0-9][A-Za-z0-9_.-]{5,127}$")
EXTENSION_KEY_RE = re.compile(r"^[A-Za-z][A-Za-z0-9_.-]{2,127}$")


class ContractError(ValueError):
    def __init__(self, code: str, path: str, message: str):
        super().__init__(message)
        self.code = code
        self.path = path
        self.message = message

    def compact(self) -> str:
        return f"{self.code}@{self.path}:{self.message}"


def fail(code: str, path: str, message: str) -> None:
    raise ContractError(code, path, message)


def require_object(value: Any, path: str) -> dict[str, Any]:
    if not isinstance(value, dict):
        fail("TYPE_OBJECT", path, "object required")
    return value


def require_array(value: Any, path: str) -> list[Any]:
    if not isinstance(value, list):
        fail("TYPE_ARRAY", path, "array required")
    return value


def require_string(value: Any, path: str, allow_empty: bool = False) -> str:
    if not isinstance(value, str):
        fail("TYPE_STRING", path, "string required")
    if not allow_empty and not value:
        fail("EMPTY_STRING", path, "non-empty string required")
    return value


def require_int(value: Any, path: str, minimum: int | None = None) -> int:
    if isinstance(value, bool) or not isinstance(value, int):
        fail("TYPE_INTEGER", path, "integer required")
    if minimum is not None and value < minimum:
        fail("INTEGER_RANGE", path, f"minimum is {minimum}")
    return value


def require_number(value: Any, path: str, positive: bool = False) -> float:
    if isinstance(value, bool) or not isinstance(value, (int, float)):
        fail("TYPE_NUMBER", path, "number required")
    result = float(value)
    if positive and result <= 0:
        fail("NUMBER_RANGE", path, "positive number required")
    return result


def require_bool(value: Any, path: str) -> bool:
    if not isinstance(value, bool):
        fail("TYPE_BOOLEAN", path, "boolean required")
    return value


def require_enum(value: Any, allowed: set[str], path: str) -> str:
    text = require_string(value, path)
    if text not in allowed:
        fail("ENUM", path, "allowed=" + ",".join(sorted(allowed)))
    return text


def require_pattern(value: Any, pattern: re.Pattern[str], path: str, code: str) -> str:
    text = require_string(value, path)
    if not pattern.fullmatch(text):
        fail(code, path, "invalid format")
    return text


def require_sha(value: Any, path: str) -> str:
    return require_pattern(value, SHA_RE, path, "SHA256")


def require_version(value: Any, path: str) -> str:
    return require_pattern(value, VERSION_RE, path, "VERSION")


def version_tuple(value: str) -> tuple[int, int, int]:
    return tuple(int(part) for part in value.split("."))  # type: ignore[return-value]


def require_iso_datetime(value: Any, path: str, nullable: bool = False) -> str | None:
    if value is None and nullable:
        return None
    text = require_string(value, path)
    try:
        datetime.fromisoformat(text.replace("Z", "+00:00"))
    except ValueError:
        fail("DATETIME", path, "ISO-8601 date-time required")
    return text


def require_relative_path(value: Any, path: str) -> str:
    text = require_string(value, path)
    normalized = text.replace("\\", "/")
    if normalized.startswith("/") or re.match(r"^[A-Za-z]:/", normalized):
        fail("ABSOLUTE_PATH", path, "relative path required")
    if any(part == ".." for part in normalized.split("/")):
        fail("PARENT_TRAVERSAL", path, "parent traversal prohibited")
    if any(part == "" for part in normalized.split("/")):
        fail("PATH_SEGMENT", path, "empty path segment prohibited")
    return text


def require_keys(
    obj: dict[str, Any],
    required: set[str],
    optional: set[str],
    path: str,
) -> None:
    missing = sorted(required - set(obj))
    if missing:
        fail("MISSING_FIELD", path, ",".join(missing))
    extra = sorted(set(obj) - required - optional)
    if extra:
        fail("UNKNOWN_FIELD", path, ",".join(extra))


def ensure_unique(values: list[Any], path: str, label: str) -> None:
    if len(values) != len(set(values)):
        fail("DUPLICATE", path, label)


def validate_extensions(value: Any, path: str) -> None:
    obj = require_object(value, path)
    for key in obj:
        if not EXTENSION_KEY_RE.fullmatch(key):
            fail("EXTENSION_KEY", f"{path}.{key}", "namespaced extension key required")


def validate_signal_selector(value: Any, path: str) -> None:
    obj = require_object(value, path)
    allowed = {
        "Channel", "CanId", "MessageName", "SignalName", "CanonicalName",
        "Unit", "ValueKind", "SignalRole",
    }
    require_keys(obj, set(), allowed, path)
    if not obj:
        fail("EMPTY_SELECTOR", path, "at least one selector field required")
    if "Channel" in obj and obj["Channel"] is not None:
        require_int(obj["Channel"], f"{path}.Channel", 0)
    for key in allowed - {"Channel", "ValueKind"}:
        if key in obj and obj[key] is not None:
            require_string(obj[key], f"{path}.{key}")
    if "ValueKind" in obj and obj["ValueKind"] is not None:
        require_enum(
            obj["ValueKind"],
            {"NUMERIC", "BOOLEAN", "ENUM", "IDENTIFIER", "COUNTER", "FLAG", "TEXT"},
            f"{path}.ValueKind",
        )


def validate_input_port(value: Any, path: str) -> None:
    obj = require_object(value, path)
    require_keys(obj, {"PortId", "Role", "MinCount", "MaxCount", "Selectors"}, set(), path)
    require_string(obj["PortId"], f"{path}.PortId")
    require_string(obj["Role"], f"{path}.Role")
    minimum = require_int(obj["MinCount"], f"{path}.MinCount", 0)
    maximum = obj["MaxCount"]
    if maximum is not None:
        maximum = require_int(maximum, f"{path}.MaxCount", 1)
        if maximum < minimum:
            fail("CARDINALITY", path, "MaxCount below MinCount")
    selectors = require_array(obj["Selectors"], f"{path}.Selectors")
    for index, selector in enumerate(selectors):
        validate_signal_selector(selector, f"{path}.Selectors[{index}]")


def validate_artifact_requirement(value: Any, path: str) -> None:
    obj = require_object(value, path)
    required = {"PortId", "ArtifactType", "ArtifactSchemaId", "MinimumSchemaVersion", "Cardinality"}
    require_keys(obj, required, set(), path)
    for key in ("PortId", "ArtifactType", "ArtifactSchemaId"):
        require_string(obj[key], f"{path}.{key}")
    require_version(obj["MinimumSchemaVersion"], f"{path}.MinimumSchemaVersion")
    require_enum(obj["Cardinality"], {"ONE", "ZERO_OR_ONE", "ONE_OR_MORE", "ZERO_OR_MORE"}, f"{path}.Cardinality")


def validate_artifact_provision(value: Any, path: str) -> None:
    obj = require_object(value, path)
    required = {"ArtifactType", "ArtifactSchemaId", "ArtifactSchemaVersion", "ExclusiveOutput"}
    require_keys(obj, required, set(), path)
    require_string(obj["ArtifactType"], f"{path}.ArtifactType")
    require_string(obj["ArtifactSchemaId"], f"{path}.ArtifactSchemaId")
    require_version(obj["ArtifactSchemaVersion"], f"{path}.ArtifactSchemaVersion")
    require_bool(obj["ExclusiveOutput"], f"{path}.ExclusiveOutput")


def validate_module_manifest(value: Any) -> None:
    path = "ModuleManifest"
    obj = require_object(value, path)
    required = {
        "ContractType", "ContractVersion", "ModuleId", "ModuleVersion", "DisplayName",
        "Stage", "ModuleKind", "AlgorithmStatus", "ExecutionRequirement", "ScopeSelector",
        "RequiredInputs", "OptionalInputs", "RequiresArtifacts", "ProvidesArtifacts",
        "OutputNamespace", "ConfigurationSchema", "FailurePolicy", "ConflictPolicy",
        "Entrypoint", "SelfTest", "Compatibility", "ImplementationSha256",
    }
    optional = {"Description", "Extensions"}
    require_keys(obj, required, optional, path)
    if obj["ContractType"] != "U002_MODULE_MANIFEST":
        fail("CONTRACT_TYPE", f"{path}.ContractType", "U002_MODULE_MANIFEST required")
    if obj["ContractVersion"] != CONTRACT_VERSION:
        fail("CONTRACT_VERSION", f"{path}.ContractVersion", CONTRACT_VERSION)
    require_pattern(obj["ModuleId"], MODULE_ID_RE, f"{path}.ModuleId", "MODULE_ID")
    require_version(obj["ModuleVersion"], f"{path}.ModuleVersion")
    require_string(obj["DisplayName"], f"{path}.DisplayName")
    if "Description" in obj:
        require_string(obj["Description"], f"{path}.Description", allow_empty=True)
    stage = require_enum(obj["Stage"], set(STAGES), f"{path}.Stage")
    kind = require_enum(
        obj["ModuleKind"],
        {
            "INTAKE_GATE", "VALIDITY_PREPROCESSOR", "SEMANTIC_BUILDER", "ARBITRATION",
            "ANALYSIS_VIEW", "PRIMARY_OUTCOME", "BASELINE", "EVIDENCE",
            "CANDIDATE_SYNTHESIS", "HANDOFF_EXPORT",
        },
        f"{path}.ModuleKind",
    )
    stage_kinds = {
        "S0": {"INTAKE_GATE"},
        "S1": {"VALIDITY_PREPROCESSOR", "SEMANTIC_BUILDER", "ARBITRATION", "ANALYSIS_VIEW"},
        "S2": {"ANALYSIS_VIEW"},
        "S3": {"PRIMARY_OUTCOME"},
        "S4": {"BASELINE"},
        "S5": {"EVIDENCE"},
        "S6": {"CANDIDATE_SYNTHESIS"},
        "S7": {"HANDOFF_EXPORT"},
    }
    if kind not in stage_kinds[stage]:
        fail("STAGE_KIND", f"{path}.ModuleKind", f"{kind} not allowed in {stage}")
    require_enum(obj["AlgorithmStatus"], {"FORMAL", "PROVISIONAL", "EXPERIMENTAL", "DEPRECATED"}, f"{path}.AlgorithmStatus")
    execution = require_enum(obj["ExecutionRequirement"], {"REQUIRED", "OPTIONAL"}, f"{path}.ExecutionRequirement")

    scope = require_object(obj["ScopeSelector"], f"{path}.ScopeSelector")
    require_keys(scope, {"SignalSelectors", "FrameRangeSupported"}, set(), f"{path}.ScopeSelector")
    selectors = require_array(scope["SignalSelectors"], f"{path}.ScopeSelector.SignalSelectors")
    for index, selector in enumerate(selectors):
        validate_signal_selector(selector, f"{path}.ScopeSelector.SignalSelectors[{index}]")
    require_bool(scope["FrameRangeSupported"], f"{path}.ScopeSelector.FrameRangeSupported")

    all_ports: list[str] = []
    for field in ("RequiredInputs", "OptionalInputs"):
        ports = require_array(obj[field], f"{path}.{field}")
        for index, port in enumerate(ports):
            validate_input_port(port, f"{path}.{field}[{index}]")
            all_ports.append(port["PortId"])
    require_artifacts = require_array(obj["RequiresArtifacts"], f"{path}.RequiresArtifacts")
    for index, requirement in enumerate(require_artifacts):
        validate_artifact_requirement(requirement, f"{path}.RequiresArtifacts[{index}]")
        all_ports.append(requirement["PortId"])
    ensure_unique(all_ports, path, "input PortId")

    provisions = require_array(obj["ProvidesArtifacts"], f"{path}.ProvidesArtifacts")
    if not provisions:
        fail("EMPTY_PROVISION", f"{path}.ProvidesArtifacts", "at least one output required")
    provision_keys: list[str] = []
    has_exclusive = False
    for index, provision in enumerate(provisions):
        validate_artifact_provision(provision, f"{path}.ProvidesArtifacts[{index}]")
        provision_keys.append(provision["ArtifactType"] + "@" + provision["ArtifactSchemaId"])
        has_exclusive = has_exclusive or provision["ExclusiveOutput"]
    ensure_unique(provision_keys, f"{path}.ProvidesArtifacts", "provided Artifact identity")

    require_pattern(obj["OutputNamespace"], re.compile(r"^[A-Za-z][A-Za-z0-9_.-]{2,95}$"), f"{path}.OutputNamespace", "OUTPUT_NAMESPACE")
    config = require_object(obj["ConfigurationSchema"], f"{path}.ConfigurationSchema")
    require_keys(config, {"SchemaId", "SchemaVersion", "RelativePath", "Sha256"}, set(), f"{path}.ConfigurationSchema")
    require_string(config["SchemaId"], f"{path}.ConfigurationSchema.SchemaId")
    require_version(config["SchemaVersion"], f"{path}.ConfigurationSchema.SchemaVersion")
    require_relative_path(config["RelativePath"], f"{path}.ConfigurationSchema.RelativePath")
    require_sha(config["Sha256"], f"{path}.ConfigurationSchema.Sha256")

    failure_policy = require_enum(obj["FailurePolicy"], {"ABORT_RUN", "BLOCK_DEPENDENT_PATH", "SKIP_OPTIONAL_PATH"}, f"{path}.FailurePolicy")
    if execution == "REQUIRED" and failure_policy == "SKIP_OPTIONAL_PATH":
        fail("FAILURE_POLICY", f"{path}.FailurePolicy", "required Module cannot use optional skip")
    if execution == "OPTIONAL" and failure_policy != "SKIP_OPTIONAL_PATH":
        fail("FAILURE_POLICY", f"{path}.FailurePolicy", "optional Module must use optional skip")
    conflict = require_enum(obj["ConflictPolicy"], {"ERROR_ON_COLLISION", "REQUIRE_ARBITRATION", "APPEND_ONLY"}, f"{path}.ConflictPolicy")
    if has_exclusive and conflict == "APPEND_ONLY":
        fail("CONFLICT_POLICY", f"{path}.ConflictPolicy", "exclusive output cannot append")

    entry = require_object(obj["Entrypoint"], f"{path}.Entrypoint")
    require_keys(entry, {"RelativeFile", "Callable"}, set(), f"{path}.Entrypoint")
    require_relative_path(entry["RelativeFile"], f"{path}.Entrypoint.RelativeFile")
    require_string(entry["Callable"], f"{path}.Entrypoint.Callable")
    self_test = require_object(obj["SelfTest"], f"{path}.SelfTest")
    require_keys(self_test, {"Arguments", "ExpectedExitCode", "ExpectedToken"}, set(), f"{path}.SelfTest")
    args = require_array(self_test["Arguments"], f"{path}.SelfTest.Arguments")
    for index, argument in enumerate(args):
        require_string(argument, f"{path}.SelfTest.Arguments[{index}]", allow_empty=True)
    require_int(self_test["ExpectedExitCode"], f"{path}.SelfTest.ExpectedExitCode")
    require_string(self_test["ExpectedToken"], f"{path}.SelfTest.ExpectedToken")

    compatibility = require_object(obj["Compatibility"], f"{path}.Compatibility")
    require_keys(compatibility, {"MinimumEngineVersion", "MaximumEngineVersion"}, set(), f"{path}.Compatibility")
    minimum_engine = require_version(compatibility["MinimumEngineVersion"], f"{path}.Compatibility.MinimumEngineVersion")
    maximum_engine = compatibility["MaximumEngineVersion"]
    if maximum_engine is not None:
        maximum_engine = require_version(maximum_engine, f"{path}.Compatibility.MaximumEngineVersion")
        if version_tuple(maximum_engine) < version_tuple(minimum_engine):
            fail("ENGINE_RANGE", f"{path}.Compatibility", "maximum below minimum")
    require_sha(obj["ImplementationSha256"], f"{path}.ImplementationSha256")
    if "Extensions" in obj:
        validate_extensions(obj["Extensions"], f"{path}.Extensions")


def validate_run_scope(value: Any, path: str) -> None:
    obj = require_object(value, path)
    require_keys(obj, {"ScopeType", "CaseId", "FrameStart", "FrameEnd"}, set(), path)
    scope_type = require_enum(obj["ScopeType"], {"FULL_CASE", "FRAME_RANGE"}, f"{path}.ScopeType")
    require_string(obj["CaseId"], f"{path}.CaseId")
    start = obj["FrameStart"]
    end = obj["FrameEnd"]
    if scope_type == "FULL_CASE":
        if start is not None or end is not None:
            fail("RUN_SCOPE", path, "FULL_CASE requires null FrameStart and FrameEnd")
    else:
        start_value = require_int(start, f"{path}.FrameStart", 0)
        end_value = require_int(end, f"{path}.FrameEnd", 0)
        if end_value < start_value:
            fail("FRAME_RANGE", path, "FrameEnd below FrameStart")


def validate_clock_domain(value: Any, path: str) -> None:
    obj = require_object(value, path)
    require_keys(obj, {"ClockDomainId", "ClockKind", "FrameRateHz", "TimeUnit"}, set(), path)
    require_string(obj["ClockDomainId"], f"{path}.ClockDomainId")
    kind = require_enum(obj["ClockKind"], {"FRAMECOUNT", "TIMESTAMP", "HYBRID"}, f"{path}.ClockKind")
    rate = obj["FrameRateHz"]
    unit = obj["TimeUnit"]
    if kind in {"FRAMECOUNT", "HYBRID"}:
        require_number(rate, f"{path}.FrameRateHz", positive=True)
    elif rate is not None:
        require_number(rate, f"{path}.FrameRateHz", positive=True)
    if kind in {"TIMESTAMP", "HYBRID"}:
        require_string(unit, f"{path}.TimeUnit")
    elif unit is not None:
        require_string(unit, f"{path}.TimeUnit")


def validate_source_reference(value: Any, path: str) -> None:
    obj = require_object(value, path)
    require_keys(obj, {"SourceId", "SourceType", "IdentitySha256"}, set(), path)
    require_string(obj["SourceId"], f"{path}.SourceId")
    require_string(obj["SourceType"], f"{path}.SourceType")
    require_sha(obj["IdentitySha256"], f"{path}.IdentitySha256")


def validate_artifact_manifest(value: Any) -> None:
    path = "ArtifactManifest"
    obj = require_object(value, path)
    required = {
        "ContractType", "ContractVersion", "ArtifactId", "ArtifactType", "ArtifactSchemaId",
        "ArtifactSchemaVersion", "ArtifactStatus", "Stage", "RunId", "CreatedAtUtc",
        "Producer", "Parents", "Payloads", "Scope", "ClockDomain", "SourceReferences",
        "Metadata",
    }
    optional = {"CoordinateFrame", "UncertaintyReference", "Extensions"}
    require_keys(obj, required, optional, path)
    if obj["ContractType"] != "U002_ARTIFACT_MANIFEST":
        fail("CONTRACT_TYPE", f"{path}.ContractType", "U002_ARTIFACT_MANIFEST required")
    if obj["ContractVersion"] != CONTRACT_VERSION:
        fail("CONTRACT_VERSION", f"{path}.ContractVersion", CONTRACT_VERSION)
    require_pattern(obj["ArtifactId"], ARTIFACT_ID_RE, f"{path}.ArtifactId", "ARTIFACT_ID")
    require_string(obj["ArtifactType"], f"{path}.ArtifactType")
    require_string(obj["ArtifactSchemaId"], f"{path}.ArtifactSchemaId")
    require_version(obj["ArtifactSchemaVersion"], f"{path}.ArtifactSchemaVersion")
    status = require_enum(obj["ArtifactStatus"], {"COMPLETE", "PARTIAL", "FAILED"}, f"{path}.ArtifactStatus")
    require_enum(obj["Stage"], set(STAGES), f"{path}.Stage")
    require_pattern(obj["RunId"], RUN_ID_RE, f"{path}.RunId", "RUN_ID")
    require_iso_datetime(obj["CreatedAtUtc"], f"{path}.CreatedAtUtc")

    producer = require_object(obj["Producer"], f"{path}.Producer")
    require_keys(producer, {"ModuleId", "ModuleVersion", "ModuleManifestSha256"}, set(), f"{path}.Producer")
    require_string(producer["ModuleId"], f"{path}.Producer.ModuleId")
    require_version(producer["ModuleVersion"], f"{path}.Producer.ModuleVersion")
    require_sha(producer["ModuleManifestSha256"], f"{path}.Producer.ModuleManifestSha256")

    parents = require_array(obj["Parents"], f"{path}.Parents")
    parent_ids: list[str] = []
    for index, parent in enumerate(parents):
        parent_path = f"{path}.Parents[{index}]"
        parent_obj = require_object(parent, parent_path)
        require_keys(parent_obj, {"ArtifactId", "ManifestSha256", "ContentSha256"}, set(), parent_path)
        parent_ids.append(require_string(parent_obj["ArtifactId"], f"{parent_path}.ArtifactId"))
        require_sha(parent_obj["ManifestSha256"], f"{parent_path}.ManifestSha256")
        require_sha(parent_obj["ContentSha256"], f"{parent_path}.ContentSha256")
    ensure_unique(parent_ids, f"{path}.Parents", "ArtifactId")

    payloads = require_array(obj["Payloads"], f"{path}.Payloads")
    if status == "COMPLETE" and not payloads:
        fail("COMPLETE_WITHOUT_PAYLOAD", f"{path}.Payloads", "complete Artifact requires payload")
    payload_ids: list[str] = []
    for index, payload in enumerate(payloads):
        payload_path = f"{path}.Payloads[{index}]"
        payload_obj = require_object(payload, payload_path)
        required_payload = {"PayloadId", "Role", "RelativePath", "MediaType", "ByteSize", "Sha256"}
        optional_payload = {"RowCount", "PayloadKind"}
        require_keys(payload_obj, required_payload, optional_payload, payload_path)
        payload_ids.append(require_string(payload_obj["PayloadId"], f"{payload_path}.PayloadId"))
        require_enum(payload_obj["Role"], {"PRIMARY", "INDEX", "DIAGNOSTIC", "PRESENTATION"}, f"{payload_path}.Role")
        require_relative_path(payload_obj["RelativePath"], f"{payload_path}.RelativePath")
        require_string(payload_obj["MediaType"], f"{payload_path}.MediaType")
        require_int(payload_obj["ByteSize"], f"{payload_path}.ByteSize", 0)
        require_sha(payload_obj["Sha256"], f"{payload_path}.Sha256")
        if "RowCount" in payload_obj and payload_obj["RowCount"] is not None:
            require_int(payload_obj["RowCount"], f"{payload_path}.RowCount", 0)
        if "PayloadKind" in payload_obj and payload_obj["PayloadKind"] is not None:
            require_enum(payload_obj["PayloadKind"], {"TABLE", "JSON", "MODEL", "POINT_CLOUD_REFERENCE", "VIDEO_REFERENCE", "OTHER"}, f"{payload_path}.PayloadKind")
    ensure_unique(payload_ids, f"{path}.Payloads", "PayloadId")

    validate_run_scope(obj["Scope"], f"{path}.Scope")
    validate_clock_domain(obj["ClockDomain"], f"{path}.ClockDomain")
    sources = require_array(obj["SourceReferences"], f"{path}.SourceReferences")
    source_ids: list[str] = []
    for index, source in enumerate(sources):
        validate_source_reference(source, f"{path}.SourceReferences[{index}]")
        source_ids.append(source["SourceId"])
    ensure_unique(source_ids, f"{path}.SourceReferences", "SourceId")
    metadata = require_object(obj["Metadata"], f"{path}.Metadata")
    if status == "FAILED" and not metadata.get("ErrorCode"):
        fail("FAILED_ARTIFACT_DIAGNOSTIC", f"{path}.Metadata.ErrorCode", "required for failed Artifact")
    for optional_text in ("CoordinateFrame", "UncertaintyReference"):
        if optional_text in obj and obj[optional_text] is not None:
            require_string(obj[optional_text], f"{path}.{optional_text}")
    if "Extensions" in obj:
        validate_extensions(obj["Extensions"], f"{path}.Extensions")


def validate_resolved_module(value: Any, path: str) -> None:
    obj = require_object(value, path)
    required = {
        "Order", "Stage", "ModuleId", "ModuleVersion", "ModuleManifestSha256",
        "ResolvedConfigSha256", "ExecutionRequirement", "Status",
    }
    require_keys(obj, required, set(), path)
    require_int(obj["Order"], f"{path}.Order", 0)
    require_enum(obj["Stage"], set(STAGES), f"{path}.Stage")
    require_string(obj["ModuleId"], f"{path}.ModuleId")
    require_version(obj["ModuleVersion"], f"{path}.ModuleVersion")
    require_sha(obj["ModuleManifestSha256"], f"{path}.ModuleManifestSha256")
    require_sha(obj["ResolvedConfigSha256"], f"{path}.ResolvedConfigSha256")
    require_enum(obj["ExecutionRequirement"], {"REQUIRED", "OPTIONAL"}, f"{path}.ExecutionRequirement")
    require_enum(obj["Status"], {"PENDING", "RUNNING", "SUCCEEDED", "SKIPPED", "FAILED"}, f"{path}.Status")


def validate_run_manifest(value: Any) -> None:
    path = "RunManifest"
    obj = require_object(value, path)
    required = {
        "ContractType", "ContractVersion", "RunId", "EngineVersion", "CaseId", "RunStatus",
        "CreatedAtUtc", "StartedAtUtc", "CompletedAtUtc", "InputPackages", "RunScope",
        "ClockDomain", "RegistrySnapshotSha256", "ConfigSnapshotSha256", "ResolvedModules",
        "ResolvedProfiles", "Stages", "Artifacts", "Environment", "DiagnosticsRelativePath",
    }
    optional = {"ResumeFromArtifactId", "Extensions"}
    require_keys(obj, required, optional, path)
    if obj["ContractType"] != "U002_RUN_MANIFEST":
        fail("CONTRACT_TYPE", f"{path}.ContractType", "U002_RUN_MANIFEST required")
    if obj["ContractVersion"] != CONTRACT_VERSION:
        fail("CONTRACT_VERSION", f"{path}.ContractVersion", CONTRACT_VERSION)
    require_pattern(obj["RunId"], RUN_ID_RE, f"{path}.RunId", "RUN_ID")
    require_version(obj["EngineVersion"], f"{path}.EngineVersion")
    case_id = require_string(obj["CaseId"], f"{path}.CaseId")
    status = require_enum(obj["RunStatus"], {"PENDING", "RUNNING", "SUCCEEDED", "PARTIAL", "FAILED", "CANCELLED"}, f"{path}.RunStatus")
    require_iso_datetime(obj["CreatedAtUtc"], f"{path}.CreatedAtUtc")
    started = require_iso_datetime(obj["StartedAtUtc"], f"{path}.StartedAtUtc", nullable=True)
    completed = require_iso_datetime(obj["CompletedAtUtc"], f"{path}.CompletedAtUtc", nullable=True)

    packages = require_array(obj["InputPackages"], f"{path}.InputPackages")
    if not packages:
        fail("INPUT_PACKAGE", f"{path}.InputPackages", "at least one input package required")
    package_ids: list[str] = []
    for index, package in enumerate(packages):
        validate_source_reference(package, f"{path}.InputPackages[{index}]")
        package_ids.append(package["SourceId"])
    ensure_unique(package_ids, f"{path}.InputPackages", "SourceId")

    validate_run_scope(obj["RunScope"], f"{path}.RunScope")
    if obj["RunScope"]["CaseId"] != case_id:
        fail("CASE_ID_MISMATCH", f"{path}.RunScope.CaseId", "must equal RunManifest.CaseId")
    validate_clock_domain(obj["ClockDomain"], f"{path}.ClockDomain")
    require_sha(obj["RegistrySnapshotSha256"], f"{path}.RegistrySnapshotSha256")
    require_sha(obj["ConfigSnapshotSha256"], f"{path}.ConfigSnapshotSha256")

    modules = require_array(obj["ResolvedModules"], f"{path}.ResolvedModules")
    if not modules:
        fail("RESOLVED_MODULE", f"{path}.ResolvedModules", "at least one Module required")
    module_ids: list[str] = []
    module_orders: list[int] = []
    for index, module in enumerate(modules):
        validate_resolved_module(module, f"{path}.ResolvedModules[{index}]")
        module_ids.append(module["ModuleId"])
        module_orders.append(module["Order"])
    ensure_unique(module_ids, f"{path}.ResolvedModules", "ModuleId")
    ensure_unique(module_orders, f"{path}.ResolvedModules", "Order")
    if module_orders != sorted(module_orders):
        fail("MODULE_ORDER", f"{path}.ResolvedModules", "Modules must be stored in execution order")

    profiles = require_array(obj["ResolvedProfiles"], f"{path}.ResolvedProfiles")
    profile_keys: list[str] = []
    for index, profile in enumerate(profiles):
        profile_path = f"{path}.ResolvedProfiles[{index}]"
        profile_obj = require_object(profile, profile_path)
        require_keys(profile_obj, {"ProfileType", "ProfileId", "ProfileVersion", "ResolvedSha256"}, set(), profile_path)
        require_string(profile_obj["ProfileType"], f"{profile_path}.ProfileType")
        require_string(profile_obj["ProfileId"], f"{profile_path}.ProfileId")
        require_version(profile_obj["ProfileVersion"], f"{profile_path}.ProfileVersion")
        require_sha(profile_obj["ResolvedSha256"], f"{profile_path}.ResolvedSha256")
        profile_keys.append(profile_obj["ProfileType"] + "@" + profile_obj["ProfileId"])
    ensure_unique(profile_keys, f"{path}.ResolvedProfiles", "Profile identity")

    stages = require_array(obj["Stages"], f"{path}.Stages")
    stage_names: list[str] = []
    has_failed_stage = False
    for index, stage in enumerate(stages):
        stage_path = f"{path}.Stages[{index}]"
        stage_obj = require_object(stage, stage_path)
        require_keys(stage_obj, {"Stage", "Status", "InputArtifactIds", "OutputArtifactIds", "ErrorCode"}, set(), stage_path)
        stage_name = require_enum(stage_obj["Stage"], set(STAGES), f"{stage_path}.Stage")
        stage_status = require_enum(stage_obj["Status"], {"PENDING", "RUNNING", "SUCCEEDED", "SKIPPED", "FAILED"}, f"{stage_path}.Status")
        stage_names.append(stage_name)
        for field in ("InputArtifactIds", "OutputArtifactIds"):
            ids = require_array(stage_obj[field], f"{stage_path}.{field}")
            for id_index, artifact_id in enumerate(ids):
                require_string(artifact_id, f"{stage_path}.{field}[{id_index}]")
            ensure_unique(ids, f"{stage_path}.{field}", "ArtifactId")
        if stage_status == "FAILED":
            has_failed_stage = True
            require_string(stage_obj["ErrorCode"], f"{stage_path}.ErrorCode")
        elif stage_obj["ErrorCode"] is not None:
            require_string(stage_obj["ErrorCode"], f"{stage_path}.ErrorCode")
    ensure_unique(stage_names, f"{path}.Stages", "Stage")
    if [STAGES.index(name) for name in stage_names] != sorted(STAGES.index(name) for name in stage_names):
        fail("STAGE_ORDER", f"{path}.Stages", "Stages must be stored S0 to S7")

    artifacts = require_array(obj["Artifacts"], f"{path}.Artifacts")
    artifact_ids: list[str] = []
    for index, artifact in enumerate(artifacts):
        artifact_path = f"{path}.Artifacts[{index}]"
        artifact_obj = require_object(artifact, artifact_path)
        require_keys(artifact_obj, {"ArtifactId", "ManifestRelativePath", "ManifestSha256"}, set(), artifact_path)
        artifact_ids.append(require_string(artifact_obj["ArtifactId"], f"{artifact_path}.ArtifactId"))
        require_relative_path(artifact_obj["ManifestRelativePath"], f"{artifact_path}.ManifestRelativePath")
        require_sha(artifact_obj["ManifestSha256"], f"{artifact_path}.ManifestSha256")
    ensure_unique(artifact_ids, f"{path}.Artifacts", "ArtifactId")

    environment = require_object(obj["Environment"], f"{path}.Environment")
    require_keys(environment, {"Platform", "PythonVersion", "Dependencies"}, set(), f"{path}.Environment")
    require_string(environment["Platform"], f"{path}.Environment.Platform")
    require_string(environment["PythonVersion"], f"{path}.Environment.PythonVersion")
    dependencies = require_object(environment["Dependencies"], f"{path}.Environment.Dependencies")
    for name, version in dependencies.items():
        require_string(name, f"{path}.Environment.Dependencies key")
        require_string(version, f"{path}.Environment.Dependencies.{name}")

    diagnostics = obj["DiagnosticsRelativePath"]
    if diagnostics is not None:
        require_relative_path(diagnostics, f"{path}.DiagnosticsRelativePath")
    if "ResumeFromArtifactId" in obj and obj["ResumeFromArtifactId"] is not None:
        require_string(obj["ResumeFromArtifactId"], f"{path}.ResumeFromArtifactId")
    if "Extensions" in obj:
        validate_extensions(obj["Extensions"], f"{path}.Extensions")

    if status == "PENDING":
        if started is not None or completed is not None:
            fail("RUN_STATE", path, "PENDING Run cannot have start or completion time")
    elif status == "RUNNING":
        if started is None or completed is not None:
            fail("RUN_STATE", path, "RUNNING Run requires start and no completion")
    elif status == "SUCCEEDED":
        if started is None or completed is None:
            fail("RUN_STATE", path, "SUCCEEDED Run requires start and completion")
        if stage_names != list(STAGES):
            fail("SUCCEEDED_STAGE_SET", f"{path}.Stages", "SUCCEEDED Run requires S0-S7")
        if any(stage["Status"] != "SUCCEEDED" for stage in stages):
            fail("SUCCEEDED_STAGE_STATUS", f"{path}.Stages", "all stages must succeed")
        if any(module["ExecutionRequirement"] == "REQUIRED" and module["Status"] != "SUCCEEDED" for module in modules):
            fail("SUCCEEDED_MODULE_STATUS", f"{path}.ResolvedModules", "all required Modules must succeed")
        if not artifacts:
            fail("SUCCEEDED_ARTIFACTS", f"{path}.Artifacts", "at least one Artifact required")
    elif status == "FAILED":
        if started is None or completed is None or not has_failed_stage:
            fail("RUN_STATE", path, "FAILED Run requires times and a failed Stage")
        if diagnostics is None:
            fail("FAILED_DIAGNOSTICS", f"{path}.DiagnosticsRelativePath", "required for failed Run")
    else:
        if started is None or completed is None or diagnostics is None:
            fail("RUN_STATE", path, f"{status} Run requires times and diagnostics")


def validate_manifest(value: Any) -> str:
    obj = require_object(value, "Manifest")
    contract_type = obj.get("ContractType")
    if contract_type == "U002_MODULE_MANIFEST":
        validate_module_manifest(obj)
        return "MODULE"
    if contract_type == "U002_ARTIFACT_MANIFEST":
        validate_artifact_manifest(obj)
        return "ARTIFACT"
    if contract_type == "U002_RUN_MANIFEST":
        validate_run_manifest(obj)
        return "RUN"
    fail("CONTRACT_TYPE", "Manifest.ContractType", "unknown U002 contract")
    raise AssertionError("unreachable")


def validate_schema_bundle(bundle: Any) -> None:
    obj = require_object(bundle, "SchemaBundle")
    if obj.get("$id") != BUNDLE_ID:
        fail("BUNDLE_ID", "SchemaBundle.$id", BUNDLE_ID)
    if obj.get("x-BundleVersion") != CONTRACT_VERSION:
        fail("BUNDLE_VERSION", "SchemaBundle.x-BundleVersion", CONTRACT_VERSION)
    definitions = require_object(obj.get("$defs"), "SchemaBundle.$defs")
    for name in ("moduleManifest", "artifactManifest", "runManifest"):
        if name not in definitions:
            fail("BUNDLE_DEFINITION", f"SchemaBundle.$defs.{name}", "required")


def canonical_bytes(value: Any) -> bytes:
    return json.dumps(value, sort_keys=True, separators=(",", ":"), ensure_ascii=False).encode("utf-8")


def canonical_sha256(value: Any) -> str:
    return hashlib.sha256(canonical_bytes(value)).hexdigest()

def expect_invalid(value: Any, expected_code: str) -> bool:
    try:
        validate_manifest(value)
    except ContractError as error:
        return error.code == expected_code
    return False


def run_self_test(bundle: Any) -> None:
    checks: list[tuple[str, Callable[[], bool]]] = []

    checks.append(("bundle identity", lambda: (validate_schema_bundle(bundle) is None)))
    checks.append(("bundle definitions", lambda: all(name in bundle["$defs"] for name in ("moduleManifest", "artifactManifest", "runManifest"))))

    module = valid_module_fixture()
    checks.append(("valid Module", lambda: (validate_module_manifest(module) is None)))
    missing_module = copy.deepcopy(module); missing_module.pop("ModuleId")
    checks.append(("missing Module field", lambda: expect_invalid(missing_module, "MISSING_FIELD")))
    bad_stage = copy.deepcopy(module); bad_stage["Stage"] = "S8"
    checks.append(("bad Stage", lambda: expect_invalid(bad_stage, "ENUM")))
    bad_hash = copy.deepcopy(module); bad_hash["ImplementationSha256"] = "bad"
    checks.append(("bad Module hash", lambda: expect_invalid(bad_hash, "SHA256")))
    traversal = copy.deepcopy(module); traversal["ConfigurationSchema"]["RelativePath"] = "../escape.json"
    checks.append(("Module path traversal", lambda: expect_invalid(traversal, "PARENT_TRAVERSAL")))
    duplicate_port = copy.deepcopy(module); duplicate_port["OptionalInputs"] = copy.deepcopy(duplicate_port["RequiredInputs"])
    checks.append(("duplicate PortId", lambda: expect_invalid(duplicate_port, "DUPLICATE")))
    bad_failure = copy.deepcopy(module); bad_failure["FailurePolicy"] = "SKIP_OPTIONAL_PATH"
    checks.append(("required failure policy", lambda: expect_invalid(bad_failure, "FAILURE_POLICY")))

    artifact = valid_artifact_fixture()
    checks.append(("valid Artifact", lambda: (validate_artifact_manifest(artifact) is None)))
    bad_parent = copy.deepcopy(artifact); bad_parent["Parents"][0]["ManifestSha256"] = "0"
    checks.append(("bad parent identity", lambda: expect_invalid(bad_parent, "SHA256")))
    bad_payload_path = copy.deepcopy(artifact); bad_payload_path["Payloads"][0]["RelativePath"] = "C:\\escape.csv"
    checks.append(("Artifact absolute path", lambda: expect_invalid(bad_payload_path, "ABSOLUTE_PATH")))
    no_payload = copy.deepcopy(artifact); no_payload["Payloads"] = []
    checks.append(("complete without payload", lambda: expect_invalid(no_payload, "COMPLETE_WITHOUT_PAYLOAD")))
    bad_range = copy.deepcopy(artifact); bad_range["Scope"]["FrameStart"] = 200; bad_range["Scope"]["FrameEnd"] = 100
    checks.append(("bad Frame range", lambda: expect_invalid(bad_range, "FRAME_RANGE")))
    bad_clock = copy.deepcopy(artifact); bad_clock["ClockDomain"]["FrameRateHz"] = None
    checks.append(("bad Clock domain", lambda: expect_invalid(bad_clock, "TYPE_NUMBER")))

    run = valid_run_fixture()
    checks.append(("valid succeeded Run", lambda: (validate_run_manifest(run) is None)))
    incomplete_stages = copy.deepcopy(run); incomplete_stages["Stages"] = incomplete_stages["Stages"][:-1]
    checks.append(("incomplete succeeded Run", lambda: expect_invalid(incomplete_stages, "SUCCEEDED_STAGE_SET")))
    incomplete_time = copy.deepcopy(run); incomplete_time["CompletedAtUtc"] = None
    checks.append(("succeeded Run missing completion", lambda: expect_invalid(incomplete_time, "RUN_STATE")))
    duplicate_order = copy.deepcopy(run); duplicate_order["ResolvedModules"][1]["Order"] = 0
    checks.append(("duplicate Module order", lambda: expect_invalid(duplicate_order, "DUPLICATE")))
    bad_artifact_path = copy.deepcopy(run); bad_artifact_path["Artifacts"][0]["ManifestRelativePath"] = "../artifact.json"
    checks.append(("Run Artifact path traversal", lambda: expect_invalid(bad_artifact_path, "PARENT_TRAVERSAL")))
    failed_run = valid_failed_run_fixture()
    checks.append(("valid failed Run record", lambda: (validate_run_manifest(failed_run) is None)))
    failed_no_diag = copy.deepcopy(failed_run); failed_no_diag["DiagnosticsRelativePath"] = None
    checks.append(("failed Run missing diagnostics", lambda: expect_invalid(failed_no_diag, "FAILED_DIAGNOSTICS")))
    unknown = {"ContractType": "UNKNOWN", "ContractVersion": "0.1"}
    checks.append(("unknown contract", lambda: expect_invalid(unknown, "CONTRACT_TYPE")))
    checks.append(("canonical hash deterministic", lambda: canonical_sha256(run) == canonical_sha256(json.loads(json.dumps(run)))))

    if len(checks) != 24:
        raise AssertionError(f"self-test definition count {len(checks)} != 24")
    failures = []
    for index, (name, check) in enumerate(checks, start=1):
        try:
            if not check():
                failures.append(f"{index}:{name}")
        except Exception as error:  # diagnostic boundary for self-test only
            failures.append(f"{index}:{name}:{type(error).__name__}:{error}")
    if failures:
        print("UDP002-U20A-SELFTEST-FAIL-" + "|".join(failures) + " C-7")
        raise SystemExit(7)
    print("UDP002-U20A-SELFTEST-PASS-24 C-0")


def write_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + "\n", encoding="utf-8")

def load_json(path: Path) -> Any:
    try:
        return json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, json.JSONDecodeError) as error:
        fail("JSON_READ", str(path), str(error))
    raise AssertionError("unreachable")


def validate_directory(directory: Path) -> int:
    files = sorted(directory.glob("*.json"))
    if not files:
        print("UDP002U20A1-ERROR-NO-FIXTURES C-7")
        return 7
    valid_expected = 0
    invalid_expected = 0
    valid_passed = 0
    invalid_rejected = 0
    unexpected: list[str] = []
    for path in files:
        should_pass = path.name.startswith("valid_")
        if should_pass:
            valid_expected += 1
        else:
            invalid_expected += 1
        try:
            validate_manifest(load_json(path))
            passed = True
        except ContractError as error:
            passed = False
            diagnostic = error.compact()
        if should_pass and passed:
            valid_passed += 1
        elif not should_pass and not passed:
            invalid_rejected += 1
        else:
            unexpected.append(path.name + (" unexpectedly passed" if passed else " " + diagnostic))
    code = 0 if not unexpected and valid_passed == valid_expected and invalid_rejected == invalid_expected else 7
    print(
        f"UDP002U20A1 N-B1-F{len(files)}-VP{valid_passed}-VE{valid_expected}"
        f"-IR{invalid_rejected}-IE{invalid_expected}-X{len(unexpected)} C-{code}"
    )
    for item in unexpected[:3]:
        print("MEMO-" + item)
    return code

    parser = argparse.ArgumentParser(description="Validate U002 Core Contract manifests")
    parser.add_argument("--schema-bundle", required=True)
    action = parser.add_mutually_exclusive_group(required=True)
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--emit-fixtures")
    action.add_argument("--validate", nargs="+")
    action.add_argument("--validate-dir")
    args = parser.parse_args()

    try:
        bundle = load_json(Path(args.schema_bundle))
        validate_schema_bundle(bundle)
        if args.self_test:
            run_self_test(bundle)
            return 0
        if args.emit_fixtures:
            emit_fixtures(Path(args.emit_fixtures))
            return 0
        if args.validate:
            for raw_path in args.validate:
                path = Path(raw_path)
                contract = validate_manifest(load_json(path))
                print(f"UDP002-U20A-VALID-{contract}-{path.name} C-0")
            return 0
        if args.validate_dir:
            return validate_directory(Path(args.validate_dir))
    except ContractError as error:
        print(f"UDP002-U20A-ERROR-{error.compact()} C-7")
        return 7
    return 7


if __name__ == "__main__":
    raise SystemExit(main())
