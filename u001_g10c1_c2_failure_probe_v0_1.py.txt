#!/usr/bin/env python3
"""Read-only C2 evidence probe for G10C1_SMOKE_10000_V0_2."""

from __future__ import annotations

import re
from pathlib import Path


RUN_ID = "G10C1_SMOKE_10000_V0_2"
CASE_ID = "TC001_20260422"
CONTROLLER = "u001_discovery_controller_v0_1b.py"
EXPECTED = (
    "u001_discovery_signal_inventory.csv",
    "u001_g2_selected_signals.csv",
    "u001_g4_signal_policy.csv",
    "u001_g5_frame_policy.csv",
    "u001_g6_hold_policy.csv",
    "u001_discovery_batch_map.csv",
    "u001_discovery_profile_manifest.json",
    "udp001_g10a_return_capsule.txt",
)


def clean(value: object, limit: int = 360) -> str:
    text = re.sub(r"[\r\n\t]+", " ", str(value).strip())
    text = re.sub(r"\s+", "_", text)
    text = re.sub(r"[^A-Za-z0-9_.:@+\\/=-]", "?", text)
    return text[:limit] if text else "EMPTY"


def lines(path: Path) -> list[str]:
    if not path.is_file():
        return []
    return [x.strip() for x in path.read_text(encoding="utf-8", errors="replace").splitlines() if x.strip()]


def one(root: Path, basename: str) -> Path | None:
    found = sorted(root.rglob(basename)) if root.exists() else []
    return next((p for p in found if p.is_file()), None)


def main() -> int:
    udp001 = Path(__file__).resolve().parent.parent
    runs = udp001 / "03_CASES" / CASE_ID / "02_RUNS"
    temp = runs / f".{RUN_ID}.g10c1_incomplete"
    profile = temp / "02_PROFILE_CONFIG"
    response = temp / "04_RESPONSE"
    controller = udp001 / "04_TEMP" / CONTROLLER

    out_bits = "".join("1" if (profile / name).is_file() and (profile / name).stat().st_size > 0 else "0" for name in EXPECTED)
    direct = sorted(p for p in profile.iterdir() if p.is_file()) if profile.is_dir() else []
    unexpected = sorted(p.name for p in direct if p.name not in EXPECTED)
    missing = [str(i + 1) for i, name in enumerate(EXPECTED) if not (profile / name).is_file() or (profile / name).stat().st_size == 0]

    source = controller.read_text(encoding="utf-8", errors="replace") if controller.is_file() else ""
    src_bits = "".join("1" if name in source else "0" for name in EXPECTED)
    profile_token_count = source.count("02_PROFILE_CONFIG")

    stdout = one(response, "C2.stdout.txt") or one(temp, "C2.stdout.txt")
    stderr = one(response, "C2.stderr.txt") or one(temp, "C2.stderr.txt")
    abort = one(response, "ABORTED_RUN.txt") or one(temp, "ABORTED_RUN.txt")
    capsule = profile / EXPECTED[-1]

    print(
        f"G10C1D3 N-T{int(temp.is_dir())}-P{int(profile.is_dir())}-R{int(response.is_dir())}"
        f"-O{len(direct)}-E{out_bits}-U{len(unexpected)}-M{'.'.join(missing) if missing else '0'}"
        f"-CS{int(controller.is_file())}-X0 C-0"
    )
    print(f"SRC-E{src_bits}-P{profile_token_count}")
    print("OUT-UN-" + clean(",".join(unexpected), 260))
    print("STDOUT-T" + clean("_|_".join(lines(stdout)[-3:]) if stdout else "MISSING"))
    print("STDERR-T" + clean("_|_".join(lines(stderr)[-3:]) if stderr else "MISSING"))
    print("CAP-T" + clean("_|_".join(lines(capsule)[-3:]) if capsule.is_file() else "MISSING"))
    print("ABORT-T" + clean("_|_".join(lines(abort)[-12:]) if abort else "MISSING", 520))
    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as exc:
        print(f"G10C1D3-ERROR:{clean(type(exc).__name__, 40)}:{clean(exc, 140)} C-7")
        raise SystemExit(7)
