#!/usr/bin/env python3
"""Install and qualify S1 Camera continuity Module version 0.2.1.

This governed installer upgrades only one immutable Module package.  It does
not mutate version 0.2.0, Core, Run configurations, case data or prior Runs.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile
from typing import Dict, Optional, Sequence, Tuple
import zipfile


PROGRAM_ID = "UDP002-U22B7T-S1-NULLABLE-AGE-MODULE-INSTALLER"
PROGRAM_VERSION = "0.1"
MODULE_ID = "CameraRadar.S1.CameraTargetContinuityArbitration"
OLD_VERSION = "0.2.0"
NEW_VERSION = "0.2.1"
PACKAGE_NAME = "u002_s1_camera_target_continuity_arbitration_v0_2_1.zip"
PACKAGE_SHA256 = "43b4eb1e1c4e7e38aa23093b1f74ad9985cad2629e0971f7e3c4e76c2e10c2de"
PACKAGE_FILES = (
    "algorithm_card.json",
    "configuration_schema.json",
    "implementation.py",
    "module_manifest.json",
    "selftest.py",
)
HISTORICAL_FAILURE_COVERAGE = ("FP028", "FP029", "FP030")
HISTORICAL_FAILURE_TEST_MAP = {
    "FP028": "installer_self_test_requires_historical_coverage",
    "FP029": "installer_self_test_requires_complete_rule_to_test_map",
    "FP030": "module_selftest_nullable_age_state_matrix",
}


class InstallerError(RuntimeError):
    """Represent one deterministic, bounded installation failure."""


def sha256_file(path: Path) -> str:
    """Return the SHA-256 identity of one regular file."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while True:
            block = stream.read(1024 * 1024)
            if not block:
                break
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    """Reject missing files and symbolic links."""

    return path.is_file() and not path.is_symlink()


def write_json_atomic(path: Path, value: Dict[str, object]) -> None:
    """Write one deterministic evidence document without a partial state."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    if temporary.exists():
        temporary.unlink()
    temporary.write_text(
        json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n",
        encoding="utf-8",
    )
    os.replace(temporary, path)


def resolve_workspace(value: str) -> Path:
    """Require the fixed U002 workspace identity."""

    root = Path(value).resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise InstallerError("WORKSPACE_ROOT")
    return root


def resolve_package(program_path: Path, explicit: Optional[str]) -> Path:
    """Resolve the adjacent governed package unless explicitly overridden."""

    package = Path(explicit).resolve() if explicit else program_path.parent / PACKAGE_NAME
    if not regular_file(package):
        raise InstallerError("PACKAGE_MISSING")
    if sha256_file(package) != PACKAGE_SHA256:
        raise InstallerError("PACKAGE_SHA256")
    return package


def read_package(package: Path) -> Dict[str, bytes]:
    """Validate archive membership and return the exact five-file package."""

    try:
        with zipfile.ZipFile(package, "r") as archive:
            names = archive.namelist()
            if sorted(names) != sorted(PACKAGE_FILES):
                raise InstallerError("PACKAGE_MEMBER_SET")
            if any(Path(name).name != name for name in names):
                raise InstallerError("PACKAGE_MEMBER_PATH")
            files = {name: archive.read(name) for name in PACKAGE_FILES}
    except zipfile.BadZipFile as exc:
        raise InstallerError("PACKAGE_ZIP") from exc

    try:
        manifest = json.loads(files["module_manifest.json"].decode("utf-8"))
        card = json.loads(files["algorithm_card.json"].decode("utf-8"))
        json.loads(files["configuration_schema.json"].decode("utf-8"))
    except Exception as exc:
        raise InstallerError("PACKAGE_DOCUMENT") from exc
    if manifest.get("ModuleId") != MODULE_ID:
        raise InstallerError("PACKAGE_MODULE_ID")
    if manifest.get("ModuleVersion") != NEW_VERSION:
        raise InstallerError("PACKAGE_MODULE_VERSION")
    if card.get("AlgorithmId") != MODULE_ID:
        raise InstallerError("PACKAGE_ALGORITHM_ID")
    if card.get("AlgorithmVersion") != NEW_VERSION:
        raise InstallerError("PACKAGE_ALGORITHM_VERSION")
    try:
        compile(files["implementation.py"], "implementation.py", "exec")
        compile(files["selftest.py"], "selftest.py", "exec")
    except Exception as exc:
        raise InstallerError("PACKAGE_PYTHON_COMPILE") from exc
    return files


def run_command(command: Sequence[str], reason: str, timeout: int = 120) -> str:
    """Run one governed child and require an explicit successful capsule."""

    completed = subprocess.run(
        list(command),
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )
    stdout = completed.stdout.strip()
    stderr = completed.stderr.strip()
    if completed.returncode != 0 or "C-0" not in stdout:
        detail = (stdout + " | " + stderr).strip()[-320:]
        raise InstallerError(reason + "__" + detail)
    return stdout.splitlines()[-1][:240]


def materialize_package(destination: Path, files: Dict[str, bytes]) -> bool:
    """Install immutably and return True only for a new installation."""

    if destination.exists():
        if not destination.is_dir() or destination.is_symlink():
            raise InstallerError("DESTINATION_TYPE")
        actual = {path.name for path in destination.iterdir() if path.is_file()}
        if actual != set(PACKAGE_FILES):
            raise InstallerError("DESTINATION_MEMBER_SET")
        for name, expected in files.items():
            path = destination / name
            if not regular_file(path) or path.read_bytes() != expected:
                raise InstallerError("DESTINATION_CONFLICT_" + name.upper())
        return False

    temporary = destination.with_name(destination.name + ".installing")
    if temporary.exists():
        raise InstallerError("INSTALLING_DIRECTORY_EXISTS")
    temporary.mkdir(parents=True)
    try:
        for name, content in files.items():
            (temporary / name).write_bytes(content)
        os.replace(temporary, destination)
    except Exception:
        shutil.rmtree(temporary, ignore_errors=True)
        raise
    return True


def validate_predecessor(root: Path) -> None:
    """Require the known immutable predecessor before applying the correction."""

    path = (
        root
        / "01_PROGRAMS/modules/S1"
        / MODULE_ID
        / OLD_VERSION
        / "module_manifest.json"
    )
    if not regular_file(path):
        raise InstallerError("PREDECESSOR_MISSING")
    try:
        document = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise InstallerError("PREDECESSOR_MANIFEST") from exc
    if document.get("ModuleId") != MODULE_ID or document.get("ModuleVersion") != OLD_VERSION:
        raise InstallerError("PREDECESSOR_IDENTITY")


def core_path(root: Path, name: str) -> Path:
    """Resolve one required accepted Core program."""

    path = root / "01_PROGRAMS/core" / name
    if not regular_file(path):
        raise InstallerError("CORE_MISSING_" + name.upper())
    return path


def install_and_qualify(root: Path, package: Path) -> Tuple[str, Dict[str, object]]:
    """Install the new Module, execute its tests and refresh governed control state."""

    validate_predecessor(root)
    files = read_package(package)
    destination = root / "01_PROGRAMS/modules/S1" / MODULE_ID / NEW_VERSION
    installed = materialize_package(destination, files)

    module_selftest = run_command(
        (
            sys.executable,
            str(destination / "selftest.py"),
            "--module-root",
            str(destination),
            "--workspace-root",
            str(root),
        ),
        "MODULE_SELFTEST",
    )
    if "N-T44-F18-IO9-X0 C-0" not in module_selftest:
        raise InstallerError("MODULE_SELFTEST_CAPSULE")

    registry = core_path(root, "u002_u22_module_registry_v0_1.py")
    qualification = core_path(root, "u002_u22_module_qualification_v0_1.py")
    registry_capsule = run_command(
        (sys.executable, str(registry), "--refresh", "--workspace-root", str(root)),
        "REGISTRY_REFRESH",
    )
    qualification_capsule = run_command(
        (
            sys.executable,
            str(qualification),
            "--qualify-module",
            MODULE_ID + "@" + NEW_VERSION,
            "--workspace-root",
            str(root),
        ),
        "MODULE_QUALIFICATION",
    )
    availability_capsule = run_command(
        (
            sys.executable,
            str(qualification),
            "--refresh-availability",
            "--workspace-root",
            str(root),
        ),
        "AVAILABILITY_REFRESH",
    )
    audit_capsule = run_command(
        (
            sys.executable,
            str(qualification),
            "--audit-module",
            MODULE_ID + "@" + NEW_VERSION,
            "--workspace-root",
            str(root),
        ),
        "MODULE_AUDIT",
    )

    receipt = (
        root
        / "00_CONTROL/registry/qualifications"
        / MODULE_ID
        / NEW_VERSION
        / "qualification_receipt.json"
    )
    if not regular_file(receipt):
        raise InstallerError("QUALIFICATION_RECEIPT_MISSING")

    evidence: Dict[str, object] = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "Status": "PASS",
        "ModuleId": MODULE_ID,
        "OldVersionPreserved": OLD_VERSION,
        "InstalledVersion": NEW_VERSION,
        "NewInstallation": installed,
        "PackageSha256": sha256_file(package),
        "InstalledFileSha256": {
            name: sha256_file(destination / name) for name in PACKAGE_FILES
        },
        "ModuleSelfTest": module_selftest,
        "RegistryRefresh": registry_capsule,
        "Qualification": qualification_capsule,
        "AvailabilityRefresh": availability_capsule,
        "QualificationAudit": audit_capsule,
        "QualificationReceiptSha256": sha256_file(receipt),
        "HistoricalFailureCoverage": list(HISTORICAL_FAILURE_COVERAGE),
        "HistoricalFailureTestMap": HISTORICAL_FAILURE_TEST_MAP,
        "SemanticCorrection": {
            "InvalidAndBlankAge": "ALLOWED_AND_INELIGIBLE",
            "InvalidAndNumericAge": "ALLOWED_AND_INELIGIBLE",
            "ValidAndBlankAge": "FAIL_CLOSED",
            "NonblankInvalidAge": "FAIL_CLOSED",
        },
    }
    evidence_path = (
        root
        / "07_OUTPUT/U22B7T_S1_MODULE_V021_INSTALL"
        / "u22b7t_module_install_evidence_v0_1.json"
    )
    write_json_atomic(evidence_path, evidence)
    capsule = (
        "B7T N-I1-S44-Q1-A1-V21-R1-X0 C-0"
        if installed
        else "B7T N-I0-S44-Q1-A1-V21-R1-X0 C-0"
    )
    return capsule, evidence


def self_test(package: Path) -> str:
    """Validate the delivery and execute the packaged Module in isolation."""

    files = read_package(package)
    if set(HISTORICAL_FAILURE_COVERAGE) != set(HISTORICAL_FAILURE_TEST_MAP):
        raise InstallerError("HISTORICAL_FAILURE_TEST_MAP")
    with tempfile.TemporaryDirectory(prefix="u22b7t_installer_selftest_") as temporary:
        root = Path(temporary) / "U-UDP002"
        module_root = root / "01_PROGRAMS/modules/S1" / MODULE_ID / NEW_VERSION
        module_root.mkdir(parents=True)
        for name, content in files.items():
            (module_root / name).write_bytes(content)
        capsule = run_command(
            (
                sys.executable,
                str(module_root / "selftest.py"),
                "--module-root",
                str(module_root),
                "--workspace-root",
                str(root),
            ),
            "DELIVERY_MODULE_SELFTEST",
        )
        if "N-T44-F18-IO9-X0 C-0" not in capsule:
            raise InstallerError("DELIVERY_SELFTEST_CAPSULE")
    return "B7TST N-Z1-F5-S44-H3-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Define explicit self-test and installation modes."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--install", action="store_true")
    parser.add_argument("--workspace-root")
    parser.add_argument("--package-zip")
    return parser.parse_args(argv)


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run the bounded installer workflow and expose one short capsule."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    program_path = Path(__file__).resolve()
    root: Optional[Path] = None
    try:
        package = resolve_package(program_path, args.package_zip)
        if args.self_test:
            print(self_test(package))
            return 0
        if args.workspace_root is None:
            raise InstallerError("WORKSPACE_ROOT_REQUIRED")
        root = resolve_workspace(args.workspace_root)
        capsule, _ = install_and_qualify(root, package)
        print(capsule)
        return 0
    except InstallerError as exc:
        reason = str(exc).replace(" ", "_")[:200]
        if root is not None:
            failure_path = (
                root
                / "07_OUTPUT/U22B7T_S1_MODULE_V021_INSTALL"
                / "u22b7t_module_install_failure_v0_1.json"
            )
            write_json_atomic(
                failure_path,
                {
                    "ProgramId": PROGRAM_ID,
                    "ProgramVersion": PROGRAM_VERSION,
                    "Status": "FAIL",
                    "FailureReason": str(exc),
                },
            )
        print("B7T N-I0-S0-Q0-A0-V21-R0-X1 C-7 K-" + reason)
        return 7
    except Exception as exc:
        reason = exc.__class__.__name__.upper()
        print("B7T N-I0-S0-Q0-A0-V21-R0-X1 C-7 K-" + reason)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
