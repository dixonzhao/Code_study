#!/usr/bin/env python3
"""Repair only a missing empty 03_CASES directory, then finalize through the Gate."""

from __future__ import annotations

import argparse
import subprocess
import sys
import tempfile
from pathlib import Path


PROGRAM_NAME = "u001_g10d2cd5d2_empty_directory_repair_finalizer_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2CD5D2-SELFTEST-PASS-20"
PACKAGE_NAME = "U001_Portable_v0_2"
GATE_NAME = "u001_g10d2c_portable_release_acceptance_gate_v0_1.py"
GATE_PASS = "G10D2CCG1-PASS P12-C4-H12-Z1-R1-X0 C-0"
INCOMPLETE_MARKER = ".G10D2C_INCOMPLETE"


def package_root(out: Path) -> Path:
    return out / PACKAGE_NAME


def portable_root(out: Path) -> Path:
    return package_root(out) / "UDP001"


def cases_root(out: Path) -> Path:
    return portable_root(out) / "03_CASES"


def safe_token(text: str, limit: int = 70) -> str:
    value = "".join(ch if ch.isalnum() or ch in "._-" else "_" for ch in text)
    return value[:limit] or "EMPTY"


def entries(path: Path) -> list[Path]:
    if not path.is_dir():
        return []
    return sorted(path.iterdir(), key=lambda item: item.name.lower())


def emit_memos(items: list[Path]) -> None:
    for index, item in enumerate(items[:3], 1):
        kind = "D" if item.is_dir() else "F"
        size = item.stat().st_size if item.is_file() else 0
        print(f"M{index}-{safe_token(item.name)}-{kind}-B{size}")


def gate_reason(stdout: str, stderr: str, rc: int) -> str:
    for line in (stdout + "\n" + stderr).splitlines():
        if "G10D2CCG1-FAIL:" in line:
            return safe_token(line.split("G10D2CCG1-FAIL:", 1)[1].split(" C-7", 1)[0])
    return f"GATE_RC{rc}"


def run_gate(gate: Path, out: Path, temp_root: Path) -> tuple[int, str, str]:
    result = subprocess.run(
        [sys.executable, str(gate.resolve()), "--release-root", str(out.resolve())],
        cwd=str(temp_root),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=2100,
        shell=False,
    )
    (temp_root / "G10D2CD5D2_GATE_STDOUT.txt").write_text(result.stdout, encoding="utf-8")
    (temp_root / "G10D2CD5D2_GATE_STDERR.txt").write_text(result.stderr, encoding="utf-8")
    return result.returncode, result.stdout, result.stderr


def finalize(udp001_root: Path, out: Path) -> int:
    root = udp001_root.resolve()
    temp_root = root / "04_TEMP"
    out = out.resolve()
    gate = temp_root / GATE_NAME
    marker = out / INCOMPLETE_MARKER
    cases = cases_root(out)
    release_ok = int(
        root.is_dir() and temp_root.is_dir() and out.is_dir()
        and package_root(out).is_dir() and portable_root(out).is_dir()
        and gate.is_file() and marker.is_file()
    )
    if not release_ok:
        print("G10D2CD5D2 N-R0-C0-E0-G0-P0-H0-Z0-M0-X1 K-PREFLIGHT C-7")
        return 7
    created = 0
    if cases.exists() and not cases.is_dir():
        print("G10D2CD5D2 N-R1-C0-E0-G0-P0-H0-Z0-M0-X0 K-CASES_NOT_DIRECTORY C-7")
        return 7
    if cases.is_dir():
        before = entries(cases)
        if before:
            print(f"G10D2CD5D2 N-R1-C0-E0-G0-P0-H0-Z0-M0-X0 K-PRE_CASES_{len(before)} C-7")
            emit_memos(before)
            return 7
    else:
        try:
            cases.mkdir(parents=False, exist_ok=False)
            created = 1
        except OSError as exc:
            print(f"G10D2CD5D2 N-R1-C0-E0-G0-P0-H0-Z0-M0-X1 K-{type(exc).__name__} C-7")
            return 7
    empty_before = int(cases.is_dir() and not entries(cases))
    if not empty_before:
        print(f"G10D2CD5D2 N-R1-C{created}-E0-G0-P0-H0-Z0-M0-X0 K-REPAIR_VERIFY C-7")
        return 7
    try:
        rc, stdout, stderr = run_gate(gate, out, temp_root)
    except subprocess.TimeoutExpired:
        print(f"G10D2CD5D2 N-R1-C{created}-E1-G0-P0-H0-Z1-M0-X0 K-GATE_TIMEOUT C-7")
        return 7
    except Exception as exc:
        print(f"G10D2CD5D2 N-R1-C{created}-E1-G0-P0-H0-Z0-M0-X1 K-{type(exc).__name__} C-7")
        return 7
    gate_ok = int(rc == 0 and GATE_PASS in stdout)
    hosted = 12 if gate_ok else 0
    after = entries(cases) if cases.is_dir() else []
    post_clean = int(cases.is_dir() and not after)
    if not gate_ok:
        print(
            f"G10D2CD5D2 N-R1-C{created}-E1-G0-P{post_clean}-H{hosted}-Z1-M0-X0 "
            f"K-{gate_reason(stdout, stderr, rc)} C-7"
        )
        emit_memos(after)
        return 7
    if not post_clean:
        print(f"G10D2CD5D2 N-R1-C{created}-E1-G1-P0-H12-Z1-M0-X0 K-POST_CASES_{len(after)} C-7")
        emit_memos(after)
        return 7
    try:
        marker.unlink()
    except OSError as exc:
        print(f"G10D2CD5D2 N-R1-C{created}-E1-G1-P1-H12-Z1-M0-X1 K-{type(exc).__name__} C-7")
        return 7
    print(f"G10D2CD5D2 N-R1-C{created}-E1-G1-P1-H12-Z1-M1-X0 C-0")
    return 0


def run_self_test() -> None:
    checks = 0
    with tempfile.TemporaryDirectory() as td:
        base = Path(td)
        out = base / "OUT"
        portable = portable_root(out)
        portable.mkdir(parents=True)
        cases = cases_root(out)
        checks += int(not cases.exists())
        cases.mkdir(parents=False)
        checks += int(cases.is_dir())
        checks += int(entries(cases) == [])
        item = cases / "x.txt"
        item.write_text("x", encoding="utf-8")
        checks += int(len(entries(cases)) == 1)
        checks += int(entries(cases)[0].name == "x.txt")
        checks += int(package_root(out).name == PACKAGE_NAME)
        checks += int(portable_root(out).name == "UDP001")
        checks += int(cases_root(out).name == "03_CASES")
        checks += int(safe_token("a/b c") == "a_b_c")
        checks += int(gate_reason("G10D2CCG1-FAIL:X C-7", "", 7) == "X")
        checks += int(gate_reason("", "", 9) == "GATE_RC9")
        checks += int(PROGRAM_VERSION == "0.1")
        checks += int(PROGRAM_NAME.endswith("v0_1.py"))
        checks += int(SELFTEST_TOKEN.endswith("PASS-20"))
        checks += int(PACKAGE_NAME.endswith("v0_2"))
        checks += int(GATE_NAME.endswith("v0_1.py"))
        checks += int(GATE_PASS.endswith("C-0"))
        checks += int(INCOMPLETE_MARKER.startswith("."))
        checks += int(len(GATE_PASS.split()) == 3)
        checks += int(sys.executable != "")
    if checks != 20:
        raise AssertionError(f"SELFTEST_COUNT:{checks}")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Repair missing empty 03_CASES and finalize release")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.udp001_root or not args.out:
        parser.error("--udp001-root and --out are required")
    return finalize(Path(args.udp001_root), Path(args.out))


if __name__ == "__main__":
    raise SystemExit(main())
