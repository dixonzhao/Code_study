class PlanError(RuntimeError):
    """Represent one deterministic B2C failure."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = token(reason, 60)
        self.detail = token(detail, 90)
        super().__init__(self.reason)


def token(value: object, limit: int = 80) -> str:
    text = re.sub(r"[^A-Z0-9_.:+-]+", "_", str(value).upper())
    return text.strip("_")[:limit] or "NA"


def canonical_bytes(value: object) -> bytes:
    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_json(path: Path, reason: str) -> Dict[str, object]:
    if not path.is_file() or path.is_symlink():
        raise PlanError(reason + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise PlanError(reason + "_JSON", path.name) from exc
    if not isinstance(value, dict):
        raise PlanError(reason + "_TYPE", path.name)
    return value


def write_exact(path: Path, value: object) -> None:
    payload = canonical_bytes(value)
    if path.exists():
        if not path.is_file() or path.is_symlink() or path.read_bytes() != payload:
            raise PlanError("EXISTING_FILE_CONFLICT", path.name)
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(payload)
    os.replace(temporary, path)


def require_workspace(path: Path) -> Path:
    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise PlanError("WORKSPACE_ROOT")
    required = (
        REGISTRY,
        QUALIFICATION,
        PREFLIGHT,
        FREEZE_AUDITOR,
        CAMERA_RECEIPT,
        NORMALIZED_CONFIG,
    )
    missing = [item.as_posix() for item in required if not (root / item).is_file()]
    if missing:
        raise PlanError("WORKSPACE_DEPENDENCY", missing[0])
    return root


def run_child(
    root: Path,
    relative_program: Path,
    arguments: Sequence[str],
    expected_prefix: str,
    timeout: int = 120,
) -> str:
    completed = subprocess.run(
        [sys.executable, str(root / relative_program), *arguments],
        cwd=root.parent,
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )
    lines = [line.strip() for line in completed.stdout.splitlines() if line.strip()]
    capsule = lines[-1] if lines else "MISSING"
    if completed.returncode != 0 or not capsule.startswith(expected_prefix) or not capsule.endswith("C-0"):
        detail = capsule if capsule != "MISSING" else completed.stderr[-160:]
        raise PlanError("CHILD_" + relative_program.stem, detail)
    return capsule


def freeze_audit(root: Path) -> str:
    return run_child(
        root,
        FREEZE_AUDITOR,
        ["--audit-freeze", "--workspace-root", str(root)],
        "U22A7F",
        timeout=90,
    )


def refresh_availability(root: Path) -> Dict[str, object]:
    run_child(
        root,
        QUALIFICATION,
        ["--refresh-availability", "--workspace-root", str(root)],
        "U22A4Q",
    )
    return load_json(root / AVAILABILITY, "AVAILABILITY")


def availability_entry(
    availability: Mapping[str, object],
    module_id: str,
    module_version: str,
) -> Mapping[str, object]:
    entries = availability.get("Entries")
    if not isinstance(entries, list):
        raise PlanError("AVAILABILITY_ENTRIES")
    matches = [
        item for item in entries
        if item.get("ModuleId") == module_id
        and item.get("ModuleVersion") == module_version
    ]
    if len(matches) != 1:
        raise PlanError("AVAILABILITY_COUNT", module_id)
    return matches[0]


def qualify_new_modules(root: Path) -> int:
    """Qualify once or audit an already qualified identical Module."""

    newly_qualified = 0
    for module_id, module_version in NEW_MODULES:
        availability = refresh_availability(root)
        entry = availability_entry(availability, module_id, module_version)
        identity = module_id + "@" + module_version
        if entry.get("RegistryStatus") == "AVAILABLE":
            run_child(
                root,
                QUALIFICATION,
                ["--audit-module", identity, "--workspace-root", str(root)],
                "U22A4Q",
            )
        else:
            run_child(
                root,
                QUALIFICATION,
                [
                    "--qualify-module", identity,
                    "--workspace-root", str(root),
                    "--timeout-seconds", "120",
                ],
                "U22A4Q",
            )
            newly_qualified += 1
    final = refresh_availability(root)
    for module_id, module_version in NEW_MODULES:
        entry = availability_entry(final, module_id, module_version)
        if entry.get("RegistryStatus") != "AVAILABLE":
            raise PlanError("MODULE_NOT_AVAILABLE", module_id)
    return newly_qualified


def discover_b1ar(root: Path) -> Tuple[Path, Dict[str, object]]:
    candidates: List[Tuple[int, int, Path, Dict[str, object]]] = []
    for path in (root / "07_OUTPUT").glob("*/" + B1AR_RECEIPT_NAME):
        try:
            receipt = load_json(path, "B1AR")
            if receipt.get("Status") != "REAL_U001_HANDOFF_ACCEPTED":
                continue
            frame = receipt.get("FrameDomain")
            if not isinstance(frame, dict):
                continue
            package = Path(str(receipt.get("SelectedPackage", ""))).resolve()
            if not package.is_dir() or package.is_symlink():
                continue
            rows = int(frame.get("FrameRows", 0))
            candidates.append((rows, path.stat().st_mtime_ns, path, receipt))
        except Exception:
            continue
    if not candidates:
        raise PlanError("B1AR_RECEIPT_NOT_FOUND")
    _, _, path, receipt = max(candidates, key=lambda item: (item[0], item[1], str(item[2])))
    return path, receipt


def accepted_s0(receipt_path: Path) -> Tuple[str, str]:
    plan = load_json(receipt_path.parent / "00_CONFIG/resolved_run_plan.json", "B1AR_PLAN")
    modules = plan.get("ModulePlan")
    if not isinstance(modules, list):
        raise PlanError("B1AR_MODULE_PLAN")
    matches = [
        item for item in modules
        if isinstance(item, dict)
        and item.get("Stage") == "S0"
        and "U001_INTAKE_ACCEPTED" in item.get("ProvidesArtifacts", [])
    ]
    if len(matches) != 1:
        raise PlanError("S0_IDENTITY_COUNT")
    return str(matches[0]["ModuleId"]), str(matches[0]["ModuleVersion"])


def discover_b1b4_normalization(root: Path) -> Dict[str, object]:
    """Find accepted provenance for the normalized executable configuration."""

    candidates: List[Tuple[int, Path, Dict[str, object]]] = []
    for path in (root / "07_OUTPUT").glob("*/" + B1B4_RUN_RECEIPT_NAME):
        try:
            receipt = load_json(path, "B1B4_RUN_RECEIPT")
            if receipt.get("Status") != "NORMALIZED_RETRY_RUN_COMPLETED":
                continue
            evidence = receipt.get("Normalization")
            if not isinstance(evidence, dict):
                continue
            if evidence.get("NormalizedConfiguration") != NORMALIZED_CONFIG.as_posix():
                continue
            candidates.append((path.stat().st_mtime_ns, path, receipt))
        except Exception:
            continue
    if not candidates:
        raise PlanError("B1B4_NORMALIZATION_RECEIPT_NOT_FOUND")
    return max(candidates, key=lambda item: (item[0], str(item[1])))[2]


def runtime_failure_reason(blobs: Sequence[str]) -> Optional[str]:
    """Extract an approved Radar MaxAge token from heterogeneous diagnostics."""

    allowed = {
        "INTEGER_RADAR_DISTANCE_MAX_AGE",
        "INTEGER_RADAR_DETECTION_FLAG_MAX_AGE",
    }
    patterns = (
        r"(?:^|\s)R[-_:]([A-Z0-9_.:+-]+)(?:\s|$)",
        r"(INTEGER_RADAR_(?:DISTANCE|DETECTION_FLAG)_MAX_AGE)",
    )
    for blob in blobs:
        upper = blob.upper()
        for pattern in patterns:
            for match in re.finditer(pattern, upper, flags=re.MULTILINE):
                candidate = match.group(1).strip("_.:+-")
                if candidate in allowed:
                    return candidate
    return None


def static_radar_contract_evidence(root: Path) -> List[str]:
    """Prove that accepted Radar bindings still contain non-executable blanks."""

    configuration = load_json(root / NORMALIZED_CONFIG, "STATIC_CONFIGURATION")
    profile_relative = Path(str(configuration.get("BindingProfilePath", "")))
    profile_path = root / profile_relative
    if sha256_file(profile_path) != configuration.get("BindingProfileSha256"):
        raise PlanError("STATIC_PROFILE_HASH")
    profile = load_json(profile_path, "STATIC_PROFILE")
    bindings = profile.get("Bindings")
    if not isinstance(bindings, dict):
        raise PlanError("STATIC_BINDINGS")
    blank_roles: List[str] = []
    for role in ("RADAR_DISTANCE", "RADAR_DETECTION_FLAG"):
        selected = bindings.get(role)
        if not isinstance(selected, list) or len(selected) != 1:
            raise PlanError("STATIC_BINDING_COUNT", role)
        binding = selected[0]
        if not isinstance(binding, dict):
            raise PlanError("STATIC_BINDING_TYPE", role)
        policy = str(binding.get("HoldPolicy", "")).strip().upper()
        age = str(
            binding.get("MaxAgeFrames")
            if binding.get("MaxAgeFrames") is not None
            else ""
        ).strip()
        if policy == "NONE" and age == "":
            blank_roles.append(role)
    if not blank_roles:
        raise PlanError("STATIC_RADAR_BLANK_NOT_FOUND")
    return blank_roles


def failed_radar_evidence(root: Path) -> Dict[str, object]:
    """Confirm that the protected source Run reached and failed Radar S1."""

    run_root = root / "07_OUTPUT" / FAILED_RUN_ID
    status = load_json(run_root / "run_status.json", "FAILED_RUN_STATUS")
    if status.get("Status") != "FAILED":
        raise PlanError("SOURCE_RUN_NOT_FAILED", status.get("Status"))
    records: List[Tuple[Path, Dict[str, object]]] = []
    for path in sorted((run_root / "01_LOGS").glob("*.json")):
        try:
            record = load_json(path, "SOURCE_EXECUTION_RECORD")
            if record.get("Status") == "FAILED":
                records.append((path, record))
        except Exception:
            continue
    if len(records) != 1:
        raise PlanError("SOURCE_FAILED_RECORD_COUNT", len(records))
    record_path, record = records[0]
    if int(record.get("Sequence", 0)) != 3:
        raise PlanError("SOURCE_FAILED_SEQUENCE", record.get("Sequence"))
    module_id = str(record.get("ModuleId", ""))
    if module_id != "CameraRadar.S1.RadarMeasurementValidity":
        raise PlanError("SOURCE_FAILED_MODULE", module_id)
    diagnostic_root = run_root / str(record.get("DiagnosticRoot", ""))
    evidence_paths = (
        diagnostic_root / "stdout.txt",
        diagnostic_root / "stderr.txt",
        diagnostic_root / "module_result.json",
        record_path,
    )
    blobs: List[str] = []
    hashes: Dict[str, Optional[str]] = {}
    for path in evidence_paths:
        if path.is_file() and not path.is_symlink():
            blobs.append(path.read_text(encoding="utf-8-sig", errors="replace"))
            hashes[path.name] = sha256_file(path)
        else:
            hashes[path.name] = None
    reason = runtime_failure_reason(blobs)
    if reason is None:
        blank_roles = static_radar_contract_evidence(root)
        reason = "STATIC_PROFILE_BLANK_RADAR_MAX_AGE"
        evidence_mode = "FAILED_RADAR_MODULE_PLUS_STATIC_CONTRACT"
    else:
        blank_roles = static_radar_contract_evidence(root)
        evidence_mode = "RUNTIME_TOKEN_PLUS_STATIC_CONTRACT"
    return {
        "RunId": FAILED_RUN_ID,
        "Sequence": 3,
        "ModuleId": module_id,
        "Reason": reason,
        "EvidenceMode": evidence_mode,
        "BlankRadarRoles": blank_roles,
        "ExecutionRecordSha256": sha256_file(record_path),
        "DiagnosticHashes": hashes,
    }


def compile_global_no_hold_profile(
    source: Mapping[str, object],
) -> Tuple[Dict[str, object], List[Dict[str, object]], List[str], int]:
    """Compile all selected bindings while preserving optional unbound roles.

    Binding states are handled exhaustively:
    - [] is a valid optional-unbound role and is preserved;
    - exactly one binding is compiled;
    - more than one binding is rejected as ambiguous;
    - NONE plus blank age becomes integer zero;
    - NONE plus a value must be integer zero;
    - another hold policy requires an explicit nonnegative integer age.
    """

    compiled = json.loads(json.dumps(source, ensure_ascii=False))
    bindings = compiled.get("Bindings")
    if not isinstance(bindings, dict):
        raise PlanError("COMPILED_BINDINGS")
    changes: List[Dict[str, object]] = []
    unbound_roles: List[str] = []
    selected_binding_count = 0
    for role in sorted(bindings):
        selected = bindings[role]
        if not isinstance(selected, list):
            raise PlanError("COMPILED_BINDING_COLLECTION", role)
        if len(selected) == 0:
            unbound_roles.append(role)
            continue
        if len(selected) > 1:
            raise PlanError("COMPILED_BINDING_AMBIGUOUS", role)
        for index, binding in enumerate(selected):
            if not isinstance(binding, dict):
                raise PlanError("COMPILED_BINDING_TYPE", role)
            selected_binding_count += 1
            policy = str(binding.get("HoldPolicy", "")).strip().upper()
            observed = binding.get("MaxAgeFrames")
            observed_text = str(observed if observed is not None else "").strip()
            if policy == "NONE" and observed_text == "":
                binding["MaxAgeFrames"] = 0
                changes.append(
                    {
                        "Role": role,
                        "BindingIndex": index,
                        "HoldPolicy": "NONE",
                        "ObservedMaxAgeFrames": "BLANK",
                        "CompiledMaxAgeFrames": 0,
                    }
                )
                continue
            if isinstance(observed, bool):
                raise PlanError("COMPILED_MAX_AGE", role)
            try:
                parsed = int(observed)
            except (TypeError, ValueError) as exc:
                raise PlanError("COMPILED_MAX_AGE", role) from exc
            if parsed < 0 or parsed != observed:
                raise PlanError("COMPILED_MAX_AGE", role)
            if policy == "NONE" and parsed != 0:
                raise PlanError("NONE_NONZERO_MAX_AGE", role)
            binding["MaxAgeFrames"] = parsed
    required_roles = (
        "CAMERA_1_DISTANCE",
        "CAMERA_3_DISTANCE",
        "RADAR_DISTANCE",
        "RADAR_DETECTION_FLAG",
    )
    for role in required_roles:
        selected = bindings.get(role)
        if not isinstance(selected, list) or len(selected) != 1:
            raise PlanError("CURRENT_PLAN_REQUIRED_BINDING", role)
    compiled["BindingProfileVersion"] = "0.3"
    return compiled, changes, unbound_roles, selected_binding_count


def accepted_camera(
    root: Path,
) -> Tuple[str, str, Dict[str, object], Dict[str, object]]:
    """Compile the accepted B1B4 profile into one executable global profile."""

    module_receipt = load_json(root / CAMERA_RECEIPT, "CAMERA_RECEIPT")
    if module_receipt.get("Status") != "S1_MODULE_INSTALLED_AND_QUALIFIED":
        raise PlanError("CAMERA_RECEIPT_STATUS")
    normalization_receipt = discover_b1b4_normalization(root)
    evidence = normalization_receipt["Normalization"]
    configuration_path = root / NORMALIZED_CONFIG
    if sha256_file(configuration_path) != evidence.get("NormalizedConfigurationSha256"):
        raise PlanError("NORMALIZED_CONFIGURATION_HASH")
    configuration = load_json(configuration_path, "NORMALIZED_CONFIGURATION")
    profile_relative = Path(str(configuration.get("BindingProfilePath", "")))
    profile_path = root / profile_relative
    if sha256_file(profile_path) != configuration.get("BindingProfileSha256"):
        raise PlanError("NORMALIZED_PROFILE_HASH")
    source_profile = load_json(profile_path, "NORMALIZED_PROFILE")
    (
        compiled_profile,
        changes,
        unbound_roles,
        selected_binding_count,
    ) = compile_global_no_hold_profile(source_profile)
    write_exact(root / COMPILED_PROFILE, compiled_profile)
    compiled_profile_sha = sha256_file(root / COMPILED_PROFILE)
    configuration = dict(configuration)
    configuration["BindingProfilePath"] = COMPILED_PROFILE.as_posix()
    configuration["BindingProfileSha256"] = compiled_profile_sha
    write_exact(root / COMPILED_CONFIG, configuration)
    evidence = {
        "Rule": "HOLD_POLICY_NONE_AND_BLANK_MAX_AGE_TO_INTEGER_ZERO",
        "SourceProfile": profile_relative.as_posix(),
        "SourceProfileSha256": sha256_file(profile_path),
        "CompiledProfile": COMPILED_PROFILE.as_posix(),
        "CompiledProfileSha256": compiled_profile_sha,
        "CompiledConfiguration": COMPILED_CONFIG.as_posix(),
        "CompiledConfigurationSha256": sha256_file(root / COMPILED_CONFIG),
        "ChangedBindingCount": len(changes),
        "SelectedBindingCount": selected_binding_count,
        "UnboundOptionalRoles": unbound_roles,
        "Changes": changes,
    }
    return (
        str(module_receipt["ModuleId"]),
        str(module_receipt["ModuleVersion"]),
        configuration,
        evidence,
    )


def validate_frame_range(receipt: Mapping[str, object]) -> None:
    frame = receipt.get("FrameDomain")
    if not isinstance(frame, dict):
        raise PlanError("FRAME_DOMAIN")
    minimum = int(frame.get("FrameMin"))
    maximum = int(frame.get("FrameMax"))
    if FRAME_START < minimum or FRAME_END > maximum:
        raise PlanError("FRAME_RANGE_OUTSIDE_SOURCE", f"{minimum}.{maximum}")

def prepare_short_plan(root: Path) -> Tuple[int, int, int, str]:
    failure = failed_radar_evidence(root)
    receipt_path, receipt = discover_b1ar(root)
    validate_frame_range(receipt)
    package = Path(str(receipt["SelectedPackage"])).resolve()
    s0 = accepted_s0(receipt_path)
    camera = accepted_camera(root)
    configuration = build_configuration(package, s0, camera[:3])
    config_path = root / "02_CONFIG/run_configs" / (RUN_ID + ".json")
    write_exact(config_path, configuration)
    capsule = run_child(
        root,
        PREFLIGHT,
        ["--prepare-run", str(config_path), "--workspace-root", str(root)],
        "U22A5E",
    )
    match = re.search(r"-R([01])-P1-X0 C-0$", capsule)
    if match is None:
        raise PlanError("PREFLIGHT_CAPSULE", capsule)
    replay = int(match.group(1))
    plan = load_json(root / "07_OUTPUT" / RUN_ID / "00_CONFIG/resolved_run_plan.json", "RESOLVED_PLAN")
    modules = plan.get("ModulePlan")
    if not isinstance(modules, list) or len(modules) != 5:
        raise PlanError("RESOLVED_MODULE_COUNT")
    if [item.get("Stage") for item in modules] != ["S0", "S1", "S1", "S2", "S3"]:
        raise PlanError("RESOLVED_STAGE_ORDER")
    receipt = {
        "ReceiptContract": "UDP002_U22B2DR3_GLOBAL_NO_HOLD_COMPILATION_V0_1",
        "RunId": RUN_ID,
        "SourceFailure": failure,
        "Compilation": camera[3],
        "ModuleCodeModified": False,
        "AnalysisMathematicsModified": False,
    }
    write_exact(root / "07_OUTPUT" / RUN_ID / COMPILATION_RECEIPT, receipt)
    return replay, len(modules), int(camera[3]["ChangedBindingCount"]), str(failure["Reason"])


def self_test() -> str:
    checks = []
    def ck(value: bool) -> None:
        if not value:

def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--qualify-and-prepare", action="store_true")
    parser.add_argument("--workspace-root")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            print(self_test())
            return 0
        if not args.workspace_root:
            raise PlanError("WORKSPACE_ROOT_REQUIRED")
        root = require_workspace(Path(args.workspace_root))
        freeze_before = freeze_audit(root)
        qualified = qualify_new_modules(root)
        replay, module_count, changed, reason = prepare_short_plan(root)
        freeze_after = freeze_audit(root)
        if freeze_before != freeze_after:
            raise PlanError("FOUNDATION_CHANGED")
        print(
            f"B2DR3P N-Q{qualified}-M{module_count}-N{changed}"
            f"-K{token(reason, 36)}-R{replay}-X0 C-0"
        )
        return 0
    except PlanError as exc:
        print(f"B2DR3PE K-{exc.reason}-D-{exc.detail}-X1 C-7")
        return 7
    except Exception as exc:
        print(f"B2DR3PE K-{token(exc.__class__.__name__)}-D-{token(exc)}-X1 C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
