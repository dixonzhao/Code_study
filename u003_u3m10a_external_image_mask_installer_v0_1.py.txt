#!/usr/bin/env python3
"""Prepare, self-test and promote U3M10A FrameMask v0.5.0."""

from __future__ import annotations

import argparse
import csv
import hashlib
import shutil
import subprocess
import sys
import zipfile
from datetime import datetime
from pathlib import Path


PACKAGE_NAME = "UDP003_U3M10A_EXTERNAL_IMAGE_MASK_v0_1.zip"
GUIDE_NAME = "U3M10A_START_HERE_CN_v0_1_2026-08-28.txt"
PREPARED_NAME = "U3M10A_PREPARED"
MODULE_ID = "CameraRadar.U003.FrameMask"
MODULE_VERSION = "0.5.0"
GATE_NAME = "u003_u3m10a_external_image_mask_acceptance_gate_v0_1.py"
REPORT_NAME = "U3M10A_node_report_external_image_redaction_v0_1_2026-08-28.txt"
MODULE_SELFTEST = "U3M10A-MODULE-SELFTEST-PASS-28 C-0"


class InstallError(RuntimeError):
    """Raised when transfer, cleanup, validation or promotion is incomplete."""


def require(condition: bool, message: str) -> None:
    if not condition:
        raise InstallError(message)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def discover_root(explicit: Path | None) -> Path:
    current = Path.cwd().resolve()
    candidates = [item for item in (explicit, current, current / "U-UDP003") if item is not None]
    candidates.extend(parent / "U-UDP003" for parent in (current, *current.parents[:3]))
    for candidate in candidates:
        if candidate.name == "U-UDP003" and candidate.is_dir():
            return candidate.resolve()
    raise InstallError("U-UDP003 root was not found. Run from C-CODE.")


def unique_archive(root: Path, label: str) -> Path:
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    candidate = root / "A-Archive" / f"{label}_{stamp}"
    index = 2
    while candidate.exists():
        candidate = root / "A-Archive" / f"{label}_{stamp}_{index:02d}"
        index += 1
    return candidate


def archive_path(path: Path, root: Path, label: str) -> None:
    if not path.exists():
        return
    destination = unique_archive(root, label) / path.name
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(destination))


def converge_temp(root: Path, installer: Path) -> dict[str, int]:
    temporary = root / "04_TEMP"
    temporary.mkdir(parents=True, exist_ok=True)
    retained_names = {PACKAGE_NAME, GUIDE_NAME, installer.name}
    package_path = installer.parent / PACKAGE_NAME
    guide_path = installer.parent / GUIDE_NAME
    require(installer.parent.resolve() == temporary.resolve(), "Installer must be in U-UDP003/04_TEMP.")
    require(package_path.is_file(), f"Transfer ZIP is missing: {PACKAGE_NAME}")
    require(guide_path.is_file(), f"START_HERE is missing: {GUIDE_NAME}")
    before = list(temporary.iterdir())
    old = [item for item in before if item.name not in retained_names]
    if old:
        destination_root = unique_archive(root, "U3M10A_PREPARE") / "04_TEMP_PREVIOUS"
        for item in old:
            destination = destination_root / item.name
            destination.parent.mkdir(parents=True, exist_ok=True)
            shutil.move(str(item), str(destination))
    unexpected = [item for item in temporary.iterdir() if item.name not in retained_names]
    require(not unexpected, "Unexpected TEMP artifact remains after convergence.")
    return {"Before": len(before), "Archived": len(old), "Retained": len(retained_names)}


def verify_package(package_root: Path) -> int:
    manifest_path = package_root / "PACKAGE_SHA256.csv"
    require(manifest_path.is_file(), "PACKAGE_SHA256.csv is missing.")
    with manifest_path.open("r", encoding="utf-8", newline="") as stream:
        rows = list(csv.DictReader(stream))
    expected: set[str] = set()
    for row in rows:
        relative = str(row.get("RelativePath", "")).replace("\\", "/")
        digest = str(row.get("SHA256", "")).lower()
        artifact = package_root / relative
        require(relative and digest, "PACKAGE_SHA256 contains an incomplete row.")
        require(artifact.is_file(), f"Package artifact is missing: {relative}")
        require(sha256_file(artifact) == digest, f"Package hash differs: {relative}")
        expected.add(relative)
    actual = {
        path.relative_to(package_root).as_posix()
        for path in package_root.rglob("*")
        if path.is_file() and path.name != "PACKAGE_SHA256.csv"
    }
    require(actual == expected, "Package inventory differs from PACKAGE_SHA256.csv.")
    return len(rows)


def module_at(base: Path) -> Path:
    return base / "02_MODULES" / "EVIDENCE_VIEW" / MODULE_ID / MODULE_VERSION


def foundation(root: Path) -> tuple[Path, ...]:
    return (
        root / "01_PROGRAMS" / "u003_module_runtime_v0_2.py",
        root / "01_PROGRAMS" / "u003_module_executor_v0_3.py",
        root / "01_PROGRAMS" / "u003_module_action_gui_v0_2.py",
        root / "01_PROGRAMS" / "u003_gui_shell_v0_8.py",
        root / "02_MODULES" / "EVIDENCE_VIEW" / MODULE_ID / "0.4.0" / "module_manifest.json",
    )


def prepare(root: Path, installer: Path) -> int:
    print(f"[CHECK] U003 root: {root}")
    print("[WHY] Installation must use the exact company U-UDP003 topology.")
    package_path = installer.parent / PACKAGE_NAME
    guide_path = installer.parent / GUIDE_NAME
    print(f"[CHECK] Installer SHA256: {sha256_file(installer)}")
    print(f"[CHECK] ZIP SHA256: {sha256_file(package_path) if package_path.is_file() else 'MISSING'}")
    print(f"[CHECK] START_HERE SHA256: {sha256_file(guide_path) if guide_path.is_file() else 'MISSING'}")
    print("[WHY] These identities are printed before old TEMP content is archived.")
    cleanup = converge_temp(root, installer)
    print(f"[TEMP] before={cleanup['Before']} archived={cleanup['Archived']} retained={cleanup['Retained']}")
    print("[WHY] Old TEMP files are archived so stale versions cannot be selected accidentally.")

    prepared = root / "04_TEMP" / PREPARED_NAME
    require(not prepared.exists(), "Prepared directory unexpectedly exists after cleanup.")
    prepared.mkdir(parents=True)
    with zipfile.ZipFile(package_path, "r") as archive:
        archive.extractall(prepared)
    count = verify_package(prepared)
    for path in foundation(root):
        print(f"[CHECK] Installed dependency: {path.relative_to(root)}")
        require(path.is_file(), f"Installed dependency is missing: {path.name}")
    print("[WHY] v0.5.0 extends the accepted v0.4.0 Module and existing action runtime.")
    for path in (
        module_at(prepared) / "implementation.py",
        module_at(prepared) / "module_manifest.json",
        module_at(prepared) / "algorithm_card.json",
        prepared / "01_PROGRAMS" / GATE_NAME,
        prepared / "03_DOCUMENTS" / REPORT_NAME,
    ):
        print(f"[CHECK] Prepared artifact: {path.relative_to(prepared)}")
        require(path.is_file(), f"Prepared artifact is missing: {path.name}")
    print(f"U3M10AP N-B{cleanup['Before']}-A{cleanup['Archived']}-F{count}-D5-X0 C-0")
    return 0


def run_child(command: list[str]) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        command,
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=180,
        check=False,
    )


def self_test(root: Path) -> int:
    prepared = root / "04_TEMP" / PREPARED_NAME
    implementation = module_at(prepared) / "implementation.py"
    require(implementation.is_file(), "Prepared release is missing. Run --prepare first.")
    result = run_child([sys.executable, str(implementation), "--self-test"])
    if result.stdout:
        print(result.stdout.strip())
    if result.stderr:
        print(result.stderr.strip())
    require(result.returncode == 0, "FrameMask v0.5.0 self-test failed.")
    require(MODULE_SELFTEST in result.stdout, "Expected Module self-test capsule is missing.")
    return 0


def execute(root: Path) -> int:
    prepared = root / "04_TEMP" / PREPARED_NAME
    module = module_at(prepared)
    gate = prepared / "01_PROGRAMS" / GATE_NAME
    report = prepared / "03_DOCUMENTS" / REPORT_NAME
    require(module.is_dir() and gate.is_file() and report.is_file(), "Prepared release is incomplete.")
    gate_result = run_child(
        [
            sys.executable,
            str(gate),
            "--root",
            str(root),
            "--module-dir",
            str(module),
        ]
    )
    if gate_result.stdout:
        print(gate_result.stdout.strip())
    if gate_result.stderr:
        print(gate_result.stderr.strip())
    require(gate_result.returncode == 0, "Independent U3M10A gate failed.")
    require("U3M10AG " in gate_result.stdout and " C-0" in gate_result.stdout, "Gate capsule is missing.")

    destination = module_at(root)
    archive_path(destination, root, "U3M10A_REPLACED_MODULE")
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copytree(module, destination)
    gate_destination = root / "01_PROGRAMS" / GATE_NAME
    report_destination = root / "03_DOCUMENTS" / REPORT_NAME
    archive_path(gate_destination, root, "U3M10A_REPLACED_GATE")
    archive_path(report_destination, root, "U3M10A_REPLACED_REPORT")
    shutil.copy2(gate, gate_destination)
    report_destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copy2(report, report_destination)
    for name in ("implementation.py", "module_manifest.json", "algorithm_card.json"):
        require(sha256_file(module / name) == sha256_file(destination / name), f"Promoted hash differs: {name}")
    print("U3M10AI N-M1-A2-G1-R1-X0 C-0")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--root", type=Path)
    arguments = parser.parse_args()
    root = discover_root(arguments.root)
    installer = Path(__file__).resolve()
    if arguments.prepare:
        return prepare(root, installer)
    if arguments.self_test:
        return self_test(root)
    return execute(root)


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as error:
        print(f"U3M10AIE K-{type(error).__name__.upper()} D-{error} X1 C-7")
        raise SystemExit(7)
