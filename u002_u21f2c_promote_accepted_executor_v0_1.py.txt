class ControlledError(Exception):
    def __init__(self, reason: str):
        super().__init__(reason)
        self.reason = reason


def duplicate_reject_object(pairs: list[tuple[str, Any]]) -> dict[str, Any]:
    result: dict[str, Any] = {}
    for key, value in pairs:
        if key in result:
            raise ControlledError("JSON")
        result[key] = value
    return result


def reject_nonfinite(value: str) -> None:
    raise ControlledError("JSON")


def read_json_strict(path: Path) -> dict[str, Any]:
    try:
        value = json.loads(
            path.read_text(encoding="utf-8-sig"),
            object_pairs_hook=duplicate_reject_object,
            parse_constant=reject_nonfinite,
        )
    except ControlledError:
        raise
    except Exception as exc:
        raise ControlledError("JSON") from exc
    if not isinstance(value, dict):
        raise ControlledError("JSON")
    return value


def require_regular_file(path: Path, expected_name: str | None = None) -> None:
    if expected_name is not None and path.name != expected_name:
        raise ControlledError("BASENAME")
    if path.is_symlink() or not path.is_file():
        raise ControlledError("DEPENDENCY_FILE")


def sha256_file(path: Path) -> str:
    require_regular_file(path)
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def collect_string_leaves(value: Any) -> list[str]:
    result: list[str] = []
    if isinstance(value, str):
        result.append(value)
    elif isinstance(value, dict):
        for child in value.values():
            result.extend(collect_string_leaves(child))
    elif isinstance(value, list):
        for child in value:
            result.extend(collect_string_leaves(child))
    return result


def read_exact_capsule(path: Path, expected: str) -> None:
    try:
        lines = [line.strip() for line in path.read_text(encoding="utf-8-sig").splitlines() if line.strip()]
    except Exception as exc:
        raise ControlledError("CAPSULE") from exc
    if lines != [expected]:
        raise ControlledError("CAPSULE")


def atomic_write_text(path: Path, text: str) -> None:
    temp = path.with_name(path.name + ".tmp")
    if temp.exists():
        raise ControlledError("OUTPUT_EXISTS")
    try:
        with temp.open("x", encoding="utf-8", newline="\n") as stream:
            stream.write(text)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temp, path)
    except ControlledError:
        raise
    except Exception as exc:
        temp.unlink(missing_ok=True)
        raise ControlledError("EVIDENCE") from exc


def install_byte_identical(source: Path, destination: Path) -> bool:
    source_hash = sha256_file(source)
    if destination.exists():
        if destination.is_symlink() or not destination.is_file():
            raise ControlledError("DESTINATION")
        if sha256_file(destination) != source_hash:
            raise ControlledError("DESTINATION")
        return False

    temp = destination.with_name(destination.name + ".promotion-tmp")
    if temp.exists():
        raise ControlledError("OUTPUT_EXISTS")
    try:
        with source.open("rb") as src, temp.open("xb") as dst:
            shutil.copyfileobj(src, dst, 1024 * 1024)
            dst.flush()
            os.fsync(dst.fileno())
        if sha256_file(temp) != source_hash:
            raise ControlledError("IDENTITY")
        os.replace(temp, destination)
    except ControlledError:
        temp.unlink(missing_ok=True)
        raise
    except Exception as exc:
        temp.unlink(missing_ok=True)
        raise ControlledError("DESTINATION") from exc
    if sha256_file(destination) != source_hash:
        raise ControlledError("IDENTITY")
    return True


def make_read_only(path: Path) -> None:
    try:
        path.chmod(stat.S_IRUSR | stat.S_IRGRP | stat.S_IROTH)
    except OSError as exc:
        raise ControlledError("DESTINATION") from exc


def terminate_tree(process: subprocess.Popen[bytes]) -> None:
    if process.poll() is not None:
        return
    try:
        if os.name == "nt":
            subprocess.run(
                ["taskkill", "/PID", str(process.pid), "/T", "/F"],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
                timeout=5,
                check=False,
            )
        else:
            os.killpg(process.pid, signal.SIGTERM)
    except Exception:
        try:
            process.terminate()
        except Exception:
            pass
    try:
        process.wait(timeout=3)
        return
    except subprocess.TimeoutExpired:
        pass
    try:
        if os.name != "nt":
            os.killpg(process.pid, signal.SIGKILL)
        else:
            process.kill()
    except Exception:
        try:
            process.kill()
        except Exception:
            pass
    try:
        process.wait(timeout=3)
    except Exception:
        pass


def run_exact(argv: list[str], expected_line: str, timeout_seconds: float) -> int:
    if not math.isfinite(timeout_seconds) or timeout_seconds <= 0:
        raise ControlledError("CHILD_TIMEOUT")
    creationflags = 0
    popen_kwargs: dict[str, Any] = {}
    if os.name == "nt" and hasattr(subprocess, "CREATE_NEW_PROCESS_GROUP"):
        creationflags = subprocess.CREATE_NEW_PROCESS_GROUP
    elif os.name != "nt":
        popen_kwargs["start_new_session"] = True
    start = time.monotonic()
    process = subprocess.Popen(
        argv,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        creationflags=creationflags,
        **popen_kwargs,
    )
    try:
        stdout, stderr = process.communicate(timeout=timeout_seconds)
    except subprocess.TimeoutExpired as exc:
        terminate_tree(process)
        raise ControlledError("CHILD_TIMEOUT") from exc
    elapsed_ms = int((time.monotonic() - start) * 1000)
    stdout_text = stdout.decode("utf-8", errors="replace")
    stderr_text = stderr.decode("utf-8", errors="replace")
    final_lines = [line.strip() for line in stdout_text.splitlines() if line.strip()]
    final_line = final_lines[-1] if final_lines else ""
    if process.returncode != 0:
        raise ControlledError("CHILD_RETURN")
    if "Traceback" in stderr_text or "Traceback" in stdout_text:
        raise ControlledError("TRACEBACK")
    if final_line != expected_line:
        raise ControlledError("CAPSULE")
    return elapsed_ms


def dependency_argv(runtime: Path, paths: dict[str, Path], action: str) -> list[str]:
    return [
        sys.executable,
        str(runtime),
        action,
        "--engine",
        str(paths["engine"]),
        "--protocol-bundle",
        str(paths["protocol_bundle"]),
        "--protocol-validator",
        str(paths["protocol_validator"]),
        "--protocol-promotion-evidence",
        str(paths["protocol_promotion_evidence"]),
        "--schema-bundle",
        str(paths["schema_bundle"]),
        "--contract-validator",
        str(paths["contract_validator"]),
    ]


def validate_inputs(
    gate_json: Path,
    gate_capsule: Path,
    source: Path,
    programs_dir: Path,
    evidence_dir: Path,
    dependencies: dict[str, Path],
) -> tuple[dict[str, Any], dict[str, str]]:
    require_regular_file(gate_json, GATE_JSON_NAME)
    require_regular_file(gate_capsule, GATE_CAPSULE_NAME)
    if gate_json.parent != gate_capsule.parent or gate_json.parent.name != GATE_DIR_NAME:
        raise ControlledError("GATE_EVIDENCE")
    require_regular_file(source, SOURCE_NAME)
    for role, expected in EXPECTED_DEPENDENCY_NAMES.items():
        require_regular_file(dependencies[role], expected)
    if programs_dir.is_symlink() or not programs_dir.is_dir():
        raise ControlledError("PROGRAMS_DIR")
    if evidence_dir.exists():
        raise ControlledError("OUTPUT_EXISTS")
    if not evidence_dir.parent.is_dir():
        raise ControlledError("OUTPUT_PARENT")

    read_exact_capsule(gate_capsule, EXPECTED_GATE_CAPSULE)
    gate_data = read_json_strict(gate_json)
    source_hash = sha256_file(source)
    if source_hash not in collect_string_leaves(gate_data):
        raise ControlledError("GATE_HASH")

    inputs = {
        "gate_json": gate_json,
        "gate_capsule": gate_capsule,
        "source_executor": source,
        **dependencies,
    }
    hashes = {role: sha256_file(path) for role, path in inputs.items()}
    if len(hashes) != 9:
        raise ControlledError("INTERNAL")
    return gate_data, hashes


def promote_impl(args: argparse.Namespace) -> str:
    gate_json = Path(args.gate_json).resolve()
    gate_capsule = Path(args.gate_capsule).resolve()
    source = Path(args.source_executor).resolve()
    programs_dir = Path(args.programs_dir).resolve()
    evidence_dir = Path(args.evidence_dir).resolve()
    dependencies = {
        "engine": Path(args.engine).resolve(),
        "protocol_bundle": Path(args.protocol_bundle).resolve(),
        "protocol_validator": Path(args.protocol_validator).resolve(),
        "protocol_promotion_evidence": Path(args.protocol_promotion_evidence).resolve(),
        "schema_bundle": Path(args.schema_bundle).resolve(),
        "contract_validator": Path(args.contract_validator).resolve(),
    }
    _, before = validate_inputs(
        gate_json, gate_capsule, source, programs_dir, evidence_dir, dependencies
    )

    runtime = programs_dir / RUNTIME_NAME
    stage_runtime = programs_dir / ("." + RUNTIME_NAME + ".promotion-stage")
    installed_now = not runtime.exists()
    if runtime.exists() and sha256_file(runtime) != before["source_executor"]:
        raise ControlledError("DESTINATION")
    if stage_runtime.exists():
        raise ControlledError("OUTPUT_EXISTS")

    if installed_now:
        install_byte_identical(source, stage_runtime)
        try:
            run_exact(
                dependency_argv(stage_runtime, dependencies, "--dependency-preflight"),
                EXPECTED_DEPENDENCY_CAPSULE,
                45.0,
            )
            run_exact(
                dependency_argv(stage_runtime, dependencies, "--self-test"),
                EXPECTED_EXECUTOR_CAPSULE,
                120.0,
            )
            if sha256_file(stage_runtime) != before["source_executor"]:
                raise ControlledError("IDENTITY")
            os.replace(stage_runtime, runtime)
        except Exception:
            stage_runtime.unlink(missing_ok=True)
            raise

    make_read_only(runtime)
    try:
        dependency_ms = run_exact(
            dependency_argv(runtime, dependencies, "--dependency-preflight"),
            EXPECTED_DEPENDENCY_CAPSULE,
            45.0,
        )
        selftest_ms = run_exact(
            dependency_argv(runtime, dependencies, "--self-test"),
            EXPECTED_EXECUTOR_CAPSULE,
            120.0,
        )
    except Exception:
        if installed_now:
            try:
                runtime.chmod(stat.S_IWUSR | stat.S_IRUSR)
                runtime.unlink()
            except Exception:
                pass
        raise

    all_inputs = {
        "gate_json": gate_json,
        "gate_capsule": gate_capsule,
        "source_executor": source,
        **dependencies,
    }
    after = {role: sha256_file(path) for role, path in all_inputs.items()}
    if after != before:
        raise ControlledError("IDENTITY")
    if sha256_file(runtime) != before["source_executor"]:
        raise ControlledError("IDENTITY")

    staging = evidence_dir.with_name("." + evidence_dir.name + ".staging")
    if staging.exists():
        raise ControlledError("OUTPUT_EXISTS")
    staging.mkdir()
    try:
        evidence = {
            "CandidateHashMembership": True,
            "CreatedUTC": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
            "DependencyPreflightCapsule": EXPECTED_DEPENDENCY_CAPSULE,
            "DependencyPreflightElapsedMs": dependency_ms,
            "GateCapsule": EXPECTED_GATE_CAPSULE,
            "GateEvidence": {
                "CapsuleName": GATE_CAPSULE_NAME,
                "CapsuleSHA256": before["gate_capsule"],
                "JsonName": GATE_JSON_NAME,
                "JsonSHA256": before["gate_json"],
            },
            "InputImmutabilityCount": 9,
            "InstalledNow": installed_now,
            "RuntimeName": RUNTIME_NAME,
            "RuntimeSHA256": before["source_executor"],
            "SelfTestCapsule": EXPECTED_EXECUTOR_CAPSULE,
            "SelfTestElapsedMs": selftest_ms,
            "SourceName": SOURCE_NAME,
            "SourceSHA256": before["source_executor"],
            "Status": "PASS",
            "ToolVersion": TOOL_VERSION,
        }
        atomic_write_text(
            staging / EVIDENCE_JSON_NAME,
            json.dumps(evidence, ensure_ascii=True, sort_keys=True, separators=(",", ":")) + "\n",
        )
        atomic_write_text(staging / EVIDENCE_CAPSULE_NAME, PROMOTION_CAPSULE + "\n")
        os.replace(staging, evidence_dir)
    except Exception:
        if staging.exists():
            shutil.rmtree(staging, ignore_errors=True)
        raise
    return PROMOTION_CAPSULE


def write_stub_executor(path: Path) -> None:
    text = """from __future__ import annotations
import sys
if '--dependency-preflight' in sys.argv:
    print('U21F2BDP N-E1-P2-C2-H6-X0 C-0')
    raise SystemExit(0)
if '--self-test' in sys.argv:
    print('U21F2B1 N-T32-R4-M12-A9-F8-I8-X0 C-0')
    raise SystemExit(0)
print('U21F2B1 N-T0-R0-M0-A0-F0-I0-X1 C-7 K-CLI')
raise SystemExit(7)
"""
    path.write_text(text, encoding="utf-8")


def build_selftest_namespace(root: Path) -> argparse.Namespace:
    temp_dir = root / "04_TEMP"
    programs_dir = root / "01_PROGRAMS"
    gate_dir = root / GATE_DIR_NAME
    evidence_parent = root / "05_TESTS"
    for directory in (temp_dir, programs_dir, gate_dir, evidence_parent):
        directory.mkdir(parents=True, exist_ok=True)

    source = temp_dir / SOURCE_NAME
    write_stub_executor(source)
    dependency_paths: dict[str, Path] = {}
    for role, name in EXPECTED_DEPENDENCY_NAMES.items():
        parent = programs_dir if role in {"engine", "protocol_bundle", "protocol_validator"} else temp_dir
        if role == "protocol_promotion_evidence":
            parent = root / "F2A_PROMOTION"
            parent.mkdir(exist_ok=True)
        path = parent / name
        path.write_text("{}\n" if path.suffix in {".json", ".txt"} else "# fixture\n", encoding="utf-8")
        dependency_paths[role] = path

    gate_json = gate_dir / GATE_JSON_NAME
    gate_json.write_text(
        json.dumps({"CandidateSHA256": sha256_file(source), "Counts": {"R": 2, "P": 4, "D": 12, "T": 4, "I": 8, "S": 6, "X": 0}}, sort_keys=True) + "\n",
        encoding="utf-8",
    )
    gate_capsule = gate_dir / GATE_CAPSULE_NAME
    gate_capsule.write_text(EXPECTED_GATE_CAPSULE + "\n", encoding="utf-8")

    return argparse.Namespace(
        gate_json=str(gate_json),
        gate_capsule=str(gate_capsule),
        source_executor=str(source),
        programs_dir=str(programs_dir),
        evidence_dir=str(evidence_parent / "PROMOTION"),
        **{role: str(path) for role, path in dependency_paths.items()},
    )


def self_test() -> int:
    checks = 0

    def check(condition: bool) -> None:
        nonlocal checks
        if not condition:
            raise ControlledError("SELFTEST_ASSERTION")
        checks += 1

    try:
        with tempfile.TemporaryDirectory(prefix="u21f2cp-") as directory:
            root = Path(directory)
            args = build_selftest_namespace(root)
            capsule = promote_impl(args)
            check(capsule == PROMOTION_CAPSULE)
            runtime = Path(args.programs_dir) / RUNTIME_NAME
            check(runtime.is_file())
            check(sha256_file(runtime) == sha256_file(Path(args.source_executor)))
            check((Path(args.evidence_dir) / EVIDENCE_JSON_NAME).is_file())
            check((Path(args.evidence_dir) / EVIDENCE_CAPSULE_NAME).read_text(encoding="utf-8").strip() == PROMOTION_CAPSULE)

            evidence = read_json_strict(Path(args.evidence_dir) / EVIDENCE_JSON_NAME)
            check(evidence["Status"] == "PASS")
            check(evidence["CandidateHashMembership"] is True)
            check(evidence["InputImmutabilityCount"] == 9)

            bad_gate = root / "bad-gate.json"
            bad_gate.write_text('{"a":1,"a":2}', encoding="utf-8")
            try:
                read_json_strict(bad_gate)
                check(False)
            except ControlledError as exc:
                check(exc.reason == "JSON")

            check(sha256_file(Path(args.source_executor)) == evidence["SourceSHA256"])
            check(evidence["RuntimeSHA256"] == evidence["SourceSHA256"])
            check(evidence["GateCapsule"] == EXPECTED_GATE_CAPSULE)
            check(evidence["DependencyPreflightCapsule"] == EXPECTED_DEPENDENCY_CAPSULE)
            check(evidence["SelfTestCapsule"] == EXPECTED_EXECUTOR_CAPSULE)

            other = root / "other.py.txt"
            other.write_text("x", encoding="utf-8")
            try:
                require_regular_file(other, SOURCE_NAME)
                check(False)
            except ControlledError as exc:
                check(exc.reason == "BASENAME")

            existing_destination = root / "existing.py"
            existing_destination.write_text("wrong", encoding="utf-8")
            try:
                install_byte_identical(Path(args.source_executor), existing_destination)
                check(False)
            except ControlledError as exc:
                check(exc.reason == "DESTINATION")

            check(len(collect_string_leaves({"a": ["x", {"b": "y"}]})) == 2)
            check(checks == 17)
        print("U21F2CPST N-T18-P6-D6-I6-X0 C-0")
        return 0
    except ControlledError as exc:
        print(f"U21F2CPST N-T{checks}-P0-D0-I0-X1 C-7 K-{exc.reason}")
        return 7
    except Exception:
        print(f"U21F2CPST N-T{checks}-P0-D0-I0-X1 C-7 K-INTERNAL")
        return 7


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(allow_abbrev=False)
    actions = parser.add_mutually_exclusive_group(required=True)
    actions.add_argument("--self-test", action="store_true")
    actions.add_argument("--promote", action="store_true")
    parser.add_argument("--gate-json")
    parser.add_argument("--gate-capsule")
    parser.add_argument("--source-executor")
    parser.add_argument("--programs-dir")
    parser.add_argument("--evidence-dir")
    parser.add_argument("--engine")
    parser.add_argument("--protocol-bundle")
    parser.add_argument("--protocol-validator")
    parser.add_argument("--protocol-promotion-evidence")
    parser.add_argument("--schema-bundle")
    parser.add_argument("--contract-validator")
    return parser


def main() -> int:
    args = build_parser().parse_args()
    if args.self_test:
        return self_test()
    required = [
        "gate_json",
        "gate_capsule",
        "source_executor",
        "programs_dir",
        "evidence_dir",
        "engine",
        "protocol_bundle",
        "protocol_validator",
        "protocol_promotion_evidence",
        "schema_bundle",
        "contract_validator",
    ]
    if any(not getattr(args, name) for name in required):
        print("U21F2CP1 N-G0-H0-C0-D0-R0-I0-X1 C-7 K-CLI")
        return 7
    try:
        print(promote_impl(args))
        return 0
    except ControlledError as exc:
        print(f"U21F2CP1 N-G0-H0-C0-D0-R0-I0-X1 C-7 K-{exc.reason}")
        return 7
    except Exception:
        print("U21F2CP1 N-G0-H0-C0-D0-R0-I0-X1 C-7 K-INTERNAL")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
