#!/usr/bin/env python3
"""Archive obsolete U003 program versions and close the U003 04_TEMP workspace.

The cleaner is intentionally move-only. It never deletes a file and never
touches U003 case/output evidence. Dry-run is the default; ``--apply`` is
required before any path is moved.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import shutil
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Iterable


VERSION = "0.1"
ACTIVE_SOURCE_NAME = "u003_evidence_review_workbench_v0_8.py"
ACTIVE_SOURCE_SHA256 = "0b5ddf0c0f0717c7a092d97571cebf74e5f8366098e98e2ab07529a75f20fd1e"
ARCHIVE_BASENAME = "U003_FREEZE_V08_20260825"


class CleanupError(RuntimeError):
    """Fail-closed cleanup error."""


def canonical_source_sha256(path: Path) -> str:
    """Hash Python source after normalizing UTF-8 BOM and line endings."""

    text = path.read_text(encoding="utf-8-sig")
    canonical = text.replace("\r\n", "\n").replace("\r", "\n").encode("utf-8")
    return hashlib.sha256(canonical).hexdigest()


def file_sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while True:
            block = stream.read(1024 * 1024)
            if not block:
                break
            digest.update(block)
    return digest.hexdigest()


def is_obsolete_program_name(raw_name: str) -> bool:
    """Classify only known historical U003 executable/source filenames."""

    name = raw_name.lower()
    if name == ACTIVE_SOURCE_NAME.lower():
        return False
    if name.startswith("framecount_video_reviewer_v"):
        return True
    if name.startswith("u003_evidence_review_workbench_v"):
        return True
    if name.startswith("u003_") and "acceptance_gate" in name:
        return True
    if name.endswith(".exe") and ("u003" in name or "framecount" in name):
        return True
    return False


def is_obsolete_program(path: Path) -> bool:
    """Classify an existing file without treating dependency directories as old."""

    return path.is_file() and is_obsolete_program_name(path.name)


def unique_destination(path: Path) -> Path:
    """Return a collision-free destination without overwriting history."""

    if not path.exists():
        return path
    index = 2
    while True:
        candidate = path.with_name(f"{path.name}_{index:03d}")
        if not candidate.exists():
            return candidate
        index += 1


def choose_archive_root(archive_parent: Path) -> Path:
    base = archive_parent / ARCHIVE_BASENAME
    if not base.exists():
        return base
    index = 2
    while True:
        candidate = archive_parent / f"{ARCHIVE_BASENAME}_{index:03d}"
        if not candidate.exists():
            return candidate
        index += 1


def tree_summary(path: Path) -> dict[str, Any]:
    """Create a compact pre-move identity record for one file or directory."""

    if path.is_file():
        return {
            "kind": "file",
            "file_count": 1,
            "byte_count": path.stat().st_size,
            "sha256": file_sha256(path),
        }
    if path.is_dir():
        digest = hashlib.sha256()
        file_count = 0
        byte_count = 0
        for child in sorted((item for item in path.rglob("*") if item.is_file())):
            relative = child.relative_to(path).as_posix()
            child_hash = file_sha256(child)
            size = child.stat().st_size
            digest.update(relative.encode("utf-8"))
            digest.update(b"\0")
            digest.update(child_hash.encode("ascii"))
            digest.update(b"\0")
            file_count += 1
            byte_count += size
        return {
            "kind": "directory",
            "file_count": file_count,
            "byte_count": byte_count,
            "tree_sha256": digest.hexdigest(),
        }
    return {"kind": "missing", "file_count": 0, "byte_count": 0}


def plan_moves(workspace: Path, current_script: Path) -> dict[str, Any]:
    program_root = workspace / "01_PROGRAMS"
    temp_root = workspace / "04_TEMP"
    archive_parent = workspace / "A-Archive"
    active_source = program_root / ACTIVE_SOURCE_NAME

    if not workspace.is_dir():
        raise CleanupError("WORKSPACE_MISSING")
    if not program_root.is_dir():
        raise CleanupError("PROGRAM_ROOT_MISSING")
    if not temp_root.is_dir():
        raise CleanupError("TEMP_ROOT_MISSING")
    if not active_source.is_file():
        raise CleanupError("ACTIVE_SOURCE_MISSING")
    if canonical_source_sha256(active_source) != ACTIVE_SOURCE_SHA256:
        raise CleanupError("ACTIVE_SOURCE_IDENTITY")

    archive_root = choose_archive_root(archive_parent)
    program_moves: list[dict[str, Any]] = []
    for source in sorted(program_root.iterdir(), key=lambda item: item.name.lower()):
        if is_obsolete_program(source):
            destination = archive_root / "01_PROGRAMS_OBSOLETE" / source.name
            program_moves.append({
                "source": source,
                "destination": destination,
                "summary": tree_summary(source),
            })

    temp_moves: list[dict[str, Any]] = []
    for source in sorted(temp_root.iterdir(), key=lambda item: item.name.lower()):
        destination = archive_root / "04_TEMP_SNAPSHOT" / source.name
        temp_moves.append({
            "source": source,
            "destination": destination,
            "summary": tree_summary(source),
            "self_script": same_path(source, current_script),
        })

    untouched_program = [
        item.name
        for item in sorted(program_root.iterdir(), key=lambda entry: entry.name.lower())
        if not is_obsolete_program(item)
    ]
    return {
        "workspace": workspace,
        "program_root": program_root,
        "temp_root": temp_root,
        "archive_root": archive_root,
        "active_source": active_source,
        "program_moves": program_moves,
        "temp_moves": temp_moves,
        "untouched_program": untouched_program,
    }


def same_path(left: Path, right: Path) -> bool:
    try:
        return left.resolve() == right.resolve()
    except OSError:
        return os.path.abspath(str(left)) == os.path.abspath(str(right))


def serializable_plan(plan: dict[str, Any]) -> dict[str, Any]:
    def entry(record: dict[str, Any]) -> dict[str, Any]:
        return {
            "source": str(record["source"]),
            "destination": str(record["destination"]),
            "summary": record["summary"],
            "self_script": bool(record.get("self_script", False)),
        }

    return {
        "contract": "U003_FREEZE_V08_CLEANUP_PLAN_V0_1",
        "version": VERSION,
        "workspace": str(plan["workspace"]),
        "archive_root": str(plan["archive_root"]),
        "active_source": str(plan["active_source"]),
        "active_source_sha256": ACTIVE_SOURCE_SHA256,
        "program_moves": [entry(item) for item in plan["program_moves"]],
        "temp_moves": [entry(item) for item in plan["temp_moves"]],
        "untouched_program": list(plan["untouched_program"]),
        "protected_roots": ["03_CASES", "07_OUTPUT"],
        "deletion_performed": False,
    }


def move_one(source: Path, destination: Path) -> Path:
    destination.parent.mkdir(parents=True, exist_ok=True)
    target = unique_destination(destination)
    shutil.move(str(source), str(target))
    return target


def apply_plan(plan: dict[str, Any]) -> tuple[list[dict[str, Any]], list[str]]:
    archive_root: Path = plan["archive_root"]
    records_root = archive_root / "00_RECORDS"
    records_root.mkdir(parents=True, exist_ok=True)

    serialized = serializable_plan(plan)
    (records_root / "cleanup_plan.json").write_text(
        json.dumps(serialized, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )

    completed: list[dict[str, Any]] = []
    failures: list[str] = []
    regular_moves = list(plan["program_moves"]) + [
        item for item in plan["temp_moves"] if not item.get("self_script", False)
    ]
    self_moves = [item for item in plan["temp_moves"] if item.get("self_script", False)]

    for record in regular_moves + self_moves:
        source: Path = record["source"]
        destination: Path = record["destination"]
        try:
            moved_to = move_one(source, destination)
            completed.append({
                "source": str(source),
                "destination": str(moved_to),
                "summary": record["summary"],
            })
        except Exception as exc:
            failures.append(f"{source.name}:{type(exc).__name__}")

    result = {
        "contract": "U003_FREEZE_V08_CLEANUP_RESULT_V0_1",
        "version": VERSION,
        "timestamp_utc": datetime.now(timezone.utc).isoformat(),
        "status": "PASS" if not failures else "FAILED",
        "active_source": str(plan["active_source"]),
        "active_source_sha256": ACTIVE_SOURCE_SHA256,
        "completed_moves": completed,
        "failures": failures,
        "rollback_rule": "Move each destination back to its recorded source path.",
        "deletion_performed": False,
    }
    (records_root / "cleanup_result.json").write_text(
        json.dumps(result, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )
    return completed, failures


def capsule(plan: dict[str, Any], apply: bool, moved: int, failures: Iterable[str]) -> str:
    failure_list = list(failures)
    code = 0 if not failure_list else 7
    return (
        f"U003CLN N-A{int(apply)}-K1-P{len(plan['program_moves'])}-"
        f"T{len(plan['temp_moves'])}-M{moved}-"
        f"U{len(plan['untouched_program'])}-R1-X{len(failure_list)} C-{code}"
    )


def run_cleanup(workspace: Path, apply: bool) -> int:
    current_script = Path(__file__).resolve()
    try:
        plan = plan_moves(workspace.resolve(), current_script)
    except Exception as exc:
        line = f"U003CLN N-A{int(apply)}-K0-P0-T0-M0-U0-R0-X1 C-7"
        print(line)
        print(f"MEMO1-{type(exc).__name__}:{exc}")
        return 7

    if not apply:
        print(capsule(plan, False, 0, []))
        print(f"MEMO1-ARCHIVE-{plan['archive_root']}")
        return 0

    completed, failures = apply_plan(plan)
    line = capsule(plan, True, len(completed), failures)
    print(line)
    for index, failure in enumerate(failures[:3], start=1):
        print(f"MEMO{index}-{failure}")
    return 0 if not failures else 7


def run_self_test() -> int:
    checks = 0

    def ok(condition: bool) -> None:
        nonlocal checks
        assert condition
        checks += 1

    ok(is_obsolete_program_name("framecount_video_reviewer_v0_5.py"))
    ok(is_obsolete_program_name("framecount_video_reviewer_v0_6.py.txt"))
    ok(is_obsolete_program_name("u003_evidence_review_workbench_v0_7.py"))
    ok(not is_obsolete_program_name(ACTIVE_SOURCE_NAME))
    ok(is_obsolete_program_name("U003_Evidence_Review_Workbench.exe"))
    ok(is_obsolete_program_name("u003_old_acceptance_gate_v0_1.py"))
    ok(not is_obsolete_program_name("fvr_vendor"))
    ok(not is_obsolete_program_name("README.txt"))

    from tempfile import TemporaryDirectory

    with TemporaryDirectory(prefix="u003_cleanup_selftest_") as temp_dir:
        root = Path(temp_dir)
        base = root / "target.txt"
        ok(unique_destination(base) == base)
        base.write_text("x", encoding="utf-8")
        ok(unique_destination(base).name == "target.txt_002")

        archive_parent = root / "A-Archive"
        ok(choose_archive_root(archive_parent).name == ARCHIVE_BASENAME)
        archive_parent.mkdir()
        (archive_parent / ARCHIVE_BASENAME).mkdir()
        ok(choose_archive_root(archive_parent).name == ARCHIVE_BASENAME + "_002")

        source = root / "source.bin"
        source.write_bytes(b"abc")
        summary = tree_summary(source)
        ok(summary["kind"] == "file")
        ok(summary["file_count"] == 1)
        ok(summary["byte_count"] == 3)
        ok(summary["sha256"] == hashlib.sha256(b"abc").hexdigest())

        folder = root / "folder"
        folder.mkdir()
        (folder / "a.txt").write_text("a", encoding="utf-8")
        folder_summary = tree_summary(folder)
        ok(folder_summary["kind"] == "directory")
        ok(folder_summary["file_count"] == 1)
        ok(folder_summary["byte_count"] == 1)

        destination = root / "moved" / "source.bin"
        moved_to = move_one(source, destination)
        ok(moved_to.is_file())
        ok(not source.exists())

    assert checks == 21, checks
    print("U003-CLEANUP-V01-SELFTEST-PASS-21")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--workspace", type=Path)
    parser.add_argument("--apply", action="store_true")
    args = parser.parse_args()

    if args.self_test:
        return run_self_test()
    if args.workspace is None:
        parser.error("cleanup requires --workspace; dry-run is default")
    return run_cleanup(args.workspace, args.apply)


if __name__ == "__main__":
    raise SystemExit(main())
