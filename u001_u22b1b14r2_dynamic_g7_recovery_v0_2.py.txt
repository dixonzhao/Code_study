#!/usr/bin/env python3
"""B14-R2 v0.2 launcher with format-tolerant failure-evidence validation.

This revision preserves the fully tested v0.1 recovery engine. It changes only
the pre-patch identity gate: the failed orchestration must be FAILURE with the
first eight stages successful, while the source patcher itself still requires
one exact AST-bound FRAME_SIGNAL_COUNT_NOT_5 guard in the run-owned G7 source.
"""

from __future__ import annotations

import argparse
import hashlib
import importlib.machinery
import importlib.util
import json
from pathlib import Path
import re
import sys
from typing import Dict, Optional, Sequence


PROGRAM_ID = "U001-U22B1B14R2-DYNAMIC-G7-RECOVERY"
PROGRAM_VERSION = "0.2"
ENGINE_BASENAME = "u001_u22b1b14r2_dynamic_g7_recovery_v0_1.py.txt"


def load_engine():
    """Load the immutable v0.1 recovery engine from the same TEMP folder."""

    engine_path = Path(__file__).resolve().with_name(ENGINE_BASENAME)
    if not engine_path.is_file() or engine_path.is_symlink():
        raise RuntimeError("V01_ENGINE_MISSING")
    loader = importlib.machinery.SourceFileLoader("u22b1b14r2_v01_engine", str(engine_path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None:
        raise RuntimeError("V01_ENGINE_SPEC")
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module


def read_optional(path: Path) -> str:
    """Read optional log evidence without encoding failure."""

    if not path.is_file() or path.is_symlink():
        return ""
    return path.read_text(encoding="utf-8-sig", errors="replace")


def verify_failed_b14_v02(engine, failed_root: Path) -> Dict[str, object]:
    """Verify structural failure identity without depending on punctuation."""

    manifest_path = (
        failed_root / "03_RUN" / "CONTROL_OUTPUT" / "u001_g8a2_orchestration_manifest.json"
    )
    manifest = engine.read_json(manifest_path, "B14_ORCHESTRATION_MANIFEST")
    stages = manifest.get("Stages")
    accepted_stage_order = ("G1", "G2", "G3", "G4", "G5", "G6P", "G6A", "G7C")
    passed_stage_ids = []
    if isinstance(stages, list):
        for index, item in enumerate(stages):
            if isinstance(item, dict) and item.get("Success") is True:
                observed_id = item.get("StageId") or item.get("Stage") or item.get("Name")
                if not observed_id and index < len(accepted_stage_order):
                    observed_id = accepted_stage_order[index]
                passed_stage_ids.append(str(observed_id or ""))
    if manifest.get("Status") != "FAILURE" or len(passed_stage_ids) != 8:
        raise engine.RecoveryError(
            "B14_FAILURE_IDENTITY",
            f"{manifest.get('Status')}:{len(passed_stage_ids)}",
        )
    if tuple(passed_stage_ids) != accepted_stage_order:
        raise engine.RecoveryError("B14_PASSED_STAGE_ORDER", ".".join(passed_stage_ids))

    control_output = failed_root / "03_RUN" / "CONTROL_OUTPUT"
    texts = []
    errors = manifest.get("Errors")
    if isinstance(errors, list):
        texts.extend(str(item) for item in errors)
    for name in (
        "G7B_stdout.txt",
        "G7B_stderr.txt",
        "udp001_g8a2_return_capsule.txt",
    ):
        texts.append(read_optional(control_output / name))
    texts.append(read_optional(failed_root / "04_DIAGNOSTICS" / "RUN_stdout.txt"))
    texts.append(read_optional(failed_root / "04_DIAGNOSTICS" / "RUN_stderr.txt"))
    evidence = "\n".join(texts)
    normalized = re.sub(r"[^A-Z0-9]+", "_", evidence.upper()).strip("_")
    semantic_token_seen = "FRAME_SIGNAL_COUNT_NOT_5" in normalized

    return {
        "SuccessfulStages": 8,
        "PassedStageOrder": list(passed_stage_ids),
        "FailedNextStage": "G7B",
        "EvidenceSha256": hashlib.sha256(evidence.encode("utf-8")).hexdigest(),
        "NormalizedLegacyTokenSeen": semantic_token_seen,
        "IdentityBasis": "STATUS_FAILURE_PLUS_EXACT_FIRST_EIGHT_STAGE_ORDER",
        "SourceGuardRequiresIndependentExactASTMatch": True,
    }


def write_v02_manifest(output_root: Path, lines) -> None:
    """Add wrapper-version evidence without replacing v0.1 engine evidence."""

    path = output_root / "u22b1b14r2_wrapper_manifest_v0_2.json"
    value = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "EngineBasename": ENGINE_BASENAME,
        "Correction": "FORMAT_TOLERANT_FAILURE_EVIDENCE_GATE",
        "PatchSemanticsChanged": False,
        "ReturnCapsule": list(lines),
    }
    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
    )


def self_test(engine) -> int:
    """Run the complete v0.1 engine tests plus v0.2 evidence-gate tests."""

    if engine.self_test() != 0:
        raise AssertionError("ENGINE_SELFTEST")
    import tempfile

    with tempfile.TemporaryDirectory() as name:
        root = Path(name)
        control = root / "03_RUN" / "CONTROL_OUTPUT"
        control.mkdir(parents=True)
        stages = [
            {"StageId": stage_id, "Success": True}
            for stage_id in ("G1", "G2", "G3", "G4", "G5", "G6P", "G6A", "G7C")
        ]
        manifest = {
            "Status": "FAILURE",
            "Stages": stages,
            "Errors": ["G7 build failed: frame signal count not 5"],
        }
        (control / "u001_g8a2_orchestration_manifest.json").write_text(
            json.dumps(manifest), encoding="utf-8"
        )
        evidence = verify_failed_b14_v02(engine, root)
        if evidence["SuccessfulStages"] != 8:
            raise AssertionError("V02_STAGE_COUNT")
        if evidence["FailedNextStage"] != "G7B":
            raise AssertionError("V02_FAILED_STAGE")
        if evidence["NormalizedLegacyTokenSeen"] is not True:
            raise AssertionError("V02_NORMALIZATION")
        if evidence["SourceGuardRequiresIndependentExactASTMatch"] is not True:
            raise AssertionError("V02_SOURCE_GUARD")
    print("U22B1B14R2V02-SELFTEST-PASS-36")
    return 0


def build_parser() -> argparse.ArgumentParser:
    """Build the wrapper CLI."""

    parser = argparse.ArgumentParser(description="B14-R2 v0.2 dynamic G7 recovery")
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--repair", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--failed-run-root")
    parser.add_argument("--out-root")
    return parser


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Program entry point."""

    args = build_parser().parse_args(argv)
    try:
        engine = load_engine()
    except Exception as exc:
        print(f"B14R2E K-{type(exc).__name__.upper()}-X1 C-7")
        print(f"B14R2D D-{str(exc).upper()} C-0")
        return 7

    if args.self_test:
        return self_test(engine)
    if not args.udp001_root or not args.failed_run_root or not args.out_root:
        print("B14R2E K-ARGUMENT_MISSING-X1 C-7")
        return 7

    engine.PROGRAM_ID = PROGRAM_ID
    engine.PROGRAM_VERSION = PROGRAM_VERSION
    engine.verify_failed_b14 = lambda failed_root: verify_failed_b14_v02(engine, failed_root)
    lines = engine.execute_recovery(
        Path(args.udp001_root),
        Path(args.failed_run_root),
        Path(args.out_root),
    )
    if lines[0].endswith("C-0"):
        write_v02_manifest(Path(args.out_root), lines)
    print(lines[0])
    print(lines[1])
    return 0 if lines[0].endswith("C-0") else 7


if __name__ == "__main__":
    raise SystemExit(main())
