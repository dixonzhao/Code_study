#!/usr/bin/env python3
"""Read-only schema probe for U21E SignalBindings[].Signals[].Columns."""

from __future__ import annotations

import argparse
import json
from pathlib import Path


PREFLIGHT_REL = Path(
    "05_TESTS/02_ACCEPTANCE/U21G_G2_VERTICAL_V06.staging/control/run_preflight.json"
)


def reject_pairs(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("DUPLICATE_KEY")
        result[key] = value
    return result


def type_code(value) -> int:
    if value is None:
        return 0
    if isinstance(value, list):
        return 1
    if isinstance(value, dict):
        return 2
    if isinstance(value, str):
        return 3
    if isinstance(value, bool):
        return 4
    if isinstance(value, (int, float)):
        return 5
    return 9


def find_bindings(value):
    found = []
    if isinstance(value, dict):
        signals = value.get("Signals")
        if isinstance(signals, list):
            found.append(value)
        for item in value.values():
            found.extend(find_bindings(item))
    elif isinstance(value, list):
        for item in value:
            found.extend(find_bindings(item))
    return found


def compact_shape(value) -> str:
    """Expose schema names and generic column names, never measurement values."""
    if isinstance(value, dict):
        parts = []
        for key in sorted(value):
            item = value[key]
            if isinstance(item, str):
                parts.append(f"{key}:{item}")
            else:
                parts.append(f"{key}:T{type_code(item)}")
        return ",".join(parts)
    if isinstance(value, list):
        parts = []
        for item in value:
            if isinstance(item, str):
                parts.append(item)
            elif isinstance(item, dict):
                parts.append("{" + compact_shape(item) + "}")
            else:
                parts.append(f"T{type_code(item)}")
        return ",".join(parts)
    return f"T{type_code(value)}"


def inspect(document) -> tuple[str, str, str]:
    bindings = find_bindings(document.get("SignalBindings"))
    if not bindings:
        raise ValueError("SIGNALS_NOT_FOUND")
    signals = bindings[0]["Signals"]
    if not signals or not isinstance(signals[0], dict):
        raise ValueError("SIGNAL_RECORD_NOT_FOUND")
    signal = signals[0]
    columns = signal.get("Columns")
    if columns is None:
        raise ValueError("COLUMNS_NOT_FOUND")
    column_count = len(columns) if isinstance(columns, (list, dict)) else 0
    summary = (
        f"G2R2SC1 N-B{len(bindings)}-ST{type_code(signals)}-SN{len(signals)}"
        f"-CT{type_code(columns)}-CN{column_count}-X0 C-0"
    )
    signal_keys = "SK-" + ".".join(sorted(str(key) for key in signal))
    column_shape = "CK-" + compact_shape(columns)
    return summary, signal_keys[:600], column_shape[:900]


def self_test() -> int:
    sample = {
        "SignalBindings": [{
            "Signals": [{
                "CanonicalName": "SYNTHETIC_SIGNAL",
                "SignalName": "Synthetic",
                "SignalRole": "PRIMARY",
                "Columns": {"value": "SYNTHETIC_SIGNAL__value"},
            }]
        }]
    }
    lines = inspect(sample)
    assert lines[0] == "G2R2SC1 N-B1-ST1-SN1-CT2-CN1-X0 C-0"
    assert "CanonicalName" in lines[1]
    assert "value:SYNTHETIC_SIGNAL__value" in lines[2]
    print("G2R2SC1-SELFTEST-PASS-3")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root")
    parser.add_argument("--self-test", action="store_true")
    args = parser.parse_args()
    if args.self_test:
        return self_test()
    if not args.workspace_root:
        parser.error("--workspace-root is required")
    path = Path(args.workspace_root).resolve() / PREFLIGHT_REL
    try:
        document = json.loads(
            path.read_text(encoding="utf-8-sig"), object_pairs_hook=reject_pairs
        )
        for line in inspect(document):
            print(line)
        return 0
    except Exception as exc:
        print(f"G2R2SC1 N-B0-ST0-SN0-CT0-CN0-X1 C-7 K-{exc.__class__.__name__.upper()}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
