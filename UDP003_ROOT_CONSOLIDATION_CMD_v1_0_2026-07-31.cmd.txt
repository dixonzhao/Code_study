@echo off
setlocal EnableExtensions EnableDelayedExpansion

rem UDP003 root consolidation v1.0
rem Run from the C-CODE root in Windows CMD.
rem This node creates UDP003, backs up known root-level sources, moves the
rem active FVR into UDP003, and routes legacy P001-P004 into UDP001/UDP002.
rem It never moves company data, DBC, video, CSV analysis data, or Case folders.

set "FVR_ACTIVE=framecount_video_reviewer_v0_5.py"
set "P001=can_header_catalog_gui.py"
set "P002=matlab_can_time_order_decoder_gui.py"
set "P003=can_analysis_csv_cleaner_gui.py"
set "P004=sre_cam_stage_residual_error_analyzer_v4_5_3_jp_compact_excel_notes.py"
set "BACKUP=.\ARCH\20260731_ROOT_CONSOLIDATION_BACKUP"

if not exist ".\UDP001" (
  echo UDP003WS1-ERROR-MISSING-UDP001 C-7
  exit /b 7
)
if not exist ".\UDP002" (
  echo UDP003WS1-ERROR-MISSING-UDP002 C-7
  exit /b 7
)
if exist ".\UDP003" (
  echo UDP003WS1-ERROR-UDP003-ALREADY-EXISTS C-7
  exit /b 7
)
if exist "%BACKUP%" (
  echo UDP003WS1-ERROR-BACKUP-ALREADY-EXISTS C-7
  exit /b 7
)

for %%F in ("%FVR_ACTIVE%" "%P001%" "%P002%" "%P003%" "%P004%") do (
  if not exist ".\%%~F" (
    echo UDP003WS1-ERROR-MISSING-ROOT-FILE-%%~F C-7
    exit /b 7
  )
)

if exist ".\UDP003\01_PROGRAMS\%FVR_ACTIVE%" (
  echo UDP003WS1-ERROR-ACTIVE-FVR-DESTINATION-EXISTS C-7
  exit /b 7
)

md "%BACKUP%" || goto :create_fail
md ".\UDP003\01_PROGRAMS" || goto :create_fail
md ".\UDP003\02_CONTROLS" || goto :create_fail
md ".\UDP003\03_CASES\TC001_20260422\00_INPUT-READONLY" || goto :create_fail
md ".\UDP003\03_CASES\TC001_20260422\01_WORK" || goto :create_fail
md ".\UDP003\03_CASES\TC001_20260422\02_OUTPUT" || goto :create_fail
md ".\UDP003\03_CASES\TC001_20260422\03_REPORTS" || goto :create_fail
md ".\UDP003\03_CASES\TC001_20260422\04_LOGS" || goto :create_fail
md ".\UDP003\04_TEMP" || goto :create_fail
md ".\UDP003\05_TESTS\01_FIXTURES" || goto :create_fail
md ".\UDP003\05_TESTS\02_EXPECTED" || goto :create_fail
md ".\UDP003\05_TESTS\03_RUNS" || goto :create_fail
md ".\UDP003\06_DOCS\01_DESIGN" || goto :create_fail
md ".\UDP003\06_DOCS\02_MANUALS" || goto :create_fail
md ".\UDP003\06_DOCS\03_REPORTS" || goto :create_fail
md ".\UDP003\ARCH\01_LEGACY_VERSIONS" || goto :create_fail
md ".\UDP003\ARCH\02_RETIRED_BUILDS" || goto :create_fail
md ".\UDP003\ARCH\03_FAILED_EXPERIMENTS" || goto :create_fail

if not exist ".\UDP001\ARCH\02_ROOT_ORIGINALS" (
  md ".\UDP001\ARCH\02_ROOT_ORIGINALS" || goto :create_fail
)
if not exist ".\UDP002\ARCH\02_ROOT_ORIGINALS" (
  md ".\UDP002\ARCH\02_ROOT_ORIGINALS" || goto :create_fail
)

set /a BACKUP_COUNT=0
for %%F in ("%FVR_ACTIVE%" "%P001%" "%P002%" "%P003%" "%P004%") do (
  copy /v /y ".\%%~F" "%BACKUP%\%%~F" >nul || goto :backup_fail
  set /a BACKUP_COUNT+=1
)

set /a OLD_FVR_COUNT=0
for %%F in (.\framecount_video_reviewer_v*.py) do (
  if exist "%%~fF" if /I not "%%~nxF"=="%FVR_ACTIVE%" (
    copy /v /y "%%~fF" "%BACKUP%\%%~nxF" >nul || goto :backup_fail
    set /a BACKUP_COUNT+=1
  )
)

move /y ".\%FVR_ACTIVE%" ".\UDP003\01_PROGRAMS\%FVR_ACTIVE%" >nul || goto :move_fail
fc /b "%BACKUP%\%FVR_ACTIVE%" ".\UDP003\01_PROGRAMS\%FVR_ACTIVE%" >nul || goto :verify_fail

for %%F in (.\framecount_video_reviewer_v*.py) do (
  if exist "%%~fF" if /I not "%%~nxF"=="%FVR_ACTIVE%" (
    move /y "%%~fF" ".\UDP003\ARCH\01_LEGACY_VERSIONS\%%~nxF" >nul || goto :move_fail
    fc /b "%BACKUP%\%%~nxF" ".\UDP003\ARCH\01_LEGACY_VERSIONS\%%~nxF" >nul || goto :verify_fail
    attrib +R ".\UDP003\ARCH\01_LEGACY_VERSIONS\%%~nxF"
    set /a OLD_FVR_COUNT+=1
  )
)

for %%F in ("%P001%" "%P002%" "%P003%") do (
  move /y ".\%%~F" ".\UDP001\ARCH\02_ROOT_ORIGINALS\%%~F" >nul || goto :move_fail
  fc /b "%BACKUP%\%%~F" ".\UDP001\ARCH\02_ROOT_ORIGINALS\%%~F" >nul || goto :verify_fail
  attrib +R ".\UDP001\ARCH\02_ROOT_ORIGINALS\%%~F"
)

move /y ".\%P004%" ".\UDP002\ARCH\02_ROOT_ORIGINALS\%P004%" >nul || goto :move_fail
fc /b "%BACKUP%\%P004%" ".\UDP002\ARCH\02_ROOT_ORIGINALS\%P004%" >nul || goto :verify_fail
attrib +R ".\UDP002\ARCH\02_ROOT_ORIGINALS\%P004%"

attrib +R "%BACKUP%\*"

for %%F in ("%FVR_ACTIVE%" "%P001%" "%P002%" "%P003%" "%P004%") do (
  if exist ".\%%~F" goto :root_not_clean
)

echo UDP003WS1 N-R1-O!OLD_FVR_COUNT!-D3-A1-B!BACKUP_COUNT!-K1-X0 C-0
exit /b 0

:create_fail
echo UDP003WS1-ERROR-CREATE-DIRECTORY C-7
exit /b 7

:backup_fail
echo UDP003WS1-ERROR-BACKUP C-7
exit /b 7

:move_fail
echo UDP003WS1-ERROR-MOVE C-7
exit /b 7

:verify_fail
echo UDP003WS1-ERROR-BINARY-VERIFY C-7
exit /b 7

:root_not_clean
echo UDP003WS1-ERROR-ROOT-NOT-CLEAN C-7
exit /b 7
