#!/usr/bin/env python3
"""U001 H6B2 source/catalog reconciliation.

The program runs the accepted H2 Adapter inventory action against the uniquely
identified real source set, reconciles eligible Legacy D3 catalogs against the
observed capability catalog, and writes a provisional H2 selection request.
It does not decode payload rows and does not authorize formal replay.
"""

from __future__ import annotations

import argparse
import csv
import json
import re
import shutil
import subprocess
import sys
import tempfile
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, Iterable, List, Optional, Sequence, Tuple


PROGRAM_ID = "U001_H6B2_SOURCE_CATALOG_RECONCILIATION"
PROGRAM_VERSION = "1.0"
OUTPUT_FOLDER = "U1H6B2_OUTPUT"
ADAPTER_ID = "CameraRadar.U001.MatlabCanCsvDbc"
FRAME_KEY = (2, "0x620", "framecount")


class ReconciliationError(RuntimeError):
    """Expected fail-closed reconciliation error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def normalize_name(value: Any) -> str:
    return "".join(char.lower() for char in str(value or "") if char.isalnum())


def normalize_can_id(value: Any) -> str:
    text = str(value or "").strip()
    if not text:
        return ""
    try:
        number = int(text, 16)
    except ValueError:
        match = re.search(r"(?:0x)?([0-9a-fA-F]+)", text)
        if not match:
            return normalize_name(text)
        number = int(match.group(1), 16)
    return f"0x{number:X}"


def integer(value: Any, default: int = -1) -> int:
    try:
        return int(str(value).strip())
    except (TypeError, ValueError):
        return default


def first_present(row: Dict[str, Any], names: Iterable[str]) -> str:
    normalized = {normalize_name(key): value for key, value in row.items()}
    for name in names:
        value = normalized.get(normalize_name(name))
        if value is not None and str(value).strip():
            return str(value).strip()
    return ""


def signal_key_from_legacy(row: Dict[str, Any]) -> Optional[Tuple[int, str, str]]:
    channel = integer(first_present(row, ("Channel", "SourceChannel")))
    can_id = normalize_can_id(
        first_present(row, ("CAN_ID", "CANID", "SourceMessageId", "MessageId"))
    )
    signal_name = normalize_name(
        first_present(row, ("SignalName", "SourceSignalName"))
    )
    if channel < 0 or not can_id or not signal_name:
        return None
    return channel, can_id, signal_name


def signal_key_from_capability(row: Dict[str, Any]) -> Tuple[int, str, str]:
    return (
        integer(row.get("SourceChannel")),
        normalize_can_id(row.get("SourceMessageId")),
        normalize_name(row.get("SourceSignalName")),
    )


def read_csv(path: Path) -> List[Dict[str, str]]:
    if not path.is_file():
        raise ReconciliationError("CSV_NOT_FOUND:" + path.name)
    with path.open("r", encoding="utf-8-sig", newline="") as handle:
        return list(csv.DictReader(handle))


def read_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        raise ReconciliationError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise ReconciliationError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise ReconciliationError(label + "_ROOT_NOT_OBJECT")
    return value


def write_json(path: Path, value: Dict[str, Any]) -> None:
    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise ReconciliationError("U_UDP001_ROOT_NOT_FOUND")
    return root


def version_key(path: Path) -> Tuple[int, ...]:
    values = re.findall(r"\d+", path.name)
    return tuple(int(value) for value in values) if values else (0,)


def locate_adapter(root: Path) -> Path:
    adapter_root = root / "01_PROGRAMS" / "adapters" / ADAPTER_ID
    candidates = [
        folder / "adapter.py"
        for folder in adapter_root.iterdir()
        if folder.is_dir() and (folder / "adapter.py").is_file()
    ] if adapter_root.is_dir() else []
    if not candidates:
        raise ReconciliationError("H2_ADAPTER_NOT_FOUND")
    return max(candidates, key=lambda path: version_key(path.parent))


def load_b1(root: Path) -> Tuple[List[Dict[str, str]], List[Dict[str, str]], Dict[str, Any]]:
    b1 = root / "04_TEMP" / "U1H6B1_OUTPUT"
    legacy = read_csv(b1 / "u001_h6b1_legacy_shortlist.csv")
    sources = read_csv(b1 / "u001_h6b1_source_sets.csv")
    provisional = read_json(
        b1 / "u001_h6b1_provisional_selection.json",
        "H6B1_PROVISIONAL_SELECTION",
    )
    if not legacy:
        raise ReconciliationError("H6B1_LEGACY_SHORTLIST_EMPTY")
    preferred = [row for row in sources if integer(row.get("PreferredSourceSet"), 0) == 1]
    if len(preferred) != 1:
        raise ReconciliationError(f"H6B1_PREFERRED_SOURCE_COUNT:{len(preferred)}")
    return legacy, preferred, provisional


def dbc_channels(source_folder: Path, dbc_names: List[str]) -> List[Tuple[int, Path]]:
    resolved: List[Tuple[int, Path]] = []
    used: set[int] = set()
    for fallback, name in enumerate(sorted(dbc_names), start=1):
        match = re.search(r"dbc[_-]?(\d+)", name, flags=re.IGNORECASE)
        channel = int(match.group(1)) if match else fallback
        if channel in used:
            raise ReconciliationError(f"DBC_CHANNEL_DUPLICATE:{channel}")
        path = source_folder / name
        if not path.is_file():
            raise ReconciliationError("DBC_FILE_NOT_FOUND:" + name)
        used.add(channel)
        resolved.append((channel, path))
    return sorted(resolved)


def create_inventory_request(
    root: Path,
    source_row: Dict[str, str],
    output_root: Path,
) -> Tuple[Path, Path]:
    source_folder = root / source_row["RelativeFolder"]
    candump = source_folder / source_row["CANDumpFile"]
    if not candump.is_file():
        raise ReconciliationError("CANDUMP_NOT_FOUND")
    dbc_names = [name for name in source_row.get("DBCFiles", "").split("|") if name]
    channels = dbc_channels(source_folder, dbc_names)
    dataset_id = source_folder.parent.name if source_folder.name.upper() == "00_SOURCE-READONLY" else source_folder.name
    inputs: List[Dict[str, Any]] = [
        {"Kind": "MATLAB_CAN_CSV", "Path": str(candump.resolve())}
    ]
    inputs.extend(
        {
            "Kind": "DBC_CHANNEL",
            "Channel": channel,
            "Path": str(path.resolve()),
        }
        for channel, path in channels
    )
    request = {
        "RequestSchemaVersion": "U001_INPUT_ADAPTER_REQUEST_V0_1",
        "RequestId": "U1H6B2_REAL_SOURCE_INVENTORY",
        "DatasetId": dataset_id,
        "RunId": "U1H6B2_PROFILE_RECONCILIATION",
        "SegmentId": "FULL_SOURCE",
        "SourceId": "CAN_SOURCE_001",
        "TestMode": False,
        "Inputs": inputs,
        "Configuration": {
            "SelectionMode": "ALL_TECHNICALLY_ELIGIBLE",
            "SelectedSignals": [],
            "FrameKey": {
                "Channel": FRAME_KEY[0],
                "CAN_ID": FRAME_KEY[1],
                "SignalName": "FrameCount",
            },
            "MaxSourceRows": None,
            "ChunkRows": 5000,
            "WorkerCount": 1,
            "MaxInFlightChunks": 2,
            "MaxResidentMemoryMB": 512,
            "OutputShardRows": 250000,
            "WriteOrderPolicy": "SOURCE_ORDER",
        },
    }
    request_path = output_root / "u001_h6b2_real_source_base_request.json"
    write_json(request_path, request)
    return request_path, source_folder


def run_adapter_inventory(adapter: Path, request: Path, output: Path) -> None:
    command = [
        sys.executable,
        str(adapter),
        "--inventory",
        "--request",
        str(request),
        "--out",
        str(output),
    ]
    completed = subprocess.run(
        command,
        cwd=adapter.parent,
        capture_output=True,
        text=True,
        timeout=1800,
        check=False,
    )
    (output.parent / "u001_h6b2_adapter_inventory_stdout.txt").write_text(
        completed.stdout, encoding="utf-8"
    )
    (output.parent / "u001_h6b2_adapter_inventory_stderr.txt").write_text(
        completed.stderr, encoding="utf-8"
    )
    if completed.returncode != 0:
        detail = (completed.stderr or completed.stdout or "NO_CHILD_OUTPUT")[-1000:]
        raise ReconciliationError("H2_INVENTORY_EXIT:" + detail.replace("\n", " "))


def fingerprint_records(value: Any) -> List[Dict[str, str]]:
    records: List[Dict[str, str]] = []
    if isinstance(value, dict):
        normalized = {normalize_name(key): item for key, item in value.items()}
        file_name = normalized.get("filename")
        digest = normalized.get("sha256")
        if file_name and digest:
            records.append({"FileName": str(file_name), "SHA256": str(digest).lower()})
        for item in value.values():
            records.extend(fingerprint_records(item))
    elif isinstance(value, list):
        for item in value:
            records.extend(fingerprint_records(item))
    return records


def compare_fingerprints(
    legacy_manifest: Dict[str, Any],
    actual_summary: Dict[str, Any],
) -> Tuple[str, int, int]:
    actual_records = actual_summary.get("InputFingerprints", [])
    actual = {
        str(item.get("FileName", "")).casefold(): str(item.get("SHA256", "")).lower()
        for item in actual_records
        if isinstance(item, dict) and item.get("FileName") and item.get("SHA256")
    }
    legacy = {
        item["FileName"].casefold(): item["SHA256"]
        for item in fingerprint_records(legacy_manifest)
    }
    comparable = sorted(set(actual).intersection(legacy))
    if not comparable:
        return "NO_EVIDENCE", 0, 0
    matched = sum(actual[name] == legacy[name] for name in comparable)
    mismatched = len(comparable) - matched
    if mismatched:
        return "MISMATCH", matched, mismatched
    if len(comparable) == len(actual):
        return "MATCH_ALL", matched, 0
    return "MATCH_PARTIAL", matched, 0


def reconcile_candidate(
    root: Path,
    row: Dict[str, str],
    capability: Dict[Tuple[int, str, str], Dict[str, str]],
    actual_summary: Dict[str, Any],
) -> Dict[str, Any]:
    folder = root / row["RelativeFolder"]
    legacy_catalog = read_csv(folder / "u001_signal_catalog.csv")
    legacy_manifest = read_json(folder / "u001_run_manifest.json", "LEGACY_MANIFEST")
    mappable_ids: List[str] = []
    unmappable: List[str] = []
    unkeyed = 0
    seen: set[str] = set()
    for legacy_signal in legacy_catalog:
        key = signal_key_from_legacy(legacy_signal)
        if key is None:
            unkeyed += 1
            continue
        capability_row = capability.get(key)
        if capability_row is None:
            unmappable.append("/".join(str(item) for item in key))
            continue
        catalog_id = capability_row["CatalogItemId"]
        if catalog_id not in seen:
            seen.add(catalog_id)
            mappable_ids.append(catalog_id)
    fingerprint_status, matched, mismatched = compare_fingerprints(
        legacy_manifest, actual_summary
    )
    total_keyed = len(mappable_ids) + len(unmappable)
    mapping_ratio = len(mappable_ids) / total_keyed if total_keyed else 0.0
    result = dict(row)
    result.update(
        {
            "LegacySignalRows": len(legacy_catalog),
            "KeyedSignalCount": total_keyed,
            "MappableSignalCount": len(mappable_ids),
            "UnmappableSignalCount": len(unmappable),
            "UnkeyedSignalCount": unkeyed,
            "MappingRatio": f"{mapping_ratio:.6f}",
            "FingerprintStatus": fingerprint_status,
            "FingerprintMatchCount": matched,
            "FingerprintMismatchCount": mismatched,
            "MappableCatalogItemIds": mappable_ids,
            "UnmappableExamples": unmappable[:10],
        }
    )
    return result


def candidate_rank(row: Dict[str, Any]) -> Tuple[int, int, float, int, int]:
    fingerprint_priority = {
        "MATCH_ALL": 3,
        "MATCH_PARTIAL": 2,
        "NO_EVIDENCE": 1,
        "MISMATCH": 0,
    }.get(str(row.get("FingerprintStatus")), 0)
    no_mismatch = int(integer(row.get("FingerprintMismatchCount"), 0) == 0)
    ratio = float(row.get("MappingRatio", 0.0))
    count = integer(row.get("MappableSignalCount"), 0)
    earlier_b1_rank = -integer(row.get("RecommendationRank"), 9999)
    return fingerprint_priority, no_mismatch, ratio, count, earlier_b1_rank


def write_reconciliation_csv(path: Path, rows: List[Dict[str, Any]]) -> None:
    columns = (
        "ReconciliationRank",
        "CandidateId",
        "IdentityGroup",
        "RelativeFolder",
        "DatasetId",
        "RunId",
        "SegmentId",
        "LegacySignalRows",
        "KeyedSignalCount",
        "MappableSignalCount",
        "UnmappableSignalCount",
        "UnkeyedSignalCount",
        "MappingRatio",
        "FingerprintStatus",
        "FingerprintMatchCount",
        "FingerprintMismatchCount",
        "RecommendationRank",
        "UnmappableExamples",
    )
    with path.open("w", encoding="utf-8-sig", newline="") as handle:
        writer = csv.DictWriter(handle, fieldnames=columns, extrasaction="ignore")
        writer.writeheader()
        for row in rows:
            serialized = dict(row)
            serialized["UnmappableExamples"] = "|".join(row["UnmappableExamples"])
            writer.writerow(serialized)


def create_selection_request(
    output_root: Path,
    selected: Dict[str, Any],
    capability: Dict[Tuple[int, str, str], Dict[str, str]],
) -> Tuple[Path, int]:
    frame_row = capability.get(FRAME_KEY)
    if frame_row is None:
        raise ReconciliationError("FRAME_KEY_NOT_IN_CAPABILITY_CATALOG")
    frame_id = frame_row["CatalogItemId"]
    selected_ids = list(selected["MappableCatalogItemIds"])
    user_ids = [item for item in selected_ids if item != frame_id]
    if not user_ids:
        raise ReconciliationError("CONTROLLED_SELECTION_EMPTY_EXCEPT_FRAME")
    value = {
        "SelectionRequestSchema": "U001_SELECTION_PROFILE_REQUEST_V0_1",
        "ProfileId": "H6_LEGACY_RECONCILIATION_" + selected["CandidateId"],
        "ProfileVersion": "1.0.0",
        "ProfileStatus": "PROVISIONAL",
        "SelectedCatalogItemIds": user_ids,
        "FrameKeyCatalogItemId": frame_id,
        "ResourcePolicy": {
            "ChunkRows": 5000,
            "WorkerCount": 1,
            "MaxInFlightChunks": 2,
            "MaxResidentMemoryMB": 512,
            "OutputShardRows": 250000,
        },
    }
    path = output_root / "u001_h6b2_selection_request.json"
    write_json(path, value)
    return path, len(set(user_ids + [frame_id]))


def execute(root_argument: Optional[str]) -> int:
    output_root: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        output_root = root / "04_TEMP" / OUTPUT_FOLDER
        if output_root.exists():
            shutil.rmtree(output_root)
        output_root.mkdir(parents=True)

        legacy_rows, preferred_sources, _ = load_b1(root)
        source_row = preferred_sources[0]
        request_path, _ = create_inventory_request(root, source_row, output_root)
        adapter = locate_adapter(root)
        inventory_output = output_root / "source_inventory"
        run_adapter_inventory(adapter, request_path, inventory_output)

        capability_rows = read_csv(inventory_output / "u001_source_capability_catalog.csv")
        capability = {signal_key_from_capability(row): row for row in capability_rows}
        if len(capability) != len(capability_rows):
            raise ReconciliationError("CAPABILITY_SIGNAL_KEY_COLLISION")
        actual_summary = read_json(
            inventory_output / "u001_source_inventory.json",
            "SOURCE_INVENTORY",
        )

        reconciled = [
            reconcile_candidate(root, row, capability, actual_summary)
            for row in legacy_rows
        ]
        reconciled.sort(key=candidate_rank, reverse=True)
        for index, row in enumerate(reconciled, start=1):
            row["ReconciliationRank"] = index
        selected = reconciled[0]
        if selected["FingerprintStatus"] == "MISMATCH":
            raise ReconciliationError("ALL_TOP_CANDIDATE_SOURCE_FINGERPRINT_MISMATCH")
        selection_path, profile_count = create_selection_request(
            output_root, selected, capability
        )

        top_key = candidate_rank(selected)[:-1]
        tied = sum(candidate_rank(row)[:-1] == top_key for row in reconciled)
        ambiguity = int(
            tied > 1
            or integer(selected["UnmappableSignalCount"], 0) > 0
            or integer(selected["UnkeyedSignalCount"], 0) > 0
        )
        write_reconciliation_csv(
            output_root / "u001_h6b2_legacy_source_reconciliation.csv",
            reconciled,
        )
        decision = {
            "DecisionSchema": "U001_H6B2_RECONCILIATION_DECISION_V1_0",
            "GeneratedUtc": utc_now(),
            "Status": "PROVISIONAL_NOT_AUTHORIZED",
            "PayloadRowsDecoded": 0,
            "ActualCapabilityCount": len(capability_rows),
            "SelectedLegacyCandidateId": selected["CandidateId"],
            "SelectedIdentityGroup": selected.get("IdentityGroup"),
            "MappingRatio": selected["MappingRatio"],
            "MappableSignalCount": selected["MappableSignalCount"],
            "UnmappableSignalCount": selected["UnmappableSignalCount"],
            "UnkeyedSignalCount": selected["UnkeyedSignalCount"],
            "FingerprintStatus": selected["FingerprintStatus"],
            "FingerprintMatchCount": selected["FingerprintMatchCount"],
            "FingerprintMismatchCount": selected["FingerprintMismatchCount"],
            "SelectionRequestBasename": selection_path.name,
            "AmbiguityFlag": ambiguity,
            "FormalReplayAuthorized": False,
            "NextRequiredAction": "COMPILE_PROFILE_AND_RUN_H2_PREFLIGHT",
        }
        write_json(output_root / "u001_h6b2_reconciliation_decision.json", decision)
        recommendation_id = str(selected["CandidateId"]).replace("PKG", "")
        capsule = (
            f"U1H6B2-V{len(capability_rows)}-L{len(reconciled)}-R{recommendation_id}"
            f"-M{selected['MappableSignalCount']}-U{selected['UnmappableSignalCount']}"
            f"-F{selected['FingerprintMatchCount']}-P{profile_count}-A{ambiguity}-X0-C0"
        )
        (output_root / "u001_h6b2_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        print("OUTPUT-04_TEMP/U1H6B2_OUTPUT")
        return 0
    except Exception as exc:
        if output_root is None:
            output_root = Path.cwd() / OUTPUT_FOLDER
        output_root.mkdir(parents=True, exist_ok=True)
        write_json(
            output_root / "u001_h6b2_diagnostic.json",
            {
                "ProgramId": PROGRAM_ID,
                "GeneratedUtc": utc_now(),
                "ExceptionType": type(exc).__name__,
                "Detail": str(exc)[:3000],
            },
        )
        print(f"U1H6B2-X1-C7-K{type(exc).__name__.upper()}")
        return 7


def self_test() -> int:
    checks: List[bool] = []
    root = Path(tempfile.mkdtemp(prefix="u1h6b2_selftest_"))
    try:
        capability_rows = [
            {
                "CatalogItemId": "CAN.CH2.ID620.FrameCount",
                "SourceChannel": "2",
                "SourceMessageId": "0x620",
                "SourceSignalName": "FrameCount",
            },
            {
                "CatalogItemId": "CAN.CH2.ID221.Distance",
                "SourceChannel": "2",
                "SourceMessageId": "0x221",
                "SourceSignalName": "Distance",
            },
            {
                "CatalogItemId": "CAN.CH1.ID380.DistanceToObject",
                "SourceChannel": "1",
                "SourceMessageId": "0x380",
                "SourceSignalName": "DistanceToObject",
            },
        ]
        capability = {signal_key_from_capability(row): row for row in capability_rows}
        checks.append(len(capability) == 3)
        checks.append(FRAME_KEY in capability)
        package = root / "legacy"
        package.mkdir()
        with (package / "u001_signal_catalog.csv").open("w", encoding="utf-8", newline="") as handle:
            writer = csv.DictWriter(handle, fieldnames=("Channel", "CAN_ID", "SignalName"))
            writer.writeheader()
            writer.writerows(
                [
                    {"Channel": 2, "CAN_ID": "0x620", "SignalName": "FrameCount"},
                    {"Channel": 2, "CAN_ID": "0x221", "SignalName": "Distance"},
                    {"Channel": 1, "CAN_ID": "0x380", "SignalName": "DistanceToObject"},
                ]
            )
        manifest = {
            "InputFingerprints": [
                {"FileName": "CANDump.csv", "SHA256": "a" * 64},
                {"FileName": "DBC1.dbc", "SHA256": "b" * 64},
            ]
        }
        write_json(package / "u001_run_manifest.json", manifest)
        actual = {
            "InputFingerprints": [
                {"FileName": "CANDump.csv", "SHA256": "a" * 64},
                {"FileName": "DBC1.dbc", "SHA256": "b" * 64},
                {"FileName": "DBC2.dbc", "SHA256": "c" * 64},
            ]
        }
        row = {
            "CandidateId": "PKG003",
            "RelativeFolder": "legacy",
            "RecommendationRank": "1",
            "IdentityGroup": "G01",
        }
        reconciled = reconcile_candidate(root, row, capability, actual)
        checks.append(reconciled["MappableSignalCount"] == 3)
        checks.append(reconciled["UnmappableSignalCount"] == 0)
        checks.append(reconciled["UnkeyedSignalCount"] == 0)
        checks.append(reconciled["MappingRatio"] == "1.000000")
        checks.append(reconciled["FingerprintStatus"] == "MATCH_PARTIAL")
        checks.append(reconciled["FingerprintMatchCount"] == 2)
        checks.append(reconciled["FingerprintMismatchCount"] == 0)
        output = root / "output"
        output.mkdir()
        path, count = create_selection_request(output, reconciled, capability)
        checks.append(path.is_file())
        checks.append(count == 3)
        selection = read_json(path, "SELECTION")
        checks.append(selection["FrameKeyCatalogItemId"] == "CAN.CH2.ID620.FrameCount")
        checks.append(len(selection["SelectedCatalogItemIds"]) == 2)
        checks.append(candidate_rank(reconciled)[0] == 2)
        if not all(checks):
            failed = ".".join(str(index) for index, value in enumerate(checks, start=1) if not value)
            raise ReconciliationError("SELFTEST_ASSERTION_FAILED_" + failed)
        print(f"U1H6B2-ST-PASS-T{len(checks)}-X0-C0")
        return 0
    except Exception as exc:
        print(
            f"U1H6B2-ST-FAIL-T{sum(checks)}-X1-C7-"
            f"K{type(exc).__name__.upper()}-D{str(exc)}"
        )
        return 7
    finally:
        shutil.rmtree(root, ignore_errors=True)


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H6B2 source/catalog reconciliation")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--reconcile", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.self_test:
        return self_test()
    return execute(args.root)


if __name__ == "__main__":
    sys.exit(main())
