#!/usr/bin/env python3
"""Non-destructive stage and permission diagnostic for failed G10D2C builds."""

from __future__ import annotations

import argparse
import os
import shutil
import stat
import sys
import time
from pathlib import Path


PROGRAM_NAME = "u001_g10d2cd1_permission_diagnostic_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2CD1-SELFTEST-PASS-16"
REPORT_NAME = "G10D2CD1_PERMISSION_DIAGNOSTIC_REPORT.txt"
CLUE_TOKENS = (
    "permissionerror",
    "permission denied",
    "access is denied",
    "winerror 5",
    "used by another process",
    "cannot access the file",
)


def latest_pk04(temp_root: Path) -> Path | None:
    if not temp_root.is_dir():
        return None
    candidates = [
        path for path in temp_root.iterdir()
        if path.is_dir() and path.name.upper().startswith("PK04")
    ]
    return max(candidates, key=lambda path: path.stat().st_mtime, default=None)


def files_named(root: Path | None, name: str) -> list[Path]:
    if root is None or not root.is_dir():
        return []
    try:
        return [path for path in root.rglob(name) if path.is_file()]
    except OSError:
        return []


def any_suffix(root: Path | None, suffix: str) -> list[Path]:
    if root is None or not root.is_dir():
        return []
    try:
        return [path for path in root.rglob("*") if path.is_file() and path.suffix.lower() == suffix]
    except OSError:
        return []


def probe_own_write(temp_root: Path) -> tuple[int, int, str]:
    token = f"g10d2cd1_{os.getpid()}_{int(time.time())}"
    first = temp_root / f"{token}.tmp"
    second = temp_root / f"{token}.renamed"
    folder1 = temp_root / f"{token}_dir"
    folder2 = temp_root / f"{token}_dir_renamed"
    file_ok = 0
    dir_ok = 0
    detail = "NONE"
    try:
        first.write_bytes(b"G10D2CD1")
        first.replace(second)
        second.unlink()
        file_ok = 1
    except Exception as exc:
        detail = "FILE_" + type(exc).__name__
        for path in (first, second):
            try:
                if path.exists():
                    path.unlink()
            except OSError:
                pass
    try:
        folder1.mkdir()
        folder1.replace(folder2)
        folder2.rmdir()
        dir_ok = 1
    except Exception as exc:
        if detail == "NONE":
            detail = "DIR_" + type(exc).__name__
        for path in (folder1, folder2):
            try:
                if path.exists():
                    path.rmdir()
            except OSError:
                pass
    return file_ok, dir_ok, detail


def scan_permission_clue(root: Path | None) -> tuple[int, str]:
    if root is None or not root.is_dir():
        return 0, "NONE"
    candidates = []
    try:
        for path in root.rglob("*"):
            if path.is_file() and path.suffix.lower() in {".txt", ".log", ".json"}:
                candidates.append(path)
    except OSError:
        return 0, "SCAN_OSERROR"
    candidates.sort(key=lambda path: path.stat().st_mtime, reverse=True)
    for path in candidates[:80]:
        try:
            size = path.stat().st_size
            with path.open("rb") as stream:
                if size > 262144:
                    stream.seek(size - 262144)
                text = stream.read().decode("utf-8", errors="replace")
        except OSError:
            continue
        for line in reversed(text.splitlines()):
            lowered = line.lower()
            if any(token in lowered for token in CLUE_TOKENS):
                clean = " ".join(line.strip().split())[:180]
                return 1, f"{path.name}:{clean}"
    return 0, "NONE"


def readonly_file_count(root: Path | None) -> int:
    if root is None or not root.is_dir():
        return 0
    count = 0
    try:
        for path in root.rglob("*"):
            if not path.is_file():
                continue
            mode_readonly = not bool(path.stat().st_mode & stat.S_IWRITE)
            attrs = getattr(path.stat(), "st_file_attributes", 0)
            win_readonly = bool(attrs & getattr(stat, "FILE_ATTRIBUTE_READONLY", 0))
            count += int(mode_readonly or win_readonly)
    except OSError:
        return -1
    return count


def longest_path_length(root: Path | None) -> int:
    if root is None or not root.exists():
        return 0
    longest = len(str(root.resolve()))
    try:
        for path in root.rglob("*"):
            longest = max(longest, len(str(path.resolve())))
    except OSError:
        pass
    return longest


def infer_stage(build_root: Path | None, out: Path) -> dict:
    py_logs = []
    executables = []
    manifests = []
    capsules = []
    zip_files = []
    staged_packages = []
    if build_root is not None:
        py_logs = files_named(build_root, "pyinstaller_host_stdout.txt") + files_named(build_root, "pyinstaller_host_stderr.txt")
        executables = files_named(build_root, "U001_Workbench.exe")
        manifests = files_named(build_root, "u001_portable_host_package_manifest.json")
        capsules = files_named(build_root, "u001_portable_host_package_capsule.txt")
        zip_files = any_suffix(build_root, ".zip")
        try:
            staged_packages = [path for path in build_root.rglob("U001_Portable_v0_2") if path.is_dir()]
        except OSError:
            staged_packages = []
    out_exe = files_named(out, "U001_Workbench.exe") if out.is_dir() else []
    out_manifest = files_named(out, "u001_portable_host_package_manifest.json") if out.is_dir() else []
    out_zip = any_suffix(out, ".zip") if out.is_dir() else []
    stage = 0
    if build_root is not None:
        stage = 1
    if py_logs:
        stage = 2
    if executables:
        stage = 3
    if staged_packages:
        stage = 4
    if manifests and capsules:
        stage = 5
    if zip_files:
        stage = 6
    if out.exists():
        stage = 7
    return {
        "Stage": stage,
        "PyLogs": len(py_logs),
        "Exe": int(bool(executables or out_exe)),
        "Manifest": int(bool(manifests or out_manifest)),
        "Zip": int(bool(zip_files or out_zip)),
        "StagedPackage": int(bool(staged_packages)),
    }


def write_report(path: Path, values: dict, clue: str) -> None:
    lines = [
        "G10D2C-D1 PERMISSION DIAGNOSTIC",
        "",
        *[f"{key}={value}" for key, value in values.items()],
        f"PermissionClue={clue}",
        "",
        "This report was created by the diagnostic only; no existing artifact was modified.",
    ]
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text("\n".join(lines) + "\n", encoding="utf-8")
    temporary.replace(path)


def run_diagnostic(udp001_root: Path, out: Path) -> int:
    root = udp001_root.resolve()
    temp_root = root / "04_TEMP"
    out = out.resolve()
    internal_errors = 0
    if not root.is_dir() or root.name.upper() != "UDP001" or not temp_root.is_dir():
        print("G10D2CD1-FAIL:UDP001_ROOT C-7")
        return 7
    try:
        build_root = latest_pk04(temp_root)
    except OSError:
        build_root = None
        internal_errors += 1
    stage = infer_stage(build_root, out)
    file_write, dir_write, write_detail = probe_own_write(temp_root)
    clue_found, clue = scan_permission_clue(build_root)
    readonly = readonly_file_count(build_root)
    longest = longest_path_length(build_root)
    free_mb = int(shutil.disk_usage(temp_root).free // (1024 * 1024))
    values = {
        "UDP001Root": str(root),
        "TempRoot": str(temp_root),
        "LatestPK04": str(build_root) if build_root else "NONE",
        "RequestedOut": str(out),
        "OutExists": int(out.exists()),
        "InferredStage": stage["Stage"],
        "PyInstallerLogs": stage["PyLogs"],
        "RuntimeExeObserved": stage["Exe"],
        "ManifestObserved": stage["Manifest"],
        "ZipObserved": stage["Zip"],
        "StagedPackageObserved": stage["StagedPackage"],
        "OwnFileWriteRenameDelete": file_write,
        "OwnDirectoryCreateRenameDelete": dir_write,
        "WriteProbeDetail": write_detail,
        "PermissionClueFound": clue_found,
        "ReadOnlyFileCount": readonly,
        "LongestPathCharacters": longest,
        "FreeSpaceMB": free_mb,
        "InternalDiagnosticErrors": internal_errors,
    }
    try:
        write_report(temp_root / REPORT_NAME, values, clue)
        report_written = 1
    except OSError:
        report_written = 0
        internal_errors += 1
    capsule = (
        f"G10D2CD1 N-R{int(build_root is not None)}-S{stage['Stage']}-O{int(out.exists())}"
        f"-E{stage['Exe']}-M{stage['Manifest']}-Z{stage['Zip']}"
        f"-W{file_write}-D{dir_write}-A{clue_found}-RO{readonly}"
        f"-L{longest}-Q{free_mb}-P{report_written}-X{internal_errors} C-0"
    )
    print(capsule)
    if clue_found:
        print("H-" + clue)
    return 0


def run_self_test() -> None:
    import tempfile

    checks = 0
    with tempfile.TemporaryDirectory() as td:
        base = Path(td)
        root = base / "UDP001"
        temp = root / "04_TEMP"
        temp.mkdir(parents=True)
        checks += int(latest_pk04(temp) is None)
        old = temp / "PK04_OLD"
        new = temp / "PK04_NEW"
        old.mkdir()
        time.sleep(0.01)
        new.mkdir()
        checks += int(latest_pk04(temp) == new)
        checks += int(files_named(new, "x.txt") == [])
        (new / "x.txt").write_text("PermissionError: denied", encoding="utf-8")
        checks += int(len(files_named(new, "x.txt")) == 1)
        found, clue = scan_permission_clue(new)
        checks += int(found == 1)
        checks += int("PermissionError" in clue)
        f_ok, d_ok, detail = probe_own_write(temp)
        checks += int(f_ok == 1)
        checks += int(d_ok == 1)
        checks += int(detail == "NONE")
        stage = infer_stage(new, temp / "OUT")
        checks += int(stage["Stage"] == 1)
        checks += int(stage["Exe"] == 0)
        checks += int(stage["Manifest"] == 0)
        checks += int(stage["Zip"] == 0)
        checks += int(readonly_file_count(new) >= 0)
        checks += int(longest_path_length(new) >= len(str(new.resolve())))
        checks += int(PROGRAM_VERSION == "0.1")
    if checks != 16:
        raise AssertionError(f"SELFTEST_COUNT:{checks}")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Diagnose G10D2C PermissionError without rerunning the build")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.udp001_root or not args.out:
        parser.error("--udp001-root and --out are required")
    try:
        return run_diagnostic(Path(args.udp001_root), Path(args.out))
    except Exception as exc:
        print(f"G10D2CD1-FAIL:UNHANDLED-{type(exc).__name__} C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
