class ProfileError(RuntimeError):
    """Represent one deterministic profiler failure."""


def machine_token(value: object, limit: int = 100) -> str:
    """Normalize one diagnostic value for a short return capsule."""

    token = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (token or "UNKNOWN")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize JSON deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def parse_optional_float(value: object) -> Optional[float]:
    """Return one finite number or None."""

    text = str(value if value is not None else "").strip()
    if not text:
        return None
    try:
        number = float(text)
    except ValueError:
        return None
    return number if math.isfinite(number) else None


def parse_boolean(value: object) -> bool:
    """Parse the controlled U001/U002 boolean vocabulary."""

    text = str(value if value is not None else "").strip().lower()
    if text in {"1", "true"}:
        return True
    if text in {"0", "false", ""}:
        return False
    raise ProfileError("BOOLEAN_" + machine_token(value))


def parse_integer(value: object, field: str) -> int:
    """Parse a finite integer encoded as integer or decimal text."""

    number = parse_optional_float(value)
    if number is None or not number.is_integer():
        raise ProfileError("INTEGER_" + machine_token(field))
    return int(number)


def normalize_can_id(value: object) -> str:
    """Normalize hexadecimal CAN identity for exact comparison."""

    text = str(value).strip().lower()
    try:
        if text.startswith("0x"):
            number = int(text, 16)
        else:
            number = int(text, 16)
    except ValueError as exc:
        raise ProfileError("CAN_ID_" + machine_token(value)) from exc
    return f"0x{number:X}".lower()


def unique_file(root: Path, name: str, reason: str) -> Path:
    """Resolve exactly one regular file by basename."""

    matches = sorted(path for path in root.rglob(name) if path.is_file())
    if len(matches) != 1:
        raise ProfileError(f"{reason}_COUNT_{len(matches)}")
    return matches[0]


def discover_run_root(workspace_root: Path) -> Path:
    """Choose the newest completed run containing S1 reconstruction evidence."""

    output_root = workspace_root / "07_OUTPUT"
    if not output_root.is_dir():
        raise ProfileError("OUTPUT_ROOT_MISSING")
    candidates: List[Tuple[int, Path]] = []
    for path in output_root.rglob("camera_target_reconstruction.csv"):
        if not path.is_file():
            continue
        current = path.parent
        while current.parent != output_root and output_root in current.parents:
            current = current.parent
        if current.parent == output_root:
            candidates.append((path.stat().st_mtime_ns, current))
    unique = {path.resolve(): stamp for stamp, path in candidates}
    if not unique:
        raise ProfileError("COMPLETED_RUN_NOT_FOUND")
    return max(unique, key=lambda path: unique[path])


def locate_input_package(run_root: Path) -> Path:
    """Read the frozen S1 request so the original U001 package is not guessed."""

    matches: List[Path] = []
    for request_path in run_root.rglob("module_request.json"):
        if not request_path.is_file():
            continue
        try:
            request = json.loads(request_path.read_text(encoding="utf-8-sig"))
        except Exception:
            continue
        if request.get("ModuleId") == "CameraRadar.S1.CameraTargetContinuityArbitration":
            candidate = Path(str(request.get("InputPackage", ""))).resolve()
            if candidate.is_dir():
                matches.append(candidate)
    identities = sorted({path for path in matches})
    if len(identities) != 1:
        raise ProfileError("S1_INPUT_PACKAGE_COUNT_" + str(len(identities)))
    return identities[0]


def resolve_signals(catalog_path: Path) -> Dict[str, str]:
    """Resolve exact Channel/CAN-ID/SignalName identities to CanonicalName."""

    with catalog_path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    required_columns = {"CanonicalName", "CAN_ID", "Channel", "SignalName"}
    if not rows or not required_columns.issubset(rows[0]):
        raise ProfileError("SIGNAL_CATALOG_SCHEMA")

    resolved: Dict[str, str] = {}
    for role, (channel, can_id, signal_name, required) in SIGNAL_REQUESTS.items():
        matches = [
            row
            for row in rows
            if str(row.get("Channel", "")).strip() == channel
            and normalize_can_id(row.get("CAN_ID", "")) == normalize_can_id(can_id)
            and str(row.get("SignalName", "")).strip().casefold()
            == signal_name.casefold()
        ]
        if len(matches) == 1:
            resolved[role] = str(matches[0]["CanonicalName"]).strip()
        elif required:
            raise ProfileError(f"SIGNAL_{role}_COUNT_{len(matches)}")
    return resolved


def group_value(
    row: Mapping[str, str],
    canonical_name: Optional[str],
) -> Tuple[Optional[float], bool, Optional[int], str]:
    """Read one U001 seven-column group without manufacturing values."""

    if not canonical_name:
        return None, False, None, "NOT_RESOLVED"
    value_column = canonical_name + "__value"
    valid_column = canonical_name + "__valid"
    age_column = canonical_name + "__age"
    reason_column = canonical_name + "__missing_reason"
    for column in (value_column, valid_column, age_column, reason_column):
        if column not in row:
            raise ProfileError("FRAME_VIEW_MISSING_" + machine_token(column))
    value = parse_optional_float(row[value_column])
    valid = parse_boolean(row[valid_column]) and value is not None
    age = parse_integer(row[age_column], age_column)
    return value, valid, age, str(row[reason_column])


def read_old_reconstruction(
    path: Path,
    frame_start: int,
    frame_end: int,
) -> Dict[int, Dict[str, str]]:
    """Load only the governed review range from the accepted S1 output."""

    result: Dict[int, Dict[str, str]] = {}
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {
            "FrameCount",
            "CanonicalCameraDistance",
            "CanonicalCameraValid",
            "SelectedCameraRole",
            "SelectionStatus",
            "SelectionReason",
        }
        if not required.issubset(reader.fieldnames or []):
            raise ProfileError("RECONSTRUCTION_SCHEMA")
        for row in reader:
            frame = parse_integer(row["FrameCount"], "FRAMECOUNT")
            if frame_start <= frame <= frame_end:
                if frame in result:
                    raise ProfileError("RECONSTRUCTION_DUPLICATE_FRAME")
                result[frame] = dict(row)
    return result


def bearing_degrees(lateral: Optional[float], distance: Optional[float]) -> Optional[float]:
    """Calculate one diagnostic observation angle, not a calibrated camera angle."""

    if lateral is None or distance is None or distance <= 0:
        return None
    return math.degrees(math.atan2(lateral, distance))


def number(value: Optional[float]) -> object:
    """Return stable CSV text while preserving missingness."""

    return "" if value is None else format(value, ".12g")


def build_rows(
    frame_view: Path,
    resolved: Mapping[str, str],
    old_rows: Mapping[int, Mapping[str, str]],
    frame_start: int,
    frame_end: int,
) -> List[Dict[str, object]]:
    """Build one compact paired-geometry evidence table."""

    output: List[Dict[str, object]] = []
    with frame_view.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        if not reader.fieldnames or reader.fieldnames[0] != "FrameCount":
            raise ProfileError("FRAME_VIEW_LEADING_COLUMN")
        for source in reader:
            frame = parse_integer(source["FrameCount"], "FRAMECOUNT")
            if frame < frame_start or frame > frame_end:
                continue
            d1, vd1, _, _ = group_value(source, resolved.get("CAMERA_1_DISTANCE"))
            l1, vl1, _, _ = group_value(
                source, resolved.get("CAMERA_1_LATERAL_POSITION")
            )
            d3, vd3, _, _ = group_value(source, resolved.get("CAMERA_3_DISTANCE"))
            l3, vl3, _, _ = group_value(
                source, resolved.get("CAMERA_3_LATERAL_POSITION")
            )
            rd, vrd, _, _ = group_value(source, resolved.get("RADAR_DISTANCE"))
            rf, vrf, _, _ = group_value(
                source, resolved.get("RADAR_DETECTION_FLAG")
            )
            old = old_rows.get(frame, {})
            old_valid = parse_boolean(old.get("CanonicalCameraValid", "0"))
            old_role = str(old.get("SelectedCameraRole", ""))
            selected_lateral: Optional[float] = None
            if old_role == "CAMERA_1_DISTANCE" and vl1:
                selected_lateral = l1
            elif old_role == "CAMERA_3_DISTANCE" and vl3:
                selected_lateral = l3

            preferred = ""
            advantage: Optional[float] = None
            status = "NO_PAIRED_GEOMETRY"
            if vd1 and vl1 and not (vd3 and vl3):
                preferred = "CAMERA_1_DISTANCE"
                status = "ONLY_CAMERA_1_GEOMETRY_VALID"
            elif vd3 and vl3 and not (vd1 and vl1):
                preferred = "CAMERA_3_DISTANCE"
                status = "ONLY_CAMERA_3_GEOMETRY_VALID"
            elif vd1 and vl1 and vd3 and vl3:
                abs_1 = abs(float(l1))
                abs_3 = abs(float(l3))
                advantage = abs(abs_1 - abs_3)
                if abs_1 < abs_3:
                    preferred = "CAMERA_1_DISTANCE"
                elif abs_3 < abs_1:
                    preferred = "CAMERA_3_DISTANCE"
                status = "BOTH_GEOMETRY_VALID"

            output.append(
                {
                    "FrameCount": frame,
                    "Camera1Distance": number(d1),
                    "Camera1DistanceValid": int(vd1),
                    "Camera1LateralPosition": number(l1),
                    "Camera1LateralValid": int(vl1),
                    "Camera1AbsLateral": number(abs(l1) if vl1 else None),
                    "Camera1BearingDeg": number(bearing_degrees(l1, d1)),
                    "Camera3Distance": number(d3),
                    "Camera3DistanceValid": int(vd3),
                    "Camera3LateralPosition": number(l3),
                    "Camera3LateralValid": int(vl3),
                    "Camera3AbsLateral": number(abs(l3) if vl3 else None),
                    "Camera3BearingDeg": number(bearing_degrees(l3, d3)),
                    "RadarDistance": number(rd),
                    "RadarDistanceValid": int(vrd),
                    "RadarDetectionFlag": number(rf if vrf else None),
                    "OldCanonicalCameraDistance": old.get(
                        "CanonicalCameraDistance", ""
                    ),
                    "OldCanonicalCameraValid": int(old_valid),
                    "OldSelectedCameraRole": old_role,
                    "OldSelectionStatus": old.get("SelectionStatus", ""),
                    "OldSelectionReason": old.get("SelectionReason", ""),
                    "OldSelectedLateralPosition": number(selected_lateral),
                    "OldSelectedAbsLateral": number(
                        abs(selected_lateral) if selected_lateral is not None else None
                    ),
                    "CenterlinePreferredRole": preferred,
                    "CenterlineAdvantageMeters": number(advantage),
                    "GeometryEvidenceStatus": status,
                }
            )
    if not output:
        raise ProfileError("FRAME_RANGE_EMPTY")
    return output


def threshold_sweep(
    rows: Sequence[Mapping[str, object]],
    thresholds: Sequence[float],
) -> List[Dict[str, object]]:
    """Measure policy consequences without choosing one production threshold."""

    output: List[Dict[str, object]] = []
    for threshold in thresholds:
        counts = {
            "Camera1AuthorizedFrames": 0,
            "Camera3AuthorizedFrames": 0,
            "ExactlyOneAuthorizedFrames": 0,
            "BothAuthorizedFrames": 0,
            "NoneAuthorizedFrames": 0,
            "OldCanonicalSelectedOutsideFrames": 0,
            "OldCanonicalInvalidExactlyOneAuthorizedFrames": 0,
            "OldSelectionDisagreesWithCenterlineFrames": 0,
        }
        for row in rows:
            l1 = parse_optional_float(row["Camera1LateralPosition"])
            l3 = parse_optional_float(row["Camera3LateralPosition"])
            a1 = bool(int(row["Camera1DistanceValid"])) and bool(
                int(row["Camera1LateralValid"])
            ) and l1 is not None and abs(l1) <= threshold
            a3 = bool(int(row["Camera3DistanceValid"])) and bool(
                int(row["Camera3LateralValid"])
            ) and l3 is not None and abs(l3) <= threshold
            counts["Camera1AuthorizedFrames"] += int(a1)
            counts["Camera3AuthorizedFrames"] += int(a3)
            counts["ExactlyOneAuthorizedFrames"] += int(a1 != a3)
            counts["BothAuthorizedFrames"] += int(a1 and a3)
            counts["NoneAuthorizedFrames"] += int(not a1 and not a3)
            selected_abs = parse_optional_float(row["OldSelectedAbsLateral"])
            old_valid = bool(int(row["OldCanonicalCameraValid"]))
            counts["OldCanonicalSelectedOutsideFrames"] += int(
                old_valid and selected_abs is not None and selected_abs > threshold
            )
            counts["OldCanonicalInvalidExactlyOneAuthorizedFrames"] += int(
                not old_valid and a1 != a3
            )
            preferred = str(row["CenterlinePreferredRole"])
            selected = str(row["OldSelectedCameraRole"])
            counts["OldSelectionDisagreesWithCenterlineFrames"] += int(
                old_valid and preferred != "" and selected != preferred
            )
        output.append(
            {
                "AbsLateralThresholdMeters": format(threshold, ".12g"),
                **counts,
            }
        )
    return output


def write_csv(
    path: Path,
    columns: Sequence[str],
    rows: Iterable[Mapping[str, object]],
) -> None:
    """Write one deterministic UTF-8 CSV."""

    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(columns), lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in columns})


def parse_thresholds(text: str) -> Tuple[float, ...]:
    """Parse a strictly increasing positive threshold list."""

    try:
        values = tuple(float(part.strip()) for part in text.split(","))
    except ValueError as exc:
        raise ProfileError("THRESHOLDS_FORMAT") from exc
    if not values or any(not math.isfinite(v) or v <= 0 for v in values):
        raise ProfileError("THRESHOLDS_VALUE")
    if tuple(sorted(set(values))) != values:
        raise ProfileError("THRESHOLDS_ORDER")
    return values


def execute(
    run_root: Path,
    output_dir: Path,
    frame_start: int,
    frame_end: int,
    thresholds: Sequence[float],
) -> str:
    """Execute one read-only evidence profile and return its capsule."""

    if not run_root.is_dir():
        raise ProfileError("RUN_ROOT_MISSING")
    if frame_start > frame_end:
        raise ProfileError("FRAME_RANGE_ORDER")
    input_package = locate_input_package(run_root)
    frame_view = unique_file(input_package, "u001_frame_values.csv", "FRAME_VIEW")
    catalog = unique_file(input_package, "u001_signal_catalog.csv", "SIGNAL_CATALOG")
    reconstruction = unique_file(
        run_root,
        "camera_target_reconstruction.csv",
        "RECONSTRUCTION",
    )
    resolved = resolve_signals(catalog)
    old_rows = read_old_reconstruction(reconstruction, frame_start, frame_end)
    rows = build_rows(
        frame_view,
        resolved,
        old_rows,
        frame_start,
        frame_end,
    )
    sweep = threshold_sweep(rows, thresholds)

    output_dir.mkdir(parents=True, exist_ok=True)
    row_path = output_dir / "b7a_camera_lateral_evidence_rows.csv"
    sweep_path = output_dir / "b7a_abs_lateral_threshold_sweep.csv"
    quality_path = output_dir / "b7a_camera_lateral_evidence_quality.json"
    capsule_path = output_dir / "b7a_return_capsule.txt"
    write_csv(row_path, OUTPUT_COLUMNS, rows)
    sweep_columns = tuple(sweep[0].keys())
    write_csv(sweep_path, sweep_columns, sweep)

    paired = sum(row["GeometryEvidenceStatus"] == "BOTH_GEOMETRY_VALID" for row in rows)
    old_invalid = sum(not int(row["OldCanonicalCameraValid"]) for row in rows)
    disagreement = sum(
        int(row["OldCanonicalCameraValid"])
        and row["CenterlinePreferredRole"]
        and row["CenterlinePreferredRole"] != row["OldSelectedCameraRole"]
        for row in rows
    )
    quality = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "CompletionCode": 0,
        "DecisionBoundary": (
            "READ_ONLY_EVIDENCE_PROFILE_NOT_A_PRODUCTION_GEOMETRY_POLICY"
        ),
        "RunRoot": str(run_root.resolve()),
        "InputPackage": str(input_package.resolve()),
        "FrameStart": frame_start,
        "FrameEnd": frame_end,
        "Rows": len(rows),
        "PairedGeometryValidRows": paired,
        "OldCanonicalInvalidRows": old_invalid,
        "OldSelectionCenterlineDisagreementRows": disagreement,
        "ResolvedCanonicalNames": dict(sorted(resolved.items())),
        "ThresholdsMeters": list(thresholds),
        "InputHashes": {
            "SignalCatalogSha256": sha256_file(catalog),
            "FrameViewSha256": sha256_file(frame_view),
            "OldReconstructionSha256": sha256_file(reconstruction),
        },
        "OutputHashes": {
            row_path.name: sha256_file(row_path),
            sweep_path.name: sha256_file(sweep_path),
        },
        "Prohibitions": [
            "NO_U001_MUTATION",
            "NO_ACCEPTED_U002_MUTATION",
            "NO_THRESHOLD_PROMOTION",
            "NO_RADAR_INPUT_TO_CAMERA_ARBITRATION",
            "NO_GROUND_TRUTH_CLAIM",
        ],
    }
    quality_path.write_bytes(canonical_bytes(quality))
    capsule = (
        f"B7A N-F{len(rows)}-G{paired}-I{old_invalid}-D{disagreement}"
        f"-S{len(sweep)}-X0 C-0"
    )
    capsule_path.write_text(capsule + "\n", encoding="utf-8")
    return capsule


def self_test() -> None:
    """Exercise discovery, exact signal resolution and threshold evidence."""

    with tempfile.TemporaryDirectory(prefix="u22b7a_selftest_") as temporary:
        root = Path(temporary) / "U-UDP002"
        input_package = root / "03_CASES/TC001/U001_OUTPUT"
        input_package.mkdir(parents=True)
        run_root = root / "07_OUTPUT/RUN_B7A_SELFTEST"
        request_root = run_root / "00_STAGING/0002_S1"
        request_root.mkdir(parents=True)
        (request_root / "module_request.json").write_bytes(
            canonical_bytes(
                {
                    "ModuleId": "CameraRadar.S1.CameraTargetContinuityArbitration",
                    "InputPackage": str(input_package),
                }
            )
        )
        catalog_columns = ("CanonicalName", "CAN_ID", "Channel", "SignalName")
        catalog_rows = [
            ("C1D", "0x221", "2", "Distance"),
            ("C1L", "0x221", "2", "LateralPosition"),
            ("C3D", "0x251", "2", "Distance"),
            ("C3L", "0x251", "2", "LateralPosition"),
            ("RD", "0x380", "1", "DistanceToObject"),
            ("RF", "0x380", "1", "PrecedingVehicleDetectionFlag"),
        ]
        with (input_package / "u001_signal_catalog.csv").open(
            "w", encoding="utf-8", newline=""
        ) as stream:
            writer = csv.writer(stream, lineterminator="\n")
            writer.writerow(catalog_columns)
            writer.writerows(catalog_rows)

        suffixes = ("__value", "__age", "__valid", "__missing_reason")
        header = ["FrameCount"]
        for canonical_name, _, _, _ in catalog_rows:
            header.extend(canonical_name + suffix for suffix in suffixes)
        with (input_package / "u001_frame_values.csv").open(
            "w", encoding="utf-8", newline=""
        ) as stream:
            writer = csv.writer(stream, lineterminator="\n")
            writer.writerow(header)
            for frame in range(100, 104):
                row: List[object] = [frame]
                values = (
                    (40.0, 0, 1, ""),
                    (-2.7, 0, 1, ""),
                    (16.0, 0, 1, ""),
                    (0.1, 0, 1, ""),
                    (15.8, 0, 1, ""),
                    (1.0, 0, 1, ""),
                )
                for value in values:
                    row.extend(value)
                writer.writerow(row)

        reconstruction_root = run_root / "03_ARTIFACTS/S1/ART_S1"
        reconstruction_root.mkdir(parents=True)
        reconstruction_columns = (
            "FrameCount",
            "CanonicalCameraDistance",
            "CanonicalCameraValid",
            "SelectedCameraRole",
            "SelectionStatus",
            "SelectionReason",
        )
        with (reconstruction_root / "camera_target_reconstruction.csv").open(
            "w", encoding="utf-8", newline=""
        ) as stream:
            writer = csv.DictWriter(
                stream,
                fieldnames=reconstruction_columns,
                lineterminator="\n",
            )
            writer.writeheader()
            for frame in range(100, 104):
                writer.writerow(
                    {
                        "FrameCount": frame,
                        "CanonicalCameraDistance": 40.0,
                        "CanonicalCameraValid": 1,
                        "SelectedCameraRole": "CAMERA_1_DISTANCE",
                        "SelectionStatus": "CONFIDENT",
                        "SelectionReason": "SYNTHETIC",
                    }
                )
        output_dir = root / "04_TEMP/B7A_OUTPUT"
        capsule = execute(
            run_root,
            output_dir,
            100,
            103,
            (1.0, 2.0, 3.0),
        )
        assert capsule == "B7A N-F4-G4-I0-D4-S3-X0 C-0"
        rows = list(
            csv.DictReader(
                (output_dir / "b7a_camera_lateral_evidence_rows.csv").open(
                    "r", encoding="utf-8"
                )
            )
        )
        assert all(row["CenterlinePreferredRole"] == "CAMERA_3_DISTANCE" for row in rows)
        assert rows[0]["OldSelectedLateralPosition"] == "-2.7"
        sweep = list(
            csv.DictReader(
                (output_dir / "b7a_abs_lateral_threshold_sweep.csv").open(
                    "r", encoding="utf-8"
                )
            )
        )
        assert sweep[0]["ExactlyOneAuthorizedFrames"] == "4"
        assert sweep[-1]["BothAuthorizedFrames"] == "4"
    print("UDP002-U22B7A-SELFTEST-PASS-18 C-0")

    parser.add_argument("--frame-end", type=int, default=41780)
    parser.add_argument(
        "--thresholds",
        default=",".join(format(value, ".12g") for value in DEFAULT_THRESHOLDS),
    )
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            self_test()
            return 0
        if not args.output_dir:
            raise ProfileError("OUTPUT_DIR_REQUIRED")
        if args.run_root:
            run_root = Path(args.run_root).resolve()
        elif args.workspace_root:
            run_root = discover_run_root(Path(args.workspace_root).resolve())
        else:
            raise ProfileError("RUN_OR_WORKSPACE_ROOT_REQUIRED")
        capsule = execute(
            run_root,
            Path(args.output_dir).resolve(),
            args.frame_start,
            args.frame_end,
            parse_thresholds(args.thresholds),
        )
        print(capsule)
        return 0
    except Exception as error:
        print(
            "B7A N-X1 C-7 K-"
            + machine_token(getattr(error, "args", [type(error).__name__])[0]),
            file=sys.stderr,
        )
        return 7
