#!/usr/bin/env python3
"""Install and execute the U001 H2 hexadecimal snapshot-semantics correction.

This sprint controller has one controlled lifecycle:

    prepare -> self-test -> execute

It installs Adapter v1.2.0, reruns the accepted H6B4 full-source H2 decode,
then immediately rebuilds the H6B5/H3 semantic FrameCount package.  It does
not perform H4 hold/reconstruction, business arbitration, or analysis.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import shutil
import subprocess
import sys
import zipfile
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Optional, Sequence


PROGRAM_ID = "U001_H2_HEX_SNAPSHOT_SPRINT_INSTALLER"
PROGRAM_VERSION = "1.0"
ZIP_NAME = "UDP001_H2_HEX_SNAPSHOT_SPRINT_v1_0.zip"
PREPARED_FOLDER = "U1SP1_PREPARED"
OUTPUT_FOLDER = "U1SP1_OUTPUT"
EXPECTED_ADAPTER_VERSION = "1.2.0"
EXPECTED_FRAME_COUNT = 56596

PAYLOAD_FILES = {
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.2.0/adapter.py",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.2.0/adapter_card.json",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.2.0/u001_matlab_can_csv_dbc_core_v1_0.py",
    "04_TEMP/u001_h6b4_real_h2_decode_v1_0.py.txt",
    "04_TEMP/u001_h6b5_h3_semantic_frame_compile_v1_0.py.txt",
}


class SprintError(RuntimeError):
    """Expected fail-closed sprint error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def write_json(path: Path, value: dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> dict[str, Any]:
    if not path.is_file():
        raise SprintError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise SprintError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise SprintError(label + "_ROOT_NOT_OBJECT")
    return value


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(4 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise SprintError("U_UDP001_ROOT_NOT_FOUND")
    return root


def locate_transfer_file(root: Path, name: str) -> Path:
    candidates = (
        root / "04_TEMP" / name,
        Path(__file__).resolve().parent / name,
        Path.cwd().resolve() / name,
    )
    for path in candidates:
        if path.is_file():
            return path
    raise SprintError("TRANSFER_FILE_NOT_FOUND:" + name)


def package_inventory(zip_path: Path) -> dict[str, str]:
    try:
        with zipfile.ZipFile(zip_path, "r") as archive:
            members = {
                name[len("payload/") :]: name
                for name in archive.namelist()
                if name.startswith("payload/") and not name.endswith("/")
            }
            if set(members) != PAYLOAD_FILES:
                missing = sorted(PAYLOAD_FILES - set(members))
                extra = sorted(set(members) - PAYLOAD_FILES)
                raise SprintError(f"PACKAGE_INVENTORY:MISSING={missing}:EXTRA={extra}")
            return {
                relative: hashlib.sha256(archive.read(member)).hexdigest()
                for relative, member in sorted(members.items())
            }
    except zipfile.BadZipFile as exc:
        raise SprintError("PACKAGE_BAD_ZIP") from exc


def extract_payload(zip_path: Path, destination: Path) -> Path:
    if destination.exists():
        shutil.rmtree(destination)
    destination.mkdir(parents=True)
    with zipfile.ZipFile(zip_path, "r") as archive:
        archive.extractall(destination)
    payload = destination / "payload"
    if not payload.is_dir():
        raise SprintError("EXTRACTED_PAYLOAD_NOT_FOUND")
    return payload


def explain_required_root(root: Path) -> None:
    checks = {
        "U001 program root": root / "01_PROGRAMS",
        "H6B3 compiled profile": root / "04_TEMP" / "U1H6B3_OUTPUT",
        "H3 high-code core": root / "01_PROGRAMS" / "core" / "u001_h3_sync_semantic_engine_v1_0.py",
    }
    reasons = {
        "U001 program root": "Adapter v1.2.0 is installed into the formal U001 registry.",
        "H6B3 compiled profile": "Preserves the approved 22-signal selection and CH2/0x620 FrameCount key.",
        "H3 high-code core": "Builds the semantic FrameCount-domain package after H2 succeeds.",
    }
    missing = []
    for label, path in checks.items():
        found = path.is_dir() if label != "H3 high-code core" else path.is_file()
        print(f"[CHECK] {label} | {'FOUND' if found else 'MISSING'} | Why: {reasons[label]}")
        if not found:
            missing.append(label)
    if missing:
        raise SprintError("ROOT_DEPENDENCY_MISSING:" + ".".join(missing))


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        temp = root / "04_TEMP"
        prepared = temp / PREPARED_FOLDER
        output = temp / OUTPUT_FOLDER
        zip_path = locate_transfer_file(root, ZIP_NAME)

        print("[PREPARE] U001 H2 hexadecimal/snapshot correction sprint")
        print("[PREPARE] Why: replace decimal byte interpretation and dense-snapshot duplication in one controlled rerun.")
        if prepared.exists():
            shutil.rmtree(prepared)
            print(f"[CLEAN] 04_TEMP/{PREPARED_FOLDER} | REMOVED | Why: stale prepare receipt cannot authorize execution.")
        if output.exists():
            shutil.rmtree(output)
            print(f"[CLEAN] 04_TEMP/{OUTPUT_FOLDER} | REMOVED | Why: stale sprint diagnostics cannot be reused.")
        explain_required_root(root)
        hashes = package_inventory(zip_path)
        payload = extract_payload(zip_path, prepared / "staged")
        for relative in sorted(PAYLOAD_FILES):
            path = payload / relative
            print(f"[CHECK] package/{relative} | FOUND | Why: controlled sprint payload.")
            if sha256_file(path) != hashes[relative]:
                raise SprintError("STAGED_HASH_MISMATCH:" + relative)
        receipt = {
            "ReceiptSchema": "U001_H2_HEX_SNAPSHOT_SPRINT_PREPARE_V1_0",
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "PreparedUtc": utc_now(),
            "PackageBasename": zip_path.name,
            "PackageSHA256": sha256_file(zip_path),
            "PayloadSHA256": hashes,
            "Status": "READY",
            "CompletionCode": 0,
        }
        write_json(prepared / "prepare_receipt.json", receipt)
        print("U1SP1P-N5-A12-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1SP1P-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def run_checked(command: list[str], cwd: Path, label: str) -> None:
    print(f"[RUN] {label}")
    result = subprocess.run(command, cwd=str(cwd), check=False)
    if result.returncode != 0:
        raise SprintError(f"{label}_EXIT_{result.returncode}")


def require_prepared(root: Path) -> tuple[Path, dict[str, Any]]:
    prepared = root / "04_TEMP" / PREPARED_FOLDER
    receipt = read_json(prepared / "prepare_receipt.json", "PREPARE_RECEIPT")
    zip_path = locate_transfer_file(root, ZIP_NAME)
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise SprintError("PREPARE_RECEIPT_NOT_READY")
    if receipt.get("PackageSHA256") != sha256_file(zip_path):
        raise SprintError("PACKAGE_CHANGED_AFTER_PREPARE")
    payload = prepared / "staged" / "payload"
    hashes = receipt.get("PayloadSHA256")
    if not isinstance(hashes, dict) or set(hashes) != PAYLOAD_FILES:
        raise SprintError("PREPARE_HASH_INVENTORY")
    for relative, expected in hashes.items():
        path = payload / relative
        if not path.is_file() or sha256_file(path) != expected:
            raise SprintError("PREPARED_PAYLOAD_CHANGED:" + relative)
    return payload, receipt


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        payload, _ = require_prepared(root)
        adapter_dir = payload / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc" / "1.2.0"
        core = adapter_dir / "u001_matlab_can_csv_dbc_core_v1_0.py"
        adapter = adapter_dir / "adapter.py"
        h6b4 = payload / "04_TEMP" / "u001_h6b4_real_h2_decode_v1_0.py.txt"
        h6b5 = payload / "04_TEMP" / "u001_h6b5_h3_semantic_frame_compile_v1_0.py.txt"

        print("[SELFTEST] Hexadecimal byte parser | Why: digit-only CAN bytes such as 10 must mean 0x10, never decimal 10.")
        run_checked([sys.executable, str(core), "--self-test"], adapter_dir, "H2_CORE_SELFTEST")
        print("[SELFTEST] Snapshot-state adapter | Why: repeated retained-buffer rows must not become fake CAN updates.")
        run_checked([sys.executable, str(adapter), "--self-test"], adapter_dir, "H2_ADAPTER_SELFTEST")
        print("[SELFTEST] Orchestration syntax | Why: both full-source stages must be executable before installation.")
        for path in (h6b4, h6b5):
            compile(path.read_text(encoding="utf-8"), str(path), "exec")
        print("U1SP1S-H26-A38-O2-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1SP1S-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def install_payload(root: Path, payload: Path) -> None:
    target_adapter = root / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc" / EXPECTED_ADAPTER_VERSION
    source_adapter = payload / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc" / EXPECTED_ADAPTER_VERSION
    if target_adapter.exists():
        shutil.rmtree(target_adapter)
    target_adapter.parent.mkdir(parents=True, exist_ok=True)
    shutil.copytree(source_adapter, target_adapter)

    for name in (
        "u001_h6b4_real_h2_decode_v1_0.py.txt",
        "u001_h6b5_h3_semantic_frame_compile_v1_0.py.txt",
    ):
        shutil.copy2(payload / "04_TEMP" / name, root / "04_TEMP" / name)


def execute(root_argument: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        payload, receipt = require_prepared(root)
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            shutil.rmtree(output)
        output.mkdir(parents=True)

        install_payload(root, payload)
        adapter_dir = root / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc" / EXPECTED_ADAPTER_VERSION
        installed_adapter = adapter_dir / "adapter.py"
        if sha256_file(installed_adapter) != receipt["PayloadSHA256"][
            "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.2.0/adapter.py"
        ]:
            raise SprintError("INSTALLED_ADAPTER_HASH")

        h6b4 = root / "04_TEMP" / "u001_h6b4_real_h2_decode_v1_0.py.txt"
        h6b5 = root / "04_TEMP" / "u001_h6b5_h3_semantic_frame_compile_v1_0.py.txt"
        stages = (
            (h6b4, "--prepare", "H6B4_PREPARE"),
            (h6b4, "--self-test", "H6B4_SELFTEST"),
            (h6b4, "--execute", "H6B4_EXECUTE"),
            (h6b5, "--prepare", "H6B5_PREPARE"),
            (h6b5, "--self-test", "H6B5_SELFTEST"),
            (h6b5, "--execute", "H6B5_EXECUTE"),
        )
        for script, action, label in stages:
            run_checked([sys.executable, str(script), action, "--root", str(root)], root, label)

        h2 = read_json(root / "04_TEMP" / "U1H6B4_OUTPUT" / "h2_real" / "u001_adapter_quality.json", "H2_QUALITY")
        h2_manifest = read_json(root / "04_TEMP" / "U1H6B4_OUTPUT" / "h2_real" / "u001_adapter_run_manifest.json", "H2_MANIFEST")
        h3 = read_json(root / "04_TEMP" / "U1H6B5_OUTPUT" / "u001_h6b5_result.json", "H3_RESULT")

        facts = {
            "AdapterVersion": h2.get("AdapterVersion"),
            "RawByteEncoding": h2.get("RawByteEncoding"),
            "ObservationSemantics": h2.get("ObservationSemantics"),
            "SourceRows": int(h2.get("SourceRowCount", -1)),
            "DecodedUpdates": int(h2.get("DecodedUpdateCount", -1)),
            "FrameKeyUpdates": int(h2.get("FrameKeyUpdateCount", -1)),
            "RepeatedSnapshots": int(h2.get("RepeatedSnapshotObservationCount", -1)),
            "H3DistinctFrames": int(h3.get("DistinctFrames", -1)),
            "H3ValidUpdates": int(h3.get("ValidUpdates", -1)),
            "H3InvalidUpdates": int(h3.get("InvalidUpdates", -1)),
        }
        if facts["AdapterVersion"] != EXPECTED_ADAPTER_VERSION:
            raise SprintError("ADAPTER_VERSION_NOT_1_2_0")
        if facts["RawByteEncoding"] != "HEX_TEXT":
            raise SprintError("RAW_BYTE_ENCODING_NOT_HEX")
        if facts["ObservationSemantics"] != "SNAPSHOT_STATE_CHANGE":
            raise SprintError("OBSERVATION_SEMANTICS_NOT_STATE_CHANGE")
        if h2_manifest.get("Configuration", {}).get("ObservationSemantics") != "SNAPSHOT_STATE_CHANGE":
            raise SprintError("MANIFEST_OBSERVATION_SEMANTICS")
        if facts["FrameKeyUpdates"] != EXPECTED_FRAME_COUNT:
            raise SprintError(f"FRAME_KEY_COUNT_{facts['FrameKeyUpdates']}_EXPECTED_{EXPECTED_FRAME_COUNT}")
        if facts["H3DistinctFrames"] != EXPECTED_FRAME_COUNT:
            raise SprintError(f"H3_FRAME_COUNT_{facts['H3DistinctFrames']}_EXPECTED_{EXPECTED_FRAME_COUNT}")

        result = {
            "ResultSchema": "U001_H2_HEX_SNAPSHOT_SPRINT_RESULT_V1_0",
            "GeneratedUtc": utc_now(),
            **facts,
            "NextRequiredAction": "H6B6_BUILD_RAW_FRAME_VIEW_AND_BOUNDED_HOLD",
            "Status": "HEX_AND_SNAPSHOT_SEMANTICS_ACCEPTED_THROUGH_H3",
            "CompletionCode": 0,
        }
        write_json(output / "u001_h2_hex_snapshot_sprint_result.json", result)
        capsule = (
            f"U1SP1-A12-R{facts['SourceRows']}-U{facts['DecodedUpdates']}"
            f"-K{facts['FrameKeyUpdates']}-F{facts['H3DistinctFrames']}"
            f"-V{facts['H3ValidUpdates']}-I{facts['H3InvalidUpdates']}-X0-C0"
        )
        (output / "u001_h2_hex_snapshot_sprint_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        diagnostic = {
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "GeneratedUtc": utc_now(),
            "ExceptionType": type(exc).__name__,
            "Detail": str(exc)[:8000],
            "Recovery": "Return only the U1SP1E capsule; detailed child evidence remains in U1H6B4_OUTPUT/U1H6B5_OUTPUT.",
        }
        write_json(output / "u001_h2_hex_snapshot_sprint_diagnostic.json", diagnostic)
        print(f"U1SP1E-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H2 hexadecimal snapshot correction sprint")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    raise SystemExit(main())
