#!/usr/bin/env python3
"""Independent acceptance and installer for the U22A2 Module Registry."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile
from typing import Dict, List, Sequence


PROGRAM_ID = "UDP002-U22A2-MODULE-REGISTRY-ACCEPTANCE"
VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"
STAGES = ("S0", "S1", "S2", "S3", "S4", "S5", "S6", "S7")
CANDIDATE = "u002_u22_module_registry_v0_1.py.txt"
INSTALLED = "u002_u22_module_registry_v0_1.py"
VALIDATOR = "u002_u22_contract_validator_v0_1.py"


class AcceptanceError(RuntimeError):
    pass


def canonical(value: object) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2, allow_nan=False) + "\n"


def regular(path: Path) -> bool:
    return path.is_file() and not path.is_symlink()


def run(command: List[str], timeout: int = 30) -> subprocess.CompletedProcess:
    return subprocess.run(command, text=True, capture_output=True, timeout=timeout, check=False)


def ck(value: bool, reason: str, checks: List[bool]) -> None:
    if not value:
        raise AcceptanceError(reason)
    checks.append(True)


def algorithm(module_id: str, version: str, status: str = "PROVISIONAL") -> Dict[str, object]:
    return {
        "ContractVersion": CONTRACT_VERSION,
        "AlgorithmId": module_id,
        "AlgorithmVersion": version,
        "Status": status,
        "Purpose": "Synthetic registry acceptance fixture",
        "ProblemSolved": "Verify fail-closed discovery without code execution",
        "MathematicalDefinition": ["y = x"],
        "Inputs": ["x"],
        "Outputs": ["y"],
        "Assumptions": ["synthetic finite input"],
        "FailureConditions": ["missing input"],
        "Limitations": ["acceptance fixture only"],
        "Alternatives": ["none"],
        "EvaluationPlan": ["compare expected registry status"],
        "DecisionReference": "D-U22A2",
    }


def manifest(module_id: str, version: str, stage: str = "S1",
             contract: str = CONTRACT_VERSION) -> Dict[str, object]:
    return {
        "ContractVersion": contract,
        "ModuleId": module_id,
        "ModuleVersion": version,
        "Stage": stage,
        "AlgorithmStatus": "PROVISIONAL",
        "ScopeSelector": {},
        "RequiredInputs": ["Signal.A"],
        "OptionalInputs": [],
        "RequiresArtifacts": ["INPUT_VIEW"],
        "ProvidesArtifacts": ["OUTPUT_VIEW." + module_id],
        "OutputNamespace": module_id.lower(),
        "ConfigurationSchema": "configuration_schema.json",
        "FailurePolicy": "REQUIRED",
        "ConflictPolicy": "EXCLUSIVE",
        "EntryPoint": "implementation.py",
        "SelfTestEntryPoint": "selftest.py",
        "AlgorithmCard": "algorithm_card.json",
    }


def write_module(root: Path, module_id: str, version: str, *, mode: str) -> Path:
    directory = root / "01_PROGRAMS/modules/S1" / module_id / version
    directory.mkdir(parents=True)
    contract = "U22.9.9" if mode == "INCOMPATIBLE" else CONTRACT_VERSION
    value = manifest(module_id, version, contract=contract)
    if mode == "INVALID":
        value["Unexpected"] = True
    (directory / "module_manifest.json").write_text(canonical(value), encoding="utf-8")
    (directory / "algorithm_card.json").write_text(canonical(algorithm(module_id, version)), encoding="utf-8")
    (directory / "configuration_schema.json").write_text(canonical({"type": "object"}), encoding="utf-8")
    sentinel = root / "implementation_was_executed.txt"
    (directory / "implementation.py").write_text(
        "from pathlib import Path\nPath(" + repr(str(sentinel)) + ").write_text('BAD')\n",
        encoding="utf-8",
    )
    (directory / "selftest.py").write_text("raise RuntimeError('SELFTEST_MUST_NOT_RUN')\n", encoding="utf-8")
    if mode == "MISSING":
        (directory / "implementation.py").unlink()
    return directory


def copy_contract_environment(source: Path, target: Path) -> None:
    target.mkdir(parents=True)
    for stage in STAGES:
        (target / "01_PROGRAMS/modules" / stage).mkdir(parents=True)
    (target / "01_PROGRAMS/core").mkdir(parents=True)
    (target / "00_CONTROL/contracts").mkdir(parents=True)
    (target / "00_CONTROL/registry").mkdir(parents=True)
    source_contracts = source / "00_CONTROL/contracts"
    names = (
        "u002_module_manifest_schema_v0_1.json",
        "u002_artifact_manifest_schema_v0_1.json",
        "u002_run_configuration_schema_v0_1.json",
        "u002_algorithm_card_schema_v0_1.json",
    )
    for name in names:
        src = source_contracts / name
        if not regular(src):
            raise AcceptanceError("CONTRACT_MISSING_" + name)
        shutil.copy2(src, target / "00_CONTROL/contracts" / name)
    validator = source / "01_PROGRAMS/core" / VALIDATOR
    if not regular(validator):
        raise AcceptanceError("VALIDATOR_MISSING")
    shutil.copy2(validator, target / "01_PROGRAMS/core" / VALIDATOR)


def archive_previous(root: Path) -> int:
    archive = root / "A-Archive/U22_COMPLETED/U22A1/04_TEMP"
    moved = 0
    for name in (
        "u002_u22_contract_validator_v0_1.py.txt",
        "udp002_u22a1_contract_validator_acceptance_v0_1.py.txt",
    ):
        source = root / "04_TEMP" / name
        if not regular(source):
            continue
        archive.mkdir(parents=True, exist_ok=True)
        destination = archive / name
        if destination.exists():
            if destination.read_bytes() != source.read_bytes():
                raise AcceptanceError("ARCHIVE_CONFLICT")
            source.unlink()
        else:
            os.replace(source, destination)
        moved += 1
    return moved


def main(argv: Sequence[str] = None) -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    checks: List[bool] = []
    try:
        root = Path(args.workspace_root).resolve()
        ck(root.is_dir() and root.name == "U-UDP002", "WORKSPACE_ROOT", checks)
        candidate = root / "04_TEMP" / CANDIDATE
        ck(regular(candidate), "CANDIDATE_MISSING", checks)
        validator = root / "01_PROGRAMS/core" / VALIDATOR
        ck(regular(validator), "VALIDATOR_MISSING", checks)
        result = run([sys.executable, str(validator), "--self-test", "--workspace-root", str(root)])
        ck(result.returncode == 0 and "C-0" in result.stdout, "VALIDATOR_SELFTEST", checks)
        source = candidate.read_text(encoding="utf-8")
        ck("manifest_paths(root)" in source and "inspect_manifest(path, root, validator)" in source,
           "DISCOVERY_PATH_AUDIT", checks)
        ck("spec_from_file_location(\"u002_u22_contract_validator\"" in source,
           "CONTROLLED_VALIDATOR_IMPORT", checks)
        ck("exec(" not in source and "eval(" not in source, "DYNAMIC_EXECUTION", checks)
        with tempfile.TemporaryDirectory(prefix="u22a2_acceptance_") as temporary:
            test_root = Path(temporary) / "U-UDP002"
            copy_contract_environment(root, test_root)
            test_candidate = test_root / "01_PROGRAMS/core" / INSTALLED
            shutil.copy2(candidate, test_candidate)
            write_module(test_root, "Synthetic.Valid", "0.1.0", mode="VALID")
            write_module(test_root, "Synthetic.Missing", "0.1.0", mode="MISSING")
            write_module(test_root, "Synthetic.Invalid", "0.1.0", mode="INVALID")
            write_module(test_root, "Synthetic.Incompatible", "0.1.0", mode="INCOMPATIBLE")
            selftest = run([sys.executable, str(test_candidate), "--self-test",
                            "--workspace-root", str(test_root)])
            ck(selftest.returncode == 0 and "N-T24" in selftest.stdout, "CANDIDATE_SELFTEST", checks)
            refresh = run([sys.executable, str(test_candidate), "--refresh",
                           "--workspace-root", str(test_root)])
            ck(refresh.returncode == 0 and "C-0" in refresh.stdout, "REFRESH_EXIT", checks)
            snapshot_path = test_root / "00_CONTROL/registry/u002_module_registry_snapshot_v0_1.json"
            ck(regular(snapshot_path), "SNAPSHOT_MISSING", checks)
            snapshot = json.loads(snapshot_path.read_text(encoding="utf-8"))
            ck(snapshot["DiscoveredCount"] == 4, "DISCOVERED_COUNT", checks)
            counts = snapshot["StatusCounts"]
            ck(counts["SELFTEST_REQUIRED"] == 1, "SELFTEST_STATUS", checks)
            ck(counts["DEPENDENCY_MISSING"] == 1, "MISSING_STATUS", checks)
            ck(counts["INVALID_MANIFEST"] == 1, "INVALID_STATUS", checks)
            ck(counts["INCOMPATIBLE"] == 1, "INCOMPATIBLE_STATUS", checks)
            ck(counts["AVAILABLE"] == 0 and counts["QUARANTINED"] == 0,
               "PREMATURE_AVAILABILITY", checks)
            ck(not (test_root / "implementation_was_executed.txt").exists(),
               "IMPLEMENTATION_EXECUTED", checks)
            ck(snapshot["DiscoveryBehavior"]["ImportsModuleImplementation"] is False,
               "DISCOVERY_DECLARATION", checks)
            paths = [entry["ManifestPath"] for entry in snapshot["Entries"]]
            ck(paths == sorted(paths, key=str.casefold), "DETERMINISTIC_ORDER", checks)
            ck(all(entry["RegistryStatus"] in (
                "AVAILABLE", "INCOMPATIBLE", "INVALID_MANIFEST", "DEPENDENCY_MISSING",
                "SELFTEST_REQUIRED", "QUARANTINED") for entry in snapshot["Entries"]),
               "STATUS_DOMAIN", checks)
            ck(refresh.stdout.strip().startswith("U22A2R N-D4-A0-I1-M1-S1-Q0-K1"),
               "CAPSULE_CONTENT", checks)
        installed = root / "01_PROGRAMS/core" / INSTALLED
        if installed.exists():
            ck(regular(installed) and installed.read_bytes() == candidate.read_bytes(),
               "INSTALL_CONFLICT", checks)
        else:
            temporary_install = installed.with_name(installed.name + ".installing")
            ck(not temporary_install.exists(), "INSTALL_TEMP_EXISTS", checks)
            temporary_install.write_bytes(candidate.read_bytes())
            os.replace(temporary_install, installed)
            ck(installed.read_bytes() == candidate.read_bytes(), "INSTALL_IDENTITY", checks)
        post = run([sys.executable, str(installed), "--self-test", "--workspace-root", str(root)])
        ck(post.returncode == 0 and "N-T24" in post.stdout, "INSTALLED_SELFTEST", checks)
        ck(len(checks) == 24, "CHECK_COUNT_PRE", checks)
        if len(checks) != 25:
            raise AcceptanceError("CHECK_COUNT")
        moved = archive_previous(root)
        print(f"U22A2 N-D4-S1-M1-I1-K1-T25-P1-A{moved}-X0 C-0")
        return 0
    except Exception as exc:
        reason = "".join(ch if ch.isalnum() else "_" for ch in str(exc).upper()).strip("_")[:80]
        print(f"U22A2 N-D0-S0-M0-I0-K0-T0-P0-A0-X1 C-7 R-{reason or exc.__class__.__name__.upper()}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
