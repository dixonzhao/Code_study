#!/usr/bin/env python3
"""Independent acceptance and installation for the U22A3 Artifact Store."""

from __future__ import annotations

import argparse
import hashlib
import io
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile
import tokenize
from typing import Dict, List, Sequence


PROGRAM_ID = "UDP002-U22A3-ARTIFACT-STORE-ACCEPTANCE"
VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"
CANDIDATE = "u002_u22_artifact_store_v0_1.py.txt"
INSTALLED = "u002_u22_artifact_store_v0_1.py"
VALIDATOR = "u002_u22_contract_validator_v0_1.py"
REGISTRY = "u002_u22_module_registry_v0_1.py"


class AcceptanceError(RuntimeError):
    """Represent one deterministic independent acceptance failure."""


def regular_file(path: Path) -> bool:
    """Accept only ordinary files during installation and dependency checks."""

    return path.is_file() and not path.is_symlink()


def canonical_text(value: object) -> str:
    """Serialize fixture manifests deterministically."""

    return json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    ) + "\n"


def run(command: List[str], timeout: int = 30) -> subprocess.CompletedProcess:
    """Run one bounded child process and preserve both output streams."""

    return subprocess.run(
        command,
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )


def check(value: bool, reason: str, checks: List[bool]) -> None:
    """Append one visible acceptance check or fail immediately."""

    if not value:
        raise AcceptanceError(reason)
    checks.append(True)


def sha256_bytes(value: bytes) -> str:
    """Create one fixture payload digest."""

    return hashlib.sha256(value).hexdigest()


def contains_semicolon_operator(source: str) -> bool:
    """Ignore prose punctuation and reject executable semicolon operators."""

    tokens = tokenize.generate_tokens(io.StringIO(source).readline)
    return any(
        token.type == tokenize.OP and token.string == ";"
        for token in tokens
    )


def artifact_manifest(
    artifact_id: str,
    run_id: str,
    stage: str,
    parent_ids: List[str],
    payload: bytes,
    *,
    declared_hash: str = "",
) -> Dict[str, object]:
    """Create one schema-valid synthetic artifact manifest."""

    digest = declared_hash or sha256_bytes(payload)
    return {
        "ContractVersion": CONTRACT_VERSION,
        "ArtifactId": artifact_id,
        "ArtifactType": "SYNTHETIC_TABLE",
        "SchemaId": "urn:u002:u22:synthetic-table",
        "SchemaVersion": "0.1.0",
        "RunId": run_id,
        "Stage": stage,
        "ProducerModuleId": "Synthetic.Producer",
        "ProducerModuleVersion": "0.1.0",
        "ParentArtifactIds": parent_ids,
        "Payloads": [
            {
                "RelativePath": "data.csv",
                "MediaType": "text/csv",
                "RowCount": 2,
                "Sha256": digest,
                "ByteSize": len(payload),
                "PayloadKind": "PRIMARY",
            }
        ],
        "Metadata": {
            "Fixture": True,
        },
    }


def write_package(
    run_root: Path,
    package_name: str,
    artifact_id: str,
    stage: str,
    parent_ids: List[str],
    *,
    bad_hash: bool = False,
) -> Path:
    """Write one complete fixture package into the Run staging area."""

    package = run_root / "00_STAGING" / package_name
    package.mkdir(parents=True)
    payload = b"FrameCount,Value\n1,10\n2,20\n"
    declared_hash = "f" * 64 if bad_hash else ""
    manifest = artifact_manifest(
        artifact_id,
        run_root.name,
        stage,
        parent_ids,
        payload,
        declared_hash=declared_hash,
    )
    (package / "data.csv").write_bytes(payload)
    (package / "artifact_manifest.json").write_text(
        canonical_text(manifest),
        encoding="utf-8",
    )
    return package


def copy_contract_environment(source: Path, target: Path) -> None:
    """Create an isolated U22 workspace without copying company data."""

    target.mkdir(parents=True)
    (target / "00_CONTROL/contracts").mkdir(parents=True)
    (target / "01_PROGRAMS/core").mkdir(parents=True)
    (target / "07_OUTPUT").mkdir(parents=True)
    contract_names = (
        "u002_module_manifest_schema_v0_1.json",
        "u002_artifact_manifest_schema_v0_1.json",
        "u002_run_configuration_schema_v0_1.json",
        "u002_algorithm_card_schema_v0_1.json",
    )
    for name in contract_names:
        source_file = source / "00_CONTROL/contracts" / name
        if not regular_file(source_file):
            raise AcceptanceError("CONTRACT_MISSING_" + name)
        shutil.copy2(
            source_file,
            target / "00_CONTROL/contracts" / name,
        )
    validator = source / "01_PROGRAMS/core" / VALIDATOR
    if not regular_file(validator):
        raise AcceptanceError("VALIDATOR_MISSING")
    shutil.copy2(
        validator,
        target / "01_PROGRAMS/core" / VALIDATOR,
    )


def archive_previous(root: Path) -> int:
    """Move completed U22A2 transfer files out of the active TEMP directory."""

    archive = root / "A-Archive/U22_COMPLETED/U22A2/04_TEMP"
    names = (
        "u002_u22_module_registry_v0_1.py.txt",
        "udp002_u22a2_module_registry_acceptance_v0_1.py.txt",
    )
    moved = 0
    for name in names:
        source = root / "04_TEMP" / name
        if not regular_file(source):
            continue
        archive.mkdir(parents=True, exist_ok=True)
        destination = archive / name
        if destination.exists():
            if destination.read_bytes() != source.read_bytes():
                raise AcceptanceError("ARCHIVE_CONFLICT")
            source.unlink()
        else:
            os.replace(source, destination)
        moved += 1
    return moved


def install_candidate(root: Path, candidate: Path, checks: List[bool]) -> Path:
    """Install the byte-identical accepted Core program atomically."""

    installed = root / "01_PROGRAMS/core" / INSTALLED
    if installed.exists():
        check(regular_file(installed), "INSTALL_NOT_REGULAR", checks)
        check(
            installed.read_bytes() == candidate.read_bytes(),
            "INSTALL_CONFLICT",
            checks,
        )
        return installed
    temporary = installed.with_name(installed.name + ".installing")
    check(not temporary.exists(), "INSTALL_TEMP_EXISTS", checks)
    temporary.write_bytes(candidate.read_bytes())
    os.replace(temporary, installed)
    check(
        installed.read_bytes() == candidate.read_bytes(),
        "INSTALL_IDENTITY",
        checks,
    )
    return installed


def main(argv: Sequence[str] = None) -> int:
    """Run independent functional acceptance and install only on success."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    checks: List[bool] = []
    try:
        root = Path(args.workspace_root).resolve()
        check(
            root.is_dir() and root.name == "U-UDP002",
            "WORKSPACE_ROOT",
            checks,
        )
        candidate = root / "04_TEMP" / CANDIDATE
        check(regular_file(candidate), "CANDIDATE_MISSING", checks)
        registry = root / "01_PROGRAMS/core" / REGISTRY
        check(regular_file(registry), "REGISTRY_MISSING", checks)
        registry_result = run(
            [
                sys.executable,
                str(registry),
                "--self-test",
                "--workspace-root",
                str(root),
            ]
        )
        check(
            registry_result.returncode == 0 and "N-T24" in registry_result.stdout,
            "REGISTRY_SELFTEST",
            checks,
        )
        source = candidate.read_text(encoding="utf-8")
        check("def commit_package(" in source, "COMMIT_INTERFACE", checks)
        check("def audit_run(" in source, "AUDIT_INTERFACE", checks)
        check(
            not contains_semicolon_operator(source),
            "SEMICOLON_STYLE",
            checks,
        )
        check("allow_nan=False" in source, "FINITE_JSON", checks)
        with tempfile.TemporaryDirectory(prefix="u22a3_acceptance_") as temporary:
            test_root = Path(temporary) / "U-UDP002"
            copy_contract_environment(root, test_root)
            test_candidate = test_root / "01_PROGRAMS/core" / INSTALLED
            shutil.copy2(candidate, test_candidate)
            selftest = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--self-test",
                    "--workspace-root",
                    str(test_root),
                ]
            )
            check(
                selftest.returncode == 0 and "N-T24" in selftest.stdout,
                "CANDIDATE_SELFTEST",
                checks,
            )
            run_root = test_root / "07_OUTPUT/RUN_TEST"
            (run_root / "00_STAGING").mkdir(parents=True)
            source_package = write_package(
                run_root,
                "source_package",
                "SOURCE_A",
                "SOURCE",
                [],
            )
            source_commit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--commit-package",
                    str(source_package),
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                source_commit.returncode == 0 and "N-C1-R0-A1-P1" in source_commit.stdout,
                "SOURCE_COMMIT",
                checks,
            )
            child_package = write_package(
                run_root,
                "child_package",
                "CHILD_A",
                "S1",
                ["SOURCE_A"],
            )
            child_commit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--commit-package",
                    str(child_package),
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                child_commit.returncode == 0 and "N-C1-R0-A1-P1" in child_commit.stdout,
                "CHILD_COMMIT",
                checks,
            )
            replay_package = write_package(
                run_root,
                "replay_package",
                "CHILD_A",
                "S1",
                ["SOURCE_A"],
            )
            replay = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--commit-package",
                    str(replay_package),
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                replay.returncode == 0 and "N-C1-R1-A1-P1" in replay.stdout,
                "IDEMPOTENT_REPLAY",
                checks,
            )
            audit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--audit-run",
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                audit.returncode == 0 and "N-C0-R0-A2-P2" in audit.stdout,
                "RUN_AUDIT",
                checks,
            )
            index_path = run_root / "run_artifact_index_v0_1.json"
            check(regular_file(index_path), "INDEX_MISSING", checks)
            index = json.loads(index_path.read_text(encoding="utf-8"))
            check(len(index["Artifacts"]) == 2, "INDEX_COUNT", checks)
            check(
                [item["ArtifactId"] for item in index["Artifacts"]]
                == ["CHILD_A", "SOURCE_A"],
                "INDEX_ORDER",
                checks,
            )
            missing_parent = write_package(
                run_root,
                "missing_parent",
                "ORPHAN_A",
                "S2",
                ["UNKNOWN_PARENT"],
            )
            missing_result = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--commit-package",
                    str(missing_parent),
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                missing_result.returncode == 7 and "PARENT_MISSING" in missing_result.stdout,
                "MISSING_PARENT_FAIL_CLOSE",
                checks,
            )
            bad_hash = write_package(
                run_root,
                "bad_hash",
                "BAD_HASH_A",
                "S2",
                ["CHILD_A"],
                bad_hash=True,
            )
            hash_result = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--commit-package",
                    str(bad_hash),
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                hash_result.returncode == 7 and "PAYLOAD_HASH" in hash_result.stdout,
                "HASH_FAIL_CLOSE",
                checks,
            )
            child_payload = run_root / "03_ARTIFACTS/S1/CHILD_A/data.csv"
            original_payload = child_payload.read_bytes()
            child_payload.write_bytes(original_payload + b"3,30\n")
            tamper_audit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--audit-run",
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                tamper_audit.returncode == 7 and "PAYLOAD_SIZE" in tamper_audit.stdout,
                "TAMPER_DETECTION",
                checks,
            )
            child_payload.write_bytes(original_payload)
            restored_audit = run(
                [
                    sys.executable,
                    str(test_candidate),
                    "--audit-run",
                    "--workspace-root",
                    str(test_root),
                    "--run-root",
                    str(run_root),
                ]
            )
            check(
                restored_audit.returncode == 0,
                "RESTORED_AUDIT",
                checks,
            )
        installed = install_candidate(root, candidate, checks)
        installed_selftest = run(
            [
                sys.executable,
                str(installed),
                "--self-test",
                "--workspace-root",
                str(root),
            ]
        )
        check(
            installed_selftest.returncode == 0 and "N-T24" in installed_selftest.stdout,
            "INSTALLED_SELFTEST",
            checks,
        )
        check(len(checks) == 23, "CHECK_COUNT_PRE", checks)
        if len(checks) != 24:
            raise AcceptanceError("CHECK_COUNT")
        archived = archive_previous(root)
        print(
            f"U22A3 N-C2-R1-A2-F3-T24-P1-V{archived}-X0 C-0"
        )
        return 0
    except Exception as exc:
        reason = reformat_reason(exc)
        print(
            f"U22A3 N-C0-R0-A0-F0-T0-P0-V0-X1 C-7 R-{reason}"
        )
        return 7


def reformat_reason(exc: Exception) -> str:
    """Keep the failure return short enough for manual transfer."""

    source = str(exc).upper() or exc.__class__.__name__.upper()
    normalized = "".join(
        character if character.isalnum() else "_"
        for character in source
    )
    return normalized.strip("_")[:80] or "INTERNAL"


if __name__ == "__main__":
    raise SystemExit(main())
