#!/usr/bin/env python3
"""Prepare, verify and install the U1G1 high-code Workbench GUI package."""

from __future__ import annotations

import argparse
import hashlib
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tempfile
import time
import zipfile


INSTALLER_VERSION = "0.3.0"
ZIP_NAME = "UDP001_U1G1_HIGH_CODE_WORKBENCH_GUI_v0_3.zip"
EXPECTED_FILES = {
    "01_PROGRAMS/gui/u001_workbench_model_v0_1.py": "92e7bcfa9ec59318626a093996695a9d26e404c815149291a77a281cd2cc5967",
    "01_PROGRAMS/gui/u001_workbench_gui_v0_3.py": "eca14063d25f0831bce241d08474e0f020132b5f11aa9cbd2b87a5ed24e80f5e",
}


class InstallError(RuntimeError):
    pass


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def locate() -> tuple[Path, Path, Path]:
    c_code = Path.cwd().resolve()
    workspace = c_code / "U-UDP001"
    temp_root = workspace / "04_TEMP"
    if c_code.name.upper() != "C-CODE":
        raise InstallError("RUN_FROM_C_CODE")
    if not workspace.is_dir():
        raise InstallError("U_UDP001_MISSING")
    if not temp_root.is_dir():
        raise InstallError("04_TEMP_MISSING")
    return c_code, workspace, temp_root


def bundle_path(temp_root: Path) -> Path:
    path = temp_root / ZIP_NAME
    if not path.is_file():
        raise InstallError("ZIP_MISSING")
    return path


def verify_zip(path: Path, extract_root: Path) -> None:
    with zipfile.ZipFile(path, "r") as archive:
        inventory = sorted(
            name for name in archive.namelist() if not name.endswith("/")
        )
        if inventory != sorted(EXPECTED_FILES):
            raise InstallError("ZIP_INVENTORY")
        archive.extractall(extract_root)
    for relative, expected_hash in EXPECTED_FILES.items():
        actual_hash = sha256_file(extract_root / relative)
        if actual_hash != expected_hash:
            raise InstallError("ZIP_SHA256_" + Path(relative).stem.upper())


def run_checked(arguments: list[str], cwd: Path) -> str:
    result = subprocess.run(
        arguments,
        cwd=str(cwd),
        capture_output=True,
        text=True,
        timeout=120,
    )
    output = (result.stdout or "") + (result.stderr or "")
    if result.returncode != 0:
        raise InstallError("CHILD_EXIT_" + str(result.returncode))
    return output.strip()


def archive_existing(workspace: Path, target: Path) -> int:
    if not target.exists():
        return 0
    archive_root = workspace / "A-Archive" / "U1G1_SUPERSEDED"
    archive_root.mkdir(parents=True, exist_ok=True)
    suffix = 1
    archived = archive_root / target.name
    while archived.exists():
        suffix += 1
        archived = archive_root / f"{target.stem}_{suffix}{target.suffix}"
    shutil.move(str(target), str(archived))
    return 1


def prepare() -> None:
    _, workspace, temp_root = locate()
    zip_path = bundle_path(temp_root)
    print("[U1G1 PREPARE] Workspace: " + str(workspace))
    print("[U1G1 PREPARE] Transfer ZIP: FOUND")
    print("[U1G1 PREPARE] Adapter root: " + ("FOUND" if (workspace / "01_PROGRAMS" / "adapters").is_dir() else "MISSING"))
    print("[U1G1 PREPARE] Catalog/Mapping GUI: " + ("FOUND" if (workspace / "01_PROGRAMS" / "core" / "u001_catalog_mapping_gui_v1_0.py").is_file() else "MISSING"))
    if not (workspace / "01_PROGRAMS" / "adapters").is_dir():
        raise InstallError("ADAPTER_ROOT_MISSING")
    if not (workspace / "01_PROGRAMS" / "core" / "u001_catalog_mapping_gui_v1_0.py").is_file():
        raise InstallError("CATALOG_GUI_MISSING")
    prepared = temp_root / "U1G1_PREPARED"
    if prepared.exists():
        shutil.rmtree(prepared)
    prepared.mkdir(parents=True)
    verify_zip(zip_path, prepared)
    print("U1G1-PREP-N2-A1-I2-X0-C0")


def self_test() -> None:
    _, workspace, temp_root = locate()
    with tempfile.TemporaryDirectory(prefix="u1g1_selftest_") as temp_name:
        extracted = Path(temp_name)
        verify_zip(bundle_path(temp_root), extracted)
        gui_root = extracted / "01_PROGRAMS" / "gui"
        model = gui_root / "u001_workbench_model_v0_1.py"
        gui = gui_root / "u001_workbench_gui_v0_3.py"
        model_result = run_checked([sys.executable, str(model), "--self-test"], gui_root)
        gui_result = run_checked([sys.executable, str(gui), "--self-test"], gui_root)
        if "MODEL-SELFTEST-PASS" not in model_result:
            raise InstallError("MODEL_SELFTEST_CAPSULE")
        if "GUI-SELFTEST-PASS" not in gui_result:
            raise InstallError("GUI_SELFTEST_CAPSULE")
    adapter_count = len(list((workspace / "01_PROGRAMS" / "adapters").glob("*/*/adapter_card.json")))
    if adapter_count < 1:
        raise InstallError("ADAPTER_CARD_EMPTY")
    print(f"U1G1-SELFTEST-N2-A{adapter_count}-T23-X0-C0")


def execute(no_launch: bool) -> None:
    _, workspace, temp_root = locate()
    with tempfile.TemporaryDirectory(prefix="u1g1_install_") as temp_name:
        extracted = Path(temp_name)
        verify_zip(bundle_path(temp_root), extracted)
        source_root = extracted / "01_PROGRAMS" / "gui"
        target_root = workspace / "01_PROGRAMS" / "gui"
        target_root.mkdir(parents=True, exist_ok=True)
        archived = 0
        for name in ("u001_workbench_model_v0_1.py", "u001_workbench_gui_v0_3.py"):
            target = target_root / name
            archived += archive_existing(workspace, target)
            shutil.copy2(source_root / name, target)
        model = target_root / "u001_workbench_model_v0_1.py"
        gui = target_root / "u001_workbench_gui_v0_3.py"
        run_checked([sys.executable, str(model), "--self-test"], target_root)
        run_checked([sys.executable, str(gui), "--self-test"], target_root)
        if not no_launch:
            launch_log = temp_root / "U1G1_GUI_LAUNCH_v0_3.log"
            with launch_log.open("w", encoding="utf-8") as stream:
                process = subprocess.Popen(
                    [sys.executable, str(gui), "--gui", "--workspace-root", str(workspace)],
                    cwd=str(target_root),
                    stdout=stream,
                    stderr=subprocess.STDOUT,
                    text=True,
                )
            time.sleep(2.0)
            exit_code = process.poll()
            if exit_code is not None:
                detail = launch_log.read_text(encoding="utf-8", errors="replace")[-300:]
                raise InstallError("GUI_EARLY_EXIT_" + str(exit_code) + "_" + detail)
    print(f"U1G1-N2-A{archived}-G{0 if no_launch else 1}-X0-C0")


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    action = parser.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    parser.add_argument("--no-launch", action="store_true", help=argparse.SUPPRESS)
    arguments = parser.parse_args()
    try:
        if arguments.prepare:
            prepare()
        elif arguments.self_test:
            self_test()
        else:
            execute(arguments.no_launch)
        return 0
    except Exception as error:
        reason = str(error).upper().replace(" ", "_")[:160]
        print("U1G1E-K" + reason + "-X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
