#!/usr/bin/env python3
"""Build and validate the same-machine U001 one-click launcher package."""

from __future__ import annotations

import argparse
import ast
import hashlib
import importlib.util
import json
import os
import shutil
import subprocess
import sys
import tempfile
from datetime import datetime
from pathlib import Path
from typing import Sequence


PROGRAM_NAME = "u001_package_builder_v0_2.py"
PROGRAM_VERSION = "0.2"
SELFTEST_TOKEN = "U001-PACKAGE-BUILDER-V02-SELFTEST-PASS-15"
REPORT_SCHEMA = "UDP001_PACKAGING_PREFLIGHT_V0_1"
LAUNCHER_PYTHON_TOKEN = "__U001_DEFAULT_PYTHON__"
LAUNCHER_ROOT_TOKEN = "__U001_DEFAULT_ROOT__"


class BuildError(RuntimeError):
    pass


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def atomic_write(path: Path, data: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(data)
    temporary.replace(path)


def atomic_json(path: Path, value: dict) -> None:
    atomic_write(
        path,
        (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode("utf-8"),
    )


def load_preflight(path: Path) -> dict:
    if not path.is_file():
        raise BuildError("PREFLIGHT_REPORT_NOT_FOUND")
    try:
        report = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, json.JSONDecodeError) as exc:
        raise BuildError("PREFLIGHT_REPORT_INVALID") from exc
    if report.get("Schema") != REPORT_SCHEMA:
        raise BuildError("PREFLIGHT_SCHEMA_MISMATCH")
    if int(report.get("Summary", {}).get("BlockingErrorCount", 1)) != 0:
        raise BuildError("PREFLIGHT_NOT_ACCEPTED")
    return report


def choose_python(report: dict) -> Path:
    environment = report.get("Environment", {})
    values = []
    if environment.get("PythonwExists"):
        values.append(environment.get("PythonwCandidate", ""))
    values.append(environment.get("PythonExecutable", ""))
    for value in values:
        if value and Path(value).is_file():
            return Path(value).resolve()
    raise BuildError("VALIDATED_PYTHON_NOT_FOUND")


def verify_dependencies(report: dict) -> list[str]:
    dependencies = report.get("Summary", {}).get("ExternalImports", [])
    if not isinstance(dependencies, list) or not all(isinstance(item, str) for item in dependencies):
        raise BuildError("EXTERNAL_IMPORT_LIST_INVALID")
    for name in dependencies:
        if importlib.util.find_spec(name) is None:
            raise BuildError(f"EXTERNAL_DEPENDENCY_NOT_IMPORTABLE:{name}")
    return sorted(set(dependencies))


def render_launcher(source: Path, python_exe: Path, udp001_root: Path, destination: Path) -> None:
    if not source.is_file():
        raise BuildError("LAUNCHER_SOURCE_NOT_FOUND")
    text = source.read_text(encoding="utf-8-sig")
    if text.count(LAUNCHER_PYTHON_TOKEN) != 1 or text.count(LAUNCHER_ROOT_TOKEN) != 1:
        raise BuildError("LAUNCHER_TEMPLATE_TOKEN_MISMATCH")
    text = text.replace(LAUNCHER_PYTHON_TOKEN, str(python_exe).replace("\\", "\\\\"))
    text = text.replace(LAUNCHER_ROOT_TOKEN, str(udp001_root).replace("\\", "\\\\"))
    atomic_write(destination, text.encode("utf-8"))


def assigned_string_constant(source_text: str, variable_name: str) -> str | None:
    tree = ast.parse(source_text)
    for node in tree.body:
        if not isinstance(node, ast.Assign) or not isinstance(node.value, ast.Constant):
            continue
        if not isinstance(node.value.value, str):
            continue
        for target in node.targets:
            if isinstance(target, ast.Name) and target.id == variable_name:
                return node.value.value
    return None


def pyinstaller_available() -> bool:
    return importlib.util.find_spec("PyInstaller") is not None


def run_builder(
    udp001_root: Path,
    preflight_report: Path,
    launcher_source: Path,
    out: Path,
) -> int:
    if os.name != "nt" or udp001_root.name != "UDP001" or not udp001_root.is_dir():
        raise BuildError("WINDOWS_UDP001_ROOT_REQUIRED")
    if out.exists() and (not out.is_dir() or any(out.iterdir())):
        raise BuildError("PACKAGE_OUTPUT_MUST_BE_ABSENT_OR_EMPTY")
    if not pyinstaller_available():
        raise BuildError("PYINSTALLER_NOT_INSTALLED")
    report = load_preflight(preflight_report)
    dependencies = verify_dependencies(report)
    python_exe = choose_python(report)
    build_root = udp001_root / "04_TEMP" / f"PK02_BUILD_WORK_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
    if build_root.exists():
        raise BuildError("PK02_BUILD_WORK_ALREADY_EXISTS")
    build_root.mkdir(parents=True)
    package_stage = build_root / "package_stage"
    package_stage.mkdir()
    evidence = package_stage / "_evidence"
    evidence.mkdir()
    generated_launcher = build_root / "u001_workbench_launcher_generated.py"
    render_launcher(launcher_source, python_exe, udp001_root.resolve(), generated_launcher)
    source_selftest = subprocess.run(
        [sys.executable, str(generated_launcher), "--self-test"],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=60,
        shell=False,
    )
    launcher_source_ok = source_selftest.returncode == 0 and "U001-LAUNCHER-SELFTEST-PASS-10" in source_selftest.stdout
    if not launcher_source_ok:
        raise BuildError("GENERATED_LAUNCHER_SELFTEST_FAILED")
    dist = build_root / "dist"
    work = build_root / "work"
    spec = build_root / "spec"
    dist.mkdir()
    work.mkdir()
    spec.mkdir()
    command = [
        sys.executable,
        "-m",
        "PyInstaller",
        "--noconfirm",
        "--clean",
        "--onefile",
        "--windowed",
        "--name",
        "U001_Workbench",
        "--distpath",
        str(dist),
        "--workpath",
        str(work),
        "--specpath",
        str(spec),
        str(generated_launcher),
    ]
    build = subprocess.run(
        command,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=600,
        shell=False,
    )
    atomic_write(evidence / "pyinstaller_stdout.txt", build.stdout.encode("utf-8"))
    atomic_write(evidence / "pyinstaller_stderr.txt", build.stderr.encode("utf-8"))
    built_exe = dist / "U001_Workbench.exe"
    build_ok = build.returncode == 0 and built_exe.is_file() and built_exe.stat().st_size > 0
    if not build_ok:
        raise BuildError("PYINSTALLER_BUILD_FAILED")
    installed_exe = package_stage / "U001_Workbench.exe"
    shutil.copy2(built_exe, installed_exe)
    probe_capsule = evidence / "u001_launcher_probe_capsule.txt"
    probe = subprocess.run(
        [
            str(installed_exe),
            "--probe",
            "--udp001-root",
            str(udp001_root.resolve()),
            "--probe-out",
            str(probe_capsule.resolve()),
        ],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        timeout=120,
        shell=False,
    )
    probe_text = probe_capsule.read_text(encoding="utf-8-sig") if probe_capsule.is_file() else ""
    probe_ok = probe.returncode == 0 and probe_text.rstrip().endswith("C-0")
    if not probe_ok:
        raise BuildError("PACKAGED_LAUNCHER_PROBE_FAILED")
    manifest = {
        "Schema": "UDP001_SAME_MACHINE_LAUNCHER_PACKAGE_V0_1",
        "CreatedLocal": datetime.now().isoformat(timespec="seconds"),
        "PackageMode": "SAME_MACHINE_ONE_CLICK_LAUNCHER",
        "PortableRuntimeIncluded": False,
        "UDP001Root": str(udp001_root.resolve()),
        "ValidatedPython": str(python_exe),
        "ExternalDependencies": dependencies,
        "Launcher": {
            "Filename": installed_exe.name,
            "Bytes": installed_exe.stat().st_size,
            "SHA256": sha256_file(installed_exe),
        },
        "PreflightReportSHA256": sha256_file(preflight_report),
        "LauncherSourceSHA256": sha256_file(launcher_source),
        "LauncherProbe": probe_text.strip().splitlines(),
        "Limitations": [
            "This package is bound to the validated company PC Python environment.",
            "The accepted Gate programs remain independent files under UDP001/01_PROGRAMS.",
            "Use PK03 only if deployment to another PC later requires an embedded runtime.",
        ],
    }
    atomic_json(evidence / "u001_package_manifest.json", manifest)
    capsule = (
        "U001PK02\n"
        f"N-R1-D{len(dependencies)}-I1-B1-E1-L1-M1-X0\n"
        "C-0\n"
    )
    atomic_write(evidence / "u001_package_build_capsule.txt", capsule.encode("utf-8"))
    out.parent.mkdir(parents=True, exist_ok=True)
    if out.exists():
        out.rmdir()
    package_stage.replace(out)
    shutil.rmtree(build_root, ignore_errors=True)
    print(capsule, end="")
    return 0


def run_self_test() -> None:
    passed = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td) / "UDP001"
        root.mkdir()
        report_path = root / "report.json"
        report = {
            "Schema": REPORT_SCHEMA,
            "Environment": {
                "PythonExecutable": sys.executable,
                "PythonwCandidate": sys.executable,
                "PythonwExists": True,
            },
            "Summary": {"BlockingErrorCount": 0, "ExternalImports": ["json"]},
        }
        report_path.write_text(json.dumps(report), encoding="utf-8")
        loaded = load_preflight(report_path)
        passed += int(loaded["Schema"] == REPORT_SCHEMA)
        passed += int(choose_python(loaded).is_file())
        passed += int(verify_dependencies(loaded) == ["json"])
        source = root / "launcher.py"
        source.write_text(
            f'P="{LAUNCHER_PYTHON_TOKEN}"\nR="{LAUNCHER_ROOT_TOKEN}"\n', encoding="utf-8"
        )
        rendered = root / "rendered.py"
        render_launcher(source, Path(sys.executable), root, rendered)
        text = rendered.read_text(encoding="utf-8")
        passed += int(LAUNCHER_PYTHON_TOKEN not in text)
        passed += int(LAUNCHER_ROOT_TOKEN not in text)
        passed += int(assigned_string_constant(text, "P") == str(Path(sys.executable)))
        passed += int(assigned_string_constant(text, "R") == str(root))
        windows_python_text = r"C:\Users\Test\venv\Scripts\pythonw.exe"
        windows_root_text = r"C:\C-CODE\UDP001"
        rendered_windows = root / "rendered_windows.py"
        render_launcher(source, Path(windows_python_text), Path(windows_root_text), rendered_windows)
        windows_text = rendered_windows.read_text(encoding="utf-8")
        passed += int(assigned_string_constant(windows_text, "P") == windows_python_text)
        passed += int(assigned_string_constant(windows_text, "R") == windows_root_text)
        passed += int(len(sha256_file(rendered)) == 64)
        payload = {"a": 1}
        atomic_json(root / "x.json", payload)
        passed += int(json.loads((root / "x.json").read_text(encoding="utf-8")) == payload)
        bad = dict(report)
        bad["Schema"] = "BAD"
        (root / "bad.json").write_text(json.dumps(bad), encoding="utf-8")
        try:
            load_preflight(root / "bad.json")
        except BuildError:
            passed += 1
        passed += int(importlib.util.find_spec("json") is not None)
        passed += int(PROGRAM_NAME.endswith("v0_2.py"))
        passed += int(PROGRAM_VERSION == "0.2")
    if passed != 15:
        raise AssertionError(f"SELFTEST_COUNT:{passed}")
    print(SELFTEST_TOKEN)


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="U001 same-machine one-click package builder")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--build", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--preflight-report")
    parser.add_argument("--launcher-source")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if args.build:
        if not all((args.udp001_root, args.preflight_report, args.launcher_source, args.out)):
            parser.error("--build requires --udp001-root --preflight-report --launcher-source --out")
        try:
            return run_builder(
                Path(args.udp001_root),
                Path(args.preflight_report),
                Path(args.launcher_source),
                Path(args.out),
            )
        except BuildError as exc:
            print(f"U001PK02-ERROR:{exc}")
            print("C-7")
            return 7
    parser.error("select --self-test or --build")
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
