#!/usr/bin/env python3
"""Install and accept the U001 H6A differential replay foundation."""
from __future__ import annotations
import argparse,datetime as dt,json,shutil,subprocess,sys,tempfile,zipfile
from pathlib import Path

ZIP="UDP001_U1H6A_DIFFERENTIAL_REPLAY_HIGH_CODE_v1_0.zip"
FILES={
"01_PROGRAMS/core/u001_h6_differential_replay_v1_0.py",
"02_CONTROL/contracts/u001_h6_differential_rules_schema_v1_0.json",
"02_CONTROL/templates/u001_h6_differential_rules_template_v1_0.json",
"05_TESTS/tools/u001_h6_differential_replay_acceptance_v1_0.py",
"06_DOCS/U001_H6_LEGACY_DIFFERENTIAL_REPLAY_TECHNICAL_SPEC_v1_0_2026-08-28.txt"}
class E(RuntimeError):pass
def root()->Path:
    p=Path.cwd()/"U-UDP001"
    if not p.is_dir():raise E("RUN_FROM_C_CODE_ROOT")
    return p
def say(c:str,w:str)->None:print(f"[CHECK] {c}\n[WHY]  {w}")
def base(r:Path)->Path:
    candidates=sorted((r/"05_TESTS"/"evidence").glob("U1H5_*/run_a"),reverse=True)
    for p in candidates:
        if all((p/n).is_file() for n in ["u001_decoded_updates.csv","u001_frame_values.csv","u001_signal_catalog.csv","u001_quality_report.csv","u001_run_manifest.json","u001_review_view.csv"]):return p
    raise E("H5_ACCEPTED_BASE_PACKAGE_NOT_FOUND")
def run(cmd:list[str],label:str)->str:
    x=subprocess.run(cmd,text=True,capture_output=True,timeout=180)
    if x.returncode:raise E(f"{label}_EXIT:{x.returncode}:{x.stdout[-250:]}:{x.stderr[-250:]}")
    return x.stdout.strip()
def prepare()->None:
    r=root();z=r/"04_TEMP"/ZIP;say(str(z),"Only the approved H6A archive may be installed.")
    if not z.is_file():raise E("ZIP_MISSING")
    with zipfile.ZipFile(z) as a:
        names={i.filename.replace("\\","/").rstrip("/") for i in a.infolist() if not i.is_dir()}
        if names!=FILES:raise E(f"ZIP_INVENTORY:{sorted(names)}")
        with tempfile.TemporaryDirectory(prefix="u1h6a_") as t:
            a.extractall(t);stamp=dt.datetime.now().strftime("%Y%m%dT%H%M%S");backup=r/"A-Archive"/f"U1H6A_REPLACED_{stamp}";moved=0
            for rel in sorted(FILES):
                dst=r/rel
                if dst.exists():old=backup/rel;old.parent.mkdir(parents=True,exist_ok=True);shutil.move(str(dst),str(old));moved+=1
                dst.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(Path(t)/rel,dst)
    b=base(r);say(str(b),"H6A acceptance reuses the already accepted H5 synthetic package and does not require company source data.");print(f"U1H6AP-N{len(FILES)}-A{moved}-B1-X0-C0")
def selftest()->None:
    r=root();prog=r/"01_PROGRAMS/core/u001_h6_differential_replay_v1_0.py";say(prog.name,"The streaming comparator must pass its internal invariants before category acceptance.")
    if "PASS-3" not in run([sys.executable,str(prog),"--self-test"],"CORE_SELFTEST"):raise E("CORE_CAPSULE")
    print("U1H6AST-P1-X0-C0")
def accept()->None:
    r=root();tool=r/"05_TESTS/tools/u001_h6_differential_replay_acceptance_v1_0.py";rules=r/"02_CONTROL/templates/u001_h6_differential_rules_template_v1_0.json";b=base(r);out=r/"05_TESTS/evidence"/("U1H6A_"+dt.datetime.now().strftime("%Y%m%dT%H%M%S"));say(str(out),"Evidence covers all six difference classes and read-only behavior.")
    result=run([sys.executable,str(tool),"--core-dir",str(r/"01_PROGRAMS/core"),"--rules",str(rules),"--base-package",str(b),"--out",str(out)],"ACCEPTANCE")
    if "-C0" not in result:raise E("ACCEPTANCE_CAPSULE")
    print(result)
def main()->int:
    p=argparse.ArgumentParser();g=p.add_mutually_exclusive_group(required=True);g.add_argument("--prepare",action="store_true");g.add_argument("--self-test",action="store_true");g.add_argument("--accept",action="store_true");a=p.parse_args()
    try:
        prepare() if a.prepare else selftest() if a.self_test else accept();return 0
    except Exception as x:
        t=Path.cwd()/"U-UDP001"/"04_TEMP";t.mkdir(parents=True,exist_ok=True);(t/"U1H6A_DIAGNOSTIC.json").write_text(json.dumps({"ErrorType":type(x).__name__,"Error":str(x)},indent=2)+"\n");print(f"U1H6A-ERROR-{type(x).__name__}-{str(x)[:180]}-X1-C7");return 7
if __name__=="__main__":raise SystemExit(main())
