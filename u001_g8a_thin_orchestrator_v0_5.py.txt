class ControlError(RuntimeError):
    """Blocking orchestration error."""


@dataclass(frozen=True)
class StageSpec:
    stage_id: str
    module_key: str
    args: tuple[str, ...]
    expected_files: tuple[Path, ...]
    capsule: Path | None


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def canonical_json(value: Any) -> str:
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":"))


def config_fingerprint(data: dict[str, Any]) -> str:
    copy = dict(data)
    copy.pop("ConfigFingerprint", None)
    return hashlib.sha256(canonical_json(copy).encode("utf-8")).hexdigest()


def sha256_file(path: Path, block_size: int = 8 * 1024 * 1024) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while True:
            block = stream.read(block_size)
            if not block:
                break
            digest.update(block)
    return digest.hexdigest()


def file_record(path: Path, include_hash: bool = True) -> dict[str, Any]:
    stat = path.stat()
    record: dict[str, Any] = {
        "Path": str(path.resolve()),
        "Basename": path.name,
        "Size": stat.st_size,
        "ModifiedTimeNs": stat.st_mtime_ns,
    }
    if include_hash:
        record["SHA256"] = sha256_file(path)
    return record


def atomic_write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8", newline="\n") as stream:
        stream.write(text)
    os.replace(temporary, path)


def atomic_write_json(path: Path, value: Any) -> None:
    atomic_write_text(path, json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def require_file(path: Path, label: str) -> Path:
    if not path.is_file():
        raise ControlError(f"{label} missing: {path}")
    return path.resolve()


def require_empty_or_absent(path: Path, label: str) -> None:
    if path.exists() and (not path.is_dir() or any(path.iterdir())):
        raise ControlError(f"{label} must be absent or an empty folder: {path}")


def discover_single_csv(source_root: Path) -> Path:
    files = sorted(p for p in source_root.iterdir() if p.is_file() and p.suffix.lower() == ".csv")
    if len(files) != 1:
        names = ",".join(p.name for p in files) or "NONE"
        raise ControlError(f"expected exactly one source CSV in {source_root}; found {len(files)}: {names}")
    return files[0].resolve()


def discover_active_files(root: Path) -> tuple[dict[str, Path], dict[str, Path]]:
    program_root = root / "01_PROGRAMS"
    control_root = root / "02_CONTROL"
    modules: dict[str, Path] = {}
    for key, basename, _ in MODULES:
        modules[key] = require_file(program_root / basename, f"{key} active module")
    controls = {name: require_file(control_root / name, "active control") for name in CONTROLS}
    return modules, controls


def read_csv_dicts(path: Path) -> tuple[list[str], list[dict[str, str]]]:
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        if reader.fieldnames is None:
            raise ControlError(f"CSV has no header: {path}")
        rows = [{str(k): ("" if v is None else str(v)) for k, v in row.items()} for row in reader]
        return [str(x) for x in reader.fieldnames], rows


def derive_frame_canonical_name(g4_policy: Path) -> str:
    headers, rows = read_csv_dicts(g4_policy)
    for required in ("CanonicalName", "SignalRole"):
        if required not in headers:
            raise ControlError(f"G4 policy missing {required}")
    matches = [r["CanonicalName"].strip() for r in rows if r["SignalRole"].strip().upper() == "FRAME_KEY"]
    matches = [x for x in matches if x]
    if len(matches) != 1:
        raise ControlError(f"G4 policy must contain exactly one FRAME_KEY; found {len(matches)}")
    return matches[0]


def validate_controls(controls: dict[str, Path]) -> dict[str, Any]:
    g4 = controls["u001_g4_signal_policy.csv"]
    g5 = controls["u001_g5_frame_policy.csv"]
    g6 = controls["u001_g6_hold_policy.csv"]
    frame_name = derive_frame_canonical_name(g4)

    mappings: dict[str, list[tuple[str, str]]] = {}
    for label, path in (("G4", g4), ("G5", g5), ("G6", g6)):
        headers, rows = read_csv_dicts(path)
        if "CanonicalName" not in headers:
            raise ControlError(f"{label} policy missing CanonicalName")
        pairs: list[tuple[str, str]] = []
        for position, row in enumerate(rows, 1):
            name = row.get("CanonicalName", "").strip()
            if not name:
                raise ControlError(f"{label} policy has blank CanonicalName at row {position}")
            index = row.get("SignalIndex", "").strip()
            pairs.append((index, name))
        if len({name for _, name in pairs}) != len(pairs):
            raise ControlError(f"{label} policy contains duplicate CanonicalName")
        mappings[label] = pairs

    g4_names = {name for _, name in mappings["G4"]}
    g5_names = {name for _, name in mappings["G5"]}
    for index, name in mappings["G6"]:
        if name not in g4_names or name not in g5_names:
            raise ControlError(f"G6 mapping not present in G4/G5: {index}:{name}")

    headers6, rows6 = read_csv_dicts(g6)
    if "HoldPolicy" not in headers6:
        raise ControlError("G6 policy missing HoldPolicy")
    bad_holds = [r.get("CanonicalName", "") for r in rows6 if r.get("HoldPolicy", "").strip().upper() != "NONE"]
    if bad_holds:
        raise ControlError("rebaseline requires every G6 HoldPolicy=NONE")
    return {
        "FrameCanonicalName": frame_name,
        "G4Rows": len(mappings["G4"]),
        "G5Rows": len(mappings["G5"]),
        "G6Rows": len(mappings["G6"]),
        "G6AllHoldNone": True,
    }


def parse_capsule_code(path: Path) -> int:
    text = path.read_text(encoding="utf-8-sig", errors="replace")
    matches = re.findall(r"(?m)^C-(\d+)\s*$", text)
    if len(matches) != 1:
        raise ControlError(f"capsule must contain exactly one C-n line: {path}")
    return int(matches[0])


def count_csv_rows(path: Path) -> int:
    with path.open("rb") as stream:
        count = sum(1 for _ in stream)
    return max(count - 1, 0)


def protected_metadata(paths: Iterable[Path]) -> dict[str, dict[str, int]]:
    result: dict[str, dict[str, int]] = {}
    for path in paths:
        stat = path.stat()
        result[str(path.resolve())] = {"Size": stat.st_size, "ModifiedTimeNs": stat.st_mtime_ns}
    return result


def verify_config(data: dict[str, Any]) -> None:
    if data.get("Schema") != CONFIG_SCHEMA:
        raise ControlError("unsupported G8A thin config schema")
    if data.get("ConfigFingerprint") != config_fingerprint(data):
        raise ControlError("config fingerprint mismatch")
    safety = data.get("Safety", {})
    required = {
        "StopOnFirstFailure": True,
        "RejectOverwrite": True,
        "ShellEqualsFalse": True,
        "NoDataTransformationInOrchestrator": True,
        "NoValidationInOrchestrator": True,
        "NoGate7ConfigMutation": True,
    }
    for key, value in required.items():
        if safety.get(key) is not value:
            raise ControlError(f"unsafe config: {key}")


def load_config(path: Path) -> dict[str, Any]:
    data = json.loads(path.read_text(encoding="utf-8-sig"))
    if not isinstance(data, dict):
        raise ControlError("config must be a JSON object")
    verify_config(data)
    return data


def build_config(args: argparse.Namespace) -> dict[str, Any]:
    udp001_root = Path(args.udp001_root).resolve()
    case_root = Path(args.case_root).resolve()
    source_root = case_root / SOURCE_FOLDER_NAME
    if not source_root.is_dir():
        raise ControlError(f"source folder missing: {source_root}")
    modules, controls = discover_active_files(udp001_root)
    source = require_file(source_root / SOURCE_CSV_NAME, "fixed Raw CAN CSV alias")
    dbc1 = require_file(source_root / DBC_CH1_NAME, "fixed Channel 1 DBC alias")
    dbc2 = require_file(source_root / DBC_CH2_NAME, "fixed Channel 2 DBC alias")
    require_file(source, "Raw CAN CSV")
    if dbc1 == dbc2:
        raise ControlError("Channel 1 and Channel 2 DBC paths must differ")
    if source.parent != source_root or dbc1.parent != source_root or dbc2.parent != source_root:
        raise ControlError(f"all three fixed aliases must be directly inside {SOURCE_FOLDER_NAME}")
    control_check = validate_controls(controls)

    module_records = {key: file_record(path) for key, path in modules.items()}
    control_records = {name: file_record(path) for name, path in controls.items()}
    input_records = {
        "Source": file_record(source),
        "DBC_CH1": file_record(dbc1),
        "DBC_CH2": file_record(dbc2),
    }
    data: dict[str, Any] = {
        "Schema": CONFIG_SCHEMA,
        "Program": {"Name": PROGRAM_NAME, "Version": PROGRAM_VERSION},
        "ContractVersion": CONTRACT_VERSION,
        "ImplementationProfile": PROFILE,
        "CreatedUTC": utc_now(),
        "UDP001Root": str(udp001_root),
        "CaseRoot": str(case_root),
        "Identity": {
            "DatasetId": args.dataset_id,
            "RunId": args.run_id,
            "SegmentId": args.segment_id,
            "SourceId": args.source_id,
        },
        "Modules": module_records,
        "ControlFiles": control_records,
        "Inputs": input_records,
        "ControlCheck": control_check,
        "Parameters": {
            "MaxSourceRows": int(args.max_source_rows),
            "DecodeChunkSize": int(args.decode_chunksize),
            "AssignmentChunkSize": int(args.assignment_chunksize),
            "ClassificationChunkSize": int(args.classification_chunksize),
            "FrameBuildChunkSize": int(args.frame_chunksize),
            "HoldApplyChunkSize": int(args.hold_chunksize),
            "MaxAssignmentAgeRows": int(args.max_assignment_age_rows),
        },
        "Safety": {
            "StopOnFirstFailure": True,
            "RejectOverwrite": True,
            "ShellEqualsFalse": True,
            "NoDataTransformationInOrchestrator": True,
            "NoValidationInOrchestrator": True,
            "NoGate7ConfigMutation": True,
        },
    }
    data["ConfigFingerprint"] = config_fingerprint(data)
    return data


def current_paths(data: dict[str, Any]) -> tuple[dict[str, Path], dict[str, Path], dict[str, Path]]:
    modules = {k: Path(v["Path"]).resolve() for k, v in data["Modules"].items()}
    controls = {k: Path(v["Path"]).resolve() for k, v in data["ControlFiles"].items()}
    inputs = {k: Path(v["Path"]).resolve() for k, v in data["Inputs"].items()}
    return modules, controls, inputs


def verify_frozen_records(records: dict[str, Any], paths: dict[str, Path], label: str) -> None:
    for key, path in paths.items():
        require_file(path, f"{label} {key}")
        current = file_record(path)
        expected = records[key]
        for field in ("Basename", "Size", "SHA256"):
            if current[field] != expected[field]:
                raise ControlError(f"{label} changed since config creation: {key}:{field}")


def run_module_self_tests(modules: dict[str, Path]) -> list[dict[str, Any]]:
    results: list[dict[str, Any]] = []
    for key, _, expected in MODULES:
        started = time.monotonic()
        completed = subprocess.run(
            [sys.executable, str(modules[key]), "--self-test"],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            shell=False,
        )
        passed = completed.returncode == 0 and expected in completed.stdout
        result = {
            "Gate": key,
            "Module": modules[key].name,
            "Expected": expected,
            "ReturnCode": completed.returncode,
            "Passed": passed,
            "DurationSeconds": round(time.monotonic() - started, 3),
            "StdoutTail": completed.stdout[-4000:],
            "StderrTail": completed.stderr[-4000:],
        }
        results.append(result)
        if not passed:
            raise ControlError(f"{key} module self-test failed")
    return results


def perform_preflight(data: dict[str, Any]) -> dict[str, Any]:
    modules, controls, inputs = current_paths(data)
    verify_frozen_records(data["Modules"], modules, "module")
    verify_frozen_records(data["ControlFiles"], controls, "control")
    verify_frozen_records(data["Inputs"], inputs, "input")
    control_check = validate_controls(controls)
    self_tests = run_module_self_tests(modules)
    return {
        "Status": "PASS",
        "CheckedUTC": utc_now(),
        "ConfigFingerprint": data["ConfigFingerprint"],
        "ControlCheck": control_check,
        "ModuleSelfTests": self_tests,
        "ModuleCount": len(modules),
        "ControlCount": len(controls),
        "InputCount": len(inputs),
    }


def write_preflight_outputs(out: Path, report: dict[str, Any] | None, error: str | None) -> int:
    require_empty_or_absent(out, "preflight output")
    out.mkdir(parents=True, exist_ok=True)
    passed = error is None and report is not None
    payload = report if passed else {"Status": "FAIL", "CheckedUTC": utc_now(), "Error": error or "UNKNOWN"}
    atomic_write_json(out / "u001_g8a2_preflight_manifest.json", payload)
    capsule = (
        "UDP001G8A2P1\n"
        f"N-M{payload.get('ModuleCount', 0)}-C{payload.get('ControlCount', 0)}-I{payload.get('InputCount', 0)}-X{0 if passed else 1}\n"
        f"C-{0 if passed else 7}\n"
    )
    atomic_write_text(out / "udp001_g8a2_preflight_capsule.txt", capsule)
    return 0 if passed else 7

def run_stage(spec: StageSpec, modules: dict[str, Path], protected: Sequence[Path], control_output: Path) -> dict[str, Any]:
    metadata_before = protected_metadata(protected)
    command = [sys.executable, str(modules[spec.module_key]), *spec.args]
    started_utc = utc_now()
    started = time.monotonic()
    completed = subprocess.run(
        command,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        shell=False,
    )
    duration = round(time.monotonic() - started, 3)
    stdout_path = control_output / f"{spec.stage_id}_stdout.txt"
    stderr_path = control_output / f"{spec.stage_id}_stderr.txt"
    atomic_write_text(stdout_path, completed.stdout)
    atomic_write_text(stderr_path, completed.stderr)
    metadata_equal = metadata_before == protected_metadata(protected)
    missing = [str(path) for path in spec.expected_files if not path.is_file()]
    capsule_code: int | None = None
    capsule_error: str | None = None
    if spec.capsule is not None and spec.capsule.is_file():
        try:
            capsule_code = parse_capsule_code(spec.capsule)
        except Exception as exc:  # recorded as a controlled stage failure
            capsule_error = str(exc)
    success = completed.returncode == 0 and not missing and metadata_equal
    if spec.capsule is not None:
        success = success and capsule_code == 0 and capsule_error is None
    result = {
        "StageId": spec.stage_id,
        "Module": modules[spec.module_key].name,
        "StartedUTC": started_utc,
        "DurationSeconds": duration,
        "ReturnCode": completed.returncode,
        "CapsuleCode": capsule_code,
        "CapsuleError": capsule_error,
        "ExpectedOutputsMissing": [Path(x).name for x in missing],
        "ProtectedMetadataEqual": metadata_equal,
        "StdoutLog": stdout_path.name,
        "StderrLog": stderr_path.name,
        "Success": success,
    }
    if not success:
        raise ControlError(f"stage {spec.stage_id} failed: {canonical_json(result)}")
    return result


def find_final_package(g7_root: Path) -> Path:
    matches: list[Path] = []
    if g7_root.is_dir():
        for folder in [g7_root, *sorted(p for p in g7_root.rglob("*") if p.is_dir())]:
            if all((folder / name).is_file() for name in FINAL_PACKAGE_FILES):
                matches.append(folder)
    unique = list(dict.fromkeys(p.resolve() for p in matches))
    if len(unique) != 1:
        raise ControlError(f"expected exactly one final six-file package; found {len(unique)}")
    if {p.name for p in unique[0].iterdir() if p.is_file()} != set(FINAL_PACKAGE_FILES):
        raise ControlError("final package must contain exactly the six contract files")
    return unique[0]


def make_run_capsule(stage_results: list[dict[str, Any]], updates: int, frames: int, files: int,
                     input_hash_equal: bool, error_count: int) -> str:
    stage_text = " ".join(
        f"S{i}-C{r.get('CapsuleCode', 0) if r.get('CapsuleCode') is not None else 0}"
        for i, r in enumerate(stage_results, 1)
    )
    final_code = 0 if error_count == 0 and input_hash_equal and len(stage_results) == 9 else 7
    return (
        "UDP001G8A2R1\n"
        f"N-S{len(stage_results)}-P7-U{updates}-R{frames}-F{files}-I{1 if input_hash_equal else 0}-X{error_count}\n"
        f"{stage_text}\n"
        f"C-{final_code}\n"
    )


def execute_run(data: dict[str, Any], out_root: Path) -> int:
    require_empty_or_absent(out_root, "run output")
    out_root.mkdir(parents=True, exist_ok=True)
    control_output = out_root / "CONTROL_OUTPUT"
    control_output.mkdir()
    manifest: dict[str, Any] = {
        "Schema": "UDP001_G8A_THIN_RUN_MANIFEST_V0_1",
        "Program": {"Name": PROGRAM_NAME, "Version": PROGRAM_VERSION},
        "StartedUTC": utc_now(),
        "ConfigFingerprint": data["ConfigFingerprint"],
        "Identity": data["Identity"],
        "Status": "RUNNING",
        "Stages": [],
        "Warnings": [],
        "Errors": [],
    }
    manifest_path = control_output / "u001_g8a2_orchestration_manifest.json"
    capsule_path = control_output / "udp001_g8a2_return_capsule.txt"
    modules, controls, inputs = current_paths(data)
    protected = [*modules.values(), *controls.values(), *inputs.values()]
    pre_hashes: dict[str, str] = {}
    stage_results: list[dict[str, Any]] = []
    updates = frames = files = 0
    input_hash_equal = False
    try:
        preflight = perform_preflight(data)
        manifest["Preflight"] = preflight
        pre_hashes = {str(p.resolve()): sha256_file(p) for p in protected}
        control_dir = out_root / "CONTROL"
        control_dir.mkdir()
        copied_controls: dict[str, Path] = {}
        for name, source in controls.items():
            destination = control_dir / name
            shutil.copy2(source, destination)
            if sha256_file(destination) != pre_hashes[str(source.resolve())]:
                raise ControlError(f"copied control hash mismatch: {name}")
            copied_controls[name] = destination
        for folder in (
            "G1_PROBE", "G2_DECODE", "G3_ASSIGN", "G4_CLASSIFY", "G5_FRAME",
            "G6_PROFILE", "G6_APPLY", "G7_CONFIG", "G7_BUILD",
        ):
            (out_root / folder).mkdir()
        for spec in stage_specs(data, out_root, copied_controls):
            result = run_stage(spec, modules, protected, control_output)
            stage_results.append(result)
            manifest["Stages"] = stage_results
            atomic_write_json(manifest_path, manifest)
        package = find_final_package(out_root / "G7_BUILD")
        updates = count_csv_rows(package / "u001_decoded_updates.csv")
        frames = count_csv_rows(package / "u001_frame_values.csv")
        files = len(FINAL_PACKAGE_FILES)
        post_hashes = {str(p.resolve()): sha256_file(p) for p in protected}
        input_hash_equal = pre_hashes == post_hashes
        if not input_hash_equal:
            raise ControlError("protected module/control/input SHA-256 changed during run")
        manifest["FinalPackage"] = {
            "Folder": str(package),
            "Files": {name: file_record(package / name) for name in FINAL_PACKAGE_FILES},
            "DecodedUpdateRows": updates,
            "FrameRows": frames,
        }
        manifest["ProtectedHashEqual"] = True
        manifest["Status"] = "SUCCESS"
        manifest["FinishedUTC"] = utc_now()
        atomic_write_json(manifest_path, manifest)
        atomic_write_text(capsule_path, make_run_capsule(stage_results, updates, frames, files, True, 0))
        return 0
    except Exception as exc:
        manifest["Status"] = "FAILURE"
        manifest["FinishedUTC"] = utc_now()
        manifest["Errors"].append(str(exc))
        manifest["Stages"] = stage_results
        atomic_write_json(manifest_path, manifest)
        atomic_write_text(capsule_path, make_run_capsule(stage_results, updates, frames, files, input_hash_equal, 1))
        print(f"ERROR: {exc}", file=sys.stderr)
        return 7

def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description="UDP001 Gate 8A thin orchestrator")
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--make-config", action="store_true")
    mode.add_argument("--preflight", action="store_true")
    mode.add_argument("--run", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--case-root")
    parser.add_argument("--config")
    parser.add_argument("--out")
    parser.add_argument("--out-root")
    parser.add_argument("--dataset-id", default="TC001_20260422")
    parser.add_argument("--run-id", default="TC001_20260422_RUN_A")
    parser.add_argument("--segment-id", default="SEGMENT_001")
    parser.add_argument("--source-id", default="SOURCE_001")
    parser.add_argument("--max-source-rows", type=int, default=100000)
    parser.add_argument("--decode-chunksize", type=int, default=50000)
    parser.add_argument("--assignment-chunksize", type=int, default=100000)
    parser.add_argument("--classification-chunksize", type=int, default=100000)
    parser.add_argument("--frame-chunksize", type=int, default=100000)
    parser.add_argument("--hold-chunksize", type=int, default=100000)
    parser.add_argument("--max-assignment-age-rows", type=int, default=200)
    return parser


def main(argv: Sequence[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    try:
        if args.self_test:
            return self_test()
        if args.make_config:
            required = (args.udp001_root, args.case_root, args.out)
            if not all(required):
                raise ControlError("--make-config requires --udp001-root --case-root --out")
            out = Path(args.out)
            if out.exists():
                raise ControlError(f"config output already exists: {out}")
            config = build_config(args)
            atomic_write_json(out, config)
            print(f"UDP001-G8A2-CONFIG-CREATED {out.name}")
            return 0
        if args.preflight:
            if not args.config or not args.out:
                raise ControlError("--preflight requires --config and --out")
            data = load_config(Path(args.config))
            try:
                report = perform_preflight(data)
                return write_preflight_outputs(Path(args.out), report, None)
            except Exception as exc:
                return write_preflight_outputs(Path(args.out), None, str(exc))
        if args.run:
            if not args.config or not args.out_root:
                raise ControlError("--run requires --config and --out-root")
            data = load_config(Path(args.config))
            return execute_run(data, Path(args.out_root))
        raise ControlError("no mode selected")
    except Exception as exc:
        print(f"ERROR: {exc}", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
