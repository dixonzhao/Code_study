@echo off
setlocal EnableExtensions EnableDelayedExpansion

rem UDP002 workspace preparation v1.1
rem Run from the C-CODE root in Windows CMD.
rem UDP002 is the development task root; U002 is the future product name.
rem This script creates folders only, copies three U001 manuals, and archives
rem one legacy P004 source when it can be found. It does not copy company data
rem and does not create U002 algorithm code.

if not exist ".\UDP001" (
  echo UDP002WS1-ERROR-NOT-C-CODE-ROOT C-7
  exit /b 7
)

if exist ".\UDP002" (
  echo UDP002WS1-ERROR-UDP002-ALREADY-EXISTS C-7
  exit /b 7
)

set "DL=%USERPROFILE%\Downloads"
set "DOC1=U001_USER_OPERATION_MANUAL_v1_0_2026-07-31.txt"
set "DOC2=U001_U002_INTERFACE_MANUAL_v1_0_2026-07-31.txt"
set "DOC3=U001_MAINTENANCE_AND_TROUBLESHOOTING_MANUAL_v1_0_2026-07-31.txt"

if not exist "%DL%\%DOC1%" (
  echo UDP002WS1-ERROR-MISSING-%DOC1% C-7
  exit /b 7
)
if not exist "%DL%\%DOC2%" (
  echo UDP002WS1-ERROR-MISSING-%DOC2% C-7
  exit /b 7
)
if not exist "%DL%\%DOC3%" (
  echo UDP002WS1-ERROR-MISSING-%DOC3% C-7
  exit /b 7
)

set "P004_NAME=sre_cam_stage_residual_error_analyzer_v4_5_3_jp_compact_excel_notes.py"
set "P004_SRC="
for /f "delims=" %%F in ('where /r . "%P004_NAME%" 2^>nul') do (
  if not defined P004_SRC set "P004_SRC=%%F"
)

md ".\UDP002\01_PROGRAMS" || goto :create_fail
md ".\UDP002\02_CONTROLS" || goto :create_fail
md ".\UDP002\03_CASES\TC001_20260422\00_U001_INPUT-READONLY" || goto :create_fail
md ".\UDP002\03_CASES\TC001_20260422\01_WORK" || goto :create_fail
md ".\UDP002\03_CASES\TC001_20260422\02_OUTPUT" || goto :create_fail
md ".\UDP002\03_CASES\TC001_20260422\03_U003_HANDOFF" || goto :create_fail
md ".\UDP002\03_CASES\TC001_20260422\04_LOGS" || goto :create_fail
md ".\UDP002\03_CASES\TC001_20260422\05_MANIFESTS" || goto :create_fail
md ".\UDP002\04_TEMP" || goto :create_fail
md ".\UDP002\05_TESTS\01_FIXTURES" || goto :create_fail
md ".\UDP002\05_TESTS\02_EXPECTED" || goto :create_fail
md ".\UDP002\05_TESTS\03_RUNS" || goto :create_fail
md ".\UDP002\06_DOCS\01_U001_REFERENCE" || goto :create_fail
md ".\UDP002\06_DOCS\02_DESIGN" || goto :create_fail
md ".\UDP002\06_DOCS\03_REPORTS" || goto :create_fail
md ".\UDP002\ARCH\01_LEGACY_REFERENCE" || goto :create_fail
md ".\UDP002\ARCH\02_RETIRED_BUILDS" || goto :create_fail
md ".\UDP002\ARCH\03_FAILED_EXPERIMENTS" || goto :create_fail

copy /y "%DL%\%DOC1%" ".\UDP002\06_DOCS\01_U001_REFERENCE\%DOC1%" >nul || goto :copy_fail
copy /y "%DL%\%DOC2%" ".\UDP002\06_DOCS\01_U001_REFERENCE\%DOC2%" >nul || goto :copy_fail
copy /y "%DL%\%DOC3%" ".\UDP002\06_DOCS\01_U001_REFERENCE\%DOC3%" >nul || goto :copy_fail
attrib +R ".\UDP002\06_DOCS\01_U001_REFERENCE\%DOC1%"
attrib +R ".\UDP002\06_DOCS\01_U001_REFERENCE\%DOC2%"
attrib +R ".\UDP002\06_DOCS\01_U001_REFERENCE\%DOC3%"

set "P004_COPIED=0"
if defined P004_SRC (
  copy /y "!P004_SRC!" ".\UDP002\ARCH\01_LEGACY_REFERENCE\%P004_NAME%" >nul || goto :copy_fail
  attrib +R ".\UDP002\ARCH\01_LEGACY_REFERENCE\%P004_NAME%"
  set "P004_COPIED=1"
)

echo UDP002WS1 N-D3-P!P004_COPIED!-K1-X0 C-0
exit /b 0

:create_fail
echo UDP002WS1-ERROR-CREATE-DIRECTORY C-7
exit /b 7

:copy_fail
echo UDP002WS1-ERROR-COPY-REFERENCE C-7
exit /b 7
