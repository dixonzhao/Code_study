#!/usr/bin/env python3
"""Read-only audit of the FrameCount path from raw CSV through H2.

The diagnostic intentionally does not repair, sort, fill, or rewrite data.
It answers whether H2 lost 0x620 messages, whether source rows form a usable
cross-message time domain, and where selected message groups occur.
"""

from __future__ import annotations

import argparse
import csv
import json
import math
import re
import shutil
import sys
from collections import defaultdict
from dataclasses import dataclass
from datetime import datetime, timezone
from decimal import Decimal, InvalidOperation
from pathlib import Path
from typing import Any, Dict, Iterable, List, Optional, Sequence, Tuple


PROGRAM_ID = "U001_H6B5D3_FRAME_KEY_PATH_DIAGNOSTIC"
PROGRAM_VERSION = "1.0"
PREPARED_FOLDER = "U1H6B5D3_PREPARED"
OUTPUT_FOLDER = "U1H6B5D3_OUTPUT"

SUPERSEDED_TRANSFER_NAMES = (
    "u001_h6b5d1_frame_validity_diagnostic_v1_0.py.txt",
    "U1H6B5D1_START_HERE_CN_v1_0_2026-08-28.txt",
)


class PathDiagnosticError(RuntimeError):
    """Fail-closed diagnostic error."""


@dataclass
class GroupStats:
    update_rows: int = 0
    first_source_row: Optional[int] = None
    last_source_row: Optional[int] = None
    after_last_frame_rows: int = 0


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def write_json(path: Path, value: Dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        raise PathDiagnosticError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise PathDiagnosticError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise PathDiagnosticError(label + "_ROOT_NOT_OBJECT")
    return value


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise PathDiagnosticError("U_UDP001_ROOT_NOT_FOUND")
    return root


def required_paths(root: Path) -> Dict[str, Path]:
    b3 = root / "04_TEMP" / "U1H6B3_OUTPUT" / "compiled_profile"
    h2 = root / "04_TEMP" / "U1H6B4_OUTPUT" / "h2_real"
    d1 = root / "04_TEMP" / "U1H6B5D1_OUTPUT"
    return {
        "ControlledRequest": b3 / "u001_h2_controlled_request.json",
        "H2Updates": h2 / "u001_adapter_source_updates.csv",
        "H2Catalog": h2 / "u001_adapter_signal_catalog.csv",
        "H2Quality": h2 / "u001_adapter_quality.json",
        "D1Diagnostic": d1 / "u001_h6b5d1_diagnostic.json",
        "D1Capsules": d1 / "u001_h6b5d1_capsules.txt",
    }


def source_from_request(request: Dict[str, Any]) -> Path:
    inputs = request.get("Inputs")
    if not isinstance(inputs, list):
        raise PathDiagnosticError("REQUEST_INPUTS")
    matches: List[Path] = []
    for item in inputs:
        if not isinstance(item, dict) or item.get("Kind") != "MATLAB_CAN_CSV":
            continue
        raw = item.get("Path", item.get("FilePath"))
        if isinstance(raw, str) and raw.strip():
            matches.append(Path(raw))
    if len(matches) != 1 or not matches[0].is_file():
        raise PathDiagnosticError("RAW_SOURCE_NOT_EXACTLY_ONE_OR_MISSING")
    return matches[0]


def frame_identity(request: Dict[str, Any]) -> Tuple[int, str, str]:
    config = request.get("Configuration")
    key = config.get("FrameKey") if isinstance(config, dict) else None
    if not isinstance(key, dict):
        raise PathDiagnosticError("FRAME_KEY")
    try:
        channel = int(key["Channel"])
    except (KeyError, TypeError, ValueError) as exc:
        raise PathDiagnosticError("FRAME_CHANNEL") from exc
    can_id = str(key.get("CAN_ID", "")).strip().lower()
    signal = str(key.get("SignalName", "")).strip()
    if not can_id or not signal:
        raise PathDiagnosticError("FRAME_IDENTITY_EMPTY")
    return channel, can_id, signal


def frame_catalog_row(path: Path, identity: Tuple[int, str, str]) -> Dict[str, str]:
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        matches = [
            row for row in reader
            if int(row["SourceChannel"]) == identity[0]
            and row["SourceMessageId"].strip().lower() == identity[1]
            and row["SourceSignalName"].strip() == identity[2]
        ]
    if len(matches) != 1:
        raise PathDiagnosticError("FRAME_CATALOG_MATCH_COUNT")
    return matches[0]


def detect_header(path: Path, required_columns: Sequence[str]) -> Tuple[str, int, List[str]]:
    required = set(required_columns)
    with path.open("r", encoding="utf-8-sig", errors="replace", newline="") as stream:
        lines = [stream.readline() for _ in range(50)]
    lines = [line for line in lines if line]
    for row_index, line in enumerate(lines):
        for delimiter in (",", "\t", ";"):
            try:
                columns = next(csv.reader([line], delimiter=delimiter))
            except csv.Error:
                continue
            normalized = [column.strip() for column in columns]
            if required.issubset(normalized):
                return delimiter, row_index, normalized
    raise PathDiagnosticError("RAW_HEADER_NOT_FOUND")


def normalized_key(value: str) -> str:
    return re.sub(r"[^a-z0-9]", "", value.casefold())


def optional_column(columns: Sequence[str], aliases: Iterable[str]) -> Optional[str]:
    index = {normalized_key(column): column for column in columns}
    for alias in aliases:
        if normalized_key(alias) in index:
            return index[normalized_key(alias)]
    return None


def decimal_or_none(value: str) -> Optional[Decimal]:
    text = str(value).strip()
    if not text:
        return None
    try:
        parsed = Decimal(text)
    except InvalidOperation:
        return None
    return parsed if parsed.is_finite() else None


def quantile(values: List[int], fraction: float) -> int:
    if not values:
        return 0
    ordered = sorted(values)
    return ordered[int(round((len(ordered) - 1) * fraction))]


def scan_raw_source(
    source: Path,
    byte_columns: Sequence[str],
) -> Dict[str, Any]:
    delimiter, header_row, columns = detect_header(source, byte_columns)
    positions = {name: index for index, name in enumerate(columns)}
    byte_positions = [positions[name] for name in byte_columns]
    datetime_name = optional_column(columns, ("DateTime", "SourceDateTime", "DateAndTime"))
    time_name = optional_column(columns, ("Time[us]", "SourceTimeUs", "Microseconds"))
    datetime_position = positions.get(datetime_name) if datetime_name else None
    time_position = positions.get(time_name) if time_name else None

    total_rows = 0
    any_rows = 0
    complete_rows = 0
    incomplete_rows = 0
    complete_positions: List[int] = []
    datetime_nonempty = 0
    datetime_decrease = 0
    time_nonempty = 0
    time_decrease = 0
    previous_datetime: Optional[str] = None
    previous_time: Optional[Decimal] = None

    with source.open("r", encoding="utf-8-sig", errors="replace", newline="") as stream:
        for _ in range(header_row):
            next(stream, None)
        reader = csv.reader(stream, delimiter=delimiter)
        observed_header = next(reader, None)
        if observed_header is None or [item.strip() for item in observed_header] != columns:
            raise PathDiagnosticError("RAW_HEADER_CHANGED")
        for source_row, cells in enumerate(reader):
            total_rows += 1
            if len(cells) < len(columns):
                cells.extend([""] * (len(columns) - len(cells)))
            present = [str(cells[index]).strip() != "" for index in byte_positions]
            if any(present):
                any_rows += 1
                if all(present):
                    complete_rows += 1
                    complete_positions.append(source_row)
                else:
                    incomplete_rows += 1
            if datetime_position is not None:
                value = str(cells[datetime_position]).strip()
                if value:
                    datetime_nonempty += 1
                    if previous_datetime is not None and value < previous_datetime:
                        datetime_decrease += 1
                    previous_datetime = value
            if time_position is not None:
                value = decimal_or_none(cells[time_position])
                if value is not None:
                    time_nonempty += 1
                    if previous_time is not None and value < previous_time:
                        time_decrease += 1
                    previous_time = value
            if total_rows % 50000 == 0:
                print(f"[PROGRESS] raw_rows={total_rows}", flush=True)

    gaps = [later - earlier for earlier, later in zip(complete_positions, complete_positions[1:])]
    return {
        "TotalRows": total_rows,
        "FrameAnyByteRows": any_rows,
        "FrameCompleteRows": complete_rows,
        "FrameIncompleteRows": incomplete_rows,
        "FrameFirstRow": complete_positions[0] if complete_positions else None,
        "FrameLastRow": complete_positions[-1] if complete_positions else None,
        "FrameGapQ50": quantile(gaps, 0.50),
        "FrameGapQ95": quantile(gaps, 0.95),
        "FrameGapMaximum": max(gaps) if gaps else 0,
        "DateTimeColumn": datetime_name,
        "DateTimeNonemptyRows": datetime_nonempty,
        "DateTimeDecreaseCount": datetime_decrease,
        "TimeUsColumn": time_name,
        "TimeUsNonemptyRows": time_nonempty,
        "TimeUsDecreaseCount": time_decrease,
    }


def exact_integer(value: str, label: str) -> int:
    parsed = decimal_or_none(value)
    if parsed is None or parsed != parsed.to_integral_value():
        raise PathDiagnosticError(label)
    return int(parsed)


def scan_h2_updates(
    path: Path,
    identity: Tuple[int, str, str],
) -> Dict[str, Any]:
    frame_rows: List[int] = []
    frame_values: List[int] = []
    group_rows: Dict[Tuple[int, str], set[int]] = defaultdict(set)
    group_first: Dict[Tuple[int, str], int] = {}
    group_last: Dict[Tuple[int, str], int] = {}
    unique_source_rows: set[int] = set()

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {
            "SourceRowIndex", "SourceChannel", "SourceMessageId",
            "SourceSignalName", "PhysicalValue", "DecodeValid",
        }
        if reader.fieldnames is None or not required.issubset(reader.fieldnames):
            raise PathDiagnosticError("H2_UPDATE_COLUMNS")
        for row in reader:
            source_row = int(row["SourceRowIndex"])
            unique_source_rows.add(source_row)
            channel = int(row["SourceChannel"])
            can_id = row["SourceMessageId"].strip().lower()
            group = (channel, can_id)
            group_rows[group].add(source_row)
            group_first[group] = min(group_first.get(group, source_row), source_row)
            group_last[group] = max(group_last.get(group, source_row), source_row)
            if (
                channel == identity[0]
                and can_id == identity[1]
                and row["SourceSignalName"].strip() == identity[2]
                and row["DecodeValid"].strip().casefold() in {"1", "true"}
            ):
                frame_rows.append(source_row)
                frame_values.append(exact_integer(row["PhysicalValue"], "H2_FRAME_VALUE"))

    if not frame_rows:
        raise PathDiagnosticError("H2_FRAME_ROWS_EMPTY")
    frame_decreases = sum(later < earlier for earlier, later in zip(frame_values, frame_values[1:]))
    frame_positive_deltas = [later - earlier for earlier, later in zip(frame_values, frame_values[1:]) if later > earlier]
    last_frame_row = frame_rows[-1]
    groups: List[Dict[str, Any]] = []
    for group in sorted(group_rows):
        rows = group_rows[group]
        groups.append({
            "Channel": group[0],
            "CAN_ID": group[1],
            "UpdateSourceRows": len(rows),
            "FirstSourceRow": group_first[group],
            "LastSourceRow": group_last[group],
            "RowsAfterLastFrame": sum(value > last_frame_row for value in rows),
        })
    return {
        "UniqueSourceRows": len(unique_source_rows),
        "FrameUpdateCount": len(frame_rows),
        "FrameUniqueValueCount": len(set(frame_values)),
        "FrameMinimum": min(frame_values),
        "FrameMaximum": max(frame_values),
        "FrameFirstSourceRow": frame_rows[0],
        "FrameLastSourceRow": frame_rows[-1],
        "FrameDecreaseCount": frame_decreases,
        "FramePositiveDeltaQ50": quantile(frame_positive_deltas, 0.50),
        "FramePositiveDeltaQ95": quantile(frame_positive_deltas, 0.95),
        "FramePositiveDeltaMaximum": max(frame_positive_deltas) if frame_positive_deltas else 0,
        "Groups": groups,
    }


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        temp = root / "04_TEMP"
        prepared = temp / PREPARED_FOLDER
        output = temp / OUTPUT_FOLDER
        archive = root / "A-Archive" / ("U1H6B5D3_HYGIENE_" + datetime.now().strftime("%Y%m%dT%H%M%S_%f"))
        print("[PREPARE] U001 H6B5D3 FrameCount path diagnostic")
        print("[PREPARE] Why: distinguish raw-source coverage, H2 decode loss, and cross-group source-domain failure.")
        if prepared.exists():
            shutil.rmtree(prepared)
            print(f"[CLEAN] 04_TEMP/{PREPARED_FOLDER} | REMOVED | Why: invalidate stale receipt.")
        if output.exists():
            archive.mkdir(parents=True, exist_ok=True)
            shutil.move(str(output), str(archive / output.name))
            print(f"[CLEAN] 04_TEMP/{OUTPUT_FOLDER} | ARCHIVED | Why: prevent stale evidence reuse.")
        for name in SUPERSEDED_TRANSFER_NAMES:
            path = temp / name
            if path.exists():
                archive.mkdir(parents=True, exist_ok=True)
                shutil.move(str(path), str(archive / name))
                print(f"[CLEAN] 04_TEMP/{name} | ARCHIVED | Why: superseded transfer file.")
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        source = source_from_request(request)
        missing = [label for label, path in paths.items() if not path.is_file()]
        for label, path in paths.items():
            print(f"[CHECK] {label} | {'FOUND' if path.is_file() else 'MISSING'} | Why: required diagnostic evidence.")
        print(f"[CHECK] RawSource | FOUND | basename={source.name} | Why: compare raw 0x620 rows against H2 output.")
        print("[KEEP] H6B3 / H6B4 / H6B5 / H6B5D1 outputs | Why: active traceability chain.")
        if missing:
            raise PathDiagnosticError("PREPARE_MISSING:" + ".".join(missing))
        prepared.mkdir(parents=True)
        write_json(prepared / "u001_h6b5d3_prepare_receipt.json", {
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "RawSource": str(source),
            "PreparedUtc": utc_now(),
            "Status": "READY",
            "CompletionCode": 0,
        })
        print("U1H6B5D3P-C7-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1H6B5D3P-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def require_prepare(root: Path) -> Dict[str, Any]:
    receipt = read_json(root / "04_TEMP" / PREPARED_FOLDER / "u001_h6b5d3_prepare_receipt.json", "PREPARE_RECEIPT")
    if receipt.get("ProgramId") != PROGRAM_ID or receipt.get("ProgramVersion") != PROGRAM_VERSION:
        raise PathDiagnosticError("PREPARE_IDENTITY")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise PathDiagnosticError("PREPARE_STATUS")
    return receipt


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        require_prepare(root)
        print("[SELFTEST] Quantiles and integer parsing | Why: protect compact distribution evidence.")
        if quantile([1, 2, 3, 100], 0.5) != 3 or exact_integer("12.0", "T") != 12:
            raise AssertionError("MATH_HELPERS")
        print("[PASS] Distribution and exact-integer helpers verified.")
        print("[SELFTEST] Frame identity and byte-column contract | Why: guarantee the raw/H2 comparison addresses CH2 0x620 FrameCount only.")
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        identity = frame_identity(request)
        catalog = frame_catalog_row(paths["H2Catalog"], identity)
        byte_columns = [value for value in catalog["SourceByteColumns"].split("|") if value]
        if len(byte_columns) != int(catalog["MessageDLC"]) or len(byte_columns) != 8:
            raise AssertionError("FRAME_BYTE_COLUMNS")
        print("[PASS] Exact FrameCount identity and eight raw byte columns verified.")
        print("U1H6B5D3-ST-T2-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1H6B5D3-ST-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def execute(root_argument: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        receipt = require_prepare(root)
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        source = source_from_request(request)
        if str(source) != str(receipt.get("RawSource")):
            raise PathDiagnosticError("PREPARED_SOURCE_CHANGED")
        identity = frame_identity(request)
        catalog = frame_catalog_row(paths["H2Catalog"], identity)
        byte_columns = [value for value in catalog["SourceByteColumns"].split("|") if value]

        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            raise PathDiagnosticError("OUTPUT_EXISTS_RUN_PREPARE")
        output.mkdir(parents=True)

        raw = scan_raw_source(source, byte_columns)
        h2 = scan_h2_updates(paths["H2Updates"], identity)
        parity = int(raw["FrameCompleteRows"] == h2["FrameUpdateCount"])
        report = {
            "DiagnosticSchema": "U001_H6B5D3_FRAME_KEY_PATH_DIAGNOSTIC_V1_0",
            "GeneratedUtc": utc_now(),
            "FrameIdentity": {"Channel": identity[0], "CAN_ID": identity[1], "SignalName": identity[2]},
            "RawSource": raw,
            "H2": h2,
            "RawCompleteEqualsH2FrameUpdates": bool(parity),
            "Status": "DIAGNOSTIC_COMPLETE",
            "CompletionCode": 0,
        }
        write_json(output / "u001_h6b5d3_frame_key_path_diagnostic.json", report)

        capsule3 = (
            f"H6B5D3-R{raw['TotalRows']}-A{raw['FrameAnyByteRows']}-C{raw['FrameCompleteRows']}"
            f"-I{raw['FrameIncompleteRows']}-H{h2['FrameUpdateCount']}-P{parity}"
            f"-U{h2['FrameUniqueValueCount']}-F{h2['FrameFirstSourceRow']}"
            f"-L{h2['FrameLastSourceRow']}-G{raw['FrameGapMaximum']}"
            f"-D{h2['FrameDecreaseCount']}-X0-C0"
        )
        capsule4 = (
            f"H6B5D4-DN{raw['DateTimeNonemptyRows']}-DD{raw['DateTimeDecreaseCount']}"
            f"-TN{raw['TimeUsNonemptyRows']}-TD{raw['TimeUsDecreaseCount']}"
            f"-Q{h2['FramePositiveDeltaQ50']}.{h2['FramePositiveDeltaQ95']}.{h2['FramePositiveDeltaMaximum']}"
            f"-N{h2['UniqueSourceRows']}-X0-C0"
        )
        group_tokens = []
        for item in h2["Groups"]:
            can_token = item["CAN_ID"].replace("0x", "")
            group_tokens.append(
                f"G{item['Channel']}.{can_token}.{item['UpdateSourceRows']}."
                f"{item['FirstSourceRow']}.{item['LastSourceRow']}.{item['RowsAfterLastFrame']}"
            )
        capsule5 = "H6B5D5-N%d-%s-X0-C0" % (len(group_tokens), "-".join(group_tokens))
        (output / "u001_h6b5d3_capsules.txt").write_text(
            capsule3 + "\n" + capsule4 + "\n" + capsule5 + "\n", encoding="utf-8"
        )
        print(capsule3)
        print(capsule4)
        print(capsule5)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        write_json(output / "u001_h6b5d3_error.json", {"Type": type(exc).__name__, "Detail": str(exc)[:3000]})
        print(f"H6B5D3-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H6B5D3 FrameCount path diagnostic")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    sys.exit(main())
