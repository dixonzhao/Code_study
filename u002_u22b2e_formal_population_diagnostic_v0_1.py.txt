#!/usr/bin/env python3
"""只读诊断B2DR3为什么没有FORMAL_ELIGIBLE样本。

本程序不执行Module，不修改Artifact，也不改变任何分析数学。
它验证B2DR3既有Artifact的Hash与FrameCount对齐后，联合检查：

1. Camera重建的有效性、SelectionStatus与SelectionReason；
2. Radar有效性与EvidenceClass；
3. S2三通道人群分类与EligibilityReason；
4. 短区间起点和Camera Seed之间是否缺少有状态模块所需的历史上下文。
"""

from __future__ import annotations

import argparse
import csv
from collections import Counter
import hashlib
import json
import math
import os
from pathlib import Path, PurePosixPath
import re
import tempfile
from typing import Dict, Iterable, List, Mapping, Sequence, Tuple


PROGRAM_ID = "UDP002-U22B2E-FORMAL-POPULATION-DIAGNOSTIC"
PROGRAM_VERSION = "0.1"
RUN_ID = "U22B2DR3_TC001_F41480_41750_V01"
FRAME_START = 41480
FRAME_END = 41750

ARTIFACT_INDEX = Path("run_artifact_index_v0_1.json")
RUN_PLAN = Path("00_CONFIG/resolved_run_plan.json")
DIAGNOSTIC_REPORT = Path(
    "02_DIAGNOSTICS/b2e_formal_population_diagnostic_v0_1.json"
)
EXAMPLE_TABLE = Path(
    "02_DIAGNOSTICS/b2e_formal_population_examples_v0_1.csv"
)

REQUIRED_TYPES = (
    "CAMERA_TARGET_RECONSTRUCTION",
    "RADAR_VALIDITY",
    "COMPARISON_POPULATION",
)


class DiagnosticError(RuntimeError):
    """表示一个可稳定编码的诊断失败。"""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = compact_token(reason, 64)
        self.detail = compact_token(detail, 120)
        super().__init__(self.reason)


def compact_token(value: object, limit: int = 80) -> str:
    """把自由文本压缩成可人工回传的稳定Token。"""

    text = re.sub(r"[^A-Z0-9_.:+-]+", "_", str(value).upper())
    return text.strip("_")[:limit] or "NA"


def canonical_bytes(value: object) -> bytes:
    """以确定性JSON格式序列化诊断记录。"""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    """流式计算文件Hash，避免把较大Artifact一次读入内存。"""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    """只接受真实普通文件，不接受符号链接。"""

    return path.is_file() and not path.is_symlink()


def load_json(path: Path, reason: str) -> Dict[str, object]:
    """读取JSON对象并拒绝缺失、链接、语法错误和非Object顶层。"""

    if not regular_file(path):
        raise DiagnosticError(reason + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise DiagnosticError(reason + "_JSON", path.name) from exc
    if not isinstance(value, dict):
        raise DiagnosticError(reason + "_TYPE", path.name)
    return value


def write_json_atomic(path: Path, value: Mapping[str, object]) -> None:
    """原子写入可重复生成的只读诊断结果。"""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def safe_relative(value: object, field: str) -> PurePosixPath:
    """只接受不越界的Portable相对路径。"""

    text = str(value)
    if "\\" in text:
        raise DiagnosticError(field + "_SEPARATOR", text)
    path = PurePosixPath(text)
    if path.is_absolute() or not path.parts:
        raise DiagnosticError(field + "_PATH", text)
    if any(part in {"", ".", ".."} for part in path.parts):
        raise DiagnosticError(field + "_PATH", text)
    return path


def resolve_inside(root: Path, relative: object, field: str) -> Path:
    """解析相对路径并保证目标仍位于所属目录内。"""

    path = root.joinpath(*safe_relative(relative, field).parts).resolve()
    try:
        path.relative_to(root.resolve())
    except ValueError as exc:
        raise DiagnosticError(field + "_ESCAPE", relative) from exc
    return path


def parse_nonnegative_integer(value: object, field: str) -> int:
    """解析非负整数，同时拒绝bool和非整数浮点数。"""

    if isinstance(value, bool):
        raise DiagnosticError(field, value)
    try:
        number = int(value)
    except (TypeError, ValueError) as exc:
        raise DiagnosticError(field, value) from exc
    if number < 0:
        raise DiagnosticError(field, value)
    if isinstance(value, float) and not value.is_integer():
        raise DiagnosticError(field, value)
    return number


def parse_boolean_text(value: object, field: str) -> bool:
    """解析CSV中的严格0/1布尔值。"""

    text = str(value).strip()
    if text not in {"0", "1"}:
        raise DiagnosticError(field, text)
    return text == "1"


def require_workspace(path: Path) -> Tuple[Path, Path]:
    """验证U-UDP002工作区和B2DR3 Run目录。"""

    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise DiagnosticError("WORKSPACE_ROOT", root.name)
    run_root = root / "07_OUTPUT" / RUN_ID
    if not run_root.is_dir() or run_root.is_symlink():
        raise DiagnosticError("RUN_ROOT", RUN_ID)
    if not regular_file(run_root / ARTIFACT_INDEX):
        raise DiagnosticError("ARTIFACT_INDEX", RUN_ID)
    if not regular_file(run_root / RUN_PLAN):
        raise DiagnosticError("RUN_PLAN", RUN_ID)
    return root, run_root


def load_artifact_entries(run_root: Path) -> Dict[str, Mapping[str, object]]:
    """读取Artifact索引并按ArtifactType建立唯一映射。"""

    index = load_json(run_root / ARTIFACT_INDEX, "ARTIFACT_INDEX")
    entries = index.get("Artifacts")
    if not isinstance(entries, list):
        raise DiagnosticError("ARTIFACTS_TYPE")
    by_type: Dict[str, Mapping[str, object]] = {}
    for entry in entries:
        if not isinstance(entry, dict):
            raise DiagnosticError("ARTIFACT_ENTRY_TYPE")
        artifact_type = str(entry.get("ArtifactType"))
        if artifact_type in by_type:
            raise DiagnosticError("ARTIFACT_TYPE_DUPLICATE", artifact_type)
        by_type[artifact_type] = entry
    missing = [name for name in REQUIRED_TYPES if name not in by_type]
    if missing:
        raise DiagnosticError("ARTIFACT_TYPE_MISSING", missing[0])
    return by_type


def artifact_payloads(
    run_root: Path,
    entry: Mapping[str, object],
) -> Tuple[Path, Dict[str, object], Dict[str, object]]:
    """Hash验证Manifest及其唯一主CSV和辅助Quality JSON。"""

    manifest_path = resolve_inside(
        run_root,
        entry.get("ManifestRelativePath"),
        "MANIFEST",
    )
    if not regular_file(manifest_path):
        raise DiagnosticError("MANIFEST_MISSING", entry.get("ArtifactType"))
    if sha256_file(manifest_path) != entry.get("ManifestSha256"):
        raise DiagnosticError("MANIFEST_HASH", entry.get("ArtifactType"))
    manifest = load_json(manifest_path, "MANIFEST")
    if manifest.get("ArtifactType") != entry.get("ArtifactType"):
        raise DiagnosticError("MANIFEST_TYPE", entry.get("ArtifactType"))
    payloads = manifest.get("Payloads")
    if not isinstance(payloads, list):
        raise DiagnosticError("PAYLOADS_TYPE", entry.get("ArtifactType"))
    primary = [
        item
        for item in payloads
        if isinstance(item, dict)
        and item.get("PayloadKind") == "PRIMARY"
        and item.get("MediaType") == "text/csv"
    ]
    quality = [
        item
        for item in payloads
        if isinstance(item, dict)
        and item.get("PayloadKind") == "AUXILIARY"
        and item.get("MediaType") == "application/json"
    ]
    if len(primary) != 1 or len(quality) != 1:
        raise DiagnosticError("PAYLOAD_ROLE_COUNT", entry.get("ArtifactType"))
    package_root = manifest_path.parent
    primary_path = resolve_inside(
        package_root,
        primary[0].get("RelativePath"),
        "PRIMARY",
    )
    quality_path = resolve_inside(
        package_root,
        quality[0].get("RelativePath"),
        "QUALITY",
    )
    for path, record in ((primary_path, primary[0]), (quality_path, quality[0])):
        if not regular_file(path):
            raise DiagnosticError("PAYLOAD_MISSING", path.name)
        if sha256_file(path) != record.get("Sha256"):
            raise DiagnosticError("PAYLOAD_HASH", path.name)
    return primary_path, load_json(quality_path, "QUALITY"), manifest


def read_csv_rows(
    path: Path,
    required_columns: Sequence[str],
) -> Tuple[List[Dict[str, str]], List[int]]:
    """读取短区间CSV并验证Header、FrameCount整数性和严格递增。"""

    if not regular_file(path):
        raise DiagnosticError("CSV_MISSING", path.name)
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        header = reader.fieldnames or []
        missing = [name for name in required_columns if name not in header]
        if missing:
            raise DiagnosticError("CSV_COLUMN", missing[0])
        rows = [dict(row) for row in reader]
    if not rows:
        raise DiagnosticError("CSV_EMPTY", path.name)
    frames: List[int] = []
    for row in rows:
        text = str(row["FrameCount"]).strip()
        try:
            number = float(text)
        except ValueError as exc:
            raise DiagnosticError("FRAMECOUNT_PARSE", text) from exc
        if not math.isfinite(number) or not number.is_integer():
            raise DiagnosticError("FRAMECOUNT_INTEGER", text)
        frames.append(int(number))
    if any(right <= left for left, right in zip(frames, frames[1:])):
        raise DiagnosticError("FRAMECOUNT_ORDER", path.name)
    return rows, frames


def counter_dict(values: Iterable[str]) -> Dict[str, int]:
    """返回按Key稳定排序的计数映射。"""

    return dict(sorted(Counter(values).items()))


def classify_root_cause(
    row_count: int,
    camera_valid_rows: int,
    formal_rows: int,
    status_counts: Mapping[str, int],
    reason_counts: Mapping[str, int],
    seed_frame: object,
    frame_start: int,
) -> Tuple[str, bool]:
    """根据完整等价类证据分类，而不是按单个案例猜测。"""

    ambiguous = int(status_counts.get("AMBIGUOUS", 0))
    unseeded = int(reason_counts.get("MULTIPLE_UNSEEDED_CANDIDATES", 0))
    seed_before_scope = isinstance(seed_frame, int) and seed_frame < frame_start
    if (
        row_count > 0
        and camera_valid_rows == 0
        and formal_rows == 0
        and ambiguous == row_count
        and unseeded == row_count
        and seed_before_scope
    ):
        return "STATEFUL_CAMERA_WARMUP_MISSING", True
    if camera_valid_rows == 0 and unseeded > 0:
        return "CAMERA_UNSEEDED_OR_WARMUP_PARTIAL", False
    if camera_valid_rows == 0:
        return "CAMERA_NOT_FORMAL", False
    if formal_rows == 0:
        return "RADAR_OR_POPULATION_NOT_FORMAL", False
    return "FORMAL_POPULATION_PRESENT", False


def example_rows(
    camera_rows: Sequence[Mapping[str, str]],
    radar_rows: Sequence[Mapping[str, str]],
    population_rows: Sequence[Mapping[str, str]],
) -> List[Dict[str, str]]:
    """每种语义组合最多保存一行代表样本，避免生成宽大人工表。"""

    examples: List[Dict[str, str]] = []
    observed = set()
    for camera, radar, population in zip(camera_rows, radar_rows, population_rows):
        key = (
            camera["SelectionStatus"],
            camera["SelectionReason"],
            radar["RadarMeasurementValid"],
            radar["RadarEvidenceClass"],
            population["ComparisonEligibility"],
            population["EligibilityReason"],
        )
        if key in observed:
            continue
        observed.add(key)
        examples.append({
            "FrameCount": camera["FrameCount"],
            "CanonicalCameraValid": camera["CanonicalCameraValid"],
            "SelectionStatus": camera["SelectionStatus"],
            "SelectionReason": camera["SelectionReason"],
            "RadarMeasurementValid": radar["RadarMeasurementValid"],
            "RadarEvidenceClass": radar["RadarEvidenceClass"],
            "ComparisonEligibility": population["ComparisonEligibility"],
            "EligibilityReason": population["EligibilityReason"],
        })
    return examples


def write_example_csv(path: Path, rows: Sequence[Mapping[str, str]]) -> None:
    """写入窄表代表样本，供人工快速核查。"""

    columns = (
        "FrameCount",
        "CanonicalCameraValid",
        "SelectionStatus",
        "SelectionReason",
        "RadarMeasurementValid",
        "RadarEvidenceClass",
        "ComparisonEligibility",
        "EligibilityReason",
    )
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    with temporary.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=columns)
        writer.writeheader()
        for row in rows:
            writer.writerow({name: row.get(name, "") for name in columns})
    os.replace(temporary, path)


def diagnose(workspace_root: Path) -> str:
    """对B2DR3现有Artifact执行一次只读联合语义诊断。"""

    _, run_root = require_workspace(workspace_root)
    plan = load_json(run_root / RUN_PLAN, "RUN_PLAN")
    if plan.get("RunId") != RUN_ID:
        raise DiagnosticError("RUN_ID", plan.get("RunId"))
    if plan.get("FrameStart") != FRAME_START or plan.get("FrameEnd") != FRAME_END:
        raise DiagnosticError("RUN_SCOPE", f"{plan.get('FrameStart')}.{plan.get('FrameEnd')}")

    entries = load_artifact_entries(run_root)
    camera_path, camera_quality, camera_manifest = artifact_payloads(
        run_root,
        entries["CAMERA_TARGET_RECONSTRUCTION"],
    )
    radar_path, radar_quality, radar_manifest = artifact_payloads(
        run_root,
        entries["RADAR_VALIDITY"],
    )
    population_path, population_quality, population_manifest = artifact_payloads(
        run_root,
        entries["COMPARISON_POPULATION"],
    )

    camera_rows, camera_frames = read_csv_rows(
        camera_path,
        (
            "FrameCount",
            "CanonicalCameraValid",
            "SelectionStatus",
            "SelectionReason",
        ),
    )
    radar_rows, radar_frames = read_csv_rows(
        radar_path,
        (
            "FrameCount",
            "RadarMeasurementValid",
            "RadarValidityReason",
            "RadarEvidenceClass",
        ),
    )
    population_rows, population_frames = read_csv_rows(
        population_path,
        (
            "FrameCount",
            "CanonicalCameraValid",
            "CameraSelectionStatus",
            "RadarMeasurementValid",
            "RadarEvidenceClass",
            "ComparisonEligibility",
            "EligibilityReason",
        ),
    )
    if camera_frames != radar_frames or camera_frames != population_frames:
        raise DiagnosticError("FRAME_ALIGNMENT")
    if camera_frames[0] != FRAME_START or camera_frames[-1] != FRAME_END:
        raise DiagnosticError("FRAME_BOUNDS", f"{camera_frames[0]}.{camera_frames[-1]}")

    for camera, radar, population in zip(camera_rows, radar_rows, population_rows):
        if camera["FrameCount"] != radar["FrameCount"]:
            raise DiagnosticError("FRAME_IDENTITY_CAMERA_RADAR")
        if camera["FrameCount"] != population["FrameCount"]:
            raise DiagnosticError("FRAME_IDENTITY_POPULATION")
        if camera["CanonicalCameraValid"] != population["CanonicalCameraValid"]:
            raise DiagnosticError("CAMERA_VALID_PROPAGATION", camera["FrameCount"])
        if camera["SelectionStatus"] != population["CameraSelectionStatus"]:
            raise DiagnosticError("CAMERA_STATUS_PROPAGATION", camera["FrameCount"])
        if radar["RadarMeasurementValid"] != population["RadarMeasurementValid"]:
            raise DiagnosticError("RADAR_VALID_PROPAGATION", camera["FrameCount"])
        if radar["RadarEvidenceClass"] != population["RadarEvidenceClass"]:
            raise DiagnosticError("RADAR_CLASS_PROPAGATION", camera["FrameCount"])

    row_count = len(camera_rows)
    camera_valid_rows = sum(
        parse_boolean_text(row["CanonicalCameraValid"], "CAMERA_VALID")
        for row in camera_rows
    )
    radar_valid_rows = sum(
        parse_boolean_text(row["RadarMeasurementValid"], "RADAR_VALID")
        for row in radar_rows
    )
    status_counts = counter_dict(row["SelectionStatus"] for row in camera_rows)
    selection_reason_counts = counter_dict(
        row["SelectionReason"] for row in camera_rows
    )
    eligibility_counts = counter_dict(
        row["ComparisonEligibility"] for row in population_rows
    )
    eligibility_reason_counts = counter_dict(
        row["EligibilityReason"] for row in population_rows
    )
    radar_evidence_counts = counter_dict(
        row["RadarEvidenceClass"] for row in radar_rows
    )
    formal_rows = eligibility_counts.get("FORMAL_ELIGIBLE", 0)
    investigation_rows = eligibility_counts.get("INVESTIGATION_ONLY", 0)
    ineligible_rows = eligibility_counts.get("INELIGIBLE", 0)
    if formal_rows + investigation_rows + ineligible_rows != row_count:
        raise DiagnosticError("ELIGIBILITY_ACCOUNTING")

    configuration = camera_quality.get("Configuration")
    if not isinstance(configuration, dict):
        raise DiagnosticError("CAMERA_CONFIGURATION")
    seed_frame = configuration.get("SeedFrame")
    root_cause, warmup_proven = classify_root_cause(
        row_count,
        camera_valid_rows,
        formal_rows,
        status_counts,
        selection_reason_counts,
        seed_frame,
        FRAME_START,
    )
    examples = example_rows(camera_rows, radar_rows, population_rows)
    write_example_csv(run_root / EXAMPLE_TABLE, examples)

    report = {
        "DiagnosticContract": "UDP002_U22B2E_FORMAL_POPULATION_DIAGNOSTIC_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "RunId": RUN_ID,
        "ReadOnlyArtifactInspection": True,
        "ArtifactMutation": False,
        "FrameStart": camera_frames[0],
        "FrameEnd": camera_frames[-1],
        "FrameRows": row_count,
        "CameraValidRows": camera_valid_rows,
        "CameraInvalidRows": row_count - camera_valid_rows,
        "RadarValidRows": radar_valid_rows,
        "RadarInvalidRows": row_count - radar_valid_rows,
        "SelectionStatusCounts": status_counts,
        "SelectionReasonCounts": selection_reason_counts,
        "RadarEvidenceClassCounts": radar_evidence_counts,
        "EligibilityCounts": eligibility_counts,
        "EligibilityReasonCounts": eligibility_reason_counts,
        "ConfiguredSeedFrame": seed_frame,
        "ConfiguredSeedRole": configuration.get("SeedRole"),
        "RunStartsAfterConfiguredSeed": (
            isinstance(seed_frame, int) and FRAME_START > seed_frame
        ),
        "RootCauseClassification": root_cause,
        "StatefulWarmupAbsenceProven": warmup_proven,
        "RepresentativeCombinationCount": len(examples),
        "RepresentativeExamplesRelativePath": EXAMPLE_TABLE.as_posix(),
        "ManifestArtifactIds": {
            "Camera": camera_manifest.get("ArtifactId"),
            "Radar": radar_manifest.get("ArtifactId"),
            "Population": population_manifest.get("ArtifactId"),
        },
        "QualityCrossChecks": {
            "CameraFrameRows": camera_quality.get("FrameRows"),
            "CameraCanonicalValidRows": camera_quality.get("CanonicalValidRows"),
            "RadarFrameRows": radar_quality.get("FrameRows"),
            "RadarValidRows": radar_quality.get("RadarValidRows"),
            "PopulationFrameRows": population_quality.get("FrameRows"),
            "FormalEligibleRows": population_quality.get("FormalEligibleRows"),
        },
        "EngineeringInterpretation": (
            "B2DR3的结构性五模块Run成功，但短区间绕过了Camera有状态重建所需的Seed或预热历史。"
            if warmup_proven
            else "B2DR3结构成功；正式人群为空的原因需要按本报告计数继续判定。"
        ),
        "NextDecisionBoundary": (
            "为有状态S1 Module分离Context/Warmup范围与Evaluation范围，或使用包含Seed的全历史重建Artifact后再裁剪。"
            if warmup_proven
            else "不得直接修改算法；先根据SelectionStatus与EligibilityReason确定责任层。"
        ),
    }
    write_json_atomic(run_root / DIAGNOSTIC_REPORT, report)
    warmup_code = 1 if warmup_proven else 0
    ambiguous_rows = status_counts.get("AMBIGUOUS", 0)
    return (
        f"B2E N-R{row_count}-CV{camera_valid_rows}-RV{radar_valid_rows}"
        f"-F{formal_rows}-I{investigation_rows}-N{ineligible_rows}"
        f"-A{ambiguous_rows}-W{warmup_code}-X0 C-0"
    )


def self_test() -> str:
    """覆盖路径、类型、分类与代表样本等关键等价类。"""

    checks = 0

    def check(condition: bool) -> None:
        nonlocal checks
        if not condition:
            raise AssertionError("SELFTEST")
        checks += 1

    check(compact_token("a b/c") == "A_B_C")
    check(safe_relative("a/b.json", "P").as_posix() == "a/b.json")
    try:
        safe_relative("../escape", "P")
    except DiagnosticError:
        check(True)
    else:
        check(False)
    try:
        parse_boolean_text("true", "B")
    except DiagnosticError:
        check(True)
    else:
        check(False)
    check(parse_boolean_text("1", "B") is True)
    check(parse_boolean_text("0", "B") is False)
    check(parse_nonnegative_integer(0, "I") == 0)
    try:
        parse_nonnegative_integer(True, "I")
    except DiagnosticError:
        check(True)
    else:
        check(False)

    root, proven = classify_root_cause(
        3,
        0,
        0,
        {"AMBIGUOUS": 3},
        {"MULTIPLE_UNSEEDED_CANDIDATES": 3},
        1,
        10,
    )
    check(root == "STATEFUL_CAMERA_WARMUP_MISSING")
    check(proven is True)
    root, proven = classify_root_cause(
        3,
        0,
        0,
        {"AMBIGUOUS": 2, "INVALID": 1},
        {"MULTIPLE_UNSEEDED_CANDIDATES": 2},
        1,
        10,
    )
    check(root == "CAMERA_UNSEEDED_OR_WARMUP_PARTIAL")
    check(proven is False)
    root, _ = classify_root_cause(3, 0, 0, {"INVALID": 3}, {}, 1, 10)
    check(root == "CAMERA_NOT_FORMAL")
    root, _ = classify_root_cause(3, 2, 0, {"CONFIDENT": 2}, {}, 1, 10)
    check(root == "RADAR_OR_POPULATION_NOT_FORMAL")
    root, _ = classify_root_cause(3, 2, 2, {"CONFIDENT": 2}, {}, 1, 10)
    check(root == "FORMAL_POPULATION_PRESENT")
    check(counter_dict(["B", "A", "B"]) == {"A": 1, "B": 2})

    camera = [{
        "FrameCount": "1",
        "CanonicalCameraValid": "0",
        "SelectionStatus": "AMBIGUOUS",
        "SelectionReason": "MULTIPLE_UNSEEDED_CANDIDATES",
    }]
    radar = [{
        "FrameCount": "1",
        "RadarMeasurementValid": "1",
        "RadarEvidenceClass": "VALID_MEASUREMENT",
    }]
    population = [{
        "FrameCount": "1",
        "ComparisonEligibility": "INVESTIGATION_ONLY",
        "EligibilityReason": "CAMERA_SELECTION_UNCERTAIN",
    }]
    examples = example_rows(camera, radar, population)
    check(len(examples) == 1)
    check(examples[0]["FrameCount"] == "1")

    with tempfile.TemporaryDirectory() as directory:
        path = Path(directory) / "a.json"
        write_json_atomic(path, {"B": 2, "A": 1})
        check(load_json(path, "J") == {"A": 1, "B": 2})
        check(len(sha256_file(path)) == 64)

def build_parser() -> argparse.ArgumentParser:
    """建立明确互斥的CLI入口。"""

    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--diagnose", action="store_true")
    parser.add_argument("--workspace-root", type=Path)
    return parser


def main() -> int:
    """CLI入口；失败时同时给出短码和一行中文诊断。"""

    arguments = build_parser().parse_args()
    try:
        if arguments.self_test:
            print(self_test())
            return 0
        if arguments.workspace_root is None:
            raise DiagnosticError("WORKSPACE_REQUIRED")
        print(diagnose(arguments.workspace_root))
        return 0
    except DiagnosticError as exc:
        print(f"B2EE K-{exc.reason}-D-{exc.detail}-X1 C-7")
        return 7
    except Exception as exc:
        print(
            "B2EE K-UNEXPECTED"
            f"-D-{compact_token(type(exc).__name__ + ':' + str(exc), 120)}"
            "-X1 C-7"
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
