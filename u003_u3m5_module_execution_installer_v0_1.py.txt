)


class InstallError(RuntimeError):
    """Raised when transfer evidence or company topology is invalid."""


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def safe_members(archive: zipfile.ZipFile) -> tuple[list[str], list[str]]:
    members: list[str] = []
    for name in archive.namelist():
        pure = PurePosixPath(name)
        if pure.is_absolute() or ".." in pure.parts:
            raise InstallError(f"Unsafe ZIP member: {name}")
        if not name.endswith("/"):
            members.append(name)
    member_set = set(members)
    install_members = member_set - SAFE_AUXILIARY_DOCUMENTS
    ignored_documents = sorted(member_set & SAFE_AUXILIARY_DOCUMENTS)
    if install_members != REQUIRED_MEMBERS:
        missing = sorted(REQUIRED_MEMBERS - install_members)
        extra = sorted(install_members - REQUIRED_MEMBERS)
        raise InstallError(f"Package inventory mismatch. Missing={missing}; Extra={extra}")
    return sorted(install_members), ignored_documents


def validate_payloads(package_root: Path) -> list[tuple[str, str, str]]:
    manifest_path = package_root / "PACKAGE_SHA256.csv"
    with manifest_path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    if len(rows) != 7:
        raise InstallError("PACKAGE_SHA256.csv must contain exactly seven payload rows.")
    checked: list[tuple[str, str, str]] = []
    for row in rows:
        relative = row.get("RelativePath", "").strip()
        expected = row.get("Sha256", "").strip().lower()
        purpose = row.get("Purpose", "").strip()
        payload = (package_root / relative).resolve()
        try:
            payload.relative_to(package_root.resolve())
        except ValueError as error:
            raise InstallError(f"Manifest path escapes package root: {relative}") from error
        if not payload.is_file() or payload.is_symlink():
            raise InstallError(f"Payload is missing: {relative}")
        actual = sha256_file(payload)
        if actual != expected:
            raise InstallError(f"Payload hash mismatch: {relative}")
        checked.append((relative, actual, purpose))
    if {row[0] for row in checked} != REQUIRED_MEMBERS - {"PACKAGE_SHA256.csv"}:
        raise InstallError("Manifest payload inventory does not match the approved package.")
    return checked


def next_archive_root(root: Path, label: str) -> Path:
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    candidate = root / "A-Archive" / f"{label}_{stamp}"
    suffix = 1
    while candidate.exists():
        candidate = root / "A-Archive" / f"{label}_{stamp}_{suffix:02d}"
        suffix += 1
    return candidate


def move_to_archive(source: Path, archive_root: Path, relative: Path) -> None:
    destination = archive_root / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(source), str(destination))


def archive_previous_temp(root: Path) -> tuple[Path | None, int]:
    temp_root = root / "04_TEMP"
    temp_root.mkdir(parents=True, exist_ok=True)
    selected: list[tuple[Path, Path]] = []
    for name in PREVIOUS_PREPARED_NAMES:
        candidate = temp_root / name
        if candidate.exists():
            selected.append((candidate, Path(name)))
    for candidate in temp_root.iterdir():
        if candidate.name in PREVIOUS_PREPARED_NAMES:
            continue
        if any(candidate.name.startswith(prefix) for prefix in PREVIOUS_TRANSFER_PREFIXES):
            selected.append((candidate, Path(candidate.name)))
    current_prepared = temp_root / "U3M5_PREPARED"
    if current_prepared.exists():
        selected.append((current_prepared, Path("U3M5_PREPARED_PREVIOUS")))
    if not selected:
        return None, 0
    archive_root = next_archive_root(root, "U3M5_TEMP_CLEANUP")
    for source, relative in selected:
        move_to_archive(source, archive_root, Path("04_TEMP") / relative)
    return archive_root, len(selected)


def archive_replaced_payloads(root: Path, relative_paths: list[str]) -> tuple[Path | None, int]:
    existing = [relative for relative in relative_paths if (root / relative).is_file()]
    if not existing:
        return None, 0
    archive_root = next_archive_root(root, "U3M5_REPLACED")
    for relative in existing:
        destination = archive_root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(root / relative, destination)
    return archive_root, len(existing)


def validate_predecessors(root: Path) -> None:
    for relative, expected in REQUIRED_PREDECESSORS.items():
        path = root / relative
        if not path.is_file() or path.is_symlink():
            raise InstallError(f"Required accepted predecessor is missing: {relative}")
        if sha256_file(path) != expected:
            raise InstallError(f"Required accepted predecessor SHA-256 is not approved: {relative}")


def install(package_path: Path, root: Path) -> int:
    package_path = package_path.resolve()
    root = root.resolve()
    if root.name.upper() != "U-UDP003":
        raise InstallError(f"Expected U-UDP003 root, got: {root}")
    if not package_path.is_file():
        raise InstallError(f"Transfer ZIP is missing: {package_path}")
    validate_predecessors(root)
    temp_archive, temp_count = archive_previous_temp(root)
    prepared = root / "04_TEMP" / "U3M5_PREPARED"
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package_path, "r") as archive:
        members, ignored_documents = safe_members(archive)
        archive.extractall(prepared)
    checked = validate_payloads(prepared)
    relative_paths = [relative for relative, _digest, _purpose in checked]
    replaced_archive, replaced_count = archive_replaced_payloads(root, relative_paths)
    for relative in relative_paths:
        source = prepared / relative
        destination = root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        writing = destination.with_name(destination.name + ".writing")
        shutil.copy2(source, writing)
        writing.replace(destination)

    print("U3M5 prepare checks:")
    print(f"  [OK] U003 root: {root} - company directory topology is explicit")
    for relative, expected in REQUIRED_PREDECESSORS.items():
        print(f"  [OK] Predecessor: {relative} | {expected[:12]} - accepted input is frozen")
    print(f"  [OK] ZIP inventory: {len(members)} files - unknown payloads are rejected")
    if ignored_documents:
        print("  [OK] Auxiliary documentation ignored: " + ", ".join(ignored_documents))
    for relative, digest, purpose in checked:
        print(f"  [OK] {relative} | {digest[:12]} - {purpose}")
    if temp_archive is None:
        print("  [OK] TEMP hygiene: no previous-node item required archival")
    else:
        print(f"  [OK] TEMP hygiene: {temp_count} previous-node items archived to {temp_archive}")
    if replaced_archive is None:
        print("  [OK] Program preservation: no installed U3M5 payload required backup")
    else:
        print(f"  [OK] Program preservation: {replaced_count} payloads backed up to {replaced_archive}")
    print("  [OK] U3M5_PREPARED recreated from the verified ZIP")
    print(f"U3M5P N-R1-F7-I9-L2-A{temp_count}-B{replaced_count}-X0 C-0")
    return 0


def parse_arguments(argv: list[str]) -> argparse.Namespace:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--root", type=Path)
    parser.add_argument("--package", type=Path)
    return parser.parse_args(argv)


def main(argv: list[str] | None = None) -> int:
    arguments = parse_arguments(sys.argv[1:] if argv is None else argv)
    try:
        if not arguments.prepare:
            raise InstallError("Use --prepare.")
        script_directory = Path(__file__).resolve().parent
        root = arguments.root if arguments.root is not None else script_directory.parent
        package = arguments.package if arguments.package is not None else script_directory / PACKAGE_NAME
        return install(package, root)
    except Exception as error:
        print(f"U3M5PE K-{type(error).__name__.upper()} D-{error} X1 C-7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
