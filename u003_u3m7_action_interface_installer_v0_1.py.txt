#!/usr/bin/env python3
"""Prepare and install the U003 U3M7 Interactive Module Action update."""

import argparse
import csv
import hashlib
import shutil
import sys
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PACKAGE_NAME = "UDP003_U3M7_ACTION_INTERFACE_v0_1_1.zip"
PROGRAMS = {
    "01_PROGRAMS/u003_evidence_review_workbench_v0_19.py",
    "01_PROGRAMS/u003_gui_shell_v0_6.py",
    "01_PROGRAMS/u003_module_action_gui_v0_1.py",
    "01_PROGRAMS/u003_module_execution_coordinator_v0_3.py",
    "01_PROGRAMS/u003_module_executor_v0_3.py",
    "01_PROGRAMS/u003_module_manager_gui_v0_4.py",
    "01_PROGRAMS/u003_module_plan_core_v0_2.py",
    "01_PROGRAMS/u003_module_runtime_v0_2.py",
    "01_PROGRAMS/u003_u3m7_action_interface_acceptance_gate_v0_1.py",
}
CONTRACTS = {
    "02_CONTROLS/contracts/u003_module_action_contract_bundle_v0_1.json",
    "02_CONTROLS/contracts/u003_module_contract_v0_2.json",
}
MODULE_FILES = {
    "02_MODULES/EXPORT/CameraRadar.U003.SampleArtifactAction/0.1.0/algorithm_card.json",
    "02_MODULES/EXPORT/CameraRadar.U003.SampleArtifactAction/0.1.0/implementation.py",

DOCUMENTS = {

}
PAYLOADS = PROGRAMS | CONTRACTS | MODULE_FILES | DOCUMENTS
REQUIRED_MEMBERS = PAYLOADS | {"PACKAGE_SHA256.csv"}
OBSOLETE_PROGRAMS = {
    "u003_evidence_review_workbench_v0_17.py",
    "u003_gui_shell_v0_4.py",
    "u003_module_execution_coordinator_v0_1.py",
    "u003_module_executor_v0_1.py",
    "u003_module_manager_gui_v0_3.py",
    "u003_module_plan_core_v0_1.py",
    "u003_module_runtime_v0_1.py",
    "u003_u3m6_source_release_acceptance_gate_v0_1.py",
}


class InstallError(RuntimeError):
    """Raised when a package or company-side topology violates the update contract."""


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def next_archive_root(root: Path, label: str) -> Path:
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    destination = root / "A-Archive" / f"{label}_{stamp}"
    suffix = 1
    while destination.exists():
        destination = root / "A-Archive" / f"{label}_{stamp}_{suffix:02d}"
        suffix += 1
    return destination


def move_preserving_root(source: Path, root: Path, archive_root: Path) -> None:
    destination = archive_root / source.relative_to(root)
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(source), str(destination))


def validate_zip_inventory(archive: zipfile.ZipFile) -> list[str]:
    members: list[str] = []
    for name in archive.namelist():
        pure = PurePosixPath(name)
        if pure.is_absolute() or ".." in pure.parts:
            raise InstallError(f"Unsafe ZIP member: {name}")
        if not name.endswith("/"):
            members.append(name)
    if set(members) != REQUIRED_MEMBERS:
        missing = sorted(REQUIRED_MEMBERS - set(members))
        extra = sorted(set(members) - REQUIRED_MEMBERS)
        raise InstallError(f"Package inventory mismatch. Missing={missing}; Extra={extra}")
    return sorted(members)


def validate_extracted_payloads(package_root: Path) -> list[tuple[str, str, str]]:
    manifest_path = package_root / "PACKAGE_SHA256.csv"
    with manifest_path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    if len(rows) != len(PAYLOADS):
        raise InstallError(
            f"Manifest row count {len(rows)} does not equal {len(PAYLOADS)}."
        )
    verified: list[tuple[str, str, str]] = []
    for row in rows:
        relative = str(row.get("RelativePath", "")).strip()
        expected = str(row.get("Sha256", "")).strip().lower()
        purpose = str(row.get("Purpose", "")).strip()
        path = (package_root / relative).resolve()
        try:
            path.relative_to(package_root.resolve())
        except ValueError as error:
            raise InstallError(f"Manifest path escapes package root: {relative}") from error
        if not path.is_file() or path.is_symlink():
            raise InstallError(f"Payload is missing or invalid: {relative}")
        actual = sha256_file(path)
        if actual != expected:
            raise InstallError(f"Payload hash mismatch: {relative}")
        verified.append((relative, actual, purpose))
    if {item[0] for item in verified} != PAYLOADS:
        raise InstallError("Manifest payload inventory differs from the release contract.")
    return verified


def archive_previous_temp(
    root: Path,
    installer_path: Path,
    package_path: Path,
) -> tuple[Path | None, int]:
    temp_root = root / "04_TEMP"
    temp_root.mkdir(parents=True, exist_ok=True)
    keep = {installer_path.resolve(), package_path.resolve()}
    selected = []
    for path in temp_root.iterdir():
        if path.resolve() in keep or path.name.startswith("U3M7_START_HERE"):
            continue
        if (
            path.name.startswith("U3M7")
            or path.name.startswith("UDP003_U3M7")
            or path.name.startswith("u003_u3m7")
        ):
            selected.append(path)
    if not selected:
        return None, 0
    archive_root = next_archive_root(root, "U3M7_TEMP_CLEANUP")
    for path in selected:
        move_preserving_root(path, root, archive_root)
    return archive_root, len(selected)


def archive_obsolete_programs(root: Path) -> tuple[Path | None, int]:
    program_root = root / "01_PROGRAMS"
    program_root.mkdir(parents=True, exist_ok=True)
    selected = [program_root / name for name in sorted(OBSOLETE_PROGRAMS)]
    selected = [path for path in selected if path.is_file()]
    if not selected:
        return None, 0
    archive_root = next_archive_root(root, "U3M7_OBSOLETE_ACTIVE_PROGRAMS")
    for path in selected:
        move_preserving_root(path, root, archive_root)
    return archive_root, len(selected)


def backup_replaced_payloads(root: Path) -> tuple[Path | None, int]:
    existing = [root / relative for relative in PAYLOADS if (root / relative).is_file()]
    if not existing:
        return None, 0
    archive_root = next_archive_root(root, "U3M7_REPLACED_PAYLOADS")
    for source in existing:
        destination = archive_root / source.relative_to(root)
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, destination)
    return archive_root, len(existing)


def require_existing_foundation(root: Path) -> None:
    required = (
        root / "01_PROGRAMS" / "u003_domain_core_v0_2.py",
        root / "01_PROGRAMS" / "u003_handoff_core_v0_2.py",
        root / "01_PROGRAMS" / "u003_localization_v0_2.py",
        root / "02_MODULES" / "REVIEW_ASSIST" / "CameraRadar.U003.SampleMemoDraft" / "0.1.0" / "module_manifest.json",
    )
    missing = [str(path) for path in required if not path.is_file()]
    if missing:
        raise InstallError(f"Accepted U3M6 foundation is incomplete: {missing}")


def install(package_path: Path, root: Path, installer_path: Path) -> int:
    package_path = package_path.resolve()
    root = root.resolve()
    installer_path = installer_path.resolve()
    if root.name.upper() != "U-UDP003":
        raise InstallError(f"Expected U-UDP003 root, got: {root}")
    if not package_path.is_file():
        raise InstallError(f"Transfer ZIP is missing: {package_path}")
    require_existing_foundation(root)

    temp_archive, temp_count = archive_previous_temp(root, installer_path, package_path)
    prepared = root / "04_TEMP" / "U3M7_PREPARED"
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package_path, "r") as archive:
        members = validate_zip_inventory(archive)
        archive.extractall(prepared)
    verified = validate_extracted_payloads(prepared)
    backup_root, backup_count = backup_replaced_payloads(root)
    obsolete_root, obsolete_count = archive_obsolete_programs(root)

    for relative, _digest, _purpose in verified:
        source = prepared / relative
        destination = root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        writing = destination.with_name(destination.name + ".writing")
        shutil.copy2(source, writing)
        writing.replace(destination)

    print("U3M7 prepare and installation checks:")
    print(f"  [OK] U003 root: {root} - company topology is explicit")
    print("  [OK] Accepted U3M6 foundation - stable domain, handoff and localization cores exist")
    print(f"  [OK] Exact ZIP inventory: {len(members)} files - no hidden payload accepted")
    print(f"  [OK] Active update programs: {len(PROGRAMS)} - high-code composition is pinned")
    print(f"  [OK] New contracts: {len(CONTRACTS)} - UserAction boundary is explicit")
    print(f"  [OK] Reference action files: {len(MODULE_FILES)} - real action execution is testable")
    for relative, digest, purpose in verified:
        print(f"  [OK] {relative} | {digest[:12]} - {purpose}")
    print(f"  [OK] Previous U3M7 TEMP items archived: {temp_count} | {temp_archive or 'none'}")
    print(f"  [OK] Replaced U3M7 payloads backed up: {backup_count} | {backup_root or 'none'}")
    print(f"  [OK] Superseded active programs archived: {obsolete_count} | {obsolete_root or 'none'}")
    print("  [OK] Videos, review data, outputs, profiles, and unknown files were not modified")
    print(
        f"U3M7P N-F{len(PAYLOADS)}-P{len(PROGRAMS)}-C{len(CONTRACTS)}-"
        f"M1-A{temp_count}-B{backup_count}-O{obsolete_count}-X0 C-0"
    )
    return 0


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--root", type=Path)
    parser.add_argument("--package", type=Path)
    arguments = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if not arguments.prepare:
            raise InstallError("Use --prepare.")
        script_path = Path(__file__).resolve()
        root = arguments.root if arguments.root is not None else script_path.parent.parent
        package = arguments.package if arguments.package is not None else script_path.parent / PACKAGE_NAME
        return install(package, root, script_path)
    except Exception as error:
        print(f"U3M7PE K-{type(error).__name__.upper()} D-{error} X1 C-7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
