#!/usr/bin/env python3
"""Prepare, test and install the U001 H2 MATLAB CAN CSV + DBC adapter."""

from __future__ import annotations

import argparse
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PROGRAM_VERSION = "1.0.0"
PACKAGE_NAME = "UDP001_U1H2_MATLAB_CAN_DBC_ADAPTER_v1_0.zip"
START_HERE_NAME = "U1H2_START_HERE_CN_v1_0_2026-08-28.txt"
ACTIVE_NAME = "U001H_ACTIVE"
PREPARED_NAME = "U1H2_PREPARED"
ADAPTER_ID = "CameraRadar.U001.MatlabCanCsvDbc"
ADAPTER_VERSION = "1.0.0"

EXPECTED_FILES = {
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.0.0/adapter.py",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.0.0/adapter_card.json",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.0.0/u001_matlab_can_csv_dbc_core_v1_0.py",
    "02_CONTROL/templates/u001_h2_adapter_request_template_v1_0.json",
    "05_TESTS/tools/u001_h2_matlab_can_dbc_acceptance_v1_0.py",
    "06_DOCS/U001_H2_MATLAB_CAN_CSV_DBC_INPUT_ADAPTER_TECHNICAL_SPEC_v1_0_2026-08-28.txt",
}

H1_REQUIRED_FILES = {
    "01_PROGRAMS/core/u001_input_adapter_registry_v0_1.py",
    "01_PROGRAMS/core/u001_input_adapter_conformance_v0_1.py",
    "01_PROGRAMS/adapters/Foundation.Test.SyntheticInputAdapter/0.1.0/adapter.py",
    "01_PROGRAMS/adapters/Foundation.Test.SyntheticInputAdapter/0.1.0/adapter_card.json",
    "02_CONTROL/contracts/u001_input_adapter_card_schema_v0_1.json",
    "02_CONTROL/contracts/u001_canonical_source_update_schema_v0_1.json",
}

OBSOLETE_TRANSFER_NAMES = {
    "UDP001_U1H1_INPUT_ADAPTER_FOUNDATION_v0_1.zip",
    "u001_u1h1_input_adapter_foundation_installer_v0_1.py.txt",
    "U1H1_START_HERE_CN_v0_1_2026-08-28.txt",
    "U1H1_EXECUTION_RESULT.json",
    "U1H2_EXECUTION_RESULT.json",
}


class InstallError(RuntimeError):
    """Fail-closed H2 installation error."""


def timestamp() -> str:
    return datetime.now().strftime("%Y%m%dT%H%M%S_%f")


def resolve_u001_root(explicit: Path | None) -> Path:
    candidates = []
    if explicit is not None:
        candidates.append(explicit)
    candidates.extend((Path.cwd() / "U-UDP001", Path.cwd()))
    for candidate in candidates:
        candidate = candidate.resolve()
        if candidate.name == "U-UDP001" and candidate.is_dir():
            return candidate
    raise InstallError("U001_ROOT_NOT_FOUND")


def transfer_dir(u001_root: Path) -> Path:
    path = u001_root / "04_TEMP"
    if not path.is_dir():
        raise InstallError("TEMP_DIRECTORY_MISSING")
    return path


def package_path(u001_root: Path) -> Path:
    path = transfer_dir(u001_root) / PACKAGE_NAME
    if not path.is_file():
        raise InstallError(f"PACKAGE_MISSING:{PACKAGE_NAME}")
    return path


def validate_h1_foundation(u001_root: Path) -> None:
    missing = sorted(relative for relative in H1_REQUIRED_FILES if not (u001_root / relative).is_file())
    if missing:
        raise InstallError(f"H1_FOUNDATION_MISSING:{missing}")


def validate_zip(path: Path) -> None:
    try:
        with zipfile.ZipFile(path, "r") as archive:
            file_names = {name for name in archive.namelist() if not name.endswith("/")}
            if file_names != EXPECTED_FILES:
                missing = sorted(EXPECTED_FILES - file_names)
                extra = sorted(file_names - EXPECTED_FILES)
                raise InstallError(f"PACKAGE_INVENTORY:MISSING={missing}:EXTRA={extra}")
            for name in file_names:
                pure = PurePosixPath(name)
                if pure.is_absolute() or ".." in pure.parts:
                    raise InstallError(f"UNSAFE_PACKAGE_PATH:{name}")
            bad_member = archive.testzip()
            if bad_member is not None:
                raise InstallError(f"PACKAGE_CRC:{bad_member}")
    except zipfile.BadZipFile as exc:
        raise InstallError("PACKAGE_ZIP") from exc


def _archive(path: Path, archive_root: Path, relative: Path) -> bool:
    if not path.exists():
        return False
    destination = archive_root / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(destination))
    return True


def clean_superseded_transfer(u001_root: Path) -> int:
    temp = transfer_dir(u001_root)
    archive_root = u001_root / "A-Archive" / f"U1H2_TEMP_HYGIENE_{timestamp()}"
    moved = 0
    for name in sorted(OBSOLETE_TRANSFER_NAMES):
        moved += int(_archive(temp / name, archive_root, Path(name)))
    active = temp / ACTIVE_NAME
    for name in ("U1H1_PREPARED", PREPARED_NAME):
        moved += int(_archive(active / name, archive_root, Path(ACTIVE_NAME) / name))
    return moved


def prepare(u001_root: Path) -> Path:
    source_zip = package_path(u001_root)
    print(f"[CHECK] U001 root exists: {u001_root}")
    print("[WHY]   H2 may be installed only inside the accepted U-UDP001 layout.")
    validate_h1_foundation(u001_root)
    print(f"[CHECK] H1 foundation is complete: {len(H1_REQUIRED_FILES)} required files")
    print("[WHY]   H2 depends on the accepted Adapter Registry, schemas and Conformance Gate.")
    print(f"[CHECK] H2 transfer package exists: {source_zip.name}")
    validate_zip(source_zip)
    print(f"[CHECK] Package inventory and CRC passed: {len(EXPECTED_FILES)} files")
    print("[WHY]   Exact inventory blocks stale Gate debris from entering the High-Code product.")

    archived = clean_superseded_transfer(u001_root)
    print(f"[CLEAN] Superseded H1/H2 transfer entries archived without deletion: {archived}")
    print("[WHY]   Prepared code must resolve only from the current H2 package and installed H1 foundation.")

    active = transfer_dir(u001_root) / ACTIVE_NAME
    active.mkdir(parents=True, exist_ok=True)
    prepared = active / PREPARED_NAME
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(source_zip, "r") as archive:
        archive.extractall(prepared)
    extracted = {
        path.relative_to(prepared).as_posix()
        for path in prepared.rglob("*")
        if path.is_file()
    }
    if extracted != EXPECTED_FILES:
        raise InstallError("EXTRACTED_INVENTORY")
    print("[CHECK] Prepared H2 payload exactly matches the approved package.")
    print(f"U1H2P-N{len(EXPECTED_FILES)}-H1-A{archived}-X0-C0")
    return prepared


def child_environment() -> dict[str, str]:
    environment = dict(os.environ)
    environment["PYTHONDONTWRITEBYTECODE"] = "1"
    environment["PYTHONPYCACHEPREFIX"] = str(Path(tempfile.gettempdir()) / "u001_h2_pycache")
    return environment


def run_child(command: list[str], label: str, timeout_seconds: int = 180) -> str:
    try:
        result = subprocess.run(
            command,
            text=True,
            capture_output=True,
            timeout=timeout_seconds,
            check=False,
            env=child_environment(),
        )
    except subprocess.TimeoutExpired as exc:
        raise InstallError(f"{label}_TIMEOUT") from exc
    if result.returncode != 0:
        detail = (result.stderr or result.stdout).strip().replace("\n", " | ")[:1000]
        raise InstallError(f"{label}_EXIT:{result.returncode}:{detail}")
    output = result.stdout.strip()
    if not output:
        raise InstallError(f"{label}_NO_OUTPUT")
    return output


def self_test(u001_root: Path) -> None:
    validate_h1_foundation(u001_root)
    prepared = transfer_dir(u001_root) / ACTIVE_NAME / PREPARED_NAME
    if not prepared.is_dir():
        raise InstallError("PREPARED_DIRECTORY_MISSING")
    adapter_dir = prepared / "01_PROGRAMS" / "adapters" / ADAPTER_ID / ADAPTER_VERSION
    core = adapter_dir / "u001_matlab_can_csv_dbc_core_v1_0.py"
    adapter = adapter_dir / "adapter.py"
    h2_gate = prepared / "05_TESTS" / "tools" / "u001_h2_matlab_can_dbc_acceptance_v1_0.py"
    h1_registry = u001_root / "01_PROGRAMS" / "core" / "u001_input_adapter_registry_v0_1.py"
    h1_conformance = u001_root / "01_PROGRAMS" / "core" / "u001_input_adapter_conformance_v0_1.py"
    if not all(path.is_file() for path in (core, adapter, h2_gate, h1_registry, h1_conformance)):
        raise InstallError("SELFTEST_PROGRAM_MISSING")

    checks = [
        (core, "PASS-24", "H2 core primitives: header, DBC, endian, signedness and source grouping."),
        (adapter, "PASS-30", "H2 production adapter: preflight, streaming run, quality and traceability."),
        (h2_gate, "PASS-30", "Independent black-box replay: production request, determinism and row limit."),
        (h1_registry, "PASS-18", "H1 registry: Adapter Card identity, versioning and fail-closed discovery."),
        (h1_conformance, "PASS-20", "H1 generic conformance: five artifacts, hashes and path safety."),
    ]
    total = 0
    for program, capsule, explanation in checks:
        print(f"[CHECK] {explanation}")
        output = run_child([sys.executable, str(program), "--self-test"], program.stem.upper())
        if capsule not in output:
            raise InstallError(f"SELFTEST_CAPSULE:{program.name}")
        total += int(capsule.rsplit("-", 1)[1])
    print(f"U1H2ST-N{len(checks)}-T{total}-X0-C0")


def copy_payload(prepared: Path, u001_root: Path) -> tuple[int, Path | None]:
    archive_root: Path | None = None
    copied = 0
    for relative_text in sorted(EXPECTED_FILES):
        relative = Path(relative_text)
        source = prepared / relative
        target = u001_root / relative
        if target.exists():
            if archive_root is None:
                archive_root = u001_root / "A-Archive" / f"U1H2_REPLACED_{timestamp()}"
            _archive(target, archive_root, relative)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        copied += 1
    return copied, archive_root


def execute(u001_root: Path) -> None:
    self_test(u001_root)
    prepared = transfer_dir(u001_root) / ACTIVE_NAME / PREPARED_NAME
    copied, replaced_archive = copy_payload(prepared, u001_root)

    adapters_root = u001_root / "01_PROGRAMS" / "adapters"
    adapter_dir = adapters_root / ADAPTER_ID / ADAPTER_VERSION
    registry = u001_root / "01_PROGRAMS" / "core" / "u001_input_adapter_registry_v0_1.py"
    conformance = u001_root / "01_PROGRAMS" / "core" / "u001_input_adapter_conformance_v0_1.py"
    h2_gate = u001_root / "05_TESTS" / "tools" / "u001_h2_matlab_can_dbc_acceptance_v1_0.py"
    registry_output = u001_root / "02_CONTROL" / "registry" / "u001_input_adapter_registry_snapshot_v0_1.json"
    registry_output.parent.mkdir(parents=True, exist_ok=True)

    scan = run_child([
        sys.executable,
        str(registry),
        "--scan",
        "--adapter-root",
        str(adapters_root),
        "--output",
        str(registry_output),
    ], "INSTALLED_REGISTRY_SCAN")
    if "C0" not in scan.replace("-", ""):
        raise InstallError("INSTALLED_REGISTRY_CAPSULE")
    snapshot = json.loads(registry_output.read_text(encoding="utf-8"))
    if snapshot.get("AdapterCount") != 2 or snapshot.get("AdapterFamilyCount") != 2:
        raise InstallError("INSTALLED_REGISTRY_COUNT")

    acceptance_root = u001_root / "05_TESTS" / f"U1H2_ACCEPTANCE_{timestamp()}"
    generic_output = acceptance_root / "generic_conformance"
    generic = run_child([
        sys.executable,
        str(conformance),
        "--qualify",
        "--adapter-dir",
        str(adapter_dir),
        "--output",
        str(generic_output),
    ], "H1_GENERIC_CONFORMANCE")
    if "C0" not in generic.replace("-", ""):
        raise InstallError("GENERIC_CONFORMANCE_CAPSULE")

    independent_output = acceptance_root / "h2_independent"
    independent = run_child([
        sys.executable,
        str(h2_gate),
        "--validate",
        "--adapter-dir",
        str(adapter_dir),
        "--output",
        str(independent_output),
    ], "H2_INDEPENDENT_ACCEPTANCE")
    if "C0" not in independent.replace("-", ""):
        raise InstallError("INDEPENDENT_ACCEPTANCE_CAPSULE")

    result = {
        "Node": "U1H2",
        "AdapterId": ADAPTER_ID,
        "AdapterVersion": ADAPTER_VERSION,
        "InstalledFileCount": copied,
        "AdapterCount": snapshot["AdapterCount"],
        "AdapterFamilyCount": snapshot["AdapterFamilyCount"],
        "GenericConformancePassed": True,
        "IndependentH2AcceptancePassed": True,
        "AcceptanceFolder": acceptance_root.name,
        "ReplacedFilesArchived": replaced_archive is not None,
        "CompletionCode": 0,
    }
    (transfer_dir(u001_root) / "U1H2_EXECUTION_RESULT.json").write_text(
        json.dumps(result, indent=2) + "\n", encoding="utf-8"
    )
    print(
        f"U1H2-N{snapshot['AdapterCount']}-F{snapshot['AdapterFamilyCount']}-"
        f"Q2-A{copied}-X0-C0"
    )


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description="Install U001 H2 MATLAB CAN CSV + DBC Adapter")
    action = result.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    result.add_argument("--u001-root", type=Path)
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    try:
        root = resolve_u001_root(args.u001_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            self_test(root)
        else:
            execute(root)
        return 0
    except (InstallError, OSError, ValueError, json.JSONDecodeError, zipfile.BadZipFile) as exc:
        print(f"U1H2E-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
