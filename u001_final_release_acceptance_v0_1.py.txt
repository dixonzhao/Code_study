class AcceptanceFailure(RuntimeError):
    pass


def fail(code: str) -> None:
    raise AcceptanceFailure(code)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def record_name(record: dict) -> str:
    for key in ("Filename", "FileName", "Name", "RelativePath", "Path"):
        value = record.get(key)
        if isinstance(value, str) and value:
            return Path(value.replace("\\", "/")).name
    return ""


def record_hash(record: dict) -> str:
    for key in ("SHA256", "Sha256", "sha256"):
        value = record.get(key)
        if isinstance(value, str):
            return value.lower()
    return ""


def manifest_records(manifest: dict, key: str, expected: tuple[str, ...]) -> dict[str, dict]:
    values = manifest.get(key)
    if not isinstance(values, list):
        fail("MANIFEST_" + key.upper())
    result = {}
    for record in values:
        if not isinstance(record, dict):
            fail("MANIFEST_RECORD")
        name = record_name(record)
        if not name or name in result:
            fail("MANIFEST_RECORD_IDENTITY")
        result[name] = record
    if set(result) != set(expected):
        fail("MANIFEST_" + key.upper() + "_INVENTORY")
    return result


def verify_hashes(records: dict[str, dict], folder: Path) -> None:
    for name, record in records.items():
        expected = record_hash(record)
        path = folder / name
        if len(expected) != 64 or not path.is_file() or sha256_file(path).lower() != expected:
            fail("HASH_" + name)


def hosted_count(manifest: dict) -> int:
    values = manifest.get("HostedProgramSelftests")
    if not isinstance(values, list):
        fail("HOSTED_MANIFEST")
    names = set()
    for item in values:
        if not isinstance(item, dict):
            fail("HOSTED_RECORD")
        name = record_name(item)
        code = item.get("ReturnCode", item.get("returncode", item.get("ExitCode")))
        passed = item.get("PassTokenObserved", item.get("Passed", item.get("Pass")))
        if name not in PROGRAMS or code not in (0, "0") or passed is not True:
            fail("HOSTED_RESULT")
        names.add(name)
    if names != set(PROGRAMS):
        fail("HOSTED_INVENTORY")
    return len(names)


def verify_zip(zip_path: Path) -> None:
    if not zip_path.is_file() or zip_path.stat().st_size == 0:
        fail("ZIP_MISSING")
    program_entries = {f"{PACKAGE_NAME}/UDP001/01_PROGRAMS/{name}" for name in PROGRAMS}
    control_entries = {f"{PACKAGE_NAME}/UDP001/02_CONTROL/{name}" for name in CONTROLS}
    cases_prefix = f"{PACKAGE_NAME}/UDP001/03_CASES/"
    with zipfile.ZipFile(zip_path, "r") as archive:
        names = set()
        cases_present = False
        for info in archive.infolist():
            raw = info.filename.replace("\\", "/")
            pure = PurePosixPath(raw)
            if pure.is_absolute() or ".." in pure.parts:
                fail("ZIP_UNSAFE")
            if raw and pure.parts and pure.parts[0] != PACKAGE_NAME:
                fail("ZIP_ROOT")
            lowered = {part.lower() for part in pure.parts}
            if "04_temp" in lowered or "arch" in lowered or "__pycache__" in lowered:
                fail("ZIP_FORBIDDEN")
            names.add(raw)
            if raw.rstrip("/") == cases_prefix.rstrip("/"):
                cases_present = True
        if not program_entries.issubset(names):
            fail("ZIP_PROGRAMS")
        if not control_entries.issubset(names):
            fail("ZIP_CONTROLS")
        if not cases_present:
            fail("ZIP_CASES_DIRECTORY")


def run_runtime(executable: Path, portable: Path, log_root: Path) -> None:
    result = subprocess.run(
        [str(executable.resolve()), "--hosted-self-test-all", "--portable-root", str(portable.resolve())],
        cwd=str(executable.parent.resolve()),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        timeout=2100,
        shell=False,
    )
    (log_root / "UDP001_FINAL_RUNTIME_STDOUT.txt").write_bytes(result.stdout)
    (log_root / "UDP001_FINAL_RUNTIME_STDERR.txt").write_bytes(result.stderr)
    if result.returncode != 0:
        fail("RUNTIME_HOSTED_SELFTEST")


def accept(release_root: Path) -> None:
    root = release_root.resolve()
    package = root / PACKAGE_NAME
    zip_path = root / f"{PACKAGE_NAME}.zip"
    portable = package / "UDP001"
    programs = portable / "01_PROGRAMS"
    controls = portable / "02_CONTROL"
    cases = portable / "03_CASES"
    launcher = portable / "00_LAUNCHER"
    evidence = package / "_internal" / "evidence"
    executable = launcher / "U001_Workbench.exe"
    manifest_path = evidence / "u001_portable_host_package_manifest.json"
    capsule_path = evidence / "u001_portable_host_package_capsule.txt"
    if not root.is_dir() or not package.is_dir() or not portable.is_dir():
        fail("LAYOUT_ROOT")
    if not cases.is_dir() or any(cases.iterdir()):
        fail("CASES_HYGIENE_PRE")
    if any(path.name.endswith("INCOMPLETE") for path in root.rglob("*")):
        fail("INCOMPLETE_MARKER")
    if tuple(sorted(path.name for path in programs.glob("*.py"))) != tuple(sorted(PROGRAMS)):
        fail("PROGRAM_INVENTORY")
    if tuple(sorted(path.name for path in controls.glob("*.csv"))) != tuple(sorted(CONTROLS)):
        fail("CONTROL_INVENTORY")
    if any(path.name == "__pycache__" for path in package.rglob("__pycache__")):
        fail("PYCACHE")
    if not executable.is_file() or not manifest_path.is_file() or not capsule_path.is_file():
        fail("EVIDENCE_OR_EXE")
    try:
        manifest = json.loads(manifest_path.read_text(encoding="utf-8-sig"))
    except (OSError, json.JSONDecodeError):
        fail("MANIFEST_PARSE")
    if manifest.get("Schema") != MANIFEST_SCHEMA or manifest.get("PackageName") != PACKAGE_NAME:
        fail("MANIFEST_IDENTITY")
    if manifest.get("CompanyDataIncluded") is not False:
        fail("COMPANY_DATA")
    serialized = json.dumps(manifest, ensure_ascii=False, sort_keys=True)
    if BUILDER_NAME not in serialized or HOST_NAME not in serialized:
        fail("MANIFEST_TOOL_IDENTITY")
    if "/Users/" in serialized or ":\\Users\\" in serialized or ":/Users/" in serialized:
        fail("ABSOLUTE_USER_PATH")
    program_records = manifest_records(manifest, "Programs", PROGRAMS)
    control_records = manifest_records(manifest, "Controls", CONTROLS)
    verify_hashes(program_records, programs)
    verify_hashes(control_records, controls)
    if hosted_count(manifest) != 12:
        fail("HOSTED_COUNT")
    capsule = tuple(line.strip() for line in capsule_path.read_text(encoding="utf-8-sig").splitlines() if line.strip())
    if capsule != EXPECTED_CAPSULE:
        fail("CAPSULE")
    verify_zip(zip_path)
    run_runtime(executable, portable, root.parent)
    if not cases.is_dir() or any(cases.iterdir()):
        fail("CASES_HYGIENE_POST")


def run_self_test() -> None:
    checks = (
        PROGRAM_VERSION == "0.1",
        PROGRAM_NAME.endswith("v0_1.py"),
        SELFTEST_TOKEN.endswith("PASS-24"),
        PACKAGE_NAME.endswith("v0_2"),
        MANIFEST_SCHEMA.endswith("V0_2"),
        BUILDER_NAME.endswith("v0_8.py"),
        HOST_NAME.endswith("v0_6.py"),
        len(EXPECTED_CAPSULE) == 3,
        EXPECTED_CAPSULE[0] == "U001PK05",
        EXPECTED_CAPSULE[-1] == "C-0",
        len(PROGRAMS) == 12,
        len(set(PROGRAMS)) == 12,
        PROGRAMS[-1].endswith("gui_v0_4a.py"),
        len(CONTROLS) == 4,
        len(set(CONTROLS)) == 4,
        all(name.endswith(".csv") for name in CONTROLS),
        record_name({"Filename": "a.py"}) == "a.py",
        record_name({"Path": "x/y/b.py"}) == "b.py",
        record_hash({"SHA256": "A" * 64}) == "a" * 64,
        len(hashlib.sha256(b"x").hexdigest()) == 64,
        PurePosixPath(f"{PACKAGE_NAME}/UDP001").parts[0] == PACKAGE_NAME,
        callable(accept),
        issubclass(AcceptanceFailure, RuntimeError),
        sys.executable != "",
    )
    if sum(bool(item) for item in checks) != 24:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Final U001 portable release acceptance")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--release-root")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.release_root:
        parser.error("--release-root is required")
    try:
        accept(Path(args.release_root))
    except AcceptanceFailure as exc:
        print(f"UDP001-FINAL-FAIL-{exc.args[0]} C-7")
        return 7
    except subprocess.TimeoutExpired:
        print("UDP001-FINAL-FAIL-TIMEOUT C-7")
        return 7
    except Exception as exc:
        print(f"UDP001-FINAL-FAIL-UNHANDLED-{type(exc).__name__} C-7")
        return 7
    print("UDP001-FINAL-ACCEPT P12-C4-H12-Z1-D1-N1-X0 C-0")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
