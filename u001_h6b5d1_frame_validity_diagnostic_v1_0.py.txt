#!/usr/bin/env python3
"""Read-only H6B5 frame-assignment and invalidity diagnostic."""

from __future__ import annotations

import argparse
import csv
import json
import shutil
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, List, Optional, Sequence


PROGRAM_ID = "U001_H6B5D1_FRAME_VALIDITY_DIAGNOSTIC"
PROGRAM_VERSION = "1.0"
PREPARED_FOLDER = "U1H6B5D1_PREPARED"
OUTPUT_FOLDER = "U1H6B5D1_OUTPUT"

SUPERSEDED_TRANSFER_NAMES = (
    "u001_h6b5_h3_semantic_frame_compile_v1_0.py.txt",
    "U1H6B5_START_HERE_CN_v1_0_2026-08-28.txt",
)


class DiagnosticError(RuntimeError):
    """Fail-closed diagnostic error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def write_json(path: Path, value: Dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        raise DiagnosticError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise DiagnosticError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise DiagnosticError(label + "_ROOT_NOT_OBJECT")
    return value


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise DiagnosticError("U_UDP001_ROOT_NOT_FOUND")
    return root


def required_paths(root: Path) -> Dict[str, Path]:
    b5 = root / "04_TEMP" / "U1H6B5_OUTPUT"
    h3 = b5 / "h3_output"
    return {
        "B5Capsule": b5 / "u001_h6b5_capsule.txt",
        "B5Result": b5 / "u001_h6b5_result.json",
        "FrameMap": h3 / "u001_h3_frame_map.csv",
        "FrameQuality": h3 / "u001_h3_frame_quality.json",
        "CanonicalQuality": h3 / "u001_h3_canonical_quality.json",
        "H3Manifest": h3 / "u001_h3_run_manifest.json",
    }


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        temp = root / "04_TEMP"
        prepared = temp / PREPARED_FOLDER
        output = temp / OUTPUT_FOLDER
        archive = root / "A-Archive" / ("U1H6B5D1_HYGIENE_" + datetime.now().strftime("%Y%m%dT%H%M%S_%f"))
        print("[PREPARE] U001 H6B5D1 read-only frame and validity diagnostic")
        print("[PREPARE] Why: explain the high invalid/unassigned counts before H4 is allowed to run.")
        if prepared.exists():
            shutil.rmtree(prepared)
            print(f"[CLEAN] 04_TEMP/{PREPARED_FOLDER} | REMOVED | Why: invalidate stale diagnostic receipt.")
        if output.exists():
            archive.mkdir(parents=True, exist_ok=True)
            shutil.move(str(output), str(archive / output.name))
            print(f"[CLEAN] 04_TEMP/{OUTPUT_FOLDER} | ARCHIVED | Why: prevent stale diagnostic reuse.")
        for name in SUPERSEDED_TRANSFER_NAMES:
            path = temp / name
            if path.exists():
                archive.mkdir(parents=True, exist_ok=True)
                shutil.move(str(path), str(archive / path.name))
                print(f"[CLEAN] 04_TEMP/{name} | ARCHIVED | Why: superseded transfer file.")
        paths = required_paths(root)
        reasons = {
            "B5Capsule": "Proves H6B5 execution completed.",
            "B5Result": "Carries aggregate H3 counts requiring diagnosis.",
            "FrameMap": "Shows exact, assigned, and unassigned Source Rows.",
            "FrameQuality": "Carries FrameCount coverage and assignment-age statistics.",
            "CanonicalQuality": "Separates frame-unassigned invalidity from intrinsic invalidity.",
            "H3Manifest": "Links counts to the accepted H3 artifact boundary.",
        }
        missing: List[str] = []
        for label, path in paths.items():
            found = path.is_file()
            print(f"[CHECK] {label} | {'FOUND' if found else 'MISSING'} | Why: {reasons[label]}")
            if not found:
                missing.append(label)
        print("[KEEP] U1H6B3_OUTPUT / U1H6B4_OUTPUT / U1H6B5_OUTPUT | Why: active traceability chain.")
        if missing:
            raise DiagnosticError("PREPARE_MISSING:" + ".".join(missing))
        if not paths["B5Capsule"].read_text(encoding="utf-8").strip().endswith("C0"):
            raise DiagnosticError("B5_CAPSULE_NOT_C0")
        prepared.mkdir(parents=True)
        receipt = {
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "PreparedUtc": utc_now(),
            "Status": "READY",
            "CompletionCode": 0,
        }
        write_json(prepared / "u001_h6b5d1_prepare_receipt.json", receipt)
        print("U1H6B5D1P-C6-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1H6B5D1P-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def require_prepare(root: Path) -> None:
    receipt = read_json(root / "04_TEMP" / PREPARED_FOLDER / "u001_h6b5d1_prepare_receipt.json", "PREPARE_RECEIPT")
    if receipt.get("ProgramId") != PROGRAM_ID or receipt.get("ProgramVersion") != PROGRAM_VERSION:
        raise DiagnosticError("PREPARE_RECEIPT_IDENTITY")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise DiagnosticError("PREPARE_RECEIPT_STATUS")


def percentile(values: List[int], fraction: float) -> int:
    if not values:
        return 0
    ordered = sorted(values)
    return ordered[int(round((len(ordered) - 1) * fraction))]


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        require_prepare(root)
        print("[SELFTEST] Quantile calculation | Why: summarize coverage gaps without loading payload tables.")
        values = [1, 2, 3, 4, 100]
        if percentile(values, 0.5) != 3 or percentile(values, 0.95) != 100:
            raise DiagnosticError("PERCENTILE")
        print("[PASS] Quantile calculation verified.")
        print("[SELFTEST] Required quality fields | Why: prevent a misleading partial diagnosis.")
        paths = required_paths(root)
        frame = read_json(paths["FrameQuality"], "FRAME_QUALITY")
        canonical = read_json(paths["CanonicalQuality"], "CANONICAL_QUALITY")
        for key in (
            "unique_source_row_count", "frame_observation_count", "distinct_frame_count",
            "exact_assignment_source_row_count", "assigned_within_limit_source_row_count",
            "unassigned_no_prior_source_row_count", "unassigned_limit_source_row_count",
        ):
            if key not in frame:
                raise DiagnosticError("FRAME_FIELD_" + key.upper())
        for key in (
            "invalid_update_count", "intrinsic_invalid_count", "frame_unassigned_update_count",
            "dtype_conversion_failure_count", "exact_invalid_count",
        ):
            if key not in canonical:
                raise DiagnosticError("CANONICAL_FIELD_" + key.upper())
        print("[PASS] Required frame and invalidity fields verified.")
        print("U1H6B5D1-ST-T2-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1H6B5D1-ST-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def execute(root_argument: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        require_prepare(root)
        paths = required_paths(root)
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            raise DiagnosticError("OUTPUT_ALREADY_EXISTS_RUN_PREPARE_AGAIN")
        output.mkdir(parents=True)
        frame_quality = read_json(paths["FrameQuality"], "FRAME_QUALITY")
        canonical = read_json(paths["CanonicalQuality"], "CANONICAL_QUALITY")

        exact_rows: List[int] = []
        exact_frames: List[int] = []
        all_rows: List[int] = []
        status_counts: Dict[str, int] = {}
        with paths["FrameMap"].open("r", encoding="utf-8-sig", newline="") as stream:
            reader = csv.DictReader(stream)
            required = {"SourceRowIndex", "FrameCount", "FrameAssignmentStatus"}
            if reader.fieldnames is None or not required.issubset(reader.fieldnames):
                raise DiagnosticError("FRAME_MAP_COLUMNS")
            for row in reader:
                source_row = int(row["SourceRowIndex"])
                all_rows.append(source_row)
                status = row["FrameAssignmentStatus"]
                status_counts[status] = status_counts.get(status, 0) + 1
                if status == "FRAME_EXACT":
                    exact_rows.append(source_row)
                    exact_frames.append(int(row["FrameCount"]))
        if not all_rows or not exact_rows:
            raise DiagnosticError("FRAME_MAP_EMPTY")
        source_gaps = [later - earlier for earlier, later in zip(exact_rows, exact_rows[1:])]
        frame_deltas = [later - earlier for earlier, later in zip(exact_frames, exact_frames[1:])]
        positive_frame_deltas = [value for value in frame_deltas if value > 0]
        gap_over_200 = sum(value > 200 for value in source_gaps)
        last_coverage_percent = round(100.0 * exact_rows[-1] / max(all_rows[-1], 1), 2)

        diagnostic = {
            "DiagnosticSchema": "U001_H6B5D1_FRAME_VALIDITY_DIAGNOSTIC_V1_0",
            "GeneratedUtc": utc_now(),
            "Frame": {
                "UniqueSourceRows": int(frame_quality["unique_source_row_count"]),
                "FrameObservations": int(frame_quality["frame_observation_count"]),
                "DistinctFrames": int(frame_quality["distinct_frame_count"]),
                "RepeatedEqualObservations": int(frame_quality["repeated_equal_frame_observation_count"]),
                "FrameMinimum": frame_quality.get("frame_min"),
                "FrameMaximum": frame_quality.get("frame_max"),
                "ExactSourceRows": int(frame_quality["exact_assignment_source_row_count"]),
                "AssignedSourceRows": int(frame_quality["assigned_within_limit_source_row_count"]),
                "UnassignedNoPrior": int(frame_quality["unassigned_no_prior_source_row_count"]),
                "UnassignedLimit": int(frame_quality["unassigned_limit_source_row_count"]),
                "AssignmentAgeQ50": frame_quality.get("assignment_age_q50"),
                "AssignmentAgeQ95": frame_quality.get("assignment_age_q95"),
                "AssignmentAgeMax": frame_quality.get("assignment_age_max"),
                "FirstExactSourceRow": exact_rows[0],
                "LastExactSourceRow": exact_rows[-1],
                "MaximumObservedSourceRow": all_rows[-1],
                "LastExactCoveragePercent": last_coverage_percent,
                "SourceGapQ50": percentile(source_gaps, 0.50),
                "SourceGapQ95": percentile(source_gaps, 0.95),
                "SourceGapMaximum": max(source_gaps) if source_gaps else 0,
                "SourceGapOver200Count": gap_over_200,
                "PositiveFrameDeltaQ50": percentile(positive_frame_deltas, 0.50),
                "PositiveFrameDeltaQ95": percentile(positive_frame_deltas, 0.95),
                "PositiveFrameDeltaMaximum": max(positive_frame_deltas) if positive_frame_deltas else 0,
                "StatusCounts": status_counts,
            },
            "Invalidity": {
                "InvalidUpdates": int(canonical["invalid_update_count"]),
                "FrameUnassignedUpdates": int(canonical["frame_unassigned_update_count"]),
                "IntrinsicInvalidUpdates": int(canonical["intrinsic_invalid_count"]),
                "ExactInvalidUpdates": int(canonical["exact_invalid_count"]),
                "DtypeFailureUpdates": int(canonical["dtype_conversion_failure_count"]),
                "UnknownEnumUpdates": int(canonical["unknown_enum_code_count"]),
                "ProvisionalPolicyUpdates": int(canonical["provisional_policy_update_count"]),
            },
            "Status": "DIAGNOSTIC_COMPLETE",
            "CompletionCode": 0,
        }
        write_json(output / "u001_h6b5d1_diagnostic.json", diagnostic)
        frame = diagnostic["Frame"]
        invalid = diagnostic["Invalidity"]
        capsule1 = (
            f"H6B5D1-N{frame['UniqueSourceRows']}-O{frame['FrameObservations']}"
            f"-F{frame['DistinctFrames']}-E{frame['ExactSourceRows']}"
            f"-A{frame['AssignedSourceRows']}-U0{frame['UnassignedNoPrior']}"
            f"-UL{frame['UnassignedLimit']}-L{frame['LastExactSourceRow']}"
            f"-M{frame['MaximumObservedSourceRow']}-C{frame['LastExactCoveragePercent']}-X0-C0"
        )
        capsule2 = (
            f"H6B5D2-G{frame['SourceGapQ50']}.{frame['SourceGapQ95']}.{frame['SourceGapMaximum']}"
            f"-H{frame['SourceGapOver200Count']}"
            f"-Q{frame['AssignmentAgeQ50']}.{frame['AssignmentAgeQ95']}.{frame['AssignmentAgeMax']}"
            f"-I{invalid['InvalidUpdates']}-FU{invalid['FrameUnassignedUpdates']}"
            f"-IN{invalid['IntrinsicInvalidUpdates']}-EX{invalid['ExactInvalidUpdates']}"
            f"-DT{invalid['DtypeFailureUpdates']}-EN{invalid['UnknownEnumUpdates']}"
            f"-P{invalid['ProvisionalPolicyUpdates']}-X0-C0"
        )
        (output / "u001_h6b5d1_capsules.txt").write_text(capsule1 + "\n" + capsule2 + "\n", encoding="utf-8")
        print(capsule1)
        print(capsule2)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        write_json(output / "u001_h6b5d1_error.json", {"ExceptionType": type(exc).__name__, "Detail": str(exc)[:3000]})
        print(f"H6B5D-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H6B5D1 frame and validity diagnostic")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    sys.exit(main())
