#!/usr/bin/env python3
"""Read-only key-shape probe for retained synthetic U21E SignalBindings."""

from __future__ import annotations

import argparse
import json
from pathlib import Path


PREFLIGHT_REL = Path(
    "05_TESTS/02_ACCEPTANCE/U21G_G2_VERTICAL_V06.staging/control/run_preflight.json"
)
KEYS = (
    "ModuleId", "ModuleVersion", "PortId", "PortName", "SelectionMode",
    "CanonicalNames", "Required", "MinCount", "MaxCount", "Selectors",
    "D3SemanticProjection", "D3Signals", "D2Columns", "D2ColumnMapping",
    "BoundSignals", "SignalRecords", "Status", "Role", "ValueKind", "Unit",
)


def reject_pairs(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("DUPLICATE_KEY")
        result[key] = value
    return result


def type_code(value, present: bool) -> int:
    if not present:
        return 0
    if isinstance(value, list):
        return 1
    if isinstance(value, dict):
        return 2
    if isinstance(value, str):
        return 3
    return 4


def mappings(value, depth=0):
    if isinstance(value, dict):
        yield value, depth
        for item in value.values():
            yield from mappings(item, depth + 1)
    elif isinstance(value, list):
        for item in value:
            yield from mappings(item, depth + 1)


def all_keys(value):
    result = set()
    if isinstance(value, dict):
        for key, item in value.items():
            result.add(str(key))
            result.update(all_keys(item))
    elif isinstance(value, list):
        for item in value:
            result.update(all_keys(item))
    return result


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    path = Path(args.workspace_root).resolve() / PREFLIGHT_REL
    try:
        document = json.loads(path.read_text(encoding="utf-8-sig"),
                              object_pairs_hook=reject_pairs)
        raw = document["SignalBindings"]
        root_type = 1 if isinstance(raw, list) else (2 if isinstance(raw, dict) else 0)
        records = []
        for item, depth in mappings(raw):
            has_port = "PortId" in item or "PortName" in item
            has_binding = any(key in item for key in (
                "CanonicalNames", "SelectionMode", "D2Columns", "D2ColumnMapping",
                "D3SemanticProjection", "D3Signals", "BoundSignals", "SignalRecords",
            ))
            if has_port and has_binding:
                records.append((item, depth))
        print(f"G2R2SB2 N-RT{root_type}-R{len(records)}-X0 C-0")
        for index, (item, depth) in enumerate(records[:8], 1):
            mask = sum((1 << bit) for bit, key in enumerate(KEYS) if key in item)
            canonical = item.get("CanonicalNames")
            d2_key = "D2Columns" if "D2Columns" in item else "D2ColumnMapping"
            projection_key = "D3SemanticProjection" if "D3SemanticProjection" in item else "D3Signals"
            canonical_count = len(canonical) if isinstance(canonical, (list, dict)) else 0
            d2 = item.get(d2_key)
            projection = item.get(projection_key)
            print(
                f"B{index}-H{mask:05X}-C{type_code(canonical, 'CanonicalNames' in item)}"
                f"N{canonical_count}-D{type_code(d2, d2_key in item)}"
                f"J{type_code(projection, projection_key in item)}-L{depth}"
            )
            relevant = sorted(
                key for key in all_keys(item)
                if any(token in key.lower() for token in (
                    "signal", "column", "projection", "semantic", "canonical", "d2", "d3"
                ))
            )
            print(f"K{index}-" + (".".join(relevant)[:260] or "NONE"))
        return 0
    except Exception as exc:
        print(f"G2R2SB2 N-RT0-R0-X1 C-7 K-{exc.__class__.__name__.upper()}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
