#!/usr/bin/env python3
"""Read-only G2 D3 diagnostic for ExecutionOrder and QualityDecision shape."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import tempfile
from typing import Dict, List, Mapping, Optional, Sequence, Tuple


SELFTEST_CAPSULE = "G2D3ST N-T18-P6-J4-I4-X0 C-0"
MAX_JSON_BYTES = 4_194_304
MAX_FILES = 4096


class D3Error(Exception):
    def __init__(self, reason: str):
        super().__init__(reason)
        self.reason = reason


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def read_json(path: Path, reason: str) -> dict:
    if not path.is_file() or path.is_symlink() or path.stat().st_size > MAX_JSON_BYTES:
        raise D3Error(reason)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise D3Error(reason) from exc
    if not isinstance(value, dict):
        raise D3Error(reason)
    return value


def inventory(root: Path) -> Dict[str, Tuple[int, str]]:
    result = {}
    count = 0
    for directory, names, files in os.walk(root, followlinks=False):
        base = Path(directory)
        names[:] = sorted(name for name in names if not (base / name).is_symlink())
        for name in sorted(files):
            path = base / name
            if path.is_symlink() or not path.is_file():
                raise D3Error("EVIDENCE_FILE_TYPE")
            count += 1
            if count > MAX_FILES:
                raise D3Error("EVIDENCE_FILE_LIMIT")
            result[path.relative_to(root).as_posix()] = (path.stat().st_size, sha256_file(path))
    return result


def selected(plan: Mapping[str, object]) -> List[dict]:
    raw = plan.get("SelectedModules")
    if not isinstance(raw, list) or not all(isinstance(item, dict) for item in raw):
        raise D3Error("SELECTED_MODULES")
    return sorted(raw, key=lambda item: item.get("PlanOrder", 10**9))


def selected_identity(item: Mapping[str, object]) -> Optional[Tuple[str, str]]:
    module_id = item.get("ModuleId")
    version = item.get("ModuleVersion")
    if isinstance(module_id, str) and isinstance(version, str):
        return module_id, version
    return None


def execution_shape(plan: Mapping[str, object]) -> dict:
    raw = plan.get("ExecutionOrder")
    if not isinstance(raw, list):
        raise D3Error("EXECUTION_ORDER_LIST")
    result = {
        "count": len(raw), "dict": 0, "string": 0, "list": 0, "other": 0,
        "at": 0, "dict_id": 0, "dict_version": 0, "nested": 0,
        "string_pairs": [], "string_ids": [], "dict_pairs": [],
    }
    for item in raw:
        if isinstance(item, str):
            result["string"] += 1
            if item.count("@") == 1 and all(item.split("@", 1)):
                result["at"] += 1
                result["string_pairs"].append(tuple(item.split("@", 1)))
            else:
                result["string_ids"].append(item)
        elif isinstance(item, dict):
            result["dict"] += 1
            source = item
            if isinstance(item.get("ModuleIdentity"), dict):
                result["nested"] += 1
                source = item["ModuleIdentity"]
            if isinstance(source.get("ModuleId"), str):
                result["dict_id"] += 1
            if isinstance(source.get("ModuleVersion"), str):
                result["dict_version"] += 1
            pair = selected_identity(source)
            if pair is not None:
                result["dict_pairs"].append(pair)
        elif isinstance(item, list):
            result["list"] += 1
        else:
            result["other"] += 1
    return result


def plan_order_class(items: List[dict]) -> str:
    values = [item.get("PlanOrder") for item in items]
    if values == list(range(len(items))):
        return "Z"
    if values == list(range(1, len(items) + 1)):
        return "O"
    return "U"


def walk_quality(value: object, key: str = "") -> Tuple[int, int, int]:
    accepted = blocking = zero_block = 0
    if isinstance(value, dict):
        for child_key, child in value.items():
            a, b, z = walk_quality(child, str(child_key).upper())
            accepted += a
            blocking += b
            zero_block += z
    elif isinstance(value, list):
        for child in value:
            a, b, z = walk_quality(child, key)
            accepted += a
            blocking += b
            zero_block += z
    elif isinstance(value, str):
        token = value.upper().replace("-", "_").replace(" ", "_")
        if any(mark in token for mark in (
                "NON_BLOCKING", "PROCEED", "ACCEPT", "ALLOW", "READY", "APPROV", "USABLE")):
            accepted += 1
        if token in {"BLOCK", "BLOCKING", "REJECT", "REJECTED", "DENY", "DENIED", "FAIL", "FAILED"}:
            blocking += 1
    elif "BLOCK" in key:
        if value is False or value == 0:
            zero_block += 1
        elif value is True or (isinstance(value, (int, float)) and value > 0):
            blocking += 1
    return accepted, blocking, zero_block


def quality_shape(preflight: Mapping[str, object]) -> Tuple[str, int, int, int, int]:
    quality = preflight.get("QualityDecision")
    if quality is None:
        return "N", 0, 0, 0, 0
    kind = "D" if isinstance(quality, dict) else "L" if isinstance(quality, list) else "S" if isinstance(quality, str) else "O"
    key_count = len(quality) if isinstance(quality, dict) else 0
    accepted, blocking, zero_block = walk_quality(quality)
    return kind, key_count, accepted, blocking, zero_block


def diagnose(staging: Path) -> str:
    staging = staging.resolve(strict=True)
    if not staging.is_dir() or staging.is_symlink() or not staging.name.endswith(".staging"):
        raise D3Error("G2_STAGING_DIR")
    before = inventory(staging)
    plan = read_json(staging / "control" / "software_plan.json", "PLAN_JSON")
    preflight = read_json(staging / "control" / "run_preflight.json", "PREFLIGHT_JSON")
    items = selected(plan)
    identities = [selected_identity(item) for item in items]
    if not all(item is not None for item in identities):
        raise D3Error("SELECTED_IDENTITY")
    expected_pairs = [item for item in identities if item is not None]
    expected_ids = [item[0] for item in expected_pairs]
    shape = execution_shape(plan)
    pair_match = int(shape["string_pairs"] == expected_pairs or shape["dict_pairs"] == expected_pairs)
    id_match = int(shape["string_ids"] == expected_ids)
    qkind, qkeys, qaccept, qblock, qzero = quality_shape(preflight)

    if shape["string"] == shape["count"] and shape["at"] == shape["count"] and pair_match:
        reason = "EXEC_ORDER_STRING_IDENTITY_CONTRACT_GAP"
    elif shape["string"] == shape["count"] and id_match:
        reason = "EXEC_ORDER_ID_ONLY_CONTRACT_GAP"
    elif shape["dict"] == shape["count"] and shape["dict_id"] == shape["count"] and shape["dict_version"] < shape["count"]:
        reason = "EXEC_ORDER_VERSION_MISSING"
    elif shape["dict"] == shape["count"] and pair_match:
        reason = "EXEC_ORDER_DICT_IDENTITY"
    else:
        reason = "EXEC_ORDER_UNCLASSIFIED"

    after = inventory(staging)
    if before != after:
        raise D3Error("EVIDENCE_MUTATED")
    return (
        f"G2D3 N-E{shape['count']}-D{shape['dict']}-S{shape['string']}-L{shape['list']}-"
        f"A{shape['at']}-I{shape['dict_id']}-V{shape['dict_version']}-N{shape['nested']}-"
        f"PM{pair_match}-IM{id_match}-PO{plan_order_class(items)}-"
        f"Q{qkind}-QK{qkeys}-QA{qaccept}-QB{qblock}-QZ{qzero}-H{len(before)}-X0 "
        f"C-0 K-{reason}"
    )


def selftest() -> None:
    checks = 0

    def ck(value: bool) -> None:
        nonlocal checks
        if not value:
            raise AssertionError("SELFTEST")
        checks += 1

    with tempfile.TemporaryDirectory() as temporary:
        staging = Path(temporary) / "G2_TEST.staging"
        control = staging / "control"
        control.mkdir(parents=True)
        selected_modules = [
            {"PlanOrder": index, "ModuleId": f"U002.Test{index + 1}", "ModuleVersion": "0.1.0"}
            for index in range(3)
        ]
        plan = {
            "SelectedModules": selected_modules,
            "ExecutionOrder": [f"U002.Test{index + 1}@0.1.0" for index in range(3)],
        }
        preflight = {
            "QualityDecision": {"Decision": "PROCEED_WITH_WARNINGS", "BlockingReasonCount": 0},
        }
        (control / "software_plan.json").write_text(json.dumps(plan), encoding="utf-8")
        (control / "run_preflight.json").write_text(json.dumps(preflight), encoding="utf-8")
        before = inventory(staging)
        capsule = diagnose(staging)
        after = inventory(staging)
        ck(capsule.startswith("G2D3 "))
        ck("E3-D0-S3-L0-A3" in capsule)
        ck("I0-V0-N0" in capsule)
        ck("PM1-IM0-POZ" in capsule)
        ck("QD-QK2-QA1-QB0-QZ1" in capsule)
        ck(capsule.endswith("C-0 K-EXEC_ORDER_STRING_IDENTITY_CONTRACT_GAP"))
        ck(before == after)
        ck(len(before) == 2)
        shape = execution_shape(plan)
        ck(shape["string"] == 3)
        ck(shape["at"] == 3)
        ck(shape["dict"] == 0)
        ck(shape["string_pairs"][0] == ("U002.Test1", "0.1.0"))
        ck(plan_order_class(selected_modules) == "Z")
        ck(quality_shape(preflight) == ("D", 2, 1, 0, 1))
        ck(selected_identity(selected_modules[0]) == ("U002.Test1", "0.1.0"))
        ck(sha256_file(control / "software_plan.json") == sha256_file(control / "software_plan.json"))
        ck(not (staging / "runs").exists())
        ck("CONTRACT_GAP" in capsule)
    if checks != 18:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_CAPSULE)


def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--diagnose", action="store_true")
    parser.add_argument("--g2-staging", type=Path)
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            if args.g2_staging is not None:
                raise D3Error("ARGUMENTS")
            selftest()
            return 0
        if args.g2_staging is None:
            raise D3Error("ARGUMENTS")
        print(diagnose(args.g2_staging))
        return 0
    except D3Error as exc:
        print(f"G2D3 N-E0-D0-S0-L0-A0-I0-V0-N0-PM0-IM0-POU-QN-QK0-QA0-QB0-QZ0-H0-X1 C-7 K-{exc.reason}")
        return 7
    except Exception:
        print("G2D3 N-E0-D0-S0-L0-A0-I0-V0-N0-PM0-IM0-POU-QN-QK0-QA0-QB0-QZ0-H0-X1 C-7 K-INTERNAL")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
