#!/usr/bin/env python3
"""Read-only diagnostic for retained U21G G2R2 synthetic vertical evidence."""

from __future__ import annotations

import argparse
from pathlib import Path
import re


STAGING_REL = Path("05_TESTS/02_ACCEPTANCE/U21G_G2_VERTICAL_V06.staging")
PHASES = (
    ("R", "registry"),
    ("A1", "activation_1"),
    ("A2", "activation_2"),
    ("A3", "activation_3"),
    ("P", "resolver"),
    ("B", "binder"),
    ("E", "executor"),
    ("V", "run_manifest_validator"),
)


def lines(path: Path) -> list[str]:
    if not path.is_file() or path.stat().st_size > 2_097_152:
        return []
    return [item.strip() for item in path.read_text(encoding="utf-8", errors="replace").splitlines()
            if item.strip()]


def stable(value: str) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", value.upper()).strip("_")[:64] or "UNKNOWN"


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    root = Path(args.workspace_root).resolve()
    staging = root / STAGING_REL
    if not staging.is_dir():
        print("G2R2D1 N-L0-C0-R0-M0-PNA-X1 C-7 K-STAGING_MISSING")
        return 7
    logs = staging / "logs"
    observed = 0
    last_phase = "NA"
    last_line = ""
    explicit_phase = "NA"
    explicit_reason = ""
    for phase, name in PHASES:
        pair_present = False
        for suffix in ("stdout.txt", "stderr.txt"):
            path = logs / f"{name}.{suffix}"
            current = lines(path)
            if path.is_file():
                pair_present = True
            if current:
                last_phase = phase
                last_line = current[-1]
                for item in current:
                    match = re.search(r"\bC-7\b.*?\bK-([A-Za-z0-9_]+)", item)
                    if match:
                        explicit_phase = phase
                        explicit_reason = stable(match.group(1))
        if pair_present:
            observed += 1
    control = staging / "control"
    control_count = sum(path.is_file() for path in (
        control / "registry_snapshot.json",
        control / "software_plan.json",
        control / "run_request.json",
        control / "run_preflight.json",
    ))
    run_root = staging / "runs"
    run_dirs = [path for path in run_root.iterdir() if path.is_dir()] if run_root.is_dir() else []
    manifests = sum((path / "run_manifest.json").is_file() for path in run_dirs)
    reason = explicit_reason or "NO_EXPLICIT_CHILD_ERROR"
    phase = explicit_phase if explicit_reason else last_phase
    print(f"G2R2D1 N-L{observed}-C{control_count}-R{len(run_dirs)}-M{manifests}-P{phase}-X0 C-0 K-{reason}")
    if not explicit_reason:
        compact = re.sub(r"\s+", "_", last_line)[:180] or "EMPTY"
        print("D-" + compact)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
