#!/usr/bin/env python3
"""Read-only semantic probe for the frozen U21F2A Promotion v0.2 evidence."""

from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path
import re


GATE_CAPSULE = "U21F2AG1 N-R2-P4-D12-I4-X0 C-0"
PROMOTION_CAPSULE = "U21F2AP2 N-G2-H2-C2-R1-I4-X0 C-0"
PROTOCOL_SELFTEST = "U21F2A1 N-T24-P4-D16-K4-X0 C-0"


def reject_pairs(pairs):
    result = {}
    for key, value in pairs:
        if key in result:
            raise ValueError("DUPLICATE_KEY")
        result[key] = value
    return result


def load(path: Path):
    return json.loads(path.read_text(encoding="utf-8-sig"), object_pairs_hook=reject_pairs,
                      parse_constant=lambda _: (_ for _ in ()).throw(ValueError("NONFINITE")))


def sha(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def scalars(value, path=""):
    if isinstance(value, dict):
        for key, item in value.items():
            child = f"{path}.{key}" if path else str(key)
            yield from scalars(item, child)
    elif isinstance(value, list):
        for index, item in enumerate(value):
            yield from scalars(item, f"{path}[{index}]")
    else:
        yield path, value


def norm(value: str) -> str:
    return re.sub(r"[^a-z0-9]", "", value.lower())


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    root = Path(args.workspace_root).resolve()
    evidence = root / "05_TESTS/02_ACCEPTANCE/U21F2A_PROMOTION_V02/u21f2a_promotion_acceptance.json"
    bundle = root / "01_PROGRAMS/u002_module_runtime_protocol_bundle_v0_2.json"
    validator = root / "01_PROGRAMS/u002_module_runtime_protocol_validator_v0_2.py"
    try:
        if not all(path.is_file() and not path.is_symlink() for path in (evidence, bundle, validator)):
            raise ValueError("FILE")
        document = load(evidence)
        if not isinstance(document, dict):
            raise ValueError("ROOT")
        flat = list(scalars(document))
        values = [value for _, value in flat]
        version = any(norm(path).endswith("promotiontoolversion") and str(value) == "0.2"
                      for path, value in flat)
        gate = GATE_CAPSULE in values
        promotion = PROMOTION_CAPSULE in values
        selftest = PROTOCOL_SELFTEST in values
        hashes = sum(sha(path) in values for path in (bundle, validator))
        equality = sum(value is True and ("byteidentical" in norm(path) or
                       "byteequal" in norm(path)) for path, value in flat)
        relocated = sum(value is True and "relocated" in norm(path) and
                        ("selftest" in norm(path) or "validation" in norm(path))
                        for path, value in flat)
        print(f"G2R1P1 N-V{int(version)}-G{int(gate)}-P{int(promotion)}-S{int(selftest)}-H{hashes}-B{equality}-R{relocated}-X0 C-0")
        return 0
    except Exception as exc:
        reason = re.sub(r"[^A-Z0-9_]+", "_", exc.__class__.__name__.upper())
        print(f"G2R1P1 N-V0-G0-P0-S0-H0-B0-R0-X1 C-7 K-{reason}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
