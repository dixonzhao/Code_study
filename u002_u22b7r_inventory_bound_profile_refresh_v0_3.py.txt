class RefreshError(RuntimeError):
    """Represent one deterministic profile-refresh failure."""

    def __init__(self, reason: str, detail: object = "NA", stage: str = "B7R"):
        self.reason = machine_token(reason, 72)
        self.detail = machine_token(detail, 180)
        self.stage = machine_token(stage, 24)
        super().__init__(self.reason)


def machine_token(value: object, limit: int = 100) -> str:
    """Return a bounded machine-readable token."""

    text = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper()).strip("_")
    return (text or "UNKNOWN")[:limit]


def load_adjacent(name: str, basename: str):
    """Load one explicitly named adjacent high-code dependency."""

    path = Path(__file__).resolve().with_name(basename)
    if not path.is_file():
        raise RefreshError("DEPENDENCY_MISSING", basename, "IMPORT")
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    specification = importlib.util.spec_from_loader(loader.name, loader)
    if specification is None:
        raise RefreshError("DEPENDENCY_SPEC", basename, "IMPORT")
    module = importlib.util.module_from_spec(specification)
    loader.exec_module(module)
    return module


def sha256_file(path: Path) -> str:
    """Hash one file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def canonical_bytes(value: object) -> bytes:
    """Serialize one control document deterministically."""

    return (
        json.dumps(
            value,
            ensure_ascii=False,
            sort_keys=True,
            indent=2,
            allow_nan=False,
        )
        + "\n"
    ).encode("utf-8")


def write_json_atomic(path: Path, value: Mapping[str, object]) -> None:
    """Write one JSON document without exposing a partial file."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def write_text_atomic(path: Path, value: str) -> None:
    """Write one UTF-8 text document atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_text(value, encoding="utf-8", newline="\n")
    os.replace(temporary, path)


def load_json(path: Path, label: str) -> Dict[str, object]:
    """Load one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise RefreshError(label + "_MISSING", path, "DISCOVER")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise RefreshError(label + "_JSON", type(error).__name__, "DISCOVER") from error
    if not isinstance(value, dict):
        raise RefreshError(label + "_TYPE", path.name, "DISCOVER")
    return value


def normalize_name(value: object) -> str:
    """Normalize a governed identity without fuzzy substring matching."""

    return re.sub(r"[^a-z0-9]+", "", str(value).casefold())


def normalize_can_id(value: object) -> str:
    """Normalize common hexadecimal CAN identity forms."""

    text = str(value).strip().upper()
    if text.startswith("ID"):
        text = text[2:]
    if text.startswith("0X"):
        text = text[2:]
    if not re.fullmatch(r"[0-9A-F]+", text):
        return ""
    return text.lstrip("0") or "0"


def normalize_channel(value: object) -> str:
    """Normalize one integer-like Channel value."""

    text = str(value).strip()
    try:
        number = float(text)
    except ValueError:
        return text
    return str(int(number)) if number.is_integer() else text


def parse_boolean(value: object, field: str) -> bool:
    """Parse the governed CSV boolean representation."""

    text = str(value).strip().casefold()
    if text in {"1", "true"}:
        return True
    if text in {"0", "false"}:
        return False
    raise RefreshError("BOOLEAN", field, "PROFILE")


def parse_nonnegative_integer(value: object, field: str) -> int:
    """Parse an integer policy without silently truncating a fraction."""

    if isinstance(value, bool):
        raise RefreshError("INTEGER", field, "PROFILE")
    try:
        number = float(str(value).strip())
    except ValueError as error:
        raise RefreshError("INTEGER", field, "PROFILE") from error
    if not number.is_integer() or number < 0:
        raise RefreshError("INTEGER", field, "PROFILE")
    return int(number)


def compile_hold_policy(
    row: Mapping[str, str],
    role: str,
) -> Tuple[int, Optional[Dict[str, object]]]:
    """Compile declarative hold metadata into one executable integer policy.

    Historical rule FP027:
    - NONE plus blank means no cross-FrameCount hold and compiles to integer 0;
    - NONE plus an explicit value must equal integer 0;
    - BOUNDED / BOUNDED_HOLD requires an explicit nonnegative integer;
    - unknown or contradictory combinations fail closed.
    """

    policy = str(row.get("HoldPolicy", "")).strip().upper()
    observed = row.get("MaxAgeFrames")
    observed_text = str(observed if observed is not None else "").strip()
    if policy == "NONE":
        if observed_text == "":
            return 0, {
                "Role": role,
                "HoldPolicy": policy,
                "ObservedMaxAgeFrames": "BLANK",
                "CompiledMaxAgeFrames": 0,
                "Rule": "NO_HOLD_IMPLIES_ZERO_FRAME_AGE",
            }
        parsed = parse_nonnegative_integer(observed, role + "_MAX_AGE")
        if parsed != 0:
            raise RefreshError("NONE_NONZERO_MAX_AGE", role, "PROFILE")
        change = None
        if not isinstance(observed, int):
            change = {
                "Role": role,
                "HoldPolicy": policy,
                "ObservedMaxAgeFrames": observed_text,
                "CompiledMaxAgeFrames": 0,
                "Rule": "CONTROL_FIELD_INTEGER_COMPILATION",
            }
        return 0, change
    if policy in {"BOUNDED", "BOUNDED_HOLD"}:
        if observed_text == "":
            raise RefreshError("BOUNDED_MAX_AGE_BLANK", role, "PROFILE")
        parsed = parse_nonnegative_integer(observed, role + "_MAX_AGE")
        change = None
        if not isinstance(observed, int):
            change = {
                "Role": role,
                "HoldPolicy": policy,
                "ObservedMaxAgeFrames": observed_text,
                "CompiledMaxAgeFrames": parsed,
                "Rule": "CONTROL_FIELD_INTEGER_COMPILATION",
            }
        return parsed, change
    raise RefreshError("UNKNOWN_HOLD_POLICY", f"{role}:{policy or 'BLANK'}", "PROFILE")


def read_inventory(path: Path) -> List[Dict[str, str]]:
    """Read and validate the exact S0 signal inventory."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        fields = tuple(reader.fieldnames or ())
        missing = [field for field in INVENTORY_REQUIRED if field not in fields]
        if missing:
            raise RefreshError("INVENTORY_SCHEMA", missing[0], "PROFILE")
        rows = [{field: row.get(field, "") for field in fields} for row in reader]
    if not rows:
        raise RefreshError("INVENTORY_EMPTY", path.name, "PROFILE")
    canonical = [row["CanonicalName"] for row in rows]
    if any(not value for value in canonical) or len(canonical) != len(set(canonical)):
        raise RefreshError("CANONICAL_IDENTITY", len(canonical), "PROFILE")
    return rows


def discover_failed_s0(
    workspace_root: Path,
    failed_run_id: str,
    expected_package: Path,
) -> Tuple[Path, Path, Dict[str, object]]:
    """Resolve the accepted S0 Artifact inside the exact failed B7Q Run."""

    run_root = (workspace_root / "07_OUTPUT" / failed_run_id).resolve()
    status = load_json(run_root / "run_status.json", "FAILED_RUN_STATUS")
    if status.get("RunId") not in {None, failed_run_id}:
        raise RefreshError("FAILED_RUN_ID", status.get("RunId"), "DISCOVER")
    if status.get("Status") not in {"FAILED", "ABORTED"}:
        raise RefreshError("FAILED_RUN_STATUS", status.get("Status"), "DISCOVER")
    config = load_json(run_root / "00_CONFIG/run_configuration.json", "FAILED_RUN_CONFIG")
    if config.get("RunId") != failed_run_id:
        raise RefreshError("FAILED_CONFIG_ID", config.get("RunId"), "DISCOVER")
    if Path(str(config.get("InputPackage", ""))).resolve() != expected_package.resolve():
        raise RefreshError("FAILED_INPUT_PACKAGE", config.get("InputPackage"), "DISCOVER")

    manifests = []
    for path in run_root.rglob("artifact_manifest.json"):
        try:
            value = load_json(path, "S0_MANIFEST")
        except RefreshError:
            continue
        if value.get("ArtifactType") == "U001_INTAKE_ACCEPTED":
            manifests.append((path.resolve(), value))
    if len(manifests) != 1:
        raise RefreshError("S0_MANIFEST_COUNT", len(manifests), "DISCOVER")
    manifest_path, manifest = manifests[0]
    inventory = manifest_path.parent / "u001_signal_inventory.csv"
    if not inventory.is_file() or inventory.is_symlink():
        raise RefreshError("S0_INVENTORY_MISSING", inventory.name, "DISCOVER")
    payloads = manifest.get("Payloads")
    if not isinstance(payloads, list):
        raise RefreshError("S0_PAYLOADS", "TYPE", "DISCOVER")
    selected = [
        row
        for row in payloads
        if isinstance(row, dict) and row.get("RelativePath") == "u001_signal_inventory.csv"
    ]
    if len(selected) != 1 or selected[0].get("Sha256") != sha256_file(inventory):
        raise RefreshError("S0_INVENTORY_PAYLOAD", len(selected), "DISCOVER")
    return run_root, inventory.resolve(), config


def exact_role_matches(
    rows: Sequence[Mapping[str, str]],
    role: str,
) -> List[Mapping[str, str]]:
    """Select one role by the governed Channel, CAN ID and Signal identity."""

    channel, can_id, signal = ROLE_IDENTITIES[role]
    matches: List[Mapping[str, str]] = []
    for row in rows:
        if channel is not None and normalize_channel(row.get("Channel")) != channel:
            continue
        if normalize_can_id(row.get("CAN_ID")) != can_id:
            continue
        if normalize_name(row.get("SignalName")) != signal:
            continue
        matches.append(row)
    return matches


def binding_record(
    row: Mapping[str, str],
    role: str,
) -> Tuple[Dict[str, object], Optional[Dict[str, object]]]:
    """Preserve inventory semantics while materializing typed policy fields."""

    if not parse_boolean(row.get("IncludeInFrameView"), role + "_FRAME"):
        raise RefreshError("NOT_IN_FRAME_VIEW", role, "PROFILE")
    compiled_max_age, compilation = compile_hold_policy(row, role)
    result: Dict[str, object] = {
        field: row.get(field, "") for field in INVENTORY_REQUIRED
    }
    result["RequiredForAnalysis"] = parse_boolean(
        row.get("RequiredForAnalysis"), role + "_REQUIRED"
    )
    result["IncludeInFrameView"] = True
    result["IncludeInReviewView"] = parse_boolean(
        row.get("IncludeInReviewView"), role + "_REVIEW"
    )
    result["MaxAgeFrames"] = compiled_max_age
    return result, compilation


def build_profile(
    inventory: Path,
    rows: Sequence[Mapping[str, str]],
    source_run_id: str,
) -> Tuple[Dict[str, object], Dict[str, int]]:
    """Create one complete profile with explicit required/optional cardinality."""

    bindings: Dict[str, List[Dict[str, object]]] = {}
    counts: Dict[str, int] = {}
    compilation_changes: List[Dict[str, object]] = []
    for role in ROLE_IDENTITIES:
        matches = exact_role_matches(rows, role)
        counts[role] = len(matches)
        if len(matches) > 1:
            raise RefreshError("ROLE_COUNT", f"{role}:{len(matches)}", "PROFILE")
        if len(matches) == 0:
            if role in REQUIRED_ROLES:
                raise RefreshError("ROLE_COUNT", f"{role}:0", "PROFILE")
            bindings[role] = []
            continue
        record, compilation = binding_record(matches[0], role)
        bindings[role] = [record]
        if compilation is not None:
            compilation_changes.append(compilation)
    inventory_hash = sha256_file(inventory)
    profile = {
        "ContractVersion": "U002_SIGNAL_BINDING_PROFILE_V0_1",
        "BindingProfileId": "TC001.CameraRadar.FullGeometry." + inventory_hash[:12],
        "BindingProfileVersion": "0.3.0",
        "Status": "CONTROLLED_INVENTORY_BOUND",
        "SourceRunId": source_run_id,
        "SourceInventorySha256": inventory_hash,
        "InventorySignalCount": len(rows),
        "RequiredRoles": list(REQUIRED_ROLES),
        "OptionalRoles": list(OPTIONAL_ROLES),
        "Bindings": bindings,
        "BindingCounts": counts,
        "DuplicateRoleCount": 0,
        "BlockingIssues": [],
        "CompilationRule": "HOLD_POLICY_TO_EXECUTABLE_INTEGER_MAX_AGE_V0_1",
        "CompilationChanges": compilation_changes,
        "HistoricalFailureCoverage": list(HISTORICAL_FAILURE_COVERAGE),
        "HistoricalFailureTestMap": {
            key: list(value) for key, value in HISTORICAL_FAILURE_TEST_MAP.items()
        },
        "Prohibitions": [
            "Recognition_ID is not authoritative target truth",
            "Radar is not used to select the Camera target",
            "No binding is inferred by a fuzzy keyword match",
            "This profile is valid only for the recorded S0 inventory hash",
        ],
    }
    return profile, counts


def persist_immutable(path: Path, value: Mapping[str, object], label: str) -> str:
    """Create one versioned document or accept a byte-identical replay."""

    payload = canonical_bytes(value)
    if path.exists():
        if not path.is_file() or path.read_bytes() != payload:
            raise RefreshError(label + "_CONFLICT", path.name, "PERSIST")
        return "REPLAY"
    write_json_atomic(path, value)
    return "CREATED"


def build_refreshed_config(
    failed_config: Mapping[str, object],
    profile: Mapping[str, object],
    profile_path: Path,
    workspace_root: Path,
    run_id: str,
) -> Dict[str, object]:
    """Clone the failed evidence intent while changing only controlled identity."""

    value = json.loads(json.dumps(failed_config))
    value["RunId"] = run_id
    value["SignalProfile"] = (
        str(profile["BindingProfileId"]) + "@" + str(profile["BindingProfileVersion"])
    )
    plan = value.get("ModulePlan")
    if not isinstance(plan, list) or len(plan) != 2:
        raise RefreshError("MODULE_PLAN", len(plan or []), "CONFIG")
    camera = [
        row
        for row in plan
        if isinstance(row, dict)
        and row.get("ModuleId") == "CameraRadar.S1.CameraTargetContinuityArbitration"
    ]
    if len(camera) != 1 or camera[0].get("ModuleVersion") != "0.2.0":
        raise RefreshError("CAMERA_MODULE", len(camera), "CONFIG")
    configuration = camera[0].get("Configuration")
    if not isinstance(configuration, dict):
        raise RefreshError("CAMERA_CONFIGURATION", "TYPE", "CONFIG")
    relative = profile_path.resolve().relative_to(workspace_root.resolve())
    configuration["BindingProfilePath"] = relative.as_posix()
    configuration["BindingProfileSha256"] = sha256_file(profile_path)
    if configuration.get("ContextFrameStart") is None:
        configuration["ContextFrameStart"] = int(value["FrameStart"])
    if configuration.get("SeedFrame") is None:
        configuration["SeedFrame"] = int(value["FrameStart"])
    return value


def execute(
    c_code_root: Path,
    workspace_root: Path,
    b7p2_root: Path,
    output_dir: Path,
    failed_run_id: str,
    run_id: str,
    module_timeout: int,
) -> Tuple[str, str]:
    """Refresh the binding profile, execute a new Run and profile geometry."""

    diagnostics: List[Dict[str, object]] = []
    try:
        b7q = load_adjacent("u22b7r_b7q_runtime", B7Q_BASENAME)
        b7a_program = Path(__file__).resolve().with_name(B7A_BASENAME)
        if c_code_root.resolve() != workspace_root.resolve().parent:
            raise RefreshError("C_CODE_ROOT", c_code_root, "WORKSPACE")
        core = b7q.require_workspace(workspace_root)
        package, verification, geometry = b7q.validate_full_package(b7p2_root)
        failed_run_root, inventory, failed_config = discover_failed_s0(
            workspace_root,
            failed_run_id,
            package,
        )
        rows = read_inventory(inventory)
        profile, role_counts = build_profile(inventory, rows, failed_run_id)
        profile_name = (
            "tc001_camera_radar_full_geometry_binding_"
            + str(profile["SourceInventorySha256"])[:12]
            + "_v0_3.json"
        )
        profile_path = workspace_root / "02_CONFIG/signal_profiles" / profile_name
        profile_disposition = persist_immutable(profile_path, profile, "PROFILE")
        config = build_refreshed_config(
            failed_config,
            profile,
            profile_path,
            workspace_root,
            run_id,
        )
        config_path = workspace_root / "02_CONFIG/run_configs" / (run_id + ".json")
        config_disposition = persist_immutable(config_path, config, "CONFIG")
        protected = [
            inventory,
            profile_path,
            config_path,
            b7a_program,
            *core.values(),
            *[package / name for name in b7q.FINAL_PACKAGE_FILES],
        ]
        before = b7q.snapshot_files(protected)
        run_root, run_replay = b7q.execute_or_replay_run(
            workspace_root,
            core,
            config_path,
            run_id,
            module_timeout,
            diagnostics,
        )
        linked = b7q.validate_linked_run(
            run_root,
            package,
            int(config["FrameStart"]),
            int(config["FrameEnd"]),
        )
        profile_dir = output_dir / "B7A_PROFILE"
        b7a_capsule, b7a_counts, profile_replay = b7q.run_or_replay_b7a(
            b7a_program,
            run_root,
            profile_dir,
            int(config["FrameStart"]),
            int(config["FrameEnd"]),
            diagnostics,
        )
        if before != b7q.snapshot_files(protected):
            raise RefreshError("PROTECTED_INPUT_CHANGED", "HASH_OR_METADATA", "VERIFY")
        output_dir.mkdir(parents=True, exist_ok=True)
        line1 = (
            f"B7R3 N-I{len(rows)}-B{len(role_counts)}-M2"
            f"-A{linked['ArtifactCount']}-F{linked['ReconstructionRows']}"
            f"-G{b7a_counts['PairedGeometryRows']}"
            f"-R{run_replay}-P{profile_replay}-X0 C-0"
        )
        write_json_atomic(
            output_dir / "u22b7r_inventory_bound_profile_refresh_receipt.json",
            {
                "CompletionCode": 0,
                "ProgramId": PROGRAM_ID,
                "ProgramVersion": PROGRAM_VERSION,
                "FailedRunId": failed_run_id,
                "FailedRunRoot": str(failed_run_root),
                "NewRunId": run_id,
                "NewRunRoot": str(run_root),
                "U001Package": str(package),
                "U001Verification": verification,
                "GeometryBindings": geometry,
                "SourceInventory": str(inventory),
                "SourceInventorySha256": profile["SourceInventorySha256"],
                "BindingProfile": str(profile_path),
                "BindingProfileSha256": sha256_file(profile_path),
                "BindingProfileDisposition": profile_disposition,
                "BindingCounts": role_counts,
                "RunConfiguration": str(config_path),
                "RunConfigurationDisposition": config_disposition,
                "LinkedRunEvidence": linked,
                "B7ACapsule": b7a_capsule,
                "B7ACounts": b7a_counts,
                "Diagnostics": diagnostics,
                "Correction": "REGENERATE_PROFILE_FROM_EXACT_S0_INVENTORY",
                "MetadataCompilation": "NONE_BLANK_MAX_AGE_TO_INTEGER_ZERO",
                "HistoricalFailureCoverage": list(HISTORICAL_FAILURE_COVERAGE),
                "HistoricalFailureTestMap": {
                    key: list(value)
                    for key, value in HISTORICAL_FAILURE_TEST_MAP.items()
                },
                "ProhibitedCorrection": "DO_NOT_DISABLE_OR_REWRITE_INVENTORY_HASH_CHECK",
                "NextAction": "AUTHOR_AND_TEST_CAMERA_S1_V0_3_GEOMETRY_EXPERIMENT",
            },
        )
        write_text_atomic(
            output_dir / "u22b7r_return_capsule.txt",
            line1 + "\n" + b7a_capsule + "\n",
        )
        return line1, b7a_capsule
    except Exception as error:
        if isinstance(error, RefreshError):
            reason, detail, stage = error.reason, error.detail, error.stage
        elif hasattr(error, "reason") and hasattr(error, "detail"):
            reason = machine_token(getattr(error, "reason"), 72)
            detail = machine_token(getattr(error, "detail"), 180)
            stage = machine_token(getattr(error, "stage", "DEPENDENCY"), 24)
        else:
            reason = machine_token(type(error).__name__, 72)
            detail = machine_token(str(error), 180)
            stage = "UNHANDLED"
        line1 = f"B7RE K-{reason}-J{stage}-X1 C-7"
        line2 = f"MEMO1-{detail}"
        try:
            output_dir.mkdir(parents=True, exist_ok=True)
            write_json_atomic(
                output_dir / "u22b7r_failure_diagnostic.json",
                {
                    "CompletionCode": 7,
                    "ProgramId": PROGRAM_ID,
                    "ProgramVersion": PROGRAM_VERSION,
                    "Reason": reason,
                    "Detail": detail,
                    "Stage": stage,
                    "Diagnostics": diagnostics,
                    "Traceback": traceback.format_exc(),
                },
            )
            write_text_atomic(
                output_dir / "u22b7r_return_capsule.txt",
                line1 + "\n" + line2 + "\n",
            )
        except Exception:
            pass
        return line1, line2


def self_test() -> None:
    """Exercise identity matching, ambiguity rejection and immutable controls."""

    checks: List[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            raise AssertionError(name)
        checks.append(name)

    def row(
        role: str,
        canonical: str,
        hold_policy: str = "NONE",
        max_age: str = "",
    ) -> Dict[str, str]:
        channel, can_id, signal = ROLE_IDENTITIES[role]
        return {
            "CanonicalName": canonical,
            "CAN_ID": "0x" + can_id,
            "Channel": channel or "1",
            "MessageName": role,
            "SignalName": signal,
            "Unit": "m" if "DISTANCE" in role or "LATERAL" in role else "",
            "ValueKind": "CONTINUOUS",
            "SignalRole": role,
            "UpdateMode": "EVENT_DRIVEN",
            "StorageDtype": "float64",
            "WithinFramePolicy": "LAST_VALID_UPDATE",
            "HoldPolicy": hold_policy,
            "MaxAgeFrames": max_age,
            "RequiredForAnalysis": "1",
            "IncludeInFrameView": "1",
            "IncludeInReviewView": "1",
        }

    with tempfile.TemporaryDirectory(prefix="u22b7r_") as temporary:
        root = Path(temporary)
        inventory = root / "u001_signal_inventory.csv"
        rows = [row(role, "C" + str(index)) for index, role in enumerate(ROLE_IDENTITIES)]
        with inventory.open("w", encoding="utf-8", newline="") as stream:
            writer = csv.DictWriter(stream, fieldnames=list(INVENTORY_REQUIRED))
            writer.writeheader()
            writer.writerows(rows)
        observed = read_inventory(inventory)
        check("T01_ROWS", len(observed) == 7)
        profile, counts = build_profile(inventory, observed, "FAILED_TEST")
        check("T02_BINDINGS", len(profile["Bindings"]) == 7)
        check("T03_COUNTS", set(counts.values()) == {1})
        check("T04_HASH", profile["SourceInventorySha256"] == sha256_file(inventory))
        check("T05_BLOCKING", profile["BlockingIssues"] == [])
        check("T06_DUPLICATE", profile["DuplicateRoleCount"] == 0)
        check("T07_PROHIBITIONS", len(profile["Prohibitions"]) == 4)
        check("T07A_COMPILE_COUNT", len(profile["CompilationChanges"]) == 7)
        check(
            "T07B_COMPILED_ZERO",
            all(
                selected[0]["MaxAgeFrames"] == 0
                for selected in profile["Bindings"].values()
                if selected
            ),
        )
        without_optional = [
            item
            for item in observed
            if len(exact_role_matches([item], "VEHICLE_SPEED")) == 0
        ]
        optional_profile, optional_counts = build_profile(
            inventory,
            without_optional,
            "FAILED_TEST",
        )
        check("T07C_OPTIONAL_ZERO_COUNT", optional_counts["VEHICLE_SPEED"] == 0)
        check("T07D_OPTIONAL_EMPTY_LIST", optional_profile["Bindings"]["VEHICLE_SPEED"] == [])
        check("T07E_OPTIONAL_NO_COMPILE", len(optional_profile["CompilationChanges"]) == 6)
        required_missing_rejected = False
        try:
            build_profile(inventory, observed[1:], "FAILED_TEST")
        except RefreshError as error:
            required_missing_rejected = (
                error.reason == "ROLE_COUNT"
                and error.detail == "CAMERA_1_DISTANCE_0"
            )
        check("T07F_REQUIRED_MISSING", required_missing_rejected)
        optional_duplicate_rejected = False
        try:
            build_profile(inventory, observed + [dict(observed[-1])], "FAILED_TEST")
        except RefreshError as error:
            optional_duplicate_rejected = (
                error.reason == "ROLE_COUNT"
                and error.detail == "VEHICLE_SPEED_2"
            )
        check("T07G_OPTIONAL_DUPLICATE", optional_duplicate_rejected)
        check("T08_ROLE_EXACT", len(exact_role_matches(observed, "CAMERA_1_DISTANCE")) == 1)
        check("T09_NO_FUZZY", normalize_name("DistanceExtra") != "distance")
        path = root / "profile.json"
        check("T10_CREATE", persist_immutable(path, profile, "PROFILE") == "CREATED")
        check("T11_REPLAY", persist_immutable(path, profile, "PROFILE") == "REPLAY")
        failed = False
        try:
            build_profile(inventory, observed + [dict(observed[0])], "FAILED_TEST")
        except RefreshError as error:
            failed = error.reason == "ROLE_COUNT"
        check("T12_AMBIGUITY", failed)
        explicit_zero, explicit_zero_change = compile_hold_policy(
            row("CAMERA_1_DISTANCE", "Z", "NONE", "0"),
            "CAMERA_1_DISTANCE",
        )
        check("T12A_NONE_ZERO", explicit_zero == 0)
        check("T12B_NONE_ZERO_TYPED", explicit_zero_change is not None)
        bounded, bounded_change = compile_hold_policy(
            row("CAMERA_1_DISTANCE", "B", "BOUNDED_HOLD", "3"),
            "CAMERA_1_DISTANCE",
        )
        check("T12C_BOUNDED", bounded == 3)
        check("T12D_BOUNDED_TYPED", bounded_change is not None)
        rejected = []
        for policy, age, expected_reason in (
            ("NONE", "1", "NONE_NONZERO_MAX_AGE"),
            ("BOUNDED_HOLD", "", "BOUNDED_MAX_AGE_BLANK"),
            ("UNKNOWN", "1", "UNKNOWN_HOLD_POLICY"),
            ("BOUNDED", "1.5", "INTEGER"),
        ):
            try:
                compile_hold_policy(
                    row("CAMERA_1_DISTANCE", "R", policy, age),
                    "CAMERA_1_DISTANCE",
                )
            except RefreshError as error:
                rejected.append(error.reason == expected_reason)
        check("T12E_POLICY_REJECTION_MATRIX", rejected == [True, True, True, True])
        source = {
            "ContractVersion": "U22.0.1",
            "RunId": "OLD",
            "RunPurpose": "EVALUATION",
            "RunScope": "FRAME_RANGE",
            "FrameStart": 10,
            "FrameEnd": 20,
            "InputPackage": str(root),
            "SignalProfile": "OLD@0.1",
            "ModulePlan": [
                {
                    "Sequence": 1,
                    "Stage": "S0",
                    "ModuleId": "CameraRadar.S0.U001ControlledProfileIntake",
                    "ModuleVersion": "0.1.1",
                    "Configuration": {},
                },
                {

                    "ModuleId": "CameraRadar.S1.CameraTargetContinuityArbitration",
                    "ModuleVersion": "0.2.0",

                        "BindingProfilePath": "old.json",
                        "BindingProfileSha256": "0" * 64,
                        "ContextFrameStart": 10,
                        "SeedFrame": 10,
                    },
                },
            ],
        }
        workspace = root / "U-UDP002"
        target_profile = workspace / "02_CONFIG/signal_profiles/profile.json"
        target_profile.parent.mkdir(parents=True)
        target_profile.write_bytes(path.read_bytes())
        config = build_refreshed_config(
            source,
            profile,
            target_profile,
            workspace,
            "NEW",
        )
        check("T13_NEW_RUN", config["RunId"] == "NEW")
        camera = config["ModulePlan"][1]["Configuration"]
        check("T14_PROFILE_PATH", camera["BindingProfilePath"].startswith("02_CONFIG/"))
        check("T15_PROFILE_HASH", camera["BindingProfileSha256"] == sha256_file(target_profile))
        check("T16_SOURCE_UNCHANGED", source["RunId"] == "OLD")
        check("T17_RANGE", FRAME_END - FRAME_START + 1 == 401)
        check("T18_ROLE_COUNT", len(ROLE_IDENTITIES) == 7)
        check("T19_PROGRAM", PROGRAM_ID.startswith("UDP002"))
        check("T20_VERSION", PROGRAM_VERSION == "0.3.0")
        check("T21_DEPENDENCIES", B7Q_BASENAME.endswith(".py.txt"))
        check("T22_DIAGNOSTIC", "u22b7r_failure_diagnostic.json" in Path(__file__).read_text(encoding="utf-8"))
        check(
            "T23_FAIL_CLOSED",
            "DO_NOT_DISABLE_OR_REWRITE_INVENTORY_HASH_CHECK"
            in Path(__file__).read_text(encoding="utf-8"),
        )
        check("T23A_HISTORY", len(HISTORICAL_FAILURE_COVERAGE) == 4)
        check(
            "T23B_HISTORY_TEST_MAP",
            set(HISTORICAL_FAILURE_TEST_MAP) == set(HISTORICAL_FAILURE_COVERAGE)
            and all(
                test_name in checks
                for test_names in HISTORICAL_FAILURE_TEST_MAP.values()
                for test_name in test_names

        check("T24_CAPSULE", len("B7R3 N-I22-B7-M2-A2-F401-G401-R0-P0-X0 C-0") < 100)
    if len(checks) != 38:
        raise AssertionError("SELFTEST_COUNT")

def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test or the company profile-refresh workflow."""

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--execute", action="store_true")
    parser.add_argument("--c-code-root", default=".")
    parser.add_argument("--workspace-root", default="U-UDP002")
    parser.add_argument(
        "--b7p2-root",
        default="U-UDP001/03_CASES/TC001_20260422/02_RUNS/U22B7P_FULL_GEOMETRY_V02",
    )
    parser.add_argument("--output-dir", default="U-UDP002/04_TEMP/U22B7R3_OUTPUT")
    parser.add_argument("--failed-run-id", default=FAILED_RUN_ID)
    parser.add_argument("--run-id", default=RUN_ID)
    parser.add_argument("--module-timeout-seconds", type=int, default=1800)
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            self_test()
            return 0
        if not args.execute:
            raise RefreshError("MODE_REQUIRED", "USE_EXECUTE", "CLI")
        line1, line2 = execute(
            Path(args.c_code_root).resolve(),
            Path(args.workspace_root).resolve(),
            Path(args.b7p2_root).resolve(),
            Path(args.output_dir).resolve(),
            args.failed_run_id,
            args.run_id,
            args.module_timeout_seconds,
        )
        print(line1)
        print(line2)
        return 0 if line1.endswith("C-0") else 7
    except Exception as error:
        if isinstance(error, RefreshError):
            reason, detail, stage = error.reason, error.detail, error.stage
        else:
            reason = machine_token(type(error).__name__, 72)
            detail = machine_token(str(error), 180)
            stage = "UNHANDLED"
        print(f"B7RE K-{reason}-J{stage}-X1 C-7")
        print(f"MEMO1-{detail}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
