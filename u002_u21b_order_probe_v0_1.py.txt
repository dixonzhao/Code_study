from __future__ import annotations

import argparse
import json
from pathlib import Path
import re
import sys


def token(entry: dict) -> str:
    path = entry.get("ManifestRelativePath") or ""
    match = re.match(r"^(\d{2})_", path)
    return match.group(1) if match else "XX"


def sort_key(entry: dict) -> tuple[str, str, str, str]:
    return (
        entry.get("Stage") or "",
        entry.get("ModuleId") or "",
        entry.get("ModuleVersion") or "",
        entry.get("ManifestRelativePath") or "",
    )


def main() -> int:
    parser = argparse.ArgumentParser(description="Read-only U21B Registry order probe")
    parser.add_argument("snapshot")
    args = parser.parse_args()
    try:
        path = Path(args.snapshot)
        value = json.loads(path.read_text(encoding="utf-8"))
        modules = value["Modules"]
        if not isinstance(modules, list) or len(modules) != 10:
            print(f"U21BD1-ERROR-MODULECOUNT-{len(modules) if isinstance(modules, list) else 'NA'} C-7")
            return 7
        actual = "".join(token(item) for item in modules)
        expected = "".join(token(item) for item in sorted(modules, key=sort_key))
        path_order = "".join(token(item) for item in sorted(modules, key=lambda item: item.get("ManifestRelativePath") or ""))
        print(
            "U21BD1 "
            f"A-{actual} "
            f"E-{expected} "
            f"P{1 if actual == path_order else 0} "
            f"S{1 if actual == expected else 0} "
            "X0 C-0"
        )
        return 0
    except Exception as exc:
        print(f"U21BD1-ERROR-{type(exc).__name__.upper()} C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
