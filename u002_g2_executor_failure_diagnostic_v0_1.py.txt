#!/usr/bin/env python3
"""Read-only diagnostic for a retained U21G G2 .staging directory.

This tool diagnoses why the frozen Controlled Stage Executor returned a
non-zero exit during G2. It never executes or imports project programs and
never writes inside the supplied evidence directory.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import sys
import tempfile
from typing import Dict, Iterable, List, Mapping, Optional, Sequence, Tuple


VERSION = "0.1"
SELFTEST_CAPSULE = "G2D1ST N-T16-P4-J4-I4-X0 C-0"
MAX_TEXT_BYTES = 131_072
MAX_JSON_BYTES = 2_097_152
MAX_FILES = 4096
MAX_MEMO = 140
CAPSULE_REASON_RE = re.compile(r"\bC-7\s+K-([A-Z0-9_]+)\b")
TOKEN_RE = re.compile(r"[^A-Z0-9_]+")
STATUS_CODE = {
    "PENDING": "P",
    "RUNNING": "R",
    "SUCCEEDED": "S",
    "SKIPPED": "K",
    "FAILED": "F",
    "ABORTED": "F",
    "DENIED": "F",
}


class D1Error(Exception):
    def __init__(self, reason: str):
        super().__init__(reason)
        self.reason = reason


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def require_regular(path: Path, reason: str) -> None:
    if not path.is_file() or path.is_symlink():
        raise D1Error(reason)


def bounded_tail(path: Path, limit: int = MAX_TEXT_BYTES) -> str:
    require_regular(path, "LOG_FILE")
    size = path.stat().st_size
    with path.open("rb") as stream:
        if size > limit:
            stream.seek(size - limit)
        data = stream.read(limit)
    return data.decode("utf-8", errors="replace")


def read_json(path: Path) -> Optional[dict]:
    if not path.is_file() or path.is_symlink() or path.stat().st_size > MAX_JSON_BYTES:
        return None
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError):
        return None
    return value if isinstance(value, dict) else None


def safe_token(value: object, fallback: str = "UNKNOWN") -> str:
    if not isinstance(value, str):
        return fallback
    token = TOKEN_RE.sub("_", value.upper()).strip("_")
    return token[:48] or fallback


def last_nonempty(text: str) -> str:
    lines = [line.strip() for line in text.splitlines() if line.strip()]
    return lines[-1] if lines else ""


def sanitize_memo(text: str) -> str:
    text = text.replace("\\", "/")
    text = re.sub(r"[A-Za-z]:/[^\s]+", "PATH", text)
    text = re.sub(r"/(?:[^\s/]+/){2,}[^\s]+", "PATH", text)
    text = re.sub(r"\s+", "_", text.strip())
    text = re.sub(r"[^A-Za-z0-9_.:+-]", "?", text)
    return text[:MAX_MEMO] or "NO_STABLE_REASON"


def inventory(root: Path) -> Dict[str, Tuple[int, str]]:
    result: Dict[str, Tuple[int, str]] = {}
    count = 0
    for directory, names, files in os.walk(root, followlinks=False):
        base = Path(directory)
        names[:] = sorted(name for name in names if not (base / name).is_symlink())
        for name in sorted(files):
            path = base / name
            if path.is_symlink() or not path.is_file():
                raise D1Error("EVIDENCE_FILE_TYPE")
            count += 1
            if count > MAX_FILES:
                raise D1Error("EVIDENCE_FILE_LIMIT")
            relative = path.relative_to(root).as_posix()
            result[relative] = (path.stat().st_size, sha256_file(path))
    return result


def find_reason(stdout_text: str, stderr_text: str) -> Optional[str]:
    for text in (stdout_text, stderr_text):
        matches = CAPSULE_REASON_RE.findall(text)
        if matches:
            return matches[-1]
    return None


def stage_mask(document: Mapping[str, object]) -> str:
    raw = document.get("Stages")
    if not isinstance(raw, list):
        return "NA"
    mapping: Dict[str, str] = {}
    for item in raw:
        if not isinstance(item, dict):
            continue
        stage = item.get("Stage")
        status = item.get("Status")
        if isinstance(stage, str) and isinstance(status, str):
            mapping[stage] = STATUS_CODE.get(status.upper(), "?")
    return "".join(mapping.get(f"S{index}", "?") for index in range(8))


def module_counts(document: Mapping[str, object]) -> Dict[str, int]:
    counts = {"S": 0, "F": 0, "K": 0, "P": 0, "R": 0, "?": 0}
    raw = document.get("ResolvedModules")
    if not isinstance(raw, list):
        return counts
    for item in raw:
        if not isinstance(item, dict):
            counts["?"] += 1
            continue
        status = str(item.get("Status", "")).upper()
        counts[STATUS_CODE.get(status, "?")] += 1
    return counts


def artifact_count(document: Mapping[str, object]) -> int:
    raw = document.get("Artifacts")
    return len(raw) if isinstance(raw, list) else 0


def diagnostic_tokens(run_dir: Path) -> List[str]:
    keys = ("ErrorCode", "FailureReason", "Reason", "ErrorMessage")
    tokens: List[str] = []
    roots = [path for path in run_dir.rglob("*")
             if path.is_dir() and path.name.upper() in {"02_DIAGNOSTICS", "DIAGNOSTICS"}]
    for root in roots:
        for path in sorted(root.rglob("*.json"))[:256]:
            document = read_json(path)
            if document is None:
                continue
            stack: List[object] = [document]
            while stack:
                value = stack.pop()
                if isinstance(value, dict):
                    for key, item in value.items():
                        if key in keys and isinstance(item, str):
                            token = safe_token(item)
                            if token not in tokens:
                                tokens.append(token)
                        elif isinstance(item, (dict, list)):
                            stack.append(item)
                elif isinstance(value, list):
                    stack.extend(item for item in value if isinstance(item, (dict, list)))
    return tokens[:16]


def diagnose(staging: Path) -> Tuple[str, Optional[str]]:
    staging = staging.resolve(strict=True)
    if not staging.is_dir() or staging.is_symlink() or not staging.name.endswith(".staging"):
        raise D1Error("G2_STAGING_DIR")
    logs = staging / "logs"
    if not logs.is_dir() or logs.is_symlink():
        raise D1Error("LOGS_DIR")
    stdout_path = logs / "executor.stdout.txt"
    stderr_path = logs / "executor.stderr.txt"
    require_regular(stdout_path, "EXECUTOR_STDOUT")
    require_regular(stderr_path, "EXECUTOR_STDERR")

    before = inventory(staging)
    stdout_text = bounded_tail(stdout_path)
    stderr_text = bounded_tail(stderr_path)
    stdout_nonempty = int(bool(stdout_text.strip()))
    stderr_nonempty = int(bool(stderr_text.strip()))
    traceback = int("Traceback (most recent call last)" in stdout_text or
                    "Traceback (most recent call last)" in stderr_text)

    runs_root = staging / "runs"
    run_dirs = []
    if runs_root.is_dir() and not runs_root.is_symlink():
        run_dirs = sorted(path for path in runs_root.iterdir()
                          if path.is_dir() and not path.is_symlink() and path.name.startswith("RUN_"))

    manifests: List[dict] = []
    masks: List[str] = []
    counts = {"S": 0, "F": 0, "K": 0, "P": 0, "R": 0, "?": 0}
    artifacts = 0
    diag: List[str] = []
    for run_dir in run_dirs:
        manifest = read_json(run_dir / "run_manifest.json")
        if manifest is not None:
            manifests.append(manifest)
            masks.append(stage_mask(manifest))
            current = module_counts(manifest)
            for key in counts:
                counts[key] += current[key]
            artifacts += artifact_count(manifest)
        diag.extend(token for token in diagnostic_tokens(run_dir) if token not in diag)

    reason = find_reason(stdout_text, stderr_text)
    if reason is None and diag:
        reason = diag[0]
    if reason is None:
        if traceback:
            reason = "TRACEBACK"
        elif run_dirs and manifests:
            statuses = {str(item.get("RunStatus", "")).upper() for item in manifests}
            reason = "RUN_" + safe_token(sorted(statuses)[0]) if statuses else "RUN_STATUS_UNKNOWN"
        elif run_dirs:
            reason = "RUN_MANIFEST_MISSING"
        else:
            reason = "EXECUTOR_BOOTSTRAP"

    before_consumed = {
        path.relative_to(staging).as_posix(): (path.stat().st_size, sha256_file(path))
        for path in (stdout_path, stderr_path)
    }
    after = inventory(staging)
    if before != after:
        raise D1Error("EVIDENCE_MUTATED")
    after_consumed = {
        path.relative_to(staging).as_posix(): (path.stat().st_size, sha256_file(path))
        for path in (stdout_path, stderr_path)
    }
    if before_consumed != after_consumed:
        raise D1Error("CONSUMED_EVIDENCE_MUTATED")

    mask = masks[0] if len(masks) == 1 else ("MULTI" if masks else "NA")
    capsule = (
        "G2D1 "
        f"N-O{stdout_nonempty}-E{stderr_nonempty}-T{traceback}-R{len(run_dirs)}-"
        f"J{len(manifests)}-S{mask}-MS{counts['S']}-MF{counts['F']}-"
        f"MK{counts['K']}-MP{counts['P']}-A{artifacts}-D{len(diag)}-X0 "
        f"C-0 K-{safe_token(reason)}"
    )
    memo = None
    if find_reason(stdout_text, stderr_text) is None and not diag:
        memo_source = last_nonempty(stderr_text) or last_nonempty(stdout_text)
        memo = "M1-" + sanitize_memo(memo_source)
    return capsule, memo


def selftest() -> None:
    checks = 0

    def ck(value: bool) -> None:
        nonlocal checks
        if not value:
            raise AssertionError("SELFTEST")
        checks += 1

    with tempfile.TemporaryDirectory() as temporary:
        staging = Path(temporary) / "G2_TEST.staging"
        logs = staging / "logs"
        run_dir = staging / "runs" / "RUN_TEST"
        diagnostics = run_dir / "02_DIAGNOSTICS"
        diagnostics.mkdir(parents=True)
        logs.mkdir(parents=True)
        (logs / "executor.stdout.txt").write_text(
            "U21F2BX N-X1 C-7 K-MODULE_RESULT\n", encoding="utf-8"
        )
        (logs / "executor.stderr.txt").write_text("", encoding="utf-8")
        manifest = {
            "RunStatus": "FAILED",
            "Stages": [
                {"Stage": "S0", "Status": "SUCCEEDED"},
                {"Stage": "S1", "Status": "FAILED"},
            ] + [{"Stage": f"S{i}", "Status": "PENDING"} for i in range(2, 8)],
            "ResolvedModules": [
                {"ModuleId": "M1", "Status": "FAILED"},
                {"ModuleId": "M2", "Status": "PENDING"},
            ],
            "Artifacts": [],
        }
        (run_dir / "run_manifest.json").write_text(
            json.dumps(manifest), encoding="utf-8"
        )
        (diagnostics / "failure.json").write_text(
            json.dumps({"FailureReason": "MODULE_RESULT"}), encoding="utf-8"
        )
        before = inventory(staging)
        capsule, memo = diagnose(staging)
        after = inventory(staging)
        ck(capsule.startswith("G2D1 "))
        ck("O1-E0-T0-R1-J1" in capsule)
        ck("SSFP" in capsule)
        ck("MF1" in capsule)
        ck("MP1" in capsule)
        ck("A0" in capsule)
        ck("D1" in capsule)
        ck(capsule.endswith("C-0 K-MODULE_RESULT"))
        ck(memo is None)
        ck(before == after)
        ck((logs / "executor.stdout.txt").is_file())
        ck((run_dir / "run_manifest.json").is_file())
        ck(len(before) == 4)
        ck(safe_token("module result") == "MODULE_RESULT")
        ck(stage_mask(manifest) == "SFPPPPPP")
        ck(module_counts(manifest)["F"] == 1)
    if checks != 16:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_CAPSULE)


def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--diagnose", action="store_true")
    parser.add_argument("--g2-staging", type=Path)
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            if args.g2_staging is not None:
                raise D1Error("ARGUMENTS")
            selftest()
            return 0
        if args.g2_staging is None:
            raise D1Error("ARGUMENTS")
        capsule, memo = diagnose(args.g2_staging)
        print(capsule)
        if memo:
            print(memo)
        return 0
    except D1Error as exc:
        print(f"G2D1 N-O0-E0-T0-R0-J0-SNA-MS0-MF0-MK0-MP0-A0-D0-X1 C-7 K-{exc.reason}")
        return 7
    except Exception:
        print("G2D1 N-O0-E0-T0-R0-J0-SNA-MS0-MF0-MK0-MP0-A0-D0-X1 C-7 K-INTERNAL")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
