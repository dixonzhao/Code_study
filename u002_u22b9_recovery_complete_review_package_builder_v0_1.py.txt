class B9Error(RuntimeError):
    """Represent one deterministic fail-closed B9 error."""


def machine_token(value: object, limit: int = 120) -> str:
    token = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (token or "UNKNOWN")[:limit]


def error_reason(error: BaseException) -> str:
    return str(error.args[0]) if error.args else type(error).__name__


def canonical_bytes(value: object) -> bytes:
    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def write_json(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def load_json(path: Path, reason: str) -> Dict[str, object]:
    if not path.is_file() or path.is_symlink():
        raise B9Error(reason + "_MISSING")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise B9Error(reason + "_JSON") from error
    if not isinstance(value, dict):
        raise B9Error(reason + "_TYPE")
    return value


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def parse_optional_float(value: object) -> Optional[float]:
    text = str(value if value is not None else "").strip()
    if not text:
        return None
    try:
        number = float(text)
    except ValueError:
        return None
    return number if math.isfinite(number) else None


def parse_integer(value: object, field: str) -> int:
    number = parse_optional_float(value)
    if number is None or not number.is_integer():
        raise B9Error("INTEGER_" + machine_token(field))
    return int(number)


def parse_boolean(value: object, field: str) -> bool:
    text = str(value if value is not None else "").strip().lower()
    if text in {"1", "true"}:
        return True
    if text in {"0", "false", ""}:
        return False
    raise B9Error("BOOLEAN_" + machine_token(field))


def write_csv(
    path: Path,
    columns: Sequence[str],
    rows: Iterable[Mapping[str, object]],
) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(
            stream,
            fieldnames=list(columns),
            lineterminator="\n",
        )
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in columns})


def verify_b8_package(package: Path) -> Tuple[Dict[str, str], Dict[str, object]]:
    """Verify B8 payload hashes and return the single accepted queue row."""

    if not package.is_dir() or package.is_symlink():
        raise B9Error("B8_PACKAGE_MISSING")
    artifact = load_json(package / "artifact_manifest.json", "B8_ARTIFACT")
    hashes = artifact.get("PayloadHashes")
    if not isinstance(hashes, dict):
        raise B9Error("B8_PAYLOAD_HASHES")
    for name, expected in hashes.items():
        path = package / str(name)
        if not path.is_file() or path.is_symlink():
            raise B9Error("B8_PAYLOAD_MISSING_" + machine_token(name))
        if sha256_file(path) != str(expected):
            raise B9Error("B8_PAYLOAD_HASH_" + machine_token(name))
    validation = load_json(
        package / "review_package_validation.json",
        "B8_VALIDATION",
    )
    if validation.get("GateStatus") != "PASS":
        raise B9Error("B8_VALIDATION_NOT_PASS")
    queue_path = package / "review_queue.csv"
    with queue_path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        missing = [column for column in QUEUE_REQUIRED if column not in (reader.fieldnames or [])]
        if missing:
            raise B9Error("B8_QUEUE_COLUMNS_" + machine_token("_".join(missing)))
        rows = [dict(row) for row in reader]
    if len(rows) != 1:
        raise B9Error("B8_CANDIDATE_COUNT_NOT_1")
    if str(rows[0]["RootCauseClaim"]).strip().lower() not in {"0", "false"}:
        raise B9Error("B8_ROOT_CAUSE_CLAIM")
    return rows[0], artifact


def verify_b7a_profile(profile: Path) -> Path:
    """Verify the immutable B7A row payload against its quality receipt."""

    row_path = profile / "b7a_camera_lateral_evidence_rows.csv"
    quality_path = profile / "b7a_camera_lateral_evidence_quality.json"
    capsule_path = profile / "b7a_return_capsule.txt"
    if not row_path.is_file() or row_path.is_symlink():
        raise B9Error("B7A_ROWS_MISSING")
    quality = load_json(quality_path, "B7A_QUALITY")
    hashes = quality.get("OutputHashes")
    if not isinstance(hashes, dict):
        raise B9Error("B7A_OUTPUT_HASHES")
    if str(hashes.get(row_path.name, "")) != sha256_file(row_path):
        raise B9Error("B7A_ROWS_HASH")
    if not capsule_path.is_file():
        raise B9Error("B7A_CAPSULE_MISSING")
    capsule = capsule_path.read_text(encoding="utf-8-sig").strip()
    if not capsule.startswith("B7A ") or not capsule.endswith("C-0"):
        raise B9Error("B7A_NOT_ACCEPTED")
    return row_path


def read_b7a_rows(path: Path) -> List[Dict[str, object]]:
    """Read strictly increasing B7A evidence rows."""

    rows: List[Dict[str, object]] = []
    previous: Optional[int] = None
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        missing = [column for column in B7A_COLUMNS if column not in (reader.fieldnames or [])]
        if missing:
            raise B9Error("B7A_COLUMNS_" + machine_token("_".join(missing[:3])))
        for source in reader:
            frame = parse_integer(source["FrameCount"], "FRAMECOUNT")
            if previous is not None and frame <= previous:
                raise B9Error("FRAMECOUNT_ORDER")
            previous = frame
            row: Dict[str, object] = dict(source)
            row["FrameCount"] = frame
            rows.append(row)
    if not rows:
        raise B9Error("B7A_ROWS_EMPTY")
    return rows


def radar_usable(row: Mapping[str, object]) -> Tuple[bool, str]:
    """Repeat the accepted B8 TC001 Radar usability rule exactly."""

    state = parse_boolean(row["RadarDistanceValid"], "RADAR_DISTANCE_VALID")
    distance = parse_optional_float(row["RadarDistance"])
    flag = parse_optional_float(row["RadarDetectionFlag"])
    if not state:
        return False, "DISTANCE_STATE_INVALID"
    if distance is None:
        return False, "DISTANCE_MISSING"
    if distance <= RADAR_MINIMUM_DISTANCE_METERS:
        return False, "DISTANCE_NONPOSITIVE"
    if distance >= RADAR_MAXIMUM_DISTANCE_METERS:
        return False, "DISTANCE_SENTINEL_OR_OUT_OF_RANGE"
    if flag is None:
        return False, "DETECTION_FLAG_MISSING"
    if flag <= 0:
        return False, "DETECTION_FLAG_OFF"
    return True, "USABLE"


def annotate(rows: Sequence[Mapping[str, object]]) -> List[Dict[str, object]]:
    output: List[Dict[str, object]] = []
    for source in rows:
        usable, reason = radar_usable(source)
        row = dict(source)
        row["RadarTargetUsable"] = int(usable)
        row["RadarLossReason"] = reason
        output.append(row)
    return output


def first_stable_start(
    rows: Sequence[Mapping[str, object]],
    after_frame: int,
    predicate,
) -> Optional[int]:
    """Return the first start of a consecutive stable recovery run."""

    run_start: Optional[int] = None
    run_length = 0
    previous: Optional[int] = None
    for row in rows:
        frame = int(row["FrameCount"])
        if frame <= after_frame:
            continue
        passes = bool(predicate(row))
        consecutive = previous is not None and frame == previous + 1
        if passes:
            if run_start is None or not consecutive:
                run_start = frame
                run_length = 1
            else:
                run_length += 1
            if run_length >= STABLE_RECOVERY_FRAMES:
                return run_start
        else:
            run_start = None
            run_length = 0
        previous = frame
    return None


def derive_recovery(
    rows: Sequence[Mapping[str, object]],
    evidence_end: int,
) -> Dict[str, object]:
    """Locate stable recovery of Radar, any raw Camera, and old canonical data."""

    radar = first_stable_start(
        rows,
        evidence_end,
        lambda row: bool(int(row["RadarTargetUsable"])),
    )
    camera = first_stable_start(
        rows,
        evidence_end,
        lambda row: parse_boolean(row["Camera1DistanceValid"], "CAMERA1_VALID")
        or parse_boolean(row["Camera3DistanceValid"], "CAMERA3_VALID"),
    )
    canonical = first_stable_start(
        rows,
        evidence_end,
        lambda row: parse_boolean(
            row["OldCanonicalCameraValid"],
            "OLD_CANONICAL_VALID",
        ),
    )
    missing = [
        name
        for name, value in (
            ("RADAR", radar),
            ("RAW_CAMERA", camera),
            ("CANONICAL", canonical),
        )
        if value is None
    ]
    return {
        "RadarStableRecoveryFrame": radar,
        "RawCameraStableRecoveryFrame": camera,
        "CanonicalStableRecoveryFrame": canonical,
        "RecoveryCoverageStatus": (
            "RECOVERY_COMPLETE" if not missing else "NOT_RECOVERED_" + "_".join(missing)
        ),
    }


def recovery_phase(
    frame: int,
    evidence_start: int,
    evidence_end: int,
    recovery: Mapping[str, object],
) -> str:
    """Label the human-review phase without creating a causal claim."""

    if frame < evidence_start:
        return "PRE_LOSS_CONTEXT"
    if frame <= evidence_end:
        return "RADAR_UNUSABLE_EVIDENCE"
    radar = recovery["RadarStableRecoveryFrame"]
    camera = recovery["RawCameraStableRecoveryFrame"]
    canonical = recovery["CanonicalStableRecoveryFrame"]
    if radar is None or frame < int(radar):
        return "RADAR_RECOVERY_PENDING"
    if camera is None or frame < int(camera):
        return "RAW_CAMERA_RECOVERY_PENDING"
    if canonical is None or frame < int(canonical):
        return "CANONICAL_RECOVERY_PENDING"
    return "RECOVERY_COMPLETE_CONTEXT"


def build_package(
    b8_package: Path,
    b7a_profile: Path,
    output_dir: Path,
) -> str:
    """Build the recovery-complete U003 package."""

    queue, b8_artifact = verify_b8_package(b8_package)
    row_path = verify_b7a_profile(b7a_profile)
    source_rows = annotate(read_b7a_rows(row_path))
    source_start = int(source_rows[0]["FrameCount"])
    source_end = int(source_rows[-1]["FrameCount"])
    review_start = parse_integer(queue["StartFrameCount"], "START_FRAME")
    old_review_end = parse_integer(queue["EndFrameCount"], "END_FRAME")
    evidence_start = parse_integer(queue["EvidenceStartFrame"], "EVIDENCE_START")
    evidence_end = parse_integer(queue["EvidenceEndFrame"], "EVIDENCE_END")
    if not source_start <= review_start <= evidence_start <= evidence_end <= source_end:
        raise B9Error("B8_RANGE_OUTSIDE_B7A")

    recovery = derive_recovery(source_rows, evidence_end)
    recovery_values = [
        recovery["RadarStableRecoveryFrame"],
        recovery["RawCameraStableRecoveryFrame"],
        recovery["CanonicalStableRecoveryFrame"],
    ]
    if any(value is None for value in recovery_values):
        review_end = source_end
    else:
        latest = max(int(value) for value in recovery_values if value is not None)
        review_end = min(
            source_end,
            max(old_review_end, latest + POST_RECOVERY_CONTEXT_FRAMES),
        )
    if review_end <= old_review_end:
        raise B9Error("REVIEW_WINDOW_NOT_EXTENDED")

    new_queue = dict(queue)
    new_queue.update(
        {
            "EndFrameCount": review_end,
            "ReviewResult": "",
            "ReviewMemo": "",
            "SuggestedReviewMemo": (
                "Review Radar loss onset and delayed recovery, then continue until "
                "raw Camera and canonical recovery can be evaluated."
            ),
            "SuggestedReviewTags": (
                "RADAR_TARGET_LOSS;RECOVERY_COMPLETENESS;CAMERA_CANONICAL_REVIEW"
            ),
            "ModuleId": PROGRAM_ID,
            "ModuleVersion": PROGRAM_VERSION,
            "AlgorithmStatus": "PROVISIONAL_RECOVERY_COVERAGE_ADAPTER",
            "OriginalReviewEndFrame": old_review_end,
            **{
                key: "" if value is None else value
                for key, value in recovery.items()
            },
        }
    )

    review_rows: List[Dict[str, object]] = []
    detail_rows: List[Dict[str, object]] = []
    for source in source_rows:
        frame = int(source["FrameCount"])
        if frame < review_start or frame > review_end:
            continue
        phase = recovery_phase(
            frame,
            evidence_start,
            evidence_end,
            recovery,
        )
        review_rows.append(
            {
                "FrameCount": frame,
                "CandidateIds": queue["CandidateId"],
                "RecoveryPhase": phase,
                "RadarDistance": source["RadarDistance"],
                "RadarDistanceValid": source["RadarDistanceValid"],
                "RadarDetectionFlag": source["RadarDetectionFlag"],
                "RadarTargetUsable": source["RadarTargetUsable"],
                "RadarLossReason": source["RadarLossReason"],
                "Camera1Distance": source["Camera1Distance"],
                "Camera1DistanceValid": source["Camera1DistanceValid"],
                "Camera1LateralPosition": source["Camera1LateralPosition"],
                "Camera3Distance": source["Camera3Distance"],
                "Camera3DistanceValid": source["Camera3DistanceValid"],
                "Camera3LateralPosition": source["Camera3LateralPosition"],
                "OldCanonicalCameraDistance": source["OldCanonicalCameraDistance"],
                "OldCanonicalCameraValid": source["OldCanonicalCameraValid"],
                "OldSelectedCameraRole": source["OldSelectedCameraRole"],
                "CenterlinePreferredRole": source["CenterlinePreferredRole"],
            }
        )
        detail_rows.append(
            {
                "CandidateIds": queue["CandidateId"],
                "RecoveryPhase": phase,
                **dict(source),
            }
        )
    if not review_rows:
        raise B9Error("REVIEW_ROWS_EMPTY")
    if int(review_rows[0]["FrameCount"]) != review_start:
        raise B9Error("REVIEW_START_COVERAGE")
    if int(review_rows[-1]["FrameCount"]) != review_end:
        raise B9Error("REVIEW_END_COVERAGE")

    if output_dir.exists():
        shutil.rmtree(output_dir)
    package = output_dir / "U003_REVIEW_PACKAGE_RECOVERY_COMPLETE"
    package.mkdir(parents=True)
    queue_path = package / "review_queue.csv"
    data_path = package / "review_data.csv"
    detail_path = package / "review_evidence_detail.csv"
    validation_path = package / "review_package_validation.json"
    manifest_path = package / "review_manifest.json"
    artifact_path = package / "artifact_manifest.json"
    write_csv(queue_path, RECOVERY_QUEUE_COLUMNS, [new_queue])
    write_csv(data_path, REVIEW_COLUMNS, review_rows)
    detail_columns = ("CandidateIds", "RecoveryPhase") + B7A_COLUMNS + (
        "RadarTargetUsable",
        "RadarLossReason",
    )
    write_csv(detail_path, detail_columns, detail_rows)

    validation = {
        "GateStatus": "PASS",
        "CandidateCount": 1,
        "ReviewRowCount": len(review_rows),
        "OriginalReviewEndFrame": old_review_end,
        "ExtendedReviewEndFrame": review_end,
        **recovery,
        "RootCauseClaim": False,
    }
    write_json(validation_path, validation)
    manifest = {
        "ArtifactType": "U003_REVIEW_PACKAGE_RECOVERY_COMPLETE",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "FrameDomain": "FrameCount",
        "ReviewStartFrame": review_start,
        "ReviewEndFrame": review_end,
        "EvidenceStartFrame": evidence_start,
        "EvidenceEndFrame": evidence_end,
        "StableRecoveryFrames": STABLE_RECOVERY_FRAMES,
        "PostRecoveryContextFrames": POST_RECOVERY_CONTEXT_FRAMES,
        **recovery,
        "DecisionBoundary": (
            "REVIEW_COVERAGE_CORRECTION_ONLY_NO_SELECTOR_OR_ROOT_CAUSE_CHANGE"
        ),
        "SourceHashes": {
            "B8ArtifactManifest": sha256_file(b8_package / "artifact_manifest.json"),
            "B7ARows": sha256_file(row_path),
        },
        "SourceB8ArtifactId": b8_artifact.get("ArtifactId"),
    }
    write_json(manifest_path, manifest)
    payload_hashes = {
        path.name: sha256_file(path)
        for path in (queue_path, data_path, detail_path, validation_path, manifest_path)
    }
    write_json(
        artifact_path,
        {
            "ArtifactId": "U22B9.RECOVERY_COMPLETE.U003_REVIEW_PACKAGE.1",
            "ArtifactType": "U003_REVIEW_PACKAGE_RECOVERY_COMPLETE",
            "ProducedBy": PROGRAM_ID,
            "PayloadHashes": payload_hashes,
            "RootCauseClaim": False,
            "CompletionCode": 0,
        },
    )
    capsule = (
        f"B9 N-C1-F{len(review_rows)}"
        f"-R{recovery['RadarStableRecoveryFrame'] or 0}"
        f"-A{recovery['RawCameraStableRecoveryFrame'] or 0}"
        f"-K{recovery['CanonicalStableRecoveryFrame'] or 0}"
        f"-E{review_end}-X0 C-0"
    )
    (output_dir / "u22b9_return_capsule.txt").write_text(
        capsule + "\n",
        encoding="utf-8",
    )
    return capsule


def prepare_run(temp_root: Path) -> str:
    """Archive stale TEMP entries while preserving the two accepted inputs."""

    if not temp_root.is_dir():
        raise B9Error("TEMP_ROOT_MISSING")
    keep = {
        "A-Archive",
        "U22B7U_OUTPUT",
        "U22B8_OUTPUT",
        "UDP002_U22B9_START_HERE_v0_1_2026-08-24.txt",
        "u002_u22b9_recovery_complete_review_package_builder_v0_1.py.txt",
    }
    archive = temp_root / "A-Archive"
    archive.mkdir(exist_ok=True)
    stamp = __import__("datetime").datetime.utcnow().strftime("%Y%m%dT%H%M%SZ")
    destination = archive / ("U22B9_PREPARE_" + stamp)
    moved: List[str] = []
    for path in sorted(temp_root.iterdir(), key=lambda item: item.name.casefold()):
        if path.name in keep:
            continue
        destination.mkdir(parents=True, exist_ok=True)
        shutil.move(str(path), str(destination / path.name))
        moved.append(path.name)
    if moved:
        write_json(
            destination / "u22b9_temp_cleanup_manifest.json",
            {"Moved": moved, "Kept": sorted(keep), "DeletionCount": 0},
        )
    return f"B9CLEAN N-M{len(moved)}-K{len(keep)}-A{int(bool(moved))}-X0 C-0"


def runtime_preflight(b8_package: Path, b7a_profile: Path) -> str:
    required = (
        b8_package / "review_queue.csv",
        b8_package / "review_data.csv",
        b8_package / "review_evidence_detail.csv",
        b8_package / "review_package_validation.json",
        b8_package / "review_manifest.json",
        b8_package / "artifact_manifest.json",
        b7a_profile / "b7a_camera_lateral_evidence_rows.csv",
        b7a_profile / "b7a_camera_lateral_evidence_quality.json",
        b7a_profile / "b7a_return_capsule.txt",
    )
    passed = sum(path.is_file() and not path.is_symlink() for path in required)
    missing = len(required) - passed
    return f"B9PF N-E{len(required)}-P{passed}-M{missing}-X{int(bool(missing))} C-{7 if missing else 0}"


def make_fixture(root: Path, canonical_recovers: bool = True) -> Tuple[Path, Path]:
    """Create accepted B8 and B7A fixtures for deterministic self-tests."""

    profile = root / "B7A_PROFILE_V03"
    profile.mkdir(parents=True)
    rows: List[Dict[str, object]] = []
    for frame in range(1000, 1101):
        radar_ok = frame < 1010 or frame >= 1021
        camera_ok = frame < 1010 or frame >= 1040
        canonical_ok = frame < 1010 or (canonical_recovers and frame >= 1060)
        row = {column: "" for column in B7A_COLUMNS}
        row.update(
            {
                "FrameCount": frame,
                "Camera1Distance": "15" if camera_ok else "",
                "Camera1DistanceValid": int(camera_ok),
                "Camera1LateralPosition": "0.2" if camera_ok else "",
                "Camera1LateralValid": int(camera_ok),
                "Camera3Distance": "20" if camera_ok else "",
                "Camera3DistanceValid": int(camera_ok),
                "Camera3LateralPosition": "2.0" if camera_ok else "",
                "Camera3LateralValid": int(camera_ok),

def self_test() -> None:
    checks = 0
    with tempfile.TemporaryDirectory(prefix="u22b9_selftest_") as temporary:
        root = Path(temporary)
        package, profile = make_fixture(root)
        capsule = build_package(package, profile, root / "OUT")
        checks += 1
        if not capsule.startswith("B9 N-C1-F101-R1021-A1040-K1060-E1100"):
            raise B9Error("SELFTEST_CAPSULE")
        output_package = root / "OUT/U003_REVIEW_PACKAGE_RECOVERY_COMPLETE"
        for name in (
            "review_queue.csv",
            "review_data.csv",
            "review_evidence_detail.csv",
            "review_package_validation.json",
            "review_manifest.json",
            "artifact_manifest.json",
        ):
            checks += 1
            if not (output_package / name).is_file():
                raise B9Error("SELFTEST_FILE_" + machine_token(name))
        validation = load_json(
            output_package / "review_package_validation.json",
            "SELFTEST_VALIDATION",
        )
        checks += 1
        if validation.get("GateStatus") != "PASS":
            raise B9Error("SELFTEST_VALIDATION")
        checks += 1
        if validation.get("OriginalReviewEndFrame") != 1030:
            raise B9Error("SELFTEST_OLD_END")
        checks += 1
        if validation.get("ExtendedReviewEndFrame") != 1100:
            raise B9Error("SELFTEST_NEW_END")

        second_root = root / "MISSING_CANONICAL"
        package2, profile2 = make_fixture(second_root, canonical_recovers=False)
        capsule2 = build_package(package2, profile2, second_root / "OUT")
        checks += 1
        if "-K0-E1100-" not in capsule2:
            raise B9Error("SELFTEST_MISSING_CANONICAL")
    print(f"UDP002-U22B9-SELFTEST-PASS-{checks} C-0")


def write_failure(output_dir: Path, stage: str, error: BaseException) -> None:
    reason = machine_token(error_reason(error))
    diagnostic = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "Stage": stage,
        "ErrorType": type(error).__name__,
        "Reason": error_reason(error),
        "CompletionCode": 7,
        "RequiredB8Package": str(DEFAULT_B8_PACKAGE),
        "RequiredB7AProfile": str(DEFAULT_B7A_PROFILE),
        "Recovery": (
            "Keep U22B7U_OUTPUT and U22B8_OUTPUT in 04_TEMP, run --self-test, "
            "then retry --execute without editing accepted inputs."
        ),
    }
    try:
        output_dir.mkdir(parents=True, exist_ok=True)
        write_json(output_dir / "u22b9_failure_diagnostic.json", diagnostic)
    except Exception:
        pass
    print(f"B9E K-{reason}-J{machine_token(stage, 24)}-X1 C-7")
    print(f"B9D K-{reason}-D{type(error).__name__}-X0 C-0")


def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare-run", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--preflight-only", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--b8-package", type=Path, default=DEFAULT_B8_PACKAGE)
    parser.add_argument("--b7a-profile", type=Path, default=DEFAULT_B7A_PROFILE)
    parser.add_argument("--output-dir", type=Path, default=DEFAULT_OUTPUT_DIR)
    parser.add_argument("--temp-root", type=Path, default=DEFAULT_TEMP_ROOT)
    args = parser.parse_args(argv)
    stage = "START"
    try:
        if args.prepare_run:
            stage = "PREPARE"
            print(prepare_run(args.temp_root))
            return 0
        if args.self_test:
            stage = "SELFTEST"
            self_test()
            summary = runtime_preflight(args.b8_package, args.b7a_profile)
            print(summary)
            return 0 if summary.endswith("C-0") else 7
        if args.preflight_only:
            stage = "PREFLIGHT"
            summary = runtime_preflight(args.b8_package, args.b7a_profile)
            print(summary)
            return 0 if summary.endswith("C-0") else 7
        stage = "PREFLIGHT"
        if not runtime_preflight(args.b8_package, args.b7a_profile).endswith("C-0"):
            raise B9Error("RUNTIME_PREFLIGHT")
        stage = "EXECUTE"
        print(build_package(args.b8_package, args.b7a_profile, args.output_dir))
        return 0
    except Exception as error:
        write_failure(args.output_dir, stage, error)
        return 7


if __name__ == "__main__":
    sys.exit(main())
