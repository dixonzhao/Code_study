#!/usr/bin/env python3
"""Safely move dependency-proven unreferenced U001 extras into ARCH."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from datetime import datetime, timezone
from pathlib import Path


ALLOWED_CATEGORIES = {"VALIDATION", "PACKAGING", "DEPRECATED_VERSION"}


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while True:
            block = stream.read(1024 * 1024)
            if not block:
                break
            digest.update(block)
    return digest.hexdigest()


def fail(code: str) -> int:
    print(f"G10D1A1-ERROR:{code} C-7")
    return 7


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--archive", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--dependency-report")
    parser.add_argument("--archive-name")
    args = parser.parse_args()
    if args.self_test:
        assert Path("a.py").name == "a.py"
        assert "OTHER" not in ALLOWED_CATEGORIES
        assert len(ALLOWED_CATEGORIES) == 3
        print("U001-G10D1A1-SELFTEST-PASS-8")
        return 0
    if not args.archive or not args.udp001_root or not args.dependency_report or not args.archive_name:
        parser.error("requires --archive --udp001-root --dependency-report --archive-name")
    if Path(args.archive_name).name != args.archive_name or not args.archive_name.startswith("G10D_"):
        return fail("ARCHIVE_NAME")

    root = Path(args.udp001_root).resolve()
    program_root = root / "01_PROGRAMS"
    report_path = Path(args.dependency_report).resolve()
    archive = root / "ARCH" / args.archive_name
    if not program_root.is_dir() or not report_path.is_file():
        return fail("INPUT")
    if archive.exists():
        return fail("ARCHIVE_EXISTS")

    report = json.loads(report_path.read_text(encoding="utf-8-sig"))
    if report.get("Schema") != "UDP001_G10D1D1_UNEXPECTED_DEPENDENCY_V0_1":
        return fail("REPORT_SCHEMA")
    rows = report.get("UnexpectedPrograms", [])
    if not rows or report.get("ReferencedCount") != 0 or report.get("ImportedCount") != 0:
        return fail("DEPENDENCY_BLOCK")
    if report.get("BlockingCount") != 0:
        return fail("REPORT_BLOCK")

    plan = []
    for row in rows:
        name = row.get("Name", "")
        if not name or Path(name).name != name or not name.endswith(".py"):
            return fail("UNSAFE_NAME")
        if row.get("Category") not in ALLOWED_CATEGORIES:
            return fail("UNAPPROVED_CATEGORY")
        if row.get("ReferencedByExpectedPrograms") or row.get("ImportedByExpectedPrograms"):
            return fail("ROW_DEPENDENCY")
        source = program_root / name
        target = archive / name
        if not source.is_file():
            return fail("SOURCE_MISSING")
        plan.append({
            "Name": name,
            "Category": row["Category"],
            "Source": source,
            "Target": target,
            "Bytes": source.stat().st_size,
            "SHA256": sha256(source),
        })

    archive.mkdir(parents=True, exist_ok=False)
    manifest_path = archive / "u001_g10d1a1_archive_manifest.json"
    manifest = {
        "Schema": "UDP001_G10D1A1_ARCHIVE_MANIFEST_V0_1",
        "Status": "PREPARED",
        "CreatedUTC": datetime.now(timezone.utc).isoformat(),
        "DependencyReport": str(report_path),
        "Moves": [
            {key: value for key, value in item.items() if key not in {"Source", "Target"}}
            | {"From": str(item["Source"]), "To": str(item["Target"])}
            for item in plan
        ],
    }
    manifest_path.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")

    moved = []
    try:
        for item in plan:
            os.replace(item["Source"], item["Target"])
            moved.append(item)
    except Exception:
        manifest["Status"] = "PARTIAL_FAILURE"
        manifest["MovedCount"] = len(moved)
        manifest_path.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")
        return fail("MOVE_FAILURE")

    for item in plan:
        if item["Source"].exists() or not item["Target"].is_file():
            return fail("POST_MOVE_LOCATION")
        if sha256(item["Target"]) != item["SHA256"]:
            return fail("POST_MOVE_HASH")

    manifest["Status"] = "COMPLETE"
    manifest["CompletedUTC"] = datetime.now(timezone.utc).isoformat()
    manifest["MovedCount"] = len(plan)
    manifest_path.write_text(json.dumps(manifest, indent=2, ensure_ascii=False), encoding="utf-8")
    categories = {name: 0 for name in sorted(ALLOWED_CATEGORIES)}
    for item in plan:
        categories[item["Category"]] += 1
    print(
        f"G10D1A1 N-E{len(plan)}-M{len(plan)}-V{categories['VALIDATION']}-"
        f"P{categories['PACKAGING']}-D{categories['DEPRECATED_VERSION']}-X0 C-0"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
