TRANSFER_NAME = "u002_u22b20c_gui_config_equivalence_gate_v0_1.py.txt"

MODEL_RELATIVE = Path("01_PROGRAMS/gui/u002_thin_workbench_model_v0_4.py")

REPORT_RELATIVE = Path("07_OUTPUT") / OUTPUT_ID / "02_DIAGNOSTICS" / (

SEMANTIC_FIELDS: Tuple[str, ...] = (

    "RunPurpose",
    "RunScope",

    "SignalProfile",

class B20CError(RuntimeError):
    """Controlled rejection with one short machine-readable reason."""

    def __init__(self, reason: object, detail: object = "NA") -> None:
        self.reason = token(reason)
        self.detail = detail_token(detail)
        super().__init__(f"{self.reason}: {self.detail}")


def token(value: object) -> str:
    """Convert one value to a short return-capsule token."""

    text = str(value).strip().upper()
    result = "".join(character if character.isalnum() else "_" for character in text)
    return result.strip("_") or "NA"


def detail_token(value: object, limit: int = 160) -> str:
    """Bound failure details so manual return remains practical."""

    result = token(value)
    return result[:limit] if result else "NA"


def canonical_bytes(value: object) -> bytes:
    """Serialize evidence deterministically for equality and hashing."""

    return (json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n").encode(
        "utf-8"
    )


def sha256(value: object) -> str:
    """Hash a canonical JSON-compatible value."""

    return hashlib.sha256(canonical_bytes(value)).hexdigest()


def read_object(path: Path, label: str) -> Dict[str, Any]:
    """Read one required regular JSON object."""

    if not path.is_file() or path.is_symlink():
        raise B20CError(label + "_MISSING", path)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as error:
        raise B20CError(label + "_JSON", error) from error
    if not isinstance(value, dict):
        raise B20CError(label + "_TYPE", type(value).__name__)
    return value


def write_object(path: Path, value: Mapping[str, object]) -> None:
    """Atomically write one controlled JSON evidence object."""

    path.parent.mkdir(parents=True, exist_ok=True)
    writing = path.with_name(path.name + ".writing")
    if writing.exists():
        raise B20CError("WRITING_EXISTS", writing)
    writing.write_bytes(canonical_bytes(value))
    os.replace(writing, path)


def load_module(path: Path, name: str):
    """Load one installed Python program without renaming it."""

    if not path.is_file() or path.is_symlink():
        raise B20CError("MODEL_MISSING", path)
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None:
        raise B20CError("MODEL_IMPORT_SPEC", path.name)
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module


def workspace_path(root: Path, value: object, label: str) -> Path:
    """Resolve one workspace-relative path without permitting escape."""

    if not isinstance(value, str) or not value.strip():
        raise B20CError(label + "_TYPE", value)
    candidate = Path(value.strip())
    resolved = candidate.resolve() if candidate.is_absolute() else (root / candidate).resolve()
    try:
        resolved.relative_to(root.resolve())
    except ValueError as error:
        raise B20CError(label + "_OUTSIDE_WORKSPACE", resolved) from error
    return resolved


def source_paths(root: Path, profile: Mapping[str, object]) -> Dict[str, Path]:
    """Resolve the accepted Golden source referenced by the GUI Profile."""

    run_id = profile.get("SourceGoldenRunId")
    if not isinstance(run_id, str) or not run_id.strip():
        raise B20CError("SOURCE_GOLDEN_RUN_ID", run_id)
    run_root = root / "07_OUTPUT" / run_id
    return {
        "RunRoot": run_root,
        "Status": run_root / "run_status.json",
        "Configuration": run_root / "00_CONFIG/run_configuration.json",
    }


def require_succeeded_source(root: Path, profile: Mapping[str, object]) -> Dict[str, Any]:
    """Load a succeeded source Golden Configuration anchored by the Profile."""

    paths = source_paths(root, profile)
    status = read_object(paths["Status"], "SOURCE_STATUS")
    if status.get("Status") != "SUCCEEDED":
        raise B20CError("SOURCE_NOT_SUCCEEDED", status.get("Status"))
    configuration = read_object(paths["Configuration"], "SOURCE_CONFIGURATION")
    expected_id = str(profile["SourceGoldenRunId"])
    if configuration.get("RunId") != expected_id:
        raise B20CError("SOURCE_RUN_ID", configuration.get("RunId"))
    return configuration


def semantic_projection(configuration: Mapping[str, object]) -> Dict[str, Any]:
    """Keep every analytical field and intentionally exclude RunId only."""

    missing = [field for field in SEMANTIC_FIELDS if field not in configuration]
    if missing:
        raise B20CError("SEMANTIC_FIELD_MISSING", ".".join(missing))
    return {field: copy.deepcopy(configuration[field]) for field in SEMANTIC_FIELDS}


def first_difference(reference: object, candidate: object, path: str = "$") -> str:
    """Return the first exact semantic difference as a compact JSON path."""

    if type(reference) is not type(candidate):
        return path + "@TYPE"
    if isinstance(reference, dict):
        reference_keys = set(reference)
        candidate_keys = set(candidate)
        if reference_keys != candidate_keys:
            changed = sorted(reference_keys.symmetric_difference(candidate_keys))[0]
            return path + "." + str(changed) + "@KEY"
        for key in sorted(reference_keys):
            difference = first_difference(reference[key], candidate[key], path + "." + str(key))
            if difference:
                return difference
        return ""
    if isinstance(reference, list):
        if len(reference) != len(candidate):
            return path + "@LENGTH"
        for index, (left, right) in enumerate(zip(reference, candidate)):
            difference = first_difference(left, right, f"{path}[{index}]")
            if difference:
                return difference
        return ""
    return "" if reference == candidate else path


def compare_configurations(
    reference: Mapping[str, object],
    candidate: Mapping[str, object],
) -> Dict[str, Any]:
    """Compare exact analytical semantics while documenting the allowed identity change."""

    reference_projection = semantic_projection(reference)
    candidate_projection = semantic_projection(candidate)
    difference = first_difference(reference_projection, candidate_projection)
    return {
        "Equivalent": difference == "",
        "FirstDifference": difference or None,
        "ComparedFields": list(SEMANTIC_FIELDS),
        "ExcludedIdentityFields": ["RunId"],
        "ReferenceRunId": reference.get("RunId"),
        "CandidateRunId": candidate.get("RunId"),
        "ReferenceProjectionSha256": sha256(reference_projection),
        "CandidateProjectionSha256": sha256(candidate_projection),
        "ReferenceConfigurationSha256": sha256(reference),
        "CandidateConfigurationSha256": sha256(candidate),
    }


def model_and_state(root: Path):
    """Load the exact GUI model and its untouched Default Qualified state."""

    model_module = load_module(root / MODEL_RELATIVE, "u002_b20c_workbench_model")
    model = model_module.ThinWorkbenchModel(root)
    state = model.new_gui_state()
    if state.get("ProfileMode") != "DEFAULT_QUALIFIED":
        raise B20CError("DEFAULT_PROFILE_MODE", state.get("ProfileMode"))
    state["RunId"] = RUN_ID
    return model_module, model, state


def install_exact(source: Path, destination: Path) -> str:
    """Install a versioned program or accept an identical replay."""

    data = source.read_bytes()
    if destination.exists():
        if not destination.is_file() or destination.is_symlink() or destination.read_bytes() != data:
            raise B20CError("INSTALL_CONFLICT", destination.name)
        return "REPLAY"
    destination.parent.mkdir(parents=True, exist_ok=True)
    writing = destination.with_name(destination.name + ".writing")
    if writing.exists():
        raise B20CError("INSTALL_WRITING_EXISTS", writing.name)
    writing.write_bytes(data)
    os.replace(writing, destination)
    return "CREATED"


def transfer_source(root: Path) -> Path:
    """Return the exact transfer program expected in 04_TEMP."""

    path = root / "04_TEMP" / TRANSFER_NAME
    if not path.is_file() or path.is_symlink():
        raise B20CError("TRANSFER_MISSING", path)
    return path


def archive_obsolete_temp(root: Path, keep: Sequence[str]) -> Tuple[int, int]:
    """Archive older B20C transfer artifacts after the current source is secured."""

    temp = root / "04_TEMP"
    stale = [
        path
        for path in temp.iterdir()
        if path.name not in keep
        and (
            path.name.startswith("u002_u22b20c_")
            or path.name.startswith("UDP002_U22B20C_")
        )
    ]
    if not stale:
        return 0, 0
    archive = temp / "A-Archive" / (
        "U22B20C_PREPARE_" + dt.datetime.now().strftime("%Y%m%d_%H%M%S_%f")
    )
    archive.mkdir(parents=True, exist_ok=False)
    moved = 0
    locked = 0
    for path in stale:
        try:
            shutil.move(str(path), str(archive / path.name))
            moved += 1
        except PermissionError:
            locked += 1
    return moved, locked


def prepare(root: Path) -> str:
    """Install B20C and verify that all comparison evidence is present."""

    source = transfer_source(root)
    destination = root / INSTALL_RELATIVE
    disposition = install_exact(source, destination)
    profile = read_object(root / PROFILE_RELATIVE, "PROFILE")
    golden = require_succeeded_source(root, profile)
    _module, model, state = model_and_state(root)
    candidate = model.materialize_run(state)
    comparison = compare_configurations(golden, candidate)
    if not comparison["Equivalent"]:
        raise B20CError("DEFAULT_NOT_EQUIVALENT", comparison["FirstDifference"])
    moved, locked = archive_obsolete_temp(
        root,
        keep=(TRANSFER_NAME, "UDP002_U22B20C_START_HERE_CN_v0_1_2026-08-26.txt"),
    )
    print("=== U22B20C PREPARE / GUI CONFIGURATION EQUIVALENCE ===")
    print("PASS | B20C installed as a versioned program | WHY: acceptance evidence must be replayable")
    print("PASS | GUI model v0.4 is present | WHY: comparison must use the same model called by the GUI")
    print("PASS | Default Profile anchors one succeeded Golden Run | WHY: comparison requires accepted reference semantics")
    print("PASS | Default state matches all 11 semantic fields | WHY: GUI convenience must not alter analysis")
    print(f"INFO | install={disposition} archived={moved} locked={locked}")
    capsule = f"B20CP-I1-M1-G1-N11-A{moved}-L{locked}-X0-C0"
    print(capsule)
    return capsule


def self_test(root: Path) -> str:
    """Exercise positive identity and broad negative semantic mutations without writing output."""

    profile = read_object(root / PROFILE_RELATIVE, "PROFILE")
    golden = require_succeeded_source(root, profile)
    _module, model, state = model_and_state(root)
    candidate = model.materialize_run(state)

    checks: List[bool] = []

    def check(value: bool, label: str) -> None:
        if not value:
            raise B20CError("SELFTEST_" + label)
        checks.append(True)

    positive = compare_configurations(golden, candidate)
    check(positive["Equivalent"] is True, "POSITIVE")
    check(positive["ReferenceRunId"] != positive["CandidateRunId"], "IDENTITY_DIFFERENCE")
    check(len(positive["ComparedFields"]) == 11, "FIELD_COUNT")
    check(len(candidate["ModulePlan"]) == 5, "MODULE_COUNT")
    check(len({item["Stage"] for item in candidate["ModulePlan"]}) == 4, "STAGE_COUNT")

    mutations = [
        ("RunPurpose", "SMOKE" if candidate["RunPurpose"] != "SMOKE" else "TUNING"),
        ("InputPackage", "03_CASES/WRONG_PACKAGE"),
        ("SignalProfile", "WRONG.PROFILE@9.9.9"),
        ("CandidatePolicy", "WRONG.POLICY@9.9.9"),
        ("OutputSelection", list(candidate["OutputSelection"])[:-1]),
        ("PrimaryOutcome", {**candidate["PrimaryOutcome"], "Unit": "wrong"}),
    ]
    for field, value in mutations:
        changed = copy.deepcopy(candidate)
        changed[field] = value
        check(not compare_configurations(golden, changed)["Equivalent"], "MUTATION_" + field)

    changed_scope = copy.deepcopy(candidate)
    changed_scope["RunScope"] = "FRAME_RANGE"
    changed_scope["FrameStart"] = 1
    changed_scope["FrameEnd"] = 2
    check(not compare_configurations(golden, changed_scope)["Equivalent"], "MUTATION_SCOPE")

    changed_plan = copy.deepcopy(candidate)
    changed_plan["ModulePlan"][1]["Configuration"]["MaxContinuityCost"] = 999.0
    plan_result = compare_configurations(golden, changed_plan)
    check(not plan_result["Equivalent"], "MUTATION_MODULE_CONFIG")
    check("ModulePlan" in str(plan_result["FirstDifference"]), "DIAGNOSTIC_PATH")

    identity_only = copy.deepcopy(candidate)
    identity_only["RunId"] = "ANOTHER_VALID_RUN_ID"
    check(compare_configurations(golden, identity_only)["Equivalent"], "RUN_ID_EXCLUSION")

    print("=== U22B20C SELF-TEST / SEMANTIC BOUNDARY ===")
    print("PASS | RunId is the only excluded field | WHY: new Runs require new identity without changing analysis")
    print("PASS | 11 semantic fields are exact | WHY: configuration equivalence is stronger than visual similarity")
    print("PASS | Scope input profile plan outcome policy and outputs fail on mutation | WHY: drift must fail closed")
    print("PASS | First-difference path is emitted | WHY: one failure should identify its owner without another probe")
    capsule = f"B20CST-N{len(checks)}-M5-S4-D1-X0-C0"
    print(capsule)
    return capsule


def execute(root: Path) -> str:
    """Save the Default GUI configuration and emit formal equivalence evidence."""

    profile = read_object(root / PROFILE_RELATIVE, "PROFILE")
    golden = require_succeeded_source(root, profile)
    _module, model, state = model_and_state(root)
    config_path, disposition = model.save_run(state)
    candidate = read_object(config_path, "GUI_CONFIGURATION")
    comparison = compare_configurations(golden, candidate)
    report = {
        "ContractId": "U002_GUI_CONFIG_EQUIVALENCE_REPORT",
        "ContractVersion": "0.1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "Status": "SUCCEEDED" if comparison["Equivalent"] else "FAILED",
        "GeneratedBy": "THIN_WORKBENCH_MODEL_SAVE_RUN",
        "SourceGoldenRunId": profile["SourceGoldenRunId"],
        "GuiRunConfigurationPath": str(config_path.relative_to(root)),
        "GuiRunConfigurationDisposition": disposition,
        "Comparison": comparison,
        "ModuleCount": len(candidate["ModulePlan"]),
        "OccupiedStageCount": len({item["Stage"] for item in candidate["ModulePlan"]}),
        "AnalyticalExecutionStarted": False,
    }
    write_object(root / REPORT_RELATIVE, report)
    if not comparison["Equivalent"]:
        raise B20CError("SEMANTIC_DIFFERENCE", comparison["FirstDifference"])
    capsule = "B20C-N11-M5-S4-D1-X0-C0"
    print(capsule)
    return capsule


def parse_arguments(argv: Sequence[str]) -> argparse.Namespace:
    """Parse one of the three controlled company commands."""

    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    return parser.parse_args(argv)


def main(argv: Sequence[str] | None = None) -> int:
    """Run one fail-closed B20C operation."""

    try:
        arguments = parse_arguments(sys.argv[1:] if argv is None else argv)
        root = Path(arguments.workspace_root).resolve()
        if not root.is_dir() or root.is_symlink():
            raise B20CError("WORKSPACE_ROOT", root)
        if arguments.prepare:
            prepare(root)
        elif arguments.self_test:
            self_test(root)
        else:
            execute(root)
        return 0
    except B20CError as error:
        print(f"B20CE-K{error.reason}-D{error.detail}-X1-C7")
        return 7
    except Exception as error:  # pragma: no cover - final diagnostic containment
        print(f"B20CE-K{token(type(error).__name__)}-D{detail_token(error)}-X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
