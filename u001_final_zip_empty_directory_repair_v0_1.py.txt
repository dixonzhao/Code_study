#!/usr/bin/env python3
"""Atomically add the required empty 03_CASES entry to the final ZIP."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import shutil
import tempfile
import zipfile
from pathlib import Path, PurePosixPath


PROGRAM_NAME = "u001_final_zip_empty_directory_repair_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-FINAL-ZIP-REPAIR-SELFTEST-PASS-18"
PACKAGE_NAME = "U001_Portable_v0_2"
CASES_ENTRY = f"{PACKAGE_NAME}/UDP001/03_CASES/"
REPORT_NAME = "UDP001_FINAL_ZIP_REPAIR_REPORT.txt"


class RepairFailure(RuntimeError):
    pass


def fail(code: str) -> None:
    raise RepairFailure(code)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def safe_zip_name(name: str) -> None:
    pure = PurePosixPath(name.replace("\\", "/"))
    if pure.is_absolute() or ".." in pure.parts:
        fail("UNSAFE_ZIP_PATH")
    if pure.parts and pure.parts[0] != PACKAGE_NAME:
        fail("ZIP_ROOT")


def zip_records(path: Path) -> dict[str, tuple[int, int, bool]]:
    result = {}
    with zipfile.ZipFile(path, "r") as archive:
        for info in archive.infolist():
            name = info.filename.replace("\\", "/")
            safe_zip_name(name)
            if name in result:
                fail("DUPLICATE_ENTRY")
            result[name] = (info.file_size, info.CRC, info.is_dir())
    return result


def copy_zip_with_cases(source: Path, target: Path) -> None:
    with zipfile.ZipFile(source, "r") as old, zipfile.ZipFile(target, "w", allowZip64=True) as new:
        new.comment = old.comment
        for info in old.infolist():
            name = info.filename.replace("\\", "/")
            safe_zip_name(name)
            if info.is_dir():
                new.writestr(info, b"")
            else:
                with old.open(info, "r") as reader, new.open(info, "w", force_zip64=True) as writer:
                    shutil.copyfileobj(reader, writer, length=1024 * 1024)
        directory = zipfile.ZipInfo(CASES_ENTRY, date_time=(1980, 1, 1, 0, 0, 0))
        directory.compress_type = zipfile.ZIP_STORED
        directory.external_attr = (0o40775 << 16) | 0x10
        directory.create_system = 3
        new.writestr(directory, b"")


def repair_zip(root: Path) -> dict:
    root = root.resolve()
    package = root / PACKAGE_NAME
    cases = package / "UDP001" / "03_CASES"
    zip_path = root / f"{PACKAGE_NAME}.zip"
    manifest_path = package / "_internal" / "evidence" / "u001_portable_host_package_manifest.json"
    if not root.is_dir() or not package.is_dir() or not zip_path.is_file() or not manifest_path.is_file():
        fail("LAYOUT")
    if not cases.is_dir() or any(cases.iterdir()):
        fail("FOLDER_CASES")
    original_records = zip_records(zip_path)
    if CASES_ENTRY in original_records:
        fail("ENTRY_ALREADY_PRESENT")
    original_hash = sha256_file(zip_path)
    try:
        manifest = json.loads(manifest_path.read_text(encoding="utf-8-sig"))
    except (OSError, json.JSONDecodeError):
        fail("MANIFEST_PARSE")
    serialized = json.dumps(manifest, ensure_ascii=False, sort_keys=True).lower()
    manifest_references = serialized.count(original_hash.lower())
    if manifest_references:
        fail("ZIP_HASH_REFERENCED")
    backup = root.parent / f"{PACKAGE_NAME}_PRE_CASES_FIX_{original_hash[:8]}.zip"
    if backup.exists():
        if not backup.is_file() or sha256_file(backup) != original_hash:
            fail("BACKUP_CONFLICT")
    else:
        shutil.copy2(zip_path, backup)
        if sha256_file(backup) != original_hash:
            fail("BACKUP_VERIFY")
    temporary = root / f".{PACKAGE_NAME}.zip.repair_tmp"
    if temporary.exists():
        fail("TEMP_EXISTS")
    try:
        copy_zip_with_cases(zip_path, temporary)
        repaired_records = zip_records(temporary)
        if CASES_ENTRY not in repaired_records or repaired_records[CASES_ENTRY][2] is not True:
            fail("CASES_ENTRY_VERIFY")
        without_cases = dict(repaired_records)
        without_cases.pop(CASES_ENTRY)
        if without_cases != original_records:
            fail("ORIGINAL_ENTRY_MISMATCH")
        repaired_hash = sha256_file(temporary)
        os.replace(temporary, zip_path)
    finally:
        if temporary.exists():
            try:
                temporary.unlink()
            except OSError:
                pass
    final_records = zip_records(zip_path)
    if final_records != repaired_records or sha256_file(zip_path) != repaired_hash:
        fail("FINAL_VERIFY")
    report = root.parent / REPORT_NAME
    report.write_text(
        "UDP001 FINAL ZIP REPAIR REPORT\n"
        f"OriginalSHA256={original_hash}\n"
        f"RepairedSHA256={repaired_hash}\n"
        f"OriginalEntryCount={len(original_records)}\n"
        f"RepairedEntryCount={len(repaired_records)}\n"
        f"AddedEntry={CASES_ENTRY}\n"
        f"ManifestOriginalZipHashReferences={manifest_references}\n"
        f"Backup={backup.name}\n",
        encoding="utf-8",
    )
    return {
        "FolderCases": 1,
        "EntryAdded": 1,
        "ManifestReferences": manifest_references,
        "Backup": 1,
        "Replaced": 1,
        "Verified": 1,
    }


def run_self_test() -> None:
    with tempfile.TemporaryDirectory() as td:
        root = Path(td) / "release"
        package = root / PACKAGE_NAME
        cases = package / "UDP001" / "03_CASES"
        evidence = package / "_internal" / "evidence"
        cases.mkdir(parents=True)
        evidence.mkdir(parents=True)
        (evidence / "u001_portable_host_package_manifest.json").write_text(
            json.dumps({"Schema": "TEST"}), encoding="utf-8"
        )
        zip_path = root / f"{PACKAGE_NAME}.zip"
        with zipfile.ZipFile(zip_path, "w") as archive:
            archive.writestr(f"{PACKAGE_NAME}/README_FIRST.txt", "readme")
        before = zip_records(zip_path)
        result = repair_zip(root)
        after = zip_records(zip_path)
        checks = (
            PROGRAM_VERSION == "0.1",
            PROGRAM_NAME.endswith("v0_1.py"),
            SELFTEST_TOKEN.endswith("PASS-18"),
            PACKAGE_NAME.endswith("v0_2"),
            CASES_ENTRY.endswith("03_CASES/"),
            REPORT_NAME.endswith(".txt"),
            CASES_ENTRY not in before,
            CASES_ENTRY in after,
            after[CASES_ENTRY][2] is True,
            len(after) == len(before) + 1,
            result["ManifestReferences"] == 0,
            result["Backup"] == 1,
            result["Replaced"] == 1,
            result["Verified"] == 1,
            len(sha256_file(zip_path)) == 64,
            callable(copy_zip_with_cases),
            issubclass(RepairFailure, RuntimeError),
            PurePosixPath(CASES_ENTRY).parts[0] == PACKAGE_NAME,
        )
    if sum(bool(item) for item in checks) != 18:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Repair final ZIP empty 03_CASES entry")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--release-root")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.release_root:
        parser.error("--release-root is required")
    try:
        result = repair_zip(Path(args.release_root))
    except RepairFailure as exc:
        print(f"UDP001-FINAL-ZIP-FAIL-{exc.args[0]} C-7")
        return 7
    except Exception as exc:
        print(f"UDP001-FINAL-ZIP-FAIL-UNHANDLED-{type(exc).__name__} C-7")
        return 7
    print(
        "UDP001-FINAL-ZIP-REPAIR "
        f"N-F{result['FolderCases']}-E{result['EntryAdded']}-M{result['ManifestReferences']}"
        f"-B{result['Backup']}-R{result['Replaced']}-V{result['Verified']}-X0 C-0"
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
