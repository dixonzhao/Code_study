#!/usr/bin/env python3
"""Read-only bounded diagnostic probe for a preserved G10C1 failed Run.

Transfer as .py.txt, remove only the final .txt on the company PC, place the
program in UDP001/04_TEMP, then execute it from C-CODE.  No project file is
created, modified, deleted, renamed, repaired, or re-run.
"""

from __future__ import annotations

import argparse
import json
import re
from pathlib import Path


DEFAULT_CASE_ID = "TC001_20260422"
DEFAULT_RUN_ID = "G10C1_SMOKE_10000_V0_1"


def clean(value: object, limit: int = 220) -> str:
    text = re.sub(r"[\r\n\t]+", " ", str(value).strip())
    text = re.sub(r"\s+", "_", text)
    text = re.sub(r"[^A-Za-z0-9_.:@+\\/=-]", "?", text)
    return text[:limit] if text else "EMPTY"


def read_lines(path: Path | None) -> list[str]:
    if path is None or not path.is_file():
        return []
    try:
        return [x.strip() for x in path.read_text(encoding="utf-8", errors="replace").splitlines() if x.strip()]
    except OSError:
        return []


def stats(path: Path | None) -> tuple[int, int]:
    if path is None or not path.is_file():
        return 0, 0
    try:
        data = path.read_bytes()
    except OSError:
        return 0, 0
    return len(data), len([x for x in data.splitlines() if x.strip()])


def find_one(root: Path, basename: str) -> Path | None:
    if not root.exists():
        return None
    found = sorted(p for p in root.rglob(basename) if p.is_file())
    return found[0] if found else None


def resolve_incomplete(runs_root: Path, run_id: str) -> tuple[Path, int]:
    exact = runs_root / f".{run_id}.g10c1_incomplete"
    if exact.is_dir():
        return exact, 1
    candidates = sorted(
        p for p in runs_root.glob(f"*{run_id}*")
        if p.is_dir() and "incomplete" in p.name.lower()
    ) if runs_root.exists() else []
    return (candidates[0], len(candidates)) if candidates else (exact, 0)


def json_evidence(path: Path | None) -> str:
    if path is None or not path.is_file():
        return "NA"
    try:
        if path.stat().st_size > 5_000_000:
            return "TOO_LARGE"
        obj = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        return "ERR=" + clean(type(exc).__name__, 40)
    if not isinstance(obj, dict):
        return "ROOT=" + clean(type(obj).__name__, 30)
    wanted = (
        "CompletionCode", "Status", "RunStatus", "ReadyForU002",
        "SourceRowCount", "UpdateRowCount", "FrameRowCount", "BatchCount",
        "WarningCount", "ErrorCount",
    )
    lower = {str(k).lower(): (k, v) for k, v in obj.items()}
    out: list[str] = []
    for key in wanted:
        hit = lower.get(key.lower())
        if hit and not isinstance(hit[1], (dict, list)):
            out.append(f"{clean(hit[0], 30)}={clean(hit[1], 50)}")
    return ",".join(out) if out else "NO_TOP_LEVEL_MATCH"


def compact_names(paths: list[Path], limit: int = 360) -> str:
    names = ",".join(sorted(p.name for p in paths))
    return clean(names, limit)


def run_probe(udp001_root: Path, case_id: str, run_id: str, stage: str) -> int:
    case_root = udp001_root / "03_CASES" / case_id
    runs_root = case_root / "02_RUNS"
    incomplete, candidate_count = resolve_incomplete(runs_root, run_id)
    final_root = runs_root / run_id
    response_root = incomplete / "04_RESPONSE"
    data_root = incomplete / "03_DISCOVERY_DATA"

    stdout_path = find_one(response_root, f"{stage}.stdout.txt") or find_one(incomplete, f"{stage}.stdout.txt")
    stderr_path = find_one(response_root, f"{stage}.stderr.txt") or find_one(incomplete, f"{stage}.stderr.txt")
    abort_path = find_one(response_root, "ABORTED_RUN.txt") or find_one(incomplete, "ABORTED_RUN.txt")

    all_files = sorted(p for p in data_root.rglob("*") if p.is_file()) if data_root.exists() else []
    root_files = sorted(p for p in data_root.iterdir() if p.is_file()) if data_root.exists() else []
    csv_files = [p for p in all_files if p.suffix.lower() == ".csv"]
    json_files = [p for p in all_files if p.suffix.lower() == ".json"]
    txt_files = [p for p in all_files if p.suffix.lower() == ".txt"]
    update_files = [p for p in csv_files if "update" in p.name.lower()]
    frame_files = [p for p in csv_files if "frame" in p.name.lower()]
    capsules = [p for p in txt_files if "capsule" in p.name.lower()]
    manifests = [p for p in json_files if "manifest" in p.name.lower()]

    sb, sl = stats(stdout_path)
    eb, el = stats(stderr_path)
    stdout_tail = "_|_".join(read_lines(stdout_path)[-3:]) if stdout_path else "MISSING"
    stderr_tail = "_|_".join(read_lines(stderr_path)[-3:]) if stderr_path else "MISSING"
    abort_tail = "_|_".join(read_lines(abort_path)[-8:]) if abort_path else "MISSING"

    print(
        f"G10C1D2 N-T{int(incomplete.exists())}-TC{candidate_count}-F{int(final_root.exists())}"
        f"-D{int(data_root.exists())}-R{int(response_root.exists())}"
        f"-SO{int(stdout_path is not None)}-SE{int(stderr_path is not None)}"
        f"-AB{int(abort_path is not None)}-X0 C-0"
    )
    print(
        f"DATA-A{len(all_files)}-R{len(root_files)}-C{len(csv_files)}-J{len(json_files)}"
        f"-T{len(txt_files)}-U{len(update_files)}-F{len(frame_files)}"
        f"-CP{len(capsules)}-MF{len(manifests)}"
    )
    print(f"ROOT-{compact_names(root_files)}")
    print(f"STDOUT-B{sb}-L{sl}-T{clean(stdout_tail)}")
    print(f"STDERR-B{eb}-L{el}-T{clean(stderr_tail)}")
    print(f"ABORT-T{clean(abort_tail, 420)}")
    if capsules:
        cap = capsules[0]
        print(f"CAP-N{clean(cap.name, 100)}-T{clean('_|_'.join(read_lines(cap)[-3:]), 260)}")
    else:
        print("CAP-NMISSING-TMISSING")
    if manifests:
        manifest = manifests[0]
        print(f"MAN-N{clean(manifest.name, 120)}-E{clean(json_evidence(manifest), 300)}")
    else:
        print("MAN-NMISSING-ENA")
    return 0


def self_test() -> int:
    assert clean("A B") == "A_B"
    assert stats(None) == (0, 0)
    assert compact_names([]) == "EMPTY"
    print("UDP001-G10C1D2-SELFTEST-PASS-3")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--case-id", default=DEFAULT_CASE_ID)
    parser.add_argument("--run-id", default=DEFAULT_RUN_ID)
    parser.add_argument("--stage", default="C4")
    args = parser.parse_args()
    if args.self_test:
        return self_test()
    udp001_root = Path(args.udp001_root).resolve() if args.udp001_root else Path(__file__).resolve().parent.parent
    return run_probe(udp001_root, args.case_id, args.run_id, args.stage)


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as exc:
        print(f"G10C1D2-ERROR:{clean(type(exc).__name__, 40)}:{clean(exc, 140)} C-7")
        raise SystemExit(7)
