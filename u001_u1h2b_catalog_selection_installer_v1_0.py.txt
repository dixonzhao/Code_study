#!/usr/bin/env python3
"""Prepare, test and install U001 H2B Catalog/Selection support."""

from __future__ import annotations

import argparse
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PROGRAM_VERSION = "1.0.0"
PACKAGE_NAME = "UDP001_U1H2B_CATALOG_SELECTION_HIGH_CODE_v1_0.zip"
ACTIVE_NAME = "U001H_ACTIVE"
PREPARED_NAME = "U1H2B_PREPARED"

EXPECTED_FILES = {
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.1.0/adapter.py",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.1.0/adapter_card.json",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.1.0/u001_matlab_can_csv_dbc_core_v1_0.py",
    "01_PROGRAMS/core/u001_selection_profile_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_catalog_mapping_gui_v1_0.py",
    "02_CONTROL/contracts/u001_input_adapter_card_schema_v0_1.json",
    "02_CONTROL/contracts/u001_source_capability_catalog_schema_v0_1.json",
    "02_CONTROL/contracts/u001_selection_profile_request_schema_v0_1.json",
    "02_CONTROL/templates/u001_selection_profile_request_template_v0_1.json",
    "05_TESTS/tools/u001_h2b_catalog_selection_acceptance_v1_0.py",
    "06_DOCS/U001_H2B_CAPABILITY_CATALOG_SELECTION_AND_RESOURCE_GOVERNANCE_TECHNICAL_SPEC_v1_0_2026-08-28.txt",
    "06_DOCS/U001_SOURCE_DRIVER_AND_CANONICAL_MAPPING_DEVELOPMENT_SPEC_v1_0_2026-08-28.txt",
}

FOUNDATION_FILES = {
    "01_PROGRAMS/core/u001_input_adapter_registry_v0_1.py",
    "01_PROGRAMS/core/u001_input_adapter_conformance_v0_1.py",
    "01_PROGRAMS/core/u001_h3_sync_semantic_engine_v1_0.py",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.0.0/adapter.py",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.0.0/adapter_card.json",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.0.0/u001_matlab_can_csv_dbc_core_v1_0.py",
    "02_CONTROL/contracts/u001_canonical_source_update_schema_v0_1.json",
}

OBSOLETE_TRANSFER_NAMES = {
    "U1H3_EXECUTION_RESULT.json",
    "U1H2B_EXECUTION_RESULT.json",
    "UDP001_U1H3_SYNC_SEMANTIC_HIGH_CODE_v1_0.zip",
    "u001_u1h3_sync_semantic_installer_v1_0.py.txt",
    "U1H3_START_HERE_CN_v1_0_2026-08-28.txt",
}


class InstallError(RuntimeError):
    """Fail-closed H2B installation error."""


def timestamp() -> str:
    return datetime.now().strftime("%Y%m%dT%H%M%S_%f")


def resolve_u001_root(explicit: Path | None) -> Path:
    candidates = []
    if explicit is not None:
        candidates.append(explicit)
    candidates.extend((Path.cwd() / "U-UDP001", Path.cwd()))
    for candidate in candidates:
        candidate = candidate.resolve()
        if candidate.name == "U-UDP001" and candidate.is_dir():
            return candidate
    raise InstallError("U001_ROOT_NOT_FOUND")


def temp_dir(root: Path) -> Path:
    path = root / "04_TEMP"
    if not path.is_dir():
        raise InstallError("TEMP_DIRECTORY_MISSING")
    return path


def validate_foundation(root: Path) -> None:
    missing = sorted(relative for relative in FOUNDATION_FILES if not (root / relative).is_file())
    if missing:
        raise InstallError(f"FOUNDATION_MISSING:{missing}")


def validate_zip(path: Path) -> None:
    if not path.is_file():
        raise InstallError(f"PACKAGE_MISSING:{PACKAGE_NAME}")
    try:
        with zipfile.ZipFile(path, "r") as archive:
            files = {name for name in archive.namelist() if not name.endswith("/")}
            if files != EXPECTED_FILES:
                raise InstallError(
                    f"PACKAGE_INVENTORY:MISSING={sorted(EXPECTED_FILES-files)}:EXTRA={sorted(files-EXPECTED_FILES)}"
                )
            for name in files:
                pure = PurePosixPath(name)
                if pure.is_absolute() or ".." in pure.parts:
                    raise InstallError(f"PACKAGE_PATH:{name}")
            bad = archive.testzip()
            if bad:
                raise InstallError(f"PACKAGE_CRC:{bad}")
    except zipfile.BadZipFile as exc:
        raise InstallError("PACKAGE_ZIP") from exc


def _archive(path: Path, archive_root: Path, relative: Path) -> bool:
    if not path.exists():
        return False
    target = archive_root / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(target))
    return True


def clean_temp(root: Path) -> int:
    transfer = temp_dir(root)
    archive_root = root / "A-Archive" / f"U1H2B_TEMP_HYGIENE_{timestamp()}"
    moved = 0
    for name in sorted(OBSOLETE_TRANSFER_NAMES):
        moved += int(_archive(transfer / name, archive_root, Path(name)))
    active = transfer / ACTIVE_NAME
    for name in ("U1H3_PREPARED", PREPARED_NAME):
        moved += int(_archive(active / name, archive_root, Path(ACTIVE_NAME) / name))
    return moved


def prepare(root: Path) -> Path:
    package = temp_dir(root) / PACKAGE_NAME
    print(f"[CHECK] U001 root exists: {root}")
    print("[WHY]   H2B extends the accepted U001 product; it is not a separate ad-hoc decoder.")
    validate_foundation(root)
    print(f"[CHECK] H1/H2/H3 foundation is complete: {len(FOUNDATION_FILES)} required files")
    print("[WHY]   Catalog/selection must preserve the accepted adapter and H3 boundaries.")
    validate_zip(package)
    print(f"[CHECK] H2B package inventory and CRC passed: {len(EXPECTED_FILES)} files")
    print("[WHY]   Exact inventory blocks stale selection or decoder files.")
    archived = clean_temp(root)
    print(f"[CLEAN] Superseded transfer/prepared entries archived: {archived}")
    print("[WHY]   04_TEMP exposes one current prepared H2B source of truth.")
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package, "r") as archive:
        archive.extractall(prepared)
    extracted = {path.relative_to(prepared).as_posix() for path in prepared.rglob("*") if path.is_file()}
    if extracted != EXPECTED_FILES:
        raise InstallError("EXTRACTED_INVENTORY")
    print("[CHECK] Prepared H2B payload exactly matches the approved package.")
    print(f"U1H2BP-N{len(EXPECTED_FILES)}-A{archived}-X0-C0")
    return prepared


def child_environment() -> dict[str, str]:
    environment = dict(os.environ)
    environment["PYTHONDONTWRITEBYTECODE"] = "1"
    environment["PYTHONPYCACHEPREFIX"] = str(Path(tempfile.gettempdir()) / "u001_h2b_pycache")
    return environment


def run_child(command: list[str], label: str, timeout: int = 240) -> str:
    try:
        result = subprocess.run(
            command,
            text=True,
            capture_output=True,
            timeout=timeout,
            check=False,
            env=child_environment(),
        )
    except subprocess.TimeoutExpired as exc:
        raise InstallError(f"{label}_TIMEOUT") from exc
    if result.returncode != 0:
        detail = (result.stderr or result.stdout).strip().replace("\n", " | ")[:1400]
        raise InstallError(f"{label}_EXIT:{result.returncode}:{detail}")
    output = result.stdout.strip()
    if not output:
        raise InstallError(f"{label}_NO_OUTPUT")
    return output


def self_test(root: Path) -> None:
    validate_foundation(root)
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    if not prepared.is_dir():
        raise InstallError("PREPARED_DIRECTORY_MISSING")
    adapter = prepared / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc" / "1.1.0"
    core = prepared / "01_PROGRAMS" / "core"
    tests = [
        (adapter / "u001_matlab_can_csv_dbc_core_v1_0.py", "PASS-24", "CAN/DBC header discovery and deterministic physical decoding primitives."),
        (adapter / "adapter.py", "PASS-36", "Inventory decodes zero payload rows and controlled runs stream only selected signals."),
        (core / "u001_selection_profile_compiler_v1_0.py", "PASS-14", "Explicit selection, visible FrameCount dependency and bounded resource policy."),
        (core / "u001_catalog_mapping_gui_v1_0.py", "PASS-14", "Source-neutral search/filter/selection/profile GUI model without opening payload data."),
        (prepared / "05_TESTS" / "tools" / "u001_h2b_catalog_selection_acceptance_v1_0.py", "PASS-15", "Independent inventory-to-selection-to-controlled-decode acceptance."),
    ]
    total = 0
    for program, marker, explanation in tests:
        if not program.is_file():
            raise InstallError(f"SELFTEST_PROGRAM:{program.name}")
        print(f"[CHECK] {explanation}")
        output = run_child([sys.executable, str(program), "--self-test"], program.stem.upper())
        if marker not in output:
            raise InstallError(f"SELFTEST_MARKER:{program.name}")
        total += int(marker.rsplit("-", 1)[1])
    print(f"U1H2BST-N{len(tests)}-T{total}-X0-C0")


def copy_payload(prepared: Path, root: Path) -> tuple[int, Path | None]:
    archive_root: Path | None = None
    copied = 0
    for relative_text in sorted(EXPECTED_FILES):
        relative = Path(relative_text)
        source = prepared / relative
        target = root / relative
        if target.exists():
            if archive_root is None:
                archive_root = root / "A-Archive" / f"U1H2B_REPLACED_{timestamp()}"
            _archive(target, archive_root, relative)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        copied += 1
    return copied, archive_root


def execute(root: Path) -> None:
    self_test(root)
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    copied, replaced = copy_payload(prepared, root)

    registry = root / "01_PROGRAMS" / "core" / "u001_input_adapter_registry_v0_1.py"
    snapshot = root / "02_CONTROL" / "registry" / "u001_input_adapter_registry_snapshot_v0_1.json"
    if snapshot.exists():
        archive_root = root / "A-Archive" / f"U1H2B_REGISTRY_REPLACED_{timestamp()}"
        _archive(snapshot, archive_root, Path("02_CONTROL/registry") / snapshot.name)
    registry_output = run_child(
        [sys.executable, str(registry), "--scan", "--adapter-root", str(root / "01_PROGRAMS" / "adapters"), "--output", str(snapshot)],
        "INSTALLED_REGISTRY_SCAN",
    )
    if "C0" not in registry_output.replace("-", ""):
        raise InstallError("REGISTRY_CAPSULE")

    conformance = root / "01_PROGRAMS" / "core" / "u001_input_adapter_conformance_v0_1.py"
    adapter_dir = root / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc" / "1.1.0"
    qualification_root = root / "05_TESTS" / f"U1H2B_ADAPTER_QUALIFICATION_{timestamp()}"
    qualification_output = run_child(
        [sys.executable, str(conformance), "--qualify", "--adapter-dir", str(adapter_dir), "--output", str(qualification_root)],
        "INSTALLED_ADAPTER_QUALIFICATION",
    )
    if "C0" not in qualification_output.replace("-", ""):
        raise InstallError("QUALIFICATION_CAPSULE")

    acceptance = root / "05_TESTS" / "tools" / "u001_h2b_catalog_selection_acceptance_v1_0.py"
    acceptance_root = root / "05_TESTS" / f"U1H2B_ACCEPTANCE_{timestamp()}"
    acceptance_output = run_child(
        [
            sys.executable,
            str(acceptance),
            "--accept",
            "--adapter",
            str(adapter_dir / "adapter.py"),
            "--compiler",
            str(root / "01_PROGRAMS" / "core" / "u001_selection_profile_compiler_v1_0.py"),
            "--out",
            str(acceptance_root),
        ],
        "INSTALLED_H2B_ACCEPTANCE",
    )
    if "C0" not in acceptance_output.replace("-", ""):
        raise InstallError("ACCEPTANCE_CAPSULE")
    receipt = json.loads((acceptance_root / "u001_h2b_acceptance_receipt.json").read_text(encoding="utf-8"))
    if receipt.get("Status") != "ACCEPTED" or receipt.get("CompletionCode") != 0:
        raise InstallError("ACCEPTANCE_RECEIPT")

    h3 = root / "01_PROGRAMS" / "core" / "u001_h3_sync_semantic_engine_v1_0.py"
    h3_output = run_child([sys.executable, str(h3), "--self-test"], "H3_COMPATIBILITY")
    if "PASS-12" not in h3_output:
        raise InstallError("H3_COMPATIBILITY_MARKER")

    result = {
        "Node": "U1H2B",
        "ProgramVersion": PROGRAM_VERSION,
        "InstalledFileCount": copied,
        "InstalledProgramCount": 5,
        "InventoryActionQualified": True,
        "IndependentGateCount": 2,
        "H3CompatibilityPassed": True,
        "AcceptanceFolder": acceptance_root.name,
        "ReplacedFilesArchived": replaced is not None,
        "CompletionCode": 0,
    }
    (temp_dir(root) / "U1H2B_EXECUTION_RESULT.json").write_text(
        json.dumps(result, indent=2) + "\n", encoding="utf-8"
    )
    print(f"U1H2B-N5-I1-Q2-H1-A{copied}-X0-C0")


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description="Install U001 H2B Catalog/Selection")
    action = result.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    result.add_argument("--u001-root", type=Path)
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    try:
        root = resolve_u001_root(args.u001_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            self_test(root)
        else:
            execute(root)
        return 0
    except (InstallError, OSError, ValueError, json.JSONDecodeError, zipfile.BadZipFile) as exc:
        print(f"U1H2BIE-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
