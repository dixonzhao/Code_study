def stable(value: object) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")[:80] or "INTERNAL"


def canonical_bytes(value: object) -> bytes:
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2,
                       allow_nan=False) + "\n").encode("utf-8")


def sha256_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()

def validate_embedded_contracts() -> None:
    if ARCHITECTURE["StageOrder"] != STAGES or STAGE_CONTRACT["StageOrder"] != STAGES:
        raise InstallError("STAGE_ORDER")
    if set(STAGE_CONTRACT["Ownership"]) != set(STAGES):
        raise InstallError("STAGE_OWNERSHIP")
    for name, schema in (("MODULE", MODULE_SCHEMA), ("ARTIFACT", ARTIFACT_SCHEMA),
                         ("RUN", RUN_SCHEMA), ("ALGORITHM", ALGORITHM_CARD_SCHEMA)):
        if schema.get("$schema") != "https://json-schema.org/draft/2020-12/schema":
            raise InstallError(name + "_SCHEMA_DRAFT")
        if schema.get("type") != "object" or schema.get("additionalProperties") is not False:
            raise InstallError(name + "_SCHEMA_STRICTNESS")
    if MODULE_SCHEMA["properties"]["Stage"]["enum"] != STAGES:
        raise InstallError("MODULE_STAGE_ENUM")
    if RUN_SCHEMA["properties"]["RunScope"]["enum"] != ["FULL_CASE", "FRAME_RANGE", "FILTERED_RANGE"]:
        raise InstallError("RUN_SCOPE_ENUM")


def install(root: Path) -> str:
    root = root.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise InstallError("WORKSPACE_ROOT")
    validate_embedded_contracts()
    created_dirs: List[Path] = []
    created_files: List[Path] = []
    try:
        for relative in REQUIRED_DIRS:
            path = root / relative
            if path.exists() and (not path.is_dir() or path.is_symlink()):
                raise InstallError("DIRECTORY_CONFLICT_" + relative)
            if not path.exists():
                path.mkdir(parents=True)
                created_dirs.append(path)
        installed = []
        for relative, payload in FILES.items():
            path = root / relative
            data = canonical_bytes(payload)
            if path.exists():
                if not path.is_file() or path.is_symlink() or path.read_bytes() != data:
                    raise InstallError("CONTRACT_CONFLICT_" + path.name)
            else:
                temporary = path.with_name(path.name + ".installing")
                if temporary.exists():
                    raise InstallError("TEMP_EXISTS_" + path.name)
                temporary.write_bytes(data)
                os.replace(temporary, path)
                created_files.append(path)
            installed.append((relative, len(data), sha256_bytes(data)))
        manifest = root / "00_CONTROL/u22a0_contract_manifest.csv"
        manifest_lines = [["RelativePath", "SizeBytes", "Sha256"]]
        manifest_lines.extend([[a, str(b), c] for a, b, c in installed])
        if not manifest.exists():
            with manifest.open("x", encoding="utf-8", newline="") as stream:
                csv.writer(stream, lineterminator="\n").writerows(manifest_lines)
            created_files.append(manifest)
        reset_script = root / "04_TEMP/udp002_u22r0_workspace_archive_reset_v0_1.py.txt"
        archived_reset = 0
        if reset_script.is_file() and not reset_script.is_symlink():
            destination = root / "A-Archive/U21_LEGACY_20260820/04_TEMP" / reset_script.name
            destination.parent.mkdir(parents=True, exist_ok=True)
            if destination.exists():
                if destination.read_bytes() != reset_script.read_bytes():
                    raise InstallError("RESET_ARCHIVE_CONFLICT")
                reset_script.unlink()
            else:
                os.replace(reset_script, destination)
            archived_reset = 1
        capsule = f"U22A0 N-D{len(REQUIRED_DIRS)}-C{len(FILES)}-M1-R{archived_reset}-X0 C-0"
        capsule_path = root / "00_CONTROL/u22a0_return_capsule.txt"
        if not capsule_path.exists():
            capsule_path.write_text(capsule + "\n", encoding="utf-8")
        return capsule
    except Exception:
        for path in reversed(created_files):
            if path.exists():
                path.unlink()
        for path in sorted(created_dirs, key=lambda item: len(item.parts), reverse=True):
            if path.exists() and not any(path.iterdir()):
                path.rmdir()
        raise


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    try:
        print(install(Path(args.workspace_root)))
        return 0
    except Exception as exc:
        print(f"U22A0 N-D0-C0-M0-R0-X1 C-7 K-{stable(exc)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
