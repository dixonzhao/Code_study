#!/usr/bin/env python3
"""Onefile entry for U003 Evidence Review Workbench MVP v0.1."""

from __future__ import annotations

import datetime as dt
import json
import os
import sys
import traceback
from pathlib import Path


PRODUCT_NAME = "U003 Evidence Review Workbench"
PRODUCT_VERSION = "0.1"
CORE_IMPLEMENTATION = "FrameCount Video Reviewer v0.5"
WORKSPACE_NAME = "U003_Workspace"


def executable_directory() -> Path:
    if getattr(sys, "frozen", False):
        return Path(sys.executable).resolve().parent
    return Path(__file__).resolve().parent


def ensure_workspace() -> dict[str, Path]:
    root = executable_directory() / WORKSPACE_NAME
    paths = {
        "root": root,
        "video": root / "01_INPUT_VIDEO",
        "events": root / "02_EVENT_QUEUE",
        "review_data": root / "03_REVIEW_DATA",
        "output": root / "04_OUTPUT",
        "logs": root / "05_LOGS",
    }
    for path in paths.values():
        path.mkdir(parents=True, exist_ok=True)

    readme = root / "README_U003_WORKSPACE.txt"
    if not readme.exists():
        readme.write_text(
            "U003 WORKSPACE\n"
            "\n"
            "01_INPUT_VIDEO  - optional local working copies of videos\n"
            "02_EVENT_QUEUE  - optional Event Queue CSV files\n"
            "03_REVIEW_DATA  - optional synchronized narrow CAN CSV files\n"
            "04_OUTPUT       - operator-selected report working area\n"
            "05_LOGS         - U003 startup and failure logs\n"
            "\n"
            "The accepted v0.5 core preserves its validated behavior: screenshots,\n"
            "plot CSVs and review outputs are normally written beside the selected\n"
            "video/Event/Review Data input. Put working copies into this workspace\n"
            "when all generated artifacts should remain under one portable folder.\n",
            encoding="utf-8",
        )

    manifest = root / "U003_WORKSPACE_MANIFEST.json"
    manifest.write_text(
        json.dumps(
            {
                "product": PRODUCT_NAME,
                "product_version": PRODUCT_VERSION,
                "core_implementation": CORE_IMPLEMENTATION,
                "workspace_version": "0.1",
                "executable_directory": str(executable_directory()),
            },
            ensure_ascii=False,
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    os.environ["U003_WORKSPACE"] = str(root)
    return paths


def append_startup_log(log_dir: Path, state: str, detail: str = "") -> None:
    timestamp = dt.datetime.now().astimezone().isoformat(timespec="seconds")
    with (log_dir / "U003_STARTUP.log").open("a", encoding="utf-8") as handle:
        handle.write(f"{timestamp}\t{state}\t{detail}\n")


def main() -> int:
    paths: dict[str, Path] | None = None
    try:
        paths = ensure_workspace()
        append_startup_log(paths["logs"], "START", CORE_IMPLEMENTATION)
        from framecount_video_reviewer_v0_5 import main as core_main

        result = int(core_main())
        append_startup_log(paths["logs"], "EXIT", str(result))
        return result
    except Exception as exc:
        detail = f"{type(exc).__name__}: {exc}"
        if paths is not None:
            append_startup_log(paths["logs"], "ERROR", detail)
            (paths["logs"] / "U003_STARTUP_ERROR.txt").write_text(
                traceback.format_exc(), encoding="utf-8"
            )
        try:
            import tkinter as tk
            from tkinter import messagebox

            root = tk.Tk()
            root.withdraw()
            messagebox.showerror(f"{PRODUCT_NAME} startup error", detail)
            root.destroy()
        except Exception:
            pass
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
