#!/usr/bin/env python3
"""Independent clean-folder acceptance gate for UDP002 U2-0A.

Uses only the accepted schema bundle and validator as inputs. It creates two
new isolated run folders, executes the validator with Python isolated mode,
compares deterministic fixtures and writes factual evidence. It never edits
the accepted source files and does not freeze U2-0A by itself.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
import re
import shutil
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path


VERSION = "0.1"
DEFAULT_TAG = "U20A_CLEAN_20260818"
EXPECTED_SELFTEST = "UDP002-U20A-SELFTEST-PASS-24 C-0"
EXPECTED_FIXTURES = "UDP002-U20A-FIXTURES-WRITTEN-N8 C-0"
EXPECTED_VALIDATE = "UDP002U20A1 N-B1-F8-VP4-VE4-IR4-IE4-X0 C-0"
EXPECTED_FIXTURE_NAMES = {
    "valid_module_manifest.json",
    "invalid_module_missing_field.json",
    "invalid_module_stage.json",
    "valid_artifact_manifest.json",
    "invalid_artifact_parent_identity.json",
    "valid_run_manifest.json",
    "invalid_run_incomplete_succeeded.json",
    "valid_failed_run_manifest.json",
}


def sha256(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def tree_hashes(directory: Path) -> dict[str, str]:
    return {
        path.relative_to(directory).as_posix(): sha256(path)
        for path in sorted(directory.rglob("*"))
        if path.is_file()
    }


def find_source(temp_dir: Path, stem: str, suffixes: tuple[str, ...]) -> Path:
    matches = [temp_dir / f"{stem}{suffix}" for suffix in suffixes]
    existing = [path for path in matches if path.is_file()]
    if len(existing) != 1:
        shown = ",".join(path.name for path in existing) or "NONE"
        raise RuntimeError(f"SOURCE_{stem.upper()}_COUNT_{len(existing)}:{shown}")
    return existing[0]


def next_clean_directory(parent: Path, tag: str) -> Path:
    candidate = parent / tag
    if not candidate.exists():
        return candidate
    for index in range(2, 100):
        candidate = parent / f"{tag}_R{index:02d}"
        if not candidate.exists():
            return candidate
    raise RuntimeError("NO_AVAILABLE_CLEAN_RUN_DIRECTORY")


def run_command(command: list[str], cwd: Path, log_prefix: Path) -> dict[str, object]:
    environment = os.environ.copy()
    environment.pop("PYTHONPATH", None)
    environment["PYTHONDONTWRITEBYTECODE"] = "1"
    completed = subprocess.run(
        command,
        cwd=cwd,
        env=environment,
        text=True,
        capture_output=True,
        timeout=120,
        check=False,
    )
    stdout = completed.stdout.replace("\r\n", "\n")
    stderr = completed.stderr.replace("\r\n", "\n")
    log_prefix.with_suffix(".stdout.txt").write_text(stdout, encoding="utf-8")
    log_prefix.with_suffix(".stderr.txt").write_text(stderr, encoding="utf-8")
    return {
        "command": command,
        "return_code": completed.returncode,
        "stdout": stdout,
        "stderr": stderr,
    }


def exact_line(result: dict[str, object], expected: str) -> bool:
    return result["return_code"] == 0 and expected in str(result["stdout"]).splitlines()


def contract_version_check(schema: Path, validator: Path) -> dict[str, object]:
    bundle = json.loads(schema.read_text(encoding="utf-8-sig"))
    schema_version = bundle.get("x-BundleVersion")
    schema_id = bundle.get("$id")
    validator_text = validator.read_text(encoding="utf-8-sig")
    version_match = re.search(r'^CONTRACT_VERSION\s*=\s*["\']([^"\']+)["\']', validator_text, re.MULTILINE)
    id_match = re.search(r'^BUNDLE_ID\s*=\s*["\']([^"\']+)["\']', validator_text, re.MULTILINE)
    validator_version = version_match.group(1) if version_match else None
    validator_id = id_match.group(1) if id_match else None
    passed = (
        schema_version == validator_version == "0.1"
        and schema_id == validator_id == "urn:u002:core-contract-schema-bundle:0.1"
    )
    return {
        "passed": passed,
        "schema_version": schema_version,
        "validator_version": validator_version,
        "schema_id": schema_id,
        "validator_id": validator_id,
    }


def execute_one_run(
    run_dir: Path,
    source_schema: Path,
    source_validator: Path,
) -> dict[str, object]:
    contract_dir = run_dir / "contract"
    fixture_dir = run_dir / "fixtures"
    log_dir = run_dir / "logs"
    contract_dir.mkdir(parents=True)
    log_dir.mkdir(parents=True)
    schema = contract_dir / "u002_core_contract_schema_bundle_v0_1.json"
    validator = contract_dir / "u002_core_contract_validator_v0_1.py"
    shutil.copy2(source_schema, schema)
    shutil.copy2(source_validator, validator)

    source_copy_ok = sha256(schema) == sha256(source_schema) and sha256(validator) == sha256(source_validator)
    version = contract_version_check(schema, validator)
    prefix = [sys.executable, "-I", str(validator), "--schema-bundle", str(schema)]
    selftest = run_command(prefix + ["--self-test"], run_dir, log_dir / "01_selftest")
    emit = run_command(prefix + ["--emit-fixtures", str(fixture_dir)], run_dir, log_dir / "02_emit")
    validate = run_command(prefix + ["--validate-dir", str(fixture_dir)], run_dir, log_dir / "03_validate")

    fixture_names = {path.name for path in fixture_dir.glob("*.json")} if fixture_dir.is_dir() else set()
    fixture_set_ok = fixture_names == EXPECTED_FIXTURE_NAMES
    status = {
        "source_copy_ok": source_copy_ok,
        "version_check": version,
        "selftest_ok": exact_line(selftest, EXPECTED_SELFTEST),
        "emit_ok": exact_line(emit, EXPECTED_FIXTURES),
        "validate_ok": exact_line(validate, EXPECTED_VALIDATE),
        "fixture_set_ok": fixture_set_ok,
        "fixture_names": sorted(fixture_names),
        "fixture_hashes": tree_hashes(fixture_dir) if fixture_dir.is_dir() else {},
        "stdout": {
            "selftest": selftest["stdout"],
            "emit": emit["stdout"],
            "validate": validate["stdout"],
        },
        "return_codes": {
            "selftest": selftest["return_code"],
            "emit": emit["return_code"],
            "validate": validate["return_code"],
        },
    }
    status["passed"] = bool(
        source_copy_ok
        and version["passed"]
        and status["selftest_ok"]
        and status["emit_ok"]
        and status["validate_ok"]
        and fixture_set_ok
    )
    return status


def render_report(evidence: dict[str, object], capsule: str) -> str:
    run1 = evidence["runs"][0]
    run2 = evidence["runs"][1]
    lines = [
        "UDP002 U2-0A INDEPENDENT CLEAN-FOLDER RERUN EVIDENCE",
        "",
        f"GateVersion: {VERSION}",
        f"GeneratedUtc: {evidence['generated_utc']}",
        f"Status: {evidence['status']}",
        f"ReturnCode: {evidence['return_code']}",
        f"Capsule: {capsule}",
        "",
        "SOURCE IDENTITY",
        f"SchemaSha256: {evidence['source']['schema_sha256']}",
        f"ValidatorSha256: {evidence['source']['validator_sha256']}",
        "",
        "RUN 1",
        f"Passed: {int(run1['passed'])}",
        f"SourceCopy: {int(run1['source_copy_ok'])}",
        f"VersionAgreement: {int(run1['version_check']['passed'])}",
        f"SelfTest: {int(run1['selftest_ok'])}",
        f"FixtureEmission: {int(run1['emit_ok'])}",
        f"DirectoryValidation: {int(run1['validate_ok'])}",
        f"FixtureSet: {int(run1['fixture_set_ok'])}",
        "",
        "RUN 2",
        f"Passed: {int(run2['passed'])}",
        f"SourceCopy: {int(run2['source_copy_ok'])}",
        f"VersionAgreement: {int(run2['version_check']['passed'])}",
        f"SelfTest: {int(run2['selftest_ok'])}",
        f"FixtureEmission: {int(run2['emit_ok'])}",
        f"DirectoryValidation: {int(run2['validate_ok'])}",
        f"FixtureSet: {int(run2['fixture_set_ok'])}",
        "",
        "DETERMINISM",
        f"FixtureTreesByteIdentical: {int(evidence['deterministic_fixtures'])}",
        f"ConsoleTokensIdentical: {int(evidence['deterministic_console'])}",
        "",
        "BOUNDARY",
        "Inputs were limited to the accepted Schema Bundle and Validator.",
        "Both executions used Python isolated mode (-I) in newly created folders.",
        "No company data, DBC, U001 output, P004 or third-party Python package was used.",
        "The gate generated factual evidence only; formal freeze requires a separate decision.",
        "",
        "RETURN ONLY THE CAPSULE LINE.",
        "",
    ]
    return "\n".join(lines)


def self_test() -> int:
    if next_clean_directory(Path("unused"), "tag").name != "tag":
        print("UDP002-U20A-CLEAN-GATE-SELFTEST-FAIL-ALLOCATOR C-7")
        return 7
    if not re.search(r"PASS", EXPECTED_SELFTEST):
        print("UDP002-U20A-CLEAN-GATE-SELFTEST-FAIL-TOKEN C-7")
        return 7
    print("UDP002-U20A-CLEAN-GATE-SELFTEST-PASS-2 C-0")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--root", help="U-UDP002 root; defaults to ./U-UDP002")
    parser.add_argument("--tag", default=DEFAULT_TAG)
    parser.add_argument("--self-test", action="store_true")
    args = parser.parse_args()
    if args.self_test:
        return self_test()

    try:
        root = Path(args.root or (Path.cwd() / "U-UDP002")).resolve()
        if not root.is_dir():
            raise FileNotFoundError(f"UDP002 root not found: {root}")
        temp_dir = root / "04_TEMP"
        source_schema = find_source(
            temp_dir,
            "u002_core_contract_schema_bundle_v0_1",
            (".json", ".json.txt"),
        )
        source_validator = find_source(
            temp_dir,
            "u002_core_contract_validator_v0_1",
            (".py", ".py.txt"),
        )
        acceptance_parent = root / "05_TESTS" / "02_ACCEPTANCE"
        acceptance_parent.mkdir(parents=True, exist_ok=True)
        output_dir = next_clean_directory(acceptance_parent, args.tag)
        output_dir.mkdir()

        run1 = execute_one_run(output_dir / "run1", source_schema, source_validator)
        run2 = execute_one_run(output_dir / "run2", source_schema, source_validator)
        deterministic_fixtures = run1["fixture_hashes"] == run2["fixture_hashes"]
        deterministic_console = run1["stdout"] == run2["stdout"]
        passed = bool(run1["passed"] and run2["passed"] and deterministic_fixtures and deterministic_console)
        return_code = "C-0" if passed else "C-7"
        capsule = (
            "U20ACG1 "
            f"N-R2-S{int(run1['selftest_ok']) + int(run2['selftest_ok'])}"
            f"-F{int(run1['emit_ok']) + int(run2['emit_ok'])}"
            f"-V{int(run1['validate_ok']) + int(run2['validate_ok'])}"
            f"-H{int(run1['source_copy_ok'] and run2['source_copy_ok'])}"
            f"-A{int(run1['version_check']['passed'] and run2['version_check']['passed'])}"
            f"-D{int(deterministic_fixtures and deterministic_console)}"
            f"-X{0 if passed else 1} {return_code}"
        )
        evidence = {
            "gate_version": VERSION,
            "generated_utc": datetime.now(timezone.utc).isoformat(timespec="seconds"),
            "status": "PASS" if passed else "FAIL",
            "return_code": return_code,
            "source": {
                "schema_name": source_schema.name,
                "schema_sha256": sha256(source_schema),
                "validator_name": source_validator.name,
                "validator_sha256": sha256(source_validator),
            },
            "runs": [run1, run2],
            "deterministic_fixtures": deterministic_fixtures,
            "deterministic_console": deterministic_console,
            "capsule": capsule,
        }
        (output_dir / "U20A_CLEAN_RERUN_EVIDENCE.json").write_text(
            json.dumps(evidence, ensure_ascii=False, indent=2) + "\n",
            encoding="utf-8",
        )
        (output_dir / "U20A_CLEAN_RERUN_REPORT.txt").write_text(
            render_report(evidence, capsule),
            encoding="utf-8",
        )
        print(capsule)
        print(f"RESULT={output_dir / 'U20A_CLEAN_RERUN_REPORT.txt'}")
        return 0 if passed else 7
    except Exception as error:
        print(f"U20ACG1 ERROR-{type(error).__name__} C-7")
        print(str(error))
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
