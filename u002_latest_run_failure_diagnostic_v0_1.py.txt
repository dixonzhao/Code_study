#!/usr/bin/env python3
"""Report the exact failed U002 module and its persisted child evidence."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import re
import sys
import tempfile


PROGRAM_ID = "U002-LATEST-RUN-FAILURE-DIAGNOSTIC"
VERSION = "0.1.0"
ALIASES = {
    "CameraRadar.S0.U001ControlledProfileIntake": "S0I",
    "CameraRadar.S1.CameraTargetContinuityArbitration": "CTA",
    "CameraRadar.S1.RadarMeasurementValidity": "RMV",
    "CameraRadar.S6.RadarMeasurementLossGeometryCandidateSynthesis": "RLG",
    "CameraRadar.S7.CandidateReviewBatchExport": "RBE",
}


class DiagnosticError(RuntimeError):
    """Represent one deterministic diagnostic failure."""


def token(value: object, limit: int = 96) -> str:
    """Convert evidence to a compact capsule-safe token."""

    text = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (text or "NONE")[:limit]


def load_json(path: Path) -> dict:
    """Load one JSON object and retain the file identity in failures."""

    if not path.is_file() or path.is_symlink():
        raise DiagnosticError("MISSING_" + token(path.name))
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise DiagnosticError("INVALID_JSON_" + token(path.name)) from exc
    if not isinstance(value, dict):
        raise DiagnosticError("INVALID_OBJECT_" + token(path.name))
    return value


def require_workspace(value: str) -> Path:
    """Accept only the controlled U002 workspace."""

    root = Path(value).resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise DiagnosticError("WORKSPACE_ROOT")
    output_root = root / "07_OUTPUT"
    if not output_root.is_dir():
        raise DiagnosticError("OUTPUT_ROOT")
    return root


def choose_run(root: Path, requested: str | None) -> Path:
    """Use the requested Run or the newest persisted failed Run."""

    output_root = root / "07_OUTPUT"
    if requested:
        run_root = output_root / requested
        if not run_root.is_dir() or run_root.is_symlink():
            raise DiagnosticError("RUN_NOT_FOUND")
        return run_root

    candidates = []
    for status_path in output_root.glob("*/run_status.json"):
        try:
            status = load_json(status_path)
        except DiagnosticError:
            continue
        if status.get("Status") == "FAILED":
            candidates.append((status_path.stat().st_mtime_ns, status_path.parent))
    if not candidates:
        raise DiagnosticError("FAILED_RUN_NOT_FOUND")
    return max(candidates, key=lambda item: item[0])[1]


def last_nonempty(path: Path) -> str:
    """Return the final non-empty child output line without flooding the capsule."""

    if not path.is_file() or path.is_symlink():
        return "MISSING"
    lines = [line.strip() for line in path.read_text(
        encoding="utf-8-sig", errors="replace"
    ).splitlines() if line.strip()]
    return lines[-1] if lines else "EMPTY"


def failed_record(run_root: Path, module_id: str) -> dict:
    """Find the unique failed execution record for the status-owned module."""

    matches = []
    for path in (run_root / "01_LOGS").glob("*.json"):
        value = load_json(path)
        if value.get("ModuleId") == module_id and value.get("Status") == "FAILED":
            matches.append(value)
    if len(matches) != 1:
        raise DiagnosticError("FAILED_RECORD_COUNT_" + str(len(matches)))
    return matches[0]


def diagnose(root: Path, requested: str | None) -> str:
    """Resolve status, frozen plan, execution record and child stdout/stderr."""

    run_root = choose_run(root, requested)
    status = load_json(run_root / "run_status.json")
    if status.get("Status") != "FAILED":
        raise DiagnosticError("RUN_NOT_FAILED_" + token(status.get("Status")))
    module_id = status.get("FailedModule")
    if not isinstance(module_id, str) or not module_id:
        raise DiagnosticError("FAILED_MODULE_MISSING")

    plan = load_json(run_root / "00_CONFIG/resolved_run_plan.json")
    modules = [
        item for item in plan.get("ModulePlan", [])
        if isinstance(item, dict) and item.get("ModuleId") == module_id
    ]
    if len(modules) != 1:
        raise DiagnosticError("FAILED_PLAN_MODULE_COUNT_" + str(len(modules)))
    module = modules[0]
    record = failed_record(run_root, module_id)
    diagnostic_root = run_root / str(record.get("DiagnosticRoot", ""))
    if not diagnostic_root.is_dir() or diagnostic_root.is_symlink():
        raise DiagnosticError("DIAGNOSTIC_ROOT")

    stdout = last_nonempty(diagnostic_root / "stdout.txt")
    stderr = last_nonempty(diagnostic_root / "stderr.txt")
    result_path = diagnostic_root / "module_result.json"
    result_capsule = "MISSING"
    if result_path.is_file() and not result_path.is_symlink():
        result = load_json(result_path)
        result_capsule = str(result.get("Capsule", "NONE"))

    alias = ALIASES.get(module_id, token(module_id, 16))
    print("[FAILED RUN] " + run_root.name)
    print(
        "[FAILED MODULE] " + str(module.get("Stage")) + " | "
        + module_id + " | version " + str(module.get("ModuleVersion"))
    )
    print("[EXECUTOR REASON] " + str(record.get("Reason", status.get("FailureReason"))))
    print("[CHILD STDOUT LAST] " + stdout)
    print("[CHILD STDERR LAST] " + stderr)
    print("[DIAGNOSTIC FOLDER] " + str(diagnostic_root))

    capsule = (
        "S1D-M" + alias
        + "-V" + token(module.get("ModuleVersion"), 12).replace("_", "")
        + "-K" + token(record.get("Reason", status.get("FailureReason")), 40)
        + "-O" + token(result_capsule if result_capsule != "MISSING" else stdout, 96)
        + "-E" + token(stderr, 64)
        + "-X0-C0"
    )
    print(capsule)
    return capsule


def self_test() -> str:
    """Exercise failed-run discovery and evidence extraction on a synthetic Run."""

    with tempfile.TemporaryDirectory() as temporary:
        root = Path(temporary) / "U-UDP002"
        run_root = root / "07_OUTPUT/RUN_TEST"
        (run_root / "00_CONFIG").mkdir(parents=True)
        (run_root / "01_LOGS").mkdir()
        diagnostic = run_root / "02_DIAGNOSTICS/modules/0002_Module.Test"
        diagnostic.mkdir(parents=True)
        (run_root / "run_status.json").write_text(json.dumps({
            "RunId": "RUN_TEST", "Status": "FAILED",
            "FailedModule": "Module.Test", "FailureReason": "MODULE_EXIT"
        }), encoding="utf-8")
        (run_root / "00_CONFIG/resolved_run_plan.json").write_text(json.dumps({
            "ModulePlan": [{"Sequence": 2, "Stage": "S1",
                            "ModuleId": "Module.Test", "ModuleVersion": "0.1.0"}]
        }), encoding="utf-8")
        (run_root / "01_LOGS/0002_Module.Test.json").write_text(json.dumps({
            "ModuleId": "Module.Test", "Status": "FAILED", "Reason": "MODULE_EXIT",
            "DiagnosticRoot": "02_DIAGNOSTICS/modules/0002_Module.Test"
        }), encoding="utf-8")
        (diagnostic / "stdout.txt").write_text("TEST-CHILD-C7\n", encoding="utf-8")
        (diagnostic / "stderr.txt").write_text("", encoding="utf-8")
        capsule = diagnose(root, "RUN_TEST")
        if "-KMODULE_EXIT-OTEST_CHILD_C7-EEMPTY-X0-C0" not in capsule:
            raise DiagnosticError("SELFTEST")
    result = "S1DST-N12-X0-C0"
    print(result)
    return result


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root")
    parser.add_argument("--run-id")
    parser.add_argument("--self-test", action="store_true")
    args = parser.parse_args()
    try:
        if args.self_test:
            self_test()
        else:
            if not args.workspace_root:
                raise DiagnosticError("WORKSPACE_ROOT_REQUIRED")
            diagnose(require_workspace(args.workspace_root), args.run_id)
        return 0
    except Exception as exc:
        print("S1DE-K" + token(exc, 80) + "-X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
