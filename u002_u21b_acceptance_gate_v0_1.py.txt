SUCCESS = "U21BG1 N-R2-M10-S6-H12-D1-N0-X0 C-0"
SELFTEST_SUCCESS = "U21BGS N-T16-X0 C-0"
ENGINE_VERSION = "0.1.0"
EXPECTED_COUNTS = {
    "AVAILABLE": 0,
    "INCOMPATIBLE": 1,
    "INVALID_MANIFEST": 2,
    "DEPENDENCY_MISSING": 2,
    "SELFTEST_REQUIRED": 1,
    "QUARANTINED": 4,
}
TOP_FIELDS = {
    "RegistrySnapshotType",
    "RegistrySnapshotVersion",
    "GeneratedAtUtc",
    "EngineVersion",
    "ModulesRootIdentity",
    "ScanPolicy",
    "StatusCounts",
    "ModuleCount",
    "Modules",
    "UnmanagedEntryCount",
    "Warnings",
}

    "ModuleId",
    "ModuleVersion",
    "DisplayName",
    "Stage",
    "ModuleKind",
    "AlgorithmStatus",
    "ExecutionRequirement",
    "ManifestRelativePath",
    "ManifestSha256",
    "EntrypointRelativePath",
    "ImplementationSha256Declared",
    "ImplementationSha256Observed",
    "ConfigurationSchemaRelativePath",
    "ConfigurationSchemaSha256Declared",

    "MinimumEngineVersion",
    "MaximumEngineVersion",
    "RegistryStatus",
    "ReasonCodes",
    "ValidatorExitCode",
}


class GateError(RuntimeError):
    def __init__(self, code: str, detail: str = "") -> None:
        super().__init__(detail or code)
        self.code = code
        self.detail = detail or code


def require(condition: bool, code: str, detail: str = "") -> None:
    if not condition:
        raise GateError(code, detail)


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(chunk)
    return digest.hexdigest()


def canonical_bytes(value: Any) -> bytes:
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")) + "\n").encode("utf-8")


def atomic_json(path: Path, value: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def atomic_text(path: Path, value: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    # Write bytes directly so Windows cannot translate LF to CRLF. Evidence
    # hashes and self-tests must be byte-stable across supported hosts.
    temporary.write_bytes(value.encode("utf-8"))
    os.replace(temporary, path)


def load_object(path: Path, code: str) -> dict[str, Any]:
    try:
        value = json.loads(path.read_text(encoding="utf-8"))
    except Exception as exc:
        raise GateError(code, f"{path.name}: {type(exc).__name__}: {exc}") from exc
    require(isinstance(value, dict), code, f"{path.name} is not a JSON object")
    return value


def bounded(value: str, limit: int = 4000) -> str:
    return value if len(value) <= limit else value[:limit] + "...[TRUNCATED]"


def ensure_clean_output(path: Path) -> None:
    if path.exists():
        require(path.is_dir(), "OUTPUT_NOT_DIRECTORY", str(path))
        require(not any(path.iterdir()), "OUTPUT_NOT_EMPTY", str(path))
    else:
        path.mkdir(parents=True)


def safe_slug(value: str) -> str:
    return "".join(ch if ch.isalnum() else "_" for ch in value)


def write_inert_files(pack: Path, marker_name: str) -> tuple[str, str]:
    pack.mkdir(parents=True, exist_ok=True)
    implementation = pack / "implementation.py"
    config = pack / "config_schema.json"
    implementation.write_text(
        "from pathlib import Path\n"
        f"Path(__file__).with_name({marker_name!r}).write_text('EXECUTED', encoding='utf-8')\n",
        encoding="utf-8",
    )
    config.write_text("{}\n", encoding="utf-8")
    return sha256_file(implementation), sha256_file(config)


def make_manifest(
    base: dict[str, Any],
    module_id: str,
    display_name: str,
    implementation_hash: str,
    config_hash: str,
) -> dict[str, Any]:
    manifest = copy.deepcopy(base)
    suffix = safe_slug(module_id.split(".")[-1])
    manifest["ModuleId"] = module_id
    manifest["ModuleVersion"] = "1.0.0"
    manifest["DisplayName"] = display_name
    manifest["OutputNamespace"] = f"Gate.{suffix}"
    manifest["Entrypoint"] = {"RelativeFile": "implementation.py", "Callable": "run"}
    manifest["ImplementationSha256"] = implementation_hash
    manifest["ConfigurationSchema"]["SchemaId"] = f"Gate.{suffix}.Config"
    manifest["ConfigurationSchema"]["SchemaVersion"] = "1.0.0"
    manifest["ConfigurationSchema"]["RelativePath"] = "config_schema.json"
    manifest["ConfigurationSchema"]["Sha256"] = config_hash
    manifest["Compatibility"] = {
        "MinimumEngineVersion": "0.1.0",
        "MaximumEngineVersion": None,
    }
    for index, provision in enumerate(manifest.get("ProvidesArtifacts", [])):
        provision["ArtifactType"] = f"GATE_{suffix.upper()}_{index}"
        provision["ArtifactSchemaId"] = f"Gate.{suffix}.Artifact{index}"
    return manifest


def write_manifest(path: Path, value: dict[str, Any]) -> None:
    atomic_json(path, value)


def build_fixture_tree(root: Path, fixture_manifest: Path) -> dict[str, Any]:
    base = load_object(fixture_manifest, "FIXTURE_INVALID")
    identities: dict[str, Any] = {}

    def pack(name: str, module_id: str | None = None) -> tuple[Path, dict[str, Any]]:
        directory = root / name
        ih, ch = write_inert_files(directory, "EXECUTED.marker")
        identity = module_id or f"U002.Gate.{safe_slug(name)}"
        manifest = make_manifest(base, identity, f"Gate {name}", ih, ch)
        identities[name] = {
            "ManifestRelativePath": f"{name}/module_manifest.json",
            "ModuleId": identity,
        }
        return directory, manifest

    directory, manifest = pack("01_valid_static", "U002.Gate.ValidStatic")
    write_manifest(directory / "module_manifest.json", manifest)

    invalid_json = root / "02_invalid_json"
    invalid_json.mkdir(parents=True)
    (invalid_json / "module_manifest.json").write_text("{\n", encoding="utf-8")
    identities["02_invalid_json"] = {"ManifestRelativePath": "02_invalid_json/module_manifest.json", "ModuleId": None}

    directory, manifest = pack("03_contract_invalid", "U002.Gate.ContractInvalid")
    manifest["ContractType"] = "NOT_U002_MODULE_MANIFEST"
    write_manifest(directory / "module_manifest.json", manifest)

    directory, manifest = pack("04_incompatible", "U002.Gate.Incompatible")
    manifest["Compatibility"] = {"MinimumEngineVersion": "99.0.0", "MaximumEngineVersion": None}
    write_manifest(directory / "module_manifest.json", manifest)

    directory, manifest = pack("05_missing_entrypoint", "U002.Gate.MissingEntrypoint")
    (directory / "implementation.py").unlink()
    write_manifest(directory / "module_manifest.json", manifest)

    directory, manifest = pack("06_missing_config", "U002.Gate.MissingConfig")
    (directory / "config_schema.json").unlink()
    write_manifest(directory / "module_manifest.json", manifest)

    directory, manifest = pack("07_bad_implementation_hash", "U002.Gate.BadImplementationHash")
    manifest["ImplementationSha256"] = "0" * 64
    write_manifest(directory / "module_manifest.json", manifest)

    directory, manifest = pack("08_bad_config_hash", "U002.Gate.BadConfigHash")
    manifest["ConfigurationSchema"]["Sha256"] = "0" * 64
    write_manifest(directory / "module_manifest.json", manifest)

    for name in ("09_duplicate_a", "10_duplicate_b"):
        directory, manifest = pack(name, "U002.Gate.Duplicate")
        write_manifest(directory / "module_manifest.json", manifest)

    return identities


def registry_command(
    registry: Path,
    modules_root: Path,
    schema_bundle: Path,
    contract_validator: Path,
    output_path: Path,
) -> list[str]:
    return [
        sys.executable,
        str(registry),
        "--scan",
        "--modules-root",
        str(modules_root),
        "--engine-version",
        ENGINE_VERSION,
        "--schema-bundle",
        str(schema_bundle),
        "--contract-validator",
        str(contract_validator),
        "--out",
        str(output_path),
    ]


def run_child(command: list[str], timeout: int = 120) -> subprocess.CompletedProcess[str]:
    try:
        return subprocess.run(
            command,
            shell=False,
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=timeout,
            check=False,
        )
    except subprocess.TimeoutExpired as exc:
        raise GateError("REGISTRY_TIMEOUT", str(exc)) from exc


def normalized_snapshot(value: dict[str, Any]) -> dict[str, Any]:
    result = copy.deepcopy(value)
    result["GeneratedAtUtc"] = "<NORMALIZED>"
    return result


def entry_by_path(snapshot: dict[str, Any]) -> dict[str, dict[str, Any]]:
    return {entry.get("ManifestRelativePath"): entry for entry in snapshot["Modules"]}


def verify_snapshot(snapshot: dict[str, Any]) -> int:
    require(TOP_FIELDS.issubset(snapshot), "SNAPSHOT_TOP_FIELDS")
    require(snapshot["RegistrySnapshotType"] == "U002_MODULE_REGISTRY_SNAPSHOT", "SNAPSHOT_TYPE")
    require(snapshot["RegistrySnapshotVersion"] == "0.1", "SNAPSHOT_VERSION")
    require(snapshot["EngineVersion"] == ENGINE_VERSION, "ENGINE_VERSION")
    require(snapshot["ScanPolicy"] == "STATIC_MANIFEST_ONLY_NO_MODULE_EXECUTION", "SCAN_POLICY")
    require(snapshot["ModuleCount"] == 10, "MODULE_COUNT", str(snapshot.get("ModuleCount")))
    require(snapshot["StatusCounts"] == EXPECTED_COUNTS, "STATUS_COUNTS", repr(snapshot.get("StatusCounts")))
    modules = snapshot["Modules"]
    require(isinstance(modules, list) and len(modules) == 10, "MODULES_LENGTH")
    for entry in modules:
        require(isinstance(entry, dict) and ENTRY_FIELDS.issubset(entry), "ENTRY_FIELDS")
        require(not Path(entry["ManifestRelativePath"]).is_absolute(), "ABSOLUTE_MANIFEST_PATH")
        require(entry["RegistryStatus"] in EXPECTED_COUNTS, "UNKNOWN_STATUS")
        require(entry["ReasonCodes"] == sorted(entry["ReasonCodes"]), "REASON_ORDER")
    ordered = sorted(
        modules,
        key=lambda entry: (
            entry.get("Stage") or "",
            entry.get("ModuleId") or "",
            entry.get("ModuleVersion") or "",
            entry.get("ManifestRelativePath") or "",
        ),
    )
    require(modules == ordered, "MODULE_ORDER")
    by_path = entry_by_path(snapshot)

    expected = {
        "01_valid_static/module_manifest.json": ("SELFTEST_REQUIRED", "SELFTEST_NOT_YET_ACCEPTED"),
        "02_invalid_json/module_manifest.json": ("INVALID_MANIFEST", "MANIFEST_JSON_INVALID"),
        "03_contract_invalid/module_manifest.json": ("INVALID_MANIFEST", "CONTRACT_VALIDATION_FAILED"),
        "04_incompatible/module_manifest.json": ("INCOMPATIBLE", "ENGINE_VERSION_OUT_OF_RANGE"),
        "05_missing_entrypoint/module_manifest.json": ("DEPENDENCY_MISSING", "ENTRYPOINT_MISSING"),
        "06_missing_config/module_manifest.json": ("DEPENDENCY_MISSING", "CONFIG_SCHEMA_MISSING"),
        "07_bad_implementation_hash/module_manifest.json": ("QUARANTINED", "IMPLEMENTATION_HASH_MISMATCH"),
        "08_bad_config_hash/module_manifest.json": ("QUARANTINED", "CONFIG_SCHEMA_HASH_MISMATCH"),
        "09_duplicate_a/module_manifest.json": ("QUARANTINED", "DUPLICATE_MODULE_IDENTITY"),
        "10_duplicate_b/module_manifest.json": ("QUARANTINED", "DUPLICATE_MODULE_IDENTITY"),
    }
    require(set(by_path) == set(expected), "MANIFEST_PATH_SET")
    for path, (status, reason) in expected.items():
        entry = by_path[path]
        require(entry["RegistryStatus"] == status, "STATUS_MISMATCH", path)
        require(reason in entry["ReasonCodes"], "REASON_MISMATCH", path)

    hash_checks = 0
    for path in (
        "01_valid_static/module_manifest.json",
        "04_incompatible/module_manifest.json",
        "09_duplicate_a/module_manifest.json",
        "10_duplicate_b/module_manifest.json",
    ):
        entry = by_path[path]
        require(entry["ImplementationSha256Observed"] == entry["ImplementationSha256Declared"], "IMPLEMENTATION_HASH_LINK", path)
        require(entry["ConfigurationSchemaSha256Observed"] == entry["ConfigurationSchemaSha256Declared"], "CONFIG_HASH_LINK", path)
        hash_checks += 2
    for path in (
        "07_bad_implementation_hash/module_manifest.json",
        "08_bad_config_hash/module_manifest.json",
    ):
        entry = by_path[path]
        require(entry["ManifestSha256"] is not None, "MANIFEST_HASH_MISSING", path)
        require(len(entry["ManifestSha256"]) == 64, "MANIFEST_HASH_LENGTH", path)
        hash_checks += 2
    require(hash_checks == 12, "HASH_CHECK_COUNT", str(hash_checks))
    return hash_checks


def no_execution_markers(modules_root: Path) -> bool:
    return not any(modules_root.rglob("EXECUTED.marker"))


def run_gate(args: argparse.Namespace) -> str:
    registry = Path(args.registry).resolve()
    schema_bundle = Path(args.schema_bundle).resolve()
    contract_validator = Path(args.contract_validator).resolve()
    fixture_manifest = Path(args.fixture_manifest).resolve()
    out = Path(args.out).resolve()
    for path, code in (
        (registry, "REGISTRY_MISSING"),
        (schema_bundle, "SCHEMA_MISSING"),
        (contract_validator, "VALIDATOR_MISSING"),
        (fixture_manifest, "FIXTURE_MISSING"),
    ):
        require(path.is_file(), code, str(path))
    ensure_clean_output(out)

    fixture_root = out / "00_FIXTURES" / "modules"
    runs_root = out / "01_RUNS"
    evidence_root = out / "02_EVIDENCE"
    fixture_root.mkdir(parents=True)
    runs_root.mkdir(parents=True)
    evidence_root.mkdir(parents=True)
    build_fixture_tree(fixture_root, fixture_manifest)

    snapshots: list[dict[str, Any]] = []
    child_records: list[dict[str, Any]] = []
    for index in (1, 2):
        output_path = runs_root / f"registry_snapshot_{index}.json"
        command = registry_command(registry, fixture_root, schema_bundle, contract_validator, output_path)
        result = run_child(command)
        atomic_text(runs_root / f"run_{index}.stdout.txt", bounded(result.stdout))
        atomic_text(runs_root / f"run_{index}.stderr.txt", bounded(result.stderr))
        require(result.returncode == 0, "REGISTRY_CHILD_FAILED", f"run={index} rc={result.returncode}")
        require(output_path.is_file(), "SNAPSHOT_MISSING", f"run={index}")
        snapshot = load_object(output_path, "SNAPSHOT_JSON_INVALID")
        verify_snapshot(snapshot)
        snapshots.append(snapshot)
        child_records.append(
            {
                "Run": index,
                "ExitCode": result.returncode,
                "SnapshotSha256": sha256_file(output_path),
                "StdoutSha256": sha256_file(runs_root / f"run_{index}.stdout.txt"),
                "StderrSha256": sha256_file(runs_root / f"run_{index}.stderr.txt"),
            }
        )

    require(normalized_snapshot(snapshots[0]) == normalized_snapshot(snapshots[1]), "REPEATABILITY")
    require(no_execution_markers(fixture_root), "MODULE_CODE_EXECUTED")

    evidence = {
        "GateId": "UDP002.U21B.AcceptanceGate",
        "GateVersion": "0.1",
        "Status": "PASS",
        "RegistryFileName": registry.name,
        "RegistrySha256": sha256_file(registry),
        "SchemaBundleFileName": schema_bundle.name,
        "SchemaBundleSha256": sha256_file(schema_bundle),
        "ContractValidatorFileName": contract_validator.name,
        "ContractValidatorSha256": sha256_file(contract_validator),
        "FixtureManifestFileName": fixture_manifest.name,
        "FixtureManifestSha256": sha256_file(fixture_manifest),
        "EngineVersion": ENGINE_VERSION,
        "ExpectedStatusCounts": EXPECTED_COUNTS,
        "ChildRuns": child_records,
        "ModuleCodeExecuted": False,
        "RepeatabilityAfterGeneratedAtNormalization": True,
        "ReturnCapsule": SUCCESS,
    }
    atomic_json(evidence_root / "U21B_ACCEPTANCE_EVIDENCE.json", evidence)
    atomic_text(
        evidence_root / "U21B_ACCEPTANCE_REPORT.txt",
        "\n".join(
            [
                "UDP002 U21B ACCEPTANCE REPORT",
                "",
                "Status: PASS",
                "Registry treated as an opaque child process.",
                "Two scans independently produced the required ten classifications.",
                "Module implementation code was not executed.",
                "Repeatability passed after normalizing GeneratedAtUtc only.",
                "",
                SUCCESS,
                "",
            ]
        ),
    )
    atomic_text(evidence_root / "U21B_RETURN_CAPSULE.txt", SUCCESS + "\n")
    return SUCCESS


def run_selftest() -> str:
    checks: list[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            reason = safe_slug(name).upper()
            raise GateError(f"GATE_SELFTEST_{reason}", name)
        checks.append(name)

    check("sha-bytes", sha256_bytes(b"abc") == hashlib.sha256(b"abc").hexdigest())
    check("canonical-order", canonical_bytes({"b": 1, "a": 2}) == canonical_bytes({"a": 2, "b": 1}))
    check("bounded-short", bounded("abc") == "abc")
    check("bounded-long", bounded("x" * 20, 5).endswith("[TRUNCATED]"))
    check("expected-status-six", len(EXPECTED_COUNTS) == 6)
    check("expected-module-ten", sum(EXPECTED_COUNTS.values()) == 10)
    check("top-fields", "Modules" in TOP_FIELDS and "StatusCounts" in TOP_FIELDS)
    check("entry-fields", "RegistryStatus" in ENTRY_FIELDS and "ManifestSha256" in ENTRY_FIELDS)
    check("slug", safe_slug("A-B.C") == "A_B_C")
    normalized = normalized_snapshot({"GeneratedAtUtc": "now", "Modules": []})
    check("normalize-time", normalized["GeneratedAtUtc"] == "<NORMALIZED>")
    with tempfile.TemporaryDirectory(prefix="u21bgate_selftest_") as temporary:
        root = Path(temporary)
        output = root / "out"
        ensure_clean_output(output)
        check("clean-output-create", output.is_dir())
        atomic_json(output / "value.json", {"ok": True})
        check("atomic-json", load_object(output / "value.json", "TEST")["ok"] is True)
        atomic_text(output / "value.txt", "ok\n")
        check("atomic-text", (output / "value.txt").read_text(encoding="utf-8") == "ok\n")
        check("file-sha", sha256_file(output / "value.txt") == sha256_bytes(b"ok\n"))
        marker_root = root / "markers"
        marker_root.mkdir()
        check("no-marker", no_execution_markers(marker_root))
        (marker_root / "EXECUTED.marker").write_text("x", encoding="utf-8")
        check("marker-detected", not no_execution_markers(marker_root))
    require(len(checks) == 16, "GATE_SELFTEST_COUNT", str(len(checks)))
    return SELFTEST_SUCCESS


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description="Independent black-box acceptance Gate for UDP002 U21B")
    mode = result.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--run", action="store_true")
    result.add_argument("--registry")
    result.add_argument("--schema-bundle")
    result.add_argument("--contract-validator")
    result.add_argument("--fixture-manifest")
    result.add_argument("--out")
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    try:
        if args.self_test:
            print(run_selftest())
            return 0
        missing = [
            name
            for name in ("registry", "schema_bundle", "contract_validator", "fixture_manifest", "out")
            if not getattr(args, name)
        ]
        require(not missing, "MISSING_ARGUMENTS", ",".join(missing))
        print(run_gate(args))
        return 0
    except GateError as exc:
        print(f"U21BG1-ERROR-{exc.code} C-7")
        return 7
    except Exception as exc:
        print(f"U21BG1-ERROR-UNEXPECTED-{type(exc).__name__} C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
