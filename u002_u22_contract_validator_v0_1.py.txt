#!/usr/bin/env python3
"""Strict standard-library validator for U002 U22 machine contracts."""

from __future__ import annotations

import argparse
import json
from pathlib import Path, PurePosixPath
import re
import sys
from typing import Any, Dict, List, Mapping, Sequence


PROGRAM_ID = "U002-U22-CONTRACT-VALIDATOR"
VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"
STAGES = ("S0", "S1", "S2", "S3", "S4", "S5", "S6", "S7")
SCHEMAS = {
    "MODULE": "00_CONTROL/contracts/u002_module_manifest_schema_v0_1.json",
    "ARTIFACT": "00_CONTROL/contracts/u002_artifact_manifest_schema_v0_1.json",
    "RUN": "00_CONTROL/contracts/u002_run_configuration_schema_v0_1.json",
    "ALGORITHM": "00_CONTROL/contracts/u002_algorithm_card_schema_v0_1.json",
}


class ContractError(RuntimeError):
    def __init__(self, reason: str, path: str = "$"):
        self.reason = stable(reason)
        self.path = path[:240]
        super().__init__(self.reason)


def stable(value: object) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")[:80] or "INTERNAL"


def reject_duplicate_pairs(pairs: list) -> dict:
    result = {}
    for key, value in pairs:
        if key in result:
            raise ContractError("DUPLICATE_JSON_KEY", "$" + "." + str(key))
        result[key] = value
    return result


def reject_constant(value: str) -> None:
    raise ContractError("NONFINITE_JSON", "$")


def load_json(path: Path, reason: str) -> Any:
    if not path.is_file() or path.is_symlink():
        raise ContractError(reason, str(path))
    try:
        return json.loads(path.read_text(encoding="utf-8-sig"),
                          object_pairs_hook=reject_duplicate_pairs,
                          parse_constant=reject_constant)
    except ContractError:
        raise
    except Exception as exc:
        raise ContractError(reason, str(path)) from exc


def json_type(value: Any) -> str:
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, int):
        return "integer"
    if isinstance(value, float):
        return "number"
    if isinstance(value, str):
        return "string"
    if isinstance(value, list):
        return "array"
    if isinstance(value, dict):
        return "object"
    return type(value).__name__


def type_matches(value: Any, expected: object) -> bool:
    allowed = expected if isinstance(expected, list) else [expected]
    actual = json_type(value)
    return actual in allowed or (actual == "integer" and "number" in allowed)


def schema_accepts(value: Any, schema: Mapping[str, Any]) -> bool:
    try:
        validate_schema(value, schema, "$")
        return True
    except ContractError:
        return False


def validate_schema(value: Any, schema: Mapping[str, Any], path: str) -> None:
    if not isinstance(schema, dict):
        raise ContractError("SCHEMA_NODE", path)
    for branch in schema.get("allOf", []):
        validate_schema(value, branch, path)
    if "if" in schema and schema_accepts(value, schema["if"]):
        if "then" in schema:
            validate_schema(value, schema["then"], path)
    expected_type = schema.get("type")
    if expected_type is not None and not type_matches(value, expected_type):
        raise ContractError("TYPE", path)
    if "const" in schema and value != schema["const"]:
        raise ContractError("CONST", path)
    if "enum" in schema and value not in schema["enum"]:
        raise ContractError("ENUM", path)
    if isinstance(value, dict):
        required = schema.get("required", [])
        for key in required:
            if key not in value:
                raise ContractError("REQUIRED", path + "." + str(key))
        properties = schema.get("properties", {})
        for key, item in value.items():
            child = path + "." + str(key)
            if key in properties:
                validate_schema(item, properties[key], child)
            else:
                additional = schema.get("additionalProperties", True)
                if additional is False:
                    raise ContractError("ADDITIONAL_PROPERTY", child)
                if isinstance(additional, dict):
                    validate_schema(item, additional, child)
    if isinstance(value, list):
        if len(value) < int(schema.get("minItems", 0)):
            raise ContractError("MIN_ITEMS", path)
        if schema.get("uniqueItems"):
            identities = [json.dumps(item, sort_keys=True, separators=(",", ":"), allow_nan=False)
                          for item in value]
            if len(identities) != len(set(identities)):
                raise ContractError("UNIQUE_ITEMS", path)
        if "items" in schema:
            for index, item in enumerate(value):
                validate_schema(item, schema["items"], f"{path}[{index}]")
    if isinstance(value, str):
        if len(value) < int(schema.get("minLength", 0)):
            raise ContractError("MIN_LENGTH", path)
        if "pattern" in schema:
            try:
                matched = re.search(schema["pattern"], value)
            except re.error as exc:
                raise ContractError("SCHEMA_PATTERN", path) from exc
            if matched is None:
                raise ContractError("PATTERN", path)
    if isinstance(value, (int, float)) and not isinstance(value, bool):
        if "minimum" in schema and value < schema["minimum"]:
            raise ContractError("MINIMUM", path)


def schema_for(root: Path, kind: str) -> dict:
    schema = load_json(root / SCHEMAS[kind], "SCHEMA_MISSING")
    if not isinstance(schema, dict) or schema.get("$schema") != "https://json-schema.org/draft/2020-12/schema":
        raise ContractError("SCHEMA_IDENTITY", str(root / SCHEMAS[kind]))
    return schema


def validate_module(value: dict) -> None:
    if set(value["RequiredInputs"]) & set(value["OptionalInputs"]):
        raise ContractError("INPUT_ROLE_OVERLAP", "$.RequiredInputs")
    if set(value["RequiresArtifacts"]) & set(value["ProvidesArtifacts"]):
        raise ContractError("ARTIFACT_SELF_DEPENDENCY", "$.ProvidesArtifacts")
    if value["ConflictPolicy"] == "EXCLUSIVE" and len(value["ProvidesArtifacts"]) != 1:
        raise ContractError("EXCLUSIVE_OUTPUT_COUNT", "$.ProvidesArtifacts")


def safe_relative_path(value: str, path: str) -> None:
    if "\\" in value:
        raise ContractError("PATH_SEPARATOR", path)
    candidate = PurePosixPath(value)
    if candidate.is_absolute() or not candidate.parts or any(part in ("", ".", "..") for part in candidate.parts):
        raise ContractError("RELATIVE_PATH", path)


def validate_artifact(value: dict) -> None:
    payload_paths: List[str] = []
    primary = 0
    for index, payload in enumerate(value["Payloads"]):
        safe_relative_path(payload["RelativePath"], f"$.Payloads[{index}].RelativePath")
        payload_paths.append(payload["RelativePath"])
        if payload["PayloadKind"] == "PRIMARY":
            primary += 1
    if len(payload_paths) != len(set(payload_paths)):
        raise ContractError("DUPLICATE_PAYLOAD_PATH", "$.Payloads")
    if primary == 0:
        raise ContractError("PRIMARY_PAYLOAD_REQUIRED", "$.Payloads")


def validate_run(value: dict) -> None:
    start, end = value["FrameStart"], value["FrameEnd"]
    if value["RunScope"] == "FULL_CASE":
        if start is not None or end is not None:
            raise ContractError("FULL_CASE_BOUNDS", "$.FrameStart")
    else:
        if start is None or end is None or start > end:
            raise ContractError("RANGE_BOUNDS", "$.FrameStart")
    plan = value["ModulePlan"]
    sequences = [item["Sequence"] for item in plan]
    if sequences != list(range(1, len(plan) + 1)):
        raise ContractError("PLAN_SEQUENCE", "$.ModulePlan")
    positions = [STAGES.index(item["Stage"]) for item in plan]
    if positions != sorted(positions):
        raise ContractError("PLAN_STAGE_ORDER", "$.ModulePlan")
    identities = [(item["ModuleId"], item["ModuleVersion"]) for item in plan]
    if len(identities) != len(set(identities)):
        raise ContractError("PLAN_DUPLICATE_MODULE", "$.ModulePlan")


def validate_algorithm(value: dict) -> None:
    if not value["Inputs"] or not value["Outputs"]:
        raise ContractError("ALGORITHM_IO", "$.Inputs")
    if not value["Assumptions"]:
        raise ContractError("ALGORITHM_ASSUMPTIONS", "$.Assumptions")
    if not value["Limitations"]:
        raise ContractError("ALGORITHM_LIMITATIONS", "$.Limitations")


SEMANTIC = {
    "MODULE": validate_module,
    "ARTIFACT": validate_artifact,
    "RUN": validate_run,
    "ALGORITHM": validate_algorithm,
}


def validate_document(root: Path, kind: str, document: Any) -> None:
    if kind not in SCHEMAS:
        raise ContractError("KIND")
    schema = schema_for(root, kind)
    validate_schema(document, schema, "$")
    if not isinstance(document, dict):
        raise ContractError("DOCUMENT_TYPE")
    SEMANTIC[kind](document)

def self_test(root: Path) -> str:
    checks: List[bool] = []
    def ck(value: bool) -> None:
        if not value:
            raise ContractError("SELFTEST")
        checks.append(True)
    ck(PROGRAM_ID.startswith("U002")); ck(VERSION == "0.1"); ck(CONTRACT_VERSION == "U22.0.1")
    ck(len(STAGES) == 8)
    fixtures = valid_fixtures()
    for kind, value in fixtures.items():
        validate_document(root, kind, value); ck(True)
    broken = json.loads(json.dumps(fixtures["MODULE"])); del broken["ModuleId"]
    try: validate_document(root, "MODULE", broken); ck(False)
    except ContractError as exc: ck(exc.reason == "REQUIRED")
    broken = json.loads(json.dumps(fixtures["ARTIFACT"])); broken["Payloads"][0]["RelativePath"] = "../escape.csv"
    try: validate_document(root, "ARTIFACT", broken); ck(False)
    except ContractError as exc: ck(exc.reason == "RELATIVE_PATH")
    broken = json.loads(json.dumps(fixtures["RUN"])); broken["FrameStart"] = 0
    try: validate_document(root, "RUN", broken); ck(False)
    except ContractError: ck(True)
    broken = json.loads(json.dumps(fixtures["ALGORITHM"])); broken["Extra"] = 1
    try: validate_document(root, "ALGORITHM", broken); ck(False)
    except ContractError as exc: ck(exc.reason == "ADDITIONAL_PROPERTY")
    ck(json_type(True) == "boolean"); ck(json_type(1) == "integer"); ck(json_type(1.0) == "number")
    ck(type_matches(1, "number")); ck(not type_matches(True, "integer")); ck(stable("a b/c") == "A_B_C")
    ck(len(SCHEMAS) == 4); ck(set(SEMANTIC) == set(SCHEMAS)); ck(SCHEMAS["MODULE"].endswith(".json"))
    ck(PurePosixPath("a/b.csv").is_absolute() is False); ck(STAGES[0] == "S0" and STAGES[-1] == "S7")
    ck(len(checks) == 23)
    if len(checks) != 24:
        raise ContractError("SELFTEST_COUNT")
    return "U22A1V-ST N-T24-S4-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    for kind in SCHEMAS:
        mode.add_argument("--validate-" + kind.lower())
    parser.add_argument("--workspace-root", required=True)
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    args = parse_args(sys.argv[1:] if argv is None else argv)
    root = Path(args.workspace_root).resolve()
    try:
        if args.self_test:
            print(self_test(root)); return 0
        for kind in SCHEMAS:
            path_value = getattr(args, "validate_" + kind.lower())
            if path_value:
                document = load_json(Path(path_value), "DOCUMENT_JSON")
                validate_document(root, kind, document)
                print(f"U22A1V K-{kind}-V1-X0 C-0")
                return 0
        raise ContractError("MODE")
    except ContractError as exc:
        print(f"U22A1V K-FAIL-V0-X1 C-7 R-{exc.reason} P-{stable(exc.path)}")
        return 7
    except Exception as exc:
        print(f"U22A1V K-FAIL-V0-X1 C-7 R-{stable(exc.__class__.__name__)} P-NA")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
