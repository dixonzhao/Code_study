#!/usr/bin/env python3
"""Finalize a copied G10D2C release only after Gate and post-Gate hygiene pass."""

from __future__ import annotations

import argparse
import subprocess
import sys
import tempfile
from pathlib import Path


PROGRAM_NAME = "u001_g10d2cd5_controlled_gate_finalizer_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2CD5-SELFTEST-PASS-18"
PACKAGE_NAME = "U001_Portable_v0_2"
GATE_NAME = "u001_g10d2c_portable_release_acceptance_gate_v0_1.py"
GATE_PASS = "G10D2CCG1-PASS P12-C4-H12-Z1-R1-X0 C-0"
INCOMPLETE_MARKER = ".G10D2C_INCOMPLETE"


def package_root(out: Path) -> Path:
    return out / PACKAGE_NAME


def cases_root(out: Path) -> Path:
    return package_root(out) / "UDP001" / "03_CASES"


def case_entries(out: Path) -> list[Path]:
    root = cases_root(out)
    if not root.is_dir():
        return []
    return sorted(root.iterdir(), key=lambda path: path.name.lower())


def safe_token(text: str, limit: int = 70) -> str:
    value = "".join(ch if ch.isalnum() or ch in "._-" else "_" for ch in text)
    return value[:limit] or "EMPTY"


def gate_reason(stdout: str, stderr: str, returncode: int) -> str:
    for line in (stdout + "\n" + stderr).splitlines():
        if "G10D2CCG1-FAIL:" in line:
            return safe_token(line.split("G10D2CCG1-FAIL:", 1)[1].split(" C-7", 1)[0])
    return f"GATE_RC{returncode}"


def run_gate(gate: Path, out: Path, temp_root: Path) -> tuple[int, str, str]:
    result = subprocess.run(
        [sys.executable, str(gate.resolve()), "--release-root", str(out.resolve())],
        cwd=str(temp_root),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=2100,
        shell=False,
    )
    (temp_root / "G10D2CD5_GATE_STDOUT.txt").write_text(result.stdout, encoding="utf-8")
    (temp_root / "G10D2CD5_GATE_STDERR.txt").write_text(result.stderr, encoding="utf-8")
    return result.returncode, result.stdout, result.stderr


def emit_case_memos(entries: list[Path]) -> None:
    for index, path in enumerate(entries[:3], 1):
        kind = "D" if path.is_dir() else "F"
        size = path.stat().st_size if path.is_file() else 0
        print(f"M{index}-{safe_token(path.name)}-{kind}-B{size}")


def finalize(udp001_root: Path, out: Path) -> int:
    root = udp001_root.resolve()
    temp_root = root / "04_TEMP"
    out = out.resolve()
    gate = temp_root / GATE_NAME
    marker = out / INCOMPLETE_MARKER
    release_ok = int(out.is_dir() and package_root(out).is_dir() and cases_root(out).is_dir())
    if not release_ok or not gate.is_file():
        print("G10D2CD5 N-R0-E0-G0-P0-H0-Z0-M0-X1 K-RELEASE_OR_GATE C-7")
        return 7
    initial = case_entries(out)
    if initial:
        print(f"G10D2CD5 N-R1-E0-G0-P0-H0-Z0-M0-X0 K-PRE_CASES_{len(initial)} C-7")
        emit_case_memos(initial)
        return 7
    if not marker.is_file():
        print("G10D2CD5 N-R1-E1-G0-P0-H0-Z0-M0-X0 K-INCOMPLETE_MARKER_MISSING C-7")
        return 7
    try:
        rc, stdout, stderr = run_gate(gate, out, temp_root)
    except subprocess.TimeoutExpired:
        print("G10D2CD5 N-R1-E1-G0-P0-H0-Z1-M0-X0 K-GATE_TIMEOUT C-7")
        return 7
    except Exception as exc:
        print(f"G10D2CD5 N-R1-E1-G0-P0-H0-Z0-M0-X1 K-{type(exc).__name__} C-7")
        return 7
    gate_ok = int(rc == 0 and GATE_PASS in stdout)
    hosted = 12 if gate_ok else 0
    after = case_entries(out)
    post_empty = int(not after)
    if not gate_ok:
        print(
            f"G10D2CD5 N-R1-E1-G0-P{post_empty}-H{hosted}-Z1-M0-X0 "
            f"K-{gate_reason(stdout, stderr, rc)} C-7"
        )
        emit_case_memos(after)
        return 7
    if not post_empty:
        print(f"G10D2CD5 N-R1-E1-G1-P0-H12-Z1-M0-X0 K-POST_CASES_{len(after)} C-7")
        emit_case_memos(after)
        return 7
    try:
        marker.unlink()
    except OSError as exc:
        print(f"G10D2CD5 N-R1-E1-G1-P1-H12-Z1-M0-X1 K-{type(exc).__name__} C-7")
        return 7
    print("G10D2CD5 N-R1-E1-G1-P1-H12-Z1-M1-X0 C-0")
    return 0


def run_self_test() -> None:
    checks = 0
    with tempfile.TemporaryDirectory() as td:
        base = Path(td)
        out = base / "OUT"
        cases = cases_root(out)
        cases.mkdir(parents=True)
        checks += int(package_root(out).name == PACKAGE_NAME)
        checks += int(cases.name == "03_CASES")
        checks += int(case_entries(out) == [])
        item = cases / "SELFTEST_CASE"
        item.mkdir()
        checks += int(len(case_entries(out)) == 1)
        checks += int(case_entries(out)[0].name == "SELFTEST_CASE")
        checks += int(safe_token("a/b c") == "a_b_c")
        checks += int(gate_reason("G10D2CCG1-FAIL:CASES_NOT_EMPTY C-7", "", 7) == "CASES_NOT_EMPTY")
        checks += int(gate_reason("", "", 9) == "GATE_RC9")
        checks += int(PROGRAM_VERSION == "0.1")
        checks += int(PROGRAM_NAME.endswith("v0_1.py"))
        checks += int(SELFTEST_TOKEN.endswith("PASS-18"))
        checks += int(PACKAGE_NAME.endswith("v0_2"))
        checks += int(GATE_NAME.endswith("v0_1.py"))
        checks += int(GATE_PASS.endswith("C-0"))
        checks += int(INCOMPLETE_MARKER.startswith("."))
        checks += int(len(GATE_PASS.split()) == 3)
        checks += int(sys.executable != "")
        checks += int(base.is_dir())
    if checks != 18:
        raise AssertionError(f"SELFTEST_COUNT:{checks}")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Controlled Gate and post-Gate hygiene finalizer")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.udp001_root or not args.out:
        parser.error("--udp001-root and --out are required")
    return finalize(Path(args.udp001_root), Path(args.out))


if __name__ == "__main__":
    raise SystemExit(main())
