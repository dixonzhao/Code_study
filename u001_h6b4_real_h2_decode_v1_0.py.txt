#!/usr/bin/env python3
"""U001 H6B4 controlled full-source H2 decode and acceptance.

This wrapper deliberately performs one responsibility only:
run the accepted H2 input adapter with the H6B3 compiled request, then
independently verify the five-file adapter boundary.  It does not perform
frame reconstruction, hold/forward-fill, semantic invalid-value policy,
camera arbitration, or analysis.

User-facing lifecycle: prepare -> self-test -> execute.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import re
import shutil
import subprocess
import sys
import time
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, List, Optional, Sequence, Tuple


PROGRAM_ID = "U001_H6B4_REAL_H2_DECODE"
PROGRAM_VERSION = "1.0"
PREPARED_FOLDER = "U1H6B4_PREPARED"
OUTPUT_FOLDER = "U1H6B4_OUTPUT"
PROGRESS_SECONDS = 30
EXECUTION_TIMEOUT_SECONDS = 6 * 60 * 60

EXPECTED_H2_FILES = {
    "u001_adapter_source_updates.csv",
    "u001_adapter_signal_catalog.csv",
    "u001_adapter_quality.json",
    "u001_adapter_run_manifest.json",
    "u001_adapter_completion_capsule.txt",
}

SUPERSEDED_TRANSFER_NAMES = (
    "u001_h6b3_profile_compile_preflight_v1_0.py.txt",
    "U1H6B3_START_HERE_CN_v1_0_2026-08-28.txt",
)


class DecodeGateError(RuntimeError):
    """Expected fail-closed orchestration or acceptance error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def timestamp_token() -> str:
    return datetime.now().strftime("%Y%m%dT%H%M%S_%f")


def write_json(path: Path, value: Dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )


def read_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        raise DecodeGateError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise DecodeGateError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise DecodeGateError(label + "_ROOT_NOT_OBJECT")
    return value


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(4 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise DecodeGateError("U_UDP001_ROOT_NOT_FOUND")
    return root


def version_key(path: Path) -> Tuple[int, ...]:
    values = re.findall(r"\d+", path.name)
    return tuple(int(value) for value in values) if values else (0,)


def locate_adapter(root: Path) -> Path:
    parent = root / "01_PROGRAMS" / "adapters" / "CameraRadar.U001.MatlabCanCsvDbc"
    candidates = [
        folder / "adapter.py"
        for folder in parent.iterdir()
        if folder.is_dir() and (folder / "adapter.py").is_file()
    ] if parent.is_dir() else []
    if not candidates:
        raise DecodeGateError("H2_ADAPTER_NOT_FOUND")
    return max(candidates, key=lambda path: version_key(path.parent))


def required_paths(root: Path) -> Dict[str, Path]:
    b3 = root / "04_TEMP" / "U1H6B3_OUTPUT"
    compiled = b3 / "compiled_profile"
    return {
        "H2Adapter": locate_adapter(root),
        "ControlledRequest": compiled / "u001_h2_controlled_request.json",
        "CompiledProfile": compiled / "u001_compiled_selection_profile.json",
        "SelectedCatalog": compiled / "u001_selected_signal_catalog.csv",
        "H2Preflight": b3 / "h2_preflight" / "u001_adapter_preflight.json",
        "B3Result": b3 / "u001_h6b3_result.json",
        "B3Capsule": b3 / "u001_h6b3_capsule.txt",
    }


def archive_path(path: Path, archive_root: Path) -> None:
    archive_root.mkdir(parents=True, exist_ok=True)
    destination = archive_root / path.name
    counter = 1
    while destination.exists():
        destination = archive_root / f"{path.stem}_{counter}{path.suffix}"
        counter += 1
    shutil.move(str(path), str(destination))


def request_inputs(request: Dict[str, Any]) -> List[Path]:
    values: List[Path] = []
    inputs = request.get("Inputs")
    if not isinstance(inputs, list) or not inputs:
        raise DecodeGateError("CONTROLLED_REQUEST_INPUTS_EMPTY")
    for index, item in enumerate(inputs):
        if not isinstance(item, dict):
            raise DecodeGateError(f"CONTROLLED_REQUEST_INPUT_{index}_NOT_OBJECT")
        raw_path = item.get("Path", item.get("FilePath"))
        if not isinstance(raw_path, str) or not raw_path.strip():
            raise DecodeGateError(f"CONTROLLED_REQUEST_INPUT_{index}_PATH")
        path = Path(raw_path)
        if not path.is_file():
            raise DecodeGateError(f"CONTROLLED_REQUEST_INPUT_{index}_NOT_FOUND")
        values.append(path)
    return values


def validate_business_request(
    request: Dict[str, Any],
    profile: Dict[str, Any],
    preflight: Dict[str, Any],
    b3_result: Dict[str, Any],
) -> Tuple[int, List[Path]]:
    if request.get("TestMode") is True:
        raise DecodeGateError("CONTROLLED_REQUEST_TEST_MODE_TRUE")
    configuration = request.get("Configuration")
    if not isinstance(configuration, dict):
        raise DecodeGateError("CONTROLLED_REQUEST_CONFIGURATION")
    if configuration.get("SelectionMode") != "CONTROLLED_PROFILE":
        raise DecodeGateError("SELECTION_MODE_NOT_CONTROLLED_PROFILE")
    selected = configuration.get("SelectedSignals")
    if not isinstance(selected, list) or not selected:
        raise DecodeGateError("SELECTED_SIGNALS_EMPTY")
    selected_count = len(selected)
    expected_counts = {
        selected_count,
        int(profile.get("EffectiveSelectedCount", -1)),
        int(preflight.get("SelectedSignalCount", -1)),
        int(b3_result.get("CompiledSignalCount", -1)),
        int(b3_result.get("ControlledRequestSignalCount", -1)),
    }
    if len(expected_counts) != 1 or selected_count <= 0:
        raise DecodeGateError("SELECTED_SIGNAL_COUNT_MISMATCH")
    frame_key = configuration.get("FrameKey")
    expected_frame_key = {"Channel": 2, "CAN_ID": "0x620", "SignalName": "FrameCount"}
    if not isinstance(frame_key, dict):
        raise DecodeGateError("FRAME_KEY_NOT_OBJECT")
    normalized_frame_key = {
        "Channel": int(frame_key.get("Channel", -1)),
        "CAN_ID": str(frame_key.get("CAN_ID", "")).lower(),
        "SignalName": str(frame_key.get("SignalName", "")),
    }
    if normalized_frame_key != expected_frame_key:
        raise DecodeGateError("FRAME_KEY_NOT_CH2_0X620_FRAMECOUNT")
    if preflight.get("Status") != "READY" or preflight.get("CompletionCode") != 0:
        raise DecodeGateError("H2_PREFLIGHT_NOT_READY")
    if preflight.get("TestMode") is True or preflight.get("FrameKeyConfigured") is not True:
        raise DecodeGateError("H2_PREFLIGHT_MODE_OR_FRAME_KEY")
    if b3_result.get("CompletionCode") != 0:
        raise DecodeGateError("B3_RESULT_NOT_C0")
    if b3_result.get("NextRequiredAction") != "RUN_FULL_H2_DECODE_WITH_COMPILED_REQUEST":
        raise DecodeGateError("B3_NEXT_ACTION")
    return selected_count, request_inputs(request)


def dependency_hashes(paths: Dict[str, Path]) -> Dict[str, str]:
    return {label: sha256_file(path) for label, path in paths.items() if path.is_file()}


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        temp_root = root / "04_TEMP"
        prepared_root = temp_root / PREPARED_FOLDER
        output_root = temp_root / OUTPUT_FOLDER
        archive_root = root / "A-Archive" / ("U1H6B4_TEMP_HYGIENE_" + timestamp_token())

        print("[PREPARE] U001 H6B4 controlled full-source H2 decode")
        print("[PREPARE] Why: establish a clean, immutable input boundary before the long real-source run.")

        if prepared_root.exists():
            shutil.rmtree(prepared_root)
            print(f"[CLEAN] 04_TEMP/{PREPARED_FOLDER} | REMOVED | Why: invalidate stale prepare receipts.")
        if output_root.exists():
            archive_path(output_root, archive_root)
            print(f"[CLEAN] 04_TEMP/{OUTPUT_FOLDER} | ARCHIVED | Why: prevent stale decode evidence reuse.")

        moved = 0
        for name in SUPERSEDED_TRANSFER_NAMES:
            path = temp_root / name
            if path.exists():
                archive_path(path, archive_root)
                moved += 1
                print(f"[CLEAN] 04_TEMP/{name} | ARCHIVED | Why: superseded transfer file.")

        paths = required_paths(root)
        reasons = {
            "H2Adapter": "Accepted streaming decoder that owns source-format interpretation.",
            "ControlledRequest": "Exact 22-signal production request compiled by H6B3.",
            "CompiledProfile": "Independent record of selected count, frame key, and resource policy.",
            "SelectedCatalog": "Human-readable record of the exact decoded variables.",
            "H2Preflight": "Proof that the real CSV/DBC source definition accepts the request.",
            "B3Result": "Machine-readable authorization boundary for the full H2 decode.",
            "B3Capsule": "Compact proof that H6B3 completed successfully on this host.",
        }
        missing: List[str] = []
        for label, path in paths.items():
            found = path.is_file()
            print(f"[CHECK] {label} | {'FOUND' if found else 'MISSING'} | Why: {reasons[label]}")
            if not found:
                missing.append(label)
        for kept in ("U1H6B0_OUTPUT", "U1H6B1_OUTPUT", "U1H6B2_OUTPUT", "U1H6B3_OUTPUT"):
            print(f"[KEEP] 04_TEMP/{kept} | Why: accepted upstream evidence required for traceability.")
        if missing:
            raise DecodeGateError("PREPARE_MISSING:" + ".".join(missing))

        capsule = paths["B3Capsule"].read_text(encoding="utf-8").strip()
        if not capsule.endswith("C0"):
            raise DecodeGateError("B3_CAPSULE_NOT_C0")
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        profile = read_json(paths["CompiledProfile"], "COMPILED_PROFILE")
        preflight = read_json(paths["H2Preflight"], "H2_PREFLIGHT")
        b3_result = read_json(paths["B3Result"], "B3_RESULT")
        selected_count, inputs = validate_business_request(request, profile, preflight, b3_result)
        input_bytes = sum(path.stat().st_size for path in inputs)
        free_bytes = shutil.disk_usage(root).free
        required_free_bytes = max(4 * 1024**3, input_bytes * 4)
        print(
            f"[CHECK] Disk capacity | {'PASS' if free_bytes >= required_free_bytes else 'FAIL'} | "
            f"input={input_bytes} bytes; free={free_bytes} bytes; required={required_free_bytes} bytes | "
            "Why: reserve space for the long update stream, evidence, and safe recovery."
        )
        if free_bytes < required_free_bytes:
            raise DecodeGateError("INSUFFICIENT_FREE_DISK")

        prepared_root.mkdir(parents=True)
        receipt = {
            "PrepareReceiptSchema": "U001_H6B4_PREPARE_RECEIPT_V1_0",
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "PreparedUtc": utc_now(),
            "SelectedSignalCount": selected_count,
            "InputFileCount": len(inputs),
            "InputTotalBytes": input_bytes,
            "FreeBytesObserved": free_bytes,
            "RequiredFreeBytes": required_free_bytes,
            "DependencySHA256": dependency_hashes(paths),
            "SupersededFilesArchived": moved,
            "Status": "READY",
            "CompletionCode": 0,
        }
        write_json(prepared_root / "u001_h6b4_prepare_receipt.json", receipt)
        prepare_capsule = f"U1H6B4P-C{len(paths)}-N{selected_count}-I{len(inputs)}-A{moved}-X0-C0"
        (prepared_root / "u001_h6b4_prepare_capsule.txt").write_text(
            prepare_capsule + "\n", encoding="utf-8"
        )
        print("[READY] H6B4 production request, source files, capacity, and evidence boundary are ready.")
        print(prepare_capsule)
        return 0
    except Exception as exc:
        print(f"U1H6B4P-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def require_prepare(root: Path) -> Dict[str, Any]:
    receipt = read_json(
        root / "04_TEMP" / PREPARED_FOLDER / "u001_h6b4_prepare_receipt.json",
        "PREPARE_RECEIPT",
    )
    if receipt.get("ProgramId") != PROGRAM_ID or receipt.get("ProgramVersion") != PROGRAM_VERSION:
        raise DecodeGateError("PREPARE_RECEIPT_IDENTITY")
    if receipt.get("Status") != "READY" or receipt.get("CompletionCode") != 0:
        raise DecodeGateError("PREPARE_RECEIPT_STATUS")
    current = dependency_hashes(required_paths(root))
    if current != receipt.get("DependencySHA256"):
        raise DecodeGateError("PREPARED_DEPENDENCY_CHANGED")
    return receipt


def run_child(command: List[str], cwd: Path, label: str, timeout: int = 600) -> subprocess.CompletedProcess[str]:
    completed = subprocess.run(
        command,
        cwd=cwd,
        capture_output=True,
        text=True,
        timeout=timeout,
        check=False,
    )
    if completed.returncode != 0:
        detail = (completed.stderr or completed.stdout or "NO_CHILD_OUTPUT")[-1600:]
        raise DecodeGateError(label + "_EXIT:" + detail.replace("\n", " "))
    return completed


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        receipt = require_prepare(root)
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        profile = read_json(paths["CompiledProfile"], "COMPILED_PROFILE")
        preflight = read_json(paths["H2Preflight"], "H2_PREFLIGHT")
        b3_result = read_json(paths["B3Result"], "B3_RESULT")

        print("[SELFTEST] H2 adapter isolated tests | Why: verify DBC parsing, byte decoding, streaming, and contracts.")
        adapter_test = run_child(
            [sys.executable, str(paths["H2Adapter"]), "--self-test"],
            paths["H2Adapter"].parent,
            "H2_ADAPTER_SELFTEST",
        )
        if "SELFTEST-PASS" not in adapter_test.stdout.upper():
            raise DecodeGateError("H2_ADAPTER_SELFTEST_PASS_TOKEN")
        print("[PASS] H2 adapter isolated synthetic tests completed.")

        print("[SELFTEST] Controlled request semantics | Why: prevent an unbounded or non-reproducible production run.")
        selected_count, _ = validate_business_request(request, profile, preflight, b3_result)
        configuration = request["Configuration"]
        expected = {
            "WorkerCount": 1,
            "MaxResidentMemoryMB": 512,
            "WriteOrderPolicy": "SOURCE_ORDER",
        }
        for key, value in expected.items():
            if configuration.get(key) != value:
                raise DecodeGateError("RESOURCE_POLICY_" + key.upper())
        if configuration.get("MaxSourceRows") not in (None, "", 0, "0"):
            raise DecodeGateError("MAX_SOURCE_ROWS_NOT_FULL_SOURCE")
        if int(receipt.get("SelectedSignalCount", -1)) != selected_count:
            raise DecodeGateError("PREPARE_SELECTED_SIGNAL_COUNT")
        print("[PASS] Full-source, single-reader, 512 MB target, source-order policy verified.")
        print("U1H6B4-ST-T2-N%d-X0-C0" % selected_count)
        return 0
    except Exception as exc:
        print(f"U1H6B4-ST-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def run_decode_with_progress(adapter: Path, request: Path, output: Path, log_root: Path) -> int:
    stdout_path = log_root / "h2_decode_stdout.txt"
    stderr_path = log_root / "h2_decode_stderr.txt"
    command = [sys.executable, str(adapter), "--run", "--request", str(request), "--out", str(output)]
    started = time.monotonic()
    next_progress = started
    with stdout_path.open("w", encoding="utf-8") as stdout_stream, stderr_path.open(
        "w", encoding="utf-8"
    ) as stderr_stream:
        process = subprocess.Popen(
            command,
            cwd=adapter.parent,
            stdout=stdout_stream,
            stderr=stderr_stream,
            text=True,
        )
        while True:
            return_code = process.poll()
            now = time.monotonic()
            if now >= next_progress:
                update_path = output / "u001_adapter_source_updates.csv"
                output_bytes = update_path.stat().st_size if update_path.is_file() else 0
                elapsed_seconds = int(now - started)
                print(
                    f"[PROGRESS] H2 decode | elapsed={elapsed_seconds}s | "
                    f"update_file_bytes={output_bytes}",
                    flush=True,
                )
                next_progress = now + PROGRESS_SECONDS
            if return_code is not None:
                return return_code
            if now - started > EXECUTION_TIMEOUT_SECONDS:
                process.kill()
                process.wait()
                raise DecodeGateError("H2_DECODE_TIMEOUT_6H")
            time.sleep(1.0)


def validate_h2_output(h2_root: Path, selected_count: int) -> Tuple[Dict[str, Any], Dict[str, Any]]:
    actual_files = {path.name for path in h2_root.iterdir() if path.is_file()}
    if actual_files != EXPECTED_H2_FILES:
        missing = sorted(EXPECTED_H2_FILES - actual_files)
        extra = sorted(actual_files - EXPECTED_H2_FILES)
        raise DecodeGateError(f"H2_FILESET_MISSING_{missing}_EXTRA_{extra}")
    manifest = read_json(h2_root / "u001_adapter_run_manifest.json", "H2_MANIFEST")
    quality = read_json(h2_root / "u001_adapter_quality.json", "H2_QUALITY")
    child_capsule = (h2_root / "u001_adapter_completion_capsule.txt").read_text(encoding="utf-8").strip()
    if not child_capsule.endswith("C0"):
        raise DecodeGateError("H2_CHILD_CAPSULE_NOT_C0")
    if manifest.get("Status") != "COMPLETE" or manifest.get("CompletionCode") != 0:
        raise DecodeGateError("H2_MANIFEST_STATUS")
    if manifest.get("Errors") or quality.get("Errors"):
        raise DecodeGateError("H2_ERRORS_NOT_EMPTY")
    if int(quality.get("AbsolutePathLeakCount", -1)) != 0:
        raise DecodeGateError("H2_ABSOLUTE_PATH_LEAK")
    if int(quality.get("SelectedSignalCount", -1)) != selected_count:
        raise DecodeGateError("H2_QUALITY_SIGNAL_COUNT")
    if int(quality.get("SourceRowCount", 0)) <= 0:
        raise DecodeGateError("H2_SOURCE_ROW_COUNT")
    if int(quality.get("DecodedUpdateCount", 0)) <= 0:
        raise DecodeGateError("H2_DECODED_UPDATE_COUNT")
    if int(quality.get("FrameKeyUpdateCount", 0)) <= 0:
        raise DecodeGateError("H2_FRAME_KEY_UPDATE_COUNT")
    if int(quality.get("SelectedMessageGroupCount", 0)) <= 0:
        raise DecodeGateError("H2_SELECTED_MESSAGE_GROUP_COUNT")

    configuration = manifest.get("Configuration")
    if not isinstance(configuration, dict):
        raise DecodeGateError("H2_MANIFEST_CONFIGURATION")
    expected_configuration = {
        "SelectionMode": "CONTROLLED_PROFILE",
        "SelectedSignalCount": selected_count,
        "MaxSourceRows": None,
        "WorkerCountEffective": 1,
        "MaxResidentMemoryMBTarget": 512,
        "WriteOrderPolicy": "SOURCE_ORDER",
        "StreamingImplementation": "ROW_STREAM_SINGLE_READER_SINGLE_WRITER",
    }
    for key, expected in expected_configuration.items():
        if configuration.get(key) != expected:
            raise DecodeGateError("H2_MANIFEST_CONFIG_" + key.upper())
    fingerprints = manifest.get("InputFingerprints")
    if not isinstance(fingerprints, list) or not fingerprints:
        raise DecodeGateError("H2_INPUT_FINGERPRINTS")
    if any(not isinstance(item, dict) or len(str(item.get("SHA256", ""))) != 64 for item in fingerprints):
        raise DecodeGateError("H2_INPUT_FINGERPRINT_SHA256")

    artifacts = manifest.get("Artifacts")
    if not isinstance(artifacts, list) or len(artifacts) != 3:
        raise DecodeGateError("H2_ARTIFACT_COUNT")
    expected_artifact_names = {
        "u001_adapter_source_updates.csv",
        "u001_adapter_signal_catalog.csv",
        "u001_adapter_quality.json",
    }
    observed_names = {str(item.get("FileName", "")) for item in artifacts if isinstance(item, dict)}
    if observed_names != expected_artifact_names:
        raise DecodeGateError("H2_ARTIFACT_NAMES")
    for item in artifacts:
        if not isinstance(item, dict):
            raise DecodeGateError("H2_ARTIFACT_NOT_OBJECT")
        artifact_path = h2_root / str(item["FileName"])
        if artifact_path.stat().st_size != int(item.get("SizeBytes", -1)):
            raise DecodeGateError("H2_ARTIFACT_SIZE_" + artifact_path.name)
        if sha256_file(artifact_path) != str(item.get("SHA256", "")):
            raise DecodeGateError("H2_ARTIFACT_SHA256_" + artifact_path.name)
    return manifest, quality


def execute(root_argument: Optional[str]) -> int:
    output_root: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        receipt = require_prepare(root)
        paths = required_paths(root)
        output_root = root / "04_TEMP" / OUTPUT_FOLDER
        if output_root.exists():
            raise DecodeGateError("OUTPUT_ALREADY_EXISTS_RUN_PREPARE_AGAIN")
        output_root.mkdir(parents=True)
        h2_root = output_root / "h2_real"

        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        profile = read_json(paths["CompiledProfile"], "COMPILED_PROFILE")
        preflight = read_json(paths["H2Preflight"], "H2_PREFLIGHT")
        b3_result = read_json(paths["B3Result"], "B3_RESULT")
        selected_count, _ = validate_business_request(request, profile, preflight, b3_result)
        if int(receipt.get("SelectedSignalCount", -1)) != selected_count:
            raise DecodeGateError("PREPARE_SELECTED_SIGNAL_COUNT")

        print("[EXECUTE] Stage 1/3 | Run the accepted H2 adapter over the complete real source.")
        return_code = run_decode_with_progress(
            paths["H2Adapter"], paths["ControlledRequest"], h2_root, output_root
        )
        if return_code != 0:
            stderr = output_root / "h2_decode_stderr.txt"
            stdout = output_root / "h2_decode_stdout.txt"
            detail_path = stderr if stderr.is_file() and stderr.stat().st_size else stdout
            detail = detail_path.read_text(encoding="utf-8", errors="replace")[-1800:] if detail_path.is_file() else ""
            raise DecodeGateError(f"H2_ADAPTER_EXIT_{return_code}:{detail.replace(chr(10), ' ')}")

        print("[EXECUTE] Stage 2/3 | Validate file set, counts, resource policy, and traceability evidence.")
        manifest, quality = validate_h2_output(h2_root, selected_count)
        print("[EXECUTE] Stage 3/3 | Seal the H6B4 node result and compact return capsule.")
        result = {
            "ResultSchema": "U001_H6B4_REAL_H2_DECODE_RESULT_V1_0",
            "GeneratedUtc": utc_now(),
            "SelectedSignalCount": selected_count,
            "SourceRowCount": int(quality["SourceRowCount"]),
            "SelectedMessageGroupCount": int(quality["SelectedMessageGroupCount"]),
            "DecodedUpdateCount": int(quality["DecodedUpdateCount"]),
            "DecodeFailureCount": int(quality["DecodeFailureCount"]),
            "FrameKeyUpdateCount": int(quality["FrameKeyUpdateCount"]),
            "WarningCount": len(quality.get("Warnings", [])),
            "StreamingImplementation": manifest["Configuration"]["StreamingImplementation"],
            "WorkerCountEffective": manifest["Configuration"]["WorkerCountEffective"],
            "NextRequiredAction": "H6B5_BUILD_HIGH_CODE_FRAME_DOMAIN_FROM_ACCEPTED_H2_OUTPUT",
            "Status": "ACCEPTED_H2_SOURCE_UPDATE_STREAM",
            "CompletionCode": 0,
        }
        write_json(output_root / "u001_h6b4_result.json", result)
        capsule = (
            f"U1H6B4-R{result['SourceRowCount']}-U{result['DecodedUpdateCount']}"
            f"-N{selected_count}-F{result['DecodeFailureCount']}"
            f"-K{result['FrameKeyUpdateCount']}-G{result['SelectedMessageGroupCount']}"
            f"-W{result['WarningCount']}-X0-C0"
        )
        (output_root / "u001_h6b4_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print("[COMPLETE] Real H2 source-update stream accepted; no reconstruction or analysis was performed.")
        print(capsule)
        return 0
    except Exception as exc:
        if output_root is None:
            output_root = Path.cwd() / OUTPUT_FOLDER
        output_root.mkdir(parents=True, exist_ok=True)
        write_json(
            output_root / "u001_h6b4_diagnostic.json",
            {
                "ProgramId": PROGRAM_ID,
                "GeneratedUtc": utc_now(),
                "ExceptionType": type(exc).__name__,
                "Detail": str(exc)[:4000],
                "Recovery": "Run --prepare again after correcting the reported cause, then repeat all three commands.",
            },
        )
        print(f"U1H6B4-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H6B4 controlled full-source H2 decode")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    sys.exit(main())
