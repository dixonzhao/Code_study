def fail(code: str, detail: str = "") -> int:
    suffix = f" D-{sanitize(detail, 120)}" if detail else ""
    print(f"G10C1CG1-ERROR:{code}{suffix} C-7")
    return 7


def sanitize(value: object, limit: int = 160) -> str:
    text = re.sub(r"\s+", "_", str(value).strip())
    text = re.sub(r"[^A-Za-z0-9_.:@+=-]", "?", text)
    return text[:limit] if text else "EMPTY"


def write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding="utf-8", newline="\n")


def run(cmd: list[str], cwd: Path, env: dict[str, str] | None = None) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        cmd,
        cwd=str(cwd),
        env=env,
        text=True,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        shell=False,
        check=False,
    )

def make_fake_tree(root: Path) -> tuple[Path, Path, Path]:
    udp = root / "UDP001"
    programs = udp / "01_PROGRAMS"
    case = udp / "03_CASES" / "TC_SYNTHETIC_001"
    source = case / "00_SOURCE-READONLY"
    programs.mkdir(parents=True)
    source.mkdir(parents=True)
    write(source / "CANDump.csv", "ID620_2_8_0\n00\n")
    write(source / "DBC1.dbc", "VERSION \"FAKE1\"\n")
    write(source / "DBC2.dbc", "VERSION \"FAKE2\"\n")
    write(programs / "u001_foundation_core_v0_1.py", FAKE_G1)
    write(programs / "u001_auto_discovery_profile_builder_v0_2.py", FAKE_G10A)
    write(programs / "u001_g10b_single_pass_discovery_engine_v0_5a.py", FAKE_G10B)
    return udp, programs, case


def make_request(controller: Path, udp: Path, case: Path, run_id: str) -> tuple[Path, subprocess.CompletedProcess[str]]:
    request = case / "01_CONFIG" / run_id / "u001_discovery_request.json"
    request.parent.mkdir(parents=True)
    cmd = [
        sys.executable, str(controller), "--make-request",
        "--case-root", str(case), "--out", str(request),
        "--request-id", run_id, "--run-id", run_id,
        "--dataset-id", case.name, "--source-id", "SOURCE_SYNTHETIC",
        "--segment-id", "SOURCE_ROWS_0_1",
        "--requested-output-root", f"02_RUNS/{run_id}",
        "--reuse-policy", "FORCE_NEW_RUN", "--max-source-rows", "2",
        "--chunksize", "2", "--max-assignment-age-rows", "20",
        "--max-rss-mb", "256", "--update-buffer-rows", "8",
        "--frame-buffer-rows", "8", "--memory-check-every-rows", "1",
        "--minimum-free-gib", "0",
    ]
    return request, run(cmd, udp.parent)


def preflight(controller: Path, udp: Path, case: Path, request: Path, run_id: str) -> subprocess.CompletedProcess[str]:
    out = case / "03_RETURN" / f"{run_id}_PREFLIGHT"
    return run([
        sys.executable, str(controller), "--preflight",
        "--udp001-root", str(udp), "--case-root", str(case),
        "--request", str(request), "--out", str(out),
    ], udp.parent)


def execute(controller: Path, udp: Path, case: Path, request: Path) -> subprocess.CompletedProcess[str]:
    return run([
        sys.executable, str(controller), "--run-request",
        "--udp001-root", str(udp), "--case-root", str(case),
        "--request", str(request),
    ], udp.parent)


def require_file(path: Path, token: str | None = None) -> bool:
    if not path.is_file():
        return False
    if token is None:
        return True
    return token in path.read_text(encoding="utf-8", errors="replace")


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--controller", required=True)
    args = parser.parse_args()
    controller = Path(args.controller).resolve()
    if not controller.is_file():
        return fail("CONTROLLER_MISSING")

    source = controller.read_text(encoding="utf-8", errors="replace")
    bits = "".join("1" if name in source else "0" for name in G10A_FILES)
    if bits != "11111111":
        return fail("STATIC_G10A_CONTRACT", bits)
    for token in ("01_G1_CATALOG", "02_PROFILE_CONFIG", "03_DISCOVERY_DATA", "ABORTED_RUN.txt", "C4.stdout.txt", "C4.stderr.txt"):
        if token not in source:
            return fail("STATIC_TOKEN", token)

    st = run([sys.executable, str(controller), "--self-test"], controller.parent)
    if st.returncode != 0 or "U001-G10C1R-SELFTEST-PASS-40" not in st.stdout:
        return fail("SELFTEST", f"E{st.returncode}:{st.stdout[-80:]}:{st.stderr[-80:]}")

    with tempfile.TemporaryDirectory(prefix="u001_g10c1cg1_") as tmp:
        udp, programs, case = make_fake_tree(Path(tmp))

        # Test A: C2 exits zero but creates no artifacts.  Evidence must remain.
        run_a = "CG1_FAIL_C2"
        req_a, mk_a = make_request(controller, udp, case, run_a)
        if mk_a.returncode != 0:
            return fail("A_REQUEST", mk_a.stdout + mk_a.stderr)
        pf_a = preflight(controller, udp, case, req_a, run_a)
        if pf_a.returncode != 0:
            return fail("A_PREFLIGHT", pf_a.stdout + pf_a.stderr)
        write(programs / "FAIL_C2_NO_ARTIFACTS", "1\n")
        ex_a = execute(controller, udp, case, req_a)
        (programs / "FAIL_C2_NO_ARTIFACTS").unlink()
        inc_a = case / "02_RUNS" / f".{run_a}.g10c1_incomplete"
        if ex_a.returncode == 0:
            return fail("A_FALSE_SUCCESS")
        if not inc_a.is_dir():
            return fail("A_INCOMPLETE_NOT_PRESERVED")
        if (case / "02_RUNS" / run_a).exists():
            return fail("A_FINAL_CREATED")
        if not require_file(inc_a / "04_RESPONSE" / "LOGS" / "C2.stdout.txt", "FAKE-G10A-NO-ARTIFACTS"):
            return fail("A_C2_STDOUT_MISSING")
        if not require_file(inc_a / "04_RESPONSE" / "LOGS" / "C2.stderr.txt"):
            return fail("A_C2_STDERR_MISSING")
        if not require_file(inc_a / "04_RESPONSE" / "ABORTED_RUN.txt"):
            return fail("A_ABORT_MISSING")

        # Test B: exact G10A output set must pass C2/C3; C4 then exits nonzero.
        run_b = "CG1_FAIL_C4"
        req_b, mk_b = make_request(controller, udp, case, run_b)
        if mk_b.returncode != 0:
            return fail("B_REQUEST", mk_b.stdout + mk_b.stderr)
        pf_b = preflight(controller, udp, case, req_b, run_b)
        if pf_b.returncode != 0:
            return fail("B_PREFLIGHT", pf_b.stdout + pf_b.stderr)
        ex_b = execute(controller, udp, case, req_b)
        inc_b = case / "02_RUNS" / f".{run_b}.g10c1_incomplete"
        if ex_b.returncode == 0:
            return fail("B_FALSE_SUCCESS")
        if not inc_b.is_dir():
            return fail("B_INCOMPLETE_NOT_PRESERVED")
        profile = inc_b / "02_PROFILE_CONFIG"
        bits_b = "".join("1" if (profile / name).is_file() else "0" for name in G10A_FILES)
        if bits_b != "11111111":
            return fail("B_G10A_NOT_ACCEPTED", bits_b)
        if not require_file(inc_b / "04_RESPONSE" / "LOGS" / "C3.stdout.txt", "UDP001-G10A2-VALIDATE-PASS"):
            return fail("B_C3_NOT_REACHED")
        if not require_file(inc_b / "04_RESPONSE" / "LOGS" / "C4.stdout.txt", "FAKE-G10B-STDOUT-BEFORE-EXIT"):
            return fail("B_C4_STDOUT_MISSING")
        if not require_file(inc_b / "04_RESPONSE" / "LOGS" / "C4.stderr.txt", "FAKE-G10B-STDERR-BEFORE-EXIT"):
            return fail("B_C4_STDERR_MISSING")
        if not require_file(inc_b / "04_RESPONSE" / "ABORTED_RUN.txt"):
            return fail("B_ABORT_MISSING")

    print("G10C1CG1-PASS S8-D2-P2-L6-X0 C-0")
    return 0


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as exc:
        raise SystemExit(fail("UNHANDLED", f"{type(exc).__name__}:{exc}"))
