#!/usr/bin/env python3
"""Independent conformance Gate for U001 portable builder v0.8."""

from __future__ import annotations

import argparse
import hashlib
import importlib.util
import json
import subprocess
import sys
import tempfile
from pathlib import Path


PROGRAM_NAME = "u001_g10d2m3_builder_v08_conformance_gate_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2M3-CG1-SELFTEST-PASS-20"
BUILDER_NAME = "u001_portable_host_package_builder_v0_8.py"
BUILDER_VERSION = "0.8"
BUILDER_TOKEN = "U001-PORTABLE-HOST-PACKAGE-BUILDER-V08-SELFTEST-PASS-44"
HOST_NAME = "u001_portable_runtime_host_v0_6.py"
HOST_VERSION = "0.6"
HOST_TOKEN = "U001-PORTABLE-HOST-V06-SELFTEST-PASS-26"
PACKAGE_NAME = "U001_Portable_v0_2"
MARKER = ".G10D2M3_INCOMPLETE"


class GateFailure(RuntimeError):
    pass


def fail(code: str) -> None:
    raise GateFailure(code)


def load(path: Path, module_name: str):
    spec = importlib.util.spec_from_file_location(module_name, path)
    if spec is None or spec.loader is None:
        fail("IMPORT_SPEC")
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    spec.loader.exec_module(module)
    return module


def run_source_selftest(path: Path, token: str) -> None:
    result = subprocess.run(
        [sys.executable, str(path.resolve()), "--self-test"],
        cwd=str(path.parent.resolve()),
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        encoding="utf-8",
        errors="replace",
        timeout=120,
        shell=False,
        check=False,
    )
    if result.returncode != 0 or token not in result.stdout:
        fail("SELFTEST_" + path.stem)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    digest.update(path.read_bytes())
    return digest.hexdigest()


def validate(builder_path: Path, host_path: Path) -> None:
    builder_path = builder_path.resolve()
    host_path = host_path.resolve()
    if not builder_path.is_file() or builder_path.name != BUILDER_NAME:
        fail("BUILDER_IDENTITY")
    if not host_path.is_file() or host_path.name != HOST_NAME:
        fail("HOST_IDENTITY")
    run_source_selftest(builder_path, BUILDER_TOKEN)
    run_source_selftest(host_path, HOST_TOKEN)
    try:
        builder = load(builder_path, "u001_builder_v08_candidate")
        host = load(host_path, "u001_host_v06_for_builder_gate")
    except GateFailure:
        raise
    except Exception as exc:
        fail("IMPORT_" + type(exc).__name__)
    if getattr(builder, "PROGRAM_NAME", None) != BUILDER_NAME:
        fail("BUILDER_NAME")
    if getattr(builder, "PROGRAM_VERSION", None) != BUILDER_VERSION:
        fail("BUILDER_VERSION")
    if getattr(builder, "SELFTEST_TOKEN", None) != BUILDER_TOKEN:
        fail("BUILDER_TOKEN")
    if getattr(host, "PROGRAM_VERSION", None) != HOST_VERSION:
        fail("HOST_VERSION")
    if getattr(host, "SELFTEST_TOKEN", None) != HOST_TOKEN:
        fail("HOST_TOKEN")
    if getattr(builder, "PACKAGE_NAME", None) != PACKAGE_NAME:
        fail("PACKAGE_NAME")
    contract_fn = getattr(builder, "get_portable_builder_contract", None)
    inventory_fn = getattr(builder, "directory_inventory", None)
    publish_fn = getattr(builder, "publish_staged_release", None)
    command_fn = getattr(builder, "pyinstaller_host_command", None)
    if not all(callable(item) for item in (contract_fn, inventory_fn, publish_fn, command_fn)):
        fail("PUBLIC_API")
    contract = contract_fn()
    serialized = json.dumps(contract, ensure_ascii=False, sort_keys=True)
    if str(contract.get("ProgramVersion")) != BUILDER_VERSION:
        fail("CONTRACT_VERSION")
    if str(contract.get("ExpectedRuntimeHostVersion")) != HOST_VERSION:
        fail("CONTRACT_HOST")
    if contract.get("PublicationMode") != "VERIFIED_FILE_COPY_WITH_EMPTY_DIRECTORY_PRESERVATION":
        fail("CONTRACT_PUBLICATION")
    if "03_CASES" not in serialized:
        fail("CONTRACT_EMPTY_DIRECTORY")
    command = command_fn(Path("host_v0_6.py"), Path("dist"), Path("work"), Path("spec"))
    joined = " ".join(str(item) for item in command)
    if "PyInstaller" not in joined or "--onedir" not in command or "--windowed" not in command:
        fail("PYINSTALLER_POLICY")
    if "--onefile" in command:
        fail("ONEFILE_FORBIDDEN")
    with tempfile.TemporaryDirectory(prefix="u001_g10d2m3cg1_") as td:
        base = Path(td)
        stage = base / "stage"
        package = stage / PACKAGE_NAME
        cases = package / "UDP001" / "03_CASES"
        programs = package / "UDP001" / "01_PROGRAMS"
        programs.mkdir(parents=True)
        cases.mkdir(parents=True)
        sample = programs / "sample.py"
        sample.write_text("print('sample')\n", encoding="utf-8")
        (package / "README_FIRST.txt").write_text("readme\n", encoding="utf-8")
        (stage / f"{PACKAGE_NAME}.zip").write_bytes(b"ZIP")
        before = inventory_fn(stage)
        if "03_CASES" not in json.dumps(before, ensure_ascii=False):
            fail("INVENTORY_EMPTY_DIRECTORY")
        out = base / "out"
        publish_fn(stage, out)
        out_cases = out / PACKAGE_NAME / "UDP001" / "03_CASES"
        out_sample = out / PACKAGE_NAME / "UDP001" / "01_PROGRAMS" / "sample.py"
        if not out_cases.is_dir() or any(out_cases.iterdir()):
            fail("PUBLISHED_CASES")
        if not out_sample.is_file() or sha256_file(out_sample) != sha256_file(sample):
            fail("PUBLISHED_HASH")
        if not (out / f"{PACKAGE_NAME}.zip").is_file():
            fail("PUBLISHED_ZIP")
        if (out / MARKER).exists():
            fail("MARKER_RETAINED")
        if inventory_fn(stage) != inventory_fn(out):
            fail("PUBLISHED_INVENTORY")
        try:
            publish_fn(stage, out)
        except Exception:
            pass
        else:
            fail("EXISTING_OUTPUT_ACCEPTED")


def run_self_test() -> None:
    checks = (
        PROGRAM_VERSION == "0.1",
        PROGRAM_NAME.endswith("v0_1.py"),
        SELFTEST_TOKEN.endswith("PASS-20"),
        BUILDER_NAME.endswith("v0_8.py"),
        BUILDER_VERSION == "0.8",
        BUILDER_TOKEN.endswith("PASS-44"),
        HOST_NAME.endswith("v0_6.py"),
        HOST_VERSION == "0.6",
        HOST_TOKEN.endswith("PASS-26"),
        PACKAGE_NAME.endswith("v0_2"),
        MARKER.startswith("."),
        callable(load),
        callable(run_source_selftest),
        callable(sha256_file),
        callable(validate),
        issubclass(GateFailure, RuntimeError),
        Path("x.py").suffix == ".py",
        len(hashlib.sha256(b"x").hexdigest()) == 64,
        sys.executable != "",
        "VERIFIED" in "VERIFIED_FILE_COPY_WITH_EMPTY_DIRECTORY_PRESERVATION",
    )
    if sum(bool(item) for item in checks) != 20:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Gate U001 portable builder v0.8")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--builder")
    parser.add_argument("--host")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.builder or not args.host:
        parser.error("--builder and --host are required")
    try:
        validate(Path(args.builder), Path(args.host))
    except GateFailure as exc:
        print(f"G10D2M3CG1-FAIL-{exc.args[0]} C-7")
        return 7
    except subprocess.TimeoutExpired:
        print("G10D2M3CG1-FAIL-TIMEOUT C-7")
        return 7
    except Exception as exc:
        print(f"G10D2M3CG1-FAIL-UNHANDLED-{type(exc).__name__} C-7")
        return 7
    print("G10D2M3CG1-PASS H06-B08-D1-P1-R1-X0 C-0")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
