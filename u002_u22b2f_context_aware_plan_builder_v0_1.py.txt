#!/usr/bin/env python3
"""Build the B2F Run by separating Camera context from evaluation scope.

The B2E diagnostic proved that B2DR3 started after the governed Camera seed and
therefore produced only unseeded ambiguous Camera rows.  This planner preserves
the accepted B2DR3 profile, stage order and analysis mathematics.  It selects
the context-aware Camera Module v0.2.0, adds ContextFrameStart=7400 and creates
a new Run identity.  No accepted Run or Artifact is modified.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import tempfile
from typing import Dict, Mapping, Sequence, Tuple


PROGRAM_ID = "UDP002-U22B2F-CONTEXT-AWARE-PLAN-BUILDER"
PROGRAM_VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"

SOURCE_RUN_ID = "U22B2DR3_TC001_F41480_41750_V01"
RUN_ID = "U22B2F_TC001_F41480_41750_V01"
FRAME_START = 41480
FRAME_END = 41750
CONTEXT_FRAME_START = 7400
CAMERA_MODULE_ID = "CameraRadar.S1.CameraTargetContinuityArbitration"
CAMERA_MODULE_VERSION = "0.2.0"

QUALIFICATION = Path("01_PROGRAMS/core/u002_u22_module_qualification_v0_1.py")
PREFLIGHT = Path("01_PROGRAMS/core/u002_u22_engine_preflight_v0_1.py")
FREEZE_AUDITOR = Path(
    "05_TESTS/integration/u002_u22_synthetic_vertical_acceptance_v0_1.py"
)
AVAILABILITY = Path(
    "00_CONTROL/registry/u002_module_availability_snapshot_v0_1.json"
)
SOURCE_PLAN = Path("00_CONFIG/resolved_run_plan.json")
B2E_REPORT = Path(
    "02_DIAGNOSTICS/b2e_formal_population_diagnostic_v0_1.json"
)
PLAN_RECEIPT = Path(
    "u22b2f_context_evaluation_plan_receipt_v0_1.json"
)


class PlanError(RuntimeError):
    """Represent one deterministic B2F planning failure."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = compact_token(reason, 64)
        self.detail = compact_token(detail, 120)
        super().__init__(self.reason)


def compact_token(value: object, limit: int = 80) -> str:
    """Convert diagnostic text into one bounded return-capsule token."""

    text = re.sub(r"[^A-Z0-9_.:+-]+", "_", str(value).upper())
    return text.strip("_")[:limit] or "NA"


def canonical_bytes(value: object) -> bytes:
    """Serialize a control record deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    """Accept an ordinary non-link file only."""

    return path.is_file() and not path.is_symlink()


def load_json(path: Path, reason: str) -> Dict[str, object]:
    """Load one JSON object and fail closed on any structural problem."""

    if not regular_file(path):
        raise PlanError(reason + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise PlanError(reason + "_JSON", path.name) from exc
    if not isinstance(value, dict):
        raise PlanError(reason + "_TYPE", path.name)
    return value


def write_exact(path: Path, value: object) -> None:
    """Create an immutable control file or accept an identical replay."""

    payload = canonical_bytes(value)
    if path.exists():
        if not regular_file(path) or path.read_bytes() != payload:
            raise PlanError("EXISTING_FILE_CONFLICT", path.name)
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(payload)
    os.replace(temporary, path)


def require_workspace(path: Path) -> Tuple[Path, Path]:
    """Validate the U002 workspace and the accepted B2DR3 evidence root."""

    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise PlanError("WORKSPACE_ROOT", root.name)
    for relative in (QUALIFICATION, PREFLIGHT, FREEZE_AUDITOR):
        if not regular_file(root / relative):
            raise PlanError("WORKSPACE_DEPENDENCY", relative.as_posix())
    source_root = root / "07_OUTPUT" / SOURCE_RUN_ID
    if not source_root.is_dir() or source_root.is_symlink():
        raise PlanError("SOURCE_RUN_ROOT", SOURCE_RUN_ID)
    return root, source_root


def run_child(
    root: Path,
    program: Path,
    arguments: Sequence[str],
    expected_prefix: str,
    timeout: int = 180,
) -> str:
    """Run one accepted Core program and retain only its final capsule."""

    completed = subprocess.run(
        [sys.executable, str(root / program), *arguments],
        cwd=root.parent,
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )
    lines = [line.strip() for line in completed.stdout.splitlines() if line.strip()]
    capsule = lines[-1] if lines else "MISSING"
    if (
        completed.returncode != 0
        or not capsule.startswith(expected_prefix)
        or not capsule.endswith("C-0")
    ):
        detail = capsule if capsule != "MISSING" else completed.stderr[-240:]
        raise PlanError("CHILD_" + program.stem, detail)
    return capsule


def freeze_audit(root: Path) -> str:
    """Prove that accepted U22 Core files remain frozen."""

    return run_child(
        root,
        FREEZE_AUDITOR,
        ["--audit-freeze", "--workspace-root", str(root)],
        "U22A7F",
        timeout=90,
    )


def refresh_availability(root: Path) -> Dict[str, object]:
    """Refresh and return the governed Module availability snapshot."""

    run_child(
        root,
        QUALIFICATION,
        ["--refresh-availability", "--workspace-root", str(root)],
        "U22A4Q",
    )
    return load_json(root / AVAILABILITY, "AVAILABILITY")


def availability_entry(
    snapshot: Mapping[str, object],
    module_id: str,
    module_version: str,
) -> Mapping[str, object]:
    """Resolve exactly one Module identity from the registry snapshot."""

    entries = snapshot.get("Entries")
    if not isinstance(entries, list):
        raise PlanError("AVAILABILITY_ENTRIES")
    matches = [
        entry
        for entry in entries
        if isinstance(entry, dict)
        and entry.get("ModuleId") == module_id
        and entry.get("ModuleVersion") == module_version
    ]
    if len(matches) != 1:
        raise PlanError("AVAILABILITY_COUNT", f"{module_id}@{module_version}")
    return matches[0]


def qualify_camera_module(root: Path) -> int:
    """Qualify the new Camera version once or audit an identical replay."""

    identity = CAMERA_MODULE_ID + "@" + CAMERA_MODULE_VERSION
    snapshot = refresh_availability(root)
    entry = availability_entry(snapshot, CAMERA_MODULE_ID, CAMERA_MODULE_VERSION)
    newly_qualified = 0
    if entry.get("RegistryStatus") == "AVAILABLE":
        run_child(
            root,
            QUALIFICATION,
            ["--audit-module", identity, "--workspace-root", str(root)],
            "U22A4Q",
        )
    else:
        run_child(
            root,
            QUALIFICATION,
            [
                "--qualify-module",
                identity,
                "--workspace-root",
                str(root),
                "--timeout-seconds",
                "180",
            ],
            "U22A4Q",
            timeout=240,
        )
        newly_qualified = 1
    final = refresh_availability(root)
    accepted = availability_entry(final, CAMERA_MODULE_ID, CAMERA_MODULE_VERSION)
    if accepted.get("RegistryStatus") != "AVAILABLE":
        raise PlanError("CAMERA_MODULE_NOT_AVAILABLE")
    return newly_qualified


def prove_b2e(source_root: Path) -> Dict[str, object]:
    """Require the accepted read-only proof of the missing warm-up condition."""

    report = load_json(source_root / B2E_REPORT, "B2E_REPORT")
    if report.get("RunId") != SOURCE_RUN_ID:
        raise PlanError("B2E_RUN_ID", report.get("RunId"))
    if report.get("ReadOnlyArtifactInspection") is not True:
        raise PlanError("B2E_NOT_READ_ONLY")
    if report.get("ArtifactMutation") is not False:
        raise PlanError("B2E_ARTIFACT_MUTATION")
    if report.get("StatefulWarmupAbsenceProven") is not True:
        raise PlanError("B2E_WARMUP_NOT_PROVEN")
    if report.get("RootCauseClassification") != "STATEFUL_CAMERA_WARMUP_MISSING":
        raise PlanError("B2E_ROOT_CAUSE", report.get("RootCauseClassification"))
    if report.get("CameraValidRows") != 0 or report.get("FrameRows") != 271:
        raise PlanError("B2E_COUNTS")
    return report


def transform_plan(source: Mapping[str, object]) -> Dict[str, object]:
    """Create the new Run configuration while changing one owned boundary only."""

    transformed = json.loads(json.dumps(source, ensure_ascii=False))
    if transformed.get("RunId") != SOURCE_RUN_ID:
        raise PlanError("SOURCE_PLAN_RUN_ID", transformed.get("RunId"))
    if transformed.get("RunScope") != "FRAME_RANGE":
        raise PlanError("SOURCE_PLAN_SCOPE", transformed.get("RunScope"))
    if transformed.get("FrameStart") != FRAME_START:
        raise PlanError("SOURCE_PLAN_FRAME_START", transformed.get("FrameStart"))
    if transformed.get("FrameEnd") != FRAME_END:
        raise PlanError("SOURCE_PLAN_FRAME_END", transformed.get("FrameEnd"))
    modules = transformed.get("ModulePlan")
    if not isinstance(modules, list) or len(modules) != 5:
        raise PlanError("SOURCE_MODULE_COUNT")
    stages = [item.get("Stage") for item in modules if isinstance(item, dict)]
    if stages != ["S0", "S1", "S1", "S2", "S3"]:
        raise PlanError("SOURCE_STAGE_ORDER", stages)
    camera_matches = [
        item
        for item in modules
        if isinstance(item, dict) and item.get("ModuleId") == CAMERA_MODULE_ID
    ]
    if len(camera_matches) != 1:
        raise PlanError("CAMERA_MODULE_COUNT", len(camera_matches))
    camera = camera_matches[0]
    configuration = camera.get("Configuration")
    if not isinstance(configuration, dict):
        raise PlanError("CAMERA_CONFIGURATION")
    if configuration.get("SeedFrame") != CONTEXT_FRAME_START:
        raise PlanError("CAMERA_SEED_FRAME", configuration.get("SeedFrame"))
    if configuration.get("SeedRole") not in {
        "CAMERA_1_DISTANCE",
        "CAMERA_3_DISTANCE",
    }:
        raise PlanError("CAMERA_SEED_ROLE", configuration.get("SeedRole"))

    transformed["RunId"] = RUN_ID
    transformed["RunPurpose"] = "CONTEXT_AWARE_EVALUATION"
    transformed["CandidatePolicy"] = "NotAppliedInU22B2F@0.1"
    camera["ModuleVersion"] = CAMERA_MODULE_VERSION
    configuration["ContextFrameStart"] = CONTEXT_FRAME_START
    return transformed


def verify_transformation(
    source: Mapping[str, object],
    transformed: Mapping[str, object],
) -> None:
    """Prove that only the approved Run and Camera context fields changed."""

    source_copy = json.loads(json.dumps(source, ensure_ascii=False))
    candidate_copy = json.loads(json.dumps(transformed, ensure_ascii=False))

    source_copy["RunId"] = RUN_ID
    source_copy["RunPurpose"] = "CONTEXT_AWARE_EVALUATION"
    source_copy["CandidatePolicy"] = "NotAppliedInU22B2F@0.1"
    source_camera = next(
        item
        for item in source_copy["ModulePlan"]
        if item["ModuleId"] == CAMERA_MODULE_ID
    )
    source_camera["ModuleVersion"] = CAMERA_MODULE_VERSION
    source_camera["Configuration"]["ContextFrameStart"] = CONTEXT_FRAME_START
    if source_copy != candidate_copy:
        raise PlanError("UNAPPROVED_PLAN_CHANGE")


def build_plan(root: Path, source_root: Path) -> str:
    """Qualify v0.2.0, create a new Plan and preserve B2E provenance."""

    freeze_before = freeze_audit(root)
    b2e = prove_b2e(source_root)
    source_plan = load_json(source_root / SOURCE_PLAN, "SOURCE_PLAN")
    source_plan_sha = sha256_file(source_root / SOURCE_PLAN)
    newly_qualified = qualify_camera_module(root)
    transformed = transform_plan(source_plan)
    verify_transformation(source_plan, transformed)

    config_path = root / "02_CONFIG/run_configs" / (RUN_ID + ".json")
    write_exact(config_path, transformed)
    preflight_capsule = run_child(
        root,
        PREFLIGHT,
        ["--prepare-run", str(config_path), "--workspace-root", str(root)],
        "U22A5E",
        timeout=180,
    )
    match = re.search(r"-R([01])-P1-X0 C-0$", preflight_capsule)
    if match is None:
        raise PlanError("PREFLIGHT_CAPSULE", preflight_capsule)
    replay = int(match.group(1))
    run_root = root / "07_OUTPUT" / RUN_ID
    resolved = load_json(run_root / SOURCE_PLAN, "RESOLVED_PLAN")
    if resolved.get("RunId") != RUN_ID:
        raise PlanError("RESOLVED_RUN_ID", resolved.get("RunId"))
    if resolved.get("RunScope") != "FRAME_RANGE":
        raise PlanError("RESOLVED_SCOPE", resolved.get("RunScope"))
    if resolved.get("FrameStart") != FRAME_START:
        raise PlanError("RESOLVED_FRAME_START", resolved.get("FrameStart"))
    if resolved.get("FrameEnd") != FRAME_END:
        raise PlanError("RESOLVED_FRAME_END", resolved.get("FrameEnd"))
    resolved_modules = resolved.get("ModulePlan")
    if not isinstance(resolved_modules, list) or len(resolved_modules) != 5:
        raise PlanError("RESOLVED_MODULE_COUNT")
    if [item.get("Stage") for item in resolved_modules] != ["S0", "S1", "S1", "S2", "S3"]:
        raise PlanError("RESOLVED_STAGE_ORDER")
    resolved_camera = [
        item
        for item in resolved_modules
        if item.get("ModuleId") == CAMERA_MODULE_ID
    ]
    if len(resolved_camera) != 1:
        raise PlanError("RESOLVED_CAMERA_COUNT", len(resolved_camera))
    if resolved_camera[0].get("ModuleVersion") != CAMERA_MODULE_VERSION:
        raise PlanError("RESOLVED_CAMERA_VERSION", resolved_camera[0].get("ModuleVersion"))
    resolved_camera_configuration = resolved_camera[0].get("Configuration")
    if not isinstance(resolved_camera_configuration, dict):
        raise PlanError("RESOLVED_CAMERA_CONFIGURATION")
    if resolved_camera_configuration.get("ContextFrameStart") != CONTEXT_FRAME_START:
        raise PlanError(
            "RESOLVED_CONTEXT_FRAME_START",
            resolved_camera_configuration.get("ContextFrameStart"),
        )

    receipt = {
        "ReceiptContract": "UDP002_U22B2F_CONTEXT_EVALUATION_PLAN_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "RunId": RUN_ID,
        "SourceRunId": SOURCE_RUN_ID,
        "SourceResolvedPlanSha256": source_plan_sha,
        "B2EDiagnosticSha256": sha256_file(source_root / B2E_REPORT),
        "RootCauseClassification": b2e["RootCauseClassification"],
        "ContextFrameStart": CONTEXT_FRAME_START,
        "EvaluationFrameStart": FRAME_START,
        "EvaluationFrameEnd": FRAME_END,
        "CameraModuleId": CAMERA_MODULE_ID,
        "CameraModuleVersion": CAMERA_MODULE_VERSION,
        "ModuleCodeModified": True,
        "AnalysisMathematicsModified": False,
        "ApprovedChangeSurface": [
            "Camera Module version 0.1.0 to 0.2.0",
            "Camera Configuration.ContextFrameStart added as 7400",
            "Run identity and purpose updated",
            "CandidatePolicy version updated",
        ],
        "PreflightCapsule": preflight_capsule,
        "FreezeAuditCapsule": freeze_before,
    }
    write_exact(run_root / PLAN_RECEIPT, receipt)
    freeze_after = freeze_audit(root)
    if freeze_before != freeze_after:
        raise PlanError("FOUNDATION_CHANGED")
    return (
        f"B2FP N-M5-Q{newly_qualified}-R{replay}"
        f"-C{CONTEXT_FRAME_START}-E{FRAME_START}.{FRAME_END}-X0 C-0"
    )


def self_test() -> str:
    """Cover the transformation boundary and all principal rejection classes."""

    checks = 0

    def check(condition: bool) -> None:
        nonlocal checks
        if not condition:
            raise AssertionError("SELFTEST")
        checks += 1

    source = {
        "ContractVersion": CONTRACT_VERSION,
        "RunId": SOURCE_RUN_ID,
        "RunPurpose": "EVALUATION",
        "RunScope": "FRAME_RANGE",
        "FrameStart": FRAME_START,
        "FrameEnd": FRAME_END,
        "InputPackage": "X",
        "SignalProfile": "P",
        "ModulePlan": [
            {"Sequence": 1, "Stage": "S0", "ModuleId": "S0", "ModuleVersion": "1", "Configuration": {}},
            {
                "Sequence": 2,
                "Stage": "S1",
                "ModuleId": CAMERA_MODULE_ID,
                "ModuleVersion": "0.1.0",
                "Configuration": {
                    "SeedFrame": CONTEXT_FRAME_START,
                    "SeedRole": "CAMERA_1_DISTANCE",
                },
            },
            {"Sequence": 3, "Stage": "S1", "ModuleId": "R", "ModuleVersion": "1", "Configuration": {}},
            {"Sequence": 4, "Stage": "S2", "ModuleId": "P", "ModuleVersion": "1", "Configuration": {}},
            {"Sequence": 5, "Stage": "S3", "ModuleId": "O", "ModuleVersion": "1", "Configuration": {}},
        ],
        "PrimaryOutcome": {"Name": "D"},
        "CandidatePolicy": "OLD",
        "OutputSelection": ["A"],
    }
    transformed = transform_plan(source)
    check(transformed["RunId"] == RUN_ID)
    check(transformed["RunPurpose"] == "CONTEXT_AWARE_EVALUATION")
    check(transformed["FrameStart"] == FRAME_START)
    check(transformed["FrameEnd"] == FRAME_END)
    check(transformed["ModulePlan"][1]["ModuleVersion"] == CAMERA_MODULE_VERSION)
    check(
        transformed["ModulePlan"][1]["Configuration"]["ContextFrameStart"]
        == CONTEXT_FRAME_START
    )
    check("ContextFrameStart" not in source["ModulePlan"][1]["Configuration"])
    verify_transformation(source, transformed)
    check(True)

    mutations = (
        ("RunId", "WRONG"),
        ("RunScope", "FULL_CASE"),
        ("FrameStart", FRAME_START + 1),
        ("FrameEnd", FRAME_END - 1),
    )
    for field, value in mutations:
        invalid = json.loads(json.dumps(source))
        invalid[field] = value
        try:
            transform_plan(invalid)
        except PlanError:
            check(True)
        else:
            check(False)

    invalid = json.loads(json.dumps(source))
    invalid["ModulePlan"] = invalid["ModulePlan"][:-1]
    try:
        transform_plan(invalid)
    except PlanError:
        check(True)
    else:
        check(False)

    invalid = json.loads(json.dumps(source))
    invalid["ModulePlan"][1]["Configuration"]["SeedFrame"] = 7399
    try:
        transform_plan(invalid)
    except PlanError:
        check(True)
    else:
        check(False)

    invalid = json.loads(json.dumps(source))
    invalid["ModulePlan"][1]["Configuration"]["SeedRole"] = None
    try:
        transform_plan(invalid)
    except PlanError:
        check(True)
    else:
        check(False)

    altered = json.loads(json.dumps(transformed))
    altered["PrimaryOutcome"] = {"Name": "CHANGED"}
    try:
        verify_transformation(source, altered)
    except PlanError:
        check(True)
    else:
        check(False)

    with tempfile.TemporaryDirectory() as directory:
        path = Path(directory) / "record.json"
        write_exact(path, {"B": 2, "A": 1})
        check(load_json(path, "RECORD") == {"A": 1, "B": 2})
        write_exact(path, {"A": 1, "B": 2})
        check(len(sha256_file(path)) == 64)
        try:
            write_exact(path, {"A": 2})
        except PlanError:
            check(True)
        else:
            check(False)

    check(compact_token("a b/c") == "A_B_C")
    check(checks == 20)
    return "UDP002-U22B2F-PLAN-SELFTEST-PASS-21 C-0"


def build_parser() -> argparse.ArgumentParser:
    """Build explicit self-test and plan modes."""

    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--build-plan", action="store_true")
    parser.add_argument("--workspace-root", type=Path)
    return parser


def main() -> int:
    """CLI entry point with bounded built-in diagnostics."""

    arguments = build_parser().parse_args()
    try:
        if arguments.self_test:
            print(self_test())
            return 0
        if arguments.workspace_root is None:
            raise PlanError("WORKSPACE_REQUIRED")
        root, source_root = require_workspace(arguments.workspace_root)
        print(build_plan(root, source_root))
        return 0
    except PlanError as exc:
        print(f"B2FPE K-{exc.reason}-D-{exc.detail}-X1 C-7")
        return 7
    except Exception as exc:
        detail = compact_token(type(exc).__name__ + ":" + str(exc), 120)
        print(f"B2FPE K-UNEXPECTED-D-{detail}-X1 C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
