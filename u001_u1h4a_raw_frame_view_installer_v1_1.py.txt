#!/usr/bin/env python3
"""Prepare, test and install U001 H4A raw Frame View support."""

from __future__ import annotations

import argparse
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PROGRAM_VERSION = "1.1.0"
PACKAGE_NAME = "UDP001_U1H4A_RAW_FRAME_VIEW_HIGH_CODE_v1_1.zip"
ACTIVE_NAME = "U001H_ACTIVE"
PREPARED_NAME = "U1H4A_PREPARED"

EXPECTED_FILES = {
    "01_PROGRAMS/core/u001_h4_contract_core_v1_0.py",
    "01_PROGRAMS/core/u001_h4_frame_policy_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_h4_raw_frame_view_builder_v1_0.py",
    "02_CONTROL/contracts/u001_h4_frame_policy_schema_v1_0.json",
    "02_CONTROL/contracts/u001_h4_raw_frame_view_contract_v1_0.json",
    "05_TESTS/tools/u001_h4a_raw_frame_view_acceptance_v1_1.py",
    "06_DOCS/U001_H4A_RAW_FRAME_VIEW_TECHNICAL_SPEC_v1_1_2026-08-28.txt",
}

FOUNDATION_FILES = {
    "01_PROGRAMS/core/u001_frame_domain_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_signal_policy_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_canonical_update_projector_v1_0.py",
    "01_PROGRAMS/core/u001_h3_sync_semantic_engine_v1_0.py",
    "01_PROGRAMS/core/u001_selection_profile_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_catalog_mapping_gui_v1_0.py",
    "01_PROGRAMS/adapters/CameraRadar.U001.MatlabCanCsvDbc/1.1.0/adapter.py",
    "05_TESTS/tools/u001_h3_sync_semantic_acceptance_v1_0.py",
}

OBSOLETE_TRANSFER_NAMES = {
    "U1H2B_EXECUTION_RESULT.json",
    "U1H4A_EXECUTION_RESULT.json",
    "UDP001_U1H2B_CATALOG_SELECTION_HIGH_CODE_v1_0.zip",
    "u001_u1h2b_catalog_selection_installer_v1_0.py.txt",
    "U1H2B_START_HERE_CN_v1_0_2026-08-28.txt",
    "UDP001_U1H4A_RAW_FRAME_VIEW_HIGH_CODE_v1_0.zip",
    "u001_u1h4a_raw_frame_view_installer_v1_0.py.txt",
    "U1H4A_START_HERE_CN_v1_0_2026-08-28.txt",
}

ACTIVE_OBSOLETE_FILES = {
    "05_TESTS/tools/u001_h4a_raw_frame_view_acceptance_v1_0.py",
    "06_DOCS/U001_H4A_RAW_FRAME_VIEW_TECHNICAL_SPEC_v1_0_2026-08-28.txt",
}

LAST_CHILD_EVIDENCE: dict[str, object] = {}


class InstallError(RuntimeError):
    """Fail-closed H4A installation error."""


def timestamp() -> str:
    return datetime.now().strftime("%Y%m%dT%H%M%S_%f")


def resolve_u001_root(explicit: Path | None) -> Path:
    candidates = [explicit] if explicit is not None else []
    candidates.extend((Path.cwd() / "U-UDP001", Path.cwd()))
    for candidate in candidates:
        candidate = candidate.resolve()
        if candidate.name == "U-UDP001" and candidate.is_dir():
            return candidate
    raise InstallError("U001_ROOT_NOT_FOUND")


def temp_dir(root: Path) -> Path:
    result = root / "04_TEMP"
    if not result.is_dir():
        raise InstallError("TEMP_DIRECTORY_MISSING")
    return result


def validate_foundation(root: Path) -> None:
    missing = sorted(relative for relative in FOUNDATION_FILES if not (root / relative).is_file())
    if missing:
        raise InstallError(f"FOUNDATION_MISSING:{missing}")


def validate_zip(path: Path) -> None:
    if not path.is_file():
        raise InstallError(f"PACKAGE_MISSING:{PACKAGE_NAME}")
    with zipfile.ZipFile(path, "r") as archive:
        files = {name for name in archive.namelist() if not name.endswith("/")}
        if files != EXPECTED_FILES:
            raise InstallError(
                f"PACKAGE_INVENTORY:MISSING={sorted(EXPECTED_FILES-files)}:EXTRA={sorted(files-EXPECTED_FILES)}"
            )
        for name in files:
            pure = PurePosixPath(name)
            if pure.is_absolute() or ".." in pure.parts:
                raise InstallError(f"PACKAGE_PATH:{name}")
        bad = archive.testzip()
        if bad:
            raise InstallError(f"PACKAGE_CRC:{bad}")


def _archive(path: Path, archive_root: Path, relative: Path) -> bool:
    if not path.exists():
        return False
    target = archive_root / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(target))
    return True


def clean_temp(root: Path) -> int:
    transfer = temp_dir(root)
    archive_root = root / "A-Archive" / f"U1H4A_TEMP_HYGIENE_{timestamp()}"
    moved = 0
    for name in sorted(OBSOLETE_TRANSFER_NAMES):
        moved += int(_archive(transfer / name, archive_root, Path(name)))
    active = transfer / ACTIVE_NAME
    for name in ("U1H2B_PREPARED", PREPARED_NAME):
        moved += int(_archive(active / name, archive_root, Path(ACTIVE_NAME) / name))
    return moved


def prepare(root: Path) -> Path:
    package = temp_dir(root) / PACKAGE_NAME
    print(f"[CHECK] U001 root exists: {root}")
    print("[WHY]   H4A extends the accepted H3/H2B product and must not replace their semantics.")
    validate_foundation(root)
    print(f"[CHECK] H3 and H2B foundation is complete: {len(FOUNDATION_FILES)} required files")
    print("[WHY]   Raw Frame View is permitted only after source, selection and semantic boundaries pass.")
    validate_zip(package)
    print(f"[CHECK] H4A package inventory and CRC passed: {len(EXPECTED_FILES)} files")
    print("[WHY]   Exact inventory blocks stale legacy frame-view or forward-fill code.")
    archived = clean_temp(root)
    print(f"[CLEAN] Superseded H2B/H4A transfer and prepared entries archived: {archived}")
    print("[WHY]   04_TEMP exposes one current H4A prepared source of truth.")
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package, "r") as archive:
        archive.extractall(prepared)
    extracted = {path.relative_to(prepared).as_posix() for path in prepared.rglob("*") if path.is_file()}
    if extracted != EXPECTED_FILES:
        raise InstallError("EXTRACTED_INVENTORY")
    print("[CHECK] Prepared H4A payload exactly matches the approved package.")
    print(f"U1H4AP-N{len(EXPECTED_FILES)}-A{archived}-X0-C0")
    return prepared


def child_environment() -> dict[str, str]:
    environment = dict(os.environ)
    environment["PYTHONDONTWRITEBYTECODE"] = "1"
    environment["PYTHONPYCACHEPREFIX"] = str(Path(tempfile.gettempdir()) / "u001_h4a_pycache")
    return environment


def run_child(command: list[str], label: str, timeout: int = 300) -> str:
    global LAST_CHILD_EVIDENCE
    try:
        result = subprocess.run(
            command,
            text=True,
            capture_output=True,
            timeout=timeout,
            check=False,
            env=child_environment(),
        )
    except subprocess.TimeoutExpired as exc:
        LAST_CHILD_EVIDENCE = {
            "Label": label,
            "Status": "TIMEOUT",
            "TimeoutSeconds": timeout,
            "Command": command,
        }
        raise InstallError(f"{label}_TIMEOUT") from exc
    LAST_CHILD_EVIDENCE = {
        "Label": label,
        "Status": "PASS" if result.returncode == 0 else "FAIL",
        "ReturnCode": result.returncode,
        "Command": command,
        "Stdout": result.stdout,
        "Stderr": result.stderr,
    }
    if result.returncode != 0:
        raise InstallError(f"{label}_EXIT:{result.returncode}")
    output = result.stdout.strip()
    if not output:
        raise InstallError(f"{label}_NO_OUTPUT")
    return output


def write_failure_diagnostic(root: Path, exc: Exception) -> None:
    """Persist full child evidence so the operator never has to retype a traceback."""
    diagnostic = {
        "DiagnosticSchema": "U001_H4A_INSTALL_FAILURE_V1_1",
        "ProgramVersion": PROGRAM_VERSION,
        "ErrorType": type(exc).__name__,
        "Error": str(exc),
        "LastChildEvidence": LAST_CHILD_EVIDENCE,
        "CreatedLocal": datetime.now().isoformat(),
    }
    path = temp_dir(root) / "U1H4A_FAILURE_DIAGNOSTIC.json"
    path.write_text(json.dumps(diagnostic, indent=2) + "\n", encoding="utf-8")


def archive_active_obsolete(root: Path) -> int:
    archive_root = root / "A-Archive" / f"U1H4A_V10_REPLACED_{timestamp()}"
    moved = 0
    for relative_text in sorted(ACTIVE_OBSOLETE_FILES):
        relative = Path(relative_text)
        moved += int(_archive(root / relative, archive_root, relative))
    return moved


def self_test(root: Path) -> None:
    validate_foundation(root)
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    if not prepared.is_dir():
        raise InstallError("PREPARED_DIRECTORY_MISSING")
    h4_core = prepared / "01_PROGRAMS" / "core"
    component_tests = [
        (h4_core / "u001_h4_contract_core_v1_0.py", "PASS-12", "H3 package identity and H4 structural primitives."),
        (h4_core / "u001_h4_frame_policy_compiler_v1_0.py", "PASS-16", "Explicit approved within-frame policy with no hold."),
        (h4_core / "u001_h4_raw_frame_view_builder_v1_0.py", "PASS-24", "Streaming one-row-per-observed-FrameCount construction."),
    ]
    total = 0
    for program, marker, explanation in component_tests:
        print(f"[CHECK] {explanation}")
        output = run_child([sys.executable, str(program), "--self-test"], program.stem.upper())
        if marker not in output:
            raise InstallError(f"SELFTEST_MARKER:{program.name}")
        total += int(marker.rsplit("-", 1)[1])
    print("[CHECK] Independent H3 -> H4A replay, policy gate, determinism and tamper rejection.")
    with tempfile.TemporaryDirectory(prefix="u001_h4a_installer_") as temporary:
        acceptance = prepared / "05_TESTS" / "tools" / "u001_h4a_raw_frame_view_acceptance_v1_1.py"
        acceptance_output = run_child(
            [
                sys.executable,
                str(acceptance),
                "--validate",
                "--h4-core-dir",
                str(h4_core),
                "--h3-core-dir",
                str(root / "01_PROGRAMS" / "core"),
                "--h3-acceptance",
                str(root / "05_TESTS" / "tools" / "u001_h3_sync_semantic_acceptance_v1_0.py"),
                "--output",
                str(Path(temporary) / "acceptance"),
            ],
            "H4A_INDEPENDENT_ACCEPTANCE",
        )
    if "T18" not in acceptance_output or "C0" not in acceptance_output.replace("-", ""):
        raise InstallError("H4A_ACCEPTANCE_MARKER")
    print(f"U1H4AST-N4-T{total + 18}-X0-C0")


def copy_payload(prepared: Path, root: Path) -> tuple[int, Path | None]:
    archive_root: Path | None = None
    copied = 0
    for relative_text in sorted(EXPECTED_FILES):
        relative = Path(relative_text)
        source = prepared / relative
        target = root / relative
        if target.exists():
            if archive_root is None:
                archive_root = root / "A-Archive" / f"U1H4A_REPLACED_{timestamp()}"
            _archive(target, archive_root, relative)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        copied += 1
    return copied, archive_root


def execute(root: Path) -> None:
    self_test(root)
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    archived_obsolete = archive_active_obsolete(root)
    copied, replaced = copy_payload(prepared, root)
    for relative_text in sorted(EXPECTED_FILES):
        source = prepared / relative_text
        installed = root / relative_text
        if source.read_bytes() != installed.read_bytes():
            raise InstallError(f"INSTALLED_FILE_IDENTITY:{relative_text}")

    acceptance = root / "05_TESTS" / "tools" / "u001_h4a_raw_frame_view_acceptance_v1_1.py"
    evidence_root = root / "05_TESTS" / "evidence" / f"U1H4A_{timestamp()}"
    evidence_root.mkdir(parents=True, exist_ok=False)
    with tempfile.TemporaryDirectory(prefix="u1h4a_post_") as temporary:
        acceptance_root = Path(temporary) / "acceptance"
        output = run_child(
            [
                sys.executable,
                str(acceptance),
                "--validate",
                "--h4-core-dir",
                str(root / "01_PROGRAMS" / "core"),
                "--h3-core-dir",
                str(root / "01_PROGRAMS" / "core"),
                "--h3-acceptance",
                str(root / "05_TESTS" / "tools" / "u001_h3_sync_semantic_acceptance_v1_0.py"),
                "--output",
                str(acceptance_root),
            ],
            "INSTALLED_H4A_ACCEPTANCE",
        )
        if "T18" not in output or "C0" not in output.replace("-", ""):
            raise InstallError("INSTALLED_ACCEPTANCE_MARKER")
        receipt_path = acceptance_root / "u001_h4a_acceptance_receipt.json"
        capsule_path = acceptance_root / "u001_h4a_acceptance_capsule.txt"
        receipt = json.loads(receipt_path.read_text(encoding="utf-8"))
        shutil.copy2(receipt_path, evidence_root / receipt_path.name)
        shutil.copy2(capsule_path, evidence_root / capsule_path.name)
    if receipt.get("Status") != "PASS" or receipt.get("CompletionCode") != 0:
        raise InstallError("INSTALLED_ACCEPTANCE_RECEIPT")
    h3_output = run_child(
        [sys.executable, str(root / "01_PROGRAMS" / "core" / "u001_h3_sync_semantic_engine_v1_0.py"), "--self-test"],
        "H3_COMPATIBILITY",
    )
    if "PASS-12" not in h3_output:
        raise InstallError("H3_COMPATIBILITY_MARKER")
    result = {
        "Node": "U1H4A",
        "ProgramVersion": PROGRAM_VERSION,
        "InstalledFileCount": copied,
        "InstalledProgramCount": 3,
        "FramePolicyCompilerQualified": True,
        "IndependentAcceptanceCount": 1,
        "H3CompatibilityPassed": True,
        "ReplacedFilesArchived": replaced is not None,
        "ObsoleteV10FilesArchived": archived_obsolete,
        "AcceptanceEvidenceFolder": evidence_root.relative_to(root).as_posix(),
        "CompletionCode": 0,
    }
    (temp_dir(root) / "U1H4A_EXECUTION_RESULT.json").write_text(
        json.dumps(result, indent=2) + "\n", encoding="utf-8"
    )
    print(f"U1H4A-N3-P1-Q1-H1-A{copied}-X0-C0")


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description="Install U001 H4A raw Frame View")
    action = result.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    result.add_argument("--u001-root", type=Path)
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    root: Path | None = None
    try:
        root = resolve_u001_root(args.u001_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            self_test(root)
        else:
            execute(root)
        return 0
    except (InstallError, OSError, ValueError, json.JSONDecodeError, zipfile.BadZipFile) as exc:
        if root is not None:
            try:
                write_failure_diagnostic(root, exc)
            except Exception:
                pass
        short_detail = " ".join(str(exc).split())[-120:] or "NO_DETAIL"
        print(
            f"U1H4AIE-K{type(exc).__name__.upper()}-D{short_detail}-"
            "L04_TEMP/U1H4A_FAILURE_DIAGNOSTIC.json-X1-C7",
            file=sys.stderr,
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
