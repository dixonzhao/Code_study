def regular_file(path: Path) -> bool:
    """Accept ordinary files only as governed inputs."""

    return path.is_file() and not path.is_symlink()


def canonical_bytes(value: object) -> bytes:
    """Serialize one control document deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def canonical_text(value: object) -> str:
    """Return deterministic UTF-8 JSON text."""

    return canonical_bytes(value).decode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file without loading it completely into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def sha256_tree(root: Path) -> str:
    """Hash all regular files under one read-only synthetic package."""

    digest = hashlib.sha256()
    for path in sorted(root.rglob("*"), key=lambda item: item.as_posix()):
        if not regular_file(path):
            continue
        relative = path.relative_to(root).as_posix().encode("utf-8")
        digest.update(len(relative).to_bytes(8, "big"))
        digest.update(relative)
        with path.open("rb") as stream:
            for block in iter(lambda: stream.read(1024 * 1024), b""):
                digest.update(block)
    return digest.hexdigest()


def check(value: bool, reason: str, checks: List[bool]) -> None:
    """Append a visible check or fail closed."""

    if not value:

    checks.append(True)

def copy_environment(source: Path, target: Path) -> None:
    """Build an isolated U22 workspace without copying company data."""

    target.mkdir(parents=True)
    for relative in (
        "00_CONTROL/contracts",
        "00_CONTROL/registry",
        "01_PROGRAMS/core",
        "02_CONFIG/run_configs",
        "03_CASES",
        "05_TESTS/integration",
        "07_OUTPUT",

        (target / relative).mkdir(parents=True, exist_ok=True)
    for stage in STAGES:
        (target / "01_PROGRAMS/modules" / stage).mkdir(parents=True)
    for name in CONTRACT_NAMES:
        source_file = source / "00_CONTROL/contracts" / name

        shutil.copy2(source_file, target / "01_PROGRAMS/core" / name)

def run_vertical(workspace_root: Path) -> str:
    """Prove all eight Stage boundaries using one isolated Golden Run."""

    checks: List[bool] = []
    with tempfile.TemporaryDirectory(prefix="u22a7_vertical_") as temporary:
        root = Path(temporary) / "U-UDP002"
        copy_environment(workspace_root, root)
        input_package = root / "03_CASES/SYNTHETIC_U001_PACKAGE"
        input_package.mkdir()
        (input_package / "package_identity.txt").write_text(
            "U22A7 SYNTHETIC READ ONLY INPUT\n",
            encoding="utf-8",
        )
        input_hash_before = sha256_tree(input_package)
        module_ids: List[str] = []
        previous_type: List[str] = []
        for index, stage in enumerate(STAGES):
            output_type = (
                "SYNTHETIC_S7_HANDOFF"
                if stage == "S7"
                else "SYNTHETIC_" + stage
            )
            module_id = write_module(
                root,
                stage,
                list(previous_type),
                output_type,
            )
            module_ids.append(module_id)
            qualification = qualify(root, module_id)
            check(
                qualification.returncode == 0,
                "QUALIFICATION_" + stage,
                checks,
            )
            previous_type = [output_type]
        check(len(module_ids) == 8, "MODULE_COUNT", checks)
        prepared = prepare(root, run_configuration(module_ids))
        check(prepared.returncode == 0, "PREFLIGHT", checks)
        first_execution = execute(root)
        check(
            first_execution.returncode == 0
            and "N-M8-A8-O0-R0-S1" in first_execution.stdout,
            "EXECUTION_" + first_execution.stdout.strip(),
            checks,
        )
        run_root = root / "07_OUTPUT/U22A7_VERTICAL"
        status = json.loads(
            (run_root / "run_status.json").read_text(encoding="utf-8")
        )
        check(status["Status"] == "SUCCEEDED", "RUN_STATUS", checks)
        check(status["CompletedModuleCount"] == 8, "COMPLETED_COUNT", checks)
        check(status["LastCompletedStage"] == "S7", "LAST_STAGE", checks)
        index = json.loads(
            (run_root / "run_artifact_index_v0_1.json").read_text(
                encoding="utf-8"
            )
        )
        artifacts = index["Artifacts"]
        check(len(artifacts) == 8, "ARTIFACT_COUNT", checks)
        lineage_links = sum(
            len(item["ParentArtifactIds"])
            for item in artifacts
        )
        check(lineage_links == 7, "LINEAGE_COUNT", checks)
        check(
            sum(item["ArtifactType"] == "SYNTHETIC_S7_HANDOFF" for item in artifacts)
            == 1,
            "HANDOFF_COUNT",
            checks,
        )
        check(
            len(list((run_root / "01_LOGS").glob("*.json"))) == 8,
            "EXECUTION_RECORD_COUNT",
            checks,
        )
        diagnostics = run_root / "02_DIAGNOSTICS/modules"
        check(
            len([path for path in diagnostics.iterdir() if path.is_dir()]) == 8,
            "DIAGNOSTIC_COUNT",
            checks,
        )
        check(
            not any((run_root / "00_STAGING").iterdir()),
            "STAGING_NOT_EMPTY",
            checks,
        )
        audit_result = audit(root)
        check(audit_result.returncode == 0, "ARTIFACT_AUDIT", checks)
        check(
            sha256_tree(input_package) == input_hash_before,
            "INPUT_MUTATED",
            checks,
        )
        replay = execute(root)
        check(
            replay.returncode == 0
            and "N-M8-A8-O0-R1-S1" in replay.stdout,
            "CONTROLLED_REPLAY",
            checks,
        )
        check(len(checks) == 23, "VERTICAL_CHECK_COUNT", checks)
    return VERTICAL_CAPSULE


def frozen_file_paths(workspace_root: Path) -> List[Path]:
    """Return the complete Phase A file set protected by the receipt."""

    paths = [

        for name in CONTRACT_NAMES

    paths.extend(

        for name in CORE_NAMES

    paths.append(workspace_root / INSTALLED_RELATIVE)
    return paths


def build_freeze_receipt(
    workspace_root: Path,
    accepted_capsule: str,
) -> Dict[str, object]:
    """Build one deterministic Phase A freeze receipt."""

    if accepted_capsule != VERTICAL_CAPSULE:
        raise VerticalError("VERTICAL_CAPSULE")
    frozen_files = []
    for path in frozen_file_paths(workspace_root):
        if not regular_file(path):
            raise VerticalError("FROZEN_FILE_MISSING_" + path.name)
        frozen_files.append(
            {
                "RelativePath": path.relative_to(workspace_root).as_posix(),
                "Sha256": sha256_file(path),
            }
        )
    receipt: Dict[str, object] = {
        "ReceiptContract": "U002_U22_FOUNDATION_FREEZE",
        "ReceiptVersion": "0.1",
        "ContractVersion": CONTRACT_VERSION,
        "Status": "FROZEN",
        "AcceptedNodes": [
            "U22A0",
            "U22A1",
            "U22A2",
            "U22A3",
            "U22A4",
            "U22A5",
            "U22A6",
            "U22A7",
        ],
        "AcceptedVerticalCapsule": accepted_capsule,
        "FrozenFiles": sorted(
            frozen_files,
            key=lambda item: str(item["RelativePath"]),
        ),
    }
    receipt["ReceiptSha256"] = hashlib.sha256(canonical_bytes(receipt)).hexdigest()
    return receipt


def write_freeze_receipt(
    workspace_root: Path,
    accepted_capsule: str,
) -> str:
    """Atomically install the deterministic Phase A freeze receipt."""

    receipt = build_freeze_receipt(workspace_root, accepted_capsule)
    path = workspace_root / FREEZE_RECEIPT_RELATIVE
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    if temporary.exists():
        raise VerticalError("FREEZE_TEMP_EXISTS")
    temporary.write_bytes(canonical_bytes(receipt))
    os.replace(temporary, path)
    return "U22A7F N-F11-N8-V1-X0 C-0"


def audit_freeze(workspace_root: Path) -> str:
    """Fail closed if any frozen Phase A file or receipt field changed."""

    path = workspace_root / FREEZE_RECEIPT_RELATIVE
    if not regular_file(path):
        raise VerticalError("FREEZE_RECEIPT_MISSING")
    try:
        receipt = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise VerticalError("FREEZE_RECEIPT_JSON") from exc
    required = {
        "ReceiptContract",
        "ReceiptVersion",
        "ContractVersion",
        "Status",
        "AcceptedNodes",
        "AcceptedVerticalCapsule",
        "FrozenFiles",
        "ReceiptSha256",
    }
    if not isinstance(receipt, dict) or set(receipt) != required:
        raise VerticalError("FREEZE_RECEIPT_SHAPE")
    expected_hash = receipt.pop("ReceiptSha256")
    observed_hash = hashlib.sha256(canonical_bytes(receipt)).hexdigest()
    if expected_hash != observed_hash:
        raise VerticalError("FREEZE_RECEIPT_HASH")
    if receipt["Status"] != "FROZEN" or len(receipt["AcceptedNodes"]) != 8:
        raise VerticalError("FREEZE_STATUS")
    if receipt["AcceptedVerticalCapsule"] != VERTICAL_CAPSULE:
        raise VerticalError("FREEZE_VERTICAL")
    files = receipt["FrozenFiles"]
    if not isinstance(files, list) or len(files) != 11:
        raise VerticalError("FREEZE_FILE_COUNT")
    for item in files:
        if not isinstance(item, dict) or set(item) != {"RelativePath", "Sha256"}:
            raise VerticalError("FREEZE_FILE_SHAPE")
        candidate = workspace_root / str(item["RelativePath"])
        if not regular_file(candidate):
            raise VerticalError("FREEZE_FILE_MISSING")
        if sha256_file(candidate) != item["Sha256"]:
            raise VerticalError("FREEZE_FILE_CHANGED")
    return "U22A7FA N-F11-N8-V1-X0 C-0"

    check(VERTICAL_CAPSULE.endswith("C-0"), "CAPSULE", checks)
    check(len(frozen_file_paths(workspace_root)) == 11, "FROZEN_PATHS", checks)
    check(canonical_bytes({"b": 1, "a": 2}).startswith(b"{"), "CANONICAL", checks)
    check(len(sha256_file(workspace_root / "00_CONTROL/contracts" / CONTRACT_NAMES[0])) == 64, "FILE_HASH", checks)
    check(sys.executable != "", "PYTHON", checks)
    check(len(checks) == 23, "SELFTEST_COUNT", checks)
    return "U22A7V-ST N-T24-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Create explicit vertical, freeze-write, and freeze-audit modes."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--run-vertical", action="store_true")
    mode.add_argument("--write-freeze-receipt", action="store_true")
    mode.add_argument("--audit-freeze", action="store_true")
    parser.add_argument("--accepted-capsule")
    parser.add_argument("--workspace-root", required=True)
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    """Run one explicit integration or freeze-control responsibility."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        workspace_root = require_workspace(Path(args.workspace_root))
        if args.self_test:
            print(self_test(workspace_root))
            return 0
        if args.run_vertical:
            print(run_vertical(workspace_root))
            return 0
        if args.write_freeze_receipt:
            if args.accepted_capsule is None:
                raise VerticalError("ACCEPTED_CAPSULE_REQUIRED")
            print(write_freeze_receipt(workspace_root, args.accepted_capsule))
            return 0
        print(audit_freeze(workspace_root))
        return 0
    except VerticalError as exc:
        print("U22A7V N-M0-S0-A0-L0-R0-I0-X1 C-7 R-" + str(exc))
        return 7
    except Exception as exc:
        print(
            "U22A7V N-M0-S0-A0-L0-R0-I0-X1 C-7 R-"
            + exc.__class__.__name__.upper()
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
