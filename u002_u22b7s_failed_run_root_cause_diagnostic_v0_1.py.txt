#!/usr/bin/env python3
"""Diagnose one failed governed U002 Run without changing any Run artifact."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import re
import sys
import tempfile
from typing import Dict, List, Mapping, Optional, Sequence, Tuple


PROGRAM_ID = "UDP002.U22B7S.FailedRunRootCauseDiagnostic"
PROGRAM_VERSION = "0.1.0"
DEFAULT_RUN_ID = "U22B7R_GEOMETRY_RELINK_F41380_41780_V03"
HISTORICAL_FAILURE_COVERAGE = (
    "FP019_LOW_BANDWIDTH_CAPSULE",
    "FP023_NESTED_CHILD_ROOT_CAUSE",
    "FP025_TWO_LAYER_DIAGNOSTIC",
    "FP028_HISTORICAL_MEMORY_GATE",
    "FP029_RULE_TO_TEST_MAPPING",
)
HISTORICAL_FAILURE_TEST_MAP = {
    "FP019_LOW_BANDWIDTH_CAPSULE": ("T15_CAPSULE",),
    "FP023_NESTED_CHILD_ROOT_CAUSE": ("T03_REASON", "T04_DETAIL"),
    "FP025_TWO_LAYER_DIAGNOSTIC": ("T03_REASON", "T04_DETAIL", "T15_CAPSULE"),
    "FP028_HISTORICAL_MEMORY_GATE": ("T18_HISTORY_COVERAGE",),
    "FP029_RULE_TO_TEST_MAPPING": ("T19_HISTORY_TEST_MAP",),
}


class DiagnosticError(RuntimeError):
    """Represent one deterministic read-only diagnostic failure."""


def token(value: object, limit: int = 100) -> str:
    """Convert one value into a bounded manual-transfer token."""

    text = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper()).strip("_")
    return (text or "NONE")[:limit]


def load_json(path: Path, label: str, required: bool = True) -> Dict[str, object]:
    """Load one JSON object, optionally accepting an absent evidence file."""

    if not path.is_file() or path.is_symlink():
        if required:
            raise DiagnosticError(label + "_MISSING")
        return {}
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise DiagnosticError(label + "_JSON_" + type(error).__name__) from error
    if not isinstance(value, dict):
        raise DiagnosticError(label + "_TYPE")
    return value


def read_text(path: Path) -> str:
    """Read one optional diagnostic text file."""

    if not path.is_file() or path.is_symlink():
        return ""
    return path.read_text(encoding="utf-8-sig", errors="replace")


def inside(child: Path, parent: Path) -> bool:
    """Return True only when a resolved path remains under its owner root."""

    try:
        child.resolve().relative_to(parent.resolve())
        return True
    except (OSError, ValueError):
        return False


def deepest_reason(stdout: str, stderr: str, fallback: object) -> Tuple[str, str]:
    """Extract the deepest stable reason and bounded terminal evidence."""

    combined = stdout + "\n" + stderr
    matches = re.findall(r"(?:^|[\s|])(?:R|K)-([A-Za-z0-9_.-]+)", combined)
    if matches:
        reason = token(matches[-1], 72)
    else:
        exception_lines = [
            line.strip()
            for line in stderr.splitlines()
            if re.search(r"(?:Error|Exception):", line)
        ]
        reason = token(exception_lines[-1] if exception_lines else fallback, 72)
    meaningful = [line.strip() for line in combined.splitlines() if line.strip()]
    detail = token(meaningful[-1] if meaningful else fallback, 180)
    return reason, detail


def correction_class(reason: str, module_id: object) -> str:
    """Map a root reason to its owning correction class."""

    reason = token(reason, 100)
    if "CONFIGURATION_SHAPE" in reason:
        return "S1_CONFIGURATION_KEYSET"
    if "CONTEXT" in reason or "SEED" in reason:
        return "S1_CONTEXT_OR_SEED_POLICY"
    if "PROFILE_INVENTORY" in reason or "BINDING_PROFILE_HASH" in reason:
        return "PROFILE_TO_S0_INVENTORY_PROVENANCE"
    if "BINDING" in reason or "SIGNAL_ROLE" in reason:
        return "PROFILE_ROLE_BINDING"
    if "MAX_AGE" in reason or reason.startswith("INTEGER_"):
        return "TYPED_HOLD_POLICY_COMPILATION"
    if "FRAME_VIEW_MISSING" in reason:
        return "U001_FRAME_VIEW_REQUIRED_COLUMNS"
    if "FRAME" in reason and ("RANGE" in reason or "SCOPE" in reason):
        return "RUN_FRAME_DOMAIN"
    if "INPUT" in reason or "PACKAGE" in reason or "PARENT" in reason:
        return "U001_PACKAGE_LINKAGE"
    if module_id == "CameraRadar.S1.CameraTargetContinuityArbitration":
        return "CAMERA_S1_RUNTIME"
    return "UNCLASSIFIED_MODULE_RUNTIME"


def resolve_diagnostic_root(
    run_root: Path,
    record: Mapping[str, object],
) -> Path:
    """Resolve one Executor-owned diagnostic directory from its execution record."""

    relative = str(record.get("DiagnosticRoot", ""))
    if not relative or "\\" in relative:
        raise DiagnosticError("DIAGNOSTIC_ROOT_IDENTITY")
    root = (run_root / Path(relative)).resolve()
    if not root.is_dir() or not inside(root, run_root):
        raise DiagnosticError("DIAGNOSTIC_ROOT_MISSING")
    return root


def collect_records(run_root: Path) -> List[Dict[str, object]]:
    """Collect all Module execution records and their retained child evidence."""

    log_root = run_root / "01_LOGS"
    if not log_root.is_dir():
        raise DiagnosticError("MODULE_LOG_ROOT_MISSING")
    records: List[Dict[str, object]] = []
    for log_path in sorted(log_root.glob("*.json")):
        record = load_json(log_path, "MODULE_LOG")
        diagnostic_root = resolve_diagnostic_root(run_root, record)
        request = load_json(
            diagnostic_root / "module_request.json",
            "MODULE_REQUEST",
            required=False,
        )
        stdout = read_text(diagnostic_root / "stdout.txt")
        stderr = read_text(diagnostic_root / "stderr.txt")
        reason, detail = deepest_reason(stdout, stderr, record.get("Reason", "UNKNOWN"))
        configuration = request.get("Configuration")
        if not isinstance(configuration, dict):
            configuration = {}
        records.append(
            {
                "Sequence": record.get("Sequence"),
                "Stage": record.get("Stage"),
                "ModuleId": record.get("ModuleId"),
                "ModuleVersion": record.get("ModuleVersion"),
                "Status": record.get("Status"),
                "ExitCode": record.get("ExitCode"),
                "ExecutorReason": record.get("Reason"),
                "DeepReason": reason,
                "DeepDetail": detail,
                "DiagnosticRoot": str(diagnostic_root),
                "StdoutBytes": len(stdout.encode("utf-8")),
                "StderrBytes": len(stderr.encode("utf-8")),
                "StdoutTail": stdout[-4000:],
                "StderrTail": stderr[-4000:],
                "Request": request,
                "Configuration": configuration,
            }
        )
    if not records:
        raise DiagnosticError("MODULE_RECORDS_EMPTY")
    return records


def profile_summary(
    workspace_root: Path,
    configuration: Mapping[str, object],
) -> Dict[str, object]:
    """Summarize the selected profile without accepting an external path."""

    relative = str(configuration.get("BindingProfilePath", ""))
    if not relative or "\\" in relative:
        return {"Present": False, "Reason": "NO_SAFE_PROFILE_PATH"}
    path = (workspace_root / Path(relative)).resolve()
    if not path.is_file() or not inside(path, workspace_root):
        return {"Present": False, "Reason": "PROFILE_MISSING"}
    profile = load_json(path, "BINDING_PROFILE")
    bindings = profile.get("Bindings")
    if not isinstance(bindings, dict):
        bindings = {}
    counts: Dict[str, int] = {}
    max_ages: Dict[str, object] = {}
    for role, selected in bindings.items():
        counts[str(role)] = len(selected) if isinstance(selected, list) else -1
        if isinstance(selected, list) and len(selected) == 1 and isinstance(selected[0], dict):
            max_ages[str(role)] = selected[0].get("MaxAgeFrames")
    return {
        "Present": True,
        "Path": str(path),
        "ProfileVersion": profile.get("BindingProfileVersion"),
        "SourceInventorySha256": profile.get("SourceInventorySha256"),
        "BindingCounts": counts,
        "MaxAgeFrames": max_ages,
        "EmptyBindingCount": sum(value == 0 for value in counts.values()),
    }


def diagnose(
    workspace_root: Path,
    run_id: str,
    b7r_output: Path,
    output_dir: Path,
) -> Tuple[str, str, str]:
    """Read one failed Run and emit a short root-cause capsule plus full JSON."""

    run_root = (workspace_root / "07_OUTPUT" / run_id).resolve()
    if not run_root.is_dir() or not inside(run_root, workspace_root):
        raise DiagnosticError("RUN_ROOT_MISSING")
    status = load_json(run_root / "run_status.json", "RUN_STATUS")
    config = load_json(run_root / "00_CONFIG/run_configuration.json", "RUN_CONFIG")
    outer = load_json(
        b7r_output / "u22b7r_failure_diagnostic.json",
        "B7R_FAILURE",
        required=False,
    )
    records = collect_records(run_root)
    failed = [record for record in records if record.get("Status") == "FAILED"]
    if len(failed) != 1:
        raise DiagnosticError("FAILED_MODULE_COUNT_" + str(len(failed)))
    selected = failed[0]
    configuration = selected.get("Configuration")
    if not isinstance(configuration, dict):
        configuration = {}
    profile = profile_summary(workspace_root, configuration)
    reason = str(selected["DeepReason"])
    owner = correction_class(reason, selected.get("ModuleId"))
    bound = profile.get("BindingCounts")
    bound_count = sum(value == 1 for value in bound.values()) if isinstance(bound, dict) else 0
    empty_count = int(profile.get("EmptyBindingCount", 0))
    line1 = (
        f"B7S N-ST{token(status.get('Status'), 12)}-CM{status.get('CompletedModuleCount', 0)}"
        f"-SQ{selected.get('Sequence', -1)}-RC{selected.get('ExitCode', -1)}"
        f"-SO{selected.get('StdoutBytes', 0)}-SE{selected.get('StderrBytes', 0)}-X0 C-0"
    )
    line2 = (
        f"B7SK K-{token(reason, 72)}-M{token(selected.get('ModuleId'), 42)}"
        f"-D{token(owner, 64)}-X0 C-0"
    )
    line3 = (
        f"B7SD N-C{len(configuration)}-B{bound_count}-Z{empty_count}"
        f"-PV{token(profile.get('ProfileVersion'), 12)}-X0 C-0"
    )
    report = {
        "CompletionCode": 0,
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "Decision": "READ_ONLY_DIAGNOSTIC_NO_RUN_OR_ARTIFACT_MUTATION",
        "WorkspaceRoot": str(workspace_root),
        "RunId": run_id,
        "RunRoot": str(run_root),
        "RunStatus": status,
        "RunConfiguration": config,
        "OuterB7RFailure": outer,
        "ModuleRecords": records,
        "FailedModule": selected,
        "DeepReason": reason,
        "CorrectionClass": owner,
        "BindingProfileSummary": profile,
        "HistoricalFailureCoverage": list(HISTORICAL_FAILURE_COVERAGE),
        "HistoricalFailureTestMap": {
            key: list(value) for key, value in HISTORICAL_FAILURE_TEST_MAP.items()
        },
        "ReturnCapsule": [line1, line2, line3],
    }
    output_dir.mkdir(parents=True, exist_ok=True)
    (output_dir / "u22b7s_failed_run_root_cause_diagnostic.json").write_text(
        json.dumps(report, ensure_ascii=False, sort_keys=True, indent=2) + "\n",
        encoding="utf-8",
    )
    (output_dir / "u22b7s_return_capsule.txt").write_text(
        line1 + "\n" + line2 + "\n" + line3 + "\n",
        encoding="utf-8",
    )
    return line1, line2, line3


def self_test() -> None:
    """Exercise reason parsing, ownership mapping and path controls."""

    checks: List[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            raise AssertionError(name)
        checks.append(name)

    check("T01_ID", PROGRAM_ID.startswith("UDP002"))
    check("T02_VERSION", PROGRAM_VERSION == "0.1.0")
    reason, detail = deepest_reason(
        "U22B2FM N-R0-V0-S0-X1 C-7 R-CONFIGURATION_SHAPE\n",
        "",
        "MODULE_EXIT",
    )
    check("T03_REASON", reason == "CONFIGURATION_SHAPE")
    check("T04_DETAIL", "CONFIGURATION_SHAPE" in detail)
    check(
        "T05_CONFIG_OWNER",
        correction_class(reason, "X") == "S1_CONFIGURATION_KEYSET",
    )
    check(
        "T06_MAXAGE_OWNER",
        correction_class("INTEGER_CAMERA_1_DISTANCE_MAX_AGE", "X")
        == "TYPED_HOLD_POLICY_COMPILATION",
    )
    check(
        "T07_PROFILE_OWNER",
        correction_class("PROFILE_INVENTORY_HASH", "X")
        == "PROFILE_TO_S0_INVENTORY_PROVENANCE",
    )
    check(
        "T08_FRAME_OWNER",
        correction_class("FRAME_VIEW_MISSING_X", "X")
        == "U001_FRAME_VIEW_REQUIRED_COLUMNS",
    )
    check("T09_TOKEN", token("a b/c") == "A_B_C")
    check("T10_DEFAULT", DEFAULT_RUN_ID.endswith("V03"))
    with tempfile.TemporaryDirectory(prefix="u22b7s_") as temporary:
        root = Path(temporary)
        item = root / "item.json"
        item.write_text('{"a": 1}\n', encoding="utf-8")
        check("T11_JSON", load_json(item, "ITEM")["a"] == 1)
        check("T12_OPTIONAL", load_json(root / "missing.json", "M", False) == {})
        check("T13_INSIDE", inside(item, root))
        check("T14_OUTSIDE", not inside(root, item))
    check("T15_CAPSULE", len("B7SK K-CONFIGURATION_SHAPE-MCAMERA_S1-DS1_CONFIGURATION_KEYSET-X0 C-0") < 100)
    check("T16_READ_ONLY", "READ_ONLY" in diagnose.__doc__.upper() or "READ" in diagnose.__doc__.upper())
    check(
        "T17_NO_CHILD_EXECUTION",
        not any(
            line.strip() == "import subprocess"
            for line in Path(__file__).read_text(encoding="utf-8").splitlines()
        ),
    )
    check("T18_HISTORY_COVERAGE", len(HISTORICAL_FAILURE_COVERAGE) == 5)
    check(
        "T19_HISTORY_TEST_MAP",
        set(HISTORICAL_FAILURE_TEST_MAP) == set(HISTORICAL_FAILURE_COVERAGE)
        and all(
            test_name in checks or test_name == "T19_HISTORY_TEST_MAP"
            for test_names in HISTORICAL_FAILURE_TEST_MAP.values()
            for test_name in test_names
        ),
    )
    check("T20_COUNT", len(checks) == 19)
    print("UDP002-U22B7S-SELFTEST-PASS-20 C-0")


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test or diagnose the immutable V03 failure evidence."""

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--diagnose", action="store_true")
    parser.add_argument("--workspace-root", default="U-UDP002")
    parser.add_argument("--run-id", default=DEFAULT_RUN_ID)
    parser.add_argument("--b7r-output", default="U-UDP002/04_TEMP/U22B7R3_OUTPUT")
    parser.add_argument("--output-dir", default="U-UDP002/04_TEMP/U22B7S_OUTPUT")
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            self_test()
            return 0
        if not args.diagnose:
            raise DiagnosticError("MODE_REQUIRED")
        lines = diagnose(
            Path(args.workspace_root).resolve(),
            args.run_id,
            Path(args.b7r_output).resolve(),
            Path(args.output_dir).resolve(),
        )
        for line in lines:
            print(line)
        return 0
    except Exception as error:
        print("B7SE K-" + token(error, 120) + "-X1 C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
