@echo off
setlocal EnableExtensions EnableDelayedExpansion

rem C-CODE top-level naming migration v1.0
rem Run from the C-CODE root in Windows CMD.
rem Renames exactly seven directories and rolls back completed renames if any
rem later rename or final verification fails.
rem Does NOT recursively rename U001 Portable internal product folders.

for %%D in (UDP001 UDP002 UDP003 ARCH) do (
  if not exist ".\%%D" (
    echo CDNAME1-ERROR-MISSING-%%D C-7
    exit /b 7
  )
)

for %%D in (U-UDP001 U-UDP002 U-UDP003 A-Archive) do (
  if exist ".\%%D" (
    echo CDNAME1-ERROR-DESTINATION-EXISTS-%%D C-7
    exit /b 7
  )
)

for %%D in (UDP001 UDP002 UDP003) do (
  if not exist ".\%%D\ARCH" (
    echo CDNAME1-ERROR-MISSING-%%D-ARCH C-7
    exit /b 7
  )
  if exist ".\%%D\A-Archive" (
    echo CDNAME1-ERROR-DESTINATION-EXISTS-%%D-A-Archive C-7
    exit /b 7
  )
)

set "S1=0"
set "S2=0"
set "S3=0"
set "S4=0"
set "S5=0"
set "S6=0"
set "S7=0"

ren ".\UDP001\ARCH" "A-Archive" || goto :rollback
set "S1=1"
ren ".\UDP002\ARCH" "A-Archive" || goto :rollback
set "S2=1"
ren ".\UDP003\ARCH" "A-Archive" || goto :rollback
set "S3=1"

ren ".\UDP001" "U-UDP001" || goto :rollback
set "S4=1"
ren ".\UDP002" "U-UDP002" || goto :rollback
set "S5=1"
ren ".\UDP003" "U-UDP003" || goto :rollback
set "S6=1"
ren ".\ARCH" "A-Archive" || goto :rollback
set "S7=1"

for %%D in (U-UDP001 U-UDP002 U-UDP003) do (
  if not exist ".\%%D\A-Archive" goto :verify_fail
)
if not exist ".\A-Archive" goto :verify_fail

for %%D in (UDP001 UDP002 UDP003 ARCH) do (
  if exist ".\%%D" goto :verify_fail
)

echo CDNAME1 N-R7-V7-P3-A4-X0 C-0
exit /b 0

:verify_fail
echo CDNAME1-ERROR-VERIFY-STARTING-ROLLBACK C-7
goto :rollback

:rollback
set "RB=0"

if "!S7!"=="1" (
  ren ".\A-Archive" "ARCH" >nul 2>&1
  if errorlevel 1 set "RB=1"
)
if "!S6!"=="1" (
  ren ".\U-UDP003" "UDP003" >nul 2>&1
  if errorlevel 1 set "RB=1"
)
if "!S5!"=="1" (
  ren ".\U-UDP002" "UDP002" >nul 2>&1
  if errorlevel 1 set "RB=1"
)
if "!S4!"=="1" (
  ren ".\U-UDP001" "UDP001" >nul 2>&1
  if errorlevel 1 set "RB=1"
)
if "!S3!"=="1" (
  ren ".\UDP003\A-Archive" "ARCH" >nul 2>&1
  if errorlevel 1 set "RB=1"
)
if "!S2!"=="1" (
  ren ".\UDP002\A-Archive" "ARCH" >nul 2>&1
  if errorlevel 1 set "RB=1"
)
if "!S1!"=="1" (
  ren ".\UDP001\A-Archive" "ARCH" >nul 2>&1
  if errorlevel 1 set "RB=1"
)

if "!RB!"=="0" (
  echo CDNAME1-ROLLBACK-COMPLETE C-7
) else (
  echo CDNAME1-ROLLBACK-INCOMPLETE C-7
)
exit /b 7
