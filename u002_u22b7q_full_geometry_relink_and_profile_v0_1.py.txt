class RelinkError(RuntimeError):
    """Represent one deterministic relink or profiling failure."""

    def __init__(self, reason: str, detail: object = "NA", stage: str = "B7Q"):
        self.reason = machine_token(reason, 72)
        self.detail = machine_token(detail, 180)
        self.stage = machine_token(stage, 24)
        super().__init__(self.reason)


def machine_token(value: object, limit: int = 100) -> str:
    """Return a bounded machine-readable token."""

    text = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper()).strip("_")
    return (text or "UNKNOWN")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize one JSON object deterministically."""

    return (
        json.dumps(
            value,
            ensure_ascii=False,
            sort_keys=True,
            indent=2,
            allow_nan=False,
        )
        + "\n"
    ).encode("utf-8")


def write_json_atomic(path: Path, value: Mapping[str, object]) -> None:
    """Write one JSON Artifact without exposing a partial file."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def write_text_atomic(path: Path, value: str) -> None:
    """Write one UTF-8 text Artifact atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_text(value, encoding="utf-8", newline="\n")
    os.replace(temporary, path)


def load_json(path: Path, label: str) -> Dict[str, object]:
    """Load one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise RelinkError(label + "_MISSING", path.name, "DISCOVER")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise RelinkError(label + "_JSON", type(error).__name__, "DISCOVER") from error
    if not isinstance(value, dict):
        raise RelinkError(label + "_TYPE", path.name, "DISCOVER")
    return value


def sha256_file(path: Path) -> str:
    """Hash one file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def snapshot_files(paths: Iterable[Path]) -> Dict[str, Tuple[int, int, str]]:
    """Capture metadata and hashes for immutable inputs."""

    result: Dict[str, Tuple[int, int, str]] = {}
    for path in paths:
        resolved = path.resolve()
        if not resolved.is_file() or resolved.is_symlink():
            raise RelinkError("SNAPSHOT_FILE", resolved.name, "VERIFY")
        stat = resolved.stat()
        result[str(resolved)] = (stat.st_size, stat.st_mtime_ns, sha256_file(resolved))
    return result


def normalize_can_id(value: object) -> str:
    """Normalize a CAN identity without changing its numeric value."""

    text = str(value).strip().lower()
    try:
        number = int(text, 16)
    except ValueError as error:
        raise RelinkError("CAN_ID", value, "PACKAGE") from error
    return f"0x{number:X}".lower()


def normalize_identifier(value: object) -> str:
    """Ignore harmless separators in one signal identifier."""

    return re.sub(r"[^a-z0-9]+", "", str(value).strip().casefold())


def read_csv_header(path: Path) -> List[str]:
    """Read only one CSV header."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        return next(csv.reader(stream), [])


def count_csv_rows(path: Path) -> int:
    """Count CSV data rows with bounded memory."""

    lines = 0
    last = b""
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(16 * 1024 * 1024), b""):
            lines += block.count(b"\n")
            last = block[-1:]
    if last and last != b"\n":
        lines += 1
    return max(lines - 1, 0)


def resolve_geometry(catalog_path: Path) -> Dict[str, str]:
    """Resolve all four paired Camera geometry identities exactly."""

    with catalog_path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    if not rows:
        raise RelinkError("CATALOG_EMPTY", catalog_path.name, "PACKAGE")
    fields = set(rows[0])
    required = {"CanonicalName", "CAN_ID", "Channel", "SignalName"}
    if not required.issubset(fields):
        raise RelinkError("CATALOG_SCHEMA", ".".join(sorted(fields)), "PACKAGE")
    resolved: Dict[str, str] = {}
    for role, (channel, can_id, signal_name) in GEOMETRY_IDENTITIES.items():
        expected = normalize_identifier(signal_name)
        matches = []
        for row in rows:
            try:
                observed_can = normalize_can_id(row.get("CAN_ID", ""))
            except RelinkError:
                continue
            if str(row.get("Channel", "")).strip() != channel:
                continue
            if observed_can != normalize_can_id(can_id):
                continue
            signal = normalize_identifier(row.get("SignalName", ""))
            canonical = normalize_identifier(row.get("CanonicalName", ""))
            if signal == expected or canonical.endswith(expected):
                matches.append(str(row.get("CanonicalName", "")).strip())
        if len(matches) != 1 or not matches[0]:
            raise RelinkError("GEOMETRY_IDENTITY", f"{role}:{len(matches)}", "PACKAGE")
        resolved[role] = matches[0]
    return resolved


def validate_full_package(
    b7p2_root: Path,
    minimum_frame_rows: int = 50_000,
) -> Tuple[Path, Dict[str, object], Dict[str, str]]:
    """Verify the accepted B7P2 manifest and its immutable six-file package."""

    manifest = load_json(
        b7p2_root / "u22b7p2_full_materialization_manifest.json",
        "B7P2_MANIFEST",
    )
    if manifest.get("CompletionCode") != 0:
        raise RelinkError("B7P2_NOT_ACCEPTED", manifest.get("CompletionCode"), "PACKAGE")
    if manifest.get("Correction") != "ZERO_TO_EXACT_POSITIVE_SOURCE_ROW_LIMIT":
        raise RelinkError("B7P2_CORRECTION", manifest.get("Correction"), "PACKAGE")
    verification = manifest.get("Verification")
    if not isinstance(verification, dict):
        raise RelinkError("B7P2_VERIFICATION", "TYPE", "PACKAGE")
    package = Path(str(verification.get("Package", ""))).resolve()
    if not package.is_dir() or package.is_symlink():
        raise RelinkError("FULL_PACKAGE", package.name, "PACKAGE")
    observed_files = sorted(path.name for path in package.iterdir() if path.is_file())
    if observed_files != sorted(FINAL_PACKAGE_FILES):
        raise RelinkError("FULL_PACKAGE_FILESET", len(observed_files), "PACKAGE")
    expected_files = verification.get("PackageFiles")
    if not isinstance(expected_files, dict):
        raise RelinkError("PACKAGE_HASH_MAP", "TYPE", "PACKAGE")
    for name in FINAL_PACKAGE_FILES:
        entry = expected_files.get(name)
        if not isinstance(entry, dict):
            raise RelinkError("PACKAGE_HASH_ENTRY", name, "PACKAGE")
        path = package / name
        if int(entry.get("Bytes", -1)) != path.stat().st_size:
            raise RelinkError("PACKAGE_BYTES", name, "PACKAGE")
        if entry.get("Sha256") != sha256_file(path):
            raise RelinkError("PACKAGE_HASH", name, "PACKAGE")
    bindings = resolve_geometry(package / "u001_signal_catalog.csv")
    header = read_csv_header(package / "u001_frame_values.csv")
    if any(canonical + "__value" not in header for canonical in bindings.values()):
        raise RelinkError("GEOMETRY_FRAME_GROUP", "MISSING", "PACKAGE")
    frame_rows = count_csv_rows(package / "u001_frame_values.csv")
    if frame_rows < minimum_frame_rows:
        raise RelinkError("NOT_FULL_FRAME_DOMAIN", frame_rows, "PACKAGE")
    if int(verification.get("FrameRows", -1)) != frame_rows:
        raise RelinkError("FRAME_ROW_MISMATCH", frame_rows, "PACKAGE")
    if int(verification.get("CatalogSignals", -1)) != 22:
        raise RelinkError("CATALOG_SIGNAL_COUNT", verification.get("CatalogSignals"), "PACKAGE")
    if int(verification.get("FrameSignals", -1)) != 21:
        raise RelinkError("FRAME_SIGNAL_COUNT", verification.get("FrameSignals"), "PACKAGE")
    if int(verification.get("GeometryBindingCount", -1)) != 4:
        raise RelinkError("GEOMETRY_COUNT", verification.get("GeometryBindingCount"), "PACKAGE")
    return package, verification, bindings


def require_workspace(workspace_root: Path) -> Dict[str, Path]:
    """Validate the governed U002 core required for this evidence Run."""

    if not workspace_root.is_dir() or workspace_root.name not in {"U-UDP002", "UDP002"}:
        raise RelinkError("WORKSPACE_ROOT", workspace_root, "WORKSPACE")
    result: Dict[str, Path] = {}
    for role, relative in CORE_FILES.items():
        path = (workspace_root / relative).resolve()
        if not path.is_file() or path.is_symlink():
            raise RelinkError("CORE_FILE", relative, "WORKSPACE")
        result[role] = path
    return result


def discover_source_config(workspace_root: Path) -> Tuple[Path, Dict[str, object]]:
    """Select the accepted B5 configuration by exact identity, never by mtime."""

    candidates = (
        workspace_root / "02_CONFIG/run_configs" / (SOURCE_RUN_ID + ".json"),
        workspace_root / "07_OUTPUT" / SOURCE_RUN_ID / "00_CONFIG/run_configuration.json",
    )
    existing = [path.resolve() for path in candidates if path.is_file()]
    if not existing:
        raise RelinkError("SOURCE_CONFIG_NOT_FOUND", SOURCE_RUN_ID, "CONFIG")
    values = [(path, load_json(path, "SOURCE_CONFIG")) for path in existing]
    canonical = canonical_bytes(values[0][1])
    if any(canonical_bytes(value) != canonical for _, value in values[1:]):
        raise RelinkError("SOURCE_CONFIG_IDENTITY", len(values), "CONFIG")
    config = values[0][1]
    if config.get("RunId") != SOURCE_RUN_ID:
        raise RelinkError("SOURCE_RUN_ID", config.get("RunId"), "CONFIG")
    if config.get("ContractVersion") != CONTRACT_VERSION:
        raise RelinkError("SOURCE_CONTRACT", config.get("ContractVersion"), "CONFIG")
    return values[0][0], config


def select_module(
    plan: object,
    module_id: str,
) -> Dict[str, object]:
    """Select one exact Module definition from the accepted source plan."""

    if not isinstance(plan, list):
        raise RelinkError("SOURCE_MODULE_PLAN", "TYPE", "CONFIG")
    matches = [item for item in plan if isinstance(item, dict) and item.get("ModuleId") == module_id]
    if len(matches) != 1:
        raise RelinkError("SOURCE_MODULE_COUNT", f"{module_id}:{len(matches)}", "CONFIG")
    return copy.deepcopy(matches[0])


def build_relink_config(
    source: Mapping[str, object],
    package: Path,
    run_id: str,
    frame_start: int,
    frame_end: int,
) -> Dict[str, object]:
    """Build a two-Module controlled evidence configuration."""

    if frame_start >= frame_end:
        raise RelinkError("FRAME_RANGE", f"{frame_start}:{frame_end}", "CONFIG")
    source_plan = source.get("ModulePlan")
    s0 = select_module(source_plan, "CameraRadar.S0.U001ControlledProfileIntake")
    camera = select_module(source_plan, "CameraRadar.S1.CameraTargetContinuityArbitration")
    if s0.get("ModuleVersion") != "0.1.1":
        raise RelinkError("S0_VERSION", s0.get("ModuleVersion"), "CONFIG")
    if camera.get("ModuleVersion") != "0.2.0":
        raise RelinkError("CAMERA_S1_VERSION", camera.get("ModuleVersion"), "CONFIG")
    s0["Sequence"] = 1
    camera["Sequence"] = 2
    value = {
        "ContractVersion": CONTRACT_VERSION,
        "RunId": run_id,
        "RunPurpose": "EVALUATION",
        "RunScope": "FRAME_RANGE",
        "FrameStart": frame_start,
        "FrameEnd": frame_end,
        "InputPackage": str(package.resolve()),
        "SignalProfile": source.get("SignalProfile"),
        "ModulePlan": [s0, camera],
        "PrimaryOutcome": {
            "Name": "CanonicalCameraDistance",
            "Formula": "S1ContinuityHysteresis(Camera1Distance,Camera3Distance)",
            "Sign": "NOT_APPLICABLE",
            "Unit": "m",
            "Purpose": "EXPLORATORY",
        },
        "CandidatePolicy": "NotAppliedInU22B7Q@0.1",
        "OutputSelection": [
            "U001_INTAKE_ACCEPTED",
            "CAMERA_TARGET_RECONSTRUCTION",
        ],
    }
    if not isinstance(value["SignalProfile"], str) or not value["SignalProfile"]:
        raise RelinkError("SIGNAL_PROFILE", value["SignalProfile"], "CONFIG")
    return value


def persist_exact(path: Path, value: Mapping[str, object]) -> str:
    """Create a versioned control or accept only a byte-identical replay."""

    payload = canonical_bytes(value)
    if path.exists():
        if not path.is_file() or path.read_bytes() != payload:
            raise RelinkError("CONFIG_CONFLICT", path.name, "CONFIG")
        return "REPLAY"
    write_json_atomic(path, value)
    return "CREATED"


def run_child(
    label: str,
    command: Sequence[str],
    timeout: int,
    diagnostics: List[Dict[str, object]],
) -> subprocess.CompletedProcess:
    """Run one child and retain bounded terminal evidence."""

    try:
        result = subprocess.run(
            list(command),
            text=True,
            encoding="utf-8",
            errors="replace",
            capture_output=True,
            timeout=timeout,
            check=False,
            shell=False,
        )
    except subprocess.TimeoutExpired as error:
        diagnostics.append({"Step": label, "TimeoutSeconds": timeout})
        raise RelinkError("CHILD_TIMEOUT", label, label) from error
    diagnostics.append(
        {
            "Step": label,
            "ReturnCode": result.returncode,
            "StdoutTail": result.stdout[-3000:],
            "StderrTail": result.stderr[-3000:],
        }
    )
    if result.returncode != 0 or "C-0" not in result.stdout:
        raise RelinkError("CHILD_EXIT", f"{label}:{result.returncode}", label)
    return result


def execute_or_replay_run(
    workspace_root: Path,
    core: Mapping[str, Path],
    config_path: Path,
    run_id: str,
    timeout: int,
    diagnostics: List[Dict[str, object]],
) -> Tuple[Path, int]:
    """Prepare, execute and audit the new two-Module Run."""

    run_root = workspace_root / "07_OUTPUT" / run_id
    if run_root.exists():
        status = load_json(run_root / "run_status.json", "RUN_STATUS")
        recorded = run_root / "00_CONFIG/run_configuration.json"
        if status.get("Status") != "SUCCEEDED":
            raise RelinkError("EXISTING_RUN_STATUS", status.get("Status"), "RUN")
        if not recorded.is_file() or recorded.read_bytes() != config_path.read_bytes():
            raise RelinkError("EXISTING_RUN_CONFIG", run_id, "RUN")
        replay = 1
    else:
        run_child(
            "PREFLIGHT",
            [
                sys.executable,
                str(core["PREFLIGHT"]),
                "--prepare-run",
                str(config_path),
                "--workspace-root",
                str(workspace_root),
            ],
            300,
            diagnostics,
        )
        run_child(
            "EXECUTOR",
            [
                sys.executable,
                str(core["EXECUTOR"]),
                "--execute-run",
                str(run_root),
                "--workspace-root",
                str(workspace_root),
                "--module-timeout-seconds",
                str(timeout),
            ],
            timeout * 2 + 300,
            diagnostics,
        )
        replay = 0
    run_child(
        "ARTIFACT_AUDIT",
        [
            sys.executable,
            str(core["STORE"]),
            "--audit-run",
            "--workspace-root",
            str(workspace_root),
            "--run-root",
            str(run_root),
        ],
        300,
        diagnostics,
    )
    return run_root, replay


def validate_linked_run(
    run_root: Path,
    package: Path,
    frame_start: int,
    frame_end: int,
) -> Dict[str, object]:
    """Verify two accepted Artifacts and the frozen S1 package linkage."""

    index = load_json(run_root / "run_artifact_index_v0_1.json", "ARTIFACT_INDEX")
    artifacts = index.get("Artifacts")
    if not isinstance(artifacts, list) or len(artifacts) != 2:
        raise RelinkError("ARTIFACT_COUNT", len(artifacts or []), "VERIFY")
    types = sorted(str(item.get("ArtifactType")) for item in artifacts if isinstance(item, dict))
    if types != ["CAMERA_TARGET_RECONSTRUCTION", "U001_INTAKE_ACCEPTED"]:
        raise RelinkError("ARTIFACT_TYPES", ".".join(types), "VERIFY")
    requests = []
    for path in run_root.rglob("module_request.json"):
        try:
            value = load_json(path, "MODULE_REQUEST")
        except RelinkError:
            continue
        if value.get("ModuleId") == "CameraRadar.S1.CameraTargetContinuityArbitration":
            requests.append(value)
    if len(requests) != 1:
        raise RelinkError("S1_REQUEST_COUNT", len(requests), "VERIFY")
    linked = Path(str(requests[0].get("InputPackage", ""))).resolve()
    if linked != package.resolve():
        raise RelinkError("S1_INPUT_LINK", linked, "VERIFY")
    reconstructions = sorted(run_root.rglob("camera_target_reconstruction.csv"))
    if len(reconstructions) != 1:
        raise RelinkError("RECONSTRUCTION_COUNT", len(reconstructions), "VERIFY")
    rows = 0
    frames: List[int] = []
    with reconstructions[0].open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {
            "FrameCount",
            "CanonicalCameraDistance",
            "CanonicalCameraValid",
            "SelectedCameraRole",
            "SelectionStatus",
            "SelectionReason",
        }
        if not required.issubset(reader.fieldnames or []):
            raise RelinkError("RECONSTRUCTION_SCHEMA", "MISSING", "VERIFY")
        for row in reader:
            frame = int(float(row["FrameCount"]))
            frames.append(frame)
            rows += 1
    expected = frame_end - frame_start + 1
    if rows != expected or frames != list(range(frame_start, frame_end + 1)):
        raise RelinkError("RECONSTRUCTION_DOMAIN", f"{rows}:{expected}", "VERIFY")
    return {
        "ArtifactCount": len(artifacts),
        "ReconstructionRows": rows,
        "ReconstructionPath": str(reconstructions[0].resolve()),
        "LinkedInputPackage": str(linked),
    }


def run_or_replay_b7a(
    b7a_program: Path,
    run_root: Path,
    output_dir: Path,
    frame_start: int,
    frame_end: int,
    diagnostics: List[Dict[str, object]],
) -> Tuple[str, Dict[str, int], int]:
    """Run B7A once or verify an existing immutable profiling result."""

    capsule_path = output_dir / "b7a_return_capsule.txt"
    quality_path = output_dir / "b7a_camera_lateral_evidence_quality.json"
    if output_dir.exists():
        if not capsule_path.is_file() or not quality_path.is_file():
            raise RelinkError("PROFILE_OUTPUT_CONFLICT", output_dir.name, "PROFILE")
        capsule = capsule_path.read_text(encoding="utf-8-sig").strip()
        quality = load_json(quality_path, "B7A_QUALITY")
        if Path(str(quality.get("RunRoot", ""))).resolve() != run_root.resolve():
            raise RelinkError("PROFILE_RUN_LINK", quality.get("RunRoot"), "PROFILE")
        replay = 1
    else:
        result = run_child(
            "B7A_PROFILE",
            [
                sys.executable,
                str(b7a_program),
                "--run-root",
                str(run_root),
                "--output-dir",
                str(output_dir),
                "--frame-start",
                str(frame_start),
                "--frame-end",
                str(frame_end),
            ],
            900,
            diagnostics,
        )
        lines = [line.strip() for line in result.stdout.splitlines() if line.strip().startswith("B7A N-")]
        if len(lines) != 1:
            raise RelinkError("B7A_CAPSULE_COUNT", len(lines), "PROFILE")
        capsule = lines[0]
        quality = load_json(quality_path, "B7A_QUALITY")
        replay = 0
    match = re.fullmatch(r"B7A N-F(\d+)-G(\d+)-I(\d+)-D(\d+)-S(\d+)-X0 C-0", capsule)
    if match is None:
        raise RelinkError("B7A_CAPSULE", capsule, "PROFILE")
    counts = {
        "FrameRows": int(match.group(1)),
        "PairedGeometryRows": int(match.group(2)),
        "OldCanonicalInvalidRows": int(match.group(3)),
        "CenterlineDisagreementRows": int(match.group(4)),
        "ThresholdCount": int(match.group(5)),
    }
    if counts["FrameRows"] != frame_end - frame_start + 1 or counts["ThresholdCount"] != 8:
        raise RelinkError("B7A_COUNTS", counts, "PROFILE")
    return capsule, counts, replay


def execute(
    c_code_root: Path,
    workspace_root: Path,
    b7p2_root: Path,
    output_dir: Path,
    b7a_program: Path,
    run_id: str,
    frame_start: int,
    frame_end: int,
    module_timeout: int,
) -> Tuple[str, str]:
    """Execute the controlled relink and read-only B7A profile."""

    diagnostics: List[Dict[str, object]] = []
    try:
        if c_code_root.resolve() != workspace_root.resolve().parent:
            raise RelinkError("C_CODE_ROOT", c_code_root, "WORKSPACE")
        core = require_workspace(workspace_root)
        if not b7a_program.is_file() or b7a_program.name != B7A_BASENAME:
            raise RelinkError("B7A_PROGRAM", b7a_program.name, "WORKSPACE")
        package, verification, bindings = validate_full_package(b7p2_root)
        source_path, source = discover_source_config(workspace_root)
        config = build_relink_config(source, package, run_id, frame_start, frame_end)
        config_path = workspace_root / "02_CONFIG/run_configs" / (run_id + ".json")
        protected = [source_path, b7a_program, *core.values(), *[package / name for name in FINAL_PACKAGE_FILES]]
        before = snapshot_files(protected)
        config_disposition = persist_exact(config_path, config)
        run_root, run_replay = execute_or_replay_run(
            workspace_root,
            core,
            config_path,
            run_id,
            module_timeout,
            diagnostics,
        )
        linked = validate_linked_run(run_root, package, frame_start, frame_end)
        profile_dir = output_dir / "B7A_PROFILE"
        b7a_capsule, profile_counts, profile_replay = run_or_replay_b7a(
            b7a_program,
            run_root,
            profile_dir,
            frame_start,
            frame_end,
            diagnostics,
        )
        after = snapshot_files(protected)
        if before != after:
            raise RelinkError("PROTECTED_INPUT_CHANGED", "HASH_OR_METADATA", "VERIFY")
        output_dir.mkdir(parents=True, exist_ok=True)
        line1 = (

def self_test(b7a_program: Path) -> None:
    """Exercise package validation, configuration projection and B7A itself."""

    checks: List[str] = []

    def check(name: str, value: bool) -> None:
        if not value:
            raise AssertionError(name)
        checks.append(name)

    check("T01_ID", PROGRAM_ID.startswith("UDP002"))
    check("T02_VERSION", PROGRAM_VERSION == "0.1.0")
    check("T03_RANGE", FRAME_END - FRAME_START + 1 == 401)
    source = {
        "ContractVersion": CONTRACT_VERSION,
        "RunId": SOURCE_RUN_ID,
        "SignalProfile": "TEST@0.1",
        "ModulePlan": [
            {
                "Sequence": 1,
                "Stage": "S0",
                "ModuleId": "CameraRadar.S0.U001ControlledProfileIntake",
                "ModuleVersion": "0.1.1",
                "Configuration": {},
            },
            {
                "Sequence": 2,
                "Stage": "S1",
                "ModuleId": "CameraRadar.S1.CameraTargetContinuityArbitration",
                "ModuleVersion": "0.2.0",
                "Configuration": {"SwitchConfirmFrames": 3},
            },
            {
                "Sequence": 3,
                "Stage": "S1",
                "ModuleId": "CameraRadar.S1.RadarMeasurementValidity",
                "ModuleVersion": "0.1.0",
                "Configuration": {},
            },
        ],
    }
    config = build_relink_config(source, Path("X"), RUN_ID, 10, 20)
    check("T04_PLAN_LENGTH", len(config["ModulePlan"]) == 2)
    check("T05_PLAN_ORDER", [row["Sequence"] for row in config["ModulePlan"]] == [1, 2])
    check("T06_PLAN_STAGE", [row["Stage"] for row in config["ModulePlan"]] == ["S0", "S1"])
    check("T07_CAMERA_VERSION", config["ModulePlan"][1]["ModuleVersion"] == "0.2.0")
    check("T08_CONFIG_COPY", len(source["ModulePlan"]) == 3)
    check("T09_SCOPE", config["RunScope"] == "FRAME_RANGE")
    check("T10_OUTPUTS", len(config["OutputSelection"]) == 2)
    check("T11_NO_POLICY_CHANGE", "Geometry" not in config["PrimaryOutcome"]["Formula"])
    check("T12_FORMULA", config["PrimaryOutcome"]["Formula"].startswith("S1Continuity"))
    check("T13_PACKAGE_ABSOLUTE", Path(config["InputPackage"]).is_absolute())
    check("T14_CANDIDATE_NONE", str(config["CandidatePolicy"]).startswith("NotApplied"))
    check("T15_JSON", len(canonical_bytes(config)) > 100)
    check("T16_HASH", len(hashlib.sha256(b"x").hexdigest()) == 64)
    check("T17_TOKEN", machine_token("a b/c") == "A_B_C")
    check("T18_GEOMETRY", len(GEOMETRY_IDENTITIES) == 4)
    check("T19_PACKAGE_FILES", len(FINAL_PACKAGE_FILES) == 6)
    check("T20_CORE", len(CORE_FILES) == 3)
    check("T21_B7A_FILE", b7a_program.is_file())
    result = subprocess.run(
        [sys.executable, str(b7a_program), "--self-test"],
        text=True,
        capture_output=True,
        timeout=60,
        check=False,
    )
    check("T22_B7A_EXIT", result.returncode == 0)
    check("T23_B7A_CAPSULE", "UDP002-U22B7A-SELFTEST-PASS-20 C-0" in result.stdout)
    with tempfile.TemporaryDirectory(prefix="u22b7q_") as temporary:
        path = Path(temporary) / "config.json"

        check("T25_REPLAY", persist_exact(path, config) == "REPLAY")

    check("T29_DIAGNOSTIC", "u22b7q_failure_diagnostic.json" in Path(__file__).read_text(encoding="utf-8"))

    print("UDP002-U22B7Q-SELFTEST-PASS-30 C-0")


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test or the company relink/profile workflow."""

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--execute", action="store_true")
    parser.add_argument("--c-code-root", default=".")
    parser.add_argument("--workspace-root", default="U-UDP002")
    parser.add_argument(
        "--b7p2-root",
        default="U-UDP001/03_CASES/TC001_20260422/02_RUNS/U22B7P_FULL_GEOMETRY_V02",
    )
    parser.add_argument("--output-dir", default="U-UDP002/04_TEMP/U22B7Q_OUTPUT")
    parser.add_argument("--run-id", default=RUN_ID)
    parser.add_argument("--frame-start", type=int, default=FRAME_START)
    parser.add_argument("--frame-end", type=int, default=FRAME_END)
    parser.add_argument("--module-timeout-seconds", type=int, default=1800)
    args = parser.parse_args(argv)
    b7a_program = Path(__file__).resolve().with_name(B7A_BASENAME)
    try:
        if args.self_test:
            self_test(b7a_program)
            return 0
        if not args.execute:
            raise RelinkError("MODE_REQUIRED", "USE_EXECUTE", "CLI")
        line1, line2 = execute(
            Path(args.c_code_root).resolve(),
            Path(args.workspace_root).resolve(),
            Path(args.b7p2_root).resolve(),
            Path(args.output_dir).resolve(),
            b7a_program,
            args.run_id,
            args.frame_start,
            args.frame_end,
            args.module_timeout_seconds,
        )
        print(line1)
        print(line2)
        return 0 if line1.endswith("C-0") else 7
    except Exception as error:
        if isinstance(error, RelinkError):
            reason, detail, stage = error.reason, error.detail, error.stage
        else:
            reason = machine_token(type(error).__name__, 72)
            detail = machine_token(str(error), 180)
            stage = "UNHANDLED"
        print(f"B7QE K-{reason}-J{stage}-X1 C-7")
        print(f"MEMO1-{detail}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
