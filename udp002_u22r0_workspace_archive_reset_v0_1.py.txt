#!/usr/bin/env python3
"""Archive the legacy U21 workspace and create a clean U22 active surface.

The operation is local, deterministic, fail-closed and reversible. It never
deletes content and never touches 03_CASES or an existing A-Archive payload.
"""

from __future__ import annotations

import argparse
import csv
from dataclasses import dataclass
import hashlib
import json
import os
from pathlib import Path
import re
from typing import Iterable


PROGRAM_ID = "UDP002-U22R0-WORKSPACE-ARCHIVE-RESET"
VERSION = "0.1"
ARCHIVE_NAME = "U21_LEGACY_20260820"
STAGING_NAME = ".U21_LEGACY_20260820.staging"
PRESERVED_DIR = "03_CASES"
ARCHIVE_DIR = "A-Archive"
ACTIVE_DIRS = (
    "00_CONTROL",
    "01_DESIGN",
    "01_PROGRAMS",
    "02_CONFIG",
    "03_CASES",
    "04_TEMP",
    "05_TESTS",
    "06_MANUALS",
    "07_OUTPUT",
)
ARCHIVED_ACTIVE_DIRS = tuple(name for name in ACTIVE_DIRS if name != PRESERVED_DIR)


class ResetError(RuntimeError):
    pass


@dataclass(frozen=True)
class FileRecord:
    source_rel: str
    archive_rel: str
    size_bytes: int
    sha256: str


@dataclass(frozen=True)
class MoveRecord:
    source: Path
    destination: Path


def stable(value: object) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")[:80] or "INTERNAL"


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def reject_symlinks(path: Path) -> None:
    if path.is_symlink():
        raise ResetError("SYMLINK_" + path.name)
    if path.is_dir():
        for child in path.iterdir():
            reject_symlinks(child)


def files_under(path: Path) -> Iterable[Path]:
    if path.is_file():
        yield path
    elif path.is_dir():
        for child in sorted(path.rglob("*"), key=lambda item: str(item).lower()):
            if child.is_file():
                yield child


def ensure_empty_or_missing(path: Path, reason: str) -> None:
    if path.exists():
        if not path.is_dir() or any(path.iterdir()):
            raise ResetError(reason)


def build_plan(root: Path, script_path: Path, staging: Path) -> tuple[list[MoveRecord], list[FileRecord], int]:
    moves: list[MoveRecord] = []
    records: list[FileRecord] = []

    for dirname in ARCHIVED_ACTIVE_DIRS:
        source_dir = root / dirname
        if not source_dir.exists():
            continue
        if not source_dir.is_dir() or source_dir.is_symlink():
            raise ResetError("ACTIVE_DIR_TYPE_" + dirname)
        for entry in sorted(source_dir.iterdir(), key=lambda item: item.name.lower()):
            if entry.resolve() == script_path:
                continue
            reject_symlinks(entry)
            destination = staging / dirname / entry.name
            if destination.exists():
                raise ResetError("ARCHIVE_COLLISION_" + entry.name)
            moves.append(MoveRecord(entry, destination))
            for item in files_under(entry):
                relative_within_entry = item.relative_to(entry.parent)
                archive_path = Path(dirname) / relative_within_entry
                records.append(FileRecord(
                    source_rel=item.relative_to(root).as_posix(),
                    archive_rel=archive_path.as_posix(),
                    size_bytes=item.stat().st_size,
                    sha256=sha256_file(item),
                ))

    protected_root_names = set(ACTIVE_DIRS) | {ARCHIVE_DIR}
    for entry in sorted(root.iterdir(), key=lambda item: item.name.lower()):
        if entry.name in protected_root_names or entry.resolve() == script_path:
            continue
        if entry.is_dir():
            continue
        reject_symlinks(entry)
        destination = staging / "ROOT" / entry.name
        moves.append(MoveRecord(entry, destination))
        records.append(FileRecord(
            source_rel=entry.relative_to(root).as_posix(),
            archive_rel=(Path("ROOT") / entry.name).as_posix(),
            size_bytes=entry.stat().st_size,
            sha256=sha256_file(entry),
        ))

    unknown_dirs = sum(
        1 for entry in root.iterdir()
        if entry.is_dir() and entry.name not in protected_root_names
    )
    return moves, records, unknown_dirs


def write_manifest(staging: Path, records: list[FileRecord]) -> None:
    manifest = staging / "archive_manifest.csv"
    with manifest.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.writer(stream, lineterminator="\n")
        writer.writerow(["SourceRelativePath", "ArchiveRelativePath", "SizeBytes", "Sha256"])
        for record in sorted(records, key=lambda item: item.source_rel.lower()):
            writer.writerow([record.source_rel, record.archive_rel, record.size_bytes, record.sha256])


def verify_records(staging: Path, records: list[FileRecord]) -> None:
    for record in records:
        destination = staging / record.archive_rel
        if not destination.is_file():
            raise ResetError("VERIFY_MISSING_" + destination.name)
        if destination.stat().st_size != record.size_bytes:
            raise ResetError("VERIFY_SIZE_" + destination.name)
        if sha256_file(destination) != record.sha256:
            raise ResetError("VERIFY_HASH_" + destination.name)


def rollback(moved: list[MoveRecord]) -> None:
    for record in reversed(moved):
        if record.destination.exists() and not record.source.exists():
            record.source.parent.mkdir(parents=True, exist_ok=True)
            os.replace(record.destination, record.source)


def execute(root: Path) -> str:
    root = root.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise ResetError("WORKSPACE_ROOT")
    cases = root / PRESERVED_DIR
    if cases.exists() and (not cases.is_dir() or cases.is_symlink()):
        raise ResetError("CASES_TYPE")
    archive_root = root / ARCHIVE_DIR
    archive_root.mkdir(exist_ok=True)
    final_archive = archive_root / ARCHIVE_NAME
    staging = archive_root / STAGING_NAME
    if final_archive.exists():
        raise ResetError("ARCHIVE_ALREADY_EXISTS")
    ensure_empty_or_missing(staging, "ARCHIVE_STAGING_NOT_EMPTY")
    staging.mkdir(exist_ok=True)

    script_path = Path(__file__).resolve()
    moves, records, unknown_dirs = build_plan(root, script_path, staging)
    if not moves:
        raise ResetError("NOTHING_TO_ARCHIVE")
    moved: list[MoveRecord] = []
    try:
        for record in moves:
            record.destination.parent.mkdir(parents=True, exist_ok=True)
            os.replace(record.source, record.destination)
            moved.append(record)
        verify_records(staging, records)
        write_manifest(staging, records)
        summary = {
            "ProgramId": PROGRAM_ID,
            "Version": VERSION,
            "Status": "ARCHIVED",
            "ArchiveName": ARCHIVE_NAME,
            "MovedTopLevelEntries": len(moves),
            "ArchivedFileCount": len(records),
            "ArchivedBytes": sum(item.size_bytes for item in records),
            "PreservedDirectory": PRESERVED_DIR,
            "UnknownRootDirectoriesPreserved": unknown_dirs,
            "NoFilesDeleted": True,
        }
        (staging / "reset_summary.json").write_text(
            json.dumps(summary, ensure_ascii=False, sort_keys=True, indent=2) + "\n",
            encoding="utf-8",
        )
        os.replace(staging, final_archive)
    except Exception:
        rollback(moved)
        raise

    for dirname in ACTIVE_DIRS:
        (root / dirname).mkdir(exist_ok=True)
    capsule = (
        f"U22R0 N-M{len(moves)}-F{len(records)}-"
        f"B{sum(item.size_bytes for item in records)}-K1-U{unknown_dirs}-X0 C-0"
    )
    control = root / "00_CONTROL"
    (control / "u22r0_reset_return_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
    return capsule


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    try:
        print(execute(Path(args.workspace_root)))
        return 0
    except Exception as exc:
        print(f"U22R0 N-M0-F0-B0-K0-U0-X1 C-7 K-{stable(exc)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
