class ValueBox:
    def __init__(self, value: Any) -> None:
        self.value = value

    def get(self) -> Any:
        return self.value

    def set(self, value: Any) -> None:
        self.value = value


def canonical_source_bytes(path: Path) -> bytes:
    text = path.read_text(encoding="utf-8-sig")
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    return text.encode("utf-8")


def canonical_sha256(path: Path) -> str:
    return hashlib.sha256(canonical_source_bytes(path)).hexdigest()


def load_source_module(path: Path):
    module_name = "udp003_fvr_candidate"
    spec = importlib.util.spec_from_file_location(module_name, path)
    if spec is None or spec.loader is None:
        raise RuntimeError("SOURCE_IMPORT_SPEC")
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    try:
        spec.loader.exec_module(module)
    except Exception:
        sys.modules.pop(module_name, None)
        raise
    return module


def ast_inventory(source_text: str) -> tuple[set[str], set[str], set[str]]:
    tree = ast.parse(source_text)
    functions: set[str] = set()
    classes: set[str] = set()
    methods: set[str] = set()
    for node in tree.body:
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            functions.add(node.name)
        elif isinstance(node, ast.ClassDef):
            classes.add(node.name)
            if node.name == "FrameCountVideoReviewer":
                for child in node.body:
                    if isinstance(child, (ast.FunctionDef, ast.AsyncFunctionDef)):
                        methods.add(child.name)
    return functions, classes, methods


def check(name: str, condition: bool, failures: list[str]) -> bool:
    if not condition:
        failures.append(name)
        return False
    return True


def make_dummy_reviewer(module, mode: str = "1:1"):
    reviewer = module.FrameCountVideoReviewer.__new__(module.FrameCountVideoReviewer)
    reviewer.anchor_video_var = ValueBox("10")
    reviewer.anchor_framecount_var = ValueBox("1000")
    reviewer.framecount_rate_var = ValueBox("20")
    reviewer.mapping_mode_var = ValueBox(mode)
    reviewer.video_fps = 10.0
    return reviewer


def run_dynamic_checks(module, temp_root: Path) -> tuple[int, list[str]]:
    failures: list[str] = []
    passed = 0

    def dynamic(name: str, condition: bool) -> None:
        nonlocal passed
        if check(name, condition, failures):
            passed += 1

    dynamic("D01_NORMALIZED_NAME", module.normalized_name("ID_620_FrameCount") == "id620framecount")
    dynamic("D02_COMPACT_TEXT", module.compact_text("A\nB   C", 20) == "A B C")
    dynamic("D03_PARSE_NUMBER", module.parse_number("12.5") == 12.5)
    dynamic("D04_PARSE_NONFINITE", module.parse_number("nan") is None)
    dynamic(
        "D05_FIND_COLUMN",
        module.find_column(["EventID", "StartFrameCount"], ("start frame count",))
        == "StartFrameCount",
    )

    fixture = temp_root / "events.csv"
    fixture.write_text(
        "EventID,StartFrameCount,EndFrameCount,ReviewMemo\n"
        "E001,100,120,test\n",
        encoding="utf-8-sig",
    )
    fields, rows = module.read_delimited_file(fixture)
    dynamic("D06_READ_CSV", fields[0] == "EventID" and rows[0]["EndFrameCount"] == "120")

    event = module.ReviewEvent("E001", 100, 120, peak_framecount=110)
    dynamic(
        "D07_REVIEW_EVENT",
        "E001" in event.display_text() and "P 110" in event.display_text(),
    )

    one_to_one = make_dummy_reviewer(module, "1:1")
    dynamic("D08_MAPPING_1TO1_FORWARD", one_to_one.video_index_to_framecount(15) == 1005)
    dynamic("D09_MAPPING_1TO1_REVERSE", one_to_one.framecount_to_video_index(1005) == 15)

    rate_based = make_dummy_reviewer(module, "Rate-based")
    dynamic("D10_MAPPING_RATE_FORWARD", rate_based.video_index_to_framecount(15) == 1010)
    dynamic("D11_MAPPING_RATE_REVERSE", rate_based.framecount_to_video_index(1010) == 15)

    columns = one_to_one._signal_display_columns(
        ["CAM1", "FrameCount", "CAM3", "Canonical"], "FrameCount"
    )
    dynamic(
        "D12_REVIEW_SCHEMA",
        columns == ["FrameCount", "CAM1", "CAM3", "Canonical"],
    )

    overlap_fields = [
        "EventID",
        "ReviewStartFrame",
        "PeakFrameCount",
        "ReviewEndFrame",
    ]
    overlap_rows = [
        {
            "EventID": "C001",
            "ReviewStartFrame": "41480",
            "PeakFrameCount": "41620",
            "ReviewEndFrame": "41750",
        },
        {
            "EventID": "C002",
            "ReviewStartFrame": "41480",
            "PeakFrameCount": "41662",
            "ReviewEndFrame": "41750",
        },
    ]
    overlap, start_column, peak_column, end_column = module.parse_review_events(
        overlap_fields, overlap_rows
    )
    dynamic("D13_OVERLAP_ROWS_RETAINED", len(overlap) == 2)
    dynamic("D14_START_COLUMN", start_column == "ReviewStartFrame")
    dynamic("D15_PEAK_COLUMN", peak_column == "PeakFrameCount")
    dynamic("D16_END_COLUMN", end_column == "ReviewEndFrame")
    dynamic("D17_SHARED_WINDOW", overlap[0].start_framecount == overlap[1].start_framecount)
    dynamic("D18_DISTINCT_PEAK_1", overlap[0].peak_framecount == 41620)
    dynamic("D19_DISTINCT_PEAK_2", overlap[1].peak_framecount == 41662)

    legacy, _, legacy_peak, _ = module.parse_review_events(
        ["EventID", "StartFrameCount", "EndFrameCount"],
        [{"EventID": "L001", "StartFrameCount": "1", "EndFrameCount": "2"}],
    )
    dynamic("D20_LEGACY_COLUMN", legacy_peak is None)
    dynamic("D21_LEGACY_EVENT", legacy[0].peak_framecount is None)

    try:
        module.parse_review_events(
            ["EventID", "StartFrameCount", "PeakFrameCount", "EndFrameCount"],
            [
                {
                    "EventID": "X001",
                    "StartFrameCount": "1",
                    "PeakFrameCount": "3",
                    "EndFrameCount": "2",
                }
            ],
        )
        invalid_peak_rejected = False
    except RuntimeError:
        invalid_peak_rejected = True
    dynamic("D22_INVALID_PEAK_REJECTED", invalid_peak_rejected)
    dynamic(
        "D23_VIDEO_FOLLOW_TARGET",
        module.highlight_target_framecount(41449, 0, None) == 41449,
    )
    dynamic(
        "D24_EXACT_PEAK_PIN_TARGET",
        module.highlight_target_framecount(41449, 0, 41450) == 41450,
    )
    dynamic(
        "D25_EXACT_PEAK_ROW",
        module.nearest_sorted_index([41449, 41450, 41451], 41450) == 1,
    )
    dynamic(
        "D26_NEAREST_VIDEO_ONLY_ROW",
        module.nearest_sorted_index([41449, 41451], 41450) == 0,
    )
    return passed, failures


def write_results(out_dir: Path, result: dict[str, Any], capsule: str, memos: list[str]) -> None:
    out_dir.mkdir(parents=True, exist_ok=True)
    (out_dir / "UDP003P5_RESULT.json").write_text(
        json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
    )
    lines = [capsule] + [f"MEMO{i + 1}-{memo}" for i, memo in enumerate(memos[:3])]
    (out_dir / "UDP003P5_RETURN_CAPSULE.txt").write_text(
        "\n".join(lines) + "\n", encoding="utf-8"
    )


def run_acceptance(source: Path, out_dir: Path) -> int:
    failures: list[str] = []
    identity_ok = source.is_file() and source.name.lower() == EXPECTED_SOURCE_NAME.lower()
    if not source.is_file():
        failures.append("SOURCE_MISSING")
        capsule = "U003P5 N-I0-S0-A0-O0-D0-X1 C-7"
        write_results(out_dir, {"failures": failures}, capsule, failures)
        print(capsule)
        print("MEMO1-SOURCE_MISSING")
        return 7

    source_text = canonical_source_bytes(source).decode("utf-8")
    observed_sha = canonical_sha256(source)
    identity_ok = identity_ok and observed_sha == EXPECTED_CANONICAL_SHA256
    if not identity_ok:
        failures.append("SOURCE_IDENTITY")

    syntax_ok = True
    try:
        compile(source_text, str(source), "exec")
    except Exception as exc:
        syntax_ok = False
        failures.append(f"SYNTAX_{type(exc).__name__}")

    functions: set[str] = set()
    classes: set[str] = set()
    methods: set[str] = set()
    ast_ok = False
    single_authority_ok = False
    if syntax_ok:
        try:
            functions, classes, methods = ast_inventory(source_text)
            ast_ok = (
                REQUIRED_FUNCTIONS <= functions
                and {"ReviewEvent", "FrameCountVideoReviewer"} <= classes
                and REQUIRED_METHODS <= methods
                and all(marker in source_text for marker in REQUIRED_TEXT_MARKERS)
            )
            if not ast_ok:
                failures.append("REQUIRED_INTERFACE")
            single_authority_ok = (
                source_text.count('text="Jump / Pin Peak"') == 1
                and source_text.count('text="Release Pin"') == 1
                and "video_peak_button" not in source_text
                and "side_peak_button" not in source_text
                and "signal_peak_button" not in source_text
                and "Release Peak Pin" not in source_text
            )
            if not single_authority_ok:
                failures.append("PEAK_CURSOR_AUTHORITY")
        except Exception as exc:
            failures.append(f"AST_{type(exc).__name__}")

    dynamic_passed = 0
    if syntax_ok and ast_ok:
        try:
            module = load_source_module(source)
            if str(getattr(module, "VERSION", "")) != EXPECTED_PRODUCT_VERSION:
                failures.append("PRODUCT_VERSION")
            else:
                with tempfile.TemporaryDirectory(prefix="udp003_a1_") as temp_dir:
                    dynamic_passed, dynamic_failures = run_dynamic_checks(
                        module, Path(temp_dir)
                    )
                failures.extend(dynamic_failures)
        except Exception as exc:
            failures.append(f"IMPORT_{type(exc).__name__}")

    passed = (
        identity_ok
        and syntax_ok
        and ast_ok
        and single_authority_ok
        and dynamic_passed == 26
        and not failures
    )
    exit_code = 0 if passed else 7
    capsule = (
        f"U003P5 N-I{int(identity_ok)}-S{int(syntax_ok)}-A{int(ast_ok)}-"
        f"O{int(single_authority_ok)}-D{dynamic_passed}-"
        f"X{len(failures)} C-{exit_code}"
    )
    result = {
        "gate_version": GATE_VERSION,
        "source": str(source.resolve()),
        "source_name": source.name,
        "canonical_sha256": observed_sha,
        "expected_canonical_sha256": EXPECTED_CANONICAL_SHA256,
        "identity_ok": identity_ok,
        "syntax_ok": syntax_ok,
        "ast_interface_ok": ast_ok,
        "single_peak_cursor_authority_ok": single_authority_ok,
        "required_function_count": len(REQUIRED_FUNCTIONS),
        "required_method_count": len(REQUIRED_METHODS),
        "dynamic_passed": dynamic_passed,
        "failures": failures,
        "capsule": capsule,
    }
    write_results(out_dir, result, capsule, failures)
    print(capsule)
    for index, failure in enumerate(failures[:3], start=1):
        print(f"MEMO{index}-{failure}")
    return exit_code


def run_self_test() -> int:
    checks = 0

    def ok(condition: bool) -> None:
        nonlocal checks
        assert condition
        checks += 1

    with tempfile.TemporaryDirectory(prefix="udp003_gate_selftest_") as temp_dir:
        root = Path(temp_dir)
        lf = root / "lf.py"
        crlf = root / "crlf.py"
        lf.write_bytes(b"x = 1\n")
        crlf.write_bytes(b"x = 1\r\n")
        ok(canonical_source_bytes(lf) == b"x = 1\n")
        ok(canonical_source_bytes(crlf) == b"x = 1\n")
        ok(canonical_sha256(lf) == canonical_sha256(crlf))

        fixture_source = (
            "def alpha():\n    return 1\n"
            "class FrameCountVideoReviewer:\n    def beta(self):\n        return 2\n"
        )
        functions, classes, methods = ast_inventory(fixture_source)
        ok(functions == {"alpha"})
        ok(classes == {"FrameCountVideoReviewer"})
        ok(methods == {"beta"})
        ok("alpha" in functions)
        ok("FrameCountVideoReviewer" in classes)
        ok("beta" in methods)

        failures: list[str] = []
        ok(check("OK", True, failures))
        ok(not check("FAIL", False, failures))
        ok(failures == ["FAIL"])

        box = ValueBox("1")
        ok(box.get() == "1")
        box.set("2")
        ok(box.get() == "2")

        output = root / "out"
        sample_result = {"ok": True}
        write_results(output, sample_result, "TEST C-0", [])
        ok((output / "UDP003P5_RESULT.json").is_file())
        ok((output / "UDP003P5_RETURN_CAPSULE.txt").read_text(encoding="utf-8").strip() == "TEST C-0")

    assert checks == 16, checks
    print("U003-P5-GATE-SELFTEST-PASS-16")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--source", type=Path)
    parser.add_argument("--out", type=Path)
    args = parser.parse_args()

    if args.self_test:
        return run_self_test()
    if args.source is None or args.out is None:
        parser.error("acceptance requires --source and --out")
    return run_acceptance(args.source, args.out)


if __name__ == "__main__":
    raise SystemExit(main())
