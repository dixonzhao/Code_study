from __future__ import annotations

import argparse
import ast
import hashlib
import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile


PROGRAM_NAME = "u001_g10b0_foundation_callable_probe"
PROGRAM_VERSION = "0.1"
TARGET_BASENAME = "u001_foundation_core_v0_1.py"

REQUIRED_FUNCTIONS = (
    "detect_text_format",
    "parse_matlab_can_column",
    "discover_source_message_groups",
    "load_dbc_text",
    "parse_dbc",
    "extract_raw_unsigned",
    "apply_signedness",
    "decode_physical_value",
)


CHILD_CODE = r'''
import importlib.util
import json
import os
from pathlib import Path
import sys

root = str(Path(sys.argv[1]).resolve())
target = str(Path(sys.argv[2]).resolve())
names = json.loads(sys.argv[3])
sys.path.insert(0, root)
module_name = "_u001_g10b0_foundation_import_probe"
spec = importlib.util.spec_from_file_location(module_name, target)
if spec is None or spec.loader is None:
    raise RuntimeError("INVALID_MODULE_SPEC")
module = importlib.util.module_from_spec(spec)
sys.modules[module_name] = module
spec.loader.exec_module(module)
bits = "".join("1" if callable(getattr(module, name, None)) else "0" for name in names)
print("IMPORT_OK:" + bits)
'''


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def ast_public_functions(path: Path) -> list[str]:
    source = path.read_text(encoding="utf-8-sig")
    tree = ast.parse(source, filename=path.name)
    return sorted(
        node.name
        for node in tree.body
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))
        and not node.name.startswith("_")
    )


def run_import_check(program_root: Path, target: Path) -> tuple[bool, str, str]:
    environment = dict(os.environ)
    environment.pop("PYTHONPATH", None)
    with tempfile.TemporaryDirectory(prefix="u001_g10b0d2_") as temp_dir:
        completed = subprocess.run(
            [
                sys.executable,
                "-c",
                CHILD_CODE,
                str(program_root),
                str(target),
                json.dumps(REQUIRED_FUNCTIONS),
            ],
            cwd=temp_dir,
            env=environment,
            capture_output=True,
            text=True,
            timeout=20,
            check=False,
        )
    stdout_lines = [line.strip() for line in completed.stdout.splitlines() if line.strip()]
    token = stdout_lines[-1] if stdout_lines else ""
    if completed.returncode == 0 and token.startswith("IMPORT_OK:"):
        return True, token.removeprefix("IMPORT_OK:"), ""
    category = "IMPORT_EXCEPTION"
    stderr = completed.stderr.lower()
    if "modulenotfounderror" in stderr:
        category = "MODULE_NOT_FOUND"
    elif "syntaxerror" in stderr:
        category = "SYNTAX_ERROR"
    elif "filenotfounderror" in stderr:
        category = "FILE_NOT_FOUND"
    return False, "00000000", category


def build_evidence(program_root: Path) -> dict[str, object]:
    root = program_root.resolve()
    if not root.is_dir():
        raise ValueError("PROGRAM_ROOT_NOT_FOUND")
    matches = list(root.glob(TARGET_BASENAME))
    if len(matches) != 1:
        raise ValueError(f"FOUNDATION_FILE_COUNT_{len(matches)}")
    target = matches[0].resolve()
    public_functions = ast_public_functions(target)
    import_safe, bit_string, import_error = run_import_check(root, target)
    if len(bit_string) != len(REQUIRED_FUNCTIONS) or any(bit not in "01" for bit in bit_string):
        raise ValueError("INVALID_IMPORT_BIT_STRING")
    found = {
        name: bit_string[index] == "1"
        for index, name in enumerate(REQUIRED_FUNCTIONS)
    }
    return {
        "program_name": PROGRAM_NAME,
        "program_version": PROGRAM_VERSION,
        "target_basename": TARGET_BASENAME,
        "target_sha256": sha256_file(target),
        "import_method": "SPEC_FROM_FILE_WITH_EXPLICIT_PROGRAM_ROOT",
        "import_safe": import_safe,
        "import_error_category": import_error,
        "required_function_order": list(REQUIRED_FUNCTIONS),
        "required_function_bits": bit_string,
        "required_function_callable": found,
        "ast_public_function_count": len(public_functions),
        "ast_public_function_names": public_functions,
        "absolute_paths_stored": 0,
    }


def make_capsule(evidence: dict[str, object], code: int) -> str:
    bits = str(evidence.get("required_function_bits", "00000000"))
    import_safe = 1 if evidence.get("import_safe") else 0
    text_format = bits[0:1].count("1")
    source_header = bits[1:3].count("1")
    dbc = bits[3:5].count("1")
    decode = bits[5:8].count("1")
    public_count = int(evidence.get("ast_public_function_count", 0))
    errors = 0 if code == 0 else 1
    return (
        "UDP001G10B0D2 "
        f"N-I{import_safe}-Q{bits}-T{text_format}-C{source_header}-"
        f"D{dbc}-B{decode}-A{public_count}-X{errors} C-{code}"
    )


def write_probe(program_root: Path, out: Path) -> str:
    output = out.resolve()
    if output.exists() and any(output.iterdir()):
        raise ValueError("OUTPUT_FOLDER_NOT_EMPTY")
    output.mkdir(parents=True, exist_ok=True)
    evidence = build_evidence(program_root)
    capsule = make_capsule(evidence, 0)
    evidence_path = output / "u001_g10b0_foundation_callable_probe.json"
    capsule_path = output / "udp001_g10b0d2_return_capsule.txt"
    evidence_path.write_text(
        json.dumps(evidence, ensure_ascii=True, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
        newline="\n",
    )
    capsule_path.write_text(capsule + "\n", encoding="utf-8", newline="\n")
    return capsule


def self_test() -> None:
    with tempfile.TemporaryDirectory(prefix="u001_g10b0d2_selftest_") as temp_dir:
        root = Path(temp_dir)
        full_source = "\n".join(
            [
                "def " + name + "(*args, **kwargs):\n    return None"
                for name in REQUIRED_FUNCTIONS
            ]
        ) + "\n"
        target = root / TARGET_BASENAME
        target.write_text(full_source, encoding="utf-8")
        evidence = build_evidence(root)
        assert evidence["import_safe"] is True
        assert evidence["required_function_bits"] == "11111111"
        assert make_capsule(evidence, 0).startswith("UDP001G10B0D2 N-I1-Q11111111")

        target.write_text("def main():\n    raise RuntimeError('must not run')\n\nif __name__ == '__main__':\n    main()\n", encoding="utf-8")
        evidence_missing = build_evidence(root)
        assert evidence_missing["import_safe"] is True
        assert evidence_missing["required_function_bits"] == "00000000"
        assert evidence_missing["absolute_paths_stored"] == 0

        target.write_text("this is not valid python", encoding="utf-8")
        try:
            build_evidence(root)
        except SyntaxError:
            pass
        else:
            raise AssertionError("syntax error was not detected")

    print("UDP001-G10B0D2-SELFTEST-PASS-6")


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--probe", action="store_true")
    parser.add_argument("--program-root", type=Path)
    parser.add_argument("--out", type=Path)
    return parser.parse_args()


def main() -> int:
    args = parse_args()
    try:
        if args.self_test:
            self_test()
            return 0
        if args.program_root is None or args.out is None:
            raise ValueError("PROBE_REQUIRES_PROGRAM_ROOT_AND_OUT")
        capsule = write_probe(args.program_root, args.out)
        print(capsule)
        return 0
    except Exception as exc:
        print(f"UDP001G10B0D2-ERROR:{type(exc).__name__}:{str(exc)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
