class ModuleError(RuntimeError):
    """Represent one deterministic U001 intake failure."""


def stable(value: object) -> str:
    """Convert one diagnostic value into a compact capsule token."""

    normalized = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper())
    return normalized.strip("_")[:80] or "INTERNAL"


def regular_file(path: Path) -> bool:
    """Accept ordinary package files only."""

    return path.is_file() and not path.is_symlink()


def canonical_bytes(value: object) -> bytes:
    """Serialize one JSON document deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def file_state(path: Path) -> Tuple[int, int]:
    """Capture the size and modified time used for concurrent-mutation checks."""

    stat = path.stat()
    return stat.st_size, stat.st_mtime_ns


def read_header(path: Path) -> List[str]:
    """Read one CSV header and reject empty or duplicated names."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        header = next(csv.reader(stream), None)
    if not header:
        raise ModuleError("EMPTY_CSV_" + path.name)
    if len(header) != len(set(header)):
        raise ModuleError("DUPLICATE_HEADER_" + path.name)
    return header


def require_columns(
    header: Sequence[str],
    required: Sequence[str],
    label: str,
) -> None:
    """Require all logical columns while permitting documented extensions."""

    missing = [name for name in required if name not in header]
    if missing:
        raise ModuleError(label + "_MISSING_" + missing[0])


def find_package_files(input_root: Path) -> Dict[str, Path]:
    """Resolve exactly one occurrence of each U001 public package file."""

    if not input_root.is_dir() or input_root.is_symlink():
        raise ModuleError("INPUT_PACKAGE_DIRECTORY")
    files: Dict[str, Path] = {}
    for name in PACKAGE_NAMES:
        matches = sorted(
            path
            for path in input_root.rglob(name)
            if regular_file(path)
        )
        if len(matches) != 1:
            raise ModuleError("PACKAGE_FILE_COUNT_" + name)
        files[name] = matches[0]
    return files


def parse_integer(value: str, field: str) -> int:
    """Parse a finite integer represented as integer or decimal text."""

    text = (value or "").strip()
    try:
        number = float(text)
    except ValueError as exc:
        raise ModuleError("NON_NUMERIC_" + field) from exc
    if not math.isfinite(number) or not number.is_integer():
        raise ModuleError("NON_INTEGER_" + field)
    return int(number)


def parse_boolean(value: str, field: str) -> bool:
    """Parse the controlled U001 CSV boolean representation."""

    text = (value or "").strip().lower()
    if text in {"1", "true"}:
        return True
    if text in {"0", "false"}:
        return False
    raise ModuleError("INVALID_BOOLEAN_" + field)


def frame_groups(header: Sequence[str]) -> List[str]:
    """Resolve and validate complete seven-column D2 signal groups."""

    if tuple(header[:4]) != D2_LEADING:
        raise ModuleError("D2_LEADING_COLUMNS")
    groups = [name[:-7] for name in header if name.endswith("__value")]
    if not groups or len(groups) != len(set(groups)):
        raise ModuleError("D2_SIGNAL_GROUPS")
    expected = list(D2_LEADING)
    for group in groups:
        expected.extend(group + suffix for suffix in D2_SUFFIXES)
    if list(header) != expected:
        raise ModuleError("D2_GROUP_ORDER")
    return groups


def inspect_frame_values(path: Path) -> Dict[str, object]:
    """Stream D2 and verify the canonical FrameCount domain."""

    groups = frame_groups(read_header(path))
    rows = 0
    frame_min = None
    frame_max = None
    previous = None
    gap_count = 0
    largest_gap = 0
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        for row in csv.DictReader(stream):
            frame = parse_integer(row["FrameCount"], "FRAMECOUNT")
            if previous is not None:
                if frame <= previous:
                    raise ModuleError("D2_FRAME_NOT_STRICT")
                step = frame - previous
                if step > 1:
                    gap_count += 1
                    largest_gap = max(largest_gap, step - 1)
            if frame_min is None:
                frame_min = frame
            frame_max = frame
            previous = frame
            rows += 1
            for group in groups:
                parse_boolean(row[group + "__valid"], group + "__valid")
                parse_boolean(row[group + "__updated"], group + "__updated")
    if rows == 0:
        raise ModuleError("D2_NO_ROWS")
    return {
        "FrameRows": rows,
        "FrameMin": frame_min,
        "FrameMax": frame_max,
        "MissingFrameGapCount": gap_count,
        "LargestMissingFrameGap": largest_gap,
        "SignalGroups": groups,
    }


def inspect_catalog(path: Path) -> Tuple[List[Dict[str, str]], List[str]]:
    """Validate D3 identity and build the compact downstream inventory."""

    header = read_header(path)
    require_columns(header, D3_REQUIRED, "D3")
    inventory: List[Dict[str, str]] = []
    included: List[str] = []
    identities = set()
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        for row in csv.DictReader(stream):
            name = (row.get("CanonicalName") or "").strip()
            if not name or name in identities:
                raise ModuleError("D3_CANONICAL_IDENTITY")
            identities.add(name)
            if parse_boolean(
                row.get("IncludeInFrameView", ""),
                name + "_IncludeInFrameView",
            ):
                included.append(name)
            inventory.append(
                {column: row.get(column, "") for column in INVENTORY_COLUMNS}
            )
    if not inventory:
        raise ModuleError("D3_NO_SIGNALS")
    return inventory, included


def inspect_quality(path: Path, signal_count: int) -> Dict[str, int]:
    """Require the minimum global U001 package quality gates."""

    header = read_header(path)
    require_columns(header, D4_REQUIRED, "D4")
    required = {
        "package_file_count",
        "package_error_count",
        "frame_view_signal_count",
    }
    found: Dict[str, int] = {}
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        for row in csv.DictReader(stream):
            if (row.get("Scope") or "").strip() != "GLOBAL":
                continue
            metric = (row.get("Metric") or "").strip()
            if metric in required:
                found[metric] = parse_integer(row.get("Value", ""), metric)
    if set(found) != required:
        raise ModuleError("D4_REQUIRED_GLOBAL_METRICS")
    if found["package_file_count"] != 6:
        raise ModuleError("D4_PACKAGE_FILE_COUNT")
    if found["package_error_count"] != 0:
        raise ModuleError("D4_PACKAGE_ERROR_COUNT")
    if found["frame_view_signal_count"] != signal_count:
        raise ModuleError("D4_SIGNAL_COUNT")
    return found


def inspect_manifest(path: Path) -> Dict[str, object]:
    """Validate D5 as a portable object and retain only safe identity fields."""

    try:
        manifest = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise ModuleError("D5_JSON") from exc
    if not isinstance(manifest, dict):
        raise ModuleError("D5_TYPE")
    serialized = json.dumps(manifest, ensure_ascii=False)
    forbidden = ("C:\\\\", "/Users/", "\\\\Users\\\\", "/home/")
    if any(marker in serialized for marker in forbidden):
        raise ModuleError("D5_ABSOLUTE_PATH")
    identity_fields = (
        "ContractVersion",
        "DatasetId",
        "RunId",
        "SegmentId",
        "SourceId",
        "ProcessingMode",
        "Status",
    )
    return {
        field: manifest[field]
        for field in identity_fields
        if field in manifest
    }


def inspect_review_view(frame_path: Path, review_path: Path) -> int:
    """Require D6 to remain a row-aligned, FrameCount-led subset of D2."""

    frame_header = read_header(frame_path)
    review_header = read_header(review_path)
    if not review_header or review_header[0] != "FrameCount":
        raise ModuleError("D6_FIRST_COLUMN")
    if any(column not in frame_header for column in review_header):
        raise ModuleError("D6_NOT_D2_SUBSET")
    rows = 0
    with frame_path.open(
        "r",
        encoding="utf-8-sig",
        newline="",
    ) as frame_stream, review_path.open(
        "r",
        encoding="utf-8-sig",
        newline="",
    ) as review_stream:
        frame_reader = csv.DictReader(frame_stream)
        review_reader = csv.DictReader(review_stream)
        while True:
            frame_row = next(frame_reader, None)
            review_row = next(review_reader, None)
            if frame_row is None or review_row is None:
                if frame_row is not None or review_row is not None:
                    raise ModuleError("D6_ROW_COUNT")
                break
            for column in review_header:
                if (frame_row.get(column) or "") != (review_row.get(column) or ""):
                    raise ModuleError("D6_VALUE_MISMATCH")
            rows += 1
    return rows


def package_file_records(files: Mapping[str, Path]) -> Tuple[List[Dict[str, object]], Dict[str, Tuple[int, int]]]:
    """Hash every public package file and capture concurrent-mutation state."""

    records: List[Dict[str, object]] = []
    states: Dict[str, Tuple[int, int]] = {}
    for name in PACKAGE_NAMES:
        path = files[name]
        states[name] = file_state(path)
        records.append(
            {
                "Basename": name,
                "Bytes": path.stat().st_size,
                "Sha256": sha256_file(path),
            }
        )
    return records, states


def package_fingerprint(records: Sequence[Mapping[str, object]]) -> str:
    """Bind the exact six-file identities into one compact fingerprint."""

    digest = hashlib.sha256()
    for record in sorted(records, key=lambda item: str(item["Basename"])):
        digest.update(str(record["Basename"]).encode("utf-8"))
        digest.update(str(record["Sha256"]).encode("ascii"))
        digest.update(str(record["Bytes"]).encode("ascii"))
    return digest.hexdigest()


def ensure_states_unchanged(
    files: Mapping[str, Path],
    states: Mapping[str, Tuple[int, int]],
) -> None:
    """Reject a package changed concurrently during the inspection."""

    for name, expected in states.items():
        if file_state(files[name]) != expected:
            raise ModuleError("INPUT_CHANGED_" + name)


def inspect_package(input_root: Path) -> Tuple[Dict[str, object], List[Dict[str, str]]]:
    """Validate all six public U001 files and return governed intake evidence."""

    files = find_package_files(input_root)
    records, states = package_file_records(files)
    require_columns(read_header(files[PACKAGE_NAMES[0]]), D1_REQUIRED, "D1")
    frame = inspect_frame_values(files[PACKAGE_NAMES[1]])
    inventory, included = inspect_catalog(files[PACKAGE_NAMES[2]])
    if included != frame["SignalGroups"]:
        raise ModuleError("D2_D3_SIGNAL_ORDER")
    quality = inspect_quality(files[PACKAGE_NAMES[3]], len(included))
    identity = inspect_manifest(files[PACKAGE_NAMES[4]])
    review_rows = inspect_review_view(
        files[PACKAGE_NAMES[1]],
        files[PACKAGE_NAMES[5]],
    )
    if review_rows != frame["FrameRows"]:
        raise ModuleError("D6_FRAME_ROWS")
    ensure_states_unchanged(files, states)
    summary: Dict[str, object] = {
        "Interface": "U001_CONTROLLED_PROFILE_SIX_FILE_PACKAGE",
        "InterfaceVersion": "0.1",
        "U001Contract": "U001_UNIFIED_DATA_CONTRACT_V0_1",
        "InputPackageBasename": input_root.name,
        "PackageFingerprintSha256": package_fingerprint(records),
        "Files": records,
        "FrameDomain": frame,
        "SignalCount": len(inventory),
        "QualityGate": quality,
        "U001ManifestIdentity": identity,
        "Status": "INTAKE_ACCEPTED",
        "Warnings": [],
        "Prohibitions": [
            "Radar is not asserted as ground truth",
            "No target selection was performed",
            "No missing value was converted to zero",
            "No discrepancy, baseline, evidence, or candidate was calculated",
        ],
    }
    return summary, inventory


def write_inventory(path: Path, rows: Iterable[Mapping[str, str]]) -> int:
    """Write the compact D3-derived inventory used by downstream Modules."""

    count = 0
    with path.open("x", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=INVENTORY_COLUMNS)
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in INVENTORY_COLUMNS})
            count += 1
    return count

            "PackageFingerprintSha256": summary["PackageFingerprintSha256"],
            "FrameRows": summary["FrameDomain"]["FrameRows"],

        raise ModuleError("REQUEST_PROTOCOL")
    if request.get("ProtocolVersion") != PROTOCOL_VERSION:
        raise ModuleError("REQUEST_PROTOCOL_VERSION")
    if request.get("ContractVersion") != CONTRACT_VERSION:
        raise ModuleError("REQUEST_CONTRACT")
    if request.get("ModuleId") != MODULE_ID:
        raise ModuleError("REQUEST_MODULE_ID")
    if request.get("ModuleVersion") != MODULE_VERSION:
        raise ModuleError("REQUEST_MODULE_VERSION")
    if request.get("Stage") != "S0":

    if request.get("ExpectedOutputTypes") != [ARTIFACT_TYPE]:
        raise ModuleError("REQUEST_OUTPUT_TYPE")
    if request.get("InputArtifacts") != []:
        raise ModuleError("REQUEST_INPUT_ARTIFACTS")

def main(argv: Sequence[str] = None) -> int:
    """Inspect one U001 package and return one governed candidate Artifact."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--request-json", required=True)
    parser.add_argument("--result-json", required=True)
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        request = json.loads(
            Path(args.request_json).read_text(encoding="utf-8-sig")
        )
        if not isinstance(request, dict):
            raise ModuleError("REQUEST_TYPE")
        validate_request(request)
        input_root = Path(str(request["InputPackage"]))
        summary, inventory = inspect_package(input_root)

def canonical_bytes(value: object) -> bytes:
    """Serialize one installed JSON document deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_bytes(value: bytes) -> str:
    """Hash one installed file payload."""

    return hashlib.sha256(value).hexdigest()


def regular_file(path: Path) -> bool:
    """Accept ordinary installer and foundation files only."""

    return path.is_file() and not path.is_symlink()


def require_workspace(path: Path) -> Path:

    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise InstallerError("WORKSPACE_ROOT")
    return root

def install_module(workspace_root: Path) -> str:
    """Install the deterministic Module package without touching frozen Core."""

    audit_foundation(workspace_root)
    destination = workspace_root / MODULE_RELATIVE
    files = module_files()
    if destination.exists():
        if not destination.is_dir() or destination.is_symlink():
            raise InstallerError("MODULE_DESTINATION_TYPE")
        for name, content in files.items():
            path = destination / name
            if not regular_file(path) or path.read_bytes() != content:
                raise InstallerError("MODULE_INSTALL_CONFLICT_" + name)
        if set(path.name for path in destination.iterdir()) != set(files):
            raise InstallerError("MODULE_INSTALL_EXTRA_FILE")
        return "U22B1AI N-F4-H1-R1-X0 C-0"
    temporary = destination.with_name(destination.name + ".installing")
    if temporary.exists():
        raise InstallerError("MODULE_INSTALL_TEMP_EXISTS")
    temporary.mkdir(parents=True)
    try:
        for name, content in files.items():
            (temporary / name).write_bytes(content)
        os.replace(temporary, destination)
    except Exception:
        if temporary.exists():
            shutil.rmtree(temporary, ignore_errors=True)
        raise
    for name, content in files.items():
        if (destination / name).read_bytes() != content:
            raise InstallerError("MODULE_INSTALL_IDENTITY_" + name)
    return "U22B1AI N-F4-H1-R0-X0 C-0"
