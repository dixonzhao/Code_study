@echo off
setlocal EnableExtensions

rem Run from the C-CODE root in Windows CMD.
rem Promote only the exact Engine accepted by U21A_GATE_20260818.

set "SRC=.\U-UDP002\04_TEMP\u002_engine_skeleton_v0_1.py"
set "DST=.\U-UDP002\01_PROGRAMS\u002_engine_skeleton_v0_1.py"
set "EVIDENCE=.\U-UDP002\05_TESTS\02_ACCEPTANCE\U21A_GATE_20260818\02_EVIDENCE\U21A_ACCEPTANCE_EVIDENCE.json"
set "SCHEMA=.\U-UDP002\04_TEMP\u002_core_contract_schema_bundle_v0_1.json.txt"
set "VALIDATOR=.\U-UDP002\04_TEMP\u002_core_contract_validator_v0_1.py.txt"

if not exist "%SRC%" goto :missing
if not exist "%EVIDENCE%" goto :missing
if not exist "%SCHEMA%" goto :missing
if not exist "%VALIDATOR%" goto :missing
if not exist ".\U-UDP002\01_PROGRAMS" goto :missing

python -c "import hashlib,json,pathlib,sys; p=pathlib.Path(r'%SRC%'); e=json.loads(pathlib.Path(r'%EVIDENCE%').read_text(encoding='utf-8')); actual=hashlib.sha256(p.read_bytes()).hexdigest(); expected=e['Inputs']['EngineSha256']; sys.exit(0 if actual==expected and e['GateStatus']=='PASS' else 7)"
if errorlevel 1 goto :identity

attrib -R "%DST%" >nul 2>nul
copy /Y "%SRC%" "%DST%" >nul
if errorlevel 1 goto :copyfail

fc /B "%SRC%" "%DST%" >nul
if errorlevel 1 goto :copyfail

attrib +R "%DST%"

python "%DST%" --self-test --schema-bundle "%SCHEMA%" --contract-validator "%VALIDATOR%" > ".\U-UDP002\04_TEMP\U21A_PROMOTION_SELFTEST.stdout.txt" 2> ".\U-UDP002\04_TEMP\U21A_PROMOTION_SELFTEST.stderr.txt"
if errorlevel 1 goto :selftest

findstr /C:"UDP002-U21A-SELFTEST-PASS-24 C-0" ".\U-UDP002\04_TEMP\U21A_PROMOTION_SELFTEST.stdout.txt" >nul
if errorlevel 1 goto :selftest

echo U21AP1 N-H1-C1-R1-X0 C-0
exit /b 0

:missing
echo U21AP1-ERROR-MISSING-INPUT C-7
exit /b 7

:identity
echo U21AP1-ERROR-ENGINE-IDENTITY C-7
exit /b 7

:copyfail
echo U21AP1-ERROR-PROMOTION-COPY C-7
exit /b 7

:selftest
echo U21AP1-ERROR-PROMOTED-SELFTEST C-7
exit /b 7
