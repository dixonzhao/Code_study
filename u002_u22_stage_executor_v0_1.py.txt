class ExecutorError(RuntimeError):
    """Represent one deterministic fail-closed Executor failure."""

    def __init__(self, reason: str, detail: str = ""):
        self.reason = stable(reason)
        self.detail = stable(detail) if detail else "NA"
        super().__init__(self.reason)


class ModuleExecutionError(ExecutorError):
    """Represent one failure attributable to a selected Module execution."""


def stable(value: object) -> str:
    """Convert one detail into a compact return-capsule token."""

    normalized = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper())
    return normalized.strip("_")[:80] or "INTERNAL"


def canonical_bytes(value: object) -> bytes:
    """Serialize control records deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_bytes(value: bytes) -> str:
    """Hash one in-memory control record."""

    return hashlib.sha256(value).hexdigest()


def sha256_file(path: Path) -> str:
    """Hash one file without loading it completely into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    """Accept ordinary files only; symbolic links are not trusted runtime input."""

    return path.is_file() and not path.is_symlink()


def inside(child: Path, parent: Path) -> bool:
    """Return True only when a resolved child remains inside its parent."""

    try:
        child.resolve().relative_to(parent.resolve())
        return True
    except (OSError, ValueError):
        return False


def safe_identifier(value: object, field: str) -> str:
    """Reject path-like or ambiguous runtime identifiers."""

    text = str(value)
    if IDENTIFIER_PATTERN.fullmatch(text) is None:
        raise ExecutorError(field + "_FORMAT", text)
    return text


def require_workspace(path: Path) -> Path:
    """Validate the fixed U002 workspace identity."""

    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise ExecutorError("WORKSPACE_ROOT", root.name)
    return root


def require_run_root(workspace_root: Path, path: Path) -> Path:
    """Restrict execution to one direct child of the governed output root."""

    run_root = path.resolve()
    output_root = (workspace_root / "07_OUTPUT").resolve()
    if not run_root.is_dir() or run_root.is_symlink():
        raise ExecutorError("RUN_ROOT", str(run_root))
    if run_root.parent != output_root:
        raise ExecutorError("RUN_ROOT_LOCATION", str(run_root))
    safe_identifier(run_root.name, "RUN_ID")
    return run_root


def load_json(path: Path, reason: str) -> Dict[str, Any]:
    """Load one regular JSON object and reject non-object documents."""

    if not regular_file(path):
        raise ExecutorError(reason + "_MISSING", str(path))
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise ExecutorError(reason + "_JSON", str(path)) from exc
    if not isinstance(value, dict):
        raise ExecutorError(reason + "_TYPE", str(path))
    return value


def write_json_atomic(path: Path, value: Mapping[str, Any]) -> None:
    """Replace one control record atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    if temporary.exists():
        raise ExecutorError("CONTROL_TEMP_EXISTS", str(temporary))
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def load_core_module(
    workspace_root: Path,
    relative_path: Path,
    module_name: str,
    required_api: Sequence[str],
) -> Any:
    """Load one approved Core helper from its exact installed path."""

    path = workspace_root / relative_path
    if not regular_file(path) or not inside(path, workspace_root):
        raise ExecutorError("CORE_DEPENDENCY_MISSING", str(relative_path))
    spec = importlib.util.spec_from_file_location(module_name, path)
    if spec is None or spec.loader is None:
        raise ExecutorError("CORE_DEPENDENCY_LOAD", str(relative_path))
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    if any(not hasattr(module, name) for name in required_api):
        raise ExecutorError("CORE_DEPENDENCY_API", str(relative_path))
    if getattr(module, "CONTRACT_VERSION", None) != CONTRACT_VERSION:
        raise ExecutorError("CORE_DEPENDENCY_VERSION", str(relative_path))
    return module


def plan_hash(plan: Mapping[str, Any]) -> str:
    """Recompute the immutable plan identity defined by Engine Preflight."""

    payload = dict(plan)
    payload.pop("PlanSha256", None)
    return sha256_bytes(canonical_bytes(payload))


def validate_plan_structure(
    workspace_root: Path,
    run_root: Path,
    plan: Mapping[str, Any],
) -> List[Dict[str, Any]]:
    """Validate plan shape, ordering, hashes and local path identities."""

    if set(plan) != PLAN_KEYS:
        raise ExecutorError("PLAN_SHAPE")
    if plan["EngineContract"] != "U002_U22_RESOLVED_RUN_PLAN":
        raise ExecutorError("PLAN_CONTRACT")
    if plan["ContractVersion"] != CONTRACT_VERSION:
        raise ExecutorError("PLAN_CONTRACT_VERSION")
    if plan["RunId"] != run_root.name:
        raise ExecutorError("PLAN_RUN_ID")
    if plan["PlanSha256"] != plan_hash(plan):
        raise ExecutorError("PLAN_HASH")
    input_package = Path(str(plan["InputPackageResolved"])).resolve()
    if not input_package.exists() or input_package.is_symlink():
        raise ExecutorError("INPUT_PACKAGE_MISSING")
    expected_kind = "DIRECTORY" if input_package.is_dir() else "FILE"
    if plan["InputPackageKind"] != expected_kind:
        raise ExecutorError("INPUT_PACKAGE_KIND")
    module_plan = plan["ModulePlan"]
    if not isinstance(module_plan, list) or not module_plan:
        raise ExecutorError("MODULE_PLAN_EMPTY")
    resolved: List[Dict[str, Any]] = []
    previous_stage_index = -1
    for expected_sequence, raw_item in enumerate(module_plan, start=1):
        if not isinstance(raw_item, dict) or set(raw_item) != MODULE_PLAN_KEYS:
            raise ExecutorError("MODULE_PLAN_SHAPE", str(expected_sequence))
        item = dict(raw_item)
        if item["Sequence"] != expected_sequence:
            raise ExecutorError("MODULE_SEQUENCE", str(expected_sequence))
        if item["Stage"] not in STAGES:
            raise ExecutorError("MODULE_STAGE", str(expected_sequence))
        stage_index = STAGES.index(str(item["Stage"]))
        if stage_index < previous_stage_index:
            raise ExecutorError("MODULE_STAGE_ORDER", str(expected_sequence))
        previous_stage_index = stage_index
        safe_identifier(item["ModuleId"], "MODULE_ID")
        safe_identifier(item["ModuleVersion"], "MODULE_VERSION")
        if item["ConfigurationSha256"] != sha256_bytes(
            canonical_bytes(item["Configuration"])
        ):
            raise ExecutorError("MODULE_CONFIGURATION_HASH", item["ModuleId"])
        module_root = workspace_root / str(item["ModuleRoot"])
        entrypoint = workspace_root / str(item["EntryPoint"])
        manifest = workspace_root / str(item["ManifestPath"])
        evidence = workspace_root / str(item["QualificationEvidence"])
        if not module_root.is_dir() or module_root.is_symlink():
            raise ExecutorError("MODULE_ROOT", item["ModuleId"])
        if not regular_file(entrypoint) or not inside(entrypoint, module_root):
            raise ExecutorError("MODULE_ENTRYPOINT", item["ModuleId"])
        if not regular_file(manifest) or not inside(manifest, module_root):
            raise ExecutorError("MODULE_MANIFEST", item["ModuleId"])
        if sha256_file(manifest) != item["ManifestSha256"]:
            raise ExecutorError("MODULE_MANIFEST_HASH", item["ModuleId"])
        if not regular_file(evidence) or not inside(evidence, workspace_root):
            raise ExecutorError("MODULE_QUALIFICATION", item["ModuleId"])
        resolved.append(item)
    return resolved


def refresh_availability(workspace_root: Path) -> Dict[str, Any]:
    """Revalidate every selected Module hash immediately before execution."""

    qualification = workspace_root / QUALIFICATION_RELATIVE
    if not regular_file(qualification):
        raise ExecutorError("QUALIFICATION_MISSING")
    result = subprocess.run(
        [
            sys.executable,
            str(qualification),
            "--refresh-availability",
            "--workspace-root",
            str(workspace_root),
        ],
        text=True,
        capture_output=True,
        timeout=60,
        check=False,
    )
    if result.returncode != 0 or "C-0" not in result.stdout:
        raise ExecutorError("AVAILABILITY_REFRESH")
    snapshot = load_json(
        workspace_root / AVAILABILITY_RELATIVE,
        "AVAILABILITY",
    )
    if not isinstance(snapshot.get("Entries"), list):
        raise ExecutorError("AVAILABILITY_SHAPE")
    return snapshot


def validate_current_module_eligibility(
    module_plan: Sequence[Mapping[str, Any]],
    availability: Mapping[str, Any],
) -> None:
    """Require every frozen Module to remain AVAILABLE with identical evidence."""

    entries = availability["Entries"]
    for module in module_plan:
        matches = [
            entry
            for entry in entries
            if entry.get("ModuleId") == module["ModuleId"]
            and entry.get("ModuleVersion") == module["ModuleVersion"]
        ]
        if len(matches) != 1:
            raise ExecutorError("MODULE_AVAILABILITY_COUNT", module["ModuleId"])
        entry = matches[0]
        if entry.get("RegistryStatus") != "AVAILABLE":
            raise ExecutorError("MODULE_NOT_AVAILABLE", module["ModuleId"])
        if entry.get("ManifestSha256") != module["ManifestSha256"]:
            raise ExecutorError("MODULE_AVAILABILITY_HASH", module["ModuleId"])
        if entry.get("QualificationEvidence") != module["QualificationEvidence"]:
            raise ExecutorError("MODULE_AVAILABILITY_EVIDENCE", module["ModuleId"])


def load_status(run_root: Path, plan: Mapping[str, Any]) -> Dict[str, Any]:
    """Load one Run lifecycle record and bind it to the current plan."""

    status = load_json(run_root / STATUS_RELATIVE, "RUN_STATUS")
    if status.get("RunId") != run_root.name:
        raise ExecutorError("RUN_STATUS_ID")
    if status.get("PlanSha256") != plan["PlanSha256"]:
        raise ExecutorError("RUN_STATUS_PLAN")
    if status.get("Status") not in {"PREPARED", "RUNNING", "FAILED", "SUCCEEDED"}:
        raise ExecutorError("RUN_STATUS_VALUE")
    return status


def artifact_inputs(
    artifact_store: Any,
    run_root: Path,
    module: Mapping[str, Any],
) -> List[Dict[str, Any]]:
    """Resolve declared required and optional Artifact types from the Run index."""

    index = artifact_store.load_index(run_root)
    by_type: Dict[str, List[Dict[str, Any]]] = {}
    for entry in index["Artifacts"]:
        by_type.setdefault(str(entry["ArtifactType"]), []).append(dict(entry))
    selected: List[Dict[str, Any]] = []
    for artifact_type in module["RequiresArtifacts"]:
        matches = by_type.get(str(artifact_type), [])
        if len(matches) != 1:
            raise ModuleExecutionError(
                "REQUIRED_ARTIFACT_COUNT",
                str(artifact_type),
            )
        selected.append(matches[0])
    return selected


def module_slug(sequence: int, module_id: str) -> str:
    """Create one bounded diagnostic directory name."""

    normalized = re.sub(r"[^A-Za-z0-9_.-]+", "_", module_id)
    return f"{sequence:04d}_{normalized[:80]}"

        "ExpectedOutputTypes": module["ProvidesArtifacts"],
        "OutputRoot": str(output_root),
    }


def terminate_process_tree(process: subprocess.Popen) -> None:
    """Terminate the bounded Module subprocess and its process group."""

    if process.poll() is not None:
        return
    if os.name == "nt":
        try:
            subprocess.run(
                ["taskkill", "/PID", str(process.pid), "/T", "/F"],
                stdout=subprocess.DEVNULL,
                stderr=subprocess.DEVNULL,
                timeout=15,
                check=False,
            )
        except Exception:
            process.kill()
    else:
        try:
            os.killpg(process.pid, signal.SIGKILL)
        except (OSError, ProcessLookupError):
            process.kill()
    try:
        process.wait(timeout=15)
    except subprocess.TimeoutExpired:
        process.kill()


def launch_module(
    entrypoint: Path,
    module_root: Path,
    request_path: Path,
    result_path: Path,
    stdout_path: Path,
    stderr_path: Path,
    timeout_seconds: int,
) -> int:
    """Run one Module without shell interpretation and without buffering logs."""

    command = [
        sys.executable,
        str(entrypoint),
        "--request-json",
        str(request_path),
        "--result-json",
        str(result_path),
    ]
    creation_flags = 0
    start_new_session = False
    if os.name == "nt":
        creation_flags = getattr(subprocess, "CREATE_NEW_PROCESS_GROUP", 0)
    else:
        start_new_session = True
    with stdout_path.open("wb") as stdout_stream:
        with stderr_path.open("wb") as stderr_stream:
            process = subprocess.Popen(
                command,
                cwd=module_root,
                stdin=subprocess.DEVNULL,
                stdout=stdout_stream,
                stderr=stderr_stream,
                shell=False,
                creationflags=creation_flags,
                start_new_session=start_new_session,
            )
            try:
                return process.wait(timeout=timeout_seconds)
            except subprocess.TimeoutExpired as exc:
                terminate_process_tree(process)
                raise ModuleExecutionError(
                    "MODULE_TIMEOUT",
                    str(timeout_seconds),
                ) from exc


def load_runtime_result(
    result_path: Path,
    module: Mapping[str, Any],
    run_id: str,
) -> Dict[str, Any]:
    """Validate the exact result envelope returned by one Module."""

    result = load_json(result_path, "MODULE_RESULT")
    if set(result) != RESULT_KEYS:
        raise ModuleExecutionError("MODULE_RESULT_SHAPE")
    expected_identity = (
        RUNTIME_PROTOCOL,
        RUNTIME_PROTOCOL_VERSION,
        CONTRACT_VERSION,
        run_id,
        module["Sequence"],
        module["Stage"],
        module["ModuleId"],
        module["ModuleVersion"],
    )
    observed_identity = (
        result["Protocol"],
        result["ProtocolVersion"],
        result["ContractVersion"],
        result["RunId"],
        result["Sequence"],
        result["Stage"],
        result["ModuleId"],
        result["ModuleVersion"],
    )
    if observed_identity != expected_identity:
        raise ModuleExecutionError("MODULE_RESULT_IDENTITY")
    if result["Status"] != "SUCCEEDED":
        raise ModuleExecutionError("MODULE_RESULT_STATUS", result["Status"])
    packages = result["ProducedPackages"]
    if not isinstance(packages, list) or len(packages) != len(set(packages)):
        raise ModuleExecutionError("MODULE_RESULT_PACKAGES")
    if not isinstance(result["Diagnostics"], list):
        raise ModuleExecutionError("MODULE_RESULT_DIAGNOSTICS")
    if not str(result["Capsule"]).endswith("C-0"):
        raise ModuleExecutionError("MODULE_RESULT_CAPSULE")
    return result


def resolve_produced_packages(
    artifact_store: Any,
    workspace_root: Path,
    run_root: Path,
    output_root: Path,
    module: Mapping[str, Any],
    result: Mapping[str, Any],
) -> List[Tuple[Path, Dict[str, Any]]]:
    """Verify produced package paths, identities and declared Artifact types."""

    resolved: List[Tuple[Path, Dict[str, Any]]] = []
    observed_types: List[str] = []
    for relative_path in result["ProducedPackages"]:
        if not isinstance(relative_path, str) or "\\" in relative_path:
            raise ModuleExecutionError("PACKAGE_RELATIVE_PATH")
        package_root = output_root.joinpath(*relative_path.split("/"))
        if not package_root.is_dir() or not inside(package_root, output_root):
            raise ModuleExecutionError("PACKAGE_LOCATION", relative_path)
        manifest, _ = artifact_store.verify_package(
            workspace_root,
            package_root,
            run_root.name,
        )
        identity = (
            manifest["Stage"],
            manifest["ProducerModuleId"],
            manifest["ProducerModuleVersion"],
        )
        expected = (
            module["Stage"],
            module["ModuleId"],
            module["ModuleVersion"],
        )
        if identity != expected:
            raise ModuleExecutionError("ARTIFACT_PRODUCER_IDENTITY")
        observed_types.append(str(manifest["ArtifactType"]))
        resolved.append((package_root, manifest))
    if sorted(observed_types) != sorted(module["ProvidesArtifacts"]):
        raise ModuleExecutionError("ARTIFACT_TYPE_SET")
    return resolved


def finalize_execution_directory(
    execution_root: Path,
    diagnostics_root: Path,
) -> Path:
    """Move complete runtime evidence out of transient staging."""

    output_root = execution_root / "packages"
    if output_root.is_dir() and not any(output_root.iterdir()):
        output_root.rmdir()
    diagnostics_root.parent.mkdir(parents=True, exist_ok=True)
    if diagnostics_root.exists():
        raise ExecutorError("DIAGNOSTIC_DIRECTORY_EXISTS", diagnostics_root.name)
    os.replace(execution_root, diagnostics_root)
    return diagnostics_root


def write_execution_record(
    run_root: Path,
    module: Mapping[str, Any],
    status: str,
    exit_code: int,
    reason: str,
    diagnostic_root: Path,
    artifact_ids: Sequence[str],
) -> None:
    """Persist one deterministic per-Module execution record."""

    relative_diagnostic = diagnostic_root.relative_to(run_root).as_posix()
    stdout_path = diagnostic_root / "stdout.txt"
    stderr_path = diagnostic_root / "stderr.txt"
    result_path = diagnostic_root / "module_result.json"
    record = {
        "ExecutionContract": "U002_U22_MODULE_EXECUTION_RECORD",
        "ExecutionVersion": VERSION,
        "ContractVersion": CONTRACT_VERSION,
        "RunId": run_root.name,
        "Sequence": module["Sequence"],
        "Stage": module["Stage"],
        "ModuleId": module["ModuleId"],
        "ModuleVersion": module["ModuleVersion"],
        "Status": status,
        "ExitCode": exit_code,
        "Reason": reason,
        "DiagnosticRoot": relative_diagnostic,
        "StdoutSha256": sha256_file(stdout_path) if regular_file(stdout_path) else None,
        "StderrSha256": sha256_file(stderr_path) if regular_file(stderr_path) else None,
        "ResultSha256": sha256_file(result_path) if regular_file(result_path) else None,
        "ArtifactIds": list(artifact_ids),
    }
    file_name = module_slug(
        int(module["Sequence"]),
        str(module["ModuleId"]),
    ) + ".json"
    write_json_atomic(run_root / "01_LOGS" / file_name, record)

    plan: Mapping[str, Any],

    next_stage: object,

    optional_failures: int,

    """Build the complete Run lifecycle record owned by the Executor."""

        "LastCompletedStage": last_stage,
        "NextStage": next_stage,
        "PlanSha256": plan["PlanSha256"],
        "ExecutionStarted": True,
        "CompletedModuleCount": completed,
        "OptionalFailureCount": optional_failures,
        "FailedModule": failed_module,
        "FailureReason": failure_reason,
    }


def execute_one_module(
    workspace_root: Path,
    run_root: Path,
    plan: Mapping[str, Any],
    module: Mapping[str, Any],
    artifact_store: Any,
    timeout_seconds: int,
) -> List[Dict[str, Any]]:
    """Execute one frozen Module and commit all declared Artifact packages."""

    slug = module_slug(int(module["Sequence"]), str(module["ModuleId"]))
    execution_root = run_root / "00_STAGING" / slug
    if execution_root.exists():
        raise ModuleExecutionError("MODULE_STAGING_EXISTS", slug)
    output_root = execution_root / "packages"
    output_root.mkdir(parents=True)
    module_root = workspace_root / str(module["ModuleRoot"])
    entrypoint = workspace_root / str(module["EntryPoint"])
    request_path = execution_root / "module_request.json"
    result_path = execution_root / "module_result.json"
    stdout_path = execution_root / "stdout.txt"
    stderr_path = execution_root / "stderr.txt"
    exit_code = -1
    artifact_entries: List[Dict[str, Any]] = []
    reason = "NA"
    try:
        inputs = artifact_inputs(artifact_store, run_root, module)
        request = build_request(
            plan,
            module,
            run_root,
            module_root,
            inputs,
            output_root,
        )
        if set(request) != REQUEST_KEYS:
            raise ModuleExecutionError("MODULE_REQUEST_SHAPE")
        write_json_atomic(request_path, request)
        exit_code = launch_module(
            entrypoint,
            module_root,
            request_path,
            result_path,
            stdout_path,
            stderr_path,
            timeout_seconds,
        )
        if exit_code != 0:
            raise ModuleExecutionError("MODULE_EXIT", str(exit_code))
        result = load_runtime_result(result_path, module, run_root.name)
        packages = resolve_produced_packages(
            artifact_store,
            workspace_root,
            run_root,
            output_root,
            module,
            result,
        )
        for package_root, _ in packages:
            _, entry = artifact_store.commit_package(
                workspace_root,
                run_root,
                package_root,
            )
            artifact_entries.append(entry)
        reason = "SUCCEEDED"
        diagnostic_root = finalize_execution_directory(
            execution_root,
            run_root / "02_DIAGNOSTICS/modules" / slug,
        )
        write_execution_record(
            run_root,
            module,
            "SUCCEEDED",
            exit_code,
            reason,
            diagnostic_root,
            [entry["ArtifactId"] for entry in artifact_entries],
        )
        return artifact_entries
    except Exception as exc:
        reason = getattr(exc, "reason", stable(exc.__class__.__name__))
        if execution_root.exists():
            diagnostic_root = finalize_execution_directory(
                execution_root,
                run_root / "02_DIAGNOSTICS/modules" / slug,
            )
            write_execution_record(
                run_root,
                module,
                "FAILED",
                exit_code,
                str(reason),
                diagnostic_root,
                [entry["ArtifactId"] for entry in artifact_entries],
            )
        if isinstance(exc, ExecutorError):
            raise
        raise ModuleExecutionError(str(reason)) from exc


def selected_output_types_present(
    artifact_store: Any,
    run_root: Path,
    output_selection: Sequence[str],
) -> bool:
    """Confirm every Run-selected output type exists exactly once."""

    index = artifact_store.load_index(run_root)
    observed = [str(entry["ArtifactType"]) for entry in index["Artifacts"]]
    return all(observed.count(str(output_type)) == 1 for output_type in output_selection)


def execute_run(
    workspace_root: Path,
    run_root: Path,
    timeout_seconds: int,
) -> Dict[str, Any]:
    """Execute one PREPARED Run serially through its frozen Module Plan."""

    plan = load_json(run_root / PLAN_RELATIVE, "RESOLVED_PLAN")
    module_plan = validate_plan_structure(workspace_root, run_root, plan)
    artifact_store = load_core_module(
        workspace_root,
        ARTIFACT_STORE_RELATIVE,
        "u002_u22_artifact_store",
        ("load_index", "verify_package", "commit_package", "audit_run"),
    )
    status = load_status(run_root, plan)
    if status["Status"] == "SUCCEEDED":
        audit = artifact_store.audit_run(workspace_root, run_root)
        return {
            "ModuleCount": int(status.get("CompletedModuleCount", len(module_plan))),
            "ArtifactCount": audit["ArtifactCount"],
            "OptionalFailureCount": int(status.get("OptionalFailureCount", 0)),
            "Replay": True,
        }
    if status["Status"] != "PREPARED" or status.get("ExecutionStarted") is not False:
        raise ExecutorError("RUN_NOT_PREPARED", status["Status"])
    availability = refresh_availability(workspace_root)
    validate_current_module_eligibility(module_plan, availability)
    first_stage = module_plan[0]["Stage"]
    write_json_atomic(
        run_root / STATUS_RELATIVE,
        status_record(plan, "RUNNING", None, first_stage, 0, 0),
    )
    completed = 0
    optional_failures = 0
    artifact_count = 0
    last_stage: object = None
    current_module_id: object = None
    try:
        for index, module in enumerate(module_plan):
            current_module_id = module["ModuleId"]
            next_stage = (
                module_plan[index + 1]["Stage"]
                if index + 1 < len(module_plan)
                else None
            )
            try:
                artifacts = execute_one_module(
                    workspace_root,
                    run_root,
                    plan,
                    module,
                    artifact_store,
                    timeout_seconds,
                )
                artifact_count += len(artifacts)
                completed += 1
                last_stage = module["Stage"]
            except ExecutorError as exc:
                if module["FailurePolicy"] != "OPTIONAL":
                    raise
                optional_failures += 1
            write_json_atomic(
                run_root / STATUS_RELATIVE,
                status_record(
                    plan,
                    "RUNNING",
                    last_stage,
                    next_stage,
                    completed,
                    optional_failures,
                ),
            )
        if not selected_output_types_present(
            artifact_store,
            run_root,
            plan["OutputSelection"],
        ):
            raise ExecutorError("SELECTED_OUTPUT_MISSING")
        audit = artifact_store.audit_run(workspace_root, run_root)
        write_json_atomic(
            run_root / STATUS_RELATIVE,
            status_record(
                plan,
                "SUCCEEDED",
                last_stage,
                None,
                completed,
                optional_failures,
            ),
        )
        return {
            "ModuleCount": completed,
            "ArtifactCount": audit["ArtifactCount"],
            "OptionalFailureCount": optional_failures,
            "Replay": False,
        }
    except Exception as exc:
        reason = getattr(exc, "reason", stable(exc.__class__.__name__))
        write_json_atomic(
            run_root / STATUS_RELATIVE,
            status_record(
                plan,
                "FAILED",
                last_stage,
                None,
                completed,
                optional_failures,
                current_module_id,
                str(reason),
            ),
        )
        if isinstance(exc, ExecutorError):
            raise
        raise ExecutorError(str(reason)) from exc


def self_test(workspace_root: Path) -> str:
    """Exercise deterministic Executor helpers without launching Modules."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise ExecutorError("SELFTEST")
        checks.append(True)

    check(PROGRAM_ID.startswith("U002"))
    check(VERSION == "0.1")
    check(CONTRACT_VERSION == "U22.0.1")
    check(RUNTIME_PROTOCOL_VERSION == "0.1")
    check(len(STAGES) == 8)
    check(STAGES[0] == "S0")
    check(STAGES[-1] == "S7")
    check(len(PLAN_KEYS) == 18)
    check(len(MODULE_PLAN_KEYS) == 18)
    check(len(REQUEST_KEYS) == 17)
    check(len(RESULT_KEYS) == 12)
    check(stable("a b/c") == "A_B_C")
    check(IDENTIFIER_PATTERN.fullmatch("RUN_001") is not None)
    check(IDENTIFIER_PATTERN.fullmatch("../bad") is None)
    check(module_slug(2, "Module.A") == "0002_Module.A")
    check(len(sha256_bytes(b"x")) == 64)
    check(canonical_bytes({"b": 1, "a": 2}).startswith(b"{"))
    check(inside(workspace_root / "01_PROGRAMS", workspace_root))
    check(not inside(workspace_root.parent, workspace_root))
    check(ARTIFACT_STORE_RELATIVE.suffix == ".py")
    check(QUALIFICATION_RELATIVE.suffix == ".py")
    check(regular_file(workspace_root / ARTIFACT_STORE_RELATIVE))
    check(regular_file(workspace_root / QUALIFICATION_RELATIVE))
    check("PlanSha256" in PLAN_KEYS)
    check("ConfigurationSha256" in MODULE_PLAN_KEYS)
    check("ExpectedOutputTypes" in REQUEST_KEYS)
    check("ProducedPackages" in RESULT_KEYS)
    sample_plan = {
        "RunId": "RUN_TEST",
        "PlanSha256": "a" * 64,
    }
    check(len(status_record(sample_plan, "RUNNING", None, None, 0, 0)) == 10)
    check(sys.executable != "")
    check(len(checks) == 29)
    if len(checks) != 30:
        raise ExecutorError("SELFTEST_COUNT")
    return "U22A6X-ST N-T30-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Create explicit self-test and Run execution modes."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute-run")
    parser.add_argument("--workspace-root", required=True)
    parser.add_argument("--module-timeout-seconds", type=int, default=300)
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    """Execute one controlled responsibility and emit one compact capsule."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        workspace_root = require_workspace(Path(args.workspace_root))
        if args.module_timeout_seconds < 1 or args.module_timeout_seconds > 86400:
            raise ExecutorError("MODULE_TIMEOUT_RANGE")
        if args.self_test:
            print(self_test(workspace_root))
            return 0
        run_root = require_run_root(
            workspace_root,
            Path(args.execute_run),
        )
        result = execute_run(
            workspace_root,
            run_root,
            args.module_timeout_seconds,
        )
        replay = 1 if result["Replay"] else 0
        print(
            f"U22A6X N-M{result['ModuleCount']}"
            f"-A{result['ArtifactCount']}"
            f"-O{result['OptionalFailureCount']}"
            f"-R{replay}-S1-X0 C-0"
        )
        return 0
    except ExecutorError as exc:
        print(
            f"U22A6X N-M0-A0-O0-R0-S0-X1 C-7 R-{exc.reason}"
        )
        return 7
    except Exception as exc:
        reason = stable(exc.__class__.__name__)
        print(
            f"U22A6X N-M0-A0-O0-R0-S0-X1 C-7 R-{reason}"
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
