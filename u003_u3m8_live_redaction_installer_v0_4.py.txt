#!/usr/bin/env python3
"""Prepare and install U3M8 v0.4 Live Evidence Redaction."""

from __future__ import annotations

import argparse
import csv
import hashlib
import shutil
import sys
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PACKAGE_NAME = "UDP003_U3M8_LIVE_EVIDENCE_REDACTION_v0_4.zip"
PAYLOADS = {
    "01_PROGRAMS/u003_evidence_review_workbench_v0_20.py",
    "01_PROGRAMS/u003_gui_shell_v0_7.py",
    "01_PROGRAMS/u003_module_action_gui_v0_2.py",
    "01_PROGRAMS/u003_live_review_context_v0_1.py",
    "01_PROGRAMS/u003_u3m8_live_redaction_acceptance_gate_v0_1.py",
    "02_CONTROLS/contracts/u003_live_review_context_contract_v0_1.json",
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.4.0/algorithm_card.json",
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.4.0/implementation.py",
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.4.0/module_manifest.json",
    "03_DOCUMENTS/U3M8_node_report_live_evidence_redaction_v0_4_2026-08-27.txt",
}
REQUIRED_MEMBERS = PAYLOADS | {"PACKAGE_SHA256.csv"}
SUPERSEDED = (
    Path("01_PROGRAMS/u003_evidence_review_workbench_v0_19.py"),
    Path("01_PROGRAMS/u003_gui_shell_v0_6.py"),
    Path("01_PROGRAMS/u003_module_action_gui_v0_1.py"),
    Path("02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.1.0"),
    Path("02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.2.0"),
    Path("02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.3.0"),
)


class InstallError(RuntimeError):
    """Raised when installation cannot preserve a deterministic topology."""


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def archive_root(root: Path, label: str) -> Path:
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    target = root / "A-Archive" / f"{label}_{stamp}"
    suffix = 1
    while target.exists():
        target = root / "A-Archive" / f"{label}_{stamp}_{suffix:02d}"
        suffix += 1
    return target


def move_to_archive(path: Path, root: Path, target_root: Path) -> None:
    destination = target_root / path.relative_to(root)
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(destination))


def require_foundation(root: Path) -> None:
    required = (
        root / "01_PROGRAMS" / "u003_evidence_review_workbench_v0_19.py",
        root / "01_PROGRAMS" / "u003_gui_shell_v0_6.py",
        root / "01_PROGRAMS" / "u003_module_action_gui_v0_1.py",
        root / "01_PROGRAMS" / "u003_module_execution_coordinator_v0_3.py",
        root / "01_PROGRAMS" / "u003_module_runtime_v0_2.py",
        root / "02_CONTROLS" / "contracts" / "u003_module_contract_v0_2.json",
        root
        / "02_MODULES"
        / "EVIDENCE_VIEW"
        / "CameraRadar.U003.FrameMask"
        / "0.3.0"
        / "module_manifest.json",
    )
    missing = [str(path) for path in required if not path.is_file()]
    if missing:
        raise InstallError(f"Accepted U3M8 v0.3 foundation is incomplete: {missing}")


def validate_zip(archive: zipfile.ZipFile) -> None:
    members = []
    for name in archive.namelist():
        pure = PurePosixPath(name)
        if pure.is_absolute() or ".." in pure.parts:
            raise InstallError(f"Unsafe ZIP member: {name}")
        if not name.endswith("/"):
            members.append(name)
    if set(members) != REQUIRED_MEMBERS:
        raise InstallError(
            f"Package inventory mismatch. Missing={sorted(REQUIRED_MEMBERS - set(members))}; "
            f"Extra={sorted(set(members) - REQUIRED_MEMBERS)}"
        )


def validate_payloads(prepared: Path) -> list[str]:
    with (prepared / "PACKAGE_SHA256.csv").open(
        "r", encoding="utf-8-sig", newline=""
    ) as stream:
        rows = list(csv.DictReader(stream))
    if len(rows) != len(PAYLOADS):
        raise InstallError("PACKAGE_SHA256 row count is invalid.")
    verified = []
    for row in rows:
        relative = str(row.get("RelativePath", "")).strip()
        expected = str(row.get("Sha256", "")).strip().lower()
        path = (prepared / relative).resolve()
        try:
            path.relative_to(prepared.resolve())
        except ValueError as error:
            raise InstallError(f"Manifest path escapes package root: {relative}") from error
        if not path.is_file() or path.is_symlink():
            raise InstallError(f"Payload is missing: {relative}")
        if sha256_file(path) != expected:
            raise InstallError(f"Payload hash mismatch: {relative}")
        verified.append(relative)
    if set(verified) != PAYLOADS:
        raise InstallError("Manifest payload inventory is invalid.")
    return verified


def clean_temp(root: Path, keep: set[Path]) -> tuple[Path | None, int]:
    temp = root / "04_TEMP"
    temp.mkdir(parents=True, exist_ok=True)
    selected = [
        item
        for item in temp.iterdir()
        if item.resolve() not in keep
        and not item.name.startswith("U3M8_START_HERE")
        and (
            item.name.startswith("U3M8")
            or item.name.startswith("UDP003_U3M8")
            or item.name.startswith("u003_u3m8")
        )
    ]
    if not selected:
        return None, 0
    target = archive_root(root, "U3M8_TEMP_CLEANUP")
    for item in selected:
        move_to_archive(item, root, target)
    return target, len(selected)


def archive_superseded(root: Path) -> tuple[Path | None, int]:
    selected = [root / relative for relative in SUPERSEDED]
    selected = [item for item in selected if item.exists()]
    if not selected:
        return None, 0
    target = archive_root(root, "U3M8_V04_SUPERSEDED")
    for item in selected:
        move_to_archive(item, root, target)
    return target, len(selected)


def install(package_path: Path, root: Path, installer_path: Path) -> int:
    package_path = package_path.resolve()
    root = root.resolve()
    if root.name.upper() != "U-UDP003":
        raise InstallError(f"Expected U-UDP003 root, got: {root}")
    if not package_path.is_file():
        raise InstallError(f"Transfer ZIP is missing: {package_path}")
    require_foundation(root)
    keep = {package_path.resolve(), installer_path.resolve()}
    temp_archive, temp_count = clean_temp(root, keep)
    prepared = root / "04_TEMP" / "U3M8_V04_PREPARED"
    if prepared.exists():
        raise InstallError(f"Prepared directory already exists: {prepared}")
    prepared.mkdir(parents=True)
    with zipfile.ZipFile(package_path, "r") as archive:
        validate_zip(archive)
        archive.extractall(prepared)
    verified = validate_payloads(prepared)
    old_archive, old_count = archive_superseded(root)
    for relative in verified:
        source = prepared / relative
        destination = root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        writing = destination.with_name(destination.name + ".writing")
        shutil.copy2(source, writing)
        writing.replace(destination)

    print("U3M8 v0.4 preparation checks:")
    print(f"- Verified payload files: {len(verified)}")
    print(f"- Archived superseded files/directories: {old_count}")
    print(f"- Archived prior U3M8 TEMP items: {temp_count}")
    print(f"- Latest workbench: {root / '01_PROGRAMS' / 'u003_evidence_review_workbench_v0_20.py'}")
    print(f"- Latest Module: {root / '02_MODULES' / 'EVIDENCE_VIEW' / 'CameraRadar.U003.FrameMask' / '0.4.0'}")
    if old_archive is not None:
        print(f"- Superseded archive: {old_archive}")
    if temp_archive is not None:
        print(f"- TEMP archive: {temp_archive}")
    print("U3M8P N-F10-M1-A1-L1-R1-X0 C-0")
    return 0


def run_self_test() -> int:
    checks = 0

    def check(condition: bool) -> None:
        nonlocal checks
        if not condition:
            raise AssertionError(f"check {checks + 1}")
        checks += 1

    check(PACKAGE_NAME.endswith("v0_4.zip"))
    check(len(PAYLOADS) == 10)
    check(len(REQUIRED_MEMBERS) == 11)
    check(any("live_review_context" in item for item in PAYLOADS))
    check(any("v0_20" in item for item in PAYLOADS))
    check(any("0.4.0" in item for item in PAYLOADS))
    check(len(SUPERSEDED) == 6)
    print(f"U3M8-INSTALLER-SELFTEST-PASS-{checks} C-0")
    return 0


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--install", action="store_true")
    parser.add_argument("--root", type=Path)
    parser.add_argument("--package", type=Path)
    arguments = parser.parse_args(sys.argv[1:] if argv is None else argv)
    if arguments.self_test:
        return run_self_test()
    if not arguments.install or arguments.root is None or arguments.package is None:
        raise InstallError("Use --self-test or --install --root <U-UDP003> --package <ZIP>.")
    return install(arguments.package, arguments.root, Path(__file__))


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as error:
        print(f"U3M8PE K-{type(error).__name__.upper()} D-{error} X1 C-7", file=sys.stderr)
        raise SystemExit(7)
