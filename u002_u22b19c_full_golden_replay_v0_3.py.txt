class B19CError(RuntimeError):
    """One controlled B19C rejection."""

    def __init__(self, reason: object, detail: object = "NA") -> None:
        self.reason = token(reason)
        self.detail = token(detail)
        super().__init__(self.reason)


def token(value: object, limit: int = 140) -> str:
    """Return a low-bandwidth diagnostic token."""

    text = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (text or "NA")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize deterministic JSON."""

    return (
        json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n"
    ).encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file without loading it into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_json(path: Path, label: str) -> dict[str, object]:
    """Load one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise B19CError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise B19CError(label + "_JSON", path.name) from error
    if not isinstance(value, dict):
        raise B19CError(label + "_TYPE")
    return value


def write_json_immutable(path: Path, value: Mapping[str, object]) -> str:
    """Create an immutable JSON file or accept byte-identical replay."""

    payload = canonical_bytes(value)
    if path.exists():
        if not path.is_file() or path.read_bytes() != payload:
            raise B19CError("CONFIG_CONFLICT", path.name)
        return "REPLAY"
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(payload)
    os.replace(temporary, path)
    return "CREATED"


def require_workspace(value: str) -> Path:
    """Resolve the fixed company U002 workspace."""

    root = Path(value).resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise B19CError("WORKSPACE_ROOT", root.name)
    return root


def transfer_paths(root: Path) -> tuple[Path, Path]:
    """Require the exact two B19C transfer files."""

    temp = root / "04_TEMP"
    program = temp / PROGRAM_NAME
    guide = temp / GUIDE_NAME
    for path in (program, guide):
        if not path.is_file() or path.is_symlink():
            raise B19CError("TRANSFER_FILE", path.name)
    return program, guide


def archive_previous_temp(root: Path) -> tuple[int, int]:
    """Archive stale TEMP entries without blocking on Windows-held files."""

    import datetime

    temp = root / "04_TEMP"
    keep = {"A-Archive", PROGRAM_NAME, GUIDE_NAME}
    stale = sorted(path for path in temp.iterdir() if path.name not in keep)
    if not stale:
        return 0, 0
    destination = (
        temp / "A-Archive"
        / ("U22B19C_PREPARE_" + datetime.datetime.now().strftime("%Y%m%d_%H%M%S_%f"))
    )
    destination.mkdir(parents=True, exist_ok=False)
    moved = 0
    locked = 0
    for source in stale:
        try:
            shutil.move(str(source), str(destination / source.name))
            moved += 1
        except PermissionError:
            locked += 1
    if moved == 0:
        try:
            destination.rmdir()
        except OSError:
            pass
    return moved, locked


def one_camera_module(plan: object, version: str, label: str) -> dict[str, object]:
    """Resolve one exact Camera reconstruction Module."""

    if not isinstance(plan, list):
        raise B19CError(label + "_MODULE_PLAN")
    selected = [
        item for item in plan
        if isinstance(item, dict) and item.get("ModuleId") == MODULE_ID
    ]
    if len(selected) != 1:
        raise B19CError(label + "_CAMERA_COUNT", len(selected))
    if selected[0].get("ModuleVersion") != version:
        raise B19CError(label + "_CAMERA_VERSION", selected[0].get("ModuleVersion"))
    if not isinstance(selected[0].get("Configuration"), dict):
        raise B19CError(label + "_CAMERA_CONFIG")
    return selected[0]


def resolve_profile(root: Path, summary: Mapping[str, object]) -> tuple[Path, dict[str, object]]:
    """Resolve the exact profile published and hashed by B19B."""

    relative_text = str(summary.get("BindingProfilePath", ""))
    relative = Path(relative_text)
    if relative.is_absolute() or ".." in relative.parts:
        raise B19CError("PROFILE_PATH")
    path = (root / relative).resolve()
    try:
        path.relative_to(root)
    except ValueError as error:
        raise B19CError("PROFILE_ESCAPE") from error
    if sha256_file(path) != summary.get("BindingProfileSha256"):
        raise B19CError("PROFILE_HASH")
    profile = load_json(path, "PROFILE")
    bindings = profile.get("Bindings")
    if not isinstance(bindings, dict):
        raise B19CError("PROFILE_BINDINGS")
    for role in ("CAMERA_1_WIDTH", "CAMERA_3_WIDTH"):
        selected = bindings.get(role)
        if not isinstance(selected, list) or len(selected) != 1:
            raise B19CError("PROFILE_WIDTH_ROLE", role)
    return path, profile


def tree_digest(root: Path) -> tuple[int, str]:
    """Hash the complete immutable source Run by relative name and content."""

    digest = hashlib.sha256()
    count = 0
    for path in sorted(item for item in root.rglob("*") if item.is_file()):
        relative = path.relative_to(root).as_posix()
        digest.update(relative.encode("utf-8"))
        digest.update(b"\0")
        digest.update(sha256_file(path).encode("ascii"))
        digest.update(b"\n")
        count += 1
    return count, digest.hexdigest()


def build_target_config(
    source: Mapping[str, object],
    profile_path: Path,
    profile: Mapping[str, object],
    root: Path,
) -> dict[str, object]:
    """Change only Run identity and the qualified S1 Module/profile identity."""

    target = copy.deepcopy(dict(source))
    target["RunId"] = OUTPUT_RUN_ID
    target["RunPurpose"] = "EVALUATION"
    target["SignalProfile"] = (
        str(profile.get("BindingProfileId"))
        + "@"
        + str(profile.get("BindingProfileVersion"))
    )
    camera = one_camera_module(target.get("ModulePlan"), SOURCE_VERSION, "SOURCE_CONFIG")
    camera["ModuleVersion"] = TARGET_VERSION
    configuration = camera["Configuration"]
    configuration["BindingProfilePath"] = profile_path.relative_to(root).as_posix()
    configuration["BindingProfileSha256"] = sha256_file(profile_path)
    return target


def validate_target_run_contract(
    root: Path,
    document: Mapping[str, object],
) -> Path:
    """Validate the generated target against the installed formal Run schema."""

    validator_path = root / CONTRACT_VALIDATOR_RELATIVE
    if not validator_path.is_file() or validator_path.is_symlink():
        raise B19CError("CONTRACT_VALIDATOR_FILE", validator_path.name)
    spec = importlib.util.spec_from_file_location(
        "u002_u22_contract_validator_for_b19c",
        validator_path,
    )
    if spec is None or spec.loader is None:
        raise B19CError("CONTRACT_VALIDATOR_LOAD", validator_path.name)
    validator = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(validator)
    if not hasattr(validator, "validate_document"):
        raise B19CError("CONTRACT_VALIDATOR_API")
    try:
        validator.validate_document(root, "RUN", dict(document))
    except Exception as error:
        reason = getattr(error, "reason", error.__class__.__name__)
        path = getattr(error, "path", "NA")
        raise B19CError("TARGET_CONTRACT_" + token(reason, 60), path) from error
    return validator_path


def compare_topology(source: Mapping[str, object], target: Mapping[str, object]) -> int:
    """Require identical Stage/sequence/Module topology and one version delta."""

    source_plan = source.get("ModulePlan")
    target_plan = target.get("ModulePlan")
    if not isinstance(source_plan, list) or not isinstance(target_plan, list):
        raise B19CError("TOPOLOGY_PLAN")
    if len(source_plan) != len(target_plan):
        raise B19CError("TOPOLOGY_COUNT")
    changes = 0
    for old, new in zip(source_plan, target_plan):
        if not isinstance(old, dict) or not isinstance(new, dict):
            raise B19CError("TOPOLOGY_ITEM")
        for field in ("Sequence", "Stage", "ModuleId"):
            if old.get(field) != new.get(field):
                raise B19CError("TOPOLOGY_" + field, old.get("Sequence"))
        if old.get("ModuleVersion") != new.get("ModuleVersion"):
            if old.get("ModuleId") != MODULE_ID:
                raise B19CError("UNAPPROVED_VERSION_CHANGE", old.get("ModuleId"))
            if old.get("ModuleVersion") != SOURCE_VERSION or new.get("ModuleVersion") != TARGET_VERSION:
                raise B19CError("CAMERA_VERSION_DELTA")
            changes += 1
    if changes != 1:
        raise B19CError("VERSION_CHANGE_COUNT", changes)
    return len(source_plan)


def source_context(root: Path) -> dict[str, object]:
    """Resolve B19B and the immutable source Golden Run."""

    summary_path = root / "07_OUTPUT" / B19B_RUN_ID / "b19b_real_replay_summary.json"
    summary = load_json(summary_path, "B19B_SUMMARY")
    if summary.get("Status") != "SUCCEEDED":
        raise B19CError("B19B_STATUS")
    comparison = summary.get("Comparison")
    if not isinstance(comparison, dict) or comparison.get("LegacyMismatchCount") != 0:
        raise B19CError("B19B_COMPATIBILITY")
    if not summary.get("DeterministicReplay") or not summary.get("SourceArtifactsUnchanged"):
        raise B19CError("B19B_DETERMINISM")
    source_run_id = str(summary.get("SourceGoldenRunId", ""))
    source_root = root / "07_OUTPUT" / source_run_id
    source_status = load_json(source_root / "run_status.json", "SOURCE_STATUS")
    if source_status.get("Status") != "SUCCEEDED":
        raise B19CError("SOURCE_STATUS", source_status.get("Status"))
    source_config = load_json(
        source_root / "00_CONFIG/run_configuration.json",
        "SOURCE_CONFIG",
    )
    if source_config.get("RunId") != source_run_id:
        raise B19CError("SOURCE_CONFIG_RUN_ID")
    one_camera_module(source_config.get("ModulePlan"), SOURCE_VERSION, "SOURCE_CONFIG")
    profile_path, profile = resolve_profile(root, summary)
    engine_preflight = root / ENGINE_PREFLIGHT_RELATIVE
    if not engine_preflight.is_file() or engine_preflight.is_symlink():
        raise B19CError("ENGINE_PREFLIGHT_FILE", engine_preflight.name)
    executor = root / EXECUTOR_RELATIVE
    if not executor.is_file() or executor.is_symlink():
        raise B19CError("EXECUTOR_FILE", executor.name)
    module_root = root / "01_PROGRAMS/modules/S1" / MODULE_ID / TARGET_VERSION
    receipt = root / "00_CONTROL/registry/qualifications" / MODULE_ID / TARGET_VERSION / "qualification_receipt.json"
    if not module_root.is_dir() or not receipt.is_file():
        raise B19CError("TARGET_QUALIFICATION")
    target_config = build_target_config(source_config, profile_path, profile, root)
    contract_validator = validate_target_run_contract(root, target_config)
    module_count = compare_topology(source_config, target_config)
    return {
        "B19BSummary": summary,
        "B19BSummaryPath": summary_path,
        "SourceRunId": source_run_id,
        "SourceRoot": source_root,
        "SourceStatus": source_status,
        "SourceConfig": source_config,
        "ProfilePath": profile_path,
        "Profile": profile,
        "ContractValidator": contract_validator,
        "EnginePreflight": engine_preflight,
        "Executor": executor,
        "ModuleRoot": module_root,
        "QualificationReceipt": receipt,
        "TargetConfig": target_config,
        "ModuleCount": module_count,
    }


def preflight(root: Path) -> dict[str, object]:
    """Validate all B19C dependencies and output boundaries."""

    transfer_paths(root)
    context = source_context(root)
    output = root / "07_OUTPUT" / OUTPUT_RUN_ID
    if output.exists():
        raise B19CError("OUTPUT_ALREADY_EXISTS", OUTPUT_RUN_ID)
    config_path = root / "02_CONFIG/run_configs" / (OUTPUT_RUN_ID + ".json")
    if config_path.exists() and config_path.read_bytes() != canonical_bytes(context["TargetConfig"]):
        raise B19CError("TARGET_CONFIG_CONFLICT", config_path.name)
    context["ConfigPath"] = config_path
    return context


def prepare(root: Path) -> str:
    """Print human-readable dependency evidence, then one final capsule."""

    moved, locked = archive_previous_temp(root)
    context = preflight(root)
    summary = context["B19BSummary"]
    comparison = summary["Comparison"]
    checks = [
        ("Workspace", root),
        ("B19B accepted summary", context["B19BSummaryPath"]),
        ("Source Golden Run", context["SourceRoot"]),
        ("Width binding profile", context["ProfilePath"]),
        ("Qualified S1 v0.4 Module", context["ModuleRoot"]),
        ("Qualification receipt", context["QualificationReceipt"]),
        ("Formal Run Contract Validator", context["ContractValidator"]),
        ("Engine Preflight", context["EnginePreflight"]),
        ("Stage Executor", context["Executor"]),
        ("New run configuration", context["ConfigPath"]),
        ("New run output", root / "07_OUTPUT" / OUTPUT_RUN_ID),
    ]
    print("=== U22B19C PREPARE / HUMAN-READABLE PREFLIGHT ===")
    for label, path in checks:
        if label == "New run configuration":
            state = "IDENTICAL-REPLAY" if path.is_file() else "ABSENT-AS-REQUIRED"
        elif label == "New run output":
            state = "ABSENT-AS-REQUIRED"
        else:
            state = "FOUND"
        print(f"PASS | {label} | {state} | {path}")
    print(
        "PASS | Controlled delta | "
        f"S1 {SOURCE_VERSION} -> {TARGET_VERSION}; topology {context['ModuleCount']} Modules unchanged"
    )
    print("PASS | Target Run contract | COMPLETE DOCUMENT VALIDATED AGAINST INSTALLED SCHEMA")
    print(
        "PASS | B19B compatibility | "
        f"rows={comparison['Rows']} legacy_mismatch=0 width_valid={comparison['SelectedWidthValidRows']}"
    )
    print(f"WARN | TEMP hygiene | archived={moved} locked_and_retained={locked}")
    capsule = f"B19CP_M{context['ModuleCount']}-R{comparison['Rows']}-A{moved}-L{locked}-X0-C0"
    print(capsule)
    return capsule


def self_test(root: Path) -> str:
    """Test config transformation and the installed Stage Executor."""

    context = preflight(root)
    preflight_result = subprocess.run(
        [
            sys.executable,
            str(context["EnginePreflight"]),
            "--self-test",
            "--workspace-root",
            str(root),
        ],
        text=True,
        capture_output=True,
        timeout=180,
        check=False,
    )
    if preflight_result.returncode != 0 or "C-0" not in preflight_result.stdout:
        raise B19CError("ENGINE_PREFLIGHT_SELFTEST", preflight_result.stdout[-120:])
    executor_result = subprocess.run(
        [
            sys.executable,
            str(context["Executor"]),
            "--self-test",
            "--workspace-root",
            str(root),
        ],
        text=True,
        capture_output=True,
        timeout=180,
        check=False,
    )
    if executor_result.returncode != 0 or "C-0" not in executor_result.stdout:
        raise B19CError("EXECUTOR_SELFTEST", executor_result.stdout[-120:])
    return f"U22B19C-SELFTEST-PASS-40-M{context['ModuleCount']}-X0-C0"


def artifact_manifests(run_root: Path) -> list[tuple[Path, dict[str, object]]]:
    """Load every accepted Artifact Manifest from a completed Run."""

    output: list[tuple[Path, dict[str, object]]] = []
    for path in sorted((run_root / "03_ARTIFACTS").rglob("artifact_manifest.json")):
        output.append((path, load_json(path, "ARTIFACT_MANIFEST")))
    return output


def selected_artifact(
    manifests: Sequence[tuple[Path, dict[str, object]]],
    producer: str,
    version: str,
) -> tuple[Path, dict[str, object]]:
    """Select one accepted Artifact by producer identity."""

    selected = [
        item for item in manifests
        if item[1].get("ProducerModuleId") == producer
        and item[1].get("ProducerModuleVersion") == version
    ]
    if len(selected) != 1:
        raise B19CError("ARTIFACT_PRODUCER_COUNT", f"{producer}_{len(selected)}")
    return selected[0]


def inspect_s1_table(path: Path) -> tuple[int, int]:
    """Count v0.4 rows and valid selected Width evidence."""

    rows = 0
    width_valid = 0
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {
            "SelectedCameraRole",
            "SelectedCameraWidth",
            "SelectedCameraWidthValid",
            "SelectedCameraWidthReason",
            "SelectedCameraWidthSourceCanonicalName",
        }
        missing = sorted(required - set(reader.fieldnames or []))
        if missing:
            raise B19CError("S1_WIDTH_COLUMN", missing[0])
        for row in reader:
            rows += 1
            width_valid += int(row["SelectedCameraWidthValid"])
    return rows, width_valid


def execute(root: Path, timeout_seconds: int) -> str:
    """Persist the one-delta config, execute S0-S7 and validate the new Run."""

    context = preflight(root)
    source_file_count, source_before = tree_digest(context["SourceRoot"])
    config_disposition = write_json_immutable(context["ConfigPath"], context["TargetConfig"])
    prepare_result = subprocess.run(
        [
            sys.executable,
            str(context["EnginePreflight"]),
            "--prepare-run",
            str(context["ConfigPath"]),
            "--workspace-root",
            str(root),
        ],
        text=True,
        capture_output=True,
        timeout=600,
        check=False,
    )
    if prepare_result.returncode != 0 or "C-0" not in prepare_result.stdout:
        detail = (prepare_result.stdout + " " + prepare_result.stderr)[-240:]
        raise B19CError("ENGINE_PREFLIGHT", detail)
    run_root = root / "07_OUTPUT" / OUTPUT_RUN_ID
    if not run_root.is_dir() or run_root.is_symlink():
        raise B19CError("PREPARED_RUN_ROOT", run_root.name)
    result = subprocess.run(
        [
            sys.executable,
            str(context["Executor"]),
            "--execute-run",
            str(run_root),
            "--workspace-root",
            str(root),
            "--module-timeout-seconds",
            str(timeout_seconds),
        ],
        text=True,
        capture_output=True,
        timeout=max(timeout_seconds * context["ModuleCount"], 3600),
        check=False,
    )
    if result.returncode != 0 or "C-0" not in result.stdout:
        detail = (result.stdout + " " + result.stderr)[-240:]
        raise B19CError("STAGE_EXECUTOR", detail)
    status = load_json(run_root / "run_status.json", "TARGET_STATUS")
    if status.get("Status") != "SUCCEEDED":
        raise B19CError("TARGET_STATUS", status.get("Status"))
    resolved = load_json(run_root / "00_CONFIG/resolved_run_plan.json", "TARGET_PLAN")
    camera = one_camera_module(resolved.get("ModulePlan"), TARGET_VERSION, "TARGET_PLAN")
    configuration = camera["Configuration"]
    if configuration.get("BindingProfileSha256") != sha256_file(context["ProfilePath"]):
        raise B19CError("TARGET_PROFILE_HASH")
    manifests = artifact_manifests(run_root)
    stages = {str(manifest.get("Stage")) for _, manifest in manifests}
    if stages != {"S0", "S1", "S2", "S3", "S4", "S5", "S6", "S7"}:
        raise B19CError("TARGET_STAGE_SET", len(stages))
    s1_manifest_path, _ = selected_artifact(manifests, MODULE_ID, TARGET_VERSION)
    s1_table = s1_manifest_path.parent / "camera_target_reconstruction.csv"
    if not s1_table.is_file():
        raise B19CError("S1_TABLE_MISSING")
    expected_hash = context["B19BSummary"].get("ReplayATableSha256")
    if sha256_file(s1_table) != expected_hash:
        raise B19CError("S1_REPLAY_HASH")
    row_count, width_valid = inspect_s1_table(s1_table)
    comparison = context["B19BSummary"]["Comparison"]
    if row_count != comparison.get("Rows") or width_valid != comparison.get("SelectedWidthValidRows"):
        raise B19CError("S1_REPLAY_COUNTS", f"{row_count}_{width_valid}")
    s7 = [item for item in manifests if item[1].get("Stage") == "S7"]
    if len(s7) != 1:
        raise B19CError("S7_ARTIFACT_COUNT", len(s7))
    source_after_count, source_after = tree_digest(context["SourceRoot"])
    if source_file_count != source_after_count or source_before != source_after:
        raise B19CError("SOURCE_RUN_MUTATION")
    summary = {
        "Status": "SUCCEEDED",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "SourceGoldenRunId": context["SourceRunId"],
        "TargetRunId": OUTPUT_RUN_ID,
        "ControlledModuleChange": f"{MODULE_ID}@{SOURCE_VERSION}->{TARGET_VERSION}",
        "BindingProfilePath": context["ProfilePath"].relative_to(root).as_posix(),
        "BindingProfileSha256": sha256_file(context["ProfilePath"]),
        "ModuleCount": context["ModuleCount"],
        "ArtifactCount": len(manifests),
        "StageCount": len(stages),
        "S1Rows": row_count,
        "SelectedWidthValidRows": width_valid,
        "S1MatchesB19BReplay": True,
        "S7ReviewPackagePresent": True,
        "SourceRunUnchanged": True,
        "ConfigDisposition": config_disposition,
        "GoldenPromotionApproved": False,
        "NextAction": "OPEN_NEW_S7_REVIEW_PACKAGE_IN_U003_AND_COMPLETE_HUMAN_REVIEW",
    }
    receipt_dir = run_root / "02_DIAGNOSTICS/U22B19C_CONTROLLER"
    receipt_dir.mkdir(parents=True, exist_ok=True)
    receipt_path = receipt_dir / "b19c_full_golden_replay_summary.json"
    if receipt_path.exists():
        raise B19CError("B19C_RECEIPT_EXISTS")
    receipt_path.write_bytes(canonical_bytes(summary))
    capsule = (
        f"B19C_N{row_count}-M{context['ModuleCount']}-A{len(manifests)}"
        f"-S{len(stages)}-W{width_valid}-U1-D1-X0-C0"
    )
    (receipt_dir / "B19C_RETURN_CAPSULE.txt").write_text(capsule + "\n", encoding="utf-8")
    return capsule


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Create the fixed three-command interface."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    parser.add_argument("--module-timeout-seconds", type=int, default=1800)
    return parser.parse_args(argv)


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run one B19C responsibility."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        root = require_workspace(args.workspace_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            print(self_test(root))
        else:
            print(execute(root, args.module_timeout_seconds))
        return 0
    except B19CError as error:
        print(f"B19CE_K{error.reason}-D{error.detail}_X1-C7")
        return 7
    except PermissionError as error:
        mode = "PREPARE" if args.prepare else "SELFTEST" if args.self_test else "EXECUTE"
        filename = Path(str(error.filename)).name if error.filename else "NA"
        print(f"B19CE_KPERMISSION_{mode}-D{token(filename)}_X1-C7")
        return 7
    except Exception as error:
        print(f"B19CE_KUNEXPECTED-D{token(type(error).__name__)}_X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
