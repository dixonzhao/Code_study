from __future__ import annotations

import argparse
import hashlib
import importlib.machinery
import importlib.util
from pathlib import Path
import re
import sys
from types import ModuleType
from typing import Dict, List


PROGRAM_ID = "UDP002-G2-V04-VERTICAL-RUN"
VERSION = "0.2"
BASE_REL = Path("04_TEMP/u002_g2_synthetic_vertical_run_v0_1.py.txt")
BASE_SHA256 = "5c06dee03372d7445cda6348cea8adeebbd61e01175e177f28e7d451b10bcd82"
OLD_EXECUTOR = "u002_controlled_stage_executor_v0_3.py"
NEW_EXECUTOR = "u002_controlled_stage_executor_v0_4.py"
OLD_CAPSULE = "G2P N-M3-B1-E1-R1-S8-A3-I17-X0 C-0"
RUN_CAPSULE = "G2V2 N-M3-B1-E1-R1-S8-A3-I17-X0 C-0"
SELFTEST_CAPSULE = "G2V2ST N-T16-B1-O3-I4-X0 C-0"


class WrapperError(RuntimeError):
    def __init__(self, reason: str):
        self.reason = reason
        super().__init__(reason)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def require_base(root: Path) -> Path:
    path = root / BASE_REL
    if not path.is_file() or path.is_symlink():
        raise WrapperError("BASE_MISSING")
    if sha256_file(path) != BASE_SHA256:
        raise WrapperError("BASE_IDENTITY")
    return path


def load_base(path: Path) -> ModuleType:
    module_name = "_u002_g2_frozen_base_v01"
    loader = importlib.machinery.SourceFileLoader(module_name, str(path))
    spec = importlib.util.spec_from_loader(module_name, loader)
    if spec is None:
        raise WrapperError("BASE_LOAD_SPEC")
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module


def configure(module: ModuleType) -> Dict[str, str]:
    if getattr(module, "VERSION", None) != "0.1":
        raise WrapperError("BASE_VERSION")
    original = dict(getattr(module, "PROGRAM_NAMES", {}))
    if original.get("executor") != OLD_EXECUTOR:
        raise WrapperError("BASE_EXECUTOR")
    if getattr(module, "RUN_CAPSULE", None) != OLD_CAPSULE:
        raise WrapperError("BASE_CAPSULE")
    updated = dict(original)
    updated["executor"] = NEW_EXECUTOR
    module.VERSION = VERSION
    module.PROGRAM_NAMES = updated
    module.RUN_CAPSULE = RUN_CAPSULE
    return original


def controlled_reason(exc: Exception) -> str:
    reason = getattr(exc, "reason", None)
    if not isinstance(reason, str):
        reason = exc.__class__.__name__.upper()
    reason = re.sub(r"[^A-Z0-9_]+", "_", reason.upper()).strip("_")
    return reason[:48] or "INTERNAL"


def self_test(root: Path) -> None:
    checks: List[bool] = []

    def ck(value: bool) -> None:
        if not value:
            raise AssertionError("SELFTEST")
        checks.append(True)

    base_path = require_base(root)
    before_hash = sha256_file(base_path)
    ck(base_path.name.endswith("_v0_1.py.txt"))
    ck(before_hash == BASE_SHA256)
    module = load_base(base_path)
    ck(getattr(module, "VERSION", None) == "0.1")
    ck(module.PROGRAM_NAMES.get("executor") == OLD_EXECUTOR)
    ck(getattr(module, "RUN_CAPSULE", None) == OLD_CAPSULE)
    ck(len(module.PROGRAM_NAMES) == 8)
    original = configure(module)
    ck(module.VERSION == VERSION)
    ck(module.PROGRAM_NAMES["executor"] == NEW_EXECUTOR)
    ck(module.RUN_CAPSULE == RUN_CAPSULE)
    ck(all(module.PROGRAM_NAMES[key] == value for key, value in original.items() if key != "executor"))
    ck((root / "01_PROGRAMS" / NEW_EXECUTOR).is_file())
    ck((root / "01_PROGRAMS" / OLD_EXECUTOR).is_file())
    ck(callable(getattr(module, "run_vertical", None)))
    ck(hasattr(module, "G2Error"))
    ck(RUN_CAPSULE.endswith("C-0"))
    ck(sha256_file(base_path) == before_hash)
    if len(checks) != 16:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_CAPSULE)


def run(root: Path, out_dir: Path) -> None:
    base_path = require_base(root)
    before_hash = sha256_file(base_path)
    module = load_base(base_path)
    configure(module)
    capsule = module.run_vertical(root, out_dir)
    if capsule != RUN_CAPSULE:
        raise WrapperError("RUN_CAPSULE")
    if sha256_file(base_path) != before_hash:
        raise WrapperError("BASE_CHANGED")
    print(capsule)


def main() -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--run", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    parser.add_argument("--out-dir")
    args = parser.parse_args()
    try:
        root = Path(args.workspace_root).resolve()
        if not root.is_dir():
            raise WrapperError("WORKSPACE_ROOT")
        if args.self_test:
            self_test(root)
        else:
            if not args.out_dir:
                raise WrapperError("OUT_DIR")
            run(root, Path(args.out_dir).resolve())
        return 0
    except Exception as exc:
        print(f"G2V2 N-M0-B0-E0-R0-S0-A0-I0-X1 C-7 K-{controlled_reason(exc)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
