class InstallError(RuntimeError):
    def __init__(self, reason: str):
        self.reason = reason
        super().__init__(reason)


def fail(reason: str) -> None:
    raise InstallError(reason)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as handle:
        for block in iter(lambda: handle.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def require_regular_file(path: Path, reason: str) -> Path:
    if not path.is_file() or path.is_symlink():
        fail(reason)
    return path


def source_import_roots(text: str) -> Set[str]:
    try:
        tree = ast.parse(text)
    except SyntaxError:
        fail("CANDIDATE_SYNTAX")
    roots: Set[str] = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            roots.update(alias.name.split(".", 1)[0] for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            roots.add(node.module.split(".", 1)[0])
    return roots


def audit_candidate(candidate: Path, baseline: Path) -> Dict[str, object]:
    candidate_text = candidate.read_text(encoding="utf-8-sig")
    baseline_text = baseline.read_text(encoding="utf-8-sig")
    candidate_imports = source_import_roots(candidate_text)
    baseline_imports = source_import_roots(baseline_text)
    standard_library = set(getattr(sys, "stdlib_module_names", ()))
    unapproved = sorted(candidate_imports - baseline_imports - standard_library)
    if unapproved:
        fail("NEW_EXTERNAL_IMPORT")
    required_tokens = ("ExecutionOrder", "BlockingIssues", "Status", "rsplit")
    missing = [token for token in required_tokens if token not in candidate_text]
    if missing:
        fail("V04_SEMANTICS_MISSING")
    if "0.4" not in candidate_text:
        fail("V04_IDENTITY_MISSING")
    candidate_hash = sha256_file(candidate)
    baseline_hash = sha256_file(baseline)
    if candidate_hash == baseline_hash:
        fail("CANDIDATE_EQUALS_BASELINE")
    return {
        "CandidateSha256": candidate_hash,
        "BaselineSha256": baseline_hash,
        "CandidateImports": sorted(candidate_imports),
        "BaselineImports": sorted(baseline_imports),
        "RequiredTokens": list(required_tokens),
    }


def snapshot(paths: Iterable[Path]) -> Dict[str, str]:
    return {str(path): sha256_file(path) for path in paths}


def run_child(path: Path, args: List[str], expected: str, log_prefix: Path) -> None:
    command = [sys.executable, str(path), *args]
    try:
        result = subprocess.run(
            command,
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=180,
            check=False,
        )
    except subprocess.TimeoutExpired as exc:
        log_prefix.with_suffix(".stdout.txt").write_text(exc.stdout or "", encoding="utf-8")
        log_prefix.with_suffix(".stderr.txt").write_text(exc.stderr or "", encoding="utf-8")
        fail("CHILD_TIMEOUT")
    log_prefix.with_suffix(".stdout.txt").write_text(result.stdout, encoding="utf-8")
    log_prefix.with_suffix(".stderr.txt").write_text(result.stderr, encoding="utf-8")
    lines = [line.strip() for line in result.stdout.splitlines() if line.strip()]
    if result.returncode != 0:
        fail("CHILD_EXIT")
    if expected not in lines:
        fail("CHILD_CAPSULE")

def install(workspace_root: Path) -> None:
    root = workspace_root.resolve()
    if not root.is_dir():
        fail("WORKSPACE_ROOT")
    candidate = require_regular_file(root / CANDIDATE_REL, "CANDIDATE_MISSING")
    baseline = require_regular_file(root / BASELINE_REL, "BASELINE_MISSING")
    destination = root / DESTINATION_REL
    protected = [require_regular_file(root / rel, "DEPENDENCY_MISSING") for rel in PROTECTED_RELS]
    audit = audit_candidate(candidate, baseline)
    before = snapshot(protected)
    final_dir = root / OUTPUT_REL
    staging_dir = final_dir.with_name(final_dir.name + ".staging")
    if final_dir.exists() or staging_dir.exists():
        fail("OUTPUT_EXISTS")
    staging_dir.mkdir(parents=True)
    created_destination = False
    try:
        run_child(candidate, dependency_args(root), EXPECTED_DEPENDENCY, staging_dir / "candidate_dependency")
        run_child(candidate, selftest_args(root), EXPECTED_SELFTEST, staging_dir / "candidate_selftest")
        if destination.exists():
            require_regular_file(destination, "DESTINATION_CONFLICT")
            if sha256_file(destination) != str(audit["CandidateSha256"]):
                fail("DESTINATION_CONFLICT")
        else:
            destination.parent.mkdir(parents=True, exist_ok=True)
            temporary = destination.with_name(destination.name + ".installing")
            if temporary.exists():
                fail("INSTALL_TEMP_EXISTS")
            shutil.copyfile(candidate, temporary)
            os.replace(temporary, destination)
            created_destination = True
        if sha256_file(destination) != str(audit["CandidateSha256"]):
            fail("DESTINATION_HASH")
        run_child(destination, dependency_args(root), EXPECTED_DEPENDENCY, staging_dir / "installed_dependency")
        run_child(destination, selftest_args(root), EXPECTED_SELFTEST, staging_dir / "installed_selftest")
        after = snapshot(protected)
        if after != before:
            fail("PROTECTED_INPUT_CHANGED")
        evidence = {
            "ProgramId": PROGRAM_ID,
            "Version": VERSION,
            "GateStatus": "PASS",
            "WorkspaceRoot": str(root),
            "Candidate": str(candidate),
            "Baseline": str(baseline),
            "Destination": str(destination),
            "Audit": audit,
            "DestinationSha256": sha256_file(destination),
            "DependencyCapsule": EXPECTED_DEPENDENCY,
            "SelfTestCapsule": EXPECTED_SELFTEST,
            "ProtectedInputCount": len(protected),
            "ProtectedInputsUnchanged": True,
            "SuccessCapsule": SUCCESS_CAPSULE,
        }
        (staging_dir / "g2f2_executor_v04_install_acceptance.json").write_text(
            json.dumps(evidence, indent=2, ensure_ascii=False) + "\n", encoding="utf-8"
        )
        (staging_dir / "g2f2_return_capsule.txt").write_text(SUCCESS_CAPSULE + "\n", encoding="utf-8")
        os.replace(staging_dir, final_dir)
    except Exception:
        if created_destination and destination.exists():
            try:
                destination.chmod(0o644)
                destination.unlink()
            except OSError:
                pass
        raise
    print(SUCCESS_CAPSULE)


def self_test() -> None:
    checks: List[bool] = []

    def ck(value: bool) -> None:
        if not value:
            raise AssertionError("SELFTEST")
        checks.append(True)

    with tempfile.TemporaryDirectory() as temp_name:
        temp = Path(temp_name)
        baseline = temp / "baseline.py"
        candidate = temp / "candidate.py"
        baseline.write_text("import json\nfrom pathlib import Path\nVERSION='0.3'\n", encoding="utf-8")
        candidate.write_text(
            "import json\nfrom pathlib import Path\nVERSION='0.4'\n"
            "def f(v):\n return v.rsplit('@',1), 'ExecutionOrder', 'BlockingIssues', 'Status'\n",
            encoding="utf-8",
        )
        audit = audit_candidate(candidate, baseline)
        ck(len(audit["CandidateSha256"]) == 64)
        ck(len(audit["BaselineSha256"]) == 64)
        ck(audit["CandidateSha256"] != audit["BaselineSha256"])
        ck("json" in audit["CandidateImports"])
        ck("pathlib" in audit["CandidateImports"])
        snap1 = snapshot([baseline, candidate])
        ck(len(snap1) == 2)
        ck(all(len(value) == 64 for value in snap1.values()))
        ck(source_import_roots(candidate.read_text(encoding="utf-8")) == {"json", "pathlib"})
        ck(CANDIDATE_REL.name.endswith(".py.txt"))
        ck(DESTINATION_REL.name.endswith(".py"))
        ck(BASELINE_REL.name.endswith("_v0_3.py"))
        ck(OUTPUT_REL.name.startswith("U21G_G2F2"))
        ck(len(PROTECTED_RELS) == 8)
        ck(EXPECTED_DEPENDENCY.endswith("C-0"))
        ck(EXPECTED_SELFTEST.endswith("C-0"))
        ck(SUCCESS_CAPSULE.endswith("C-0"))
        ck("X0" in SELFTEST_CAPSULE)
        ck(PROGRAM_ID.startswith("UDP002"))
        ck(VERSION == "0.1")
        ck(snapshot([baseline, candidate]) == snap1)

    with tempfile.TemporaryDirectory() as temp_name:
        root = Path(temp_name)
        (root / "04_TEMP").mkdir()
        (root / "01_PROGRAMS").mkdir()
        (root / "05_TESTS/02_ACCEPTANCE/U21F2A_PROMOTION_V02").mkdir(parents=True)
        baseline = root / BASELINE_REL
        candidate = root / CANDIDATE_REL
        baseline.write_text("import sys\nVERSION='0.3'\n", encoding="utf-8")
        candidate.write_text(
            "import sys\nVERSION='0.4'\n"
            "def contract(v):\n return v.rsplit('@',1), 'ExecutionOrder', 'BlockingIssues', 'Status'\n"
            f"print('{EXPECTED_DEPENDENCY}' if '--dependency-preflight' in sys.argv else '{EXPECTED_SELFTEST}')\n",
            encoding="utf-8",
        )
        for rel in PROTECTED_RELS[2:]:
            path = root / rel
            path.parent.mkdir(parents=True, exist_ok=True)
            path.write_text(str(rel) + "\n", encoding="utf-8")
        captured = io.StringIO()
        with contextlib.redirect_stdout(captured):
            install(root)
        destination = root / DESTINATION_REL
        final_dir = root / OUTPUT_REL
        ck(captured.getvalue().strip() == SUCCESS_CAPSULE)
        ck(destination.is_file())
        ck(sha256_file(destination) == sha256_file(candidate))
        ck(baseline.read_text(encoding="utf-8") == "import sys\nVERSION='0.3'\n")
        ck((final_dir / "g2f2_executor_v04_install_acceptance.json").is_file())
        ck((final_dir / "g2f2_return_capsule.txt").read_text(encoding="utf-8").strip() == SUCCESS_CAPSULE)
        ck(len(list(final_dir.glob("*.stdout.txt"))) == 4)
        ck(not final_dir.with_name(final_dir.name + ".staging").exists())
    if len(checks) != 28:
        raise AssertionError("SELFTEST_COUNT")
    print(SELFTEST_CAPSULE)


def main() -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--install", action="store_true")
    parser.add_argument("--workspace-root")
    args = parser.parse_args()
    try:
        if args.self_test:
            self_test()
        else:
            if not args.workspace_root:
                fail("WORKSPACE_ROOT")
            install(Path(args.workspace_root))
        return 0
    except InstallError as exc:
        print(f"G2F2 N-A0-D0-S0-H0-R0-I0-X1 C-7 K-{exc.reason}")
        return 7
    except Exception:
        print("G2F2 N-A0-D0-S0-H0-R0-I0-X1 C-7 K-INTERNAL")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
