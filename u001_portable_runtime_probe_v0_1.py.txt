#!/usr/bin/env python3
"""Read-only probe for building a cross-PC U001 portable runtime."""

from __future__ import annotations

import argparse
import importlib.metadata
import importlib.util
import json
import os
import site
import subprocess
import sys
import sysconfig
import tempfile
from datetime import datetime
from pathlib import Path
from typing import Sequence


PROGRAM_NAME = "u001_portable_runtime_probe_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "U001-PORTABLE-RUNTIME-PROBE-SELFTEST-PASS-16"
REPORT_SCHEMA = "UDP001_PACKAGING_PREFLIGHT_V0_1"


class ProbeError(RuntimeError):
    pass


def atomic_write(path: Path, data: bytes) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(data)
    temporary.replace(path)


def atomic_json(path: Path, value: dict) -> None:
    atomic_write(
        path,
        (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode("utf-8"),
    )


def load_preflight(path: Path) -> dict:
    if not path.is_file():
        raise ProbeError("PREFLIGHT_REPORT_NOT_FOUND")
    try:
        report = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, json.JSONDecodeError) as exc:
        raise ProbeError("PREFLIGHT_REPORT_INVALID") from exc
    if report.get("Schema") != REPORT_SCHEMA:
        raise ProbeError("PREFLIGHT_SCHEMA_MISMATCH")
    if int(report.get("Summary", {}).get("BlockingErrorCount", 1)) != 0:
        raise ProbeError("PREFLIGHT_NOT_ACCEPTED")
    return report


def scan_tree(root: Path) -> dict:
    files = 0
    bytes_total = 0
    errors: list[str] = []
    if not root.is_dir():
        return {"Exists": False, "Files": 0, "Bytes": 0, "Errors": ["NOT_A_DIRECTORY"]}
    for folder, dirnames, filenames in os.walk(root, followlinks=False):
        dirnames[:] = [name for name in dirnames if name not in {"__pycache__", ".git"}]
        base = Path(folder)
        for name in filenames:
            path = base / name
            try:
                if path.is_symlink():
                    continue
                bytes_total += path.stat().st_size
                files += 1
            except OSError as exc:
                if len(errors) < 10:
                    errors.append(f"{path.name}:{type(exc).__name__}")
    return {"Exists": True, "Files": files, "Bytes": bytes_total, "Errors": errors}


def locate_site_packages(prefix: Path) -> Path | None:
    candidates: list[Path] = []
    try:
        candidates.extend(Path(value) for value in site.getsitepackages())
    except AttributeError:
        pass
    purelib = sysconfig.get_path("purelib")
    if purelib:
        candidates.append(Path(purelib))
    candidates.extend((prefix / "Lib" / "site-packages", prefix / "lib" / "site-packages"))
    seen: set[str] = set()
    for candidate in candidates:
        key = os.path.normcase(str(candidate))
        if key in seen:
            continue
        seen.add(key)
        if candidate.is_dir():
            return candidate.resolve()
    return None


def distribution_names(import_roots: list[str]) -> dict[str, list[str]]:
    if hasattr(importlib.metadata, "packages_distributions"):
        mapping = importlib.metadata.packages_distributions()
    else:
        mapping: dict[str, list[str]] = {}
        for distribution in importlib.metadata.distributions():
            name = distribution.metadata.get("Name", "")
            top_level = distribution.read_text("top_level.txt") or ""
            for package in (line.strip() for line in top_level.splitlines()):
                if package and name:
                    mapping.setdefault(package, []).append(name)
    return {name: sorted(mapping.get(name, [])) for name in import_roots}


def imports_available(import_roots: list[str]) -> bool:
    if not import_roots:
        return True
    statement = ";".join(f"import {name}" for name in import_roots)
    try:
        result = subprocess.run(
            [sys.executable, "-c", statement],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            timeout=60,
            shell=False,
        )
        return result.returncode == 0
    except (OSError, subprocess.SubprocessError):
        return False


def base_tkinter_available(base_python: Path) -> bool:
    if not base_python.is_file():
        return False
    try:
        result = subprocess.run(
            [str(base_python), "-c", "import tkinter;print(tkinter.TclVersion)"],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            timeout=30,
            shell=False,
        )
        return result.returncode == 0
    except (OSError, subprocess.SubprocessError):
        return False


def tcl_library_path() -> str:
    try:
        import tkinter

        interpreter = tkinter.Tcl()
        return str(interpreter.eval("info library"))
    except Exception:
        return ""


def environment_report(preflight: dict) -> dict:
    prefix = Path(sys.prefix).resolve()
    base = Path(sys.base_prefix).resolve()
    is_venv = os.path.normcase(str(prefix)) != os.path.normcase(str(base))
    base_python = base / "python.exe"
    base_pythonw = base / "pythonw.exe"
    lib = base / "Lib"
    dlls = base / "DLLs"
    tcl = base / "tcl"
    site_packages = locate_site_packages(prefix)
    dependencies = sorted(set(preflight.get("Summary", {}).get("ExternalImports", [])))
    if not all(isinstance(name, str) and name for name in dependencies):
        raise ProbeError("EXTERNAL_IMPORT_LIST_INVALID")
    base_scan = scan_tree(base)
    site_scan = scan_tree(site_packages) if site_packages else {
        "Exists": False,
        "Files": 0,
        "Bytes": 0,
        "Errors": ["SITE_PACKAGES_NOT_FOUND"],
    }
    dependency_ok = imports_available(dependencies)
    base_tk_ok = base_tkinter_available(base_python)
    dll_candidates = sorted(
        path.name
        for pattern in ("python*.dll", "vcruntime*.dll")
        for path in base.glob(pattern)
        if path.is_file()
    )
    flags = {
        "Venv": is_venv,
        "BasePrefix": base.is_dir(),
        "BasePython": base_python.is_file(),
        "BasePythonw": base_pythonw.is_file(),
        "Lib": lib.is_dir(),
        "DLLs": dlls.is_dir(),
        "Tcl": tcl.is_dir(),
        "SitePackages": bool(site_packages and site_packages.is_dir()),
        "ExternalImportsAvailable": dependency_ok,
        "BaseTkinterAvailable": base_tk_ok,
        "RuntimeDLLCandidates": bool(dll_candidates),
        "BaseScanClean": not base_scan["Errors"],
        "SiteScanClean": not site_scan["Errors"],
    }
    errors = sum(
        not flags[name]
        for name in (
            "Venv",
            "BasePrefix",
            "BasePython",
            "BasePythonw",
            "Lib",
            "DLLs",
            "Tcl",
            "SitePackages",
            "ExternalImportsAvailable",
            "BaseTkinterAvailable",
            "RuntimeDLLCandidates",
            "BaseScanClean",
            "SiteScanClean",
        )
    )
    mib = 1024 * 1024
    base_mib = (int(base_scan["Bytes"]) + mib - 1) // mib
    site_mib = (int(site_scan["Bytes"]) + mib - 1) // mib
    dependency_text = ",".join(dependencies) if dependencies else "NONE"
    capsule = (
        "U001PK03P1\n"
        f"N-V{int(flags['Venv'])}-B{int(flags['BasePrefix'])}-P{int(flags['BasePython'])}-"
        f"W{int(flags['BasePythonw'])}-L{int(flags['Lib'])}-D{int(flags['DLLs'])}-"
        f"T{int(flags['Tcl'])}-S{int(flags['SitePackages'])}-E{int(flags['ExternalImportsAvailable'])}-"
        f"K{int(flags['BaseTkinterAvailable'])}-R{int(flags['RuntimeDLLCandidates'])}-X{errors}\n"
        f"Z-B{base_mib}-S{site_mib}-F{int(base_scan['Files']) + int(site_scan['Files'])}\n"
        f"D-{dependency_text}\n"
        f"C-{0 if errors == 0 else 7}\n"
    )
    return {
        "Schema": "UDP001_PORTABLE_RUNTIME_PROBE_V0_1",
        "ProgramName": PROGRAM_NAME,
        "ProgramVersion": PROGRAM_VERSION,
        "CreatedLocal": datetime.now().isoformat(timespec="seconds"),
        "Environment": {
            "Prefix": str(prefix),
            "BasePrefix": str(base),
            "BasePython": str(base_python),
            "BasePythonw": str(base_pythonw),
            "Lib": str(lib),
            "DLLs": str(dlls),
            "Tcl": str(tcl),
            "SitePackages": str(site_packages) if site_packages else "",
            "TclLibrary": tcl_library_path(),
            "RuntimeDLLCandidates": dll_candidates,
            "PythonVersion": sys.version,
        },
        "ExternalImports": dependencies,
        "Distributions": distribution_names(dependencies),
        "Flags": flags,
        "BaseTree": base_scan,
        "SitePackagesTree": site_scan,
        "BlockingErrorCount": errors,
        "Capsule": capsule.strip().splitlines(),
    }


def ensure_output(path: Path) -> None:
    if path.exists() and (not path.is_dir() or any(path.iterdir())):
        raise ProbeError("OUTPUT_MUST_BE_ABSENT_OR_EMPTY")
    path.mkdir(parents=True, exist_ok=True)


def run_inspect(preflight_path: Path, out: Path) -> int:
    ensure_output(out)
    report = environment_report(load_preflight(preflight_path))
    capsule = "\n".join(report["Capsule"]) + "\n"
    atomic_json(out / "u001_portable_runtime_probe_report.json", report)
    atomic_write(out / "u001_portable_runtime_probe_capsule.txt", capsule.encode("utf-8"))
    print(capsule, end="")
    return 0 if int(report["BlockingErrorCount"]) == 0 else 7


def run_self_test() -> None:
    passed = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td)
        tree = root / "tree"
        tree.mkdir()
        (tree / "a").write_bytes(b"123")
        (tree / "b").write_bytes(b"4567")
        scan = scan_tree(tree)
        passed += int(scan["Exists"])
        passed += int(scan["Files"] == 2)
        passed += int(scan["Bytes"] == 7)
        passed += int(scan["Errors"] == [])
        missing = scan_tree(root / "missing")
        passed += int(not missing["Exists"])
        passed += int(imports_available(["json"]))
        passed += int(distribution_names([]) == {})
        report = {
            "Schema": REPORT_SCHEMA,
            "Summary": {"BlockingErrorCount": 0, "ExternalImports": ["json"]},
        }
        report_path = root / "preflight.json"
        report_path.write_text(json.dumps(report), encoding="utf-8")
        passed += int(load_preflight(report_path)["Schema"] == REPORT_SCHEMA)
        output = root / "out"
        ensure_output(output)
        passed += int(output.is_dir())
        (output / "x").write_text("x", encoding="utf-8")
        try:
            ensure_output(output)
        except ProbeError:
            passed += 1
        atomic_json(root / "a.json", {"x": 1})
        passed += int(json.loads((root / "a.json").read_text(encoding="utf-8"))["x"] == 1)
        passed += int(locate_site_packages(Path(sys.prefix)) is not None)
        passed += int(bool(tcl_library_path()))
        passed += int(PROGRAM_NAME.endswith("v0_1.py"))
        passed += int(PROGRAM_VERSION == "0.1")
        passed += int(REPORT_SCHEMA.endswith("V0_1"))
    if passed != 16:
        raise AssertionError(f"SELFTEST_COUNT:{passed}")
    print(SELFTEST_TOKEN)


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="U001 portable runtime probe")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--inspect", action="store_true")
    parser.add_argument("--preflight-report")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if args.inspect:
        if not args.preflight_report or not args.out:
            parser.error("--inspect requires --preflight-report --out")
        try:
            return run_inspect(Path(args.preflight_report), Path(args.out))
        except ProbeError as exc:
            print(f"U001PK03P1-ERROR:{exc}")
            print("C-7")
            return 7
    parser.error("select --self-test or --inspect")
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
