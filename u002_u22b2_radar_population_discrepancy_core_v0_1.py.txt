#!/usr/bin/env python3
"""Pure B2 kernels for Radar validity, population and signed discrepancy.

This executable is deliberately independent of the U22 Module envelope.  It
freezes and tests the mathematical/semantic core before the same functions are
installed as S1, S2 and S3 Modules.  Standard library only.
"""

from __future__ import annotations

import argparse
import csv
import json
import math
import os
from pathlib import Path
import sys
import tempfile
from typing import Dict, Iterable, List, Mapping, Optional, Sequence, Tuple


PROGRAM_ID = "UDP002-U22B2-RADAR-POPULATION-DISCREPANCY-CORE"
PROGRAM_VERSION = "0.1"
OUTCOME_DIRECTION = "RADAR_MINUS_CAMERA"
OUTCOME_UNIT = "m"

FORMAL = "FORMAL_ELIGIBLE"
INVESTIGATION = "INVESTIGATION_ONLY"
INELIGIBLE = "INELIGIBLE"
CONFIDENT_CAMERA_STATUSES = {"SEEDED", "CONFIDENT"}
INVESTIGATION_CAMERA_STATUSES = {
    "ACQUIRING",
    "SWITCH_PENDING",
    "AMBIGUOUS",
    "SEGMENT_RESET",
}

INPUT_COLUMNS = (
    "FrameCount",
    "RadarDistance",
    "RadarDistanceValid",
    "RadarDistanceMissingReason",
    "RadarDetectionFlag",
    "RadarDetectionFlagValid",
    "CanonicalCameraDistance",
    "CanonicalCameraValid",
    "CameraSelectionStatus",
    "CameraSelectionReason",
)

OUTPUT_COLUMNS = (
    "FrameCount",
    "RadarDistance",
    "RadarDetectionFlag",
    "RadarMeasurementValid",
    "RadarValidityReason",
    "CanonicalCameraDistance",
    "CanonicalCameraValid",
    "CameraSelectionStatus",
    "ComparisonEligibility",
    "EligibilityReason",
    "CameraRadarDiscrepancy",
    "OutcomeValid",
    "OutcomeDirection",
    "OutcomeUnit",
)


class CoreError(RuntimeError):
    """Represent one deterministic semantic-core failure."""


def parse_boolean(value: object, field: str) -> bool:
    """Parse one explicit controlled boolean."""

    text = str(value if value is not None else "").strip().lower()
    if text in {"1", "true"}:
        return True
    if text in {"0", "false"}:
        return False
    raise CoreError("BOOLEAN_" + field)


def parse_frame(value: object) -> int:
    """Parse a finite integer-valued FrameCount."""

    try:
        number = float(str(value).strip())
    except Exception as exc:
        raise CoreError("FRAMECOUNT") from exc
    if not math.isfinite(number) or not number.is_integer():
        raise CoreError("FRAMECOUNT")
    return int(number)


def optional_float(value: object) -> Optional[float]:
    """Return one finite float, or None for blank/nonfinite input."""

    text = str(value if value is not None else "").strip()
    if not text:
        return None
    try:
        number = float(text)
    except ValueError:
        return None
    return number if math.isfinite(number) else None


def radar_validity(row: Mapping[str, object]) -> Dict[str, object]:
    """S1 kernel: preserve U001 validity and interpret the detection flag."""

    distance_valid = parse_boolean(row.get("RadarDistanceValid"), "RADAR_DISTANCE_VALID")
    flag_valid = parse_boolean(row.get("RadarDetectionFlagValid"), "RADAR_FLAG_VALID")
    distance = optional_float(row.get("RadarDistance"))

    if not distance_valid:
        reason = str(row.get("RadarDistanceMissingReason", "")).strip()
        return {
            "RadarDistance": distance,
            "RadarDetectionFlag": None,
            "RadarMeasurementValid": False,
            "RadarValidityReason": "UPSTREAM_DISTANCE_INVALID" + (":" + reason if reason else ""),
            "RadarEvidenceClass": "INVALID_MEASUREMENT",
        }
    if distance is None:
        return {
            "RadarDistance": None,
            "RadarDetectionFlag": None,
            "RadarMeasurementValid": False,
            "RadarValidityReason": "DISTANCE_NONFINITE_WITH_VALID_FLAG",
            "RadarEvidenceClass": "CONTRACT_VIOLATION",
        }
    if not flag_valid:
        return {
            "RadarDistance": distance,
            "RadarDetectionFlag": None,
            "RadarMeasurementValid": False,
            "RadarValidityReason": "DETECTION_FLAG_INVALID",
            "RadarEvidenceClass": "INVALID_STATUS",
        }

    detected = parse_boolean(row.get("RadarDetectionFlag"), "RADAR_DETECTION_FLAG")
    if not detected:
        return {
            "RadarDistance": distance,
            "RadarDetectionFlag": False,
            "RadarMeasurementValid": False,
            "RadarValidityReason": "TARGET_NOT_DETECTED",
            "RadarEvidenceClass": "TARGET_LOSS_EVIDENCE",
        }
    return {
        "RadarDistance": distance,
        "RadarDetectionFlag": True,
        "RadarMeasurementValid": True,
        "RadarValidityReason": "VALID_MEASUREMENT",
        "RadarEvidenceClass": "MEASUREMENT",
    }


def camera_state(row: Mapping[str, object]) -> Dict[str, object]:
    """Read the accepted S1 Camera state without changing its decision."""

    valid = parse_boolean(row.get("CanonicalCameraValid"), "CAMERA_VALID")
    distance = optional_float(row.get("CanonicalCameraDistance"))
    status = str(row.get("CameraSelectionStatus", "")).strip().upper()
    reason = str(row.get("CameraSelectionReason", "")).strip()

    if valid and distance is None:
        return {
            "CameraDistance": None,
            "CameraValid": False,
            "CameraStatus": status,
            "CameraReason": "NONFINITE_WITH_VALID_FLAG",
            "CameraContractViolation": True,
        }
    if valid and status not in CONFIDENT_CAMERA_STATUSES:
        return {
            "CameraDistance": distance,
            "CameraValid": False,
            "CameraStatus": status,
            "CameraReason": "VALID_STATUS_CONTRADICTION",
            "CameraContractViolation": True,
        }
    return {
        "CameraDistance": distance,
        "CameraValid": valid,
        "CameraStatus": status,
        "CameraReason": reason,
        "CameraContractViolation": False,
    }


def comparison_population(
    radar: Mapping[str, object],
    camera: Mapping[str, object],
) -> Tuple[str, str]:
    """S2 kernel: assign one explicit analysis-eligibility lane."""

    if camera["CameraContractViolation"] or radar["RadarEvidenceClass"] == "CONTRACT_VIOLATION":
        return INELIGIBLE, "CONTRACT_VIOLATION"

    if camera["CameraValid"] and radar["RadarMeasurementValid"]:
        return FORMAL, "BOTH_MEASUREMENTS_VALID"

    if camera["CameraValid"] and radar["RadarEvidenceClass"] == "TARGET_LOSS_EVIDENCE":
        return INVESTIGATION, "RADAR_TARGET_LOSS_WITH_CAMERA_TRACK"

    if (
        camera["CameraStatus"] in INVESTIGATION_CAMERA_STATUSES
        and (radar["RadarMeasurementValid"] or radar["RadarEvidenceClass"] == "TARGET_LOSS_EVIDENCE")
    ):
        return INVESTIGATION, "CAMERA_SELECTION_UNCERTAIN"

    if camera["CameraValid"] and radar["RadarEvidenceClass"] in {
        "INVALID_MEASUREMENT",
        "INVALID_STATUS",
    }:
        return INVESTIGATION, "RADAR_DATA_INVALID_WITH_CAMERA_TRACK"

    if radar["RadarMeasurementValid"] and not camera["CameraValid"]:
        return INVESTIGATION, "CAMERA_NOT_FORMAL_WITH_RADAR_MEASUREMENT"

    return INELIGIBLE, "NO_COMPARABLE_MEASUREMENT_PAIR"


def signed_discrepancy(
    radar: Mapping[str, object],
    camera: Mapping[str, object],
    eligibility: str,
) -> Tuple[Optional[float], bool]:
    """S3 kernel: calculate Radar minus Camera for formal rows only."""

    if eligibility != FORMAL:
        return None, False
    radar_distance = radar["RadarDistance"]
    camera_distance = camera["CameraDistance"]
    if not isinstance(radar_distance, (float, int)) or not isinstance(camera_distance, (float, int)):
        raise CoreError("FORMAL_VALUE_MISSING")
    return float(radar_distance) - float(camera_distance), True


def evaluate_row(row: Mapping[str, object]) -> Dict[str, object]:
    """Apply the ordered S1 -> S2 -> S3 pure kernels to one Frame."""

    frame = parse_frame(row.get("FrameCount"))
    radar = radar_validity(row)
    camera = camera_state(row)
    eligibility, eligibility_reason = comparison_population(radar, camera)
    discrepancy, outcome_valid = signed_discrepancy(radar, camera, eligibility)
    return {
        "FrameCount": frame,
        "RadarDistance": radar["RadarDistance"],
        "RadarDetectionFlag": radar["RadarDetectionFlag"],
        "RadarMeasurementValid": radar["RadarMeasurementValid"],
        "RadarValidityReason": radar["RadarValidityReason"],
        "CanonicalCameraDistance": camera["CameraDistance"],
        "CanonicalCameraValid": camera["CameraValid"],
        "CameraSelectionStatus": camera["CameraStatus"],
        "ComparisonEligibility": eligibility,
        "EligibilityReason": eligibility_reason,
        "CameraRadarDiscrepancy": discrepancy,
        "OutcomeValid": outcome_valid,
        "OutcomeDirection": OUTCOME_DIRECTION,
        "OutcomeUnit": OUTCOME_UNIT,
    }


def evaluate_rows(rows: Iterable[Mapping[str, object]]) -> Tuple[List[Dict[str, object]], Dict[str, object]]:
    """Evaluate an ordered Frame stream and return transparent population counts."""

    outputs: List[Dict[str, object]] = []
    previous: Optional[int] = None
    for source in rows:
        result = evaluate_row(source)
        frame = int(result["FrameCount"])
        if previous is not None and frame <= previous:
            raise CoreError("FRAME_ORDER")
        previous = frame
        outputs.append(result)
    if not outputs:
        raise CoreError("EMPTY_INPUT")

    lane_counts = {FORMAL: 0, INVESTIGATION: 0, INELIGIBLE: 0}
    outcome_count = 0
    target_loss_count = 0
    for row in outputs:
        lane_counts[str(row["ComparisonEligibility"])] += 1
        outcome_count += bool(row["OutcomeValid"])
        target_loss_count += row["RadarValidityReason"] == "TARGET_NOT_DETECTED"
    summary = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "InputRows": len(outputs),
        "FormalEligibleRows": lane_counts[FORMAL],
        "InvestigationOnlyRows": lane_counts[INVESTIGATION],
        "IneligibleRows": lane_counts[INELIGIBLE],
        "OutcomeValidRows": outcome_count,
        "RadarTargetLossEvidenceRows": target_loss_count,
        "OutcomeFormula": "RadarDistance-CanonicalCameraDistance",
        "OutcomeDirection": OUTCOME_DIRECTION,
        "OutcomeUnit": OUTCOME_UNIT,
        "RadarIsGroundTruth": False,
        "BaselineApplied": False,
        "ZeroFillApplied": False,
    }
    return outputs, summary


def write_csv(path: Path, rows: Sequence[Mapping[str, object]]) -> None:
    """Write the transparent kernel output."""

    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(OUTPUT_COLUMNS), lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({name: "" if row.get(name) is None else row.get(name) for name in OUTPUT_COLUMNS})
    os.replace(temporary, path)


def write_json(path: Path, value: object) -> None:
    """Write deterministic summary JSON."""

    temporary = path.with_name(path.name + ".tmp")
    temporary.write_text(
        json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2, allow_nan=False) + "\n",
        encoding="utf-8",
    )
    os.replace(temporary, path)


def fixture_rows() -> List[Dict[str, object]]:
    """Return synthetic Frames covering formal, loss, ambiguous and invalid lanes."""

    base: Dict[str, object] = {
        "RadarDistance": 12.0,
        "RadarDistanceValid": 1,
        "RadarDistanceMissingReason": "",
        "RadarDetectionFlag": 1,
        "RadarDetectionFlagValid": 1,
        "CanonicalCameraDistance": 10.0,
        "CanonicalCameraValid": 1,
        "CameraSelectionStatus": "CONFIDENT",
        "CameraSelectionReason": "CONTINUITY_CONFIRMED",
    }
    rows = []
    for frame in range(100, 108):
        row = dict(base)
        row["FrameCount"] = frame
        rows.append(row)
    rows[1]["RadarDetectionFlag"] = 0
    rows[1]["RadarDistance"] = 0
    rows[1]["RadarDistanceValid"] = 0
    rows[1]["RadarDistanceMissingReason"] = "INVALID_DEFAULT"
    rows[2]["CanonicalCameraDistance"] = ""
    rows[2]["CanonicalCameraValid"] = 0
    rows[2]["CameraSelectionStatus"] = "AMBIGUOUS"
    rows[3]["RadarDistanceValid"] = 0
    rows[3]["RadarDistance"] = 327.66
    rows[3]["RadarDistanceMissingReason"] = "INVALID_DEFAULT"
    rows[4]["CanonicalCameraDistance"] = ""
    rows[4]["CanonicalCameraValid"] = 0
    rows[4]["CameraSelectionStatus"] = "INVALID"
    rows[4]["RadarDistanceValid"] = 0
    rows[4]["RadarDistance"] = ""
    rows[4]["RadarDistanceMissingReason"] = "NO_MEASUREMENT"
    rows[5]["RadarDistance"] = 0.0
    rows[5]["RadarDistanceValid"] = 1
    rows[6]["CanonicalCameraValid"] = 1
    rows[6]["CameraSelectionStatus"] = "AMBIGUOUS"
    # A finite bus value can remain while the authoritative detection flag is
    # already false.  Preserve this row as target-loss evidence, never as a
    # formal Camera-Radar measurement pair.
    rows[7]["RadarDistance"] = 11.5
    rows[7]["RadarDistanceValid"] = 1
    rows[7]["RadarDetectionFlag"] = 0
    return rows


def self_test() -> int:
    """Verify ordered semantic and mathematical behavior."""

    checks: List[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            raise AssertionError(name)
        checks.append(name)

    rows = fixture_rows()
    outputs, summary = evaluate_rows(rows)
    check("T01_ROWS", len(outputs) == 8)
    check("T02_VALID_FORMAL", outputs[0]["ComparisonEligibility"] == FORMAL)
    check("T03_SIGN", outputs[0]["CameraRadarDiscrepancy"] == 2.0)
    check("T04_DIRECTION", outputs[0]["OutcomeDirection"] == OUTCOME_DIRECTION)
    check("T05_UNIT", outputs[0]["OutcomeUnit"] == "m")
    check("T06_LOSS_NOT_FORMAL", outputs[1]["ComparisonEligibility"] != FORMAL)
    check("T07_LOSS_BLANK", outputs[1]["CameraRadarDiscrepancy"] is None)
    check("T08_AMBIGUOUS", outputs[2]["ComparisonEligibility"] == INVESTIGATION)
    check("T09_INVALID_DEFAULT", outputs[3]["RadarMeasurementValid"] is False)
    check("T10_BOTH_INVALID", outputs[4]["ComparisonEligibility"] == INELIGIBLE)
    check("T11_ZERO_NOT_GLOBAL", outputs[5]["RadarMeasurementValid"] is True)
    check("T12_ZERO_FORMAL", outputs[5]["OutcomeValid"] is True)
    check("T13_CONTRADICTION", outputs[6]["ComparisonEligibility"] == INELIGIBLE)
    check("T14_COUNTS", sum(summary[name] for name in ("FormalEligibleRows", "InvestigationOnlyRows", "IneligibleRows")) == 8)
    check("T15_OUTCOME_COUNT", summary["OutcomeValidRows"] == 2)
    check("T16_NO_GROUND_TRUTH", summary["RadarIsGroundTruth"] is False)
    check("T17_NO_BASELINE", summary["BaselineApplied"] is False)
    check("T18_NO_FILL", summary["ZeroFillApplied"] is False)
    check("T19_BOOLEAN_TRUE", parse_boolean("true", "T") is True)
    check("T20_BOOLEAN_FALSE", parse_boolean("0", "T") is False)
    check("T21_FRAME", parse_frame("100.0") == 100)
    check("T22_OPTIONAL", optional_float("") is None)
    try:
        evaluate_rows([rows[1], rows[0]])
        order_failed = False
    except CoreError:
        order_failed = True
    check("T23_ORDER_FAIL", order_failed)
    with tempfile.TemporaryDirectory() as temp_name:
        root = Path(temp_name)
        write_csv(root / "result.csv", outputs)
        write_json(root / "summary.json", summary)
        check("T24_CSV", (root / "result.csv").is_file())
        check("T25_JSON", (root / "summary.json").is_file())
    check("T26_TARGET_LOSS_REASON", outputs[7]["RadarValidityReason"] == "TARGET_NOT_DETECTED")
    check("T27_TARGET_LOSS_COUNT", summary["RadarTargetLossEvidenceRows"] == 1)
    if len(checks) != 27:
        raise AssertionError("SELFTEST_COUNT")
    print("U22B2-CORE-SELFTEST-PASS-27")
    return 0


def run_demo(out_root: Path) -> int:
    """Generate one small explainable fixture result for human inspection."""

    if out_root.exists():
        raise CoreError("OUTPUT_EXISTS")
    out_root.mkdir(parents=True)
    outputs, summary = evaluate_rows(fixture_rows())
    write_csv(out_root / "u22b2_core_demo.csv", outputs)
    write_json(out_root / "u22b2_core_demo_summary.json", summary)
    print(
        f"B2A N-R{summary['InputRows']}-F{summary['FormalEligibleRows']}"
        f"-I{summary['InvestigationOnlyRows']}-X{summary['IneligibleRows']}"
        f"-O{summary['OutcomeValidRows']}-L{summary['RadarTargetLossEvidenceRows']} C-0"
    )
    return 0


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test or the synthetic demonstration."""

    parser = argparse.ArgumentParser(description="U22 B2 pure semantic kernels")
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--demo", action="store_true")
    parser.add_argument("--out-root")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            return self_test()
        if not args.out_root:
            raise CoreError("OUT_ROOT_REQUIRED")
        return run_demo(Path(args.out_root))
    except Exception as exc:
        print("B2AE K-" + str(exc).upper().replace(" ", "_")[:80] + "-X1 C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
