#!/usr/bin/env python3
"""Prepare, test and install U001 H4B cadence and bounded-hold support."""

from __future__ import annotations

import argparse
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PROGRAM_VERSION = "1.0.0"
PACKAGE_NAME = "UDP001_U1H4B_CADENCE_BOUNDED_HOLD_HIGH_CODE_v1_0.zip"
ACTIVE_NAME = "U001H_ACTIVE"
PREPARED_NAME = "U1H4B_PREPARED"

EXPECTED_FILES = {
    "01_PROGRAMS/core/u001_h4b_contract_core_v1_0.py",
    "01_PROGRAMS/core/u001_h4b_cadence_profiler_v1_0.py",
    "01_PROGRAMS/core/u001_h4b_hold_policy_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_h4b_bounded_hold_engine_v1_0.py",
    "02_CONTROL/contracts/u001_h4b_cadence_profile_schema_v1_0.json",
    "02_CONTROL/contracts/u001_h4b_hold_policy_schema_v1_0.json",
    "02_CONTROL/contracts/u001_h4b_bounded_frame_view_contract_v1_0.json",
    "05_TESTS/tools/u001_h4b_bounded_hold_acceptance_v1_0.py",
    "06_DOCS/U001_H4B_CADENCE_AND_BOUNDED_HOLD_TECHNICAL_SPEC_v1_0_2026-08-28.txt",
}

FOUNDATION_FILES = {
    "01_PROGRAMS/core/u001_h4_contract_core_v1_0.py",
    "01_PROGRAMS/core/u001_h4_frame_policy_compiler_v1_0.py",
    "01_PROGRAMS/core/u001_h4_raw_frame_view_builder_v1_0.py",
    "02_CONTROL/contracts/u001_h4_frame_policy_schema_v1_0.json",
    "02_CONTROL/contracts/u001_h4_raw_frame_view_contract_v1_0.json",
    "05_TESTS/tools/u001_h4a_raw_frame_view_acceptance_v1_1.py",
}

OBSOLETE_TRANSFER_NAMES = {
    "U1H4A_EXECUTION_RESULT.json",
    "UDP001_U1H4A_RAW_FRAME_VIEW_HIGH_CODE_v1_1.zip",
    "u001_u1h4a_raw_frame_view_installer_v1_1.py.txt",
    "U1H4A_START_HERE_CN_v1_1_2026-08-28.txt",
    "U1H4B_EXECUTION_RESULT.json",
    "U1H4B_FAILURE_DIAGNOSTIC.json",
}

LAST_CHILD_EVIDENCE: dict[str, object] = {}


class InstallError(RuntimeError):
    """Fail-closed H4B installation error."""


def timestamp() -> str:
    return datetime.now().strftime("%Y%m%dT%H%M%S_%f")


def resolve_u001_root(explicit: Path | None) -> Path:
    candidates = [explicit] if explicit is not None else []
    candidates.extend((Path.cwd() / "U-UDP001", Path.cwd()))
    for candidate in candidates:
        candidate = candidate.resolve()
        if candidate.name == "U-UDP001" and candidate.is_dir():
            return candidate
    raise InstallError("U001_ROOT_NOT_FOUND")


def temp_dir(root: Path) -> Path:
    result = root / "04_TEMP"
    if not result.is_dir():
        raise InstallError("TEMP_DIRECTORY_MISSING")
    return result


def validate_foundation(root: Path) -> None:
    missing = sorted(relative for relative in FOUNDATION_FILES if not (root / relative).is_file())
    if missing:
        raise InstallError(f"H4A_FOUNDATION_MISSING:{missing}")


def validate_zip(path: Path) -> None:
    if not path.is_file():
        raise InstallError(f"PACKAGE_MISSING:{PACKAGE_NAME}")
    with zipfile.ZipFile(path, "r") as archive:
        files = {name for name in archive.namelist() if not name.endswith("/")}
        if files != EXPECTED_FILES:
            raise InstallError(
                f"PACKAGE_INVENTORY:MISSING={sorted(EXPECTED_FILES-files)}:EXTRA={sorted(files-EXPECTED_FILES)}"
            )
        for name in files:
            pure = PurePosixPath(name)
            if pure.is_absolute() or ".." in pure.parts:
                raise InstallError(f"PACKAGE_PATH:{name}")
        bad = archive.testzip()
        if bad:
            raise InstallError(f"PACKAGE_CRC:{bad}")


def _archive(path: Path, archive_root: Path, relative: Path) -> bool:
    if not path.exists():
        return False
    target = archive_root / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(target))
    return True


def clean_temp(root: Path) -> int:
    transfer = temp_dir(root)
    archive_root = root / "A-Archive" / f"U1H4B_TEMP_HYGIENE_{timestamp()}"
    moved = 0
    for name in sorted(OBSOLETE_TRANSFER_NAMES):
        moved += int(_archive(transfer / name, archive_root, Path(name)))
    active = transfer / ACTIVE_NAME
    for name in ("U1H4A_PREPARED", PREPARED_NAME):
        moved += int(_archive(active / name, archive_root, Path(ACTIVE_NAME) / name))
    return moved


def prepare(root: Path) -> Path:
    package = temp_dir(root) / PACKAGE_NAME
    print(f"[CHECK] U001 root exists: {root}")
    print("[WHY]   H4B extends the accepted H4A boundary and must not mutate raw frame evidence.")
    validate_foundation(root)
    print(f"[CHECK] Accepted H4A foundation is complete: {len(FOUNDATION_FILES)} required files")
    print("[WHY]   Cadence and hold are allowed only after raw one-row-per-FrameCount construction is qualified.")
    validate_zip(package)
    print(f"[CHECK] H4B package inventory and CRC passed: {len(EXPECTED_FILES)} files")
    print("[WHY]   Exact inventory blocks legacy unlimited-forward-fill or stale policy code.")
    archived = clean_temp(root)
    print(f"[CLEAN] Superseded H4A/H4B transfer and prepared entries archived: {archived}")
    print("[WHY]   04_TEMP exposes one current H4B prepared source of truth.")
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package, "r") as archive:
        archive.extractall(prepared)
    extracted = {path.relative_to(prepared).as_posix() for path in prepared.rglob("*") if path.is_file()}
    if extracted != EXPECTED_FILES:
        raise InstallError("EXTRACTED_INVENTORY")
    print("[CHECK] Prepared H4B payload exactly matches the approved package.")
    print(f"U1H4BP-N{len(EXPECTED_FILES)}-A{archived}-X0-C0")
    return prepared


def child_environment(python_paths: list[Path] | None = None) -> dict[str, str]:
    environment = dict(os.environ)
    environment["PYTHONDONTWRITEBYTECODE"] = "1"
    environment["PYTHONPYCACHEPREFIX"] = str(Path(tempfile.gettempdir()) / "u001_h4b_pycache")
    if python_paths:
        inherited = environment.get("PYTHONPATH", "")
        parts = [str(path.resolve()) for path in python_paths]
        if inherited:
            parts.append(inherited)
        environment["PYTHONPATH"] = os.pathsep.join(parts)
    return environment


def run_child(
    command: list[str],
    label: str,
    timeout: int = 300,
    python_paths: list[Path] | None = None,
) -> str:
    global LAST_CHILD_EVIDENCE
    try:
        result = subprocess.run(
            command,
            text=True,
            capture_output=True,
            timeout=timeout,
            check=False,
            env=child_environment(python_paths),
        )
    except subprocess.TimeoutExpired as exc:
        LAST_CHILD_EVIDENCE = {
            "Label": label,
            "Status": "TIMEOUT",
            "TimeoutSeconds": timeout,
            "Command": command,
        }
        raise InstallError(f"{label}_TIMEOUT") from exc
    LAST_CHILD_EVIDENCE = {
        "Label": label,
        "Status": "PASS" if result.returncode == 0 else "FAIL",
        "ReturnCode": result.returncode,
        "Command": command,
        "Stdout": result.stdout,
        "Stderr": result.stderr,
    }
    if result.returncode != 0:
        raise InstallError(f"{label}_EXIT:{result.returncode}")
    output = result.stdout.strip()
    if not output:
        raise InstallError(f"{label}_NO_OUTPUT")
    return output


def write_failure_diagnostic(root: Path, exc: Exception) -> None:
    diagnostic = {
        "DiagnosticSchema": "U001_H4B_INSTALL_FAILURE_V1_0",
        "ProgramVersion": PROGRAM_VERSION,
        "ErrorType": type(exc).__name__,
        "Error": str(exc),
        "LastChildEvidence": LAST_CHILD_EVIDENCE,
        "CreatedLocal": datetime.now().isoformat(),
    }
    path = temp_dir(root) / "U1H4B_FAILURE_DIAGNOSTIC.json"
    path.write_text(json.dumps(diagnostic, indent=2) + "\n", encoding="utf-8")


def self_test(root: Path) -> None:
    validate_foundation(root)
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    if not prepared.is_dir():
        raise InstallError("PREPARED_DIRECTORY_MISSING")
    core = prepared / "01_PROGRAMS" / "core"
    component_tests = [
        (core / "u001_h4b_contract_core_v1_0.py", "PASS-14", "H4A identity and dynamic coupled-state contract."),
        (core / "u001_h4b_cadence_profiler_v1_0.py", "PASS-19", "Streaming real-FrameCount cadence profiling."),
        (core / "u001_h4b_hold_policy_compiler_v1_0.py", "PASS-21", "Engineer-approved finite hold-policy compilation."),
        (core / "u001_h4b_bounded_hold_engine_v1_0.py", "PASS-25", "Bounded hold, age, stale and invalid-update barriers."),
    ]
    total = 0
    for program, marker, explanation in component_tests:
        print(f"[CHECK] {explanation}")
        output = run_child(
            [sys.executable, str(program), "--self-test"],
            program.stem.upper(),
            python_paths=[core, root / "01_PROGRAMS" / "core"],
        )
        if marker not in output:
            raise InstallError(f"SELFTEST_MARKER:{program.name}")
        total += int(marker.rsplit("-", 1)[1])
    print("[CHECK] Independent H4B replay, state matrix, determinism, true-zero and tamper rejection.")
    acceptance = prepared / "05_TESTS" / "tools" / "u001_h4b_bounded_hold_acceptance_v1_0.py"
    with tempfile.TemporaryDirectory(prefix="u001_h4b_installer_") as temporary:
        output = run_child(
            [
                sys.executable,
                str(acceptance),
                "--validate",
                "--core-dir",
                str(core),
                "--out",
                str(Path(temporary) / "acceptance"),
                "--foundation-core-dir",
                str(root / "01_PROGRAMS" / "core"),
            ],
            "H4B_INDEPENDENT_ACCEPTANCE",
            python_paths=[core, root / "01_PROGRAMS" / "core"],
        )
    if "T36" not in output or "C0" not in output.replace("-", ""):
        raise InstallError("H4B_ACCEPTANCE_MARKER")
    print(f"U1H4BST-N5-T{total + 36}-X0-C0")


def copy_payload(prepared: Path, root: Path) -> tuple[int, Path | None]:
    archive_root: Path | None = None
    copied = 0
    for relative_text in sorted(EXPECTED_FILES):
        relative = Path(relative_text)
        source = prepared / relative
        target = root / relative
        if target.exists():
            if archive_root is None:
                archive_root = root / "A-Archive" / f"U1H4B_REPLACED_{timestamp()}"
            _archive(target, archive_root, relative)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        copied += 1
    return copied, archive_root


def execute(root: Path) -> None:
    self_test(root)
    prepared = temp_dir(root) / ACTIVE_NAME / PREPARED_NAME
    copied, replaced = copy_payload(prepared, root)
    for relative_text in sorted(EXPECTED_FILES):
        source = prepared / relative_text
        installed = root / relative_text
        if source.read_bytes() != installed.read_bytes():
            raise InstallError(f"INSTALLED_FILE_IDENTITY:{relative_text}")

    acceptance = root / "05_TESTS" / "tools" / "u001_h4b_bounded_hold_acceptance_v1_0.py"
    evidence_root = root / "05_TESTS" / "evidence" / f"U1H4B_{timestamp()}"
    evidence_root.mkdir(parents=True, exist_ok=False)
    with tempfile.TemporaryDirectory(prefix="u1h4b_post_") as temporary:
        acceptance_root = Path(temporary) / "acceptance"
        output = run_child(
            [
                sys.executable,
                str(acceptance),
                "--validate",
                "--core-dir",
                str(root / "01_PROGRAMS" / "core"),
                "--out",
                str(acceptance_root),
                "--foundation-core-dir",
                str(root / "01_PROGRAMS" / "core"),
            ],
            "INSTALLED_H4B_ACCEPTANCE",
            python_paths=[root / "01_PROGRAMS" / "core"],
        )
        if "T36" not in output or "C0" not in output.replace("-", ""):
            raise InstallError("INSTALLED_ACCEPTANCE_MARKER")
        receipt_path = acceptance_root / "u001_h4b_acceptance_receipt.json"
        capsule_path = acceptance_root / "u001_h4b_acceptance_capsule.txt"
        receipt = json.loads(receipt_path.read_text(encoding="utf-8"))
        shutil.copy2(receipt_path, evidence_root / receipt_path.name)
        shutil.copy2(capsule_path, evidence_root / capsule_path.name)
    if receipt.get("Status") != "PASS" or receipt.get("CompletionCode") != 0:
        raise InstallError("INSTALLED_ACCEPTANCE_RECEIPT")
    h4a_output = run_child(
        [sys.executable, str(root / "01_PROGRAMS" / "core" / "u001_h4_raw_frame_view_builder_v1_0.py"), "--self-test"],
        "H4A_COMPATIBILITY",
    )
    if "PASS-24" not in h4a_output:
        raise InstallError("H4A_COMPATIBILITY_MARKER")
    result = {
        "Node": "U1H4B",
        "ProgramVersion": PROGRAM_VERSION,
        "InstalledFileCount": copied,
        "InstalledProgramCount": 4,
        "CadenceProfilerQualified": True,
        "HoldPolicyCompilerQualified": True,
        "IndependentAcceptanceCount": 1,
        "H4ACompatibilityPassed": True,
        "ReplacedFilesArchived": replaced is not None,
        "AcceptanceEvidenceFolder": evidence_root.relative_to(root).as_posix(),
        "CompletionCode": 0,
    }
    (temp_dir(root) / "U1H4B_EXECUTION_RESULT.json").write_text(
        json.dumps(result, indent=2) + "\n", encoding="utf-8"
    )
    print(f"U1H4B-N4-P1-Q1-H1-A{copied}-X0-C0")


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description="Install U001 H4B cadence and bounded hold")
    action = result.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    result.add_argument("--u001-root", type=Path)
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    root: Path | None = None
    try:
        root = resolve_u001_root(args.u001_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            self_test(root)
        else:
            execute(root)
        return 0
    except (InstallError, OSError, ValueError, json.JSONDecodeError, zipfile.BadZipFile) as exc:
        if root is not None:
            try:
                write_failure_diagnostic(root, exc)
            except Exception:
                pass
        short_detail = " ".join(str(exc).split())[-120:] or "NO_DETAIL"
        print(
            f"U1H4BIE-K{type(exc).__name__.upper()}-D{short_detail}-"
            "L04_TEMP/U1H4B_FAILURE_DIAGNOSTIC.json-X1-C7",
            file=sys.stderr,
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
