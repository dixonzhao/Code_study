#!/usr/bin/env python3
"""Read-only diagnosis for a failed U001 PK03 copied-runtime validation."""

from __future__ import annotations

import argparse
import json
import os
import re
import subprocess
import tempfile
from pathlib import Path
from typing import Sequence


PROGRAM_NAME = "u001_pk03_runtime_failure_probe_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "U001-PK03-RUNTIME-FAILURE-PROBE-SELFTEST-PASS-14"
PACKAGE_NAME = "U001_Portable_v0_1"


class ProbeError(RuntimeError):
    pass


def atomic_write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(text.encode("utf-8"))
    temporary.replace(path)


def latest_build_root(udp001_root: Path) -> Path:
    candidates = sorted(
        (path for path in (udp001_root / "04_TEMP").glob("PK03_BUILD_WORK_*") if path.is_dir()),
        key=lambda path: path.stat().st_mtime,
        reverse=True,
    )
    if not candidates:
        raise ProbeError("PK03_BUILD_WORK_NOT_FOUND")
    return candidates[0]


def runtime_environment(runtime: Path) -> dict[str, str]:
    environment = os.environ.copy()
    environment.pop("VIRTUAL_ENV", None)
    environment.pop("PYTHONPATH", None)
    environment["PYTHONHOME"] = str(runtime.resolve())
    environment["PYTHONNOUSERSITE"] = "1"
    system_root = environment.get("SystemRoot", r"C:\Windows")
    environment["PATH"] = os.pathsep.join(
        (str(runtime.resolve()), str((runtime / "DLLs").resolve()), str(Path(system_root) / "System32"), system_root)
    )
    return environment


def inside(root: Path, value: str) -> bool:
    if not value:
        return False
    try:
        Path(value).resolve().relative_to(root.resolve())
        return True
    except (OSError, ValueError):
        return False


def run_test(python: Path, runtime: Path, script: str) -> dict:
    try:
        result = subprocess.run(
            [str(python), "-I", "-c", script],
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=120,
            shell=False,
            env=runtime_environment(runtime),
        )
        return {
            "ReturnCode": result.returncode,
            "Stdout": result.stdout,
            "Stderr": result.stderr,
        }
    except (OSError, subprocess.SubprocessError) as exc:
        return {"ReturnCode": 999, "Stdout": "", "Stderr": f"{type(exc).__name__}:{exc}"}


def last_nonempty(text: str) -> str:
    values = [line.strip() for line in text.splitlines() if line.strip()]
    return values[-1] if values else ""


def compact_memo(tag: str, result: dict) -> str:
    value = last_nonempty(result.get("Stderr", "")) or last_nonempty(result.get("Stdout", "")) or "NO_DETAIL"
    value = re.sub(r"\s+", "_", value)
    value = re.sub(r"[^A-Za-z0-9_.:+\\/-]", "?", value)
    return f"M-{tag}-{value[:160]}"


def test_origin(result: dict, runtime: Path) -> bool:
    return result.get("ReturnCode") == 0 and inside(runtime, last_nonempty(result.get("Stdout", "")))


def run_probe(udp001_root: Path, out: Path) -> int:
    if not udp001_root.is_dir() or udp001_root.name != "UDP001":
        raise ProbeError("UDP001_ROOT_INVALID")
    if out.exists() and (not out.is_dir() or any(out.iterdir())):
        raise ProbeError("OUTPUT_MUST_BE_ABSENT_OR_EMPTY")
    out.mkdir(parents=True, exist_ok=True)
    build = latest_build_root(udp001_root)
    package = build / "release_stage" / PACKAGE_NAME
    runtime = package / "_internal" / "runtime"
    evidence = package / "_internal" / "evidence"
    python = runtime / "python.exe"
    flags = {
        "Work": build.is_dir(),
        "Runtime": runtime.is_dir(),
        "Python": python.is_file(),
        "Lib": (runtime / "Lib").is_dir(),
        "DLLs": (runtime / "DLLs").is_dir(),
        "SitePackages": (runtime / "Lib" / "site-packages").is_dir(),
    }
    if not flags["Python"]:
        raise ProbeError("COPIED_PYTHON_NOT_FOUND")

    core_script = "import json,sys;print(json.dumps({'prefix':sys.prefix,'base_prefix':sys.base_prefix,'executable':sys.executable}))"
    tkinter_script = "import tkinter;print(tkinter.__file__)"
    pandas_script = "import pandas;print(pandas.__file__)"
    validation_script = (
        "import importlib,json,os,sys;from pathlib import Path;"
        "r=Path(sys.prefix).resolve();o={};mods=['tkinter','pandas'];"
        "[(o.__setitem__(n,str(Path(importlib.import_module(n).__file__).resolve()))) for n in mods];"
        "ok=all(os.path.normcase(os.path.commonpath([str(r),p]))==os.path.normcase(str(r)) for p in o.values());"
        "print(json.dumps({'Prefix':str(r),'Origins':o,'InsideRuntime':ok,'Executable':sys.executable}));"
        "raise SystemExit(0 if ok else 7)"
    )
    core = run_test(python, runtime, core_script)
    tkinter = run_test(python, runtime, tkinter_script)
    pandas = run_test(python, runtime, pandas_script)
    validation = run_test(python, runtime, validation_script)

    core_ok = False
    if core["ReturnCode"] == 0:
        try:
            payload = json.loads(last_nonempty(core["Stdout"]))
            core_ok = inside(runtime, payload.get("prefix", "")) and inside(runtime, payload.get("executable", ""))
        except json.JSONDecodeError:
            core_ok = False
    tkinter_ok = test_origin(tkinter, runtime)
    pandas_ok = test_origin(pandas, runtime)
    validation_ok = validation["ReturnCode"] == 0

    old_stdout_path = evidence / "runtime_validation_stdout.txt"
    old_stderr_path = evidence / "runtime_validation_stderr.txt"
    old_stdout = old_stdout_path.read_text(encoding="utf-8", errors="replace") if old_stdout_path.is_file() else ""
    old_stderr = old_stderr_path.read_text(encoding="utf-8", errors="replace") if old_stderr_path.is_file() else ""
    old_evidence = bool(old_stdout or old_stderr)

    all_flags = [*flags.values(), core_ok, tkinter_ok, pandas_ok, validation_ok]
    errors = sum(not value for value in all_flags)
    tests = {"Core": core, "Tkinter": tkinter, "Pandas": pandas, "Validation": validation}
    report = {
        "Schema": "UDP001_PK03_RUNTIME_FAILURE_PROBE_V0_1",
        "BuildRoot": str(build),
        "Runtime": str(runtime),
        "Flags": flags,
        "CoreOK": core_ok,
        "TkinterOK": tkinter_ok,
        "PandasOK": pandas_ok,
        "ValidationOK": validation_ok,
        "PreviousEvidenceFound": old_evidence,
        "PreviousStdout": old_stdout,
        "PreviousStderr": old_stderr,
        "Tests": tests,
        "BlockingCount": errors,
    }
    atomic_write(out / "u001_pk03_runtime_failure_probe_report.json", json.dumps(report, ensure_ascii=False, indent=2) + "\n")
    for name, result in tests.items():
        atomic_write(out / f"{name.lower()}_stdout.txt", result["Stdout"])
        atomic_write(out / f"{name.lower()}_stderr.txt", result["Stderr"])

    priority = (
        ("V", validation),
        ("P", pandas),
        ("T", tkinter),
        ("K", core),
    )
    memo = next((compact_memo(tag, result) for tag, result in priority if result["ReturnCode"] != 0), "M-NONE")
    capsule = (
        "U001PK03D1\n"
        f"N-W{int(flags['Work'])}-R{int(flags['Runtime'])}-E{int(flags['Python'])}-"
        f"L{int(flags['Lib'])}-D{int(flags['DLLs'])}-S{int(flags['SitePackages'])}-"
        f"K{int(core_ok)}-T{int(tkinter_ok)}-P{int(pandas_ok)}-V{int(validation_ok)}-"
        f"O{int(old_evidence)}-X{errors}\n"
        f"RC-K{core['ReturnCode']}-T{tkinter['ReturnCode']}-P{pandas['ReturnCode']}-V{validation['ReturnCode']}\n"
        f"{memo}\n"
        "C-0\n"
    )
    atomic_write(out / "u001_pk03_runtime_failure_probe_capsule.txt", capsule)
    print(capsule, end="")
    return 0


def run_self_test() -> None:
    passed = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td) / "UDP001"
        work1 = root / "04_TEMP" / "PK03_BUILD_WORK_20260727_100000"
        work2 = root / "04_TEMP" / "PK03_BUILD_WORK_20260727_110000"
        work1.mkdir(parents=True)
        work2.mkdir(parents=True)
        os.utime(work1, (1, 1))
        os.utime(work2, (2, 2))
        passed += int(latest_build_root(root) == work2)
        passed += int(inside(root, str(work2)))
        passed += int(not inside(work2, str(root)))
        environment = runtime_environment(Path(td))
        passed += int(environment["PYTHONHOME"] == str(Path(td).resolve()))
        passed += int(environment["PYTHONNOUSERSITE"] == "1")
        passed += int(last_nonempty("a\n\nb\n") == "b")
        memo = compact_memo("T", {"Stdout": "", "Stderr": "A bad error\n", "ReturnCode": 1})
        passed += int(memo == "M-T-A_bad_error")
        passed += int(PROGRAM_NAME.endswith("v0_1.py"))
        passed += int(PROGRAM_VERSION == "0.1")
        passed += int(PACKAGE_NAME == "U001_Portable_v0_1")
        passed += int(Path(sys_executable()).is_file())
        simple = run_test(Path(sys_executable()), Path(sys_prefix()), "print('ok')")
        passed += int(simple["ReturnCode"] == 0)
        passed += int(last_nonempty(simple["Stdout"]) == "ok")
        passed += int(SELFTEST_TOKEN.endswith("PASS-14"))
    if passed != 14:
        raise AssertionError(f"SELFTEST_COUNT:{passed}")
    print(SELFTEST_TOKEN)


def sys_executable() -> str:
    import sys

    return sys.executable


def sys_prefix() -> str:
    import sys

    return sys.prefix


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Diagnose a U001 PK03 copied-runtime import failure")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--probe", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if args.probe:
        if not args.udp001_root or not args.out:
            parser.error("--probe requires --udp001-root --out")
        try:
            return run_probe(Path(args.udp001_root), Path(args.out))
        except ProbeError as exc:
            print(f"U001PK03D1-ERROR:{exc}")
            print("C-7")
            return 7
    parser.error("select --self-test or --probe")
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
