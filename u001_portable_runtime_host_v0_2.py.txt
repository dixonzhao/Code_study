#!/usr/bin/env python3
"""PyInstaller runtime host for the accepted external U001 programs."""

from __future__ import annotations

import argparse
import ctypes
import json
import os
import runpy
import sys
import tempfile
from pathlib import Path
from typing import Sequence

import pandas as _embedded_pandas
import tkinter as _embedded_tkinter


PROGRAM_NAME = "u001_portable_runtime_host_v0_2.py"
PROGRAM_VERSION = "0.2"
SELFTEST_TOKEN = "U001-PORTABLE-RUNTIME-HOST-V02-SELFTEST-PASS-18"
EXE_NAME = "U001_Workbench.exe"
GUI_NAME = "u001_data_preparation_workbench_gui_v0_3.py"
ORCHESTRATOR_NAME = "u001_g8a_thin_orchestrator_v0_5.py"
EXPECTED_PROGRAMS = {
    "u001_foundation_core_v0_1.py",
    "u001_selected_update_decoder_v0_1.py",
    "u001_framecount_assignment_v0_1.py",
    "u001_semantic_validity_classifier_v0_1.py",
    "u001_framecount_view_builder_v0_1.py",
    "u001_signal_cadence_bounded_hold_v0_1.py",
    "u001_output_package_assembler_v0_3.py",
    ORCHESTRATOR_NAME,
    GUI_NAME,
}


class HostError(RuntimeError):
    pass


class WindowsHandleWriter:
    """Minimal text writer backed by a Windows pipe/standard handle."""

    encoding = "utf-8"
    errors = "replace"

    def __init__(self, handle: int) -> None:
        self.handle = handle

    def write(self, text: object) -> int:
        value = str(text)
        if not value:
            return 0
        raw = value.encode(self.encoding, self.errors)
        buffer = ctypes.create_string_buffer(raw)
        written = ctypes.c_ulong(0)
        success = ctypes.windll.kernel32.WriteFile(
            ctypes.c_void_p(self.handle), buffer, len(raw), ctypes.byref(written), None
        )
        return len(value) if success else 0

    def flush(self) -> None:
        return None

    def isatty(self) -> bool:
        return False

    def writable(self) -> bool:
        return True


def bind_windows_pipe_streams() -> None:
    """Restore stdout/stderr in a windowed frozen child launched with PIPEs."""
    if os.name != "nt":
        return
    invalid = ctypes.c_void_p(-1).value
    for attribute, identifier in (("stdout", -11), ("stderr", -12)):
        if getattr(sys, attribute) is not None:
            continue
        handle = ctypes.windll.kernel32.GetStdHandle(identifier)
        if handle not in (0, invalid, None):
            setattr(sys, attribute, WindowsHandleWriter(int(handle)))


def executable_folder() -> Path:
    if getattr(sys, "frozen", False):
        return Path(sys.executable).resolve().parent
    return Path(__file__).resolve().parent


def valid_udp001_root(path: Path) -> bool:
    programs = path / "01_PROGRAMS"
    return (
        path.name == "UDP001"
        and programs.is_dir()
        and all((programs / name).is_file() for name in EXPECTED_PROGRAMS)
    )


def resolve_udp001_root(explicit: str | None = None) -> Path:
    candidates: list[Path] = []
    if explicit:
        candidates.append(Path(explicit))
    folder = executable_folder()
    candidates.extend((folder, folder.parent, folder.parent.parent))
    for candidate in candidates:
        try:
            resolved = candidate.resolve()
        except OSError:
            continue
        if valid_udp001_root(resolved):
            return resolved
    raise HostError("UDP001_ROOT_NOT_FOUND")


def atomic_write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(text.encode("utf-8"))
    temporary.replace(path)


def normalize_exit_code(value: object) -> int:
    if value is None:
        return 0
    if isinstance(value, bool):
        return int(value)
    if isinstance(value, int):
        return value
    return 1


def install_program_import_path(programs: Path) -> str:
    """Make sibling U001 modules importable without weakening program approval."""
    value = str(programs.resolve())
    while value in sys.path:
        sys.path.remove(value)
    sys.path.insert(0, value)
    return value


def run_external_program(script: Path, arguments: list[str]) -> int:
    bind_windows_pipe_streams()
    root = resolve_udp001_root()
    try:
        resolved = script.resolve()
    except OSError as exc:
        raise HostError("PROGRAM_PATH_INVALID") from exc
    programs = (root / "01_PROGRAMS").resolve()
    if resolved.parent != programs or resolved.name not in EXPECTED_PROGRAMS:
        raise HostError("PROGRAM_NOT_APPROVED")
    install_program_import_path(programs)
    sys.argv = [str(resolved), *arguments]
    try:
        runpy.run_path(str(resolved), run_name="__main__")
        return 0
    except SystemExit as exc:
        return normalize_exit_code(exc.code)


def run_gui(explicit_root: str | None = None) -> int:
    root = resolve_udp001_root(explicit_root)
    install_program_import_path(root / "01_PROGRAMS")
    gui = root / "01_PROGRAMS" / GUI_NAME
    sys.argv = [str(gui)]
    try:
        runpy.run_path(str(gui), run_name="__main__")
        return 0
    except SystemExit as exc:
        return normalize_exit_code(exc.code)


def probe_capsule(root_ok: bool, programs_ok: bool) -> str:
    frozen_ok = bool(getattr(sys, "frozen", False))
    pandas_ok = bool(getattr(_embedded_pandas, "__version__", ""))
    tkinter_ok = bool(getattr(_embedded_tkinter, "TkVersion", 0))
    gui_ok = root_ok and programs_ok
    orchestrator_ok = root_ok and programs_ok
    values = (frozen_ok, root_ok, programs_ok, pandas_ok, tkinter_ok, gui_ok, orchestrator_ok)
    errors = sum(not value for value in values)
    return (
        "U001HOST1\n"
        f"N-F{int(frozen_ok)}-R{int(root_ok)}-A{int(programs_ok)}-P{int(pandas_ok)}-"
        f"T{int(tkinter_ok)}-G{int(gui_ok)}-O{int(orchestrator_ok)}-X{errors}\n"
        f"C-{0 if errors == 0 else 7}\n"
    )


def run_probe(explicit_root: str | None, output: Path) -> int:
    root_ok = programs_ok = False
    try:
        root = resolve_udp001_root(explicit_root)
        root_ok = True
        programs_ok = all((root / "01_PROGRAMS" / name).is_file() for name in EXPECTED_PROGRAMS)
    except HostError:
        pass
    capsule = probe_capsule(root_ok, programs_ok)
    atomic_write(output, capsule)
    return 0 if capsule.rstrip().endswith("C-0") else 7


def show_error(message: str) -> None:
    if os.name == "nt":
        try:
            ctypes.windll.user32.MessageBoxW(0, message, "U001 Workbench", 0x10)
            return
        except Exception:
            pass
    if sys.stderr is not None:
        print(message, file=sys.stderr)


def run_self_test() -> None:
    passed = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td) / "UDP001"
        programs = root / "01_PROGRAMS"
        programs.mkdir(parents=True)
        for name in EXPECTED_PROGRAMS:
            (programs / name).write_text("x", encoding="utf-8")
        passed += int(valid_udp001_root(root))
        passed += int(not valid_udp001_root(root.parent))
        passed += int(len(EXPECTED_PROGRAMS) == 9)
        passed += int(GUI_NAME in EXPECTED_PROGRAMS)
        passed += int(ORCHESTRATOR_NAME in EXPECTED_PROGRAMS)
        passed += int(normalize_exit_code(None) == 0)
        passed += int(normalize_exit_code(7) == 7)
        passed += int(normalize_exit_code("bad") == 1)
        output = root / "probe.txt"
        atomic_write(output, "x")
        passed += int(output.read_text(encoding="utf-8") == "x")
        source_probe = probe_capsule(True, True)
        passed += int(source_probe.startswith("U001HOST1\n"))
        passed += int("F0-R1-A1-P1-T1-G1-O1-X1" in source_probe)
        passed += int(source_probe.endswith("C-7\n"))
        passed += int(bool(getattr(_embedded_pandas, "__version__", "")))
        passed += int(bool(getattr(_embedded_tkinter, "TkVersion", 0)))
        passed += int(PROGRAM_NAME.endswith("v0_2.py"))
        passed += int(PROGRAM_VERSION == "0.2")
        import_path = install_program_import_path(programs)
        passed += int(sys.path[0] == import_path)
        install_program_import_path(programs)
        passed += int(sys.path.count(import_path) == 1)
        sys.path.remove(import_path)
    if passed != 18:
        raise AssertionError(f"SELFTEST_COUNT:{passed}")
    print(SELFTEST_TOKEN)


def main(argv: Sequence[str] | None = None) -> int:
    values = list(argv if argv is not None else sys.argv[1:])
    if values and values[0].lower().endswith(".py"):
        try:
            return run_external_program(Path(values[0]), values[1:])
        except Exception as exc:
            bind_windows_pipe_streams()
            if sys.stderr is not None:
                print(f"U001HOST-ERROR:{exc}", file=sys.stderr)
            return 7

    parser = argparse.ArgumentParser(description="U001 portable runtime host")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--probe", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--probe-out")
    args = parser.parse_args(values)
    bind_windows_pipe_streams()
    if args.self_test:
        run_self_test()
        return 0
    if args.probe:
        if not args.probe_out:
            parser.error("--probe requires --probe-out")
        return run_probe(args.udp001_root, Path(args.probe_out))
    try:
        return run_gui(args.udp001_root)
    except Exception as exc:
        show_error(str(exc))
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
