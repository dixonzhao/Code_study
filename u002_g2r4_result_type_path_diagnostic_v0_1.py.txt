#!/usr/bin/env python3
"""Read-only type-path diagnostic for the retained G2R4 synthetic result."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import re
from typing import Any


STAGING_REL = Path("05_TESTS/02_ACCEPTANCE/U21G_G2_VERTICAL_V08.staging")
BUNDLE_REL = Path("01_PROGRAMS/u002_module_runtime_protocol_bundle_v0_2.json")


def load_json(path: Path) -> Any:
    return json.loads(path.read_text(encoding="utf-8-sig"))


def mappings(value: Any):
    if isinstance(value, dict):
        yield value
        for item in value.values():
            yield from mappings(item)
    elif isinstance(value, list):
        for item in value:
            yield from mappings(item)


def json_type(value: Any) -> str:
    if value is None:
        return "null"
    if isinstance(value, bool):
        return "boolean"
    if isinstance(value, int):
        return "integer"
    if isinstance(value, float):
        return "number"
    if isinstance(value, str):
        return "string"
    if isinstance(value, list):
        return "array"
    if isinstance(value, dict):
        return "object"
    return type(value).__name__


def type_matches(value: Any, expected: Any) -> bool:
    allowed = expected if isinstance(expected, list) else [expected]
    actual = json_type(value)
    return actual in allowed or (actual == "integer" and "number" in allowed)


def resolve_ref(bundle: dict, reference: str) -> dict:
    if not reference.startswith("#/"):
        return {}
    value: Any = bundle
    for token in reference[2:].split("/"):
        token = token.replace("~1", "/").replace("~0", "~")
        if not isinstance(value, dict) or token not in value:
            return {}
        value = value[token]
    return value if isinstance(value, dict) else {}


def issues(value: Any, schema: Any, bundle: dict, path: str) -> list[tuple[str, str, str]]:
    if not isinstance(schema, dict):
        return []
    if "$ref" in schema:
        return issues(value, resolve_ref(bundle, schema["$ref"]), bundle, path)
    if "allOf" in schema:
        found = []
        for branch in schema["allOf"]:
            found.extend(issues(value, branch, bundle, path))
        return found
    for keyword in ("anyOf", "oneOf"):
        if keyword in schema:
            branches = [issues(value, branch, bundle, path) for branch in schema[keyword]]
            passing = [branch for branch in branches if not branch]
            required = 1 if keyword == "oneOf" else 1
            if len(passing) >= required:
                return []
            return min(branches, key=len) if branches else [(path, keyword, json_type(value))]
    expected_type = schema.get("type")
    if expected_type is not None and not type_matches(value, expected_type):
        expected = "/".join(expected_type) if isinstance(expected_type, list) else str(expected_type)
        return [(path, expected, json_type(value))]
    if "const" in schema and value != schema["const"]:
        return [(path, "const", json_type(value))]
    if "enum" in schema and value not in schema["enum"]:
        return [(path, "enum", json_type(value))]
    found: list[tuple[str, str, str]] = []
    if isinstance(value, dict):
        properties = schema.get("properties", {})
        required = schema.get("required", [])
        for key in required:
            if key not in value:
                found.append((f"{path}.{key}", "required", "missing"))
        if isinstance(properties, dict):
            for key, item in value.items():
                if key in properties:
                    found.extend(issues(item, properties[key], bundle, f"{path}.{key}"))
                else:
                    additional = schema.get("additionalProperties", True)
                    if additional is False:
                        found.append((f"{path}.{key}", "prohibited", json_type(item)))
                    elif isinstance(additional, dict):
                        found.extend(issues(item, additional, bundle, f"{path}.{key}"))
    elif isinstance(value, list) and "items" in schema:
        for index, item in enumerate(value):
            found.extend(issues(item, schema["items"], bundle, f"{path}[{index}]"))
    return found


def locate_result(root: Path) -> tuple[Path, dict]:
    candidates = []
    staging = root / STAGING_REL
    for path in staging.rglob("result.json"):
        if not path.is_file():
            continue
        try:
            value = load_json(path)
        except Exception:
            continue
        if isinstance(value, dict) and value.get("ContractType") == "U002_MODULE_EXECUTION_RESULT":
            candidates.append((path, value))
    if len(candidates) != 1:
        raise RuntimeError(f"RESULT_COUNT_{len(candidates)}")
    return candidates[0]


def locate_result_schema(bundle: dict) -> dict:
    candidates = []
    for item in mappings(bundle):
        props = item.get("properties")
        if not isinstance(props, dict):
            continue
        contract = props.get("ContractType")
        if isinstance(contract, dict) and contract.get("const") == "U002_MODULE_EXECUTION_RESULT":
            candidates.append(item)
    if not candidates:
        raise RuntimeError("RESULT_SCHEMA_NOT_FOUND")
    return candidates[0]


def token(value: str) -> str:
    return re.sub(r"[^A-Za-z0-9_.\[\]-]+", "_", value)[:180]


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    try:
        root = Path(args.workspace_root).resolve()
        _, result = locate_result(root)
        bundle = load_json(root / BUNDLE_REL)
        if not isinstance(bundle, dict):
            raise RuntimeError("BUNDLE_TYPE")
        schema = locate_result_schema(bundle)
        found = issues(result, schema, bundle, "Result")
        type_only = [item for item in found if item[1] not in ("const", "enum", "required", "prohibited")]
        selected = type_only[0] if type_only else (found[0] if found else ("NONE", "NONE", "NONE"))
        print(
            f"G2R4TD1 N-J1-S1-I{len(found)}-T{len(type_only)}-X0 C-0 "
            f"P-{token(selected[0])} E-{token(selected[1])} A-{token(selected[2])}"
        )
        return 0
    except Exception as exc:
        print(f"G2R4TD1 N-J0-S0-I0-T0-X1 C-7 K-{token(exc.__class__.__name__ + '_' + str(exc))}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
