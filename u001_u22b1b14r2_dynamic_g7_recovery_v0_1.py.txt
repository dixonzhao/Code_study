#!/usr/bin/env python3
"""Recover B14 after the legacy G7 five-signal output guard fails.

The tool never changes the accepted U001 installation or the failed B14 run.
It creates a run-owned patched copy of G7, rebuilds only Gate 7 from the
already accepted G1-G6 artifacts, and independently validates the six-file
package for the 22-signal / 21-frame-signal Controlled Profile.
"""

from __future__ import annotations

import argparse
import ast
import csv
import hashlib
import json
import os
from pathlib import Path
import re
import shutil
import subprocess
import sys
import tempfile
import traceback
from typing import Dict, Iterable, List, Mapping, Optional, Sequence, Tuple


PROGRAM_ID = "U001-U22B1B14R2-DYNAMIC-G7-RECOVERY"
PROGRAM_VERSION = "0.1"
LEGACY_G7_NAME = "u001_output_package_assembler_v0_3.py"
FAILURE_TOKEN = "FRAME_SIGNAL_COUNT_NOT_5"
REPLACEMENT_TOKEN = "FRAME_SIGNAL_COUNT_NOT_POSITIVE"
EXPECTED_SELECTED_SIGNALS = 22
EXPECTED_FRAME_SIGNALS = 21

FINAL_PACKAGE_FILES = (
    "u001_decoded_updates.csv",
    "u001_frame_values.csv",
    "u001_signal_catalog.csv",
    "u001_quality_report.csv",
    "u001_run_manifest.json",
    "u001_review_view.csv",
)


class RecoveryError(RuntimeError):
    """Fail-closed recovery error with bounded evidence."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = token(reason, 64)
        self.detail = token(detail, 180)
        super().__init__(self.reason)


def token(value: object, limit: int = 100) -> str:
    """Return a bounded low-bandwidth token."""

    text = re.sub(r"[^A-Z0-9_.:+-]+", "_", str(value).upper())
    return text.strip("_")[:limit] or "NA"


def sha256_file(path: Path) -> str:
    """Hash a file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def write_text(path: Path, text: str) -> None:
    """Write UTF-8 text atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("w", encoding="utf-8", newline="\n") as stream:
        stream.write(text)
    os.replace(temporary, path)


def write_json(path: Path, value: Mapping[str, object]) -> None:
    """Write deterministic JSON atomically."""

    write_text(path, json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n")


def read_json(path: Path, label: str) -> Dict[str, object]:
    """Read one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise RecoveryError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise RecoveryError(label + "_INVALID", type(exc).__name__) from exc
    if not isinstance(value, dict):
        raise RecoveryError(label + "_NOT_OBJECT", path.name)
    return value


def locate_one(root: Path, basename: str, label: str) -> Path:
    """Locate exactly one basename below a root."""

    matches = [
        path.resolve()
        for path in root.rglob(basename)
        if path.is_file() and not path.is_symlink()
    ]
    if len(matches) != 1:
        raise RecoveryError(label + "_COUNT", len(matches))
    return matches[0]


def locate_program(root: Path, basename: str, label: str) -> Path:
    """Locate one runtime program with optional transfer-only .txt suffix."""

    matches = []
    for name in (basename, basename + ".txt"):
        path = root / name
        if path.is_file() and not path.is_symlink():
            matches.append(path.resolve())
    if len(matches) != 1:
        raise RecoveryError(label + "_COUNT", len(matches))
    return matches[0]


def snapshot_tree(roots: Iterable[Path]) -> Dict[str, Tuple[int, str]]:
    """Snapshot protected files by relative identity, byte size, and hash."""

    result: Dict[str, Tuple[int, str]] = {}
    for root in roots:
        if not root.is_dir() or root.is_symlink():
            raise RecoveryError("PROTECTED_ROOT_MISSING", root.name)
        for path in sorted(root.rglob("*")):
            if path.is_symlink():
                raise RecoveryError("PROTECTED_SYMLINK", path.name)
            if path.is_file():
                key = root.name + "/" + path.relative_to(root).as_posix()
                result[key] = (path.stat().st_size, sha256_file(path))
    return result


def tree_contains_string(node: ast.AST, expected: str) -> bool:
    """Return whether an AST subtree contains one exact string constant."""

    return any(
        isinstance(child, ast.Constant) and child.value == expected
        for child in ast.walk(node)
    )


def patch_dynamic_signal_guard(source: str) -> Tuple[str, Dict[str, object]]:
    """Replace only the unique legacy five-signal failure guard.

    The accepted G7 implementation is preserved. The condition that rejected
    every count other than five becomes a fail-closed positive-count check.
    All later schema, catalog, frame, review, and six-file checks remain active.
    """

    try:
        tree = ast.parse(source)
    except SyntaxError as exc:
        raise RecoveryError("G7_SOURCE_SYNTAX", exc.lineno) from exc

    matches: List[Tuple[ast.If, ast.expr]] = []
    for node in ast.walk(tree):
        if not isinstance(node, ast.If) or not tree_contains_string(node, FAILURE_TOKEN):
            continue
        test = node.test
        if not isinstance(test, ast.Compare) or len(test.ops) != 1 or len(test.comparators) != 1:
            raise RecoveryError("G7_GUARD_SHAPE", type(test).__name__)
        if not isinstance(test.ops[0], ast.NotEq):
            raise RecoveryError("G7_GUARD_OPERATOR", type(test.ops[0]).__name__)
        left = test.left
        right = test.comparators[0]
        if isinstance(right, ast.Constant) and right.value == 5:
            dynamic_expression = left
        elif isinstance(left, ast.Constant) and left.value == 5:
            dynamic_expression = right
        else:
            raise RecoveryError("G7_GUARD_LITERAL", ast.get_source_segment(source, test) or "NA")
        matches.append((node, dynamic_expression))

    if len(matches) != 1:
        raise RecoveryError("G7_GUARD_COUNT", len(matches))

    node, dynamic_expression = matches[0]
    old_test = ast.get_source_segment(source, node.test)
    expression = ast.get_source_segment(source, dynamic_expression)
    if not old_test or not expression or source.count(old_test) != 1:
        raise RecoveryError("G7_GUARD_SOURCE_IDENTITY", old_test or "NONE")
    if source.count(FAILURE_TOKEN) != 1:
        raise RecoveryError("G7_FAILURE_TOKEN_COUNT", source.count(FAILURE_TOKEN))

    new_test = f"({expression}) <= 0"
    patched = source.replace(old_test, new_test, 1)
    patched = patched.replace(FAILURE_TOKEN, REPLACEMENT_TOKEN, 1)
    try:
        ast.parse(patched)
    except SyntaxError as exc:
        raise RecoveryError("PATCHED_SOURCE_SYNTAX", exc.lineno) from exc

    if FAILURE_TOKEN in patched or patched.count(REPLACEMENT_TOKEN) != 1:
        raise RecoveryError("PATCH_TOKEN_POSTCONDITION", "FAILED")
    evidence = {
        "OriginalCondition": old_test,
        "PatchedCondition": new_test,
        "DynamicExpression": expression,
        "OriginalFailureToken": FAILURE_TOKEN,
        "PatchedFailureToken": REPLACEMENT_TOKEN,
        "GuardMatchCount": 1,
    }
    return patched, evidence


def run_child(label: str, command: Sequence[str], log_root: Path) -> Dict[str, object]:
    """Run one child process and preserve complete local evidence."""

    result = subprocess.run(
        list(command),
        text=True,
        encoding="utf-8",
        errors="replace",
        capture_output=True,
        timeout=1800,
        check=False,
    )
    stdout_path = log_root / f"{label}_stdout.txt"
    stderr_path = log_root / f"{label}_stderr.txt"
    write_text(stdout_path, result.stdout)
    write_text(stderr_path, result.stderr)
    record = {
        "Label": label,
        "ExitCode": result.returncode,
        "StdoutBytes": len(result.stdout.encode("utf-8")),
        "StderrBytes": len(result.stderr.encode("utf-8")),
        "StdoutSha256": hashlib.sha256(result.stdout.encode("utf-8")).hexdigest(),
        "StderrSha256": hashlib.sha256(result.stderr.encode("utf-8")).hexdigest(),
    }
    if result.returncode != 0:
        lines = [line.strip() for line in (result.stderr + "\n" + result.stdout).splitlines() if line.strip()]
        detail = lines[-1] if lines else f"RC{result.returncode}"
        raise RecoveryError(label + "_EXIT", detail)
    return record


def count_csv_rows(path: Path) -> int:
    """Count data records through csv.reader without loading the table."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        try:
            next(reader)
        except StopIteration:
            return 0
        return sum(1 for _ in reader)


def csv_header(path: Path) -> List[str]:
    """Read one CSV header."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        return [str(value) for value in next(csv.reader(stream), [])]


def truthy(value: object) -> bool:
    """Interpret controlled CSV booleans."""

    return str(value).strip().upper() in {"1", "TRUE", "T", "YES", "Y"}


def catalog_counts(path: Path) -> Tuple[int, int]:
    """Count catalog signals and dynamic Frame View inclusions."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        if "CanonicalName" not in (reader.fieldnames or []):
            raise RecoveryError("CATALOG_CANONICALNAME_MISSING", path.name)
        if "IncludeInFrameView" not in (reader.fieldnames or []):
            raise RecoveryError("CATALOG_INCLUDE_FLAG_MISSING", path.name)
        rows = list(reader)
    return len(rows), sum(truthy(row.get("IncludeInFrameView", "")) for row in rows)


def find_final_package(root: Path) -> Path:
    """Locate exactly one complete final six-file package."""

    matches: List[Path] = []
    for folder in [root, *sorted(path for path in root.rglob("*") if path.is_dir())]:
        if all((folder / name).is_file() for name in FINAL_PACKAGE_FILES):
            matches.append(folder.resolve())
    matches = list(dict.fromkeys(matches))
    if len(matches) != 1:
        raise RecoveryError("FINAL_PACKAGE_COUNT", len(matches))
    actual = {path.name for path in matches[0].iterdir() if path.is_file()}
    if actual != set(FINAL_PACKAGE_FILES):
        raise RecoveryError("FINAL_PACKAGE_FILE_SET", len(actual))
    if any((matches[0] / name).stat().st_size == 0 for name in FINAL_PACKAGE_FILES):
        raise RecoveryError("FINAL_PACKAGE_EMPTY", "SIX")
    return matches[0]


def verify_failed_b14(failed_root: Path) -> Dict[str, object]:
    """Require the exact B14 failure that this recovery is authorized to fix."""

    manifest = read_json(
        failed_root / "03_RUN" / "CONTROL_OUTPUT" / "u001_g8a2_orchestration_manifest.json",
        "B14_ORCHESTRATION_MANIFEST",
    )
    stages = manifest.get("Stages")
    passed = 0
    if isinstance(stages, list):
        passed = sum(isinstance(item, dict) and item.get("Success") is True for item in stages)
    errors = manifest.get("Errors")
    joined_errors = "\n".join(str(item) for item in errors) if isinstance(errors, list) else ""
    g7_stderr = failed_root / "03_RUN" / "CONTROL_OUTPUT" / "G7B_stderr.txt"
    evidence = joined_errors
    if g7_stderr.is_file():
        evidence += "\n" + g7_stderr.read_text(encoding="utf-8-sig", errors="replace")
    if manifest.get("Status") != "FAILURE" or passed != 8:
        raise RecoveryError("B14_FAILURE_IDENTITY", f"{manifest.get('Status')}:{passed}")
    if FAILURE_TOKEN not in evidence:
        raise RecoveryError("B14_FAILURE_TOKEN_MISSING", FAILURE_TOKEN)
    return {"SuccessfulStages": passed, "FailureEvidenceSha256": hashlib.sha256(evidence.encode()).hexdigest()}


def execute_recovery(udp001_root: Path, failed_root: Path, output_root: Path) -> Tuple[str, str]:
    """Patch the isolated legacy guard, rebuild G7, and validate the package."""

    child_results: List[Dict[str, object]] = []
    try:
        if output_root.exists():
            raise RecoveryError("OUTPUT_ALREADY_EXISTS", output_root.name)
        if not failed_root.is_dir() or failed_root.is_symlink():
            raise RecoveryError("FAILED_RUN_ROOT_MISSING", failed_root.name)
        if not udp001_root.is_dir() or udp001_root.is_symlink():
            raise RecoveryError("UDP001_ROOT_MISSING", udp001_root.name)

        failure_identity = verify_failed_b14(failed_root)
        shadow_program_root = failed_root / "00_ISOLATED_UDP001" / "01_PROGRAMS"
        legacy_source = locate_program(shadow_program_root, LEGACY_G7_NAME, "FAILED_SHADOW_G7")
        active_source = locate_program(udp001_root / "01_PROGRAMS", LEGACY_G7_NAME, "ACTIVE_G7")
        if sha256_file(legacy_source) != sha256_file(active_source):
            raise RecoveryError("SHADOW_ACTIVE_G7_MISMATCH", "HASH")

        g7_config = locate_one(failed_root / "03_RUN", "u001_g7_package_config.json", "G7_CONFIG")
        protected_roots = (udp001_root / "01_PROGRAMS", udp001_root / "02_CONTROL")
        protected_before = snapshot_tree(protected_roots)
        evidence_before = {
            str(path): sha256_file(path)
            for path in (legacy_source, g7_config)
        }

        output_root.mkdir(parents=True)
        patched_root = output_root / "00_PATCHED_RUNTIME"
        log_root = output_root / "01_DIAGNOSTICS"
        build_root = output_root / "02_G7_BUILD"
        patched_path = patched_root / LEGACY_G7_NAME
        patched_text, patch_evidence = patch_dynamic_signal_guard(
            legacy_source.read_text(encoding="utf-8-sig")
        )
        write_text(patched_path, patched_text)
        patch_evidence.update(
            {
                "SourceBasename": legacy_source.name,
                "OriginalSha256": sha256_file(legacy_source),
                "PatchedSha256": sha256_file(patched_path),
                "Scope": "RUN_OWNED_ISOLATED_G7_ONLY",
                "AcceptedU001Modified": False,
            }
        )
        write_json(output_root / "u22b1b14r2_patch_manifest_v0_1.json", patch_evidence)

        child_results.append(
            run_child("PATCHED_G7_SELFTEST", [sys.executable, str(patched_path), "--self-test"], log_root)
        )
        child_results.append(
            run_child(
                "PATCHED_G7_BUILD",
                [
                    sys.executable,
                    str(patched_path),
                    "--build",
                    "--config",
                    str(g7_config),
                    "--out-root",
                    str(build_root),
                ],
                log_root,
            )
        )

        package = find_final_package(build_root)
        capsule = locate_one(build_root, "udp001_g7_return_capsule.txt", "G7_CAPSULE")
        capsule_text = capsule.read_text(encoding="utf-8-sig", errors="replace")
        if not re.search(r"(?m)^C-0\s*$", capsule_text):
            raise RecoveryError("G7_CAPSULE_NOT_C0", token(capsule_text[-200:]))

        decoded = package / "u001_decoded_updates.csv"
        frames = package / "u001_frame_values.csv"
        catalog = package / "u001_signal_catalog.csv"
        review = package / "u001_review_view.csv"
        decoded_rows = count_csv_rows(decoded)
        frame_rows = count_csv_rows(frames)
        review_rows = count_csv_rows(review)
        selected_count, included_count = catalog_counts(catalog)
        frame_value_count = sum(name.endswith("__value") for name in csv_header(frames))
        review_header = csv_header(review)

        if decoded_rows <= 0 or frame_rows <= 0:
            raise RecoveryError("PACKAGE_ROWS_NOT_POSITIVE", f"U{decoded_rows}R{frame_rows}")
        if frame_rows != review_rows:
            raise RecoveryError("FRAME_REVIEW_ROW_MISMATCH", f"F{frame_rows}V{review_rows}")
        if selected_count != EXPECTED_SELECTED_SIGNALS:
            raise RecoveryError("CATALOG_SIGNAL_COUNT", selected_count)
        if included_count != EXPECTED_FRAME_SIGNALS:
            raise RecoveryError("CATALOG_FRAME_SIGNAL_COUNT", included_count)
        if frame_value_count != EXPECTED_FRAME_SIGNALS:
            raise RecoveryError("FRAME_VALUE_GROUP_COUNT", frame_value_count)
        if len(review_header) != EXPECTED_FRAME_SIGNALS + 1 or review_header[0] != "FrameCount":
            raise RecoveryError("REVIEW_VIEW_WIDTH", len(review_header))

        protected_after = snapshot_tree(protected_roots)
        if protected_before != protected_after:
            raise RecoveryError("ACCEPTED_U001_CHANGED", "PROGRAM_OR_CONTROL")
        evidence_after = {
            str(path): sha256_file(path)
            for path in (legacy_source, g7_config)
        }
        if evidence_before != evidence_after:
            raise RecoveryError("FAILED_RUN_EVIDENCE_CHANGED", "SOURCE_OR_CONFIG")

        line1 = (
            f"B14R2 N-S8-G7P-F6-C{selected_count}-V{included_count}"
            f"-U{decoded_rows}-R{frame_rows}-H1-X0 C-0"
        )
        line2 = "B14R2I N-DYN21-P1-A0-E1-X0 C-0"
        manifest = {
            "CompletionCode": 0,
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "ExecutionMode": "G7_ONLY_ISOLATED_COMPATIBILITY_RECOVERY",
            "FailedB14Root": str(failed_root.resolve()),
            "FailureIdentity": failure_identity,
            "PatchEvidence": patch_evidence,
            "G7ConfigBasename": g7_config.name,
            "G7ConfigSha256": sha256_file(g7_config),
            "FinalPackageRelativePath": package.relative_to(output_root.resolve()).as_posix(),
            "FinalPackageFileCount": 6,
            "SelectedSignalCount": selected_count,
            "FrameSignalCount": included_count,
            "FrameValueGroupCount": frame_value_count,
            "ReviewViewColumnCount": len(review_header),
            "DecodedUpdateRows": decoded_rows,
            "FrameRows": frame_rows,
            "ReviewRows": review_rows,
            "AcceptedU001Unchanged": True,
            "FailedRunEvidenceUnchanged": True,
            "ChildResults": child_results,
            "ReturnCapsule": [line1, line2],
        }
        write_json(output_root / "u22b1b14r2_recovery_manifest_v0_1.json", manifest)
        write_text(output_root / "u22b1b14r2_return_capsule.txt", line1 + "\n" + line2 + "\n")
        write_text(output_root / "U22B1B14R2_COMPLETE.marker", "C-0\n")
        return line1, line2
    except Exception as exc:
        if isinstance(exc, RecoveryError):
            reason, detail = exc.reason, exc.detail
        else:
            reason, detail = token(type(exc).__name__), token(str(exc), 180)
        line1 = f"B14R2E K-{reason}-X1 C-7"
        line2 = f"B14R2D D-{detail} C-0"
        try:
            output_root.mkdir(parents=True, exist_ok=True)
            write_json(
                output_root / "u22b1b14r2_failure_diagnostic_v0_1.json",
                {
                    "CompletionCode": 7,
                    "ProgramId": PROGRAM_ID,
                    "ProgramVersion": PROGRAM_VERSION,
                    "Reason": reason,
                    "Detail": detail,
                    "ChildResults": child_results,
                    "Traceback": traceback.format_exc(),
                    "ReturnCapsule": [line1, line2],
                },
            )
            write_text(output_root / "u22b1b14r2_return_capsule.txt", line1 + "\n" + line2 + "\n")
        except Exception:
            pass
        return line1, line2


def self_test() -> int:
    """Exercise the patch and bounded validation helpers."""

    checks: List[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            raise AssertionError(name)
        checks.append(name)

    sample = '''\
def build(frame_signal_count):
    if frame_signal_count != 5:
        raise RuntimeError("FRAME_SIGNAL_COUNT_NOT_5")
    return frame_signal_count
'''
    patched, evidence = patch_dynamic_signal_guard(sample)
    namespace: Dict[str, object] = {}
    exec(patched, namespace)
    build = namespace["build"]
    check("T01_PATCHED_21", callable(build) and build(21) == 21)
    check("T02_PATCHED_5", build(5) == 5)
    try:
        build(0)
        zero_failed = False
    except RuntimeError as exc:
        zero_failed = REPLACEMENT_TOKEN in str(exc)
    check("T03_ZERO_REJECTED", zero_failed)
    check("T04_OLD_TOKEN_GONE", FAILURE_TOKEN not in patched)
    check("T05_NEW_TOKEN_ONE", patched.count(REPLACEMENT_TOKEN) == 1)
    check("T06_OLD_CONDITION", evidence["OriginalCondition"] == "frame_signal_count != 5")
    check("T07_NEW_CONDITION", evidence["PatchedCondition"] == "(frame_signal_count) <= 0")
    check("T08_DYNAMIC_EXPRESSION", evidence["DynamicExpression"] == "frame_signal_count")

    with tempfile.TemporaryDirectory() as temp_name:
        root = Path(temp_name)
        text_path = root / "text.txt"
        write_text(text_path, "abc\n")
        check("T09_WRITE", text_path.read_bytes() == b"abc\n")
        check("T10_SHA", sha256_file(text_path) == hashlib.sha256(b"abc\n").hexdigest())
        check("T11_TOKEN", token("a b/c") == "A_B_C")

        csv_path = root / "table.csv"
        write_text(csv_path, "A,B\n1,2\n3,4\n")
        check("T12_ROWS", count_csv_rows(csv_path) == 2)
        check("T13_HEADER", csv_header(csv_path) == ["A", "B"])

        catalog = root / "catalog.csv"
        write_text(
            catalog,
            "CanonicalName,IncludeInFrameView\nFC,0\nA,1\nB,TRUE\n",
        )
        check("T14_CATALOG_ROWS", catalog_counts(catalog)[0] == 3)
        check("T15_CATALOG_INCLUDED", catalog_counts(catalog)[1] == 2)
        check("T16_TRUTHY_1", truthy("1"))
        check("T17_TRUTHY_TRUE", truthy("true"))
        check("T18_TRUTHY_ZERO", not truthy("0"))

        package = root / "build" / "package"
        package.mkdir(parents=True)
        for name in FINAL_PACKAGE_FILES:
            write_text(package / name, "x\n")
        check("T19_PACKAGE", find_final_package(root / "build") == package.resolve())
        check("T20_PACKAGE_FILE_COUNT", len(FINAL_PACKAGE_FILES) == 6)

        protected_a = root / "programs"
        protected_b = root / "controls"
        write_text(protected_a / "a.py", "pass\n")
        write_text(protected_b / "a.csv", "A\n1\n")
        before = snapshot_tree((protected_a, protected_b))
        after = snapshot_tree((protected_a, protected_b))
        check("T21_SNAPSHOT_EQUAL", before == after)
        check("T22_SNAPSHOT_COUNT", len(before) == 2)

        duplicate_source = sample + "\n" + sample.replace("build", "build2")
        try:
            patch_dynamic_signal_guard(duplicate_source)
            duplicate_failed = False
        except RecoveryError as exc:
            duplicate_failed = exc.reason == "G7_GUARD_COUNT"
        check("T23_DUPLICATE_FAIL", duplicate_failed)

        wrong_source = sample.replace("!= 5", "> 5")
        try:
            patch_dynamic_signal_guard(wrong_source)
            operator_failed = False
        except RecoveryError as exc:
            operator_failed = exc.reason == "G7_GUARD_OPERATOR"
        check("T24_OPERATOR_FAIL", operator_failed)

        missing_source = "def build(n):\n    return n\n"
        try:
            patch_dynamic_signal_guard(missing_source)
            missing_failed = False
        except RecoveryError as exc:
            missing_failed = exc.reason == "G7_GUARD_COUNT"
        check("T25_MISSING_FAIL", missing_failed)

    check("T26_SELECTED_COUNT", EXPECTED_SELECTED_SIGNALS == 22)
    check("T27_FRAME_COUNT", EXPECTED_FRAME_SIGNALS == 21)
    check("T28_PROGRAM_NAME", LEGACY_G7_NAME.endswith("v0_3.py"))
    check("T29_POSITIVE_RELATION", EXPECTED_FRAME_SIGNALS < EXPECTED_SELECTED_SIGNALS)
    own_tree = ast.parse(Path(__file__).read_text(encoding="utf-8"))
    imported_roots = {
        alias.name.split(".")[0]
        for node in ast.walk(own_tree)
        if isinstance(node, (ast.Import, ast.ImportFrom))
        for alias in node.names
    }
    check("T30_NO_PANDAS", "pandas" not in imported_roots)
