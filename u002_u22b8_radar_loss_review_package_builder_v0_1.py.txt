class B8Error(RuntimeError):
    """Represent one deterministic fail-closed B8 error."""


def machine_token(value: object, limit: int = 120) -> str:
    """Normalize one diagnostic value for a compact return capsule."""

    token = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (token or "UNKNOWN")[:limit]


def error_reason(error: BaseException) -> str:
    """Return a stable reason even for exceptions with empty args."""

    return str(error.args[0]) if error.args else type(error).__name__


def canonical_bytes(value: object) -> bytes:
    """Serialize governed JSON deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def write_json(path: Path, value: object) -> None:
    """Write JSON atomically."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def sha256_file(path: Path) -> str:
    """Hash one file without loading it entirely into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_json(path: Path, reason: str) -> Dict[str, object]:
    """Load one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise B8Error(reason + "_MISSING")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise B8Error(reason + "_JSON") from error
    if not isinstance(value, dict):
        raise B8Error(reason + "_TYPE")
    return value


def parse_optional_float(value: object) -> Optional[float]:
    """Return one finite number or None."""

    text = str(value if value is not None else "").strip()
    if not text:
        return None
    try:
        number = float(text)
    except ValueError:
        return None
    return number if math.isfinite(number) else None


def parse_integer(value: object, field: str) -> int:
    """Parse an integer represented as integer or decimal text."""

    number = parse_optional_float(value)
    if number is None or not number.is_integer():
        raise B8Error("INTEGER_" + machine_token(field))
    return int(number)


def parse_boolean(value: object, field: str) -> bool:
    """Parse the controlled boolean vocabulary."""

    text = str(value if value is not None else "").strip().lower()
    if text in {"1", "true"}:
        return True
    if text in {"0", "false", ""}:
        return False
    raise B8Error("BOOLEAN_" + machine_token(field))


def write_csv(
    path: Path,
    columns: Sequence[str],
    rows: Iterable[Mapping[str, object]],
) -> None:
    """Write a deterministic UTF-8 CSV."""

    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(
            stream,
            fieldnames=list(columns),
            lineterminator="\n",
        )
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in columns})


def verify_b7u_capsule(path: Path) -> str:
    """Require the accepted B7U completion capsule."""

    if not path.is_file() or path.is_symlink():
        raise B8Error("B7U_CAPSULE_MISSING")
    text = path.read_text(encoding="utf-8-sig").strip()
    if not text.startswith("B7U ") or not text.endswith("C-0"):
        raise B8Error("B7U_NOT_ACCEPTED")
    return text


def verify_profile(profile_dir: Path) -> Tuple[Path, Path, Path, Dict[str, object]]:
    """Verify the accepted B7A profile payload and hashes."""

    if not profile_dir.is_dir() or profile_dir.is_symlink():
        raise B8Error("B7A_PROFILE_DIR_MISSING")
    row_path = profile_dir / "b7a_camera_lateral_evidence_rows.csv"
    sweep_path = profile_dir / "b7a_abs_lateral_threshold_sweep.csv"
    quality_path = profile_dir / "b7a_camera_lateral_evidence_quality.json"
    capsule_path = profile_dir / "b7a_return_capsule.txt"
    for path, reason in (
        (row_path, "B7A_ROWS"),
        (sweep_path, "B7A_SWEEP"),
        (quality_path, "B7A_QUALITY"),
        (capsule_path, "B7A_CAPSULE"),
    ):
        if not path.is_file() or path.is_symlink():
            raise B8Error(reason + "_MISSING")
    capsule = capsule_path.read_text(encoding="utf-8-sig").strip()
    if not capsule.startswith("B7A ") or not capsule.endswith("C-0"):
        raise B8Error("B7A_NOT_ACCEPTED")
    quality = load_json(quality_path, "B7A_QUALITY")
    if quality.get("CompletionCode") != 0:
        raise B8Error("B7A_QUALITY_NOT_ACCEPTED")
    if quality.get("DecisionBoundary") != (
        "READ_ONLY_EVIDENCE_PROFILE_NOT_A_PRODUCTION_GEOMETRY_POLICY"
    ):
        raise B8Error("B7A_DECISION_BOUNDARY")
    hashes = quality.get("OutputHashes")
    if not isinstance(hashes, dict):
        raise B8Error("B7A_OUTPUT_HASHES")
    for path in (row_path, sweep_path):
        expected = str(hashes.get(path.name, ""))
        if expected != sha256_file(path):
            raise B8Error("B7A_HASH_" + machine_token(path.name))
    return row_path, sweep_path, quality_path, quality


def read_source_rows(path: Path) -> List[Dict[str, object]]:
    """Read and validate strictly ordered B7A FrameCount evidence."""

    rows: List[Dict[str, object]] = []
    previous: Optional[int] = None
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        fieldnames = tuple(reader.fieldnames or ())
        missing = [column for column in SOURCE_COLUMNS if column not in fieldnames]
        if missing:
            raise B8Error("SOURCE_COLUMNS_" + machine_token("_".join(missing[:3])))
        for source in reader:
            frame = parse_integer(source["FrameCount"], "FRAMECOUNT")
            if previous is not None and frame <= previous:
                raise B8Error("FRAMECOUNT_NOT_STRICTLY_INCREASING")
            previous = frame
            row: Dict[str, object] = dict(source)
            row["FrameCount"] = frame
            rows.append(row)
    if not rows:
        raise B8Error("SOURCE_ROWS_EMPTY")
    return rows


def radar_usability(row: Mapping[str, object]) -> Tuple[bool, str]:
    """Evaluate the explicit TC001 Radar review eligibility rule."""

    distance_valid = parse_boolean(row["RadarDistanceValid"], "RADAR_DISTANCE_VALID")
    distance = parse_optional_float(row["RadarDistance"])
    flag = parse_optional_float(row["RadarDetectionFlag"])
    if not distance_valid:
        return False, "DISTANCE_STATE_INVALID"
    if distance is None:
        return False, "DISTANCE_MISSING"
    if distance <= RADAR_MINIMUM_DISTANCE_METERS:
        return False, "DISTANCE_NONPOSITIVE"
    if distance >= RADAR_MAXIMUM_DISTANCE_METERS:
        return False, "DISTANCE_SENTINEL_OR_OUT_OF_RANGE"
    if flag is None:
        return False, "DETECTION_FLAG_MISSING"
    if flag <= 0:
        return False, "DETECTION_FLAG_OFF"
    return True, "USABLE"


def annotate_rows(rows: Sequence[Mapping[str, object]]) -> List[Dict[str, object]]:
    """Add transparent Radar usability evidence without mutating source rows."""

    output: List[Dict[str, object]] = []
    for source in rows:
        usable, reason = radar_usability(source)
        row = dict(source)
        row["RadarTargetUsable"] = int(usable)
        row["RadarLossReason"] = reason
        output.append(row)
    return output


def contiguous_loss_spans(rows: Sequence[Mapping[str, object]]) -> List[Tuple[int, int]]:
    """Return contiguous unusable spans meeting the minimum persistence rule."""

    spans: List[Tuple[int, int]] = []
    start: Optional[int] = None
    end: Optional[int] = None
    for row in rows:
        frame = int(row["FrameCount"])
        unusable = not bool(int(row["RadarTargetUsable"]))
        if unusable and start is None:
            start = frame
            end = frame
        elif unusable and end is not None and frame == end + 1:
            end = frame
        else:
            if start is not None and end is not None:
                if end - start + 1 >= MINIMUM_LOSS_FRAMES:
                    spans.append((start, end))
            start = frame if unusable else None
            end = frame if unusable else None
    if start is not None and end is not None:
        if end - start + 1 >= MINIMUM_LOSS_FRAMES:
            spans.append((start, end))
    return spans


def merge_loss_spans(spans: Sequence[Tuple[int, int]]) -> List[Tuple[int, int]]:
    """Merge brief usable gaps so one physical episode stays one candidate."""

    merged: List[List[int]] = []
    for start, end in spans:
        if not merged or start - merged[-1][1] - 1 > MERGE_GAP_FRAMES:
            merged.append([start, end])
        else:
            merged[-1][1] = max(merged[-1][1], end)
    return [(start, end) for start, end in merged]


def build_candidates(
    rows: Sequence[Mapping[str, object]],
) -> List[Dict[str, object]]:
    """Build bounded, non-overlapping U003 review candidates."""

    frame_min = int(rows[0]["FrameCount"])
    frame_max = int(rows[-1]["FrameCount"])
    spans = merge_loss_spans(contiguous_loss_spans(rows))
    provisional: List[Dict[str, object]] = []
    for evidence_start, evidence_end in spans:
        review_start = max(frame_min, evidence_start - REVIEW_PADDING_BEFORE)
        review_end = min(frame_max, evidence_end + REVIEW_PADDING_AFTER)
        provisional.append(
            {
                "EvidenceStartFrame": evidence_start,
                "EvidenceEndFrame": evidence_end,
                "ReviewStartFrame": review_start,
                "ReviewEndFrame": review_end,
            }
        )

    combined: List[Dict[str, object]] = []
    for item in provisional:
        if not combined or int(item["ReviewStartFrame"]) > int(
            combined[-1]["ReviewEndFrame"]
        ):
            combined.append(dict(item))
        else:
            combined[-1]["EvidenceStartFrame"] = min(
                int(combined[-1]["EvidenceStartFrame"]),
                int(item["EvidenceStartFrame"]),
            )
            combined[-1]["EvidenceEndFrame"] = max(
                int(combined[-1]["EvidenceEndFrame"]),
                int(item["EvidenceEndFrame"]),
            )
            combined[-1]["ReviewEndFrame"] = max(
                int(combined[-1]["ReviewEndFrame"]),
                int(item["ReviewEndFrame"]),
            )

    if not combined:
        raise B8Error("NO_PERSISTENT_RADAR_LOSS_CANDIDATE")
    ranked = sorted(
        combined,
        key=lambda item: (
            -(int(item["EvidenceEndFrame"]) - int(item["EvidenceStartFrame"]) + 1),
            int(item["EvidenceStartFrame"]),
        ),
    )[:MAXIMUM_CANDIDATES]
    rank_by_start = {
        int(item["EvidenceStartFrame"]): rank
        for rank, item in enumerate(ranked, start=1)
    }
    selected = sorted(ranked, key=lambda item: int(item["EvidenceStartFrame"]))
    output: List[Dict[str, object]] = []
    for index, item in enumerate(selected, start=1):
        evidence_start = int(item["EvidenceStartFrame"])
        evidence_end = int(item["EvidenceEndFrame"])
        duration = evidence_end - evidence_start + 1
        priority = "CRITICAL" if duration >= 40 else "HIGH" if duration >= 20 else "MEDIUM"
        candidate_id = f"RL{index:04d}"
        output.append(
            {
                **item,
                "CandidateId": candidate_id,
                "CandidateRank": rank_by_start[evidence_start],
                "LossDurationFrames": duration,
                "CandidateScore": duration,
                "PriorityClass": priority,
            }
        )
    return output


def candidate_ids_for_frame(
    frame: int,
    candidates: Sequence[Mapping[str, object]],
) -> str:
    """Return all review candidates whose padded window contains a FrameCount."""

    ids = [
        str(candidate["CandidateId"])
        for candidate in candidates
        if int(candidate["ReviewStartFrame"]) <= frame <= int(candidate["ReviewEndFrame"])
    ]
    return ";".join(ids)


def build_queue(candidates: Sequence[Mapping[str, object]]) -> List[Dict[str, object]]:
    """Map machine candidates to the U003-compatible queue grain."""

    queue: List[Dict[str, object]] = []
    for candidate in candidates:
        candidate_id = str(candidate["CandidateId"])
        duration = int(candidate["LossDurationFrames"])
        start = int(candidate["EvidenceStartFrame"])
        queue.append(
            {
                "EventID": candidate_id,
                "StartFrameCount": candidate["ReviewStartFrame"],
                "PeakFrameCount": start,
                "EndFrameCount": candidate["ReviewEndFrame"],
                "ReviewResult": "",
                "ReviewMemo": "",
                "CandidateId": candidate_id,
                "EvidenceStartFrame": start,
                "EvidenceEndFrame": candidate["EvidenceEndFrame"],
                "CandidateType": "RADAR_TARGET_LOSS_CANDIDATE",
                "CandidateScore": duration,
                "CandidateRank": candidate["CandidateRank"],
                "PriorityClass": candidate["PriorityClass"],
                "PriorityReason": f"PersistentRadarUnusableFrames={duration}",
                "SuggestedReviewMemo": (
                    "Check target lateral displacement, Radar loss onset/recovery, "
                    "Camera target identity, and synchronization around the episode."
                ),
                "SuggestedReviewTags": (
                    "RADAR_TARGET_LOSS;TURNING_CONTEXT;VIDEO_CONFIRMATION_REQUIRED"
                ),
                "SuggestedConfidence": "PROVISIONAL_RULE_EVIDENCE",
                "PrimarySignal": "RadarDistance",
                "RelatedSignals": (
                    "RadarDetectionFlag;Camera1LateralPosition;"
                    "Camera3LateralPosition;OldCanonicalCameraDistance"
                ),
                "ModuleId": PROGRAM_ID,
                "ModuleVersion": PROGRAM_VERSION,
                "AlgorithmStatus": ALGORITHM_STATUS,
                "CandidateStatus": "INVESTIGATION_CANDIDATE",
                "RootCauseClaim": 0,
                "AnchorSemantics": "RADAR_LOSS_ONSET_NOT_SCORE_PEAK",
            }
        )
    return queue


def build_review_rows(
    rows: Sequence[Mapping[str, object]],
    candidates: Sequence[Mapping[str, object]],
) -> Tuple[List[Dict[str, object]], List[Dict[str, object]]]:
    """Build compact and detailed FrameCount-grain tables."""

    compact: List[Dict[str, object]] = []
    detail: List[Dict[str, object]] = []
    for source in rows:
        frame = int(source["FrameCount"])
        candidate_ids = candidate_ids_for_frame(frame, candidates)
        if not candidate_ids:
            continue
        compact.append(
            {
                "FrameCount": frame,
                "CandidateIds": candidate_ids,
                "RadarDistance": source["RadarDistance"],
                "RadarDistanceValid": source["RadarDistanceValid"],
                "RadarDetectionFlag": source["RadarDetectionFlag"],
                "RadarTargetUsable": source["RadarTargetUsable"],
                "RadarLossReason": source["RadarLossReason"],
                "Camera1Distance": source["Camera1Distance"],
                "Camera1LateralPosition": source["Camera1LateralPosition"],
                "Camera3Distance": source["Camera3Distance"],
                "Camera3LateralPosition": source["Camera3LateralPosition"],
                "OldCanonicalCameraDistance": source["OldCanonicalCameraDistance"],
                "OldSelectedCameraRole": source["OldSelectedCameraRole"],
                "CenterlinePreferredRole": source["CenterlinePreferredRole"],
            }
        )
        detail.append({"CandidateIds": candidate_ids, **dict(source)})
    if not compact:
        raise B8Error("REVIEW_ROWS_EMPTY")
    return compact, detail


def validate_package(
    queue: Sequence[Mapping[str, object]],
    review_rows: Sequence[Mapping[str, object]],
    candidates: Sequence[Mapping[str, object]],
) -> Dict[str, object]:
    """Validate U003 queue coverage and bounded candidate semantics."""

    failures: List[str] = []
    frames = {int(row["FrameCount"]) for row in review_rows}
    if len(queue) != len(candidates):
        failures.append("QUEUE_CANDIDATE_COUNT")
    if len(queue) > MAXIMUM_CANDIDATES:
        failures.append("MAXIMUM_CANDIDATES")
    for row in queue:
        start = int(row["StartFrameCount"])
        peak = int(row["PeakFrameCount"])
        end = int(row["EndFrameCount"])
        if not start <= peak <= end:
            failures.append("QUEUE_RANGE_" + str(row["EventID"]))
        if start not in frames or peak not in frames or end not in frames:
            failures.append("QUEUE_COVERAGE_" + str(row["EventID"]))
        if str(row["RootCauseClaim"]) not in {"0", "False", "false"}:
            failures.append("ROOT_CAUSE_CLAIM_" + str(row["EventID"]))
    return {
        "GateStatus": "PASS" if not failures else "FAIL",
        "FailureCount": len(failures),
        "Failures": failures,
        "CandidateCount": len(candidates),
        "ReviewRowCount": len(review_rows),
        "RequiredQueueColumns": list(QUEUE_COLUMNS),
        "RequiredReviewColumns": list(REVIEW_COLUMNS),
    }


def execute(profile_dir: Path, b7u_capsule: Path, output_dir: Path) -> str:
    """Build and validate one traceable U003 review package."""

    accepted_b7u = verify_b7u_capsule(b7u_capsule)
    row_path, sweep_path, quality_path, quality = verify_profile(profile_dir)
    source_rows = read_source_rows(row_path)
    rows = annotate_rows(source_rows)
    candidates = build_candidates(rows)
    queue = build_queue(candidates)
    compact, detail = build_review_rows(rows, candidates)
    validation = validate_package(queue, compact, candidates)
    if validation["GateStatus"] != "PASS":
        raise B8Error("PACKAGE_VALIDATION")

    package_dir = output_dir / "U003_REVIEW_PACKAGE"
    if output_dir.exists():
        shutil.rmtree(output_dir)
    package_dir.mkdir(parents=True)
    queue_path = package_dir / "review_queue.csv"
    compact_path = package_dir / "review_data.csv"
    detail_path = package_dir / "review_evidence_detail.csv"
    validation_path = package_dir / "review_package_validation.json"
    manifest_path = package_dir / "review_manifest.json"
    artifact_path = package_dir / "artifact_manifest.json"

    write_csv(queue_path, QUEUE_COLUMNS, queue)
    write_csv(compact_path, REVIEW_COLUMNS, compact)
    detail_columns = ("CandidateIds",) + SOURCE_COLUMNS + (
        "RadarTargetUsable",
        "RadarLossReason",
    )
    write_csv(detail_path, detail_columns, detail)
    write_json(validation_path, validation)

    manifest = {
        "ArtifactType": "U003_REVIEW_PACKAGE",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "AlgorithmStatus": ALGORITHM_STATUS,
        "FrameDomain": "FrameCount",
        "FrameStart": int(source_rows[0]["FrameCount"]),
        "FrameEnd": int(source_rows[-1]["FrameCount"]),
        "CandidateCount": len(candidates),
        "ReviewRowCount": len(compact),
        "CandidateRule": {
            "MinimumLossFrames": MINIMUM_LOSS_FRAMES,
            "MergeGapFrames": MERGE_GAP_FRAMES,
            "ReviewPaddingBeforeFrames": REVIEW_PADDING_BEFORE,
            "ReviewPaddingAfterFrames": REVIEW_PADDING_AFTER,
            "MaximumCandidates": MAXIMUM_CANDIDATES,
            "RadarMinimumDistanceMetersExclusive": RADAR_MINIMUM_DISTANCE_METERS,
            "RadarMaximumDistanceMetersExclusive": RADAR_MAXIMUM_DISTANCE_METERS,
            "DetectionFlagMustBePositive": True,
        },
        "DecisionBoundary": (
            "REVIEW_CANDIDATES_ONLY_NO_SENSOR_ERROR_OR_ROOT_CAUSE_CLAIM"
        ),
        "B7UAcceptanceCapsule": accepted_b7u,
        "B7AQualityProgramVersion": quality.get("ProgramVersion"),
        "SourceHashes": {
            row_path.name: sha256_file(row_path),
            sweep_path.name: sha256_file(sweep_path),
            quality_path.name: sha256_file(quality_path),
            b7u_capsule.name: sha256_file(b7u_capsule),
        },
        "Files": [
            "review_queue.csv",
            "review_data.csv",
            "review_evidence_detail.csv",
            "review_package_validation.json",
            "review_manifest.json",
            "artifact_manifest.json",
        ],
    }
    write_json(manifest_path, manifest)
    payload_hashes = {
        path.name: sha256_file(path)
        for path in (queue_path, compact_path, detail_path, validation_path, manifest_path)
    }
    artifact = {
        "ArtifactId": "U22B8.RADAR_LOSS.U003_REVIEW_PACKAGE.1",
        "ArtifactType": "U003_REVIEW_PACKAGE",
        "ProducedBy": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "PayloadHashes": payload_hashes,
        "CandidateCount": len(candidates),
        "RootCauseClaim": False,
        "CompletionCode": 0,
    }
    write_json(artifact_path, artifact)

    loss_frames = sum(not bool(int(row["RadarTargetUsable"])) for row in rows)
    capsule = (
        f"B8 N-C{len(candidates)}-F{len(compact)}-L{loss_frames}"
        f"-S{source_rows[0]['FrameCount']}-E{source_rows[-1]['FrameCount']}"
        "-X0 C-0"
    )
    (output_dir / "u22b8_return_capsule.txt").write_text(
        capsule + "\n",
        encoding="utf-8",
    )
    return capsule


def prepare_run(temp_root: Path) -> str:
    """Archive stale top-level TEMP entries while preserving B7U input evidence."""

    if not temp_root.is_dir():
        raise B8Error("TEMP_ROOT_MISSING")
    keep = {
        "A-Archive",
        "U22B7U_OUTPUT",
        "UDP002_U22B8_START_HERE_v0_1_2026-08-24.txt",
        "u002_u22b8_radar_loss_review_package_builder_v0_1.py.txt",
    }
    archive_root = temp_root / "A-Archive"
    archive_root.mkdir(exist_ok=True)
    stamp = __import__("datetime").datetime.utcnow().strftime("%Y%m%dT%H%M%SZ")
    destination = archive_root / ("U22B8_PREPARE_" + stamp)
    moved: List[str] = []
    for path in sorted(temp_root.iterdir(), key=lambda item: item.name.casefold()):
        if path.name in keep:
            continue
        destination.mkdir(parents=True, exist_ok=True)
        target = destination / path.name
        if target.exists():
            raise B8Error("ARCHIVE_COLLISION_" + machine_token(path.name))
        shutil.move(str(path), str(target))
        moved.append(path.name)
    manifest = {
        "Action": "U22B8_PREPARE_RUN",
        "Moved": moved,
        "Kept": sorted(keep),
        "DeletionCount": 0,
    }
    if moved:
        write_json(destination / "u22b8_temp_cleanup_manifest.json", manifest)
    return f"B8CLEAN N-M{len(moved)}-K{len(keep)}-A{int(bool(moved))}-X0 C-0"


def runtime_preflight(profile_dir: Path, b7u_capsule: Path) -> str:
    """Check every real-run dependency and print one short summary."""

    checks = [
        b7u_capsule.is_file(),
        profile_dir.is_dir(),
        (profile_dir / "b7a_camera_lateral_evidence_rows.csv").is_file(),
        (profile_dir / "b7a_abs_lateral_threshold_sweep.csv").is_file(),
        (profile_dir / "b7a_camera_lateral_evidence_quality.json").is_file(),
        (profile_dir / "b7a_return_capsule.txt").is_file(),
    ]
    missing = len(checks) - sum(checks)
    return f"B8PF N-E{len(checks)}-P{sum(checks)}-M{missing}-X{int(bool(missing))} C-{7 if missing else 0}"

def self_test() -> None:
    """Exercise persistence, merging, package generation and hash validation."""

    checks = 0
    with tempfile.TemporaryDirectory(prefix="u22b8_selftest_") as temporary:
        root = Path(temporary)
        profile, b7u = make_synthetic_profile(root)
        output = root / "OUT"
        capsule = execute(profile, b7u, output)
        checks += 1
        if not capsule.startswith("B8 N-C1-") or not capsule.endswith("C-0"):
            raise B8Error("SELFTEST_CAPSULE")
        package = output / "U003_REVIEW_PACKAGE"
        for name in (
            "review_queue.csv",
            "review_data.csv",
            "review_evidence_detail.csv",
            "review_package_validation.json",
            "review_manifest.json",
            "artifact_manifest.json",
        ):
            checks += 1
            if not (package / name).is_file():
                raise B8Error("SELFTEST_FILE_" + machine_token(name))
        with (package / "review_queue.csv").open(
            "r", encoding="utf-8-sig", newline=""
        ) as stream:
            queue = list(csv.DictReader(stream))
        checks += 1
        if len(queue) != 1:
            raise B8Error("SELFTEST_QUEUE_COUNT")
        checks += 1
        if queue[0]["PeakFrameCount"] != "1010":
            raise B8Error("SELFTEST_ONSET_ANCHOR")
        checks += 1
        if queue[0]["RootCauseClaim"] != "0":
            raise B8Error("SELFTEST_ROOT_CAUSE")
        validation = load_json(package / "review_package_validation.json", "SELFTEST_VALIDATION")
        checks += 1
        if validation.get("GateStatus") != "PASS":
            raise B8Error("SELFTEST_VALIDATION_STATUS")

        rows = read_source_rows(profile / "b7a_camera_lateral_evidence_rows.csv")
        annotated = annotate_rows(rows)
        checks += 1
        if sum(not bool(int(row["RadarTargetUsable"])) for row in annotated) != 10:
            raise B8Error("SELFTEST_LOSS_COUNT")
        checks += 1
        if build_candidates(annotated)[0]["LossDurationFrames"] != 10:
            raise B8Error("SELFTEST_PERSISTENCE")

        tampered = profile / "b7a_camera_lateral_evidence_rows.csv"
        tampered.write_text(tampered.read_text(encoding="utf-8") + "\n", encoding="utf-8")
        checks += 1
        try:
            verify_profile(profile)
        except B8Error as error:
            if not error_reason(error).startswith("B7A_HASH_"):
                raise
        else:
            raise B8Error("SELFTEST_HASH_FAIL_OPEN")

    print(f"UDP002-U22B8-SELFTEST-PASS-{checks} C-0")


def write_failure(output_dir: Path, stage: str, error: BaseException) -> None:
    """Write detailed diagnostics and emit two compact failure lines."""

    reason = machine_token(error_reason(error))
    diagnostic = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "Stage": stage,
        "ErrorType": type(error).__name__,
        "Reason": error_reason(error),
        "CompletionCode": 7,
        "RequiredFiles": [
            str(DEFAULT_B7U_CAPSULE),
            str(DEFAULT_PROFILE_DIR / "b7a_camera_lateral_evidence_rows.csv"),
            str(DEFAULT_PROFILE_DIR / "b7a_abs_lateral_threshold_sweep.csv"),
            str(DEFAULT_PROFILE_DIR / "b7a_camera_lateral_evidence_quality.json"),
            str(DEFAULT_PROFILE_DIR / "b7a_return_capsule.txt"),
        ],
        "Recovery": (
            "Keep the accepted U22B7U_OUTPUT intact, run --self-test, and use "
            "the exact RequiredFiles paths before retrying --execute."
        ),
    }
    try:
        output_dir.mkdir(parents=True, exist_ok=True)
        write_json(output_dir / "u22b8_failure_diagnostic.json", diagnostic)
    except Exception:
        pass
    print(f"B8E K-{reason}-J{machine_token(stage, 24)}-X1 C-7")
    print(f"B8D K-{reason}-D{type(error).__name__}-X0 C-0")


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Command-line entry point."""

    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare-run", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    mode.add_argument("--preflight-only", action="store_true")
    parser.add_argument("--profile-dir", type=Path, default=DEFAULT_PROFILE_DIR)
    parser.add_argument("--b7u-capsule", type=Path, default=DEFAULT_B7U_CAPSULE)
    parser.add_argument("--output-dir", type=Path, default=DEFAULT_OUTPUT_DIR)
    parser.add_argument("--temp-root", type=Path, default=DEFAULT_TEMP_ROOT)
    args = parser.parse_args(argv)

    stage = "START"
    try:
        if args.prepare_run:
            stage = "PREPARE"
            print(prepare_run(args.temp_root))
            return 0
        if args.self_test:
            stage = "SELFTEST"
            self_test()
            summary = runtime_preflight(args.profile_dir, args.b7u_capsule)
            print(summary)
            return 0 if summary.endswith("C-0") else 7
        if args.preflight_only:
            stage = "PREFLIGHT"
            summary = runtime_preflight(args.profile_dir, args.b7u_capsule)
            print(summary)
            return 0 if summary.endswith("C-0") else 7
        stage = "PREFLIGHT"
        summary = runtime_preflight(args.profile_dir, args.b7u_capsule)
        if not summary.endswith("C-0"):
            raise B8Error("RUNTIME_PREFLIGHT")
        stage = "EXECUTE"
        print(execute(args.profile_dir, args.b7u_capsule, args.output_dir))
        return 0
    except Exception as error:
        write_failure(args.output_dir, stage, error)
        return 7


if __name__ == "__main__":
    sys.exit(main())
