#!/usr/bin/env python3
"""Controlled schema-completion layer for U001 Gate 7 v0.5.

This file intentionally reuses the independently tested v0.4 dynamic package
engine.  It changes only the final Update Stream handoff required by the
accepted G8R3 correction specification:

* complete the final D1 schema in deterministic order;
* set ClockDomain to FRAMECOUNT_VIDEO;
* keep NativeTimestamp and MessageCounter blank for the current profile;
* preserve every upstream data cell semantically and preserve row order;
* record explicit schema-completion evidence.

The v0.4 file must be in the same directory.  No CAN decoding, frame
assignment, validity classification, holding, or U002 analysis is performed.
"""

from __future__ import annotations

import csv
import importlib.machinery
import importlib.util
import os
from pathlib import Path
import shutil
import sys
import tempfile
import traceback
from typing import Dict, List, Mapping, Optional, Sequence


PROGRAM_VERSION = "0.5"
ENGINE_BASENAME = "u001_output_package_assembler_v0_4.py.txt"
ALLOWED_IDENTITY_STATUS = {"PROVISIONAL_SMOKE", "CONTROLLED_REPLAY"}
COMPLETION_COLUMNS = ("ClockDomain", "NativeTimestamp", "MessageCounter")


def load_engine():
    """Load the adjacent audited v0.4 engine from its transport filename."""

    engine_path = Path(__file__).resolve().with_name(ENGINE_BASENAME)
    if not engine_path.is_file():
        raise RuntimeError(f"ENGINE_MISSING:{ENGINE_BASENAME}")
    loader = importlib.machinery.SourceFileLoader("u001_g7_v04_engine", str(engine_path))
    specification = importlib.util.spec_from_loader(loader.name, loader)
    if specification is None:
        raise RuntimeError("ENGINE_SPEC_MISSING")
    module = importlib.util.module_from_spec(specification)
    loader.exec_module(module)
    return module


ENGINE = load_engine()
ORIGINAL_COPY = ENGINE.copy_binary_verified
ORIGINAL_QUALITY = ENGINE.build_quality_rows
ORIGINAL_WRITE_JSON = ENGINE.write_json
ORIGINAL_MAKE_CONFIG = ENGINE.make_config

SCHEMA_STATE: Dict[str, object] = {
    "InputRows": 0,
    "OutputRows": 0,
    "InputMissingColumns": [],
    "ClockDomainBad": 0,
    "NativeTimestampNonblank": 0,
    "MessageCounterNonblank": 0,
}


def reset_schema_state() -> None:
    """Reset evidence before inspecting one source Update Stream."""

    SCHEMA_STATE.update(
        {
            "InputRows": 0,
            "OutputRows": 0,
            "InputMissingColumns": [],
            "ClockDomainBad": 0,
            "NativeTimestampNonblank": 0,
            "MessageCounterNonblank": 0,
        }
    )


def final_header(source_header: Sequence[str]) -> List[str]:
    """Create exact contract-leading order and preserve extra-column order."""

    if len(source_header) != len(set(source_header)):
        raise ENGINE.PackageError("UPDATE_DUPLICATE_COLUMN", len(source_header))
    missing_noncompletion = [
        name
        for name in ENGINE.D1_REQUIRED
        if name not in source_header and name not in COMPLETION_COLUMNS
    ]
    if missing_noncompletion:
        raise ENGINE.PackageError("UPDATE_COLUMN", missing_noncompletion[0])
    extras = [name for name in source_header if name not in ENGINE.D1_REQUIRED]
    return list(ENGINE.D1_REQUIRED) + extras


def inspect_updates_v05(path: Path) -> Dict[str, object]:
    """Validate the G4 source while allowing only three controlled omissions."""

    reset_schema_state()
    source_header = ENGINE.read_header(path)
    output_header = final_header(source_header)
    missing = [name for name in COMPLETION_COLUMNS if name not in source_header]
    SCHEMA_STATE["InputMissingColumns"] = missing

    index_position = source_header.index("SourceRowIndex")
    positions = {name: source_header.index(name) for name in COMPLETION_COLUMNS if name in source_header}
    previous: Optional[int] = None
    rows = 0

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        next(reader, None)
        for row in reader:
            if len(row) != len(source_header):
                raise ENGINE.PackageError("UPDATE_ROW_WIDTH", rows)
            try:
                current = int(row[index_position].strip())
            except Exception as exc:
                raise ENGINE.PackageError("SOURCE_ROW_INDEX", rows) from exc
            if previous is not None and current < previous:
                raise ENGINE.PackageError("SOURCE_ORDER_DECREASE", current)
            previous = current

            if "ClockDomain" in positions:
                value = row[positions["ClockDomain"]].strip()
                if value not in {"", "FRAMECOUNT_VIDEO"}:
                    SCHEMA_STATE["ClockDomainBad"] = int(SCHEMA_STATE["ClockDomainBad"]) + 1
            if "NativeTimestamp" in positions and row[positions["NativeTimestamp"]].strip():
                SCHEMA_STATE["NativeTimestampNonblank"] = int(SCHEMA_STATE["NativeTimestampNonblank"]) + 1
            if "MessageCounter" in positions and row[positions["MessageCounter"]].strip():
                SCHEMA_STATE["MessageCounterNonblank"] = int(SCHEMA_STATE["MessageCounterNonblank"]) + 1
            rows += 1

    if rows == 0:
        raise ENGINE.PackageError("UPDATE_EMPTY", path.name)
    if int(SCHEMA_STATE["ClockDomainBad"]):
        raise ENGINE.PackageError("CLOCK_DOMAIN_VALUE", SCHEMA_STATE["ClockDomainBad"])
    if int(SCHEMA_STATE["NativeTimestampNonblank"]):
        raise ENGINE.PackageError("NATIVE_TIMESTAMP_NONBLANK", SCHEMA_STATE["NativeTimestampNonblank"])
    if int(SCHEMA_STATE["MessageCounterNonblank"]):
        raise ENGINE.PackageError("MESSAGE_COUNTER_NONBLANK", SCHEMA_STATE["MessageCounterNonblank"])

    SCHEMA_STATE["InputRows"] = rows
    return {
        "Rows": rows,
        "Header": output_header,
        "SourceHeader": source_header,
        "LastSourceRowIndex": previous,
        "SchemaCompletionRequired": bool(missing or source_header != output_header),
    }


def write_completed_updates(source: Path, destination: Path, expected_hash: str) -> None:
    """Stream G4 into final D1 without sorting, filling measurements, or inference."""

    if ENGINE.sha256_file(source) != expected_hash:
        raise ENGINE.PackageError("UPDATE_SOURCE_HASH", source.name)

    source_header = ENGINE.read_header(source)
    output_header = final_header(source_header)

    # Preserve exact bytes when the source already satisfies the final schema.
    if list(source_header) == output_header and not SCHEMA_STATE["InputMissingColumns"]:
        ORIGINAL_COPY(source, destination, expected_hash)
        SCHEMA_STATE["OutputRows"] = ENGINE.count_csv_records(destination)
        return

    temporary = destination.with_name(destination.name + ".tmp")
    rows = 0
    with source.open("r", encoding="utf-8-sig", newline="") as source_stream, temporary.open(
        "w", encoding="utf-8", newline=""
    ) as destination_stream:
        reader = csv.DictReader(source_stream)
        if reader.fieldnames != source_header:
            raise ENGINE.PackageError("UPDATE_HEADER_CHANGED", source.name)
        writer = csv.DictWriter(
            destination_stream,
            fieldnames=output_header,
            extrasaction="raise",
            lineterminator="\n",
        )
        writer.writeheader()
        for row in reader:
            if None in row:
                raise ENGINE.PackageError("UPDATE_ROW_WIDTH", rows)
            completed = {name: row.get(name, "") for name in output_header}
            completed["ClockDomain"] = "FRAMECOUNT_VIDEO"
            completed["NativeTimestamp"] = ""
            completed["MessageCounter"] = ""
            writer.writerow(completed)
            rows += 1
    os.replace(temporary, destination)

    if rows != int(SCHEMA_STATE["InputRows"]):
        raise ENGINE.PackageError("SCHEMA_COMPLETION_ROW_COUNT", f"{SCHEMA_STATE['InputRows']}:{rows}")
    SCHEMA_STATE["OutputRows"] = rows


def copy_binary_v05(source: Path, destination: Path, expected_hash: str) -> None:
    """Complete D1 only; keep D2 byte-identical through the v0.4 engine."""

    if destination.name == "u001_decoded_updates.csv":
        write_completed_updates(source, destination, expected_hash)
        return
    ORIGINAL_COPY(source, destination, expected_hash)


def build_quality_rows_v05(
    inputs: Mapping[str, Path],
    catalog,
    update_info: Mapping[str, object],
    frame_info: Mapping[str, object],
    review_column_count: int,
):
    """Add frozen schema-completion evidence to the tidy quality table."""

    rows = ORIGINAL_QUALITY(inputs, catalog, update_info, frame_info, review_column_count)
    required_metrics = {
        "update_required_column_missing_count": 0,
        "clock_domain_bad_value_count": int(SCHEMA_STATE["ClockDomainBad"]),
        "native_timestamp_nonblank_count": int(SCHEMA_STATE["NativeTimestampNonblank"]),
        "message_counter_nonblank_count": int(SCHEMA_STATE["MessageCounterNonblank"]),
        "update_row_count_before_schema_completion": int(SCHEMA_STATE["InputRows"]),
        "update_row_count_after_schema_completion": int(SCHEMA_STATE["InputRows"]),
    }
    rows = [row for row in rows if not (row[0] == "GLOBAL" and row[2] in required_metrics)]
    for metric, value in required_metrics.items():
        rows.append(
            [
                "GLOBAL",
                "",
                metric,
                value,
                "",
                "G7",
                "PASS",
                "Controlled final Update schema completion",
            ]
        )
    for row in rows:
        if row[0] == "GLOBAL" and row[2] == "update_source_hash_match":
            row[7] = "Accepted G4 source hash verified before controlled schema completion"
    return rows


def write_json_v05(path: Path, value: Mapping[str, object]) -> None:
    """Add portable manifest evidence and apply the accepted identity rule."""

    if path.name != "u001_run_manifest.json":
        ORIGINAL_WRITE_JSON(path, value)
        return

    manifest = dict(value)
    identity_status = str(manifest.get("IdentityStatus", ""))
    if identity_status not in ALLOWED_IDENTITY_STATUS:
        raise ENGINE.PackageError("IDENTITY_STATUS", identity_status)
    warnings = [
        str(item)
        for item in manifest.get("Warnings", [])
        if not str(item).replace("_", "-").upper().startswith("IDENTITYSTATUS-")
    ]
    if identity_status == "PROVISIONAL_SMOKE":
        warnings.append("IdentityStatus-PROVISIONAL_SMOKE")
    manifest["Warnings"] = warnings
    manifest["IdentityWarningConsistency"] = 1
    manifest["ProgramVersion"] = PROGRAM_VERSION
    manifest["UpdateSchemaCompletion"] = {
        "Mode": "CONTROLLED_FINAL_SCHEMA_COMPLETION",
        "InputRows": int(SCHEMA_STATE["InputRows"]),
        "OutputRows": int(SCHEMA_STATE["OutputRows"] or SCHEMA_STATE["InputRows"]),
        "InputMissingColumns": list(SCHEMA_STATE["InputMissingColumns"]),
        "ClockDomain": "FRAMECOUNT_VIDEO",
        "NativeTimestamp": "BLANK_CURRENT_PROFILE",
        "MessageCounter": "BLANK_CURRENT_PROFILE",
        "MeasurementFillApplied": False,
        "SortingApplied": False,
    }
    ORIGINAL_WRITE_JSON(path, manifest)


def make_config_v05(*args, **kwargs):
    """Create v0.5 config and explicitly declare controlled D1 rewriting."""

    config = ORIGINAL_MAKE_CONFIG(*args, **kwargs)
    safety = config.setdefault("Safety", {})
    safety["PreserveAcceptedUpdateBytes"] = False
    safety["PreserveAcceptedUpdateCellSemantics"] = True
    safety["ControlledUpdateSchemaCompletion"] = True
    config["ConfigFingerprint"] = ENGINE.config_fingerprint(config)
    output_path = Path(args[1]) if len(args) > 1 else Path(kwargs["out"])
    ORIGINAL_WRITE_JSON(output_path, config)
    return config


def recover_b14_v05(
    udp001_root: Path,
    failed_run_root: Path,
    out_root: Path,
    dataset_id: str,
    run_id: str,
    segment_id: str,
    source_id: str,
):
    """Rebuild B14 with the v0.5 schema completion and no legacy G7 execution."""

    if out_root.exists():
        raise ENGINE.PackageError("RECOVERY_OUTPUT_EXISTS", out_root.name)
    manifest_path = failed_run_root / "03_RUN" / "CONTROL_OUTPUT" / "u001_g8a2_orchestration_manifest.json"
    old_manifest = ENGINE.read_json(manifest_path, "B14_MANIFEST")
    stages = old_manifest.get("Stages")
    passed = sum(
        isinstance(item, dict) and item.get("Success") is True
        for item in stages
    ) if isinstance(stages, list) else 0
    if old_manifest.get("Status") != "FAILURE" or passed != 8:
        raise ENGINE.PackageError("B14_FAILURE_IDENTITY", f"{old_manifest.get('Status')}:{passed}")

    protected_roots = (udp001_root / "01_PROGRAMS", udp001_root / "02_CONTROL")
    protected_before = [ENGINE.snapshot_small_tree(root) for root in protected_roots]

    out_root.mkdir(parents=True)
    config_path = out_root / "u001_g7v05_package_config.json"
    ENGINE.make_config(
        failed_run_root / "03_RUN",
        config_path,
        dataset_id,
        run_id,
        segment_id,
        source_id,
        "CONTROLLED_REPLAY",
    )
    capsule, report = ENGINE.build_package(config_path, out_root / "G7V05_BUILD")

    protected_after = [ENGINE.snapshot_small_tree(root) for root in protected_roots]
    if protected_before != protected_after:
        raise ENGINE.PackageError("ACCEPTED_U001_CHANGED", "PROGRAM_OR_CONTROL")

    line1 = (
        f"B14R4 N-G7V05-F6-C{report['CatalogSignals']}-V{report['FrameSignals']}"
        f"-U{report['UpdateRows']}-R{report['FrameRows']}-H1-X0 C-0"
    )
    line2 = (
        f"B14R4S N-M{len(SCHEMA_STATE['InputMissingColumns'])}"
        f"-C{SCHEMA_STATE['ClockDomainBad']}-N{SCHEMA_STATE['NativeTimestampNonblank']}"
        f"-G{SCHEMA_STATE['MessageCounterNonblank']}-X0 C-0"
    )
    ENGINE.write_json(
        out_root / "u22b1b14r4_recovery_manifest_v0_1.json",
        {
            "CompletionCode": 0,
            "ProgramId": ENGINE.PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "ExecutionMode": "G7V05_CONTROLLED_SCHEMA_COMPLETION_FROM_ACCEPTED_G1_G6",
            "LegacyG7Executed": False,
            "AcceptedU001Changed": False,
            "OldSuccessfulStageCount": passed,
            "InputMissingCompletionColumns": list(SCHEMA_STATE["InputMissingColumns"]),
            "ReturnCapsule": [line1, line2],
        },
    )
    ENGINE.write_text(out_root / "u22b1b14r4_return_capsule.txt", line1 + "\n" + line2 + "\n")
    return line1, line2


def install_v05_behavior() -> None:
    """Install only the accepted narrow corrections into the v0.4 engine."""

    ENGINE.PROGRAM_VERSION = PROGRAM_VERSION
    ENGINE.inspect_updates = inspect_updates_v05
    ENGINE.copy_binary_verified = copy_binary_v05
    ENGINE.build_quality_rows = build_quality_rows_v05
    ENGINE.write_json = write_json_v05
    ENGINE.make_config = make_config_v05
    ENGINE.recover_b14 = recover_b14_v05


def self_test() -> int:
    """Run the original 40 checks and sixteen v0.5 schema/build checks."""

    # First run the dependency's unchanged suite.  Its synthetic identity is
    # intentionally outside the narrower production identity vocabulary.
    ENGINE.self_test()
    install_v05_behavior()

    checks: List[str] = []

    def check(name: str, condition: bool) -> None:
        if not condition:
            raise AssertionError(name)
        checks.append(name)

    with tempfile.TemporaryDirectory() as temp_name:
        root = Path(temp_name)
        source = root / "source.csv"
        output = root / "output.csv"
        header = [name for name in ENGINE.D1_REQUIRED if name not in COMPLETION_COLUMNS]
        row = {name: "" for name in header}
        row.update(
            {
                "SourceRowIndex": "1",
                "SourceId": "SOURCE_001",
                "SourceType": "CAN",
                "NativeSequence": "1",
                "FrameCount": "7400",
                "SyncKey": "7400",
                "SyncQuality": "VALID",
                "FrameAssignmentStatus": "ASSIGNED",
                "FrameAssignmentAgeRows": "0",
                "CAN_ID": "0x221",
                "Channel": "2",
                "MessageName": "CAMERA",
                "CanonicalName": "CAMERA_DISTANCE",
                "RawCode": "123",
                "PhysicalValue": "12.3",
                "ValueKind": "FLOAT",
                "SignalRole": "MEASUREMENT",
                "Valid": "1",
                "MissingReason": "VALID_UPDATE",
            }
        )
        ENGINE.write_csv(source, header, ([row[name] for name in header],))
        info = inspect_updates_v05(source)
        check("T41_THREE_MISSING", len(SCHEMA_STATE["InputMissingColumns"]) == 3)
        check("T42_ONE_ROW", info["Rows"] == 1)
        check("T43_FINAL_ORDER", info["Header"][:25] == list(ENGINE.D1_REQUIRED))
        write_completed_updates(source, output, ENGINE.sha256_file(source))
        check("T44_OUTPUT_ROW", ENGINE.count_csv_records(output) == 1)
        with output.open("r", encoding="utf-8", newline="") as stream:
            completed = next(csv.DictReader(stream))
        check("T45_CLOCK", completed["ClockDomain"] == "FRAMECOUNT_VIDEO")
        check("T46_NATIVE_BLANK", completed["NativeTimestamp"] == "")
        check("T47_COUNTER_BLANK", completed["MessageCounter"] == "")
        check("T48_VALUE_PRESERVED", completed["PhysicalValue"] == "12.3")

        gate_root = root / "gate_inputs"
        gate_root.mkdir()
        ENGINE.make_synthetic_gate_inputs(gate_root, included_count=3)
        update_path = gate_root / "u001_g4_semantic_validity_updates.csv"
        with update_path.open("r", encoding="utf-8", newline="") as stream:
            source_rows = list(csv.DictReader(stream))
        reduced_header = [name for name in ENGINE.D1_REQUIRED if name not in COMPLETION_COLUMNS]
        ENGINE.write_csv(
            update_path,
            reduced_header,
            ([row.get(name, "") for name in reduced_header] for row in source_rows),
        )
        config_path = root / "v05_config.json"
        ENGINE.make_config(
            gate_root,
            config_path,
            "SELFTEST_DATASET",
            "SELFTEST_RUN",
            "SELFTEST_SEGMENT",
            "SOURCE_001",
            "CONTROLLED_REPLAY",
        )
        capsule, report = ENGINE.build_package(config_path, root / "v05_build")
        package = Path(report["PackagePath"])
        check("T49_BUILD_CAPSULE", capsule.endswith("C-0"))
        check("T50_SIX_FILES", len(list(package.iterdir())) == 6)
        check("T51_D1_HEADER", ENGINE.read_header(package / "u001_decoded_updates.csv")[:25] == list(ENGINE.D1_REQUIRED))
        check("T52_D1_ROWS", ENGINE.count_csv_records(package / "u001_decoded_updates.csv") == len(source_rows))
        manifest = ENGINE.read_json(package / "u001_run_manifest.json", "SELFTEST_MANIFEST")
        check("T53_MANIFEST_VERSION", manifest["ProgramVersion"] == PROGRAM_VERSION)
        check("T54_MANIFEST_SCHEMA", len(manifest["UpdateSchemaCompletion"]["InputMissingColumns"]) == 3)
        quality = ENGINE.read_control(package / "u001_quality_report.csv", ENGINE.D4_COLUMNS, "SELFTEST_QUALITY")
        metrics = {row["Metric"]: row["Value"] for row in quality if row["Scope"] == "GLOBAL"}
        check("T55_QUALITY_ROWS", metrics["update_row_count_before_schema_completion"] == metrics["update_row_count_after_schema_completion"])
        check("T56_IDENTITY", manifest["IdentityWarningConsistency"] == 1)

    if len(checks) != 16:
        raise AssertionError("V05_SELFTEST_COUNT")
    print("UDP001-G7V05-SELFTEST-PASS-56")
    return 0


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run v0.5 with bounded integrated diagnostics."""

    arguments = list(sys.argv[1:] if argv is None else argv)
    try:
        if arguments == ["--self-test"]:
            return self_test()
        install_v05_behavior()
        return ENGINE.main(arguments)
    except Exception as exc:
        if isinstance(exc, ENGINE.PackageError):
            reason, detail = exc.reason, exc.detail
        else:
            reason = ENGINE.machine_token(type(exc).__name__)
            detail = ENGINE.machine_token(str(exc), 180)
        print(f"B14R4E K-{reason}-X1 C-7")
        print(f"B14R4D D-{detail} C-0")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
