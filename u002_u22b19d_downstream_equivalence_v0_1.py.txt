"""Validate S6/S7 business-result equivalence across two accepted U002 Runs."""

PROGRAM_ID = "CameraRadar.B19D.DownstreamEquivalence"
PROGRAM_VERSION = "0.1.0"
PROGRAM_NAME = "u002_u22b19d_downstream_equivalence_v0_1.py.txt"
GUIDE_NAME = "UDP002_U22B19D_START_HERE_CN_v0_1_2026-08-26.txt"
B19B_RUN_ID = "U22B19B_S1_WIDTH_REAL_REPLAY_V01"
B19C_RUN_ID = "U22B19C_S1_V04_FULL_GOLDEN_V03"
OUTPUT_ID = "U22B19D_DOWNSTREAM_EQUIVALENCE_V01"
S6_MODULE_ID = "CameraRadar.S6.RadarMeasurementLossGeometryCandidateSynthesis"
S7_MODULE_ID = "CameraRadar.S7.CandidateReviewBatchExport"

# These values describe execution identity, not an engineering observation.
# They may legitimately differ when the same business computation is replayed.
LINEAGE_KEYS = {
    "ArtifactId",
    "ParentArtifactId",
    "ParentArtifactIds",
    "RunId",
    "SourceArtifactId",
    "SourceArtifactIds",
    "SourceRunId",
    "TargetRunId",
}


class B19DError(RuntimeError):
    """One controlled B19D technical rejection."""

    def __init__(self, reason: object, detail: object = "NA") -> None:
        self.reason = token(reason)
        self.detail = detail_token(detail)
        super().__init__(self.reason)


def token(value: object, limit: int = 120) -> str:
    """Return one bounded machine token."""

    text = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (text or "NA")[:limit]


def detail_token(value: object, limit: int = 180) -> str:
    """Keep hyphens available as capsule separators."""

    text = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper()).strip("_.-")
    return (text or "NA")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize deterministic UTF-8 JSON."""

    return (
        json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n"
    ).encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file without loading it into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_json(path: Path, label: str) -> Dict[str, Any]:
    """Load one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise B19DError(label + "_MISSING", path.name)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise B19DError(label + "_JSON", path.name) from error
    if not isinstance(value, dict):
        raise B19DError(label + "_TYPE", path.name)
    return value


def require_workspace(value: str) -> Path:
    """Resolve the fixed company U002 workspace."""

    root = Path(value).resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise B19DError("WORKSPACE_ROOT", root.name)
    return root


def transfer_paths(root: Path) -> Tuple[Path, Path]:
    """Require the exact two B19D transfer files in 04_TEMP."""

    temp = root / "04_TEMP"
    program = temp / PROGRAM_NAME
    guide = temp / GUIDE_NAME
    for path in (program, guide):
        if not path.is_file() or path.is_symlink():
            raise B19DError("TRANSFER_FILE", path.name)
    return program, guide


def archive_previous_temp(root: Path) -> Tuple[int, int]:
    """Archive obsolete TEMP entries while preserving this node and A-Archive."""

    temp = root / "04_TEMP"
    temp.mkdir(parents=True, exist_ok=True)
    keep = {"A-Archive", PROGRAM_NAME, GUIDE_NAME}
    stale = sorted(path for path in temp.iterdir() if path.name not in keep)
    if not stale:
        return 0, 0
    destination = (
        temp
        / "A-Archive"
        / ("U22B19D_PREPARE_" + datetime.datetime.now().strftime("%Y%m%d_%H%M%S_%f"))
    )
    destination.mkdir(parents=True, exist_ok=False)
    moved = 0
    locked = 0
    for source in stale:
        try:
            shutil.move(str(source), str(destination / source.name))
            moved += 1
        except PermissionError:
            locked += 1
    if moved == 0:
        try:
            destination.rmdir()
        except OSError:
            pass
    return moved, locked


def safe_payload_path(artifact_root: Path, relative_text: object) -> Path:
    """Resolve a declared Payload without allowing absolute paths or traversal."""

    relative = Path(str(relative_text))
    if relative.is_absolute() or ".." in relative.parts:
        raise B19DError("PAYLOAD_RELATIVE_PATH", relative_text)
    path = (artifact_root / relative).resolve()
    try:
        path.relative_to(artifact_root.resolve())
    except ValueError as error:
        raise B19DError("PAYLOAD_ESCAPE", relative_text) from error
    if not path.is_file() or path.is_symlink():
        raise B19DError("PAYLOAD_FILE", relative_text)
    return path


def csv_row_count(path: Path) -> int:
    """Count logical CSV data rows, including files with quoted newlines."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        try:
            next(reader)
        except StopIteration:
            return 0
        return sum(1 for _ in reader)


def normalize_lineage(value: object) -> object:
    """Remove only execution-lineage fields from JSON business evidence."""

    if isinstance(value, dict):
        return {
            key: normalize_lineage(item)
            for key, item in sorted(value.items())
            if key not in LINEAGE_KEYS
        }
    if isinstance(value, list):
        return [normalize_lineage(item) for item in value]
    return value


def semantic_digest(path: Path, media_type: str) -> str:
    """Hash business content while allowing only documented lineage changes."""

    suffix = path.suffix.lower()
    digest = hashlib.sha256()
    if media_type == "text/csv" or suffix == ".csv":
        with path.open("r", encoding="utf-8-sig", newline="") as stream:
            rows = list(csv.reader(stream))
        digest.update(canonical_bytes(rows))
        return digest.hexdigest()
    if media_type == "application/json" or suffix == ".json":
        value = json.loads(path.read_text(encoding="utf-8-sig"))
        digest.update(canonical_bytes(normalize_lineage(value)))
        return digest.hexdigest()
    if media_type == "application/x-ndjson" or suffix == ".jsonl":
        values: List[object] = []
        for line in path.read_text(encoding="utf-8-sig").splitlines():
            if line.strip():
                values.append(normalize_lineage(json.loads(line)))
        digest.update(canonical_bytes(values))
        return digest.hexdigest()
    return sha256_file(path)


def find_artifact(run_root: Path, module_id: str, stage: str) -> Tuple[Path, Dict[str, Any]]:
    """Find one exact accepted Artifact by Module identity and Stage."""

    selected: List[Tuple[Path, Dict[str, Any]]] = []
    artifact_root = run_root / "03_ARTIFACTS"
    if not artifact_root.is_dir():
        raise B19DError("ARTIFACT_ROOT", run_root.name)
    for path in sorted(artifact_root.rglob("artifact_manifest.json")):
        manifest = load_json(path, "ARTIFACT_MANIFEST")
        if manifest.get("ProducerModuleId") == module_id and manifest.get("Stage") == stage:
            selected.append((path.parent, manifest))
    if len(selected) != 1:
        raise B19DError("ARTIFACT_COUNT", f"{stage}-{len(selected)}")
    return selected[0]


def validate_payload_manifest(
    artifact_root: Path,
    manifest: Mapping[str, object],
    label: str,
) -> Dict[str, Dict[str, object]]:
    """Validate every declared Payload before using it for comparison."""

    payloads = manifest.get("Payloads")
    if not isinstance(payloads, list) or not payloads:
        raise B19DError(label + "_PAYLOADS")
    records: Dict[str, Dict[str, object]] = {}
    for index, item in enumerate(payloads, start=1):
        if not isinstance(item, dict):
            raise B19DError(label + "_PAYLOAD_ITEM", index)
        relative = str(item.get("RelativePath", ""))
        if not relative or relative in records:
            raise B19DError(label + "_PAYLOAD_PATH", relative or index)
        path = safe_payload_path(artifact_root, relative)
        actual_sha = sha256_file(path)
        if actual_sha != item.get("Sha256"):
            raise B19DError(label + "_PAYLOAD_SHA", relative)
        actual_bytes = path.stat().st_size
        if item.get("ByteSize") != actual_bytes:
            raise B19DError(label + "_PAYLOAD_BYTES", relative)
        declared_rows = item.get("RowCount")
        media_type = str(item.get("MediaType", ""))
        if declared_rows is not None:
            if media_type != "text/csv" and path.suffix.lower() != ".csv":
                raise B19DError(label + "_ROW_MEDIA", relative)
            if csv_row_count(path) != declared_rows:
                raise B19DError(label + "_PAYLOAD_ROWS", relative)
        records[relative] = {
            "Path": path,
            "Sha256": actual_sha,
            "ByteSize": actual_bytes,
            "RowCount": declared_rows,
            "MediaType": media_type,
            "SemanticSha256": semantic_digest(path, media_type),
        }
    return records


def compare_payload_sets(
    stage: str,
    source: Mapping[str, Mapping[str, object]],
    target: Mapping[str, Mapping[str, object]],
) -> List[Dict[str, object]]:
    """Compare declared Payload paths and their semantic content."""

    rows: List[Dict[str, object]] = []
    for relative in sorted(set(source) | set(target)):
        old = source.get(relative)
        new = target.get(relative)
        if old is None:
            status = "TARGET_ONLY"
        elif new is None:
            status = "SOURCE_ONLY"
        elif old["SemanticSha256"] != new["SemanticSha256"]:
            status = "BUSINESS_MISMATCH"
        elif old["Sha256"] == new["Sha256"]:
            status = "EXACT"
        else:
            status = "LINEAGE_NORMALIZED"
        rows.append(
            {
                "Stage": stage,
                "RelativePath": relative,
                "Status": status,
                "SourceSha256": "" if old is None else old["Sha256"],
                "TargetSha256": "" if new is None else new["Sha256"],
                "SourceSemanticSha256": "" if old is None else old["SemanticSha256"],
                "TargetSemanticSha256": "" if new is None else new["SemanticSha256"],
                "SourceRowCount": "" if old is None else old["RowCount"],
                "TargetRowCount": "" if new is None else new["RowCount"],
            }
        )
    return rows


def status_succeeded(run_root: Path, label: str) -> Dict[str, Any]:
    """Require one immutable completed Run."""

    status = load_json(run_root / "run_status.json", label + "_STATUS")
    if status.get("Status") != "SUCCEEDED":
        raise B19DError(label + "_NOT_SUCCEEDED", status.get("Status"))
    return status


def resolve_context(root: Path) -> Dict[str, object]:
    """Resolve B19B source, B19C target and exact S6/S7 Artifacts."""

    transfer_paths(root)
    b19b_summary_path = root / "07_OUTPUT" / B19B_RUN_ID / "b19b_real_replay_summary.json"
    b19b_summary = load_json(b19b_summary_path, "B19B_SUMMARY")
    if b19b_summary.get("Status") != "SUCCEEDED":
        raise B19DError("B19B_STATUS")
    source_run_id = str(b19b_summary.get("SourceGoldenRunId", ""))
    source_root = root / "07_OUTPUT" / source_run_id
    target_root = root / "07_OUTPUT" / B19C_RUN_ID
    status_succeeded(source_root, "SOURCE")
    status_succeeded(target_root, "TARGET")

    receipt_path = (
        target_root
        / "02_DIAGNOSTICS/U22B19C_CONTROLLER/b19c_full_golden_replay_summary_v0_5.json"
    )
    receipt = load_json(receipt_path, "B19C_RECEIPT")
    if receipt.get("Status") != "SUCCEEDED" or receipt.get("TargetRunId") != B19C_RUN_ID:
        raise B19DError("B19C_RECEIPT_STATUS")
    if receipt.get("SourceGoldenRunId") != source_run_id:
        raise B19DError("B19C_SOURCE_ID")

    stages: Dict[str, Dict[str, object]] = {}
    for stage, module_id in (("S6", S6_MODULE_ID), ("S7", S7_MODULE_ID)):
        source_artifact, source_manifest = find_artifact(source_root, module_id, stage)
        target_artifact, target_manifest = find_artifact(target_root, module_id, stage)
        source_version = source_manifest.get("ProducerModuleVersion")
        target_version = target_manifest.get("ProducerModuleVersion")
        if source_version != target_version:
            raise B19DError("MODULE_VERSION_DELTA", f"{stage}-{source_version}-{target_version}")
        source_payloads = validate_payload_manifest(
            source_artifact,
            source_manifest,
            "SOURCE_" + stage,
        )
        target_payloads = validate_payload_manifest(
            target_artifact,
            target_manifest,
            "TARGET_" + stage,
        )
        stages[stage] = {
            "ModuleId": module_id,
            "ModuleVersion": source_version,
            "SourceArtifact": source_artifact,
            "TargetArtifact": target_artifact,
            "SourceManifest": source_manifest,
            "TargetManifest": target_manifest,
            "SourcePayloads": source_payloads,
            "TargetPayloads": target_payloads,
        }

    return {
        "B19BSummaryPath": b19b_summary_path,
        "B19CReceiptPath": receipt_path,
        "SourceRunId": source_run_id,
        "SourceRoot": source_root,
        "TargetRunId": B19C_RUN_ID,
        "TargetRoot": target_root,
        "Stages": stages,
    }


def compare_context(context: Mapping[str, object]) -> List[Dict[str, object]]:
    """Compare both downstream Stage boundaries."""

    rows: List[Dict[str, object]] = []
    stages = context["Stages"]
    if not isinstance(stages, dict):
        raise B19DError("STAGE_CONTEXT")
    for stage in ("S6", "S7"):
        item = stages[stage]
        rows.extend(
            compare_payload_sets(
                stage,
                item["SourcePayloads"],
                item["TargetPayloads"],
            )
        )
    return rows


def metadata_int(manifest: Mapping[str, object], key: str) -> int:
    """Read one integer quality field from Artifact Metadata."""

    metadata = manifest.get("Metadata")
    if not isinstance(metadata, dict):
        raise B19DError("METADATA_TYPE", key)
    value = metadata.get(key)
    if isinstance(value, bool) or not isinstance(value, int):
        raise B19DError("METADATA_INTEGER", key)
    return value


def prepare(root: Path) -> str:
    """Archive stale TEMP and show dependency checks with their engineering reasons."""

    moved, locked = archive_previous_temp(root)
    context = resolve_context(root)
    rows = compare_context(context)
    stages = context["Stages"]
    total_payloads = len(rows)
    print("=== U22B19D PREPARE / 检查对象与工程意义 ===")
    print("PASS | B19B来源摘要 | WHY: 固定原Golden Run，防止比较错对象")
    print("PASS | B19C v0.5验收凭据 | WHY: 只比较已经正式通过的目标Run")
    print("PASS | Source/Target均为SUCCEEDED | WHY: 未完成Run不能用于结论继承")
    for stage in ("S6", "S7"):
        item = stages[stage]
        print(
            f"PASS | {stage} {item['ModuleId']}@{item['ModuleVersion']} | "
            "WHY: 同一算法版本才允许比较业务输出"
        )
        print(
            f"PASS | {stage} Manifest与Payload完整 | "
            f"WHY: hash、字节数与行数通过后才信任数据；比较项={len(item['SourcePayloads'])}"
        )
    print(
        f"INFO | 当前预比较 | payloads={total_payloads} "
        f"business_mismatch={sum(row['Status'] not in ('EXACT', 'LINEAGE_NORMALIZED') for row in rows)}"
    )
    print(f"INFO | TEMP整理 | archived={moved} locked={locked}")
    capsule = f"B19DP-S2-P{total_payloads}-A{moved}-L{locked}-X0-C0"
    print(capsule)
    return capsule


def self_test(root: Path) -> str:
    """Exercise the real boundary plus positive and negative comparator behavior."""

    context = resolve_context(root)
    actual_rows = compare_context(context)
    print("=== U22B19D SELF-TEST / 检查内容与原因 ===")
    print("PASS | Run身份与成功状态 | WHY: 防止把失败残留当成正式结果")
    print("PASS | S6/S7 Module身份与版本 | WHY: 算法变化时禁止自动继承人工结论")
    print("PASS | 所有声明Payload的SHA-256、ByteSize、RowCount | WHY: 先证明文件自身未损坏")
    print("PASS | CSV逻辑行比较 | WHY: 候选、证据和Review表必须逐字段相同")
    print("PASS | JSON/JSONL lineage规范化 | WHY: RunId变化不应伪装成业务差异")

    positive = compare_payload_sets(
        "SX",
        {"a.csv": {"Sha256": "A", "SemanticSha256": "S", "RowCount": 1}},
        {"a.csv": {"Sha256": "A", "SemanticSha256": "S", "RowCount": 1}},
    )
    if positive[0]["Status"] != "EXACT":
        raise B19DError("SELFTEST_POSITIVE")
    normalized = compare_payload_sets(
        "SX",
        {"a.json": {"Sha256": "A", "SemanticSha256": "S", "RowCount": None}},
        {"a.json": {"Sha256": "B", "SemanticSha256": "S", "RowCount": None}},
    )
    if normalized[0]["Status"] != "LINEAGE_NORMALIZED":
        raise B19DError("SELFTEST_NORMALIZED")
    negative = compare_payload_sets(
        "SX",
        {"a.csv": {"Sha256": "A", "SemanticSha256": "S1", "RowCount": 1}},
        {"a.csv": {"Sha256": "B", "SemanticSha256": "S2", "RowCount": 1}},
    )
    if negative[0]["Status"] != "BUSINESS_MISMATCH":
        raise B19DError("SELFTEST_NEGATIVE")
    print("PASS | Comparator正例 | WHY: 相同业务内容必须判定可继承")
    print("PASS | Lineage差异例 | WHY: 只含Run/Artifact身份变化时仍可继承")
    print("PASS | Comparator反例 | WHY: 任一业务字段变化必须触发重新人工复核")
    print("PASS | 只读边界 | WHY: 本程序只读取两个Run，报告写入独立B19D目录")
    capsule = f"U22B19D-ST-N10-P{len(actual_rows)}-X0-C0"
    print(capsule)
    return capsule


def write_csv(path: Path, rows: Sequence[Mapping[str, object]]) -> None:
    """Write a deterministic comparison table."""

    fields = [
        "Stage",
        "RelativePath",
        "Status",
        "SourceSha256",
        "TargetSha256",
        "SourceSemanticSha256",
        "TargetSemanticSha256",
        "SourceRowCount",
        "TargetRowCount",
    ]
    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields, lineterminator="\n")
        writer.writeheader()
        writer.writerows(rows)


def publish_output(root: Path, summary: Mapping[str, object], rows: Sequence[Mapping[str, object]], capsule: str) -> str:
    """Create an immutable B19D report or accept a byte-identical replay."""

    output = root / "07_OUTPUT" / OUTPUT_ID
    summary_bytes = canonical_bytes(summary)
    if output.exists():
        if not output.is_dir() or output.is_symlink():
            raise B19DError("OUTPUT_PATH", output.name)
        summary_path = output / "b19d_downstream_equivalence_summary.json"
        capsule_path = output / "B19D_RETURN_CAPSULE.txt"
        if (
            not summary_path.is_file()
            or summary_path.read_bytes() != summary_bytes
            or not capsule_path.is_file()
            or capsule_path.read_text(encoding="utf-8-sig").strip() != capsule
        ):
            raise B19DError("OUTPUT_CONFLICT", output.name)
        return capsule

    staging = output.with_name(output.name + ".writing")
    if staging.exists():
        shutil.rmtree(staging)
    staging.mkdir(parents=True)
    (staging / "b19d_downstream_equivalence_summary.json").write_bytes(summary_bytes)
    write_csv(staging / "b19d_payload_comparison.csv", rows)
    (staging / "B19D_RETURN_CAPSULE.txt").write_text(capsule + "\n", encoding="utf-8")
    os.replace(staging, output)
    return capsule


def execute(root: Path) -> str:
    """Decide whether existing U003 human review can be inherited."""

    context = resolve_context(root)
    rows = compare_context(context)
    mismatches = [row for row in rows if row["Status"] not in ("EXACT", "LINEAGE_NORMALIZED")]
    exact = sum(row["Status"] == "EXACT" for row in rows)
    normalized = sum(row["Status"] == "LINEAGE_NORMALIZED" for row in rows)
    stages = context["Stages"]
    s7_target_manifest = stages["S7"]["TargetManifest"]
    candidates = metadata_int(s7_target_manifest, "CompleteCandidateInventoryCount")
    batches = metadata_int(s7_target_manifest, "ReviewBatchCount")
    equivalent = int(not mismatches)
    human_review_required = int(bool(mismatches))
    summary = {
        "Status": "SUCCEEDED",
        "NodeType": "POST_RUN_VALIDATION",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "SourceRunId": context["SourceRunId"],
        "TargetRunId": context["TargetRunId"],
        "ComparedStages": ["S6", "S7"],
        "PayloadCount": len(rows),
        "ExactPayloadCount": exact,
        "LineageNormalizedPayloadCount": normalized,
        "BusinessMismatchCount": len(mismatches),
        "CandidateCount": candidates,
        "ReviewBatchCount": batches,
        "BusinessPayloadEquivalent": bool(equivalent),
        "ExistingHumanReviewTransferApproved": bool(equivalent),
        "NewHumanReviewRequired": bool(human_review_required),
        "NormalizationPolicy": {
            "CSV": "LOGICAL_ROWS_EXACT",
            "JSON_JSONL": "RECURSIVE_EXACT_AFTER_DOCUMENTED_LINEAGE_KEYS_REMOVED",
            "Other": "SHA256_EXACT",
            "RemovedLineageKeys": sorted(LINEAGE_KEYS),
        },
        "Decision": (
            "INHERIT_EXISTING_16_EVENT_U003_REVIEW"
            if equivalent
            else "REVIEW_CHANGED_TARGET_S7_PACKAGE_IN_U003"
        ),
        "GoldenPromotionApproved": False,
    }
    capsule = (
        f"B19D-S2-P{len(rows)}-E{exact}-N{normalized}-M{len(mismatches)}"
        f"-C{candidates}-B{batches}-D{equivalent}-H{human_review_required}-X0-C0"
    )
    return publish_output(root, summary, rows, capsule)


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Create the fixed three-command interface."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    return parser.parse_args(argv)


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run one B19D responsibility."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        root = require_workspace(args.workspace_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            self_test(root)
        else:
            print(execute(root))
        return 0
    except B19DError as error:
        print(f"B19DE-K{error.reason}-D{error.detail}-X1-C7")
        return 7
    except Exception as error:
        print(f"B19DE-KUNEXPECTED-D{token(error.__class__.__name__)}-X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
