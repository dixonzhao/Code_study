#!/usr/bin/env python3
"""Correct the missing C-0 suffix in three unqualified B2 Module self-tests."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import tempfile
from typing import Dict, Optional, Sequence, Tuple


PROGRAM_ID = "UDP002-U22B2CR1-MODULE-SELFTEST-CAPSULE-CORRECTION"
PROGRAM_VERSION = "0.2"
MODULES = (
    ("S1", "CameraRadar.S1.RadarMeasurementValidity", "RADAR"),
    ("S2", "CameraRadar.S2.ComparisonPopulation", "POPULATION"),
    ("S3", "CameraRadar.S3.SignedDiscrepancy", "OUTCOME"),
)
VERSION = "0.1.0"
OLD_LINE = b'print("B2B-MODULE-ST-" + MODE + "-PASS-12")'
NEW_LINE = b'print("B2B-MODULE-ST-" + MODE + "-PASS-12 C-0")'


class CorrectionError(RuntimeError):
    """Represent one bounded correction failure."""


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def require_workspace(path: Path) -> Path:
    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise CorrectionError("WORKSPACE_ROOT")
    return root


def module_root(workspace: Path, stage: str, module_id: str) -> Path:
    root = workspace / "01_PROGRAMS/modules" / stage / module_id / VERSION
    if not root.is_dir() or root.is_symlink():
        raise CorrectionError("MODULE_ROOT_" + stage)
    expected = {
        "module_manifest.json", "algorithm_card.json", "configuration_schema.json",
        "implementation.py", "selftest.py",
    }
    actual_files = {
        path.name
        for path in root.iterdir()
        if path.is_file() and not path.is_symlink()
    }
    if actual_files != expected:
        raise CorrectionError("MODULE_FILESET_" + stage)
    unexpected_entries = [
        path.name
        for path in root.iterdir()
        if path.name not in expected and path.name != "__pycache__"
    ]
    if unexpected_entries:
        raise CorrectionError("MODULE_EXTRA_ENTRY_" + stage)
    manifest = json.loads((root / "module_manifest.json").read_text(encoding="utf-8-sig"))
    if manifest.get("ModuleId") != module_id or manifest.get("ModuleVersion") != VERSION:
        raise CorrectionError("MODULE_IDENTITY_" + stage)
    return root


def protected_hashes(root: Path) -> Dict[str, str]:
    return {
        name: sha256_file(root / name)
        for name in (
            "module_manifest.json", "algorithm_card.json",
            "configuration_schema.json", "implementation.py",
        )
    }


def qualification_receipt(workspace: Path, module_id: str) -> Path:
    return (
        workspace / "00_CONTROL/registry/qualifications"
        / module_id / VERSION / "qualification_receipt.json"
    )


def correct_one(workspace: Path, stage: str, module_id: str, mode: str) -> bool:
    root = module_root(workspace, stage, module_id)
    selftest = root / "selftest.py"
    before_protected = protected_hashes(root)
    payload = selftest.read_bytes()
    if NEW_LINE in payload and OLD_LINE not in payload:
        changed = False
    elif payload.count(OLD_LINE) == 1 and NEW_LINE not in payload:
        if qualification_receipt(workspace, module_id).exists():
            raise CorrectionError("QUALIFIED_PACKAGE_MUTATION_" + stage)
        corrected = payload.replace(OLD_LINE, NEW_LINE, 1)
        temporary = selftest.with_name("selftest.py.correcting")
        if temporary.exists():
            raise CorrectionError("TEMP_EXISTS_" + stage)
        temporary.write_bytes(corrected)
        os.replace(temporary, selftest)
        changed = True
    else:
        raise CorrectionError("UNEXPECTED_SELFTEST_BYTES_" + stage)
    if protected_hashes(root) != before_protected:
        raise CorrectionError("PROTECTED_FILE_CHANGED_" + stage)
    completed = subprocess.run(
        [
            sys.executable, str(selftest), "--module-root", str(root),
            "--workspace-root", str(workspace),
        ],
        cwd=root,
        text=True,
        capture_output=True,
        timeout=30,
        check=False,
    )
    expected = "B2B-MODULE-ST-" + mode + "-PASS-12 C-0"
    if completed.returncode != 0 or expected not in completed.stdout:
        raise CorrectionError("CORRECTED_SELFTEST_" + stage)
    return changed


def apply_correction(workspace: Path) -> str:
    changed = 0
    for stage, module_id, mode in MODULES:
        changed += int(correct_one(workspace, stage, module_id, mode))
    return f"B2CR1 N-M3-P{changed}-R{3 - changed}-T3-X0 C-0"


def self_test() -> str:
    checks = []
    def ck(value: bool) -> None:
        if not value:
            raise AssertionError("SELFTEST")
        checks.append(True)
    ck(PROGRAM_ID.startswith("UDP002"))
    ck(PROGRAM_VERSION == "0.2")
    ck(len(MODULES) == 3)
    ck(OLD_LINE != NEW_LINE)
    ck(NEW_LINE.endswith(b'C-0")'))
    with tempfile.TemporaryDirectory() as temporary:
        workspace = Path(temporary) / "U-UDP002"
        for stage, module_id, mode in MODULES:
            root = workspace / "01_PROGRAMS/modules" / stage / module_id / VERSION
            root.mkdir(parents=True)
            manifest = {"ModuleId": module_id, "ModuleVersion": VERSION}
            (root / "module_manifest.json").write_text(json.dumps(manifest), encoding="utf-8")
            for name in ("algorithm_card.json", "configuration_schema.json"):
                (root / name).write_text("{}", encoding="utf-8")
            (root / "implementation.py").write_text(
                "def semantic_selftest():\n    return 12\nMODE=" + repr(mode) + "\n",
                encoding="utf-8",
            )
            selftest_source = (
                "import argparse,importlib.util\n"
                "from pathlib import Path\n"
                "MODE=" + repr(mode) + "\n"
                "p=argparse.ArgumentParser();p.add_argument('--module-root');"
                "p.add_argument('--workspace-root');a=p.parse_args()\n"
                "s=importlib.util.spec_from_file_location('m',Path(a.module_root)/'implementation.py');"
                "m=importlib.util.module_from_spec(s);s.loader.exec_module(m)\n"
                "print(\"B2B-MODULE-ST-\" + MODE + \"-PASS-12\")\n"
            )
            (root / "selftest.py").write_text(selftest_source, encoding="utf-8")
            # Qualification may leave this normal Python runtime cache directory.
            (root / "__pycache__").mkdir()
        result = apply_correction(workspace)
        ck(result == "B2CR1 N-M3-P3-R0-T3-X0 C-0")
        ck(apply_correction(workspace) == "B2CR1 N-M3-P0-R3-T3-X0 C-0")
        for stage, module_id, _ in MODULES:
            ck(NEW_LINE in (module_root(workspace, stage, module_id) / "selftest.py").read_bytes())
    if len(checks) != 10:
        raise AssertionError("SELFTEST_COUNT")
    return "B2CR1-ST N-T10-X0 C-0"


def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--apply-correction", action="store_true")
    parser.add_argument("--workspace-root")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            print(self_test())
            return 0
        if not args.workspace_root:
            raise CorrectionError("WORKSPACE_ROOT_REQUIRED")
        print(apply_correction(require_workspace(Path(args.workspace_root))))
        return 0
    except Exception as exc:
        reason = re.sub(r"[^A-Z0-9_]+", "_", str(exc).upper()).strip("_")[:100]
        print("B2CR1E K-" + (reason or exc.__class__.__name__.upper()) + "-X1 C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
