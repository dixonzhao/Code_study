#!/usr/bin/env python3
"""Fail-closed, non-executing module discovery for the U002 U22 runtime."""

from __future__ import annotations

import argparse
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import re
import sys
from typing import Any, Dict, List, Mapping, Sequence, Tuple


PROGRAM_ID = "U002-U22-MODULE-REGISTRY"
VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"
STAGES: Tuple[str, ...] = ("S0", "S1", "S2", "S3", "S4", "S5", "S6", "S7")
VALIDATOR_RELATIVE = Path("01_PROGRAMS/core/u002_u22_contract_validator_v0_1.py")
MODULES_RELATIVE = Path("01_PROGRAMS/modules")
SNAPSHOT_RELATIVE = Path("00_CONTROL/registry/u002_module_registry_snapshot_v0_1.json")
ALLOWED_STATUS = (
    "AVAILABLE", "INCOMPATIBLE", "INVALID_MANIFEST", "DEPENDENCY_MISSING",
    "SELFTEST_REQUIRED", "QUARANTINED",
)


class RegistryError(RuntimeError):
    def __init__(self, reason: str, detail: str = ""):
        self.reason = stable(reason)
        self.detail = stable(detail) if detail else "NA"
        super().__init__(self.reason)


def stable(value: object) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")[:80] or "INTERNAL"


def canonical_bytes(value: object) -> bytes:
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2,
                       allow_nan=False) + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    return path.is_file() and not path.is_symlink()


def within(child: Path, parent: Path) -> bool:
    try:
        child.resolve().relative_to(parent.resolve())
        return True
    except (OSError, ValueError):
        return False


def workspace(path: Path) -> Path:
    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise RegistryError("WORKSPACE_ROOT", root.name)
    return root


def load_validator(root: Path):
    path = root / VALIDATOR_RELATIVE
    if not regular_file(path) or not within(path, root):
        raise RegistryError("VALIDATOR_MISSING", str(path))
    spec = importlib.util.spec_from_file_location("u002_u22_contract_validator", path)
    if spec is None or spec.loader is None:
        raise RegistryError("VALIDATOR_LOAD", str(path))
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    required = ("load_json", "validate_document", "ContractError")
    if any(not hasattr(module, name) for name in required):
        raise RegistryError("VALIDATOR_API", str(path))
    if getattr(module, "CONTRACT_VERSION", None) != CONTRACT_VERSION:
        raise RegistryError("VALIDATOR_VERSION", str(path))
    return module


def relative(path: Path, root: Path) -> str:
    return path.resolve().relative_to(root.resolve()).as_posix()


def manifest_paths(root: Path) -> List[Path]:
    modules = root / MODULES_RELATIVE
    if not modules.is_dir() or modules.is_symlink():
        raise RegistryError("MODULE_ROOT", str(modules))
    found: List[Path] = []
    for stage in STAGES:
        stage_root = modules / stage
        if not stage_root.is_dir() or stage_root.is_symlink():
            raise RegistryError("STAGE_DIRECTORY", stage)
        for path in stage_root.glob("*/*/module_manifest.json"):
            if path.is_symlink() or not within(path, stage_root):
                continue
            found.append(path)
    return sorted(found, key=lambda item: relative(item, root).casefold())


def base_entry(path: Path, root: Path) -> Dict[str, Any]:
    return {
        "ManifestPath": relative(path, root),
        "ModuleId": None,
        "ModuleVersion": None,
        "Stage": path.parents[2].name if len(path.parents) >= 3 else None,
        "AlgorithmStatus": None,
        "RegistryStatus": "INVALID_MANIFEST",
        "Reason": "UNVALIDATED",
        "ManifestSha256": sha256_file(path) if regular_file(path) else None,
        "PackageFiles": {},
    }


def referenced_files(module_dir: Path, manifest: Mapping[str, Any]) -> Dict[str, Path]:
    result = {
        "EntryPoint": module_dir / str(manifest["EntryPoint"]),
        "SelfTestEntryPoint": module_dir / str(manifest["SelfTestEntryPoint"]),
        "AlgorithmCard": module_dir / str(manifest["AlgorithmCard"]),
    }
    configuration = manifest.get("ConfigurationSchema")
    if configuration is not None:
        result["ConfigurationSchema"] = module_dir / str(configuration)
    return result


def safe_member(path: Path, module_dir: Path) -> bool:
    return regular_file(path) and within(path, module_dir)


def inspect_manifest(path: Path, root: Path, validator: Any) -> Dict[str, Any]:
    entry = base_entry(path, root)
    if not regular_file(path):
        entry["Reason"] = "MANIFEST_NOT_REGULAR"
        return entry
    try:
        document = validator.load_json(path, "DOCUMENT_JSON")
    except Exception as exc:
        entry["Reason"] = stable(getattr(exc, "reason", exc.__class__.__name__))
        return entry
    if isinstance(document, dict):
        entry["ModuleId"] = document.get("ModuleId")
        entry["ModuleVersion"] = document.get("ModuleVersion")
        entry["Stage"] = document.get("Stage", entry["Stage"])
        entry["AlgorithmStatus"] = document.get("AlgorithmStatus")
        if document.get("ContractVersion") != CONTRACT_VERSION:
            entry["RegistryStatus"] = "INCOMPATIBLE"
            entry["Reason"] = "CONTRACT_VERSION"
            return entry
    try:
        validator.validate_document(root, "MODULE", document)
    except Exception as exc:
        entry["Reason"] = stable(getattr(exc, "reason", exc.__class__.__name__))
        return entry
    stage_dir = path.parents[2].name
    module_dir_name = path.parents[1].name
    version_dir_name = path.parent.name
    if (document["Stage"] != stage_dir or document["ModuleId"] != module_dir_name
            or document["ModuleVersion"] != version_dir_name):
        entry["Reason"] = "PATH_IDENTITY_MISMATCH"
        return entry
    files = referenced_files(path.parent, document)
    missing: List[str] = []
    for role, member in sorted(files.items()):
        if not safe_member(member, path.parent):
            missing.append(role)
        else:
            entry["PackageFiles"][role] = {
                "RelativePath": relative(member, root),
                "Sha256": sha256_file(member),
            }
    if missing:
        entry["RegistryStatus"] = "DEPENDENCY_MISSING"
        entry["Reason"] = "MISSING_" + "_".join(missing)
        return entry
    try:
        algorithm = validator.load_json(files["AlgorithmCard"], "ALGORITHM_CARD_JSON")
        validator.validate_document(root, "ALGORITHM", algorithm)
    except Exception as exc:
        entry["Reason"] = "ALGORITHM_CARD_" + stable(getattr(exc, "reason", exc.__class__.__name__))
        return entry
    if (algorithm.get("AlgorithmId") != document["ModuleId"]
            or algorithm.get("AlgorithmVersion") != document["ModuleVersion"]
            or algorithm.get("Status") != document["AlgorithmStatus"]):
        entry["Reason"] = "ALGORITHM_IDENTITY_MISMATCH"
        return entry
    entry["RegistryStatus"] = "SELFTEST_REQUIRED"
    entry["Reason"] = "STRUCTURE_VALID_NOT_ACTIVATED"
    return entry


def quarantine_duplicates(entries: List[Dict[str, Any]]) -> None:
    grouped: Dict[Tuple[object, object], List[Dict[str, Any]]] = {}
    for entry in entries:
        identity = (entry.get("ModuleId"), entry.get("ModuleVersion"))
        if None not in identity:
            grouped.setdefault(identity, []).append(entry)
    for group in grouped.values():
        if len(group) > 1:
            for entry in group:
                entry["RegistryStatus"] = "QUARANTINED"
                entry["Reason"] = "DUPLICATE_IDENTITY"


def status_counts(entries: Sequence[Mapping[str, Any]]) -> Dict[str, int]:
    counts = {name: 0 for name in ALLOWED_STATUS}
    for entry in entries:
        status = str(entry["RegistryStatus"])
        if status not in counts:
            raise RegistryError("UNKNOWN_STATUS", status)
        counts[status] += 1
    return counts


def build_snapshot(root: Path) -> Dict[str, Any]:
    validator = load_validator(root)
    entries = [inspect_manifest(path, root, validator) for path in manifest_paths(root)]
    quarantine_duplicates(entries)
    counts = status_counts(entries)
    return {
        "RegistryContract": "U002_U22_MODULE_REGISTRY_SNAPSHOT",
        "RegistryVersion": VERSION,
        "ContractVersion": CONTRACT_VERSION,
        "DiscoveryBehavior": {
            "ImportsModuleImplementation": False,
            "ExecutesModuleSelfTest": False,
            "ActivatesModules": False,
        },
        "StageOrder": list(STAGES),
        "DiscoveredCount": len(entries),
        "StatusCounts": counts,
        "Entries": entries,
    }


def write_snapshot(root: Path, snapshot: Mapping[str, Any]) -> Path:
    path = root / SNAPSHOT_RELATIVE
    path.parent.mkdir(parents=True, exist_ok=True)
    data = canonical_bytes(snapshot)
    temporary = path.with_name(path.name + ".writing")
    if temporary.exists():
        raise RegistryError("SNAPSHOT_TEMP_EXISTS", str(temporary))
    temporary.write_bytes(data)
    os.replace(temporary, path)
    return path


def capsule(snapshot: Mapping[str, Any]) -> str:
    counts = snapshot["StatusCounts"]
    return (
        f"U22A2R N-D{snapshot['DiscoveredCount']}-A{counts['AVAILABLE']}"
        f"-I{counts['INVALID_MANIFEST']}-M{counts['DEPENDENCY_MISSING']}"
        f"-S{counts['SELFTEST_REQUIRED']}-Q{counts['QUARANTINED']}"
        f"-K{counts['INCOMPATIBLE']}-X0 C-0"
    )


def self_test(root: Path) -> str:
    checks: List[bool] = []
    def ck(value: bool) -> None:
        if not value:
            raise RegistryError("SELFTEST")
        checks.append(True)
    ck(PROGRAM_ID.startswith("U002")); ck(VERSION == "0.1"); ck(CONTRACT_VERSION == "U22.0.1")
    ck(len(STAGES) == 8); ck(STAGES[0] == "S0"); ck(STAGES[-1] == "S7")
    ck(len(ALLOWED_STATUS) == 6); ck("AVAILABLE" in ALLOWED_STATUS); ck("QUARANTINED" in ALLOWED_STATUS)
    ck(stable("a b/c") == "A_B_C"); ck(within(root / "01_PROGRAMS", root))
    ck(not within(root.parent, root)); ck(MODULES_RELATIVE.as_posix() == "01_PROGRAMS/modules")
    sample = [{"RegistryStatus": name} for name in ALLOWED_STATUS]
    counts = status_counts(sample)
    ck(all(counts[name] == 1 for name in ALLOWED_STATUS)); ck(sum(counts.values()) == 6)
    empty = {"StatusCounts": {name: 0 for name in ALLOWED_STATUS}, "DiscoveredCount": 0}
    ck(capsule(empty).startswith("U22A2R N-D0-A0")); ck(capsule(empty).endswith("X0 C-0"))
    ck(SNAPSHOT_RELATIVE.suffix == ".json"); ck(VALIDATOR_RELATIVE.suffix == ".py")
    ck(len(canonical_bytes({"b": 1, "a": 2})) > 0); ck(hashlib.sha256(b"x").hexdigest().__len__() == 64)
    ck(regular_file(root / VALIDATOR_RELATIVE)); ck(load_validator(root).CONTRACT_VERSION == CONTRACT_VERSION)
    ck(len(checks) == 23)
    if len(checks) != 24:
        raise RegistryError("SELFTEST_COUNT")
    return "U22A2R-ST N-T24-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--refresh", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        root = workspace(Path(args.workspace_root))
        if args.self_test:
            print(self_test(root))
        else:
            snapshot = build_snapshot(root)
            write_snapshot(root, snapshot)
            print(capsule(snapshot))
        return 0
    except RegistryError as exc:
        print(f"U22A2R N-D0-A0-I0-M0-S0-Q0-K0-X1 C-7 R-{exc.reason}")
        return 7
    except Exception as exc:
        print(f"U22A2R N-D0-A0-I0-M0-S0-Q0-K0-X1 C-7 R-{stable(exc.__class__.__name__)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
