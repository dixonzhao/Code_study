#!/usr/bin/env python3
"""Prepare and install the U003 U3M8 Frame Privacy Mask Module."""

from __future__ import annotations

import argparse
import csv
import hashlib
import shutil
import sys
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PACKAGE_NAME = "UDP003_U3M8_FRAME_MASK_v0_2.zip"
PAYLOADS = {
    "01_PROGRAMS/u003_u3m8_frame_mask_acceptance_gate_v0_2.py",
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.2.0/algorithm_card.json",
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.2.0/implementation.py",
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.2.0/module_manifest.json",
    "03_DOCUMENTS/U3M8_node_report_frame_privacy_mask_v0_2_2026-08-27.txt",
}
REQUIRED_MEMBERS = PAYLOADS | {"PACKAGE_SHA256.csv"}
SUPERSEDED_MODULE_RELATIVE = Path(
    "02_MODULES/EVIDENCE_VIEW/CameraRadar.U003.FrameMask/0.1.0"
)


class InstallError(RuntimeError):
    """Raised when the transfer or installed U3M7 foundation is invalid."""


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def next_archive_root(root: Path, label: str) -> Path:
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    path = root / "A-Archive" / f"{label}_{stamp}"
    suffix = 1
    while path.exists():
        path = root / "A-Archive" / f"{label}_{stamp}_{suffix:02d}"
        suffix += 1
    return path


def move_preserving_root(source: Path, root: Path, archive_root: Path) -> None:
    destination = archive_root / source.relative_to(root)
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(source), str(destination))


def validate_zip(archive: zipfile.ZipFile) -> list[str]:
    members = []
    for name in archive.namelist():
        pure = PurePosixPath(name)
        if pure.is_absolute() or ".." in pure.parts:
            raise InstallError(f"Unsafe ZIP member: {name}")
        if not name.endswith("/"):
            members.append(name)
    if set(members) != REQUIRED_MEMBERS:
        raise InstallError(
            f"Package inventory mismatch. Missing={sorted(REQUIRED_MEMBERS - set(members))}; "
            f"Extra={sorted(set(members) - REQUIRED_MEMBERS)}"
        )
    return sorted(members)


def validate_payloads(package_root: Path) -> list[tuple[str, str, str]]:
    with (package_root / "PACKAGE_SHA256.csv").open(
        "r", encoding="utf-8-sig", newline=""
    ) as stream:
        rows = list(csv.DictReader(stream))
    if len(rows) != len(PAYLOADS):
        raise InstallError("PACKAGE_SHA256 row count is invalid.")
    verified = []
    for row in rows:
        relative = str(row.get("RelativePath", "")).strip()
        expected = str(row.get("Sha256", "")).strip().lower()
        purpose = str(row.get("Purpose", "")).strip()
        path = (package_root / relative).resolve()
        try:
            path.relative_to(package_root.resolve())
        except ValueError as error:
            raise InstallError(f"Manifest path escapes package root: {relative}") from error
        if not path.is_file() or path.is_symlink():
            raise InstallError(f"Payload is missing: {relative}")
        actual = sha256_file(path)
        if actual != expected:
            raise InstallError(f"Payload hash mismatch: {relative}")
        verified.append((relative, actual, purpose))
    if {row[0] for row in verified} != PAYLOADS:
        raise InstallError("Manifest payload inventory is invalid.")
    return verified


def require_u3m7(root: Path) -> None:
    required = (
        root / "01_PROGRAMS" / "u003_evidence_review_workbench_v0_19.py",
        root / "01_PROGRAMS" / "u003_module_runtime_v0_2.py",
        root / "01_PROGRAMS" / "u003_module_action_gui_v0_1.py",
        root / "02_CONTROLS" / "contracts" / "u003_module_contract_v0_2.json",
    )
    missing = [str(path) for path in required if not path.is_file()]
    if missing:
        raise InstallError(f"U3M7 foundation is incomplete: {missing}")


def archive_previous_temp(
    root: Path,
    installer_path: Path,
    package_path: Path,
) -> tuple[Path | None, int]:
    temp_root = root / "04_TEMP"
    temp_root.mkdir(parents=True, exist_ok=True)
    keep = {installer_path.resolve(), package_path.resolve()}
    selected = []
    for path in temp_root.iterdir():
        if path.resolve() in keep or path.name.startswith("U3M8_START_HERE"):
            continue
        if (
            path.name.startswith("U3M8")
            or path.name.startswith("UDP003_U3M8")
            or path.name.startswith("u003_u3m8")
        ):
            selected.append(path)
    if not selected:
        return None, 0
    archive_root = next_archive_root(root, "U3M8_TEMP_CLEANUP")
    for path in selected:
        move_preserving_root(path, root, archive_root)
    return archive_root, len(selected)


def backup_replaced(root: Path) -> tuple[Path | None, int]:
    existing = [root / relative for relative in PAYLOADS if (root / relative).is_file()]
    if not existing:
        return None, 0
    archive_root = next_archive_root(root, "U3M8_REPLACED_PAYLOADS")
    for source in existing:
        destination = archive_root / source.relative_to(root)
        destination.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, destination)
    return archive_root, len(existing)


def archive_superseded_module(root: Path) -> tuple[Path | None, int]:
    source = root / SUPERSEDED_MODULE_RELATIVE
    if not source.is_dir():
        return None, 0
    archive_root = next_archive_root(root, "U3M8_SUPERSEDED_MODULE")
    move_preserving_root(source, root, archive_root)
    return archive_root, 1


def install(package_path: Path, root: Path, installer_path: Path) -> int:
    package_path = package_path.resolve()
    root = root.resolve()
    if root.name.upper() != "U-UDP003":
        raise InstallError(f"Expected U-UDP003 root, got: {root}")
    if not package_path.is_file():
        raise InstallError(f"Transfer ZIP is missing: {package_path}")
    require_u3m7(root)
    temp_archive, temp_count = archive_previous_temp(root, installer_path, package_path)
    prepared = root / "04_TEMP" / "U3M8_PREPARED"
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package_path, "r") as archive:
        members = validate_zip(archive)
        archive.extractall(prepared)
    verified = validate_payloads(prepared)
    backup_root, backup_count = backup_replaced(root)
    superseded_root, superseded_count = archive_superseded_module(root)
    for relative, _digest, _purpose in verified:
        source = prepared / relative
        destination = root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        writing = destination.with_name(destination.name + ".writing")
        shutil.copy2(source, writing)
        writing.replace(destination)

    print("U3M8 prepare and installation checks:")
    print(f"  [OK] U003 root: {root} - company topology is explicit")
    print("  [OK] U3M7 Action interface - interactive Module dependency is present")
    print(f"  [OK] Exact ZIP inventory: {len(members)} files - no extra payload accepted")
    for relative, digest, purpose in verified:
        print(f"  [OK] {relative} | {digest[:12]} - {purpose}")
    print(f"  [OK] Previous U3M8 TEMP items archived: {temp_count} | {temp_archive or 'none'}")
    print(f"  [OK] Replaced payloads backed up: {backup_count} | {backup_root or 'none'}")
    print(f"  [OK] Superseded Module versions archived: {superseded_count} | {superseded_root or 'none'}")
    print("  [OK] Review data, videos, outputs, Profiles, and unknown files were not modified")
    print(f"U3M8P N-F{len(PAYLOADS)}-M1-A{temp_count}-B{backup_count}-O{superseded_count}-X0 C-0")
    return 0


def main(argv: list[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--prepare", action="store_true")
    parser.add_argument("--root", type=Path)
    parser.add_argument("--package", type=Path)
    arguments = parser.parse_args(argv)
    try:
        if not arguments.prepare:
            raise InstallError("Use --prepare.")
        script_path = Path(__file__).resolve()
        root = arguments.root if arguments.root is not None else script_path.parent.parent
        package = arguments.package if arguments.package is not None else script_path.parent / PACKAGE_NAME
        return install(package, root, script_path)
    except Exception as error:
        print(f"U3M8PE K-{type(error).__name__.upper()} D-{error} X1 C-7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
