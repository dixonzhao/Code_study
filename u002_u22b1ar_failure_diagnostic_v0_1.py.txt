#!/usr/bin/env python3
"""Extract the actual Module rejection from the latest failed B1AR Run."""

from __future__ import annotations

import argparse
import json
from pathlib import Path
import sys
from typing import List, Sequence


class DiagnosticError(RuntimeError):
    """Represent one deterministic diagnostic failure."""


def regular_file(path: Path) -> bool:
    """Accept ordinary files only."""

    return path.is_file() and not path.is_symlink()


def compact(value: object, limit: int = 300) -> str:
    """Keep one diagnostic line safe for manual return."""

    text = " ".join(str(value).replace("\r", "\n").split())
    return text[:limit] if text else "EMPTY"


def latest_run(root: Path) -> Path:
    """Select the most recently modified B1AR Run directory."""

    output = root / "07_OUTPUT"
    runs = [
        path
        for path in output.glob("U22B1AR_*")
        if path.is_dir() and not path.is_symlink()
    ]
    if not runs:
        raise DiagnosticError("NO_B1AR_RUN")
    return max(runs, key=lambda path: path.stat().st_mtime_ns)


def first_file(root: Path, name: str) -> Path | None:
    """Return the only matching diagnostic file when present."""

    matches = [path for path in root.rglob(name) if regular_file(path)]
    if not matches:
        return None
    return sorted(matches, key=lambda path: str(path).lower())[0]


def read_text(path: Path | None) -> str:
    """Read one optional diagnostic file safely."""

    if path is None:
        return ""
    return path.read_text(encoding="utf-8-sig", errors="replace").strip()


def diagnostic(root: Path) -> List[str]:
    """Collect the child Module reason without reading business signal values."""

    run = latest_run(root)
    module_root = run / "02_DIAGNOSTICS/modules"
    stdout_path = first_file(module_root, "stdout.txt")
    stderr_path = first_file(module_root, "stderr.txt")
    result_path = first_file(module_root, "module_result.json")
    log_path = first_file(run / "01_LOGS", "*.json")
    stdout = read_text(stdout_path)
    stderr = read_text(stderr_path)
    result = read_text(result_path)
    log = read_text(log_path)
    reason = stdout or stderr
    if not reason and result:
        try:
            payload = json.loads(result)
            reason = payload.get("Capsule") or payload.get("Status") or result
        except Exception:
            reason = result
    if not reason and log:
        try:
            payload = json.loads(log)
            reason = (
                payload.get("FailureReason")
                or payload.get("Capsule")
                or payload.get("Status")
                or log
            )
        except Exception:
            reason = log
    if not reason:
        raise DiagnosticError("NO_CHILD_REASON")
    flags = (
        int(bool(stdout)),
        int(bool(stderr)),
        int(bool(result)),
        int(bool(log)),
    )
    return [
        (
            "U22B1ARD1 N-R1-S"
            + str(flags[0])
            + "-E"
            + str(flags[1])
            + "-J"
            + str(flags[2])
            + "-L"
            + str(flags[3])
            + "-X0 C-0"
        ),
        "M-" + compact(reason),
    ]


def self_test() -> int:
    """Exercise latest-run selection and stdout precedence."""

    import tempfile

    with tempfile.TemporaryDirectory(prefix="u22b1ard1_") as temp:
        root = Path(temp)
        run = root / "07_OUTPUT/U22B1AR_TEST"
        module = run / "02_DIAGNOSTICS/modules/0001_Module"
        module.mkdir(parents=True)
        (run / "01_LOGS").mkdir(parents=True)
        (module / "stdout.txt").write_text(
            "U22B1AM N-F0-R0-K0-Q0-X1 C-7 R-D2_TEST\n",
            encoding="utf-8",
        )
        (module / "stderr.txt").write_text("", encoding="utf-8")
        (module / "module_result.json").write_text("{}\n", encoding="utf-8")
        (run / "01_LOGS/0001.json").write_text("{}\n", encoding="utf-8")
        lines = diagnostic(root)
        if len(lines) != 2 or "R-D2_TEST" not in lines[1]:
            raise DiagnosticError("SELFTEST")
    print("U22B1ARD1-ST N-T2-X0 C-0")
    return 0


def main(argv: Sequence[str] = None) -> int:
    """Run self-test or inspect the latest real B1AR failure."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--workspace-root")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            return self_test()
        if not args.workspace_root:
            raise DiagnosticError("WORKSPACE_ROOT_REQUIRED")
        for line in diagnostic(Path(args.workspace_root).resolve()):
            print(line)
        return 0
    except Exception as exc:
        print("U22B1ARD1 N-R0-S0-E0-J0-L0-X1 C-7 R-" + compact(exc, 80))
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
