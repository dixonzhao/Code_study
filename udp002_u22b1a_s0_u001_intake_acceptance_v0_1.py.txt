class AcceptanceError(RuntimeError):
    """Represent one deterministic independent acceptance failure."""


def regular_file(path: Path) -> bool:
    """Accept ordinary files only during acceptance."""

    return path.is_file() and not path.is_symlink()


def canonical_text(value: object) -> str:
    """Serialize one control document deterministically."""

    return json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    ) + "\n"


def run(command: List[str], timeout: int = 180) -> subprocess.CompletedProcess:
    """Run one bounded dependency process and preserve evidence streams."""

    return subprocess.run(
        command,
        text=True,
        capture_output=True,
        timeout=timeout,
        check=False,
    )


def check(value: bool, reason: str, checks: List[bool]) -> None:
    """Append one visible check or fail immediately."""

    if not value:
        raise AcceptanceError(reason)
    checks.append(True)


def sha256_tree(root: Path) -> str:
    """Hash all regular files in one synthetic U001 package."""

    digest = hashlib.sha256()
    for path in sorted(root.rglob("*"), key=lambda item: item.as_posix()):
        if not regular_file(path):
            continue
        relative = path.relative_to(root).as_posix().encode("utf-8")
        digest.update(relative)
        with path.open("rb") as stream:
            for block in iter(lambda: stream.read(1024 * 1024), b""):
                digest.update(block)
    return digest.hexdigest()


def write_csv(path: Path, header: Sequence[str], rows: Sequence[Sequence[object]]) -> None:
    """Write one independent fixture CSV."""

    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.writer(stream)
        writer.writerow(header)
        writer.writerows(rows)

def copy_environment(source: Path, target: Path) -> None:
    """Copy only frozen Foundation files into one isolated test workspace."""

    target.mkdir(parents=True)
    for relative in (
        "00_CONTROL/contracts",
        "00_CONTROL/foundation",
        "00_CONTROL/registry",
        "01_PROGRAMS/core",
        "02_CONFIG/run_configs",
        "03_CASES",
        "04_TEMP",
        "05_TESTS/integration",
        "07_OUTPUT",
    ):
        (target / relative).mkdir(parents=True, exist_ok=True)
    for stage in ("S0", "S1", "S2", "S3", "S4", "S5", "S6", "S7"):
        (target / "01_PROGRAMS/modules" / stage).mkdir(parents=True)
    for name in CONTRACT_NAMES:
        source_file = source / "00_CONTROL/contracts" / name
        if not regular_file(source_file):
            raise AcceptanceError("CONTRACT_MISSING_" + name)
        shutil.copy2(source_file, target / "00_CONTROL/contracts" / name)
    for name in CORE_NAMES:
        source_file = source / "01_PROGRAMS/core" / name
        if not regular_file(source_file):
            raise AcceptanceError("CORE_MISSING_" + name)
        shutil.copy2(source_file, target / "01_PROGRAMS/core" / name)
    receipt = source / "00_CONTROL/foundation/u002_u22_foundation_freeze_receipt_v0_1.json"
    if not regular_file(receipt):
        raise AcceptanceError("FREEZE_RECEIPT_MISSING")
    shutil.copy2(receipt, target / "00_CONTROL/foundation" / receipt.name)
    auditor = source / "05_TESTS/integration" / FREEZE_AUDITOR
    if not regular_file(auditor):
        raise AcceptanceError("FREEZE_AUDITOR_MISSING")
    shutil.copy2(auditor, target / "05_TESTS/integration" / auditor.name)

def inspect_summary(root: Path, run_id: str) -> Dict[str, object]:
    """Load the primary S0 summary through the governed Artifact index."""

    run_root = root / "07_OUTPUT" / run_id
    index = json.loads(
        (run_root / "run_artifact_index_v0_1.json").read_text(encoding="utf-8")
    )
    if len(index["Artifacts"]) != 1:
        raise AcceptanceError("ARTIFACT_COUNT")
    entry = index["Artifacts"][0]
    manifest_path = run_root / entry["ManifestRelativePath"]
    summary_path = manifest_path.parent / "u001_intake_summary.json"
    if not regular_file(summary_path):
        raise AcceptanceError("SUMMARY_MISSING")
    return json.loads(summary_path.read_text(encoding="utf-8"))


def archive_previous(root: Path) -> int:
    """Move completed A7 transfer programs out of active TEMP."""

    names = (
        "u002_u22_synthetic_vertical_acceptance_v0_1.py.txt",
        "udp002_u22a7_foundation_freeze_acceptance_v0_1.py.txt",
    )
    archive = root / "A-Archive/U22_COMPLETED/U22A7/04_TEMP"
    moved = 0
    for name in names:
        source = root / "04_TEMP" / name
        if not regular_file(source):
            continue
        archive.mkdir(parents=True, exist_ok=True)
        destination = archive / name
        if destination.exists():
            if destination.read_bytes() != source.read_bytes():
                raise AcceptanceError("ARCHIVE_CONFLICT")
            source.unlink()
        else:
            os.replace(source, destination)
        moved += 1
    return moved


def main(argv: Sequence[str] = None) -> int:
    """Test, install, qualify, and independently audit the first real Module."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    checks: List[bool] = []
    try:
        root = Path(args.workspace_root).resolve()
        check(root.is_dir() and root.name == "U-UDP002", "WORKSPACE_ROOT", checks)
        candidate = root / "04_TEMP" / CANDIDATE
        check(regular_file(candidate), "CANDIDATE_MISSING", checks)
        freeze_audit = run(
            [
                sys.executable,
                str(root / "05_TESTS/integration" / FREEZE_AUDITOR),
                "--audit-freeze",
                "--workspace-root",
                str(root),
            ],
            timeout=60,
        )
        check(
            freeze_audit.returncode == 0
            and "N-F11-N8-V1-X0 C-0" in freeze_audit.stdout,
            "FOUNDATION_FREEZE",
            checks,
        )
        installer_selftest = run(
            [sys.executable, str(candidate), "--self-test"],
            timeout=60,
        )
        check(
            installer_selftest.returncode == 0
            and "N-T8-F4-X0 C-0" in installer_selftest.stdout,
            "INSTALLER_SELFTEST_" + installer_selftest.stdout.strip(),
            checks,
        )
        with tempfile.TemporaryDirectory(prefix="u22b1a_acceptance_") as temporary:
            test_root = Path(temporary) / "U-UDP002"
            copy_environment(root, test_root)
            test_candidate = test_root / "04_TEMP" / CANDIDATE
            shutil.copy2(candidate, test_candidate)
            installed = install(test_candidate, test_root)
            check(installed.returncode == 0, "TEST_INSTALL", checks)
            qualification = qualify(test_root)
            check(qualification.returncode == 0, "TEST_QUALIFICATION", checks)
            good_package = test_root / "03_CASES/GOOD_U001_PACKAGE"
            create_u001_package(good_package, 0)
            input_hash = sha256_tree(good_package)
            good_config = run_configuration("U22B1A_GOOD", good_package)
            good_prepare = prepare(
                test_root,
                "u22b1a_good.json",
                good_config,
            )
            check(good_prepare.returncode == 0, "GOOD_PREFLIGHT", checks)
            good_execute = execute(test_root, "U22B1A_GOOD")
            check(
                good_execute.returncode == 0
                and "N-M1-A1-O0-R0-S1-X0 C-0" in good_execute.stdout,
                "GOOD_EXECUTION_" + good_execute.stdout.strip(),
                checks,
            )
            check(sha256_tree(good_package) == input_hash, "INPUT_MUTATED", checks)
            good_status = json.loads(
                (
                    test_root / "07_OUTPUT/U22B1A_GOOD/run_status.json"
                ).read_text(encoding="utf-8")
            )
            check(good_status["Status"] == "SUCCEEDED", "GOOD_STATUS", checks)
            summary = inspect_summary(test_root, "U22B1A_GOOD")
            check(summary["Status"] == "INTAKE_ACCEPTED", "SUMMARY_STATUS", checks)
            check(summary["FrameDomain"]["FrameRows"] == 3, "SUMMARY_ROWS", checks)
            check(summary["SignalCount"] == 2, "SUMMARY_SIGNALS", checks)
            check(len(summary["Files"]) == 6, "SUMMARY_FILES", checks)
            check(len(summary["PackageFingerprintSha256"]) == 64, "SUMMARY_HASH", checks)
            good_audit = audit(test_root, "U22B1A_GOOD")
            check(good_audit.returncode == 0, "GOOD_AUDIT", checks)
            replay = execute(test_root, "U22B1A_GOOD")
            check(
                replay.returncode == 0
                and "N-M1-A1-O0-R1-S1-X0 C-0" in replay.stdout,
                "GOOD_REPLAY",
                checks,
            )
            bad_package = test_root / "03_CASES/BAD_QUALITY_U001_PACKAGE"
            create_u001_package(bad_package, 1)
            bad_config = run_configuration("U22B1A_BAD_QUALITY", bad_package)
            bad_prepare = prepare(
                test_root,
                "u22b1a_bad_quality.json",
                bad_config,
            )
            check(bad_prepare.returncode == 0, "BAD_PREFLIGHT", checks)
            bad_execute = execute(test_root, "U22B1A_BAD_QUALITY")
            check(
                bad_execute.returncode == 7
                and "MODULE_EXIT" in bad_execute.stdout,
                "BAD_QUALITY_NOT_REJECTED",
                checks,
            )
            bad_status = json.loads(
                (
                    test_root / "07_OUTPUT/U22B1A_BAD_QUALITY/run_status.json"
                ).read_text(encoding="utf-8")
            )
            check(bad_status["Status"] == "FAILED", "BAD_STATUS", checks)
            check(
                bad_status["FailedModule"] == MODULE_ID,
                "BAD_FAILED_MODULE",
                checks,
            )
            test_freeze = run(
                [
                    sys.executable,
                    str(test_root / "05_TESTS/integration" / FREEZE_AUDITOR),
                    "--audit-freeze",
                    "--workspace-root",
                    str(test_root),
                ],
                timeout=60,
            )
            check(test_freeze.returncode == 0, "TEST_FREEZE_AFTER_MODULE", checks)
        actual_install = install(candidate, root)
        check(actual_install.returncode == 0, "ACTUAL_INSTALL", checks)
        actual_qualification = qualify(root)
        check(actual_qualification.returncode == 0, "ACTUAL_QUALIFICATION", checks)
        availability_path = root / "00_CONTROL/registry/u002_module_availability_snapshot_v0_1.json"
        check(regular_file(availability_path), "AVAILABILITY_MISSING", checks)
        availability = json.loads(availability_path.read_text(encoding="utf-8"))
        matches = [
            item
            for item in availability["Entries"]
            if item["ModuleId"] == MODULE_ID
            and item["ModuleVersion"] == MODULE_VERSION
        ]
        check(len(matches) == 1, "AVAILABILITY_IDENTITY", checks)
        check(
            matches[0]["RegistryStatus"] == "AVAILABLE",
            "AVAILABILITY_STATUS",
            checks,
        )
        final_freeze = run(
            [
                sys.executable,
                str(root / "05_TESTS/integration" / FREEZE_AUDITOR),
                "--audit-freeze",
                "--workspace-root",
                str(root),
            ],
            timeout=60,
        )
        check(final_freeze.returncode == 0, "FINAL_FOUNDATION_FREEZE", checks)
        archived = archive_previous(root)
        check(archived in (0, 1, 2), "ARCHIVE_COUNT", checks)
        print(
            f"U22B1A N-S1-P1-N1-Q1-V1-F1-H1-T{len(checks)}"
            f"-I1-A{archived}-X0 C-0"
        )
        return 0
    except AcceptanceError as exc:
        print("U22B1A N-S0-P0-N0-Q0-V0-F0-H0-T0-I0-A0-X1 C-7 R-" + str(exc))
        return 7
    except subprocess.TimeoutExpired:
        print("U22B1A N-S0-P0-N0-Q0-V0-F0-H0-T0-I0-A0-X1 C-7 R-TIMEOUT")
        return 7
    except Exception as exc:
        detail = str(exc).upper().replace(" ", "_")[:80]
        print(
            "U22B1A N-S0-P0-N0-Q0-V0-F0-H0-T0-I0-A0-X1 C-7 R-"
            + exc.__class__.__name__.upper()
            + "_"
            + detail
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
