"""Assess Radar loss against the selected Camera target angular footprint.

B17 is a bounded post-Golden-Run experiment.  It does not modify U001, the
accepted S1 Camera arbitration Module, or the accepted Candidate identities.
It reconnects the two governed Camera Width signals from the U001 Frame View
to the already selected Camera role and compares target-center bearing with
nearest-body-edge bearing.
"""

class B17Error(RuntimeError):
    """One deterministic fail-closed B17 error."""

    def __init__(self, reason: object, detail: object = "NA") -> None:
        self.reason = machine_token(reason, 80)
        self.detail = machine_token(detail, 180)
        super().__init__(self.reason)


def machine_token(value: object, limit: int = 100) -> str:
    """Return a bounded token suitable for low-bandwidth manual transfer."""

    text = "".join(
        character if character.isalnum() or character in ".-" else "_"
        for character in str(value).upper()
    ).strip("_")
    return (text or "UNKNOWN")[:limit]


def sha256_file(path: Path) -> str:
    """Hash one regular file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_json(path: Path, label: str) -> dict[str, object]:
    """Read one JSON object and reject missing, linked, or non-object files."""

    if not path.is_file() or path.is_symlink():
        raise B17Error(label + "_MISSING", path)
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as error:
        raise B17Error(label + "_JSON", path.name) from error
    if not isinstance(value, dict):
        raise B17Error(label + "_TYPE", type(value).__name__)
    return value


def read_csv(path: Path) -> tuple[list[dict[str, str]], list[str]]:
    """Read one governed CSV while preserving its text representation."""

    if not path.is_file() or path.is_symlink():
        raise B17Error("CSV_MISSING", path)
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        columns = list(reader.fieldnames or [])
        rows = [dict(row) for row in reader]
    if not columns or len(columns) != len(set(columns)):
        raise B17Error("CSV_HEADER", path.name)
    return rows, columns


def write_csv(
    path: Path,
    columns: Sequence[str],
    rows: Iterable[Mapping[str, object]],
) -> None:
    """Write deterministic UTF-8 CSV output."""

    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8-sig", newline="") as stream:
        writer = csv.DictWriter(
            stream,
            fieldnames=list(columns),
            extrasaction="ignore",
            lineterminator="\n",
        )
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in columns})


def write_json(path: Path, value: Mapping[str, object]) -> None:
    """Write deterministic JSON output."""

    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
    )


def parse_int(value: object, label: str) -> int:
    """Parse one integer without accepting fractional values."""

    if isinstance(value, bool):
        raise B17Error(label, value)
    try:
        number = float(str(value).strip())
    except Exception as error:
        raise B17Error(label, value) from error
    if not math.isfinite(number) or not number.is_integer():
        raise B17Error(label, value)
    return int(number)


def parse_float(value: object) -> float | None:
    """Return a finite float, otherwise None."""

    try:
        number = float(str(value).strip())
    except Exception:
        return None
    return number if math.isfinite(number) else None


def parse_bool(value: object, label: str) -> bool:
    """Parse the controlled zero/one or true/false representation."""

    text = str(value).strip().lower()
    if text in {"1", "1.0", "true", "yes"}:
        return True
    if text in {"0", "0.0", "false", "no"}:
        return False
    raise B17Error(label, value)


def require_columns(
    columns: Sequence[str],
    required: Sequence[str],
    label: str,
) -> None:
    """Require all named columns while permitting documented extensions."""

    missing = [column for column in required if column not in columns]
    if missing:
        raise B17Error(label + "_COLUMN", missing[0])


def normalize_role(value: object) -> str:
    """Normalize only controlled role spelling, never signal identity."""

    return "".join(character for character in str(value).upper() if character.isalnum())


def normalize_can_id(value: object) -> int:
    """Parse one hexadecimal or decimal CAN identifier for identity matching."""

    text = str(value).strip().lower()
    try:
        return int(text, 16) if text.startswith("0x") else int(text)
    except ValueError as error:
        raise B17Error("CAN_ID", value) from error


def normalize_channel(value: object) -> int:
    """Parse the governed numeric channel identity."""

    return parse_int(value, "CHANNEL")


def one_binding(plan: Mapping[str, object], role: str) -> Mapping[str, object]:
    """Load exactly one role from the source run's fingerprint-bound profile."""

    module_plan = plan.get("ModulePlan")
    if not isinstance(module_plan, list):
        raise B17Error("SOURCE_MODULE_PLAN")
    camera_modules = [
        item
        for item in module_plan
        if isinstance(item, dict) and item.get("ModuleId") == CAMERA_MODULE_ID
    ]
    if len(camera_modules) != 1:
        raise B17Error("CAMERA_MODULE_COUNT", len(camera_modules))
    configuration = camera_modules[0].get("Configuration")
    if not isinstance(configuration, dict):
        raise B17Error("CAMERA_CONFIGURATION")
    relative = Path(str(configuration.get("BindingProfilePath", "")))
    workspace_root = Path(str(plan["_WorkspaceRoot"]))
    profile_path = (workspace_root / relative).resolve()
    try:
        profile_path.relative_to(workspace_root.resolve())
    except ValueError as error:
        raise B17Error("BINDING_PROFILE_ESCAPE") from error
    if sha256_file(profile_path) != configuration.get("BindingProfileSha256"):
        raise B17Error("BINDING_PROFILE_HASH")
    profile = load_json(profile_path, "BINDING_PROFILE")
    bindings = profile.get("Bindings")
    selected = bindings.get(role) if isinstance(bindings, dict) else None
    if not isinstance(selected, list) or len(selected) != 1 or not isinstance(selected[0], dict):
        raise B17Error("BINDING_COUNT", role)
    return selected[0]


def discover_width_groups(
    workspace_root: Path,
    source_run_root: Path,
) -> tuple[Path, dict[str, str], dict[str, object]]:
    """Resolve paired Width identities without hard-coding company CAN IDs."""

    plan = load_json(source_run_root / "00_CONFIG/resolved_run_plan.json", "SOURCE_PLAN")
    plan["_WorkspaceRoot"] = str(workspace_root.resolve())
    package_root = Path(str(plan.get("InputPackageResolved", ""))).resolve()
    if not package_root.is_dir() or package_root.is_symlink():
        raise B17Error("U001_PACKAGE", package_root)
    frame_path = package_root / "u001_frame_values.csv"
    catalog_path = package_root / "u001_signal_catalog.csv"
    if not frame_path.is_file() or not catalog_path.is_file():
        raise B17Error("U001_PUBLIC_FILES")

    catalog_rows, catalog_columns = read_csv(catalog_path)
    require_columns(
        catalog_columns,
        (
            "CanonicalName",
            "CAN_ID",
            "Channel",
            "SignalName",
            "IncludeInFrameView",
            "HoldPolicy",
            "Unit",
        ),
        "CATALOG",
    )

    groups: dict[str, str] = {}
    metadata: dict[str, object] = {}
    for role in ("CAMERA_1_DISTANCE", "CAMERA_3_DISTANCE"):
        binding = one_binding(plan, role)
        matches = [
            row
            for row in catalog_rows
            if normalize_channel(row["Channel"]) == normalize_channel(binding.get("Channel"))
            and normalize_can_id(row["CAN_ID"]) == normalize_can_id(binding.get("CAN_ID"))
            and str(row["SignalName"]).strip().casefold() == "width"
        ]
        if len(matches) != 1:
            raise B17Error("WIDTH_IDENTITY_COUNT", f"{role}-{len(matches)}")
        row = matches[0]
        if not parse_bool(row["IncludeInFrameView"], role + "_WIDTH_FRAME"):
            raise B17Error("WIDTH_NOT_IN_FRAME_VIEW", role)
        if str(row["HoldPolicy"]).strip().upper() != "NONE":
            raise B17Error("WIDTH_HOLD_POLICY", row["HoldPolicy"])
        if str(row["Unit"]).strip().lower() not in {"m", "meter", "metre"}:
            raise B17Error("WIDTH_UNIT", row["Unit"])
        groups[role] = str(row["CanonicalName"])
        metadata[role] = {
            "DistanceCanonicalName": binding.get("CanonicalName"),
            "WidthCanonicalName": row["CanonicalName"],
            "CAN_ID": row["CAN_ID"],
            "Channel": row["Channel"],
            "SignalName": row["SignalName"],
            "Unit": row["Unit"],
            "HoldPolicy": row["HoldPolicy"],
        }
    return frame_path, groups, metadata


def load_selected_widths(
    frame_path: Path,
    groups: Mapping[str, str],
    wanted_frames: set[int],
) -> dict[int, dict[str, object]]:
    """Stream the large U001 Frame View and retain only B14 frame identities."""

    suffixes = ("__value", "__updated", "__age", "__valid", "__missing_reason")
    required = ["FrameCount"]
    for canonical in groups.values():
        required.extend(canonical + suffix for suffix in suffixes)

    output: dict[int, dict[str, object]] = {}
    with frame_path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        columns = list(reader.fieldnames or [])
        require_columns(columns, required, "FRAME_VIEW")
        for row in reader:
            frame = parse_int(row.get("FrameCount"), "FRAMECOUNT")
            if frame not in wanted_frames:
                continue
            if frame in output:
                raise B17Error("FRAME_VIEW_DUPLICATE", frame)
            record: dict[str, object] = {}
            for role, canonical in groups.items():
                record[role] = {
                    "value": row[canonical + "__value"],
                    "updated": row[canonical + "__updated"],
                    "age": row[canonical + "__age"],
                    "valid": row[canonical + "__valid"],
                    "missing_reason": row[canonical + "__missing_reason"],
                }
            output[frame] = record
    if set(output) != wanted_frames:
        missing = sorted(wanted_frames - set(output))
        raise B17Error("FRAME_DOMAIN_MISSING", missing[0] if missing else "NA")
    return output


def selected_width_evidence(
    role: object,
    width_record: Mapping[str, object],
) -> tuple[float | None, bool, str]:
    """Select Width with the same governed Camera role selected by S1."""

    role_map = {
        "CAMERA1DISTANCE": "CAMERA_1_DISTANCE",
        "CAMERA1": "CAMERA_1_DISTANCE",
        "CAM1": "CAMERA_1_DISTANCE",
        "CAMERA3DISTANCE": "CAMERA_3_DISTANCE",
        "CAMERA3": "CAMERA_3_DISTANCE",
        "CAM3": "CAMERA_3_DISTANCE",
    }
    selected = role_map.get(normalize_role(role))
    if selected is None:
        return None, False, "SELECTED_ROLE_UNSUPPORTED"
    evidence = width_record.get(selected)
    if not isinstance(evidence, dict):
        return None, False, "WIDTH_ROLE_MISSING"
    value = parse_float(evidence.get("value"))
    try:
        valid = parse_bool(evidence.get("valid"), "WIDTH_VALID")
        updated = parse_bool(evidence.get("updated"), "WIDTH_UPDATED")
    except B17Error:
        return value, False, "WIDTH_BOOLEAN_INVALID"
    age_text = str(evidence.get("age", "")).strip()
    age = parse_int(age_text, "WIDTH_AGE") if age_text else None
    if not valid:
        return value, False, "WIDTH_U001_INVALID"
    if not updated:
        return value, False, "WIDTH_NOT_UPDATED_THIS_FRAME"
    if age != 0:
        return value, False, "WIDTH_AGE_NOT_ZERO"
    if value is None:
        return None, False, "WIDTH_NONNUMERIC"
    if value <= 0.0:
        return value, False, "WIDTH_NONPOSITIVE"
    return value, True, "VALID"


def derive_footprint_rows(
    b14_rows: Sequence[Mapping[str, str]],
    b14_columns: Sequence[str],
    widths: Mapping[int, Mapping[str, object]],
) -> list[dict[str, object]]:
    """Calculate center and nearest-body-edge geometry for every B14 frame."""

    require_columns(
        b14_columns,
        (
            "FrameCount",
            "CanonicalCameraDistance",
            "SelectedCameraLateralPosition",
            "TargetBearingProxyValid",
            "AbsTargetBearingProxyDeg",
            "SelectedCameraRole",
            "RadarMeasurementValid",
            "RadarValidityReason",
        ),
        "B14_FRAME",
    )
    output: list[dict[str, object]] = []
    seen: set[int] = set()
    for source in b14_rows:
        frame = parse_int(source["FrameCount"], "B14_FRAMECOUNT")
        if frame in seen:
            raise B17Error("B14_DUPLICATE_FRAME", frame)
        seen.add(frame)
        distance = parse_float(source["CanonicalCameraDistance"])
        lateral = parse_float(source["SelectedCameraLateralPosition"])
        center_valid = parse_bool(source["TargetBearingProxyValid"], "CENTER_VALID")
        radar_valid = parse_bool(source["RadarMeasurementValid"], "RADAR_VALID")
        width, width_valid, width_reason = selected_width_evidence(
            source["SelectedCameraRole"],
            widths[frame],
        )

        footprint_reason = "VALID"
        if not center_valid:
            footprint_reason = "CENTER_GEOMETRY_INVALID"
        elif distance is None or distance <= 0.0:
            footprint_reason = "DISTANCE_INVALID"
        elif lateral is None:
            footprint_reason = "LATERAL_INVALID"
        elif not width_valid or width is None:
            footprint_reason = width_reason

        footprint_valid = footprint_reason == "VALID"
        abs_lateral = abs(lateral) if lateral is not None else None
        half_width = width / 2.0 if width is not None else None
        near_offset = (
            max(abs_lateral - half_width, 0.0)
            if footprint_valid and abs_lateral is not None and half_width is not None
            else None
        )
        far_offset = (
            abs_lateral + half_width
            if footprint_valid and abs_lateral is not None and half_width is not None
            else None
        )
        near_angle = (
            math.degrees(math.atan2(near_offset, distance))
            if footprint_valid and near_offset is not None and distance is not None
            else None
        )
        far_angle = (
            math.degrees(math.atan2(far_offset, distance))
            if footprint_valid and far_offset is not None and distance is not None
            else None
        )
        output.append(
            {
                "FrameCount": frame,
                "CanonicalCameraDistance": source["CanonicalCameraDistance"],
                "SelectedCameraLateralPosition": source["SelectedCameraLateralPosition"],
                "SelectedCameraRole": source["SelectedCameraRole"],
                "SelectedCameraWidth": width if width is not None else "",
                "SelectedCameraWidthValid": int(width_valid),
                "SelectedCameraWidthReason": width_reason,
                "TargetCenterBearingValid": int(center_valid),
                "AbsTargetCenterBearingDeg": source["AbsTargetBearingProxyDeg"],
                "TargetFootprintValid": int(footprint_valid),
                "TargetFootprintInvalidReason": footprint_reason,
                "HalfTargetWidthM": half_width if half_width is not None else "",
                "NearestBodyEdgeLateralOffsetM": near_offset if near_offset is not None else "",
                "FarthestBodyEdgeLateralOffsetM": far_offset if far_offset is not None else "",
                "NearestBodyEdgeBearingDeg": near_angle if near_angle is not None else "",
                "FarthestBodyEdgeBearingDeg": far_angle if far_angle is not None else "",
                "BodyFootprintCrossesSensorCenterline": (
                    int(near_offset == 0.0) if near_offset is not None else ""
                ),
                "RadarMeasurementValid": int(radar_valid),
                "RadarValidityReason": source["RadarValidityReason"],
            }
        )
    return output


def bin_index(value: float) -> int | None:
    """Locate one absolute angle in the fixed descriptive bins."""

    for index, (lower, upper) in enumerate(zip(BEARING_EDGES, BEARING_EDGES[1:])):
        if lower <= value < upper or (index == len(BEARING_EDGES) - 2 and value == upper):
            return index
    return None


def rate_tables(rows: Sequence[Mapping[str, object]]) -> list[dict[str, object]]:
    """Build directly comparable center-angle and near-edge rate tables."""

    metrics = (
        ("CENTER", "TargetCenterBearingValid", "AbsTargetCenterBearingDeg"),
        ("NEAREST_BODY_EDGE", "TargetFootprintValid", "NearestBodyEdgeBearingDeg"),
    )
    output: list[dict[str, object]] = []
    for metric, valid_column, value_column in metrics:
        counts = [dict(exposure=0, invalid=0) for _ in range(len(BEARING_EDGES) - 1)]
        for row in rows:
            if row[valid_column] != 1:
                continue
            value = parse_float(row[value_column])
            located = bin_index(value) if value is not None else None
            if located is None:
                continue
            counts[located]["exposure"] += 1
            if row["RadarMeasurementValid"] == 0:
                counts[located]["invalid"] += 1
        for index, count in enumerate(counts):
            exposure = count["exposure"]
            invalid = count["invalid"]
            output.append(
                {
                    "GeometryMetric": metric,
                    "AbsAngleLowerDeg": BEARING_EDGES[index],
                    "AbsAngleUpperDeg": BEARING_EDGES[index + 1],
                    "ExposureFrameCount": exposure,
                    "RadarInvalidFrameCount": invalid,
                    "RadarInvalidRate": invalid / exposure if exposure else "",
                    "InterpretationAllowed": int(exposure >= 100),
                }
            )
    return output


def rank_auc(scores_and_labels: Sequence[tuple[float, int]]) -> float | None:
    """Return tie-aware AUC using average ranks and no third-party library."""

    positives = sum(label == 1 for _, label in scores_and_labels)
    negatives = len(scores_and_labels) - positives
    if positives == 0 or negatives == 0:
        return None
    ordered = sorted(scores_and_labels, key=lambda item: item[0])
    positive_rank_sum = 0.0
    index = 0
    while index < len(ordered):
        end = index + 1
        while end < len(ordered) and ordered[end][0] == ordered[index][0]:
            end += 1
        average_rank = ((index + 1) + end) / 2.0
        positive_rank_sum += average_rank * sum(label == 1 for _, label in ordered[index:end])
        index = end
    return (positive_rank_sum - positives * (positives + 1) / 2.0) / (positives * negatives)


def auc_summary(rows: Sequence[Mapping[str, object]]) -> dict[str, object]:
    """Compare ranking association, not classification or causal accuracy."""

    paired_rows = [row for row in rows if row["TargetFootprintValid"] == 1]
    center_values = [
        (float(row["AbsTargetCenterBearingDeg"]), int(row["RadarMeasurementValid"] == 0))
        for row in paired_rows
    ]
    edge_values = [
        (float(row["NearestBodyEdgeBearingDeg"]), int(row["RadarMeasurementValid"] == 0))
        for row in paired_rows
    ]
    center_auc = rank_auc(center_values)
    edge_auc = rank_auc(edge_values)
    return {
        "PairedEligibleFrameCount": len(paired_rows),
        "PairedRadarInvalidFrameCount": sum(label for _, label in edge_values),
        "CenterBearingRankAUC": center_auc,
        "NearestBodyEdgeBearingRankAUC": edge_auc,
        "AucDifferenceEdgeMinusCenter": (
            edge_auc - center_auc
            if edge_auc is not None and center_auc is not None
            else None
        ),
        "Interpretation": "DESCRIPTIVE_ASSOCIATION_ONLY",
    }


def candidate_summaries(
    transition_rows: Sequence[Mapping[str, str]],
    transition_columns: Sequence[str],
    frame_rows: Sequence[Mapping[str, object]],
) -> list[dict[str, object]]:
    """Preserve B16 identities and summarize footprint evidence during loss."""

    require_columns(
        transition_columns,
        (
            "CandidateId",
            "CandidateRank",
            "RadarInvalidOnsetFrame",
            "RadarLastInvalidFrame",
            "RadarInvalidDurationFrames",
            "TransitionClass",
        ),
        "B16_CANDIDATE",
    )
    by_frame = {int(row["FrameCount"]): row for row in frame_rows}
    output: list[dict[str, object]] = []
    seen: set[str] = set()
    for candidate in sorted(
        transition_rows,
        key=lambda row: parse_int(row["CandidateRank"], "CANDIDATE_RANK"),
    ):
        identity = str(candidate["CandidateId"])
        if identity in seen:
            raise B17Error("CANDIDATE_DUPLICATE", identity)
        seen.add(identity)
        onset = parse_int(candidate["RadarInvalidOnsetFrame"], "LOSS_ONSET")
        end = parse_int(candidate["RadarLastInvalidFrame"], "LOSS_END")
        segment = [by_frame[frame] for frame in range(onset, end + 1) if frame in by_frame]
        eligible = [row for row in segment if row["TargetFootprintValid"] == 1]
        center_angles = [float(row["AbsTargetCenterBearingDeg"]) for row in eligible]
        edge_angles = [float(row["NearestBodyEdgeBearingDeg"]) for row in eligible]
        widths = [float(row["SelectedCameraWidth"]) for row in eligible]
        distances = [float(row["CanonicalCameraDistance"]) for row in eligible]
        centerline_frames = sum(
            row["BodyFootprintCrossesSensorCenterline"] == 1 for row in eligible
        )
        max_center = max(center_angles) if center_angles else ""
        max_edge = max(edge_angles) if edge_angles else ""
        memo_en = (
            f"Angular-footprint evidence: Radar invalid FC {onset}-{end}; "
            f"max target-center angle {max_center:.2f} deg and max nearest-body-edge "
            f"angle {max_edge:.2f} deg; body crossed sensor centerline for "
            f"{centerline_frames}/{len(eligible)} eligible loss frames. "
            f"Descriptive evidence only."
            if eligible
            else f"Angular-footprint evidence unavailable during Radar invalid FC {onset}-{end}."
        )
        memo_ja = (
            f"車体角度範囲の証拠：Radar無効 FC {onset}-{end}、対象中心角最大 "
            f"{max_center:.2f} deg、車体最近縁角最大 {max_edge:.2f} deg、中心線を覆うFrame "
            f"{centerline_frames}/{len(eligible)}。記述的証拠であり原因確定ではありません。"
            if eligible
            else f"Radar無効 FC {onset}-{end} では車体角度範囲の証拠を取得できませんでした。"
        )
        output.append(
            {
                "CandidateId": identity,
                "CandidateRank": candidate["CandidateRank"],
                "RadarInvalidOnsetFrame": onset,
                "RadarLastInvalidFrame": end,
                "RadarInvalidDurationFrames": candidate["RadarInvalidDurationFrames"],
                "FootprintEligibleInvalidFrameCount": len(eligible),
                "MedianSelectedCameraWidthM": statistics.median(widths) if widths else "",
                "MedianCameraDistanceM": statistics.median(distances) if distances else "",
                "MaxAbsTargetCenterBearingDeg": max_center,
                "MedianAbsTargetCenterBearingDeg": statistics.median(center_angles) if center_angles else "",
                "MaxNearestBodyEdgeBearingDeg": max_edge,
                "MedianNearestBodyEdgeBearingDeg": statistics.median(edge_angles) if edge_angles else "",
                "CenterlineCoverageFrameCount": centerline_frames,
                "CenterlineCoverageFraction": centerline_frames / len(eligible) if eligible else "",
                "PreviousTransitionClass": candidate["TransitionClass"],
                "ReviewMemoEnglish": memo_en,
                "ReviewMemoJapanese": memo_ja,
                "RootCauseClaim": 0,
            }
        )
    return output


def discover_source_run(
    workspace_root: Path,
    b14_run_id: str,
) -> tuple[Path, dict[str, object]]:
    """Resolve the Golden Run recorded by the accepted B14 output."""

    b14_root = workspace_root / "07_OUTPUT" / b14_run_id
    quality = load_json(b14_root / "b14_quality_report.json", "B14_QUALITY")
    if quality.get("Status") != "SUCCEEDED":
        raise B17Error("B14_STATUS", quality.get("Status"))
    source_run_id = str(quality.get("SourceRunId", ""))
    source_root = workspace_root / "07_OUTPUT" / source_run_id
    status = load_json(source_root / "run_status.json", "SOURCE_RUN_STATUS")
    if status.get("Status") != "SUCCEEDED":
        raise B17Error("SOURCE_RUN_STATUS", status.get("Status"))
    return source_root, quality


def prepare(
    workspace_root: Path,
    b14_run_id: str,
    b16_run_id: str,
    output_run_id: str,
    card_path: Path,
) -> str:
    """Fail early on all required paths and semantic Width interfaces."""

    source_root, _ = discover_source_run(workspace_root, b14_run_id)
    required = (
        workspace_root / "07_OUTPUT" / b14_run_id / "frame_bearing_evidence.csv",
        workspace_root / "07_OUTPUT" / b16_run_id / "b16_candidate_transition_assessment.csv",
        card_path,
    )
    missing = [path.name for path in required if not path.is_file()]
    if missing:
        raise B17Error("PREPARE_INPUT", "|".join(missing))
    if (workspace_root / "07_OUTPUT" / output_run_id).exists():
        raise B17Error("OUTPUT_ALREADY_EXISTS", output_run_id)
    frame_path, groups, _ = discover_width_groups(workspace_root, source_root)
    return f"B17P_N3-W{len(groups)}-F{frame_path.stat().st_size}-X0-C0"


def execute(
    workspace_root: Path,
    b14_run_id: str,
    b16_run_id: str,
    output_run_id: str,
    card_path: Path,
) -> str:
    """Run the bounded angular-footprint assessment."""

    source_root, b14_quality = discover_source_run(workspace_root, b14_run_id)
    output_root = workspace_root / "07_OUTPUT" / output_run_id
    if output_root.exists():
        raise B17Error("OUTPUT_ALREADY_EXISTS", output_run_id)
    card = load_json(card_path, "ALGORITHM_CARD")
    if card.get("AlgorithmId") != PROGRAM_ID or card.get("AlgorithmVersion") != PROGRAM_VERSION:
        raise B17Error("ALGORITHM_CARD_IDENTITY")

    b14_path = workspace_root / "07_OUTPUT" / b14_run_id / "frame_bearing_evidence.csv"
    b16_path = (
        workspace_root
        / "07_OUTPUT"
        / b16_run_id
        / "b16_candidate_transition_assessment.csv"
    )
    b14_rows, b14_columns = read_csv(b14_path)
    b16_rows, b16_columns = read_csv(b16_path)
    wanted_frames = {parse_int(row.get("FrameCount"), "B14_FRAME") for row in b14_rows}
    frame_path, groups, width_metadata = discover_width_groups(workspace_root, source_root)
    widths = load_selected_widths(frame_path, groups, wanted_frames)
    frames = derive_footprint_rows(b14_rows, b14_columns, widths)
    rates = rate_tables(frames)
    association = auc_summary(frames)
    candidates = candidate_summaries(b16_rows, b16_columns, frames)

    eligible = sum(row["TargetFootprintValid"] == 1 for row in frames)
    invalid = sum(
        row["TargetFootprintValid"] == 1 and row["RadarMeasurementValid"] == 0
        for row in frames
    )
    centerline_invalid = sum(
        row["TargetFootprintValid"] == 1
        and row["RadarMeasurementValid"] == 0
        and row["BodyFootprintCrossesSensorCenterline"] == 1
        for row in frames
    )
    center_auc = association["CenterBearingRankAUC"]
    edge_auc = association["NearestBodyEdgeBearingRankAUC"]

    try:
        output_root.mkdir(parents=True, exist_ok=False)
        write_csv(output_root / "b17_frame_angular_footprint_evidence.csv", list(frames[0]), frames)
        write_csv(output_root / "b17_loss_rate_by_geometry_bin.csv", list(rates[0]), rates)
        write_csv(output_root / "b17_candidate_footprint_summary.csv", list(candidates[0]), candidates)
        summary = {
            "Status": "SUCCEEDED",
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "SourceGoldenRunId": b14_quality["SourceRunId"],
            "B14RunId": b14_run_id,
            "B16RunId": b16_run_id,
            "OutputRunId": output_run_id,
            "FrameCount": len(frames),
            "FootprintEligibleFrameCount": eligible,
            "FootprintEligibleRadarInvalidFrameCount": invalid,
            "CenterlineCoveredRadarInvalidFrameCount": centerline_invalid,
            "CandidateCount": len(candidates),
            "AssociationAssessment": association,
            "WidthBindings": width_metadata,
            "WidthSourceFrameViewSha256": sha256_file(frame_path),
            "B14FrameEvidenceSha256": sha256_file(b14_path),
            "B16CandidateEvidenceSha256": sha256_file(b16_path),
            "AlgorithmCardSha256": sha256_file(card_path),
            "CandidateIdentityPreserved": True,
            "S1Modified": False,
            "U001Modified": False,
            "AngleThresholdApproved": False,
            "RootCauseClaim": False,
        }
        write_json(output_root / "b17_assessment_summary.json", summary)
    except Exception:
        if output_root.exists():
            shutil.rmtree(output_root)
        raise

    center_thousand = round(float(center_auc) * 1000) if center_auc is not None else 9999
    edge_thousand = round(float(edge_auc) * 1000) if edge_auc is not None else 9999
    capsule = (
        f"B17_N{len(frames)}-E{eligible}-I{invalid}-O{centerline_invalid}_"
        f"AUC-C{center_thousand}-N{edge_thousand}_K{len(candidates)}-X0-C0"
    )
    (output_root / "B17_RETURN_CAPSULE.txt").write_text(capsule + "\n", encoding="utf-8")
    return capsule


def self_test() -> None:
    """Exercise geometry, Width selection, rate tables, AUC, and candidates."""

    checks = 0

    def check(condition: bool) -> None:
        nonlocal checks
        if not condition:
            raise AssertionError("SELFTEST")
        checks += 1

    record = {
        "CAMERA_1_DISTANCE": {
            "value": "2.0",
            "updated": "1",
            "age": "0",
            "valid": "1",
            "missing_reason": "",
        },
        "CAMERA_3_DISTANCE": {
            "value": "1.8",
            "updated": "1",
            "age": "0",
            "valid": "1",
            "missing_reason": "",
        },
    }
    width, valid, reason = selected_width_evidence("CAMERA_1_DISTANCE", record)
    check(width == 2.0)
    check(valid)
    check(reason == "VALID")
    check(selected_width_evidence("NONE", record)[1] is False)
    stale = dict(record)
    stale["CAMERA_1_DISTANCE"] = dict(record["CAMERA_1_DISTANCE"], age="1")
    check(selected_width_evidence("CAMERA_1_DISTANCE", stale)[2] == "WIDTH_AGE_NOT_ZERO")

    columns = [
        "FrameCount",
        "CanonicalCameraDistance",
        "SelectedCameraLateralPosition",
        "TargetBearingProxyValid",
        "AbsTargetBearingProxyDeg",
        "SelectedCameraRole",
        "RadarMeasurementValid",
        "RadarValidityReason",
    ]
    source_rows = []
    width_rows: dict[int, Mapping[str, object]] = {}
    for frame in range(1, 7):
        lateral = 1.1 if frame <= 3 else 2.5
        center = math.degrees(math.atan2(lateral, 7.0))
        source_rows.append(
            {
                "FrameCount": str(frame),
                "CanonicalCameraDistance": "7",
                "SelectedCameraLateralPosition": str(lateral),
                "TargetBearingProxyValid": "1",
                "AbsTargetBearingProxyDeg": str(center),
                "SelectedCameraRole": "CAMERA_1_DISTANCE",
                "RadarMeasurementValid": "0" if frame >= 4 else "1",
                "RadarValidityReason": "NO_TARGET" if frame >= 4 else "VALID",
            }
        )
        width_rows[frame] = record
    frames = derive_footprint_rows(source_rows, columns, width_rows)
    check(len(frames) == 6)
    check(frames[0]["BodyFootprintCrossesSensorCenterline"] == 0)
    check(abs(float(frames[0]["NearestBodyEdgeLateralOffsetM"]) - 0.1) < 1e-9)
    check(float(frames[0]["NearestBodyEdgeBearingDeg"]) < 1.0)
    check(float(frames[-1]["NearestBodyEdgeBearingDeg"]) > 10.0)
    check(len(rate_tables(frames)) == 2 * (len(BEARING_EDGES) - 1))
    summary = auc_summary(frames)
    check(summary["PairedEligibleFrameCount"] == 6)
    check(summary["NearestBodyEdgeBearingRankAUC"] == 1.0)
    check(rank_auc([(0.0, 0), (0.0, 1)]) == 0.5)

    transition = [
        {
            "CandidateId": "C0001",
            "CandidateRank": "1",
            "RadarInvalidOnsetFrame": "4",
            "RadarLastInvalidFrame": "6",
            "RadarInvalidDurationFrames": "3",
            "TransitionClass": "OFF_AXIS_WITH_OBSERVED_RECOVERY",
        }
    ]
    candidates = candidate_summaries(transition, list(transition[0]), frames)
    check(len(candidates) == 1)
    check(candidates[0]["CandidateId"] == "C0001")
    check(candidates[0]["FootprintEligibleInvalidFrameCount"] == 3)
    check(candidates[0]["RootCauseClaim"] == 0)
    check("Descriptive evidence only" in str(candidates[0]["ReviewMemoEnglish"]))
    check(parse_int("7.0", "T") == 7)
    check(parse_float("NA") is None)
    check(parse_bool("true", "T"))
    check(normalize_can_id("0x221") == 0x221)
    check(normalize_channel("2") == 2)
    check(PROGRAM_VERSION == "0.1.0")

    with tempfile.TemporaryDirectory() as temporary:
        path = Path(temporary) / "x.txt"
        path.write_text("x", encoding="utf-8")
        check(len(sha256_file(path)) == 64)

    print(f"U22B17-SELFTEST-PASS-{checks}-C0")


def parser() -> argparse.ArgumentParser:
    """Build the three-command B17 interface."""

    result = argparse.ArgumentParser(description=PROGRAM_ID)
    mode = result.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    result.add_argument("--workspace-root", type=Path, default=Path("U-UDP002"))
    result.add_argument("--b14-run-id", default=DEFAULT_B14_RUN_ID)
    result.add_argument("--b16-run-id", default=DEFAULT_B16_RUN_ID)
    result.add_argument("--output-run-id", default=DEFAULT_OUTPUT_RUN_ID)
    result.add_argument(
        "--algorithm-card",
        type=Path,
        default=Path(
            "U-UDP002/04_TEMP/"
            "u002_u22b17_target_angular_footprint_assessment_algorithm_card_v0_1.json"
        ),
    )
    return result


def main() -> int:
    """Execute one explicit mode and emit one bounded result line."""

    arguments = parser().parse_args()
    try:
        if arguments.self_test:
            self_test()
        elif arguments.prepare:
            print(
                prepare(
                    arguments.workspace_root,
                    arguments.b14_run_id,
                    arguments.b16_run_id,
                    arguments.output_run_id,
                    arguments.algorithm_card,
                )
            )
        else:
            print(
                execute(
                    arguments.workspace_root,
                    arguments.b14_run_id,
                    arguments.b16_run_id,
                    arguments.output_run_id,
                    arguments.algorithm_card,
                )
            )
        return 0
    except B17Error as error:
        print(f"B17E_K{error.reason}-D{error.detail}_X1-C7")
        return 7
    except Exception as error:
        print(f"B17E_KUNEXPECTED-D{machine_token(type(error).__name__)}_X1-C7")
        return 7


if __name__ == "__main__":
    sys.exit(main())
