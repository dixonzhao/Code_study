#!/usr/bin/env python3
"""Read-only source/target pandas path probe for U001 PK03."""

from __future__ import annotations

import argparse
import json
import os
import re
import subprocess
import sys
import tempfile
from pathlib import Path
from typing import Sequence


PROGRAM_NAME = "u001_pk03_pandas_path_probe_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "U001-PK03-PANDAS-PATH-PROBE-SELFTEST-PASS-14"
PACKAGE_NAME = "U001_Portable_v0_1"


class ProbeError(RuntimeError):
    pass


def atomic_write(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_bytes(text.encode("utf-8"))
    temporary.replace(path)


def latest_build_root(udp001_root: Path) -> Path:
    candidates = sorted(
        (path for path in (udp001_root / "04_TEMP").glob("PK03_BUILD_WORK_*") if path.is_dir()),
        key=lambda path: path.stat().st_mtime,
        reverse=True,
    )
    if not candidates:
        raise ProbeError("PK03_BUILD_WORK_NOT_FOUND")
    return candidates[0]


def load_runtime_report(path: Path) -> dict:
    if not path.is_file():
        raise ProbeError("RUNTIME_REPORT_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, json.JSONDecodeError) as exc:
        raise ProbeError("RUNTIME_REPORT_INVALID") from exc
    if value.get("Schema") != "UDP001_PORTABLE_RUNTIME_PROBE_V0_1":
        raise ProbeError("RUNTIME_REPORT_SCHEMA_MISMATCH")
    return value


def runtime_environment(runtime: Path) -> dict[str, str]:
    environment = os.environ.copy()
    environment.pop("VIRTUAL_ENV", None)
    environment.pop("PYTHONPATH", None)
    environment["PYTHONHOME"] = str(runtime.resolve())
    environment["PYTHONNOUSERSITE"] = "1"
    system_root = environment.get("SystemRoot", r"C:\Windows")
    environment["PATH"] = os.pathsep.join(
        (str(runtime.resolve()), str((runtime / "DLLs").resolve()), str(Path(system_root) / "System32"), system_root)
    )
    return environment


def run_command(command: list[str], environment: dict[str, str] | None = None) -> dict:
    try:
        result = subprocess.run(
            command,
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=120,
            shell=False,
            env=environment,
        )
        return {"ReturnCode": result.returncode, "Stdout": result.stdout, "Stderr": result.stderr}
    except (OSError, subprocess.SubprocessError) as exc:
        return {"ReturnCode": 999, "Stdout": "", "Stderr": f"{type(exc).__name__}:{exc}"}


def last_nonempty(text: str) -> str:
    values = [line.strip() for line in text.splitlines() if line.strip()]
    return values[-1] if values else ""


def inside(root: Path, value: str) -> bool:
    try:
        Path(value).resolve().relative_to(root.resolve())
        return True
    except (OSError, ValueError):
        return False


def path_list(result: dict) -> list[str]:
    if result.get("ReturnCode") != 0:
        return []
    try:
        value = json.loads(last_nonempty(result.get("Stdout", "")))
    except json.JSONDecodeError:
        return []
    return value if isinstance(value, list) else []


def compact_memo(tag: str, result: dict) -> str:
    value = last_nonempty(result.get("Stderr", "")) or last_nonempty(result.get("Stdout", "")) or "NO_DETAIL"
    value = re.sub(r"\s+", "_", value)
    value = re.sub(r"[^A-Za-z0-9_.:+\\/-]", "?", value)
    return f"M-{tag}-{value[:160]}"


def run_probe(udp001_root: Path, runtime_report_path: Path, out: Path) -> int:
    if not udp001_root.is_dir() or udp001_root.name != "UDP001":
        raise ProbeError("UDP001_ROOT_INVALID")
    if out.exists() and (not out.is_dir() or any(out.iterdir())):
        raise ProbeError("OUTPUT_MUST_BE_ABSENT_OR_EMPTY")
    out.mkdir(parents=True, exist_ok=True)
    report = load_runtime_report(runtime_report_path)
    source_site = Path(report.get("Environment", {}).get("SitePackages", ""))
    build = latest_build_root(udp001_root)
    runtime = build / "release_stage" / PACKAGE_NAME / "_internal" / "runtime"
    target_site = runtime / "Lib" / "site-packages"
    target_python = runtime / "python.exe"

    source_import = run_command([sys.executable, "-c", "import pandas;print(pandas.__file__)"])
    source_origin = last_nonempty(source_import["Stdout"])
    source_import_ok = source_import["ReturnCode"] == 0 and inside(source_site, source_origin)

    import_script = "import pandas;print(pandas.__file__)"
    path_script = "import json,sys;print(json.dumps(sys.path))"
    environment = runtime_environment(runtime)
    normal_import = run_command([str(target_python), "-c", import_script], environment)
    isolated_import = run_command([str(target_python), "-I", "-c", import_script], environment)
    normal_path_result = run_command([str(target_python), "-c", path_script], environment)
    isolated_path_result = run_command([str(target_python), "-I", "-c", path_script], environment)
    normal_paths = path_list(normal_path_result)
    isolated_paths = path_list(isolated_path_result)

    flags = {
        "Report": runtime_report_path.is_file(),
        "SourceSite": source_site.is_dir(),
        "SourceImport": source_import_ok,
        "SourcePandas": (source_site / "pandas").is_dir(),
        "TargetPandas": (target_site / "pandas").is_dir(),
        "TargetInit": (target_site / "pandas" / "__init__.py").is_file(),
        "TargetMetadata": any(target_site.glob("pandas-*.dist-info")),
        "NormalPath": any(inside(target_site, value) or Path(value).resolve() == target_site.resolve() for value in normal_paths),
        "IsolatedPath": any(inside(target_site, value) or Path(value).resolve() == target_site.resolve() for value in isolated_paths),
        "NormalImport": normal_import["ReturnCode"] == 0 and inside(runtime, last_nonempty(normal_import["Stdout"])),
        "IsolatedImport": isolated_import["ReturnCode"] == 0 and inside(runtime, last_nonempty(isolated_import["Stdout"])),
    }
    errors = sum(not value for value in flags.values())
    detail = {
        "Schema": "UDP001_PK03_PANDAS_PATH_PROBE_V0_1",
        "BuildRoot": str(build),
        "SourceSitePackages": str(source_site),
        "SourcePandasOrigin": source_origin,
        "TargetSitePackages": str(target_site),
        "Flags": flags,
        "NormalSysPath": normal_paths,
        "IsolatedSysPath": isolated_paths,
        "SourceImport": source_import,
        "NormalImport": normal_import,
        "IsolatedImport": isolated_import,
        "NormalPathResult": normal_path_result,
        "IsolatedPathResult": isolated_path_result,
        "BlockingCount": errors,
    }
    atomic_write(out / "u001_pk03_pandas_path_probe_report.json", json.dumps(detail, ensure_ascii=False, indent=2) + "\n")
    priority = (("N", normal_import), ("I", isolated_import), ("S", source_import))
    memo = next((compact_memo(tag, result) for tag, result in priority if result["ReturnCode"] != 0), "M-NONE")
    capsule = (
        "U001PK03D2\n"
        f"N-R{int(flags['Report'])}-S{int(flags['SourceSite'])}-O{int(flags['SourceImport'])}-"
        f"D{int(flags['SourcePandas'])}-T{int(flags['TargetPandas'])}-I{int(flags['TargetInit'])}-"
        f"M{int(flags['TargetMetadata'])}-P{int(flags['NormalPath'])}-J{int(flags['IsolatedPath'])}-"
        f"N{int(flags['NormalImport'])}-Q{int(flags['IsolatedImport'])}-X{errors}\n"
        f"RC-S{source_import['ReturnCode']}-N{normal_import['ReturnCode']}-I{isolated_import['ReturnCode']}\n"
        f"{memo}\n"
        "C-0\n"
    )
    atomic_write(out / "u001_pk03_pandas_path_probe_capsule.txt", capsule)
    print(capsule, end="")
    return 0


def run_self_test() -> None:
    passed = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td)
        nested = root / "a" / "b"
        nested.mkdir(parents=True)
        passed += int(inside(root, str(nested)))
        passed += int(not inside(nested, str(root)))
        passed += int(last_nonempty("a\n\nb\n") == "b")
        good = {"ReturnCode": 0, "Stdout": '["a", "b"]\n', "Stderr": ""}
        passed += int(path_list(good) == ["a", "b"])
        passed += int(path_list({"ReturnCode": 1, "Stdout": "[]", "Stderr": "x"}) == [])
        memo = compact_memo("N", {"ReturnCode": 1, "Stdout": "", "Stderr": "No module named x"})
        passed += int(memo == "M-N-No_module_named_x")
        environment = runtime_environment(root)
        passed += int(environment["PYTHONHOME"] == str(root.resolve()))
        passed += int(environment["PYTHONNOUSERSITE"] == "1")
        command = run_command([sys.executable, "-c", "print('ok')"])
        passed += int(command["ReturnCode"] == 0)
        passed += int(last_nonempty(command["Stdout"]) == "ok")
        passed += int(PROGRAM_NAME.endswith("v0_1.py"))
        passed += int(PROGRAM_VERSION == "0.1")
        passed += int(PACKAGE_NAME == "U001_Portable_v0_1")
        passed += int(SELFTEST_TOKEN.endswith("PASS-14"))
    if passed != 14:
        raise AssertionError(f"SELFTEST_COUNT:{passed}")
    print(SELFTEST_TOKEN)


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description="Diagnose pandas source/copy/import paths for U001 PK03")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--probe", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--runtime-report")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if args.probe:
        if not args.udp001_root or not args.runtime_report or not args.out:
            parser.error("--probe requires --udp001-root --runtime-report --out")
        try:
            return run_probe(Path(args.udp001_root), Path(args.runtime_report), Path(args.out))
        except ProbeError as exc:
            print(f"U001PK03D2-ERROR:{exc}")
            print("C-7")
            return 7
    parser.error("select --self-test or --probe")
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
