#!/usr/bin/env python3
"""Atomic, lineage-aware artifact persistence for the U002 U22 runtime."""

from __future__ import annotations

import argparse
import csv
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import re
import sys
from typing import Any, Dict, List, Mapping, Sequence, Tuple


PROGRAM_ID = "U002-U22-ARTIFACT-STORE"
VERSION = "0.1"
CONTRACT_VERSION = "U22.0.1"
STAGE_ORDER: Tuple[str, ...] = (
    "SOURCE",
    "S0",
    "S1",
    "S2",
    "S3",
    "S4",
    "S5",
    "S6",
    "S7",
)
VALIDATOR_RELATIVE = Path("01_PROGRAMS/core/u002_u22_contract_validator_v0_1.py")
INDEX_NAME = "run_artifact_index_v0_1.json"
MANIFEST_NAME = "artifact_manifest.json"
IDENTIFIER_PATTERN = re.compile(r"^[A-Za-z0-9][A-Za-z0-9_.-]{0,127}$")


class StoreError(RuntimeError):
    """Represent one deterministic fail-closed Artifact Store failure."""

    def __init__(self, reason: str, detail: str = ""):
        self.reason = stable(reason)
        self.detail = stable(detail) if detail else "NA"
        super().__init__(self.reason)


def stable(value: object) -> str:
    """Convert an exception detail into a short return-capsule token."""

    normalized = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper())
    return normalized.strip("_")[:80] or "INTERNAL"


def canonical_bytes(value: object) -> bytes:
    """Serialize control metadata deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file without loading it entirely into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    """Accept ordinary files only; symbolic links are never trusted artifacts."""

    return path.is_file() and not path.is_symlink()


def inside(child: Path, parent: Path) -> bool:
    """Return True only when the resolved child remains inside the parent."""

    try:
        child.resolve().relative_to(parent.resolve())
        return True
    except (OSError, ValueError):
        return False


def require_workspace(path: Path) -> Path:
    """Validate the U002 workspace identity."""

    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise StoreError("WORKSPACE_ROOT", root.name)
    return root


def require_run_root(workspace_root: Path, path: Path) -> Path:
    """Restrict formal Run output to the workspace 07_OUTPUT directory."""

    run_root = path.resolve()
    output_root = (workspace_root / "07_OUTPUT").resolve()
    if not run_root.is_dir() or run_root.is_symlink():
        raise StoreError("RUN_ROOT", str(run_root))
    if run_root.parent != output_root:
        raise StoreError("RUN_ROOT_LOCATION", str(run_root))
    if IDENTIFIER_PATTERN.fullmatch(run_root.name) is None:
        raise StoreError("RUN_ID_FORMAT", run_root.name)
    return run_root


def load_validator(root: Path) -> Any:
    """Load only the approved Core validator from its exact workspace path."""

    path = root / VALIDATOR_RELATIVE
    if not regular_file(path) or not inside(path, root):
        raise StoreError("VALIDATOR_MISSING", str(path))
    spec = importlib.util.spec_from_file_location(
        "u002_u22_contract_validator",
        path,
    )
    if spec is None or spec.loader is None:
        raise StoreError("VALIDATOR_LOAD", str(path))
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    required_api = ("load_json", "validate_document")
    if any(not hasattr(module, name) for name in required_api):
        raise StoreError("VALIDATOR_API", str(path))
    if getattr(module, "CONTRACT_VERSION", None) != CONTRACT_VERSION:
        raise StoreError("VALIDATOR_VERSION", str(path))
    return module


def safe_identifier(value: object, field: str) -> str:
    """Protect canonical directory names from traversal and ambiguous spelling."""

    text = str(value)
    if IDENTIFIER_PATTERN.fullmatch(text) is None:
        raise StoreError(field + "_FORMAT", text)
    return text


def load_index(run_root: Path) -> Dict[str, Any]:
    """Load and structurally validate the deterministic per-Run index."""

    path = run_root / INDEX_NAME
    if not path.exists():
        return {
            "StoreContract": "U002_U22_ARTIFACT_INDEX",
            "StoreVersion": VERSION,
            "ContractVersion": CONTRACT_VERSION,
            "RunId": run_root.name,
            "Artifacts": [],
        }
    if not regular_file(path):
        raise StoreError("INDEX_NOT_REGULAR", str(path))
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise StoreError("INDEX_JSON", str(path)) from exc
    expected_keys = {
        "StoreContract",
        "StoreVersion",
        "ContractVersion",
        "RunId",
        "Artifacts",
    }
    if not isinstance(value, dict) or set(value) != expected_keys:
        raise StoreError("INDEX_SHAPE", str(path))
    if value["StoreContract"] != "U002_U22_ARTIFACT_INDEX":
        raise StoreError("INDEX_CONTRACT", str(path))
    if value["StoreVersion"] != VERSION:
        raise StoreError("INDEX_VERSION", str(path))
    if value["ContractVersion"] != CONTRACT_VERSION:
        raise StoreError("INDEX_CONTRACT_VERSION", str(path))
    if value["RunId"] != run_root.name:
        raise StoreError("INDEX_RUN_ID", str(path))
    if not isinstance(value["Artifacts"], list):
        raise StoreError("INDEX_ARTIFACTS", str(path))
    identities = [item.get("ArtifactId") for item in value["Artifacts"] if isinstance(item, dict)]
    if len(identities) != len(value["Artifacts"]):
        raise StoreError("INDEX_ENTRY_TYPE", str(path))
    if len(identities) != len(set(identities)):
        raise StoreError("INDEX_DUPLICATE_ID", str(path))
    return value


def write_index(run_root: Path, value: Mapping[str, Any]) -> None:
    """Replace the index atomically so partial JSON is never accepted."""

    path = run_root / INDEX_NAME
    temporary = path.with_name(path.name + ".writing")
    if temporary.exists():
        raise StoreError("INDEX_TEMP_EXISTS", str(temporary))
    temporary.write_bytes(canonical_bytes(value))
    os.replace(temporary, path)


def count_csv_rows(path: Path) -> int:
    """Count CSV data records while respecting quoted newlines."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        try:
            next(reader)
        except StopIteration:
            return 0
        return sum(1 for _ in reader)


def resolve_payload(package_root: Path, relative_path: str) -> Path:
    """Resolve one manifest payload without allowing escape from its package."""

    if "\\" in relative_path:
        raise StoreError("PAYLOAD_SEPARATOR", relative_path)
    candidate = package_root.joinpath(*relative_path.split("/"))
    if not regular_file(candidate) or not inside(candidate, package_root):
        raise StoreError("PAYLOAD_MISSING", relative_path)
    return candidate


def declared_files(package_root: Path, manifest: Mapping[str, Any]) -> List[str]:
    """Return the sorted set of files that the manifest is allowed to contain."""

    declared = [MANIFEST_NAME]
    declared.extend(str(item["RelativePath"]) for item in manifest["Payloads"])
    return sorted(declared, key=str.casefold)


def actual_files(package_root: Path) -> List[str]:
    """List regular package members and reject symbolic links or nested escapes."""

    result: List[str] = []
    for path in package_root.rglob("*"):
        if path.is_symlink():
            raise StoreError("PACKAGE_SYMLINK", str(path))
        if path.is_file():
            result.append(path.relative_to(package_root).as_posix())
    return sorted(result, key=str.casefold)


def verify_package(
    workspace_root: Path,
    package_root: Path,
    expected_run_id: str,
) -> Tuple[Dict[str, Any], List[Dict[str, Any]]]:
    """Validate one complete artifact package without modifying it."""

    if not package_root.is_dir() or package_root.is_symlink():
        raise StoreError("PACKAGE_ROOT", str(package_root))
    manifest_path = package_root / MANIFEST_NAME
    validator = load_validator(workspace_root)
    manifest = validator.load_json(manifest_path, "ARTIFACT_MANIFEST_JSON")
    validator.validate_document(workspace_root, "ARTIFACT", manifest)
    artifact_id = safe_identifier(manifest["ArtifactId"], "ARTIFACT_ID")
    safe_identifier(manifest["RunId"], "RUN_ID")
    if manifest["RunId"] != expected_run_id:
        raise StoreError("MANIFEST_RUN_ID", manifest["RunId"])
    if manifest["Stage"] not in STAGE_ORDER:
        raise StoreError("MANIFEST_STAGE", manifest["Stage"])
    if declared_files(package_root, manifest) != actual_files(package_root):
        raise StoreError("PACKAGE_FILE_SET", artifact_id)
    verified_payloads: List[Dict[str, Any]] = []
    for payload in manifest["Payloads"]:
        relative_path = str(payload["RelativePath"])
        path = resolve_payload(package_root, relative_path)
        observed_size = path.stat().st_size
        observed_hash = sha256_file(path)
        if observed_size != payload["ByteSize"]:
            raise StoreError("PAYLOAD_SIZE", relative_path)
        if observed_hash != payload["Sha256"]:
            raise StoreError("PAYLOAD_HASH", relative_path)
        if payload["RowCount"] is not None and payload["MediaType"] == "text/csv":
            observed_rows = count_csv_rows(path)
            if observed_rows != payload["RowCount"]:
                raise StoreError("PAYLOAD_ROW_COUNT", relative_path)
        verified_payloads.append(
            {
                "RelativePath": relative_path,
                "Sha256": observed_hash,
                "ByteSize": observed_size,
                "RowCount": payload["RowCount"],
                "PayloadKind": payload["PayloadKind"],
            }
        )
    return manifest, verified_payloads


def canonical_artifact_directory(run_root: Path, manifest: Mapping[str, Any]) -> Path:
    """Map artifact identity to its single authoritative Run directory."""

    stage = str(manifest["Stage"])
    artifact_id = safe_identifier(manifest["ArtifactId"], "ARTIFACT_ID")
    return run_root / "03_ARTIFACTS" / stage / artifact_id


def index_entry(
    run_root: Path,
    package_root: Path,
    manifest: Mapping[str, Any],
    payloads: Sequence[Mapping[str, Any]],
) -> Dict[str, Any]:
    """Build a stable index record from an already verified package."""

    manifest_path = package_root / MANIFEST_NAME
    return {
        "ArtifactId": manifest["ArtifactId"],
        "ArtifactType": manifest["ArtifactType"],
        "Stage": manifest["Stage"],
        "ProducerModuleId": manifest["ProducerModuleId"],
        "ProducerModuleVersion": manifest["ProducerModuleVersion"],
        "ParentArtifactIds": list(manifest["ParentArtifactIds"]),
        "ManifestRelativePath": manifest_path.relative_to(run_root).as_posix(),
        "ManifestSha256": sha256_file(manifest_path),
        "Payloads": list(payloads),
    }


def validate_parent_lineage(
    manifest: Mapping[str, Any],
    artifacts: Sequence[Mapping[str, Any]],
) -> None:
    """Require every parent to exist and precede or share the child Stage."""

    by_id = {entry["ArtifactId"]: entry for entry in artifacts}
    child_order = STAGE_ORDER.index(str(manifest["Stage"]))
    for parent_id in manifest["ParentArtifactIds"]:
        if parent_id not in by_id:
            raise StoreError("PARENT_MISSING", parent_id)
        parent_order = STAGE_ORDER.index(str(by_id[parent_id]["Stage"]))
        if parent_order > child_order:
            raise StoreError("PARENT_STAGE_ORDER", parent_id)


def entries_equal(left: Mapping[str, Any], right: Mapping[str, Any]) -> bool:
    """Compare two records using their canonical JSON representation."""

    return canonical_bytes(left) == canonical_bytes(right)


def commit_package(
    workspace_root: Path,
    run_root: Path,
    staging_package: Path,
) -> Tuple[str, Dict[str, Any]]:
    """Promote one verified staging package and update the Run index atomically."""

    staging_root = run_root / "00_STAGING"
    if not inside(staging_package, staging_root):
        raise StoreError("PACKAGE_LOCATION", str(staging_package))
    manifest, payloads = verify_package(
        workspace_root,
        staging_package,
        run_root.name,
    )
    index = load_index(run_root)
    validate_parent_lineage(manifest, index["Artifacts"])
    destination = canonical_artifact_directory(run_root, manifest)
    existing_by_id = {
        entry["ArtifactId"]: entry
        for entry in index["Artifacts"]
    }
    if destination.exists():
        existing_manifest, existing_payloads = verify_package(
            workspace_root,
            destination,
            run_root.name,
        )
        existing_entry = index_entry(
            run_root,
            destination,
            existing_manifest,
            existing_payloads,
        )
        candidate_entry = index_entry(
            run_root,
            destination,
            manifest,
            payloads,
        )
        candidate_entry["ManifestSha256"] = sha256_file(staging_package / MANIFEST_NAME)
        if not entries_equal(existing_entry, candidate_entry):
            raise StoreError("ARTIFACT_COLLISION", manifest["ArtifactId"])
        indexed = existing_by_id.get(manifest["ArtifactId"])
        if indexed is None or not entries_equal(indexed, existing_entry):
            raise StoreError("INDEX_DESTINATION_GAP", manifest["ArtifactId"])
        return "REPLAY", existing_entry
    if manifest["ArtifactId"] in existing_by_id:
        raise StoreError("ARTIFACT_ID_COLLISION", manifest["ArtifactId"])
    destination.parent.mkdir(parents=True, exist_ok=True)
    os.replace(staging_package, destination)
    final_manifest, final_payloads = verify_package(
        workspace_root,
        destination,
        run_root.name,
    )
    entry = index_entry(
        run_root,
        destination,
        final_manifest,
        final_payloads,
    )
    index["Artifacts"].append(entry)
    index["Artifacts"].sort(key=lambda item: str(item["ArtifactId"]).casefold())
    write_index(run_root, index)
    return "COMMITTED", entry


def audit_run(workspace_root: Path, run_root: Path) -> Dict[str, int]:
    """Recompute package integrity and lineage for every indexed artifact."""

    index = load_index(run_root)
    artifacts = index["Artifacts"]
    observed_manifest_paths: List[str] = []
    for entry in artifacts:
        manifest_path = run_root / str(entry["ManifestRelativePath"])
        if not regular_file(manifest_path) or not inside(manifest_path, run_root):
            raise StoreError("INDEX_MANIFEST_MISSING", entry.get("ArtifactId"))
        package_root = manifest_path.parent
        manifest, payloads = verify_package(
            workspace_root,
            package_root,
            run_root.name,
        )
        rebuilt = index_entry(run_root, package_root, manifest, payloads)
        if not entries_equal(entry, rebuilt):
            raise StoreError("INDEX_ARTIFACT_MISMATCH", entry.get("ArtifactId"))
        validate_parent_lineage(manifest, artifacts)
        observed_manifest_paths.append(str(entry["ManifestRelativePath"]))
    disk_manifest_paths = [
        path.relative_to(run_root).as_posix()
        for path in (run_root / "03_ARTIFACTS").glob("*/*/artifact_manifest.json")
        if regular_file(path)
    ]
    if sorted(observed_manifest_paths) != sorted(disk_manifest_paths):
        raise StoreError("ORPHAN_OR_MISSING_ARTIFACT")
    payload_count = sum(len(entry["Payloads"]) for entry in artifacts)
    total_bytes = sum(
        payload["ByteSize"]
        for entry in artifacts
        for payload in entry["Payloads"]
    )
    return {
        "ArtifactCount": len(artifacts),
        "PayloadCount": payload_count,
        "PayloadBytes": total_bytes,
    }


def self_test(workspace_root: Path) -> str:
    """Exercise deterministic helpers without creating Run artifacts."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise StoreError("SELFTEST")
        checks.append(True)

    check(PROGRAM_ID.startswith("U002"))
    check(VERSION == "0.1")
    check(CONTRACT_VERSION == "U22.0.1")
    check(len(STAGE_ORDER) == 9)
    check(STAGE_ORDER[0] == "SOURCE")
    check(STAGE_ORDER[-1] == "S7")
    check(stable("a b/c") == "A_B_C")
    check(IDENTIFIER_PATTERN.fullmatch("RUN_001") is not None)
    check(IDENTIFIER_PATTERN.fullmatch("../bad") is None)
    check(IDENTIFIER_PATTERN.fullmatch("bad/name") is None)
    check(inside(workspace_root / "01_PROGRAMS", workspace_root))
    check(not inside(workspace_root.parent, workspace_root))
    check(len(hashlib.sha256(b"x").hexdigest()) == 64)
    check(canonical_bytes({"b": 1, "a": 2}).startswith(b"{"))
    check(INDEX_NAME.endswith(".json"))
    check(MANIFEST_NAME == "artifact_manifest.json")
    check(VALIDATOR_RELATIVE.suffix == ".py")
    check(regular_file(workspace_root / VALIDATOR_RELATIVE))
    check(load_validator(workspace_root).CONTRACT_VERSION == CONTRACT_VERSION)
    empty_index = {
        "StoreContract": "U002_U22_ARTIFACT_INDEX",
        "StoreVersion": VERSION,
        "ContractVersion": CONTRACT_VERSION,
        "RunId": "RUN_TEST",
        "Artifacts": [],
    }
    check(json.loads(canonical_bytes(empty_index))["Artifacts"] == [])
    check(entries_equal(empty_index, json.loads(canonical_bytes(empty_index))))
    check(safe_identifier("A.001", "ARTIFACT_ID") == "A.001")
    check(STAGE_ORDER.index("S2") < STAGE_ORDER.index("S7"))
    check(len(checks) == 23)
    if len(checks) != 24:
        raise StoreError("SELFTEST_COUNT")
    return "U22A3S-ST N-T24-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Build an explicit command line with no implicit operating mode."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--commit-package")
    mode.add_argument("--audit-run", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    parser.add_argument("--run-root")
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    """Execute one Artifact Store operation and emit one compact capsule."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        root = require_workspace(Path(args.workspace_root))
        if args.self_test:
            print(self_test(root))
            return 0
        if args.run_root is None:
            raise StoreError("RUN_ROOT_REQUIRED")
        run_root = require_run_root(root, Path(args.run_root))
        if args.commit_package is not None:
            disposition, entry = commit_package(
                root,
                run_root,
                Path(args.commit_package).resolve(),
            )
            replay = 1 if disposition == "REPLAY" else 0
            payload_count = len(entry["Payloads"])
            print(
                f"U22A3S N-C1-R{replay}-A1-P{payload_count}-X0 C-0"
            )
            return 0
        result = audit_run(root, run_root)
        print(
            f"U22A3S N-C0-R0-A{result['ArtifactCount']}"
            f"-P{result['PayloadCount']}-X0 C-0"
        )
        return 0
    except StoreError as exc:
        print(f"U22A3S N-C0-R0-A0-P0-X1 C-7 R-{exc.reason}")
        return 7
    except Exception as exc:
        reason = stable(exc.__class__.__name__)
        print(f"U22A3S N-C0-R0-A0-P0-X1 C-7 R-{reason}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
