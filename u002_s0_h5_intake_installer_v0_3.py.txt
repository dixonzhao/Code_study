#!/usr/bin/env python3
"""Install and qualify the U001 H5 compatible U002 S0 Module."""

from __future__ import annotations

import argparse
import hashlib
import json
import os
from pathlib import Path
import py_compile
import re
import shutil
import subprocess
import sys
import zipfile


PROGRAM_ID = "U002-S0-H5-INSTALLER"
PROGRAM_VERSION = "0.3.0"
MODULE_ID = "CameraRadar.S0.U001ControlledProfileIntake"
MODULE_VERSION = "0.2.0"
ZIP_NAME = "UDP002_S0_H5_INTAKE_MODULE_V020.zip"
ZIP_SHA256 = "78fa44bebab42c4c570760e76541cb5c1bf810ccae1360e57bc700059e2252a8"
PREPARED_NAME = "U2S0H5_V030_PREPARED"
PROFILE_NAME = "u002_tc001_h5_radar_loss_candidate_v0_2.json"
SIGNAL_PROFILE_ID = "TC001.CAMERA_RADAR.GEOMETRY.WIDTH_V04@0.4.0"
EXPECTED_FILES = {
    "algorithm_card.json": "90531d454b1873cd05fbaac33b50842677e5a7cf2e3859a35de1065afdb2efa7",
    "implementation.py": "015d850305a0c3142ffb6bcd7788ab6ee3aae5b6de7d6cb9111ab155fe91af3f",
    "module_manifest.json": "58608d25515a1298d152ea853f1e33f4a97fc8928dec43a95c2dc0439bd155d9",
    "selftest.py": "1e86a6ba0bd24ec3d5a6c1abecdfd918a108a9439af6a40643afe262ebf131e7",
}


class InstallError(RuntimeError):
    """One bounded installation failure."""


def token(value: object, limit: int = 100) -> str:
    text = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (text or "INTERNAL")[:limit]


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def canonical_bytes(value: object) -> bytes:
    return (
        json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n"
    ).encode("utf-8")


def require_workspace(value: object) -> Path:
    root = Path(str(value)).resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise InstallError("WORKSPACE_ROOT")
    required = (
        "01_PROGRAMS/core/u002_u22_contract_validator_v0_1.py",
        "01_PROGRAMS/core/u002_u22_module_registry_v0_1.py",
        "01_PROGRAMS/core/u002_u22_module_qualification_v0_1.py",
        "02_CONFIG/gui_profiles/u002_radar_loss_golden_reference_v0_1.json",
    )
    for relative in required:
        path = root / relative
        if not path.is_file() or path.is_symlink():
            raise InstallError("REQUIRED_FILE_" + token(relative))
    return root


def transfer_root() -> Path:
    return Path(__file__).resolve().parent


def transfer_zip() -> Path:
    path = transfer_root() / ZIP_NAME
    if not path.is_file() or path.is_symlink():
        raise InstallError("TRANSFER_ZIP_MISSING")
    if sha256_file(path) != ZIP_SHA256:
        raise InstallError("TRANSFER_ZIP_SHA256")
    return path


def prepared_root(workspace: Path) -> Path:
    return workspace / "04_TEMP" / PREPARED_NAME


def prepared_module(workspace: Path) -> Path:
    return prepared_root(workspace) / "S0_module"


def target_module(workspace: Path) -> Path:
    return (
        workspace
        / "01_PROGRAMS/modules/S0"
        / MODULE_ID
        / MODULE_VERSION
    )


def profile_path(workspace: Path) -> Path:
    return workspace / "02_CONFIG/gui_profiles/user" / PROFILE_NAME


def validate_module_files(root: Path) -> None:
    observed = {path.name for path in root.iterdir() if path.is_file()}
    if observed != set(EXPECTED_FILES):
        raise InstallError("MODULE_FILESET")
    for name, expected in EXPECTED_FILES.items():
        if sha256_file(root / name) != expected:
            raise InstallError("MODULE_HASH_" + token(name))
    py_compile.compile(str(root / "implementation.py"), doraise=True)
    py_compile.compile(str(root / "selftest.py"), doraise=True)


def safe_extract(zip_path: Path, destination: Path) -> None:
    with zipfile.ZipFile(zip_path) as archive:
        names = set(archive.namelist())
        expected = {"S0_module/"} | {
            "S0_module/" + name for name in EXPECTED_FILES
        }
        if names != expected:
            raise InstallError("ZIP_FILESET")
        for member in archive.infolist():
            path = Path(member.filename)
            if path.is_absolute() or ".." in path.parts:
                raise InstallError("ZIP_PATH")
        archive.extractall(destination)


def run_module_selftest(workspace: Path, module_root: Path) -> str:
    command = [
        sys.executable,
        str(module_root / "selftest.py"),
        "--module-root",
        str(module_root),
        "--workspace-root",
        str(workspace),
    ]
    result = subprocess.run(
        command,
        cwd=module_root,
        text=True,
        capture_output=True,
        timeout=120,
        check=False,
    )
    lines = [line.strip() for line in result.stdout.splitlines() if line.strip()]
    capsule = lines[-1] if lines else "MISSING"
    if result.returncode != 0 or not capsule.endswith("C-0"):
        raise InstallError("MODULE_SELFTEST_" + token(capsule))
    return capsule


def build_saved_profile(workspace: Path) -> dict:
    default_path = (
        workspace
        / "02_CONFIG/gui_profiles/u002_radar_loss_golden_reference_v0_1.json"
    )
    profile = json.loads(default_path.read_text(encoding="utf-8-sig"))
    plan = profile.get("ModulePlan")
    if not isinstance(plan, list) or len(plan) != 5:
        raise InstallError("DEFAULT_PROFILE_MODULE_COUNT")
    s0 = [item for item in plan if item.get("Stage") == "S0"]
    s6 = [item for item in plan if item.get("Stage") == "S6"]
    if (
        len(s0) != 1
        or s0[0].get("ModuleId") != MODULE_ID
        or s0[0].get("ModuleVersion") != "0.1.1"
    ):
        raise InstallError("DEFAULT_PROFILE_S0_IDENTITY")
    if (
        len(s6) != 1
        or s6[0].get("ModuleId")
        != "CameraRadar.S6.RadarMeasurementLossGeometryCandidateSynthesis"
        or s6[0].get("ModuleVersion") != "0.1.2"
    ):
        raise InstallError("DEFAULT_PROFILE_S6_IDENTITY")
    s0[0]["ModuleVersion"] = MODULE_VERSION
    return {
        "ContractId": "U002_GUI_SAVED_PROFILE",
        "ContractVersion": "0.1",
        "ProfileName": "TC001 H5 Radar-loss Candidate",
        # The legacy GUI profile is only a structural template. Its current
        # SignalProfile may have been changed by an earlier GUI save and must
        # not control the new, explicitly identified TC001 H5 profile.
        "SignalProfile": SIGNAL_PROFILE_ID,
        "ModulePlan": plan,
        "PrimaryOutcome": profile["PrimaryOutcome"],
        "CandidatePolicy": profile["CandidatePolicy"],
        "OutputSelection": profile["OutputSelection"],
    }


def prepare(workspace: Path) -> None:
    zip_path = transfer_zip()
    destination = prepared_root(workspace)
    if destination.exists():
        shutil.rmtree(destination)
    obsolete = workspace / "04_TEMP/U2S0H5_V020_PREPARED"
    if obsolete.exists():
        shutil.rmtree(obsolete)
    destination.mkdir(parents=True)
    safe_extract(zip_path, destination)
    validate_module_files(prepared_module(workspace))
    capsule = run_module_selftest(workspace, prepared_module(workspace))
    build_saved_profile(workspace)
    print("[CHECK] U002 core and frozen Golden Profile exist | WHY: install must extend the accepted workspace, not invent a new root")
    print("[CHECK] Transfer ZIP inventory and SHA-256 match | WHY: company-side bytes must equal the tested release")
    print("[CHECK] S0 0.2.0 self-test passed: " + capsule + " | WHY: H5 intake is tested before promotion")
    print("[CHECK] New profile explicitly pins TC001 and S6 0.1.2 | WHY: a mutable legacy GUI profile must not select the new run identity")
    print("S0H5P3-N4-S1-X0-C0")


def self_test(workspace: Path) -> None:
    validate_module_files(prepared_module(workspace))
    saved = build_saved_profile(workspace)
    plan = saved["ModulePlan"]
    checks = [
        saved["ContractId"] == "U002_GUI_SAVED_PROFILE",
        saved["SignalProfile"] == SIGNAL_PROFILE_ID,
        len(plan) == 5,
        plan[0]["ModuleVersion"] == "0.2.0",
        any(item["Stage"] == "S6" and item["ModuleVersion"] == "0.1.2" for item in plan),
        run_module_selftest(workspace, prepared_module(workspace)).endswith("C-0"),
    ]
    if not all(checks):
        raise InstallError("SELFTEST")
    print("[CHECK] Saved Profile contract is loadable | WHY: GUI Open Profile must not require manual JSON editing")
    print("[CHECK] TC001 signal profile is unchanged | WHY: only S0 compatibility is under repair")
    print("[CHECK] S0 0.2.0 and S6 0.1.2 are explicit | WHY: Registry discovery never silently upgrades a frozen plan")
    print("S0H5ST3-N6-X0-C0")


def install_module(workspace: Path) -> str:
    source = prepared_module(workspace)
    validate_module_files(source)
    target = target_module(workspace)
    disposition = "NEW"
    if target.exists():
        validate_module_files(target)
        disposition = "REPLAY"
    else:
        target.parent.mkdir(parents=True, exist_ok=True)
        writing = target.with_name(target.name + ".writing")
        if writing.exists():
            shutil.rmtree(writing)
        shutil.copytree(source, writing)
        os.replace(writing, target)
    return disposition


def qualify(workspace: Path) -> None:
    qualification = workspace / "01_PROGRAMS/core/u002_u22_module_qualification_v0_1.py"
    result = subprocess.run(
        [
            sys.executable,
            str(qualification),
            "--qualify-module",
            MODULE_ID + "@" + MODULE_VERSION,
            "--workspace-root",
            str(workspace),
            "--timeout-seconds",
            "180",
        ],
        cwd=workspace,
        text=True,
        capture_output=True,
        timeout=240,
        check=False,
    )
    if result.returncode != 0 or "C-0" not in result.stdout:
        detail = result.stdout.strip() or result.stderr.strip() or "NO_OUTPUT"
        raise InstallError("QUALIFICATION_" + token(detail))


def execute(workspace: Path) -> None:
    disposition = install_module(workspace)
    qualify(workspace)
    profile = build_saved_profile(workspace)
    destination = profile_path(workspace)
    destination.parent.mkdir(parents=True, exist_ok=True)
    data = canonical_bytes(profile)
    if destination.exists() and destination.read_bytes() != data:
        raise InstallError("PROFILE_CONFLICT")
    writing = destination.with_name(destination.name + ".writing")
    writing.write_bytes(data)
    os.replace(writing, destination)
    receipt_root = workspace / "07_OUTPUT/U2S0H5_V030_INSTALL"
    receipt_root.mkdir(parents=True, exist_ok=True)
    receipt = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "ModuleId": MODULE_ID,
        "ModuleVersion": MODULE_VERSION,
        "ModuleDisposition": disposition,
        "QualificationStatus": "AVAILABLE",
        "SavedProfile": destination.relative_to(workspace).as_posix(),
        "SignalProfile": profile["SignalProfile"],
        "CandidatePolicy": profile["CandidatePolicy"],
        "S6ModuleVersion": "0.1.2",
        "Status": "SUCCEEDED",
    }
    (receipt_root / "installation_receipt.json").write_bytes(canonical_bytes(receipt))
    print("S0H5I3-NM1-Q1-P1-S6V012-X0-C0")


def main() -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    try:
        workspace = require_workspace(args.workspace_root)
        if args.prepare:
            prepare(workspace)
        elif args.self_test:
            self_test(workspace)
        else:
            execute(workspace)
        return 0
    except Exception as exc:
        print("S0H5E-K" + token(exc) + "-X1-C7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
