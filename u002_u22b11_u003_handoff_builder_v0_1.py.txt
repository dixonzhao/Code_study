#!/usr/bin/env python3
"""Build one narrow U003 review package from the accepted U22B11 replay."""

from __future__ import annotations

import argparse
import csv
import hashlib
import json
import os
from pathlib import Path
import re
import sys
import tempfile
from typing import Dict, Iterable, List, Mapping, Sequence


PROGRAM_ID = "UDP002.U22B11.U003HandoffBuilder"
PROGRAM_VERSION = "0.1.0"
DATA_START = 41630
EVENT_START = 41635
PEAK_FRAME = 41650
EVENT_END = 41660
DATA_END = 41670

REVIEW_COLUMNS = (
    "FrameCount",
    "Camera1Distance",
    "Camera1LateralPosition",
    "Camera3Distance",
    "Camera3LateralPosition",
    "CanonicalCameraDistance",
    "SelectedCameraRole",
    "SelectionStatus",
    "SelectionReason",
    "CenterlinePreferredRole",
)


class HandoffError(RuntimeError):
    """Represent one deterministic handoff failure."""


def machine_token(value: object, limit: int = 160) -> str:
    """Return one bounded diagnostic token."""

    token = re.sub(r"[^A-Z0-9_.-]+", "_", str(value).upper()).strip("_")
    return (token or "NONE")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize one JSON object deterministically."""

    return (
        json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n"
    ).encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_json(path: Path) -> Dict[str, object]:
    """Load one required JSON object."""

    if not path.is_file() or path.is_symlink():
        raise HandoffError("RECEIPT_MISSING")
    value = json.loads(path.read_text(encoding="utf-8-sig"))
    if not isinstance(value, dict):
        raise HandoffError("RECEIPT_TYPE")
    return value


def read_profile(path: Path) -> List[Dict[str, str]]:
    """Read and validate the accepted B7A frame evidence."""

    if not path.is_file() or path.is_symlink():
        raise HandoffError("B7A_ROWS_MISSING")
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        required = {
            "FrameCount",
            "Camera1Distance",
            "Camera1LateralPosition",
            "Camera3Distance",
            "Camera3LateralPosition",
            "OldCanonicalCameraDistance",
            "OldSelectedCameraRole",
            "OldSelectionStatus",
            "OldSelectionReason",
            "CenterlinePreferredRole",
        }
        if not required.issubset(reader.fieldnames or []):
            raise HandoffError("B7A_SCHEMA")
        rows = [dict(row) for row in reader]
    if len(rows) != 401:
        raise HandoffError("B7A_ROW_COUNT")
    return rows


def write_csv(path: Path, columns: Sequence[str], rows: Iterable[Mapping[str, object]]) -> None:
    """Write one deterministic UTF-8 CSV."""

    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".writing")
    with temporary.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(columns), lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in columns})
    os.replace(temporary, path)


def build_handoff(temp_root: Path) -> str:
    """Create one queue row and one compact synchronized review table."""

    if temp_root.name != "04_TEMP":
        raise HandoffError("PROGRAM_NOT_IN_04_TEMP")
    b11_root = temp_root / "U22B11_OUTPUT"
    receipt_path = b11_root / "u22b11_module_v030_controlled_replay_receipt.json"
    receipt = load_json(receipt_path)
    if int(receipt.get("CompletionCode", 7)) != 0:
        raise HandoffError("B11_NOT_ACCEPTED")
    metrics = receipt.get("ReviewedRecoveryMetrics")
    if not isinstance(metrics, dict):
        raise HandoffError("RECOVERY_METRICS")
    expected = {
        "WrongOffCenterCamera1Rows": 0,
        "FirstCamera3CanonicalFrame": PEAK_FRAME,
    }
    if any(int(metrics.get(key, -1)) != value for key, value in expected.items()):
        raise HandoffError("RECOVERY_METRIC_MISMATCH")

    b7a_path = (
        b11_root
        / "B7A_PROFILE_V03"
        / "b7a_camera_lateral_evidence_rows.csv"
    )
    profile_rows = read_profile(b7a_path)
    review_rows: List[Dict[str, object]] = []
    for source in profile_rows:
        frame = int(float(source["FrameCount"]))
        if not DATA_START <= frame <= DATA_END:
            continue
        review_rows.append(
            {
                "FrameCount": frame,
                "Camera1Distance": source["Camera1Distance"],
                "Camera1LateralPosition": source["Camera1LateralPosition"],
                "Camera3Distance": source["Camera3Distance"],
                "Camera3LateralPosition": source["Camera3LateralPosition"],
                "CanonicalCameraDistance": source["OldCanonicalCameraDistance"],
                "SelectedCameraRole": source["OldSelectedCameraRole"],
                "SelectionStatus": source["OldSelectionStatus"],
                "SelectionReason": source["OldSelectionReason"],
                "CenterlinePreferredRole": source["CenterlinePreferredRole"],
            }
        )
    if len(review_rows) != DATA_END - DATA_START + 1:
        raise HandoffError("REVIEW_FRAME_COVERAGE")

    output_root = temp_root / "U22B11_U003_HANDOFF"
    queue_path = output_root / "review_queue.csv"
    data_path = output_root / "review_data.csv"
    manifest_path = output_root / "u22b11_u003_handoff_manifest.json"
    queue_columns = (
        "EventID",
        "StartFrameCount",
        "PeakFrameCount",
        "EndFrameCount",
        "ReviewResult",
        "ReviewMemo",
        "SuggestedReviewMemo",
    )
    write_csv(
        queue_path,
        queue_columns,
        [
            {
                "EventID": "B11V01",
                "StartFrameCount": EVENT_START,
                "PeakFrameCount": PEAK_FRAME,
                "EndFrameCount": EVENT_END,
                "ReviewResult": "",
                "ReviewMemo": "",
                "SuggestedReviewMemo": (
                    "Confirm that off-center Camera 1 is not selected and "
                    "Camera 3 becomes canonical after three valid centered frames."
                ),
            }
        ],
    )
    write_csv(data_path, REVIEW_COLUMNS, review_rows)
    manifest = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "CompletionCode": 0,
        "SourceB11Receipt": str(receipt_path),
        "SourceB11ReceiptSha256": sha256_file(receipt_path),
        "SourceB7AEvidence": str(b7a_path),
        "SourceB7AEvidenceSha256": sha256_file(b7a_path),
        "ReviewWindow": {
            "DataStart": DATA_START,
            "EventStart": EVENT_START,
            "Peak": PEAK_FRAME,
            "EventEnd": EVENT_END,
            "DataEnd": DATA_END,
        },
        "ReviewRows": len(review_rows),
        "ReviewColumns": list(REVIEW_COLUMNS),
        "Outputs": {
            queue_path.name: sha256_file(queue_path),
            data_path.name: sha256_file(data_path),
        },
        "DecisionBoundary": "HUMAN_VIDEO_CONFIRMATION_REQUIRED",
    }
    manifest_path.write_bytes(canonical_bytes(manifest))
    return (
        f"B11H N-Q1-F{len(review_rows)}-K{len(REVIEW_COLUMNS)}"
        f"-S{EVENT_START}-P{PEAK_FRAME}-E{EVENT_END}-X0 C-0"
    )


def self_test() -> None:
    """Exercise the complete handoff builder with a synthetic accepted replay."""

    with tempfile.TemporaryDirectory(prefix="u22b11_handoff_") as temporary:
        temp_root = Path(temporary) / "04_TEMP"
        b11_root = temp_root / "U22B11_OUTPUT"
        profile_root = b11_root / "B7A_PROFILE_V03"
        profile_root.mkdir(parents=True)
        receipt = {
            "CompletionCode": 0,
            "ReviewedRecoveryMetrics": {
                "WrongOffCenterCamera1Rows": 0,
                "FirstCamera3CanonicalFrame": PEAK_FRAME,
            },
        }
        (b11_root / "u22b11_module_v030_controlled_replay_receipt.json").write_bytes(
            canonical_bytes(receipt)
        )
        source_columns = (
            "FrameCount",
            "Camera1Distance",
            "Camera1LateralPosition",
            "Camera3Distance",
            "Camera3LateralPosition",
            "OldCanonicalCameraDistance",
            "OldSelectedCameraRole",
            "OldSelectionStatus",
            "OldSelectionReason",
            "CenterlinePreferredRole",
        )
        source_rows = []
        for frame in range(41380, 41781):
            selected = "CAMERA_3_DISTANCE" if frame >= PEAK_FRAME else ""
            source_rows.append(
                {
                    "FrameCount": frame,
                    "Camera1Distance": 40.0,
                    "Camera1LateralPosition": -1.95,
                    "Camera3Distance": 16.5,
                    "Camera3LateralPosition": 0.15,
                    "OldCanonicalCameraDistance": 16.5 if selected else "",
                    "OldSelectedCameraRole": selected,
                    "OldSelectionStatus": "CONFIDENT" if selected else "AMBIGUOUS",
                    "OldSelectionReason": "GEOMETRY_ACQUIRED" if selected else "NONE",
                    "CenterlinePreferredRole": "CAMERA_3_DISTANCE",
                }
            )
        write_csv(
            profile_root / "b7a_camera_lateral_evidence_rows.csv",
            source_columns,
            source_rows,
        )
        capsule = build_handoff(temp_root)
        assert capsule == "B11H N-Q1-F41-K10-S41635-P41650-E41660-X0 C-0"
        output_root = temp_root / "U22B11_U003_HANDOFF"
        assert (output_root / "review_queue.csv").is_file()
        assert (output_root / "review_data.csv").is_file()
        assert (output_root / "u22b11_u003_handoff_manifest.json").is_file()
    print("UDP002-U22B11-U003-HANDOFF-SELFTEST-PASS-12 C-0")


def main(argv: Sequence[str] | None = None) -> int:
    """Run self-test or build the company U003 handoff."""

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            self_test()
            return 0
        capsule = build_handoff(Path(__file__).resolve().parent)
        print(capsule)
        return 0
    except Exception as error:
        print(f"B11HE K-{machine_token(type(error).__name__, 72)} X1 C-7")
        print(f"B11HD D-{machine_token(str(error))} X0 C-0")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
