#!/usr/bin/env python3
"""Locate and independently validate the completed G10D2C release stage."""

from __future__ import annotations

import argparse
import subprocess
import sys
import tempfile
import time
from pathlib import Path


PROGRAM_NAME = "u001_g10d2cd2_staged_release_validator_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2CD2-SELFTEST-PASS-16"
GATE_NAME = "u001_g10d2c_portable_release_acceptance_gate_v0_1.py"
PASS_TOKEN = "G10D2CCG1-PASS P12-C4-H12-Z1-R1-X0 C-0"


def latest_pk04(temp_root: Path) -> Path | None:
    if not temp_root.is_dir():
        return None
    candidates = [
        path for path in temp_root.iterdir()
        if path.is_dir() and path.name.upper().startswith("PK04")
    ]
    return max(candidates, key=lambda path: path.stat().st_mtime, default=None)


def release_stage(build_root: Path | None) -> Path | None:
    if build_root is None or not build_root.is_dir():
        return None
    direct = build_root / "release_stage"
    if (direct / "U001_Portable_v0_2").is_dir() and (direct / "U001_Portable_v0_2.zip").is_file():
        return direct
    candidates = []
    try:
        for path in build_root.rglob("*"):
            if not path.is_dir():
                continue
            if (path / "U001_Portable_v0_2").is_dir() and (path / "U001_Portable_v0_2.zip").is_file():
                candidates.append(path)
    except OSError:
        return None
    return min(candidates, key=lambda path: len(path.parts), default=None)


def compact_reason(stdout: str, stderr: str, returncode: int) -> str:
    for source in (stdout, stderr):
        for line in source.splitlines():
            text = line.strip()
            if text.startswith("G10D2CCG1-FAIL:"):
                reason = text.split("G10D2CCG1-FAIL:", 1)[1].split(" C-7", 1)[0]
                return "".join(ch if ch.isalnum() or ch in "_-" else "_" for ch in reason)[:80]
    return f"GATE_RC{returncode}"


def atomic_text(path: Path, text: str) -> None:
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text(text, encoding="utf-8")
    temporary.replace(path)


def validate(udp001_root: Path) -> int:
    root = udp001_root.resolve()
    temp_root = root / "04_TEMP"
    if not root.is_dir() or root.name.upper() != "UDP001" or not temp_root.is_dir():
        print("G10D2CD2 N-R0-S0-G0-E7-X1 K-UDP001_ROOT C-7")
        return 7
    build = latest_pk04(temp_root)
    stage = release_stage(build)
    gate = temp_root / GATE_NAME
    if build is None:
        print("G10D2CD2 N-R0-S0-G0-E7-X0 K-PK04_MISSING C-7")
        return 7
    if stage is None:
        print("G10D2CD2 N-R1-S0-G0-E7-X0 K-RELEASE_STAGE_MISSING C-7")
        return 7
    if not gate.is_file():
        print("G10D2CD2 N-R1-S1-G0-E7-X0 K-GATE_MISSING C-7")
        return 7
    try:
        result = subprocess.run(
            [sys.executable, str(gate.resolve()), "--release-root", str(stage.resolve())],
            cwd=str(temp_root),
            stdout=subprocess.PIPE,
            stderr=subprocess.PIPE,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=2100,
            shell=False,
        )
        atomic_text(temp_root / "G10D2CD2_GATE_STDOUT.txt", result.stdout)
        atomic_text(temp_root / "G10D2CD2_GATE_STDERR.txt", result.stderr)
    except subprocess.TimeoutExpired:
        print("G10D2CD2 N-R1-S1-G0-E124-X0 K-GATE_TIMEOUT C-7")
        return 7
    except Exception as exc:
        print(f"G10D2CD2 N-R1-S1-G0-E7-X1 K-{type(exc).__name__} C-7")
        return 7
    if result.returncode == 0 and PASS_TOKEN in result.stdout:
        print("G10D2CD2 N-R1-S1-G1-E0-X0 C-0")
        return 0
    reason = compact_reason(result.stdout, result.stderr, result.returncode)
    print(f"G10D2CD2 N-R1-S1-G0-E{result.returncode}-X0 K-{reason} C-7")
    return 7


def run_self_test() -> None:
    checks = 0
    with tempfile.TemporaryDirectory() as td:
        base = Path(td)
        root = base / "UDP001"
        temp_root = root / "04_TEMP"
        temp_root.mkdir(parents=True)
        checks += int(latest_pk04(temp_root) is None)
        old = temp_root / "PK04_OLD"
        new = temp_root / "PK04_NEW"
        old.mkdir()
        time.sleep(0.01)
        new.mkdir()
        checks += int(latest_pk04(temp_root) == new)
        checks += int(release_stage(new) is None)
        stage = new / "release_stage"
        (stage / "U001_Portable_v0_2").mkdir(parents=True)
        (stage / "U001_Portable_v0_2.zip").write_bytes(b"ZIP")
        checks += int(release_stage(new) == stage)
        checks += int(compact_reason("G10D2CCG1-FAIL:ZIP_MISSING C-7", "", 7) == "ZIP_MISSING")
        checks += int(compact_reason("", "", 9) == "GATE_RC9")
        checks += int(PROGRAM_VERSION == "0.1")
        checks += int(PROGRAM_NAME.endswith("v0_1.py"))
        checks += int(SELFTEST_TOKEN.endswith("PASS-16"))
        checks += int(GATE_NAME.endswith("v0_1.py"))
        checks += int(PASS_TOKEN.endswith("C-0"))
        checks += int(root.name == "UDP001")
        checks += int(temp_root.name == "04_TEMP")
        checks += int(stage.name == "release_stage")
        checks += int((stage / "U001_Portable_v0_2").is_dir())
        checks += int((stage / "U001_Portable_v0_2.zip").is_file())
    if checks != 16:
        raise AssertionError(f"SELFTEST_COUNT:{checks}")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Validate the completed staged G10D2C release")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--udp001-root")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.udp001_root:
        parser.error("--udp001-root is required")
    return validate(Path(args.udp001_root))


if __name__ == "__main__":
    raise SystemExit(main())
