#!/usr/bin/env python3
"""Read-only inspection of unexpected portable 03_CASES artifacts."""

from __future__ import annotations

import argparse
import hashlib
import os
import tempfile
import time
from collections import Counter
from pathlib import Path


PROGRAM_NAME = "u001_g10d2cd4_cases_mutation_inspector_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "UDP001-G10D2CD4-SELFTEST-PASS-16"
PACKAGE_NAME = "U001_Portable_v0_2"
REPORT_NAME = "G10D2CD4_CASES_MUTATION_REPORT.txt"
SYNTHETIC_MARKERS = (
    "selftest", "self_test", "self-test", "synthetic", "smoke", "probe",
    "fixture", "pytest", "test_case", "testcase", "dummy", "temporary",
)


def latest_pk04(temp_root: Path) -> Path | None:
    candidates = []
    if temp_root.is_dir():
        candidates = [p for p in temp_root.iterdir() if p.is_dir() and p.name.upper().startswith("PK04")]
    return max(candidates, key=lambda path: path.stat().st_mtime, default=None)


def release_stage(build_root: Path | None) -> Path | None:
    if build_root is None:
        return None
    direct = build_root / "release_stage"
    if (direct / PACKAGE_NAME).is_dir():
        return direct
    try:
        found = [p for p in build_root.rglob(PACKAGE_NAME) if p.is_dir()]
    except OSError:
        return None
    return found[0].parent if found else None


def cases_root(release_root: Path) -> Path:
    return release_root / PACKAGE_NAME / "UDP001" / "03_CASES"


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def records(root: Path) -> tuple[list[dict], int]:
    result = []
    directories = 0
    if not root.is_dir():
        return result, directories
    for path in sorted(root.rglob("*"), key=lambda item: item.relative_to(root).as_posix()):
        if path.is_dir():
            directories += 1
            continue
        if not path.is_file():
            continue
        relative = path.relative_to(root).as_posix()
        size = path.stat().st_size
        digest = sha256_file(path)
        sample = ""
        try:
            if size <= 262144:
                sample = path.read_bytes()[:8192].decode("utf-8", errors="replace").lower()
        except OSError:
            pass
        searchable = (relative + " " + sample).lower()
        synthetic = int(any(marker in searchable for marker in SYNTHETIC_MARKERS))
        result.append({
            "relative": relative,
            "bytes": size,
            "sha256": digest,
            "synthetic": synthetic,
            "mtime": path.stat().st_mtime,
        })
    return result, directories


def signatures(items: list[dict]) -> list[tuple[str, int, str]]:
    return [(item["relative"], item["bytes"], item["sha256"]) for item in items]


def safe_token(text: str, limit: int = 50) -> str:
    value = "".join(ch if ch.isalnum() or ch in "._-" else "_" for ch in text)
    return value[:limit] or "EMPTY"


def write_report(path: Path, source_items: list[dict], output_items: list[dict], source_dirs: int, output_dirs: int) -> None:
    lines = [
        "G10D2C-D4 03_CASES MUTATION REPORT",
        f"SourceFiles={len(source_items)}",
        f"SourceDirectories={source_dirs}",
        f"OutputFiles={len(output_items)}",
        f"OutputDirectories={output_dirs}",
        f"TreesEqual={int(signatures(source_items) == signatures(output_items))}",
        "",
        "SOURCE FILES",
    ]
    for item in source_items:
        lines.append(
            f"{item['relative']}|Bytes={item['bytes']}|SHA256={item['sha256']}|SyntheticMarker={item['synthetic']}|MTime={item['mtime']}"
        )
    lines.extend(("", "OUTPUT FILES"))
    for item in output_items:
        lines.append(
            f"{item['relative']}|Bytes={item['bytes']}|SHA256={item['sha256']}|SyntheticMarker={item['synthetic']}|MTime={item['mtime']}"
        )
    temporary = path.with_suffix(path.suffix + ".tmp")
    temporary.write_text("\n".join(lines) + "\n", encoding="utf-8")
    temporary.replace(path)


def inspect(udp001_root: Path, out: Path) -> int:
    root = udp001_root.resolve()
    temp_root = root / "04_TEMP"
    out = out.resolve()
    build = latest_pk04(temp_root)
    stage = release_stage(build)
    if stage is None or not out.is_dir():
        print("G10D2CD4 N-SF0-SD0-OF0-OD0-EQ0-A0-B0-K0-X1 C-7")
        return 7
    source_cases = cases_root(stage)
    output_cases = cases_root(out)
    source_items, source_dirs = records(source_cases)
    output_items, output_dirs = records(output_cases)
    equal = int(signatures(source_items) == signatures(output_items))
    synthetic = sum(item["synthetic"] for item in source_items)
    total_bytes = sum(item["bytes"] for item in source_items)
    extensions = Counter(Path(item["relative"]).suffix.lower() or "none" for item in source_items)
    try:
        write_report(temp_root / REPORT_NAME, source_items, output_items, source_dirs, output_dirs)
        report_ok = 1
    except OSError:
        report_ok = 0
    print(
        f"G10D2CD4 N-SF{len(source_items)}-SD{source_dirs}-OF{len(output_items)}-OD{output_dirs}"
        f"-EQ{equal}-A{synthetic}-B{total_bytes}-K{len(extensions)}-P{report_ok}-X0 C-0"
    )
    for index, item in enumerate(source_items[:3], 1):
        print(
            f"M{index}-{safe_token(item['relative'])}-B{item['bytes']}-H{item['sha256'][:8]}-A{item['synthetic']}"
        )
    return 0


def run_self_test() -> None:
    checks = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td) / "cases"
        (root / "SELFTEST_CASE").mkdir(parents=True)
        (root / "SELFTEST_CASE" / "result.json").write_text('{"synthetic":"selftest"}', encoding="utf-8")
        items, dirs = records(root)
        checks += int(len(items) == 1)
        checks += int(dirs == 1)
        checks += int(items[0]["synthetic"] == 1)
        checks += int(items[0]["bytes"] > 0)
        checks += int(len(items[0]["sha256"]) == 64)
        checks += int(signatures(items) == signatures(items))
        checks += int(safe_token("a/b c") == "a_b_c")
        checks += int(PROGRAM_VERSION == "0.1")
        checks += int(PROGRAM_NAME.endswith("v0_1.py"))
        checks += int(SELFTEST_TOKEN.endswith("PASS-16"))
        checks += int(PACKAGE_NAME.endswith("v0_2"))
        checks += int(REPORT_NAME.endswith(".txt"))
        checks += int("selftest" in SYNTHETIC_MARKERS)
        checks += int(latest_pk04(Path(td)) is None)
        stage = Path(td) / "release_stage"
        (stage / PACKAGE_NAME).mkdir(parents=True)
        checks += int(cases_root(stage).name == "03_CASES")
        checks += int(release_stage(None) is None)
    if checks != 16:
        raise AssertionError(f"SELFTEST_COUNT:{checks}")
    print(SELFTEST_TOKEN)


def main(argv=None) -> int:
    parser = argparse.ArgumentParser(description="Inspect unexpected portable 03_CASES artifacts")
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--udp001-root")
    parser.add_argument("--out")
    args = parser.parse_args(argv)
    if args.self_test:
        run_self_test()
        return 0
    if not args.udp001_root or not args.out:
        parser.error("--udp001-root and --out are required")
    try:
        return inspect(Path(args.udp001_root), Path(args.out))
    except Exception as exc:
        print(f"G10D2CD4 N-SF0-SD0-OF0-OD0-EQ0-A0-B0-K0-X1 T-{type(exc).__name__} C-7")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
