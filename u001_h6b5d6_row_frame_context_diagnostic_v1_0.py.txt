#!/usr/bin/env python3
"""Decode the carried FrameCount bytes on every raw source row.

This read-only diagnostic determines whether the MATLAB CAN wide table carries
a usable per-row FrameCount context even when 0x620 is not a new message event.
"""

from __future__ import annotations

import argparse
import csv
import json
import math
import re
import shutil
import sys
from datetime import datetime, timezone
from pathlib import Path
from typing import Any, Dict, List, Optional, Sequence, Tuple


PROGRAM_ID = "U001_H6B5D6_ROW_FRAME_CONTEXT_DIAGNOSTIC"
PROGRAM_VERSION = "1.0"
PREPARED_FOLDER = "U1H6B5D6_PREPARED"
OUTPUT_FOLDER = "U1H6B5D6_OUTPUT"
SUPERSEDED_TRANSFER_NAMES = (
    "u001_h6b5d3_frame_key_path_diagnostic_v1_0.py.txt",
    "U1H6B5D3_START_HERE_CN_v1_0_2026-08-28.txt",
)


class RowFrameError(RuntimeError):
    """Fail-closed diagnostic error."""


def utc_now() -> str:
    return datetime.now(timezone.utc).isoformat(timespec="seconds")


def write_json(path: Path, value: Dict[str, Any]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def read_json(path: Path, label: str) -> Dict[str, Any]:
    if not path.is_file():
        raise RowFrameError(label + "_NOT_FOUND")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise RowFrameError(label + "_INVALID_JSON") from exc
    if not isinstance(value, dict):
        raise RowFrameError(label + "_ROOT_NOT_OBJECT")
    return value


def resolve_root(argument: Optional[str]) -> Path:
    if argument:
        root = Path(argument).expanduser().resolve()
    else:
        cwd = Path.cwd().resolve()
        root = cwd if cwd.name.casefold() == "u-udp001" else cwd / "U-UDP001"
    if not root.is_dir():
        raise RowFrameError("U_UDP001_ROOT_NOT_FOUND")
    return root


def required_paths(root: Path) -> Dict[str, Path]:
    b3 = root / "04_TEMP" / "U1H6B3_OUTPUT" / "compiled_profile"
    h2 = root / "04_TEMP" / "U1H6B4_OUTPUT" / "h2_real"
    d3 = root / "04_TEMP" / "U1H6B5D3_OUTPUT"
    return {
        "ControlledRequest": b3 / "u001_h2_controlled_request.json",
        "H2Catalog": h2 / "u001_adapter_signal_catalog.csv",
        "H2Quality": h2 / "u001_adapter_quality.json",
        "D3Diagnostic": d3 / "u001_h6b5d3_frame_key_path_diagnostic.json",
        "D3Capsules": d3 / "u001_h6b5d3_capsules.txt",
    }


def raw_source(request: Dict[str, Any]) -> Path:
    values: List[Path] = []
    for item in request.get("Inputs", []):
        if isinstance(item, dict) and item.get("Kind") == "MATLAB_CAN_CSV":
            raw = item.get("Path", item.get("FilePath"))
            if isinstance(raw, str) and raw.strip():
                values.append(Path(raw))
    if len(values) != 1 or not values[0].is_file():
        raise RowFrameError("RAW_SOURCE")
    return values[0]


def frame_identity(request: Dict[str, Any]) -> Tuple[int, str, str]:
    config = request.get("Configuration")
    key = config.get("FrameKey") if isinstance(config, dict) else None
    if not isinstance(key, dict):
        raise RowFrameError("FRAME_KEY")
    return int(key["Channel"]), str(key["CAN_ID"]).strip().lower(), str(key["SignalName"]).strip()


def frame_catalog(path: Path, identity: Tuple[int, str, str]) -> Dict[str, str]:
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.DictReader(stream)
        rows = [
            row for row in reader
            if int(row["SourceChannel"]) == identity[0]
            and row["SourceMessageId"].strip().lower() == identity[1]
            and row["SourceSignalName"].strip() == identity[2]
        ]
    if len(rows) != 1:
        raise RowFrameError("FRAME_CATALOG_COUNT")
    return rows[0]


def detect_header(path: Path, required: Sequence[str]) -> Tuple[str, int, List[str]]:
    with path.open("r", encoding="utf-8-sig", errors="replace", newline="") as stream:
        lines = [stream.readline() for _ in range(50)]
    required_set = set(required)
    for row_index, line in enumerate(line for line in lines if line):
        for delimiter in (",", "\t", ";"):
            try:
                columns = [item.strip() for item in next(csv.reader([line], delimiter=delimiter))]
            except csv.Error:
                continue
            if required_set.issubset(columns):
                return delimiter, row_index, columns
    raise RowFrameError("RAW_HEADER")


def parse_byte(value: str) -> int:
    text = str(value).strip()
    if not text:
        raise RowFrameError("BYTE_BLANK")
    try:
        number = int(text, 16) if text.casefold().startswith("0x") else int(float(text))
    except ValueError as exc:
        raise RowFrameError("BYTE_FORMAT") from exc
    if not 0 <= number <= 255:
        raise RowFrameError("BYTE_RANGE")
    return number


def raw_unsigned(payload: Sequence[int], start_bit: int, bit_length: int, byte_order: str) -> int:
    if byte_order == "little_endian":
        packed = int.from_bytes(bytes(payload), byteorder="little", signed=False)
        return (packed >> start_bit) & ((1 << bit_length) - 1)
    if byte_order != "big_endian":
        raise RowFrameError("BYTE_ORDER")
    position = start_bit
    result = 0
    for _ in range(bit_length):
        byte_index = position // 8
        bit_index = position % 8
        if byte_index < 0 or byte_index >= len(payload):
            raise RowFrameError("BIT_OUT_OF_PAYLOAD")
        result = (result << 1) | ((payload[byte_index] >> bit_index) & 1)
        position = position + 15 if bit_index == 0 else position - 1
    return result


def signed_value(value: int, bit_length: int, is_signed: bool) -> int:
    if is_signed and value & (1 << (bit_length - 1)):
        return value - (1 << bit_length)
    return value


def decode(payload: Sequence[int], catalog: Dict[str, str]) -> float:
    start_bit = int(catalog["StartBit"])
    bit_length = int(catalog["BitLength"])
    unsigned = raw_unsigned(payload, start_bit, bit_length, catalog["ByteOrder"].strip())
    raw = signed_value(unsigned, bit_length, catalog["Signed"].strip() in {"1", "true", "True"})
    result = raw * float(catalog["Scale"]) + float(catalog["Offset"])
    if not math.isfinite(result):
        raise RowFrameError("FRAME_NONFINITE")
    return result


def quantile(values: List[int], fraction: float) -> int:
    if not values:
        return 0
    ordered = sorted(values)
    return ordered[int(round((len(ordered) - 1) * fraction))]


def scan(source: Path, catalog: Dict[str, str]) -> Dict[str, Any]:
    byte_columns = [value for value in catalog["SourceByteColumns"].split("|") if value]
    if len(byte_columns) != int(catalog["MessageDLC"]):
        raise RowFrameError("BYTE_COLUMN_COUNT")
    delimiter, header_row, columns = detect_header(source, byte_columns)
    positions = {name: index for index, name in enumerate(columns)}
    byte_positions = [positions[name] for name in byte_columns]
    complete_rows = 0
    invalid_rows = 0
    unique_values: set[int] = set()
    values: List[int] = []
    payload_change_count = 0
    previous_payload: Optional[Tuple[int, ...]] = None

    with source.open("r", encoding="utf-8-sig", errors="replace", newline="") as stream:
        for _ in range(header_row):
            next(stream, None)
        reader = csv.reader(stream, delimiter=delimiter)
        next(reader, None)
        for row_number, cells in enumerate(reader, start=1):
            if len(cells) < len(columns):
                cells.extend([""] * (len(columns) - len(cells)))
            selected = [cells[index] for index in byte_positions]
            if not all(str(value).strip() for value in selected):
                previous_payload = None
                continue
            complete_rows += 1
            try:
                payload = tuple(parse_byte(value) for value in selected)
                physical = decode(payload, catalog)
                if not float(physical).is_integer():
                    raise RowFrameError("FRAME_FRACTIONAL")
                frame = int(physical)
            except RowFrameError:
                invalid_rows += 1
                previous_payload = None
                continue
            if previous_payload is None or payload != previous_payload:
                payload_change_count += 1
            previous_payload = payload
            values.append(frame)
            unique_values.add(frame)
            if row_number % 50000 == 0:
                print(f"[PROGRESS] raw_rows={row_number}", flush=True)

    if not values:
        raise RowFrameError("NO_DECODED_FRAME_CONTEXT")
    decreases = sum(later < earlier for earlier, later in zip(values, values[1:]))
    positive = [later - earlier for earlier, later in zip(values, values[1:]) if later > earlier]
    return {
        "CompleteRows": complete_rows,
        "InvalidRows": invalid_rows,
        "DecodedRows": len(values),
        "UniqueFrameValues": len(unique_values),
        "FrameMinimum": min(values),
        "FrameMaximum": max(values),
        "PayloadChangeCount": payload_change_count,
        "FrameDecreaseCount": decreases,
        "PositiveDeltaQ50": quantile(positive, 0.50),
        "PositiveDeltaQ95": quantile(positive, 0.95),
        "PositiveDeltaMaximum": max(positive) if positive else 0,
    }


def prepare(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        temp = root / "04_TEMP"
        prepared = temp / PREPARED_FOLDER
        output = temp / OUTPUT_FOLDER
        archive = root / "A-Archive" / ("U1H6B5D6_HYGIENE_" + datetime.now().strftime("%Y%m%dT%H%M%S_%f"))
        print("[PREPARE] U001 H6B5D6 carried FrameCount diagnostic")
        print("[PREPARE] Why: verify whether raw 0x620 bytes provide a usable per-row video-frame context.")
        if prepared.exists():
            shutil.rmtree(prepared)
            print(f"[CLEAN] 04_TEMP/{PREPARED_FOLDER} | REMOVED | Why: invalidate stale receipt.")
        if output.exists():
            archive.mkdir(parents=True, exist_ok=True)
            shutil.move(str(output), str(archive / output.name))
            print(f"[CLEAN] 04_TEMP/{OUTPUT_FOLDER} | ARCHIVED | Why: prevent stale evidence reuse.")
        for name in SUPERSEDED_TRANSFER_NAMES:
            path = temp / name
            if path.exists():
                archive.mkdir(parents=True, exist_ok=True)
                shutil.move(str(path), str(archive / name))
                print(f"[CLEAN] 04_TEMP/{name} | ARCHIVED | Why: superseded transfer file.")
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        source = raw_source(request)
        missing = [label for label, path in paths.items() if not path.is_file()]
        for label, path in paths.items():
            print(f"[CHECK] {label} | {'FOUND' if path.is_file() else 'MISSING'} | Why: required accepted evidence.")
        print(f"[CHECK] RawSource | FOUND | basename={source.name} | Why: decode carried FrameCount bytes directly.")
        print("[KEEP] H6B3/H6B4/H6B5/H6B5D1/H6B5D3 outputs | Why: active traceability chain.")
        if missing:
            raise RowFrameError("PREPARE_MISSING:" + ".".join(missing))
        prepared.mkdir(parents=True)
        write_json(prepared / "u001_h6b5d6_prepare_receipt.json", {
            "ProgramId": PROGRAM_ID,
            "ProgramVersion": PROGRAM_VERSION,
            "RawSource": str(source),
            "PreparedUtc": utc_now(),
            "Status": "READY",
            "CompletionCode": 0,
        })
        print("U1H6B5D6P-C6-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1H6B5D6P-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def require_prepare(root: Path) -> Dict[str, Any]:
    value = read_json(root / "04_TEMP" / PREPARED_FOLDER / "u001_h6b5d6_prepare_receipt.json", "PREPARE_RECEIPT")
    if value.get("ProgramId") != PROGRAM_ID or value.get("ProgramVersion") != PROGRAM_VERSION:
        raise RowFrameError("PREPARE_IDENTITY")
    if value.get("Status") != "READY" or value.get("CompletionCode") != 0:
        raise RowFrameError("PREPARE_STATUS")
    return value


def self_test(root_argument: Optional[str]) -> int:
    try:
        root = resolve_root(root_argument)
        require_prepare(root)
        print("[SELFTEST] DBC bit extraction | Why: protect direct FrameCount decoding from byte-order errors.")
        if raw_unsigned([0x34, 0x12], 0, 16, "little_endian") != 0x1234:
            raise AssertionError("LITTLE_ENDIAN")
        if raw_unsigned([0xAB, 0xCD], 7, 16, "big_endian") != 0xABCD:
            raise AssertionError("BIG_ENDIAN")
        print("[PASS] Little-endian and Motorola bit extraction verified.")
        print("[SELFTEST] Real FrameCount catalog identity | Why: use the exact accepted CH/CAN/Signal definition and both-DBC-derived catalog.")
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        catalog = frame_catalog(paths["H2Catalog"], frame_identity(request))
        if len([value for value in catalog["SourceByteColumns"].split("|") if value]) != 8:
            raise AssertionError("FRAME_BYTE_COLUMNS")
        print("[PASS] Exact FrameCount catalog and eight-byte source group verified.")
        print("U1H6B5D6-ST-T2-X0-C0")
        return 0
    except Exception as exc:
        print(f"U1H6B5D6-ST-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def execute(root_argument: Optional[str]) -> int:
    output: Optional[Path] = None
    try:
        root = resolve_root(root_argument)
        receipt = require_prepare(root)
        paths = required_paths(root)
        request = read_json(paths["ControlledRequest"], "CONTROLLED_REQUEST")
        source = raw_source(request)
        if str(source) != str(receipt.get("RawSource")):
            raise RowFrameError("PREPARED_SOURCE_CHANGED")
        catalog = frame_catalog(paths["H2Catalog"], frame_identity(request))
        quality = read_json(paths["H2Quality"], "H2_QUALITY")
        h2_count = int(quality.get("FrameKeyUpdateCount", -1))
        if h2_count < 0:
            raise RowFrameError("H2_FRAME_COUNT")
        output = root / "04_TEMP" / OUTPUT_FOLDER
        if output.exists():
            raise RowFrameError("OUTPUT_EXISTS_RUN_PREPARE")
        output.mkdir(parents=True)
        result = scan(source, catalog)
        result["H2FrameUpdateCount"] = h2_count
        result["PayloadChangesEqualH2"] = result["PayloadChangeCount"] == h2_count
        write_json(output / "u001_h6b5d6_row_frame_context_diagnostic.json", {
            "DiagnosticSchema": "U001_H6B5D6_ROW_FRAME_CONTEXT_DIAGNOSTIC_V1_0",
            "GeneratedUtc": utc_now(),
            "Result": result,
            "Status": "DIAGNOSTIC_COMPLETE",
            "CompletionCode": 0,
        })
        capsule = (
            f"H6B5D6-R{result['DecodedRows']}-U{result['UniqueFrameValues']}"
            f"-V{result['FrameMinimum']}.{result['FrameMaximum']}"
            f"-P{result['PayloadChangeCount']}-H{h2_count}"
            f"-E{int(result['PayloadChangesEqualH2'])}-D{result['FrameDecreaseCount']}"
            f"-Q{result['PositiveDeltaQ50']}.{result['PositiveDeltaQ95']}.{result['PositiveDeltaMaximum']}"
            f"-X0-C0"
        )
        (output / "u001_h6b5d6_capsule.txt").write_text(capsule + "\n", encoding="utf-8")
        print(capsule)
        return 0
    except Exception as exc:
        if output is None:
            output = Path.cwd() / OUTPUT_FOLDER
        output.mkdir(parents=True, exist_ok=True)
        write_json(output / "u001_h6b5d6_error.json", {"Type": type(exc).__name__, "Detail": str(exc)[:3000]})
        print(f"H6B5D6-X1-C7-K{type(exc).__name__.upper()}-D{str(exc)}")
        return 7


def parser() -> argparse.ArgumentParser:
    value = argparse.ArgumentParser(description="U001 H6B5D6 carried FrameCount diagnostic")
    action = value.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    value.add_argument("--root")
    return value


def main(argv: Optional[Sequence[str]] = None) -> int:
    args = parser().parse_args(argv)
    if args.prepare:
        return prepare(args.root)
    if args.self_test:
        return self_test(args.root)
    return execute(args.root)


if __name__ == "__main__":
    sys.exit(main())
