#!/usr/bin/env python3
"""Prepare, test and promote the U3M9B editable XLSX export core."""

from __future__ import annotations

import argparse
import csv
import hashlib
import shutil
import subprocess
import sys
import zipfile
from datetime import datetime
from pathlib import Path


PACKAGE_NAME = "UDP003_U3M9B_EDITABLE_XLSX_EXPORT_CORE_v0_1.zip"
PREPARED_NAME = "U3M9B_PREPARED"
CORE_NAME = "u003_editable_xlsx_export_core_v0_1.py"
GATE_NAME = "u003_u3m9b_export_core_acceptance_gate_v0_1.py"
PROFILE_NAME = "U3M9A_sample_chart_profile_v0_1.json"
CONTRACT_NAME = "u003_editable_xlsx_chart_profile_contract_v0_1.json"
REPORT_NAME = "U3M9B_node_report_editable_xlsx_export_core_v0_1_2026-08-27.txt"


class InstallError(RuntimeError):
    """Raised when a transfer package is incomplete or fails qualification."""


def require(condition: bool, message: str) -> None:
    if not condition:
        raise InstallError(message)


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def discover_udp003_root(explicit: Path | None) -> Path:
    candidates = []
    if explicit is not None:
        candidates.append(explicit)
    current = Path.cwd().resolve()
    candidates.extend([current, current / "U-UDP003"])
    for parent in [current, *current.parents[:3]]:
        candidates.append(parent / "U-UDP003")
    for candidate in candidates:
        if candidate.name == "U-UDP003" and candidate.is_dir():
            return candidate.resolve()
    raise InstallError("U-UDP003 root was not found. Run this command from C-CODE.")


def archive_existing(path: Path, archive_root: Path, label: str) -> None:
    if not path.exists():
        return
    timestamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    destination = archive_root / f"{label}_{timestamp}" / path.name
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(destination))


def verify_manifest(package_root: Path) -> int:
    manifest_path = package_root / "PACKAGE_SHA256.csv"
    require(manifest_path.is_file(), "PACKAGE_SHA256.csv is missing.")
    with manifest_path.open("r", encoding="utf-8", newline="") as stream:
        rows = list(csv.DictReader(stream))
    require(bool(rows), "PACKAGE_SHA256.csv is empty.")
    expected_paths = set()
    for row in rows:
        relative = row.get("RelativePath", "")
        expected_hash = row.get("SHA256", "")
        require(relative and expected_hash, "PACKAGE_SHA256.csv contains an incomplete row.")
        artifact = package_root / relative
        require(artifact.is_file(), f"Package artifact is missing: {relative}")
        require(sha256_file(artifact) == expected_hash, f"Package hash differs: {relative}")
        expected_paths.add(relative.replace("\\", "/"))
    actual_paths = {
        path.relative_to(package_root).as_posix()
        for path in package_root.rglob("*")
        if path.is_file() and path.name != "PACKAGE_SHA256.csv"
    }
    require(actual_paths == expected_paths, "Package inventory differs from PACKAGE_SHA256.csv.")
    return len(rows)


def prepare(root: Path, installer_path: Path) -> int:
    temporary_root = root / "04_TEMP"
    archive_root = root / "A-Archive"
    prepared = temporary_root / PREPARED_NAME
    package_path = installer_path.parent / PACKAGE_NAME

    print(f"[CHECK] U003 root: {root}")
    print("[WHY] The package must use the same company directory layout as the accepted U003 installation.")
    print(f"[CHECK] Transfer ZIP: {package_path}")
    print("[WHY] Exact package identity prevents a stale program and a new guide from being mixed.")
    require(package_path.is_file(), f"Transfer ZIP is missing beside the installer: {PACKAGE_NAME}")
    temporary_root.mkdir(parents=True, exist_ok=True)
    archive_root.mkdir(parents=True, exist_ok=True)
    archive_existing(prepared, archive_root, "U3M9B_TEMP_CLEANUP")
    prepared.mkdir(parents=True, exist_ok=False)
    with zipfile.ZipFile(package_path, "r") as archive:
        archive.extractall(prepared)
    manifest_count = verify_manifest(prepared)

    required = [
        prepared / "01_PROGRAMS" / CORE_NAME,
        prepared / "01_PROGRAMS" / GATE_NAME,
        prepared / "02_CONTROLS" / "chart_profiles" / PROFILE_NAME,
        prepared / "02_CONTROLS" / "chart_profiles" / CONTRACT_NAME,
        prepared / "03_DOCUMENTS" / REPORT_NAME,
    ]
    for path in required:
        print(f"[CHECK] Required artifact: {path.relative_to(prepared)}")
        require(path.is_file(), f"Required prepared artifact is missing: {path.name}")
    print("[WHY] Core, independent gate, Profile, contract and report must travel together.")
    print(f"U3M9BP N-F{manifest_count}-A{len(required)}-X0 C-0")
    return 0


def run_child(command: list[str], timeout_seconds: int) -> subprocess.CompletedProcess[str]:
    return subprocess.run(
        command,
        text=True,
        capture_output=True,
        timeout=timeout_seconds,
        check=False,
    )


def self_test(root: Path) -> int:
    prepared = root / "04_TEMP" / PREPARED_NAME
    core = prepared / "01_PROGRAMS" / CORE_NAME
    require(core.is_file(), "Prepared core is missing. Run --prepare first.")
    result = run_child([sys.executable, str(core), "--self-test"], timeout_seconds=60)
    if result.stdout:
        print(result.stdout.strip())
    if result.stderr:
        print(result.stderr.strip())
    require(result.returncode == 0, "U3M9B core self-test failed.")
    require("U3M9B-SELFTEST-PASS-26 C-0" in result.stdout, "Expected self-test capsule is missing.")
    return 0


def promote_file(source: Path, destination: Path, archive_root: Path) -> None:
    destination.parent.mkdir(parents=True, exist_ok=True)
    if destination.exists():
        archive_existing(destination, archive_root, "U3M9B_REPLACED_PAYLOAD")
    shutil.copy2(source, destination)
    require(sha256_file(source) == sha256_file(destination), f"Promoted file hash differs: {destination.name}")


def execute(root: Path) -> int:
    prepared = root / "04_TEMP" / PREPARED_NAME
    core = prepared / "01_PROGRAMS" / CORE_NAME
    gate = prepared / "01_PROGRAMS" / GATE_NAME
    profile = prepared / "02_CONTROLS" / "chart_profiles" / PROFILE_NAME
    contract = prepared / "02_CONTROLS" / "chart_profiles" / CONTRACT_NAME
    report = prepared / "03_DOCUMENTS" / REPORT_NAME
    output = prepared / "05_OUTPUT"
    for path in (core, gate, profile, contract, report):
        require(path.is_file(), f"Prepared artifact is missing: {path.name}")

    gate_result = run_child(
        [
            sys.executable,
            str(gate),
            "--core",
            str(core),
            "--profile",
            str(profile),
            "--out",
            str(output),
        ],
        timeout_seconds=90,
    )
    if gate_result.stdout:
        print(gate_result.stdout.strip())
    if gate_result.stderr:
        print(gate_result.stderr.strip())
    require(gate_result.returncode == 0, "U3M9B independent acceptance gate failed.")
    require("U3M9BG1 " in gate_result.stdout and " C-0" in gate_result.stdout, "Gate pass capsule is missing.")

    archive_root = root / "A-Archive"
    promoted = [
        (core, root / "01_PROGRAMS" / CORE_NAME),
        (contract, root / "02_CONTROLS" / "chart_profiles" / CONTRACT_NAME),
        (profile, root / "02_CONTROLS" / "chart_profiles" / PROFILE_NAME),
        (report, root / "03_DOCUMENTS" / REPORT_NAME),
    ]
    for source, destination in promoted:
        promote_file(source, destination, archive_root)
    print(f"U3M9BI1 N-P{len(promoted)}-S1-G1-R1-X0 C-0")
    return 0


def main() -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--prepare", action="store_true")
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--execute", action="store_true")
    parser.add_argument("--root", type=Path)
    args = parser.parse_args()
    root = discover_udp003_root(args.root)
    if args.prepare:
        return prepare(root, Path(__file__).resolve())
    if args.self_test:
        return self_test(root)
    return execute(root)


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except (InstallError, OSError, ValueError, zipfile.BadZipFile, subprocess.SubprocessError) as error:
        print(f"U3M9BIE K-{type(error).__name__.upper()} D-{error} X1 C-7")
        raise SystemExit(7)
