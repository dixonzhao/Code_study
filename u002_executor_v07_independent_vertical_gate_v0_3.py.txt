class GateError(RuntimeError):
    def __init__(self, reason: str):
        self.reason = stable_reason(reason)
        super().__init__(self.reason)


def stable_reason(value: object) -> str:
    return re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")[:64] or "INTERNAL"


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def require_file(path: Path, reason: str) -> Path:
    if not path.is_file() or path.is_symlink():
        raise GateError(reason)
    return path


def atomic_json(path: Path, value: object) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    if temporary.exists():
        raise GateError("ATOMIC_TEMP_EXISTS")
    data = (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2,
                       allow_nan=False) + "\n").encode("utf-8")
    try:
        with temporary.open("xb") as stream:
            stream.write(data)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)
    finally:
        if temporary.exists():
            temporary.unlink()


def read_json(path: Path, reason: str) -> dict:
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except (OSError, UnicodeError, json.JSONDecodeError) as exc:
        raise GateError(reason) from exc
    if not isinstance(value, dict):
        raise GateError(reason)
    return value


def source_audit(path: Path) -> Dict[str, object]:
    text = path.read_text(encoding="utf-8-sig")
    try:
        tree = ast.parse(text)
    except SyntaxError as exc:
        raise GateError("CANDIDATE_SYNTAX") from exc
    functions = {node.name: node for node in tree.body if isinstance(node, ast.FunctionDef)}
    required_functions = {
        "execute_governed_run", "module_worker", "commit_result",
        "persist_run_manifest", "parse_execution_order", "validate_quality_decision",
    }
    if not required_functions.issubset(functions):
        raise GateError("EXECUTION_FUNCTION_MISSING")
    calls: Set[str] = set()
    for node in ast.walk(functions["execute_governed_run"]):
        if isinstance(node, ast.Call):
            if isinstance(node.func, ast.Name):
                calls.add(node.func.id)
            elif isinstance(node.func, ast.Attribute):
                calls.add(node.func.attr)
    required_calls = {
        "dependency_preflight", "validate_plan_preflight", "init_run_via_engine",
        "worker_command", "commit_result", "persist_run_manifest",
    }
    if not required_calls.issubset(calls):
        raise GateError("EXECUTION_CALL_GRAPH")
    forbidden = (
        "FORMAL EXECUTION REQUIRES INDEPENDENT GATE",
        "FORMAL_EXECUTION_REQUIRES_INDEPENDENT_GATE",
        "NOTIMPLEMENTEDERROR",
    )
    upper = text.upper()
    if any(token in upper for token in forbidden):
        raise GateError("EXECUTION_PLACEHOLDER")
    if "--execute-run" not in text or "execute_governed_run(" not in text:
        raise GateError("EXECUTION_DISPATCH")
    if "VERSION = \"0.7\"" not in text:
        raise GateError("CANDIDATE_VERSION")
    if "normalize_nested_signal_records" not in functions:
        raise GateError("SIGNAL_ADAPTER_MISSING")
    return {
        "CandidateSha256": sha256_file(path),
        "RequiredFunctions": sorted(required_functions),
        "RequiredExecutionCalls": sorted(required_calls),
        "ForbiddenPlaceholderCount": 0,
    }

    ]


def run_candidate(candidate: Path, arguments: Sequence[str], expected: str,
                  log_prefix: Path, timeout: int = 300) -> None:
    result = subprocess.run(
        [sys.executable, str(candidate), *arguments],
        stdin=subprocess.DEVNULL, capture_output=True,
        timeout=timeout, check=False,
    )
    stdout = result.stdout[:MAX_CAPTURE]
    stderr = result.stderr[:MAX_CAPTURE]
    log_prefix.parent.mkdir(parents=True, exist_ok=True)
    log_prefix.with_suffix(".stdout.txt").write_bytes(stdout)
    log_prefix.with_suffix(".stderr.txt").write_bytes(stderr)
    if result.returncode != 0:
        combined = (stdout + b"\n" + stderr).decode("utf-8", errors="replace")
        matches = re.findall(r"(?:^|\s)K-([A-Z0-9_]+)", combined.upper())
        detail = matches[-1] if matches else f"RC_{result.returncode}"
        raise GateError(log_prefix.name.upper() + "_" + detail)
    lines = [line.strip() for line in stdout.decode("utf-8", errors="replace").splitlines() if line.strip()]
    if expected not in lines:
        raise GateError("CANDIDATE_CAPSULE")


def load_g2(path: Path) -> ModuleType:
    loader = importlib.machinery.SourceFileLoader("_u002_g2_v07_frozen_base", str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None:
        raise GateError("G2_LOAD_SPEC")
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)
    return module


def configure_g2(module: ModuleType) -> None:
    if getattr(module, "VERSION", None) != "0.1":
        raise GateError("G2_BASE_VERSION")
    names = dict(getattr(module, "PROGRAM_NAMES", {}))
    if names.get("executor") != "u002_controlled_stage_executor_v0_3.py":
        raise GateError("G2_BASE_EXECUTOR")
    names["executor"] = DESTINATION_REL.name
    module.PROGRAM_NAMES = names
    module.VERSION = "0.7"
    module.RUN_CAPSULE = EXPECTED_VERTICAL


def snapshot(paths: Sequence[Path]) -> Dict[str, str]:
    return {str(path.resolve()): sha256_file(require_file(path, "PROTECTED_FILE")) for path in paths}

    ]


def install_candidate(candidate: Path, destination: Path) -> bool:
    if destination.exists():
        require_file(destination, "DESTINATION_CONFLICT")
        if sha256_file(destination) != sha256_file(candidate):
            raise GateError("DESTINATION_CONFLICT")
        return False
    destination.parent.mkdir(parents=True, exist_ok=True)
    temporary = destination.with_name(destination.name + ".installing")
    if temporary.exists():
        raise GateError("INSTALL_TEMP_EXISTS")
    shutil.copyfile(candidate, temporary)
    os.replace(temporary, destination)
    if sha256_file(destination) != sha256_file(candidate):
        raise GateError("INSTALL_IDENTITY")
    return True


def validate_synthetic_payloads(out_dir: Path) -> None:
    expected = {
        "g2_semantic.json": (1, 1, 0),
        "g2_baseline.json": (2, 0, 1),
        "g2_handoff.json": (3, 0, 1),
    }
    observed: Dict[str, tuple] = {}
    for path in out_dir.rglob("g2_*.json"):
        if path.name not in expected:
            continue
        if path.name in observed:
            raise GateError("SYNTHETIC_PAYLOAD_DUPLICATE")
        document = read_json(path, "SYNTHETIC_PAYLOAD_JSON")
        observed[path.name] = (
            document.get("Sequence"), document.get("BoundSignalCount"),
            document.get("InputArtifactCount"),
        )
    if observed != expected:
        raise GateError("SYNTHETIC_PAYLOAD_SEMANTICS")


def run_gate(root: Path) -> None:
    root = root.resolve()
    if not root.is_dir():
        raise GateError("WORKSPACE_ROOT")
    candidate = require_file(root / CANDIDATE_REL, "CANDIDATE_MISSING")
    base_g2 = require_file(root / BASE_G2_REL, "G2_BASE_MISSING")
    if sha256_file(base_g2) != BASE_G2_SHA256:
        raise GateError("G2_BASE_IDENTITY")
    destination = root / DESTINATION_REL
    out_dir = root / OUT_REL
    if out_dir.exists() or out_dir.with_name(out_dir.name + ".staging").exists():
        raise GateError("OUTPUT_EXISTS")
    audit = source_audit(candidate)
    protected = protected_paths(root, candidate, base_g2)
    before = snapshot(protected)
    logs = root / "04_TEMP" / ".G2R3_V07_GATE_LOGS"
    if logs.exists():
        raise GateError("GATE_LOG_EXISTS")
    logs.mkdir()
    installed_now = False
    try:
        common = dependency_args(root)
        run_candidate(candidate, ["--dependency-preflight", *common], EXPECTED_DP,
                      logs / "candidate_dependency")
        run_candidate(candidate, ["--self-test", *common], EXPECTED_ST,
                      logs / "candidate_selftest")
        installed_now = install_candidate(candidate, destination)
        run_candidate(destination, ["--dependency-preflight", *common], EXPECTED_DP,
                      logs / "installed_dependency")
        run_candidate(destination, ["--self-test", *common], EXPECTED_ST,
                      logs / "installed_selftest")
        module = load_g2(base_g2)
        configure_g2(module)
        capsule = module.run_vertical(root, out_dir)
        if capsule != EXPECTED_VERTICAL:
            raise GateError("VERTICAL_CAPSULE")
        vertical = read_json(out_dir / "u21g2_vertical_acceptance.json", "VERTICAL_EVIDENCE")
        if vertical.get("Status") != "SYNTHETIC_VERTICAL_RUN_ACCEPTED":
            raise GateError("VERTICAL_STATUS")
        if vertical.get("ModuleIds") != [
            "U002.VerticalSemantic", "U002.VerticalBaseline", "U002.VerticalHandoff"
        ]:
            raise GateError("VERTICAL_SAMPLE_MODULES")
        if vertical.get("StageOrder") != ["S1", "S4", "S7"]:
            raise GateError("VERTICAL_SAMPLE_STAGES")
        if vertical.get("ExpectedArtifactTypes") != [
            "SYNTHETIC_SEMANTIC", "SYNTHETIC_BASELINE", "SYNTHETIC_HANDOFF"
        ]:
            raise GateError("VERTICAL_SAMPLE_ARTIFACTS")
        if vertical.get("CompanyDataUsed") is not False or vertical.get("InputsUnchanged") is not True:
            raise GateError("VERTICAL_SAMPLE_ISOLATION")
        validate_synthetic_payloads(out_dir)
        after = snapshot(protected)
        if before != after:
            raise GateError("PROTECTED_INPUT_MUTATION")
        if sha256_file(destination) != str(audit["CandidateSha256"]):
            raise GateError("INSTALLED_IDENTITY")
        evidence = {
            "AcceptanceType": PROGRAM_ID,
            "AcceptanceVersion": VERSION,
            "GateStatus": "PASS",
            "CandidateAudit": audit,
            "CandidateSha256": sha256_file(candidate),
            "InstalledSha256": sha256_file(destination),
            "InstalledByteIdentical": True,
            "DependencyPreflightAccepted": True,
            "InternalSelfTestAccepted": True,
            "FormalExecuteRunExercised": True,
            "SyntheticVerticalCapsule": EXPECTED_VERTICAL,
            "SyntheticModuleCount": 3,
            "SyntheticModuleStages": ["S1", "S4", "S7"],
            "SyntheticArtifactCount": 3,
            "ProtectedInputCount": len(protected),
            "ProtectedInputsUnchanged": True,
            "SuccessCapsule": SUCCESS_CAPSULE,
        }
        atomic_json(out_dir / "g2r3_executor_v07_independent_acceptance.json", evidence)
        (out_dir / "g2r3_return_capsule.txt").write_text(SUCCESS_CAPSULE + "\n", encoding="utf-8")
        shutil.rmtree(logs)
    except Exception:
        if installed_now and destination.exists():
            destination.unlink()
        raise
    print(SUCCESS_CAPSULE)


def self_test() -> None:
    checks: List[str] = []

    def ck(name: str, value: bool) -> None:
        if not value:
            raise GateError("SELFTEST_" + name)
        checks.append(name)

    source = Path(__file__).read_text(encoding="utf-8-sig")
    ck("program", PROGRAM_ID.startswith("UDP002"))
    ck("version", VERSION == "0.3")
    ck("candidate", CANDIDATE_REL.name.endswith("_v0_7.py.txt"))
    ck("destination", DESTINATION_REL.name.endswith("_v0_7.py"))
    ck("base_hash", len(BASE_G2_SHA256) == 64)
    ck("out", OUT_REL.name.endswith("V07"))
    ck("dp", EXPECTED_DP.endswith("C-0"))
    ck("st", EXPECTED_ST.endswith("C-0"))
    ck("vertical", EXPECTED_VERTICAL.endswith("C-0"))
    ck("success", SUCCESS_CAPSULE.endswith("C-0"))
    ck("ast", "ast.parse" in source)
    ck("call_graph", "required_calls" in source)
    ck("placeholder", "EXECUTION_PLACEHOLDER" in source)
    ck("byte_identity", "InstalledByteIdentical" in source)
    ck("rollback", "destination.unlink" in source)
    ck("mutation", "PROTECTED_INPUT_MUTATION" in source)
    ck("formal_run", "run_vertical" in source)
    ck("sample_modules", '"SyntheticModuleCount": 3' in source)
    ck("sample_stages", '"SyntheticModuleStages": ["S1", "S4", "S7"]' in source)
    ck("signal_adapter", "SIGNAL_ADAPTER_MISSING" in source)
    ck("payload_semantics", "validate_synthetic_payloads(out_dir)" in source)
    ck("count", len(checks) == 21)
    if len(checks) != 22:
        raise GateError("SELFTEST_COUNT")
    print(SELFTEST_CAPSULE)


def main() -> int:
    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--run", action="store_true")
    parser.add_argument("--workspace-root")
    args = parser.parse_args()
    try:
        if args.self_test:
            self_test()
        else:
            if not args.workspace_root:
                raise GateError("WORKSPACE_ROOT")
            run_gate(Path(args.workspace_root))
        return 0
    except GateError as exc:
        print(f"G2R3 N-E0-I0-V0-M0-A0-X1 C-7 K-{exc.reason}")
        return 7
    except Exception as exc:
        print(f"G2R3 N-E0-I0-V0-M0-A0-X1 C-7 K-{stable_reason(exc.__class__.__name__)}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
