TC001_VIDEO_FRAME_END = 57597


class RunError(RuntimeError):
    """Represent one deterministic fail-closed orchestration failure."""


def stable(value: object) -> str:
    """Convert diagnostic text into a compact manual return token."""

    normalized = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper())
    return normalized.strip("_")[:80] or "INTERNAL"


def canonical_bytes(value: object) -> bytes:
    """Serialize one control object deterministically."""

    text = json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        indent=2,
        allow_nan=False,
    )
    return (text + "\n").encode("utf-8")


def sha256_bytes(value: bytes) -> str:
    """Return a hexadecimal SHA-256 digest."""

    return hashlib.sha256(value).hexdigest()


def sha256_file(path: Path) -> str:
    """Hash one file without loading it completely into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def regular_file(path: Path) -> bool:
    """Accept ordinary files only."""

    return path.is_file() and not path.is_symlink()


def require_workspace(path: Path) -> Path:
    """Validate the fixed U002 workspace identity."""

    root = path.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise RunError("WORKSPACE_ROOT")
    return root


def load_json(path: Path, reason: str) -> Dict[str, object]:
    """Load one JSON object and reject arrays or missing files."""

    if not regular_file(path):
        raise RunError(reason + "_MISSING")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise RunError(reason + "_JSON") from exc
    if not isinstance(value, dict):
        raise RunError(reason + "_TYPE")
    return value


def require_foundation(root: Path) -> None:
    """Require every frozen core program used by this Run."""

    required = (PREFLIGHT, EXECUTOR, ARTIFACT_STORE, FREEZE_AUDITOR, B1B2_RECEIPT)
    for relative in required:
        if not regular_file(root / relative):
            raise RunError("FOUNDATION_MISSING_" + relative.name)


def run_child(command: Sequence[str], timeout: int = 7200) -> subprocess.CompletedProcess:
    """Run one controlled child process and retain both diagnostic streams."""

    return subprocess.run(
        list(command),
        capture_output=True,
        text=True,
        timeout=timeout,
        check=False,
    )


def require_child(result: subprocess.CompletedProcess, label: str) -> None:
    """Convert child failure into one stable orchestration reason."""

    if result.returncode == 0 and "C-0" in result.stdout:
        return
    detail = result.stderr.strip() or result.stdout.strip() or "NO_DETAIL"
    raise RunError(label + "_" + stable(detail))


def freeze_audit(root: Path) -> str:
    """Verify that the U22 Phase-A foundation remains byte-identical."""

    result = run_child(
        [
            sys.executable,
            str(root / FREEZE_AUDITOR),
            "--audit-freeze",
            "--workspace-root",
            str(root),
        ],
        timeout=120,
    )
    require_child(result, "FOUNDATION_AUDIT")
    return result.stdout.strip().splitlines()[-1]


def package_snapshot(package: Path, receipt: Mapping[str, object]) -> Dict[str, str]:
    """Verify the exact six U001 files accepted by the B1AR receipt."""

    input_files = receipt.get("InputFiles")
    if not isinstance(input_files, list) or len(input_files) != 6:
        raise RunError("B1AR_INPUT_FILES")
    snapshot: Dict[str, str] = {}
    for item in input_files:
        if not isinstance(item, dict):
            raise RunError("B1AR_INPUT_ITEM")
        basename = str(item.get("Basename", ""))
        expected = str(item.get("Sha256", ""))
        path = package / basename
        if not regular_file(path):
            raise RunError("U001_FILE_MISSING_" + basename)
        observed = sha256_file(path)
        if observed != expected:
            raise RunError("U001_FILE_HASH_" + basename)
        snapshot[basename] = observed
    return snapshot


def discover_b1ar_receipt(root: Path) -> Tuple[Path, Dict[str, object]]:
    """Choose the largest, then newest, accepted real U001 handoff receipt."""

    candidates: List[Tuple[int, int, Path, Dict[str, object]]] = []
    for path in (root / "07_OUTPUT").glob("*/" + B1AR_RECEIPT_NAME):
        try:
            receipt = load_json(path, "B1AR_RECEIPT")
            if receipt.get("Status") != "REAL_U001_HANDOFF_ACCEPTED":
                continue
            frame = receipt.get("FrameDomain")
            if not isinstance(frame, dict):
                continue
            rows = int(frame.get("FrameRows", 0))
            package = Path(str(receipt.get("SelectedPackage", ""))).resolve()
            package_snapshot(package, receipt)
            candidates.append((rows, path.stat().st_mtime_ns, path, receipt))
        except Exception:
            continue
    if not candidates:
        raise RunError("B1AR_ACCEPTED_RECEIPT_NOT_FOUND")
    _, _, path, receipt = max(candidates, key=lambda item: (item[0], item[1], str(item[2])))
    return path, receipt


def accepted_s0_identity(
    receipt_path: Path,
) -> Tuple[str, str]:
    """Read the exact S0 identity already proven by the real B1AR Run."""

    plan = load_json(
        receipt_path.parent / "00_CONFIG/resolved_run_plan.json",
        "B1AR_RESOLVED_PLAN",
    )
    modules = plan.get("ModulePlan")
    if not isinstance(modules, list):
        raise RunError("B1AR_MODULE_PLAN")
    matches = []
    for item in modules:
        if not isinstance(item, dict):
            continue
        if item.get("Stage") != "S0":
            continue
        provided = item.get("ProvidesArtifacts")
        if isinstance(provided, list) and "U001_INTAKE_ACCEPTED" in provided:
            matches.append(item)
    if len(matches) != 1:
        raise RunError("B1AR_S0_IDENTITY_COUNT")
    return str(matches[0]["ModuleId"]), str(matches[0]["ModuleVersion"])


def accepted_s1(
    root: Path,
) -> Tuple[str, str, Dict[str, object], str]:
    """Load the exact B1B2-qualified S1 identity and configuration."""

    receipt = load_json(root / B1B2_RECEIPT, "B1B2_RECEIPT")
    if receipt.get("Status") != "S1_MODULE_INSTALLED_AND_QUALIFIED":
        raise RunError("B1B2_STATUS")
    relative = Path(str(receipt.get("ConfigurationRelativePath", "")))
    config_path = root / relative
    if sha256_file(config_path) != receipt.get("ConfigurationSha256"):
        raise RunError("B1B2_CONFIGURATION_HASH")
    configuration = load_json(config_path, "B1B2_CONFIGURATION")
    return (
        str(receipt["ModuleId"]),
        str(receipt["ModuleVersion"]),
        configuration,
        str(receipt["ConfigurationSha256"]),
    )


def frame_range(receipt: Mapping[str, object]) -> Tuple[int, int]:
    """Restrict TC001 to the declared road/video-overlap range."""

    frame = receipt.get("FrameDomain")
    if not isinstance(frame, dict):
        raise RunError("B1AR_FRAME_DOMAIN")
    frame_min = int(frame.get("FrameMin"))
    frame_max = int(frame.get("FrameMax"))
    start = max(FRAME_START, frame_min)
    end = min(TC001_VIDEO_FRAME_END, frame_max)
    if start >= end:
        raise RunError("TC001_FRAME_RANGE_EMPTY")
    return start, end

    package: Path,
    fingerprint: str,
    s0_identity: Tuple[str, str],
    s1_identity: Tuple[str, str],
    s1_configuration: Mapping[str, object],
    start: int,
    end: int,

    """Build the explicit two-Module S0-to-S1 Run configuration."""

    return {

        "RunId": run_id,
        "RunPurpose": "EVALUATION",
        "RunScope": "FRAME_RANGE",
        "FrameStart": start,
        "FrameEnd": end,
        "InputPackage": str(package.resolve()),
        "SignalProfile": "TC001.CameraRadar.Base.PROVISIONAL@0.1",
        "ModulePlan": [
            {
                "Sequence": 1,
                "Stage": "S0",
                "ModuleId": s0_identity[0],
                "ModuleVersion": s0_identity[1],
                "Configuration": {},
            },
            {
                "Sequence": 2,
                "Stage": "S1",
                "ModuleId": s1_identity[0],
                "ModuleVersion": s1_identity[1],
                "Configuration": dict(s1_configuration),
            },
        ],
        "PrimaryOutcome": {
            "Name": "CanonicalCameraDistance",
            "Formula": "S1ContinuityHysteresis(Camera1Distance,Camera3Distance)",
            "Sign": "NOT_APPLICABLE",
            "Unit": "m",
            "Purpose": "EXPLORATORY",
        },
        "CandidatePolicy": "NotAppliedInU22B1B3@0.1",
        "OutputSelection": [
            "U001_INTAKE_ACCEPTED",
            "CAMERA_TARGET_RECONSTRUCTION",
        ],
        "SourcePackageFingerprint": fingerprint,
    }


def schema_run_configuration(configuration: Mapping[str, object]) -> Dict[str, object]:
    """Remove the local identity helper before frozen schema validation."""

    result = dict(configuration)
    result.pop("SourcePackageFingerprint", None)
    return result


def deterministic_run_id(configuration: Mapping[str, object]) -> str:
    """Bind Run identity to source, Modules, scope and configuration."""

    digest = sha256_bytes(canonical_bytes(configuration))[:16]
    return "U22B1B3_" + digest


def write_exact(path: Path, value: Mapping[str, object]) -> None:
    """Create one control file or accept only an identical replay."""

    payload = canonical_bytes(value)
    path.parent.mkdir(parents=True, exist_ok=True)
    if path.exists():
        if not regular_file(path) or path.read_bytes() != payload:
            raise RunError("CONTROL_FILE_CONFLICT_" + path.name)
        return
    path.write_bytes(payload)


def execute_pipeline(root: Path, config_path: Path, run_id: str) -> None:
    """Prepare, execute and audit the governed two-stage Run."""

    run_root = root / "07_OUTPUT" / run_id
    already_succeeded = False
    if run_root.exists():
        status = load_json(run_root / "run_status.json", "RUN_STATUS")
        state = str(status.get("Status", ""))
        if state == "SUCCEEDED":
            already_succeeded = True
        elif state != "PREPARED":
            raise RunError("EXISTING_RUN_STATE_" + state)
    else:
        prepared = run_child(
            [
                sys.executable,
                str(root / PREFLIGHT),
                "--prepare-run",
                str(config_path),
                "--workspace-root",
                str(root),
            ]
        )
        require_child(prepared, "PREFLIGHT")
    if not already_succeeded:
        executed = run_child(
            [
                sys.executable,
                str(root / EXECUTOR),
                "--execute-run",
                str(run_root),
                "--workspace-root",
                str(root),
                "--module-timeout-seconds",
                "7200",
            ]
        )
        require_child(executed, "EXECUTOR")
    audited = run_child(
        [
            sys.executable,
            str(root / ARTIFACT_STORE),
            "--audit-run",
            "--run-root",
            str(run_root),
            "--workspace-root",
            str(root),
        ]
    )
    require_child(audited, "ARTIFACT_AUDIT")


def write_run_receipt(
    root: Path,
    run_id: str,
    b1ar_path: Path,
    b1ar_receipt: Mapping[str, object],
    source_snapshot: Mapping[str, str],
    s0_identity: Tuple[str, str],
    s1_identity: Tuple[str, str],
    configuration_sha: str,
    start: int,
    end: int,
    freeze_before: str,
    freeze_after: str,
) -> None:
    """Persist the exact technical handoff consumed by the B1B3 gate."""

    receipt = {
        "ReceiptContract": "UDP002_U22B1B3_REAL_S1_RUN_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": VERSION,
        "Status": "REAL_S1_RECONSTRUCTION_RUN_COMPLETED",
        "RunId": run_id,
        "SourceB1ARRunId": b1ar_receipt["RunId"],
        "SourceB1ARReceipt": b1ar_path.relative_to(root).as_posix(),
        "SelectedPackage": b1ar_receipt["SelectedPackage"],
        "PackageFingerprintSha256": b1ar_receipt["PackageFingerprintSha256"],
        "InputFileSha256": dict(source_snapshot),
        "S0Module": {"ModuleId": s0_identity[0], "ModuleVersion": s0_identity[1]},
        "S1Module": {"ModuleId": s1_identity[0], "ModuleVersion": s1_identity[1]},
        "S1ConfigurationSha256": configuration_sha,
        "FrameStart": start,
        "FrameEnd": end,
        "FoundationBefore": freeze_before,
        "FoundationAfter": freeze_after,
        "SemanticStatus": "HUMAN_REVIEW_PENDING",
    }
    write_exact(root / "07_OUTPUT" / run_id / RUN_RECEIPT_NAME, receipt)


def run_real(root: Path) -> str:
    """Execute the complete real S0-to-S1 technical Run."""

    require_foundation(root)
    b1ar_path, b1ar_receipt = discover_b1ar_receipt(root)
    package = Path(str(b1ar_receipt["SelectedPackage"])).resolve()
    before = package_snapshot(package, b1ar_receipt)
    fingerprint = str(b1ar_receipt["PackageFingerprintSha256"])
    s0_identity = accepted_s0_identity(b1ar_path)
    s1_id, s1_version, s1_configuration, configuration_sha = accepted_s1(root)
    start, end = frame_range(b1ar_receipt)
    identity_seed = build_run_configuration(
        "IDENTITY_PENDING",
        package,
        fingerprint,
        s0_identity,
        (s1_id, s1_version),
        s1_configuration,
        start,
        end,
    )
    identity_seed["RunId"] = ""
    run_id = deterministic_run_id(identity_seed)
    configuration = build_run_configuration(
        run_id,
        package,
        fingerprint,
        s0_identity,
        (s1_id, s1_version),
        s1_configuration,
        start,
        end,
    )
    config_path = root / "02_CONFIG/run_configs" / (run_id + ".json")
    write_exact(config_path, schema_run_configuration(configuration))
    freeze_before = freeze_audit(root)
    execute_pipeline(root, config_path, run_id)
    after = package_snapshot(package, b1ar_receipt)
    if before != after:
        raise RunError("U001_INPUT_CHANGED")
    freeze_after = freeze_audit(root)
    if freeze_before != freeze_after:
        raise RunError("FOUNDATION_CHANGED")
    write_run_receipt(
        root,
        run_id,
        b1ar_path,
        b1ar_receipt,
        before,
        s0_identity,
        (s1_id, s1_version),
        configuration_sha,
        start,
        end,
        freeze_before,
        freeze_after,
    )
    return "U22B1B3R N-M2-S2-A2-I1-X0 C-0"


def self_test() -> str:
    """Exercise deterministic identity, scope and replay helpers."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise RunError("SELFTEST")
        checks.append(True)

    check(PROGRAM_ID.startswith("U002"))
    check(VERSION == "0.1")
    check(CONTRACT_VERSION == "U22.0.1")
    check(stable("a b/c") == "A_B_C")
    check(len(sha256_bytes(b"x")) == 64)
    check(canonical_bytes({"b": 1, "a": 2}).startswith(b"{"))
    with tempfile.TemporaryDirectory(prefix="u22b1b3_runner_selftest_") as temporary:
        root = Path(temporary)
        control = root / "control.json"
        write_exact(control, {"a": 1})
        check(regular_file(control))
        write_exact(control, {"a": 1})
        check(load_json(control, "CONTROL") == {"a": 1})
        package = root / "package"
        package.mkdir()
        input_files = []
        for index in range(6):
            name = "f" + str(index)
            path = package / name
            path.write_text(name, encoding="utf-8")
            input_files.append({"Basename": name, "Sha256": sha256_file(path)})
        receipt = {"InputFiles": input_files}
        check(len(package_snapshot(package, receipt)) == 6)
    synthetic = build_run_configuration(
        "RUN_TEST",
        Path("package"),
        "a" * 64,
        ("S0.Module", "0.1.0"),
        ("S1.Module", "0.1.0"),
        {"A": 1},
        7400,
        7500,
    )
    check(len(synthetic["ModulePlan"]) == 2)
    check(synthetic["ModulePlan"][0]["Stage"] == "S0")
    check(synthetic["ModulePlan"][1]["Stage"] == "S1")
    check(synthetic["OutputSelection"][-1] == "CAMERA_TARGET_RECONSTRUCTION")
    check(schema_run_configuration(synthetic).get("SourcePackageFingerprint") is None)
    identity = dict(synthetic)
    identity["RunId"] = ""
    run_id = deterministic_run_id(identity)
    check(run_id.startswith("U22B1B3_"))
    check(len(run_id) == len("U22B1B3_") + 16)
    check(FRAME_START == 7400)
    check(TC001_VIDEO_FRAME_END == 57597)
    check(RUN_RECEIPT_NAME.endswith(".json"))
    return "U22B1B3R-ST N-T" + str(len(checks)) + "-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Create explicit self-test and real-run modes."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--run", action="store_true")
    parser.add_argument("--workspace-root")
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    """Program entry point."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            print(self_test())
            return 0
        if not args.workspace_root:
            raise RunError("WORKSPACE_ROOT_REQUIRED")
        print(run_real(require_workspace(Path(args.workspace_root))))
        return 0
    except subprocess.TimeoutExpired:
        print("U22B1B3R N-M0-S0-A0-I0-X1 C-7 R-TIMEOUT")
        return 7
    except Exception as exc:
        print(
            "U22B1B3R N-M0-S0-A0-I0-X1 C-7 R-"
            + stable(exc)
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
