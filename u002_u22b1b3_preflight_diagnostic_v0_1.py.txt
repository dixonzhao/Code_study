#!/usr/bin/env python3
"""Diagnose the exact frozen U22A5 Preflight rejection without preparing a Run.

The diagnostic reuses the installed Preflight's validation functions but reads
the already-refreshed availability snapshot instead of refreshing it again.
Consequently it does not execute Modules or create/modify a Run directory.
"""

from __future__ import annotations

import argparse
import importlib.util
import json
import os
from pathlib import Path
import re
import sys
from typing import Any, Dict, List, Mapping, Sequence, Set, Tuple


PROGRAM_ID = "U002-U22B1B3-PREFLIGHT-DIAGNOSTIC"
VERSION = "0.1"
PREFLIGHT_RELATIVE = Path("01_PROGRAMS/core/u002_u22_engine_preflight_v0_1.py")
AVAILABILITY_RELATIVE = Path(
    "00_CONTROL/registry/u002_module_availability_snapshot_v0_1.json"
)
CONFIG_ROOT_RELATIVE = Path("02_CONFIG/run_configs")
RESULT_RELATIVE = Path("04_TEMP/u22b1b3_preflight_diagnostic_result_v0_1.json")


class DiagnosticError(RuntimeError):
    """Represent a failure of the diagnostic itself."""


def stable(value: object, limit: int = 48) -> str:
    """Convert text to a short manual-transfer token."""

    token = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (token or "NA")[:limit]


def regular_file(path: Path) -> bool:
    """Accept ordinary non-symbolic-link files only."""

    return path.is_file() and not path.is_symlink()


def require_workspace(value: Path) -> Path:
    """Resolve and validate the fixed U002 workspace identity."""

    root = value.resolve()
    if not root.is_dir() or root.name != "U-UDP002":
        raise DiagnosticError("WORKSPACE_ROOT")
    return root


def load_json(path: Path, reason: str) -> Dict[str, Any]:
    """Load one JSON object with deterministic error labels."""

    if not regular_file(path):
        raise DiagnosticError(reason + "_MISSING")
    try:
        value = json.loads(path.read_text(encoding="utf-8-sig"))
    except Exception as exc:
        raise DiagnosticError(reason + "_JSON") from exc
    if not isinstance(value, dict):
        raise DiagnosticError(reason + "_TYPE")
    return value


def load_preflight(root: Path) -> Any:
    """Load the exact installed frozen Preflight implementation."""

    path = root / PREFLIGHT_RELATIVE
    if not regular_file(path):
        raise DiagnosticError("PREFLIGHT_MISSING")
    spec = importlib.util.spec_from_file_location("u22a5_diagnostic_target", path)
    if spec is None or spec.loader is None:
        raise DiagnosticError("PREFLIGHT_LOAD")
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    required = (
        "load_validator",
        "load_run_configuration",
        "resolve_input_package",
        "availability_entry",
        "load_qualified_manifest",
        "validate_module_configuration",
        "resolved_module_record",
        "PreflightError",
    )
    if any(not hasattr(module, name) for name in required):
        raise DiagnosticError("PREFLIGHT_API")
    return module


def select_configuration(root: Path) -> Tuple[Path, int]:
    """Select the newest B1B3 configuration left by the failed runner."""

    config_root = root / CONFIG_ROOT_RELATIVE
    candidates = [
        path
        for path in config_root.glob("U22B1B3_*.json")
        if regular_file(path)
    ]
    if not candidates:
        raise DiagnosticError("B1B3_CONFIG_NOT_FOUND")
    selected = max(candidates, key=lambda path: (path.stat().st_mtime_ns, path.name))
    return selected, len(candidates)


def module_summary(
    item: Mapping[str, Any],
    entry: Mapping[str, Any],
    manifest: Mapping[str, Any],
) -> Dict[str, Any]:
    """Create evidence sufficient to diagnose identity and dependency drift."""

    return {
        "Sequence": item.get("Sequence"),
        "RequestedStage": item.get("Stage"),
        "RequestedModuleId": item.get("ModuleId"),
        "RequestedModuleVersion": item.get("ModuleVersion"),
        "RegistryStage": entry.get("Stage"),
        "RegistryStatus": entry.get("RegistryStatus"),
        "RegistryReason": entry.get("Reason"),
        "ManifestPath": entry.get("ManifestPath"),
        "ManifestSha256": entry.get("ManifestSha256"),
        "PackageFileRoles": sorted(
            entry.get("PackageFiles", {}).keys()
            if isinstance(entry.get("PackageFiles"), dict)
            else []
        ),
        "RequiresArtifacts": list(manifest.get("RequiresArtifacts", [])),
        "ProvidesArtifacts": list(manifest.get("ProvidesArtifacts", [])),
    }


def inspect_plan(root: Path) -> Dict[str, Any]:
    """Replay Preflight plan validation in memory and retain the exact reason."""

    preflight = load_preflight(root)
    config_path, config_count = select_configuration(root)
    availability_path = root / AVAILABILITY_RELATIVE
    availability = load_json(availability_path, "AVAILABILITY")
    if not isinstance(availability.get("Entries"), list):
        raise DiagnosticError("AVAILABILITY_SHAPE")

    result: Dict[str, Any] = {
        "DiagnosticId": PROGRAM_ID,
        "DiagnosticVersion": VERSION,
        "WorkspaceRoot": str(root),
        "ConfigurationPath": str(config_path),
        "ConfigurationCandidateCount": config_count,
        "AvailabilityPath": str(availability_path),
        "AvailabilityEntryCount": len(availability["Entries"]),
        "FailureReason": None,
        "FailureDetail": None,
        "FailureStage": None,
        "Modules": [],
        "ProvidedBeforeFailure": [],
    }

    current_stage = "CONFIG"
    provided: Set[str] = set()
    matched_entries = 0
    package_metadata_count = 0
    dependency_ok = True
    order_ok = True
    module_count = 0

    try:
        validator = preflight.load_validator(root)
        configuration = preflight.load_run_configuration(root, config_path, validator)
        preflight.resolve_input_package(root, str(configuration["InputPackage"]))
        plan = configuration.get("ModulePlan")
        if not isinstance(plan, list):
            raise DiagnosticError("MODULE_PLAN_TYPE")
        module_count = len(plan)
        expected_sequence = 1
        availability["_WorkspaceRoot"] = str(root)

        for item in plan:
            if not isinstance(item, dict):
                raise DiagnosticError("MODULE_PLAN_ITEM")
            current_stage = str(item.get("Stage", "UNKNOWN"))
            if item.get("Sequence") != expected_sequence:
                order_ok = False
            expected_sequence += 1

            entry = preflight.availability_entry(
                availability,
                str(item.get("ModuleId", "")),
                str(item.get("ModuleVersion", "")),
            )
            matched_entries += 1
            if isinstance(entry.get("PackageFiles"), dict):
                package_metadata_count += 1
            if entry.get("Stage") != item.get("Stage"):
                raise preflight.PreflightError(
                    "PLAN_STAGE_IDENTITY",
                    str(item.get("ModuleId", "")),
                )

            manifest = preflight.load_qualified_manifest(root, entry, validator)
            result["Modules"].append(module_summary(item, entry, manifest))
            preflight.validate_module_configuration(
                root,
                entry,
                item.get("Configuration", {}),
                validator,
            )

            missing = [
                artifact
                for artifact in manifest.get("RequiresArtifacts", [])
                if artifact not in provided
            ]
            if missing:
                dependency_ok = False
                raise preflight.PreflightError(
                    "PLAN_ARTIFACT_DEPENDENCY",
                    str(missing[0]),
                )
            for artifact in manifest.get("ProvidesArtifacts", []):
                if artifact in provided:
                    dependency_ok = False
                    raise preflight.PreflightError(
                        "PLAN_ARTIFACT_CONFLICT",
                        str(artifact),
                    )
                provided.add(str(artifact))

            # This catches missing EntryPoint metadata and path errors as A5 does.
            preflight.resolved_module_record(root, item, entry, manifest)

        unavailable = [
            artifact
            for artifact in configuration.get("OutputSelection", [])
            if artifact not in provided
        ]
        if unavailable:
            dependency_ok = False
            raise preflight.PreflightError("OUTPUT_NOT_PROVIDED", str(unavailable[0]))
    except preflight.PreflightError as exc:
        result["FailureReason"] = str(exc.reason)
        result["FailureDetail"] = str(exc.detail)
        result["FailureStage"] = current_stage
    finally:
        availability.pop("_WorkspaceRoot", None)

    result["ModulePlanCount"] = module_count
    result["MatchedAvailabilityEntries"] = matched_entries
    result["PackageMetadataCount"] = package_metadata_count
    result["StageOrderValid"] = order_ok
    result["ArtifactDependencyValid"] = dependency_ok
    result["ProvidedBeforeFailure"] = sorted(provided)
    result["DiagnosticSucceeded"] = True
    return result


def write_result(root: Path, result: Mapping[str, Any]) -> Path:
    """Atomically replace the diagnostic evidence file, never a Run artifact."""

    path = root / RESULT_RELATIVE
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = (
        json.dumps(result, ensure_ascii=False, sort_keys=True, indent=2) + "\n"
    ).encode("utf-8")
    temporary = path.with_name(path.name + ".writing")
    if temporary.exists():
        temporary.unlink()
    temporary.write_bytes(payload)
    os.replace(temporary, path)
    return path


def capsule(result: Mapping[str, Any]) -> str:
    """Emit one concise capsule while retaining full details in JSON."""

    failure = result.get("FailureReason") or "NONE"
    detail = result.get("FailureDetail") or "NA"
    stage = result.get("FailureStage") or "NA"
    return (
        f"U22B1B3D1 N-C{result['ConfigurationCandidateCount']}"
        f"-M{result['ModulePlanCount']}"
        f"-O{int(bool(result['StageOrderValid']))}"
        f"-A{result['MatchedAvailabilityEntries']}"
        f"-D{int(bool(result['ArtifactDependencyValid']))}"
        f"-P{result['PackageMetadataCount']}-X0 C-0"
        f" K-{stable(failure, 40)} S-{stable(stage, 8)} R-{stable(detail, 40)}"
    )


def self_test() -> str:
    """Exercise compact tokens and structural helpers without workspace data."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise DiagnosticError("SELFTEST")
        checks.append(True)

    check(PROGRAM_ID.startswith("U002"))
    check(VERSION == "0.1")
    check(stable("plan artifact/dependency") == "PLAN_ARTIFACT_DEPENDENCY")
    check(stable("") == "NA")
    check(PREFLIGHT_RELATIVE.name.endswith(".py"))
    check(AVAILABILITY_RELATIVE.name.endswith(".json"))
    check(CONFIG_ROOT_RELATIVE.name == "run_configs")
    check(RESULT_RELATIVE.parts[0] == "04_TEMP")
    check(regular_file(Path(__file__)))
    check(len(checks) == 9)
    return "U22B1B3D1-ST N-T10-X0 C-0"


def parse_args(argv: Sequence[str]) -> argparse.Namespace:
    """Expose only self-test and one deterministic diagnostic mode."""

    parser = argparse.ArgumentParser()
    mode = parser.add_mutually_exclusive_group(required=True)
    mode.add_argument("--self-test", action="store_true")
    mode.add_argument("--diagnose", action="store_true")
    parser.add_argument("--workspace-root")
    return parser.parse_args(argv)


def main(argv: Sequence[str] = None) -> int:
    """Run the requested mode and preserve one-line manual feedback."""

    args = parse_args(sys.argv[1:] if argv is None else argv)
    try:
        if args.self_test:
            print(self_test())
            return 0
        if not args.workspace_root:
            raise DiagnosticError("WORKSPACE_ROOT_REQUIRED")
        root = require_workspace(Path(args.workspace_root))
        result = inspect_plan(root)
        write_result(root, result)
        print(capsule(result))
        return 0
    except DiagnosticError as exc:
        print(f"U22B1B3D1 N-C0-M0-O0-A0-D0-P0-X1 C-7 K-{stable(exc)}")
        return 7
    except Exception as exc:
        print(
            "U22B1B3D1 N-C0-M0-O0-A0-D0-P0-X1 C-7"
            f" K-{stable(exc.__class__.__name__)}"
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
