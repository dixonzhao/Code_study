"""Read-only core replay comparator for UDP001 Run A and GUI Replay."""

from __future__ import annotations

import hashlib
import json
import math
import os
import re
import sys
import tempfile
from pathlib import Path
from typing import Any, Dict, Iterable, List, Mapping, Sequence, Tuple


PROGRAM_NAME = "u001_core_replay_comparator_v0_1.py"
PROGRAM_VERSION = "0.1"
SELFTEST_TOKEN = "U001-REPLAY-COMPARATOR-SELFTEST-PASS-16"

PACKAGE_NAMES = (
    "u001_decoded_updates.csv",
    "u001_frame_values.csv",
    "u001_signal_catalog.csv",
    "u001_quality_report.csv",
    "u001_run_manifest.json",
    "u001_review_view.csv",
)

D3_KEY = ("CanonicalName",)
D4_KEY = ("Scope", "CanonicalName", "Metric")
STABLE_MANIFEST_KEYS = (
    "DatasetId", "SegmentId", "SourceId", "ContractVersion",
    "ImplementationProfile", "ProcessingMode",
)
ALLOWED_MANIFEST_DIFFERENCE_KEYS = (
    "RunId", "RunTimestamp", "StartedUTC", "FinishedUTC", "ExecutionDuration",
    "ConfigurationFingerprint", "IdentityStatus", "Warnings",
)


class CompareError(RuntimeError):
    pass


def sha256_file(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open("rb") as f:
        for block in iter(lambda: f.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def exact_equal(a: Path, b: Path) -> bool:
    return a.stat().st_size == b.stat().st_size and sha256_file(a) == sha256_file(b)


def find_package(run_root: Path) -> Dict[str, Path]:
    if not run_root.is_dir():
        raise CompareError("RUN_ROOT_NOT_DIRECTORY")
    result: Dict[str, Path] = {}
    for name in PACKAGE_NAMES:
        matches = sorted(p for p in run_root.rglob(name) if p.is_file())
        if len(matches) != 1:
            raise CompareError(f"PACKAGE_FILE_COUNT:{name}:{len(matches)}")
        result[name] = matches[0]
    parents = {path.parent.resolve() for path in result.values()}
    if len(parents) != 1:
        raise CompareError("PACKAGE_FILES_NOT_IN_ONE_FOLDER")
    actual = {p.name for p in next(iter(parents)).iterdir() if p.is_file()}
    if actual != set(PACKAGE_NAMES):
        raise CompareError("PACKAGE_NOT_EXACTLY_SIX_FILES")
    return result


def count_csv_rows(path: Path) -> int:
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        reader = csv.reader(f)
        next(reader, None)
        return sum(1 for _ in reader)


def csv_header(path: Path) -> List[str]:
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        header = next(csv.reader(f), None)
    if not header:
        raise CompareError(f"EMPTY_CSV:{path.name}")
    if len(header) != len(set(header)):
        raise CompareError(f"DUPLICATE_HEADER:{path.name}")
    return header


def semantic_cell(value: str) -> str:
    text = (value or "").strip().replace("\r\n", "\n").replace("\r", "\n")
    if text.startswith(("{", "[")):
        try:
            return json.dumps(json.loads(text), ensure_ascii=False, sort_keys=True, separators=(",", ":"))
        except json.JSONDecodeError:
            pass
    return text


def semantic_table(path: Path, key_columns: Sequence[str]) -> Tuple[List[str], Dict[Tuple[str, ...], Dict[str, str]]]:
    header = csv_header(path)
    missing = [name for name in key_columns if name not in header]
    if missing:
        raise CompareError(f"SEMANTIC_KEY_MISSING:{path.name}:{','.join(missing)}")
    rows: Dict[Tuple[str, ...], Dict[str, str]] = {}
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        for raw in csv.DictReader(f):
            normalized = {name: semantic_cell(raw.get(name, "")) for name in header}
            key = tuple(normalized[name] for name in key_columns)
            if tuple(key_columns) == D4_KEY:
                key_invalid = not key[0] or not key[2]
            else:
                key_invalid = any(not part for part in key)
            if key_invalid or key in rows:
                raise CompareError(f"SEMANTIC_KEY_INVALID:{path.name}")
            rows[key] = normalized
    return header, rows


def compare_semantic_csv(a: Path, b: Path, key_columns: Sequence[str]) -> Tuple[bool, int, str]:
    ha, ra = semantic_table(a, key_columns)
    hb, rb = semantic_table(b, key_columns)
    if set(ha) != set(hb):
        return False, len(ra), "HEADER_SET_DIFFERENCE"
    if set(ra) != set(rb):
        return False, len(ra), "KEY_SET_DIFFERENCE"
    columns = sorted(set(ha))
    for key in sorted(ra):
        for column in columns:
            if ra[key].get(column, "") != rb[key].get(column, ""):
                return False, len(ra), f"VALUE_DIFFERENCE:{column}"
    return True, len(ra), "SEMANTIC_EQUAL"


def flatten_strings(value: Any) -> List[str]:
    result: List[str] = []
    if isinstance(value, str):
        result.append(value)
    elif isinstance(value, Mapping):
        for key, child in value.items():
            result.append(str(key))
            result.extend(flatten_strings(child))
    elif isinstance(value, list):
        for child in value:
            result.extend(flatten_strings(child))
    return result


def normalized_key(value: str) -> str:
    return re.sub(r"[^a-z0-9]", "", value.lower())


def find_key_values(value: Any, wanted: str) -> List[str]:
    target = normalized_key(wanted)
    result: List[str] = []
    if isinstance(value, Mapping):
        for key, child in value.items():
            if normalized_key(str(key)) == target and isinstance(child, (str, int, float, bool)):
                result.append(str(child))
            result.extend(find_key_values(child, wanted))
    elif isinstance(value, list):
        for child in value:
            result.extend(find_key_values(child, wanted))
    return result


def has_absolute_path(strings: Iterable[str]) -> bool:
    for value in strings:
        text = value.strip()
        if re.match(r"^[A-Za-z]:[\\/]", text):
            return True
        if text.startswith(("\\\\", "/Users/", "/home/", "/private/", "/var/")):
            return True
    return False


def load_manifest(path: Path, package: Mapping[str, Path]) -> dict:
    try:
        data = json.loads(path.read_text(encoding="utf-8-sig"))
    except (json.JSONDecodeError, UnicodeDecodeError) as exc:
        raise CompareError("D5_INVALID_JSON") from exc
    if not isinstance(data, dict):
        raise CompareError("D5_NOT_OBJECT")
    strings = flatten_strings(data)
    if has_absolute_path(strings):
        raise CompareError("D5_ABSOLUTE_PATH_LEAK")
    statuses = [value.upper() for value in find_key_values(data, "Status")]
    if "SUCCESS" not in statuses:
        raise CompareError("D5_SUCCESS_STATUS_MISSING")
    string_set = set(strings)
    for name in PACKAGE_NAMES:
        if name == "u001_run_manifest.json":
            continue
        if name not in string_set:
            raise CompareError(f"D5_OUTPUT_BASENAME_MISSING:{name}")
        if sha256_file(package[name]) not in string_set:
            raise CompareError(f"D5_OUTPUT_HASH_MISSING:{name}")
    return data


def unique_values(data: dict, key: str, required: bool) -> set[str]:
    values = {value for value in find_key_values(data, key) if value != ""}
    if required and len(values) != 1:
        raise CompareError(f"D5_IDENTITY_KEY:{key}:{len(values)}")
    return values


def compare_manifests(
    a_path: Path,
    b_path: Path,
    a_package: Mapping[str, Path],
    b_package: Mapping[str, Path],
) -> Tuple[bool, str]:
    a = load_manifest(a_path, a_package)
    b = load_manifest(b_path, b_package)
    for key in ("DatasetId", "SegmentId", "SourceId"):
        if unique_values(a, key, True) != unique_values(b, key, True):
            return False, f"STABLE_IDENTITY_DIFFERENCE:{key}"
    for key in STABLE_MANIFEST_KEYS[3:]:
        av = unique_values(a, key, False)
        bv = unique_values(b, key, False)
        if av or bv:
            if av != bv:
                return False, f"STABLE_METADATA_DIFFERENCE:{key}"
    if len(unique_values(a, "RunId", True)) != 1 or len(unique_values(b, "RunId", True)) != 1:
        return False, "RUN_ID_INVALID"
    return True, "INTERNAL_INTEGRITY_PASS_RUNTIME_DIFFERENCES_ALLOWED"


def parse_int(value: str) -> int:
    number = float(value)
    if not math.isfinite(number) or not number.is_integer():
        raise ValueError(value)
    return int(number)


def inspect_frame_view(path: Path) -> Tuple[int, int]:
    header = csv_header(path)
    if not header or header[0] != "FrameCount":
        raise CompareError("D2_FRAMECOUNT_FIRST")
    groups = [name[:-7] for name in header if name.endswith("__value")]
    previous = None
    rows = 0
    with path.open("r", encoding="utf-8-sig", newline="") as f:
        for row in csv.DictReader(f):
            try:
                frame = parse_int(row["FrameCount"])
            except (ValueError, TypeError) as exc:
                raise CompareError("D2_FRAMECOUNT_INVALID") from exc
            if previous is not None and frame <= previous:
                raise CompareError("D2_FRAMECOUNT_NOT_STRICT")
            previous = frame
            rows += 1
    if rows == 0 or not groups:
        raise CompareError("D2_EMPTY")
    return rows, len(groups)


def atomic_write_text(path: Path, text: str) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_name(path.name + ".tmp")
    temp.write_bytes(text.encode("utf-8"))
    os.replace(temp, path)


REPORT_COLUMNS = (
    "CheckId", "Artifact", "Policy", "Status", "ReferenceRows",
    "CandidateRows", "Detail",
)


def write_report(path: Path, rows: Sequence[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_name(path.name + ".tmp")
    with temp.open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=REPORT_COLUMNS)
        writer.writeheader()
        writer.writerows(rows)
    os.replace(temp, path)


def compare(reference_root: Path, candidate_root: Path, out: Path) -> str:
    if out.exists() and any(out.iterdir()):
        raise CompareError("OUTPUT_NOT_EMPTY")
    out.mkdir(parents=True, exist_ok=True)
    ref = find_package(reference_root)
    can = find_package(candidate_root)
    rows: List[dict] = []
    failures = 0

    def record(check: str, artifact: str, policy: str, passed: bool, rr: int, cr: int, detail: str) -> None:
        nonlocal failures
        if not passed:
            failures += 1
        rows.append({
            "CheckId": check, "Artifact": artifact, "Policy": policy,
            "Status": "PASS" if passed else "FAIL", "ReferenceRows": rr,
            "CandidateRows": cr, "Detail": detail,
        })

    d1 = PACKAGE_NAMES[0]
    d2 = PACKAGE_NAMES[1]
    d3 = PACKAGE_NAMES[2]
    d4 = PACKAGE_NAMES[3]
    d5 = PACKAGE_NAMES[4]
    d6 = PACKAGE_NAMES[5]

    u_ref, u_can = count_csv_rows(ref[d1]), count_csv_rows(can[d1])
    record("D1", d1, "EXACT_BYTES_SHA256", exact_equal(ref[d1], can[d1]), u_ref, u_can, "CORE_UPDATE_STREAM")

    f_ref, k_ref = inspect_frame_view(ref[d2])
    f_can, k_can = inspect_frame_view(can[d2])
    record("D2", d2, "EXACT_BYTES_SHA256", exact_equal(ref[d2], can[d2]), f_ref, f_can, "FRAME_VIEW")

    d3_equal, d3_rows, d3_detail = compare_semantic_csv(ref[d3], can[d3], D3_KEY)
    record("D3", d3, "SEMANTIC_KEY_CANONICAL_NAME", d3_equal, d3_rows, d3_rows, d3_detail)

    d4_equal, d4_rows, d4_detail = compare_semantic_csv(ref[d4], can[d4], D4_KEY)
    record("D4", d4, "SEMANTIC_KEY_SCOPE_NAME_METRIC", d4_equal, d4_rows, d4_rows, d4_detail)

    d5_equal, d5_detail = compare_manifests(ref[d5], can[d5], ref, can)
    record("D5", d5, "INTERNAL_INTEGRITY_RUNTIME_DIFFERENCES_ALLOWED", d5_equal, 1, 1, d5_detail)

    h_ref, h_can = count_csv_rows(ref[d6]), count_csv_rows(can[d6])
    record("D6", d6, "EXACT_BYTES_SHA256", exact_equal(ref[d6], can[d6]), h_ref, h_can, "HUMAN_REVIEW_VIEW")

    if k_ref != k_can:
        failures += 1
        rows.append({
            "CheckId": "K", "Artifact": d2, "Policy": "SIGNAL_GROUP_COUNT",
            "Status": "FAIL", "ReferenceRows": k_ref, "CandidateRows": k_can,
            "Detail": "FRAME_SIGNAL_COUNT_DIFFERENCE",
        })

    write_report(out / "u001_core_replay_comparison_report.csv", rows)
    flags = {row["CheckId"]: 1 if row["Status"] == "PASS" else 0 for row in rows}
    capsule = (
        "U001RV1\n"
        f"N-D1E{flags.get('D1', 0)}-D2E{flags.get('D2', 0)}-D3S{flags.get('D3', 0)}-D4S{flags.get('D4', 0)}-D5I{flags.get('D5', 0)}-D6E{flags.get('D6', 0)}-X{failures}\n"
        f"R-U{u_ref}-F{f_ref}-K{k_ref}\n"
        f"C-{0 if failures == 0 else 1}\n"
    )
    atomic_write_text(out / "u001_core_replay_comparison_capsule.txt", capsule)
    return capsule


def write_csv(path: Path, header: Sequence[str], rows: Sequence[Sequence[Any]]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(header)
        writer.writerows(rows)


def make_synthetic_package(root: Path, run_id: str, row_order_reversed: bool = False) -> Dict[str, Path]:
    package = root / "FINAL_PACKAGE"
    package.mkdir(parents=True)
    d1_header = ["SourceRowIndex", "SourceId", "FrameCount", "CanonicalName", "PhysicalValue"]
    d1_rows = [[1, "SRC", 10, "RADAR", 12.0], [2, "SRC", 10, "CAM", 10.0]]
    write_csv(package / PACKAGE_NAMES[0], d1_header, d1_rows)
    d2_header = [
        "FrameCount", "SourceRowFirst", "SourceRowLast", "SourceUpdateCount",
        "RADAR__value", "RADAR__label", "RADAR__updated", "RADAR__update_count",
        "RADAR__age", "RADAR__valid", "RADAR__missing_reason",
    ]
    d2_rows = [[10, 1, 2, 2, 12.0, "", 1, 1, 0, 1, "VALID_UPDATE"]]
    write_csv(package / PACKAGE_NAMES[1], d2_header, d2_rows)
    d3_header = ["CanonicalName", "CAN_ID", "Channel", "SignalName", "Unit"]
    d3_rows = [["RADAR", "0x380", "1", "DistanceToObject", "m"], ["FC", "0x620", "2", "FrameCount", "count"]]
    if row_order_reversed:
        d3_rows.reverse()
    write_csv(package / PACKAGE_NAMES[2], d3_header, d3_rows)
    d4_header = ["Scope", "CanonicalName", "Metric", "Value", "Unit", "SourceGate", "Status", "Notes"]
    d4_rows = [["GLOBAL", "", "package_file_count", "6", "files", "G7", "PASS", ""]]
    write_csv(package / PACKAGE_NAMES[3], d4_header, d4_rows)
    write_csv(package / PACKAGE_NAMES[5], ["FrameCount", "RADAR__value"], [[10, 12.0]])
    outputs = {}
    for name in (PACKAGE_NAMES[0], PACKAGE_NAMES[1], PACKAGE_NAMES[2], PACKAGE_NAMES[3], PACKAGE_NAMES[5]):
        path = package / name
        outputs[name] = {"Basename": name, "Size": path.stat().st_size, "SHA256": sha256_file(path)}
    manifest = {
        "ContractVersion": "0.1", "ImplementationProfile": "P", "ProcessingMode": "BATCH",
        "DatasetId": "D", "RunId": run_id, "SegmentId": "S", "SourceId": "SRC",
        "RunTimestamp": "VOLATILE_" + run_id, "Status": "SUCCESS", "Outputs": outputs,
        "IntermediateQualityHashes": {"volatile": run_id},
    }
    (package / PACKAGE_NAMES[4]).write_text(json.dumps(manifest, indent=2), encoding="utf-8")
    return {name: package / name for name in PACKAGE_NAMES}


def run_self_test() -> None:
    passed = 0
    with tempfile.TemporaryDirectory() as td:
        root = Path(td)
        ref_root, can_root = root / "A", root / "B"
        ref = make_synthetic_package(ref_root, "RUN_A", False)
        can = make_synthetic_package(can_root, "RUN_B", True)
        assert len(find_package(ref_root)) == 6
        passed += 1
        assert exact_equal(ref[PACKAGE_NAMES[0]], can[PACKAGE_NAMES[0]])
        passed += 1
        assert exact_equal(ref[PACKAGE_NAMES[1]], can[PACKAGE_NAMES[1]])
        passed += 1
        assert compare_semantic_csv(ref[PACKAGE_NAMES[2]], can[PACKAGE_NAMES[2]], D3_KEY)[0]
        passed += 1
        assert compare_semantic_csv(ref[PACKAGE_NAMES[3]], can[PACKAGE_NAMES[3]], D4_KEY)[0]
        passed += 1
        assert compare_manifests(ref[PACKAGE_NAMES[4]], can[PACKAGE_NAMES[4]], ref, can)[0]
        passed += 1
        assert exact_equal(ref[PACKAGE_NAMES[5]], can[PACKAGE_NAMES[5]])
        passed += 1
        assert inspect_frame_view(ref[PACKAGE_NAMES[1]]) == (1, 1)
        passed += 1
        out = root / "OUT"
        capsule = compare(ref_root, can_root, out)
        assert "D1E1-D2E1-D3S1-D4S1-D5I1-D6E1-X0" in capsule
        passed += 1
        assert capsule.endswith("C-0\n")
        passed += 1
        assert count_csv_rows(out / "u001_core_replay_comparison_report.csv") == 6
        passed += 1
        assert parse_int("10.0") == 10
        passed += 1
        assert semantic_cell('{"b":2,"a":1}') == '{"a":1,"b":2}'
        passed += 1
        assert not has_absolute_path(["CANDump.csv", "folder/file.csv"])
        passed += 1
        assert has_absolute_path([r"C:\\company\\file.csv"])
        passed += 1
        assert unique_values(json.loads(ref[PACKAGE_NAMES[4]].read_text()), "RunId", True) == {"RUN_A"}
        passed += 1
    if passed != 16:
        raise AssertionError(f"SELFTEST_COUNT:{passed}")
    print(SELFTEST_TOKEN)


def main(argv: Sequence[str] | None = None) -> int:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--compare", action="store_true")
    parser.add_argument("--reference-run", type=Path)
    parser.add_argument("--candidate-run", type=Path)
    parser.add_argument("--out", type=Path)
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            run_self_test()
            return 0
        if args.compare:
            if args.reference_run is None or args.candidate_run is None or args.out is None:
                raise CompareError("COMPARE_REQUIRES_REFERENCE_CANDIDATE_OUT")
            capsule = compare(args.reference_run.resolve(), args.candidate_run.resolve(), args.out.resolve())
            print(capsule, end="")
            return 0 if capsule.endswith("C-0\n") else 1
        parser.error("choose --self-test or --compare")
    except (CompareError, OSError, csv.Error, ValueError) as exc:
        print(f"U001RV1\nN-D1E0-D2E0-D3S0-D4S0-D5I0-D6E0-X1\nC-1\nMEMO-{type(exc).__name__}:{exc}")
        return 1
    return 2


if __name__ == "__main__":
    raise SystemExit(main())
