#!/usr/bin/env python3
"""Build versioned G2 sample/gate files after the frozen protocol exposed scalar-type drift.

Read-only with respect to accepted files: exact approved inputs are verified by
SHA-256 and two new files are created in 04_TEMP. Existing files are never
overwritten.
"""

from __future__ import annotations

import argparse
import hashlib
from pathlib import Path


BASE_RUNNER = "u002_g2_synthetic_vertical_run_v0_1.py.txt"
OUT_RUNNER = "u002_g2_synthetic_vertical_run_v0_2.py.txt"
BASE_GATE = "u002_executor_v08_independent_vertical_gate_v0_5.py.txt"
OUT_GATE = "u002_executor_v08_independent_vertical_gate_v0_6.py.txt"
BASE_RUNNER_SHA256 = "5c06dee03372d7445cda6348cea8adeebbd61e01175e177f28e7d451b10bcd82"
BASE_GATE_SHA256 = "238fb43a8d0f42352b9743c174daee007d442e8b357f4400134ca06a64636b2b"


class BuildError(RuntimeError):
    pass


def sha256_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def load_exact(path: Path, expected_sha256: str) -> str:
    if not path.is_file() or path.is_symlink():
        raise BuildError("INPUT_MISSING_" + path.name)
    data = path.read_bytes()
    if sha256_bytes(data) != expected_sha256:
        raise BuildError("INPUT_IDENTITY_" + path.name)
    return data.decode("utf-8-sig")


def replace_once(text: str, old: str, new: str, label: str) -> str:
    if text.count(old) != 1:
        raise BuildError("PATCH_SITE_" + label)
    return text.replace(old, new, 1)


def write_new(path: Path, text: str) -> None:
    if path.exists():
        raise BuildError("OUTPUT_EXISTS_" + path.name)
    path.write_bytes(text.replace("\r\n", "\n").replace("\r", "\n").encode("utf-8"))


def build(temp: Path) -> tuple[Path, Path]:
    runner = load_exact(temp / BASE_RUNNER, BASE_RUNNER_SHA256)
    runner = replace_once(runner, 'VERSION = "0.1"', 'VERSION = "0.2"', "RUNNER_VERSION")
    runner = replace_once(
        runner,
        '"Metadata": {{"Synthetic": True, "Sequence": SEQUENCE}},',
        '"Metadata": {{"Synthetic": [True], "Sequence": [SEQUENCE]}},',
        "ARTIFACT_METADATA_SCALAR_ARRAY",
    )
    runner = replace_once(
        runner,
        '"Metrics": {{"SyntheticRows": 1}},',
        '"Metrics": {{"SyntheticRows": [1]}},',
        "RESULT_METRICS_SCALAR_ARRAY",
    )
    runner_path = temp / OUT_RUNNER
    write_new(runner_path, runner)
    runner_sha = sha256_bytes(runner_path.read_bytes())

    gate = load_exact(temp / BASE_GATE, BASE_GATE_SHA256)
    gate = replace_once(gate, 'VERSION = "0.5"', 'VERSION = "0.6"', "GATE_VERSION")
    gate = replace_once(gate, BASE_RUNNER, OUT_RUNNER, "GATE_RUNNER_NAME")
    gate = replace_once(gate, BASE_RUNNER_SHA256, runner_sha, "GATE_RUNNER_HASH")
    gate = replace_once(gate, 'OUT_REL = Path("05_TESTS/02_ACCEPTANCE/U21G_G2_VERTICAL_V08")',
                        'OUT_REL = Path("05_TESTS/02_ACCEPTANCE/U21G_G2_VERTICAL_V08R2")',
                        "GATE_OUTPUT")
    gate = replace_once(gate, 'if getattr(module, "VERSION", None) != "0.1":',
                        'if getattr(module, "VERSION", None) != "0.2":',
                        "GATE_RUNNER_VERSION")
    gate = replace_once(gate, 'logs = root / "04_TEMP" / ".G2R4_V08_GATE_LOGS"',
                        'logs = root / "04_TEMP" / ".G2R4_V08R2_GATE_LOGS"',
                        "GATE_LOG_ROOT")
    gate = replace_once(gate, 'ck("version", VERSION == "0.5")',
                        'ck("version", VERSION == "0.6")', "GATE_SELFTEST_VERSION")
    gate = replace_once(gate, 'ck("out", OUT_REL.name.endswith("V08"))',
                        'ck("out", OUT_REL.name.endswith("V08R2"))', "GATE_SELFTEST_OUTPUT")
    gate_path = temp / OUT_GATE
    write_new(gate_path, gate)
    return runner_path, gate_path


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--workspace-root", required=True)
    args = parser.parse_args()
    try:
        temp = Path(args.workspace_root).resolve() / "04_TEMP"
        runner, gate = build(temp)
        print(f"G2R4FIX1 N-R1-G1-X0 C-0 H-{sha256_bytes(runner.read_bytes())[:8]}.{sha256_bytes(gate.read_bytes())[:8]}")
        return 0
    except Exception as exc:
        reason = "".join(ch if ch.isalnum() or ch == "_" else "_" for ch in str(exc).upper())[:80]
        print(f"G2R4FIX1 N-R0-G0-X1 C-7 K-{reason or exc.__class__.__name__.upper()}")
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
