#!/usr/bin/env python3
"""Prepare, test and install U001 H1 Input Adapter Foundation."""

from __future__ import annotations

import argparse
import json
import os
import shutil
import subprocess
import sys
import tempfile
import zipfile
from datetime import datetime
from pathlib import Path, PurePosixPath


PROGRAM_VERSION = "0.1.0"
PACKAGE_NAME = "UDP001_U1H1_INPUT_ADAPTER_FOUNDATION_v0_1.zip"
START_HERE_NAME = "U1H1_START_HERE_CN_v0_1_2026-08-28.txt"
ACTIVE_NAME = "U001H_ACTIVE"
PREPARED_NAME = "U1H1_PREPARED"
EXPECTED_FILES = {
    "01_PROGRAMS/adapters/Foundation.Test.SyntheticInputAdapter/0.1.0/adapter.py",
    "01_PROGRAMS/adapters/Foundation.Test.SyntheticInputAdapter/0.1.0/adapter_card.json",
    "01_PROGRAMS/core/u001_input_adapter_conformance_v0_1.py",
    "01_PROGRAMS/core/u001_input_adapter_registry_v0_1.py",
    "02_CONTROL/contracts/u001_canonical_source_update_schema_v0_1.json",
    "02_CONTROL/contracts/u001_input_adapter_card_schema_v0_1.json",
    "06_DOCS/U001_HIGH_CODE_REWRITE_ARCHITECTURE_v1_0_2026-08-28.txt",
}


class InstallError(RuntimeError):
    """Fail-closed installation error."""


def timestamp() -> str:
    return datetime.now().strftime("%Y%m%dT%H%M%S_%f")


def resolve_u001_root(explicit: Path | None) -> Path:
    candidates = []
    if explicit is not None:
        candidates.append(explicit)
    candidates.extend((Path.cwd() / "U-UDP001", Path.cwd()))
    for candidate in candidates:
        candidate = candidate.resolve()
        if candidate.name == "U-UDP001" and candidate.is_dir():
            return candidate
    raise InstallError("U001_ROOT_NOT_FOUND")


def transfer_dir(u001_root: Path) -> Path:
    path = u001_root / "04_TEMP"
    if not path.is_dir():
        raise InstallError("TEMP_DIRECTORY_MISSING")
    return path


def package_path(u001_root: Path) -> Path:
    path = transfer_dir(u001_root) / PACKAGE_NAME
    if not path.is_file():
        raise InstallError(f"PACKAGE_MISSING:{PACKAGE_NAME}")
    return path


def freeze_legacy_temp_once(u001_root: Path) -> int:
    """Archive pre-high-code TEMP debt once, preserving the current transfer."""
    marker = u001_root / "02_CONTROL" / "migrations" / "u001_high_code_temp_freeze_v0_1.json"
    if marker.is_file():
        return 0

    temp = transfer_dir(u001_root)
    preserved = {
        PACKAGE_NAME,
        Path(__file__).name,
        START_HERE_NAME,
        ACTIVE_NAME,
    }
    candidates = [child for child in temp.iterdir() if child.name not in preserved]
    archive_root = u001_root / "A-Archive" / f"U001_LEGACY_TEMP_FREEZE_{timestamp()}"
    moved_names: list[str] = []
    for child in sorted(candidates, key=lambda item: item.name.lower()):
        archive_root.mkdir(parents=True, exist_ok=True)
        shutil.move(str(child), str(archive_root / child.name))
        moved_names.append(child.name)

    marker.parent.mkdir(parents=True, exist_ok=True)
    marker.write_text(
        json.dumps(
            {
                "Migration": "U001_HIGH_CODE_TEMP_FREEZE_V0_1",
                "ArchivedEntryCount": len(moved_names),
                "ArchivedBasenames": moved_names,
                "SourceFilesDeleted": False,
                "CompletionCode": 0,
            },
            indent=2,
        )
        + "\n",
        encoding="utf-8",
    )
    return len(moved_names)


def validate_zip(path: Path) -> None:
    try:
        with zipfile.ZipFile(path, "r") as archive:
            file_names = {name for name in archive.namelist() if not name.endswith("/")}
            if file_names != EXPECTED_FILES:
                missing = sorted(EXPECTED_FILES - file_names)
                extra = sorted(file_names - EXPECTED_FILES)
                raise InstallError(f"PACKAGE_INVENTORY:MISSING={missing}:EXTRA={extra}")
            for name in file_names:
                pure = PurePosixPath(name)
                if pure.is_absolute() or ".." in pure.parts:
                    raise InstallError(f"UNSAFE_PACKAGE_PATH:{name}")
            bad_member = archive.testzip()
            if bad_member is not None:
                raise InstallError(f"PACKAGE_CRC:{bad_member}")
    except zipfile.BadZipFile as exc:
        raise InstallError("PACKAGE_ZIP") from exc


def archive_path(path: Path, archive_root: Path, relative_path: Path) -> None:
    if not path.exists():
        return
    destination = archive_root / relative_path
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.move(str(path), str(destination))


def prepare(u001_root: Path) -> Path:
    temp = transfer_dir(u001_root)
    source_zip = package_path(u001_root)
    print(f"[CHECK] U001 root exists: {u001_root}")
    print(f"[WHY]   Installation is valid only inside the company U-UDP001 layout.")
    print(f"[CHECK] Transfer package exists: {source_zip.name}")
    print(f"[WHY]   The package is the immutable H1 payload being tested and installed.")
    validate_zip(source_zip)
    print(f"[CHECK] Package inventory and CRC passed: {len(EXPECTED_FILES)} files")
    print(f"[WHY]   Exact inventory prevents stale or unapproved files from entering the product.")

    legacy_archived_count = freeze_legacy_temp_once(u001_root)
    if legacy_archived_count:
        print(f"[CLEAN] Legacy TEMP entries archived without deletion: {legacy_archived_count}")
        print("[WHY]   High-Code work must not resolve dependencies from superseded Gate debris.")
    else:
        print("[CLEAN] One-time Legacy TEMP freeze was already completed.")

    active = temp / ACTIVE_NAME
    active.mkdir(parents=True, exist_ok=True)
    prepared = active / PREPARED_NAME
    if prepared.exists():
        archive_root = u001_root / "A-Archive" / f"U1H1_PREPARED_{timestamp()}"
        archive_root.parent.mkdir(parents=True, exist_ok=True)
        shutil.move(str(prepared), str(archive_root))
        print(f"[CLEAN] Previous H1 prepared directory archived: {archive_root.name}")
    prepared.mkdir(parents=True)
    with zipfile.ZipFile(source_zip, "r") as archive:
        archive.extractall(prepared)
    extracted = {
        path.relative_to(prepared).as_posix()
        for path in prepared.rglob("*")
        if path.is_file()
    }
    if extracted != EXPECTED_FILES:
        raise InstallError("EXTRACTED_INVENTORY")
    print(f"[CHECK] Prepared payload matches the approved inventory.")
    print(f"U1H1P-N{len(EXPECTED_FILES)}-A1-L{legacy_archived_count}-X0-C0")
    return prepared


def child_environment() -> dict[str, str]:
    environment = dict(os.environ)
    environment["PYTHONDONTWRITEBYTECODE"] = "1"
    environment["PYTHONPYCACHEPREFIX"] = str(Path(tempfile.gettempdir()) / "u001_h1_pycache")
    return environment


def run_child(command: list[str], label: str, timeout_seconds: int = 120) -> str:
    try:
        result = subprocess.run(
            command,
            text=True,
            capture_output=True,
            timeout=timeout_seconds,
            check=False,
            env=child_environment(),
        )
    except subprocess.TimeoutExpired as exc:
        raise InstallError(f"{label}_TIMEOUT") from exc
    if result.returncode != 0:
        detail = (result.stderr or result.stdout).strip().replace("\n", " | ")[:800]
        raise InstallError(f"{label}_EXIT:{result.returncode}:{detail}")
    output = result.stdout.strip()
    if not output:
        raise InstallError(f"{label}_NO_OUTPUT")
    return output


def self_test(u001_root: Path) -> None:
    prepared = transfer_dir(u001_root) / ACTIVE_NAME / PREPARED_NAME
    if not prepared.is_dir():
        raise InstallError("PREPARED_DIRECTORY_MISSING")
    core = prepared / "01_PROGRAMS" / "core"
    adapter = (
        prepared
        / "01_PROGRAMS"
        / "adapters"
        / "Foundation.Test.SyntheticInputAdapter"
        / "0.1.0"
        / "adapter.py"
    )
    registry = core / "u001_input_adapter_registry_v0_1.py"
    conformance = core / "u001_input_adapter_conformance_v0_1.py"
    required = (registry, conformance, adapter)
    if not all(path.is_file() for path in required):
        raise InstallError("SELFTEST_PROGRAM_MISSING")

    print("[CHECK] Registry self-test: card identity, versioning, package layout and fail-closed cases.")
    registry_output = run_child([sys.executable, str(registry), "--self-test"], "REGISTRY_SELFTEST")
    if "PASS-18" not in registry_output:
        raise InstallError("REGISTRY_SELFTEST_CAPSULE")
    print("[CHECK] Synthetic Adapter self-test: describe, preflight, run and canonical source artifacts.")
    adapter_output = run_child([sys.executable, str(adapter), "--self-test"], "ADAPTER_SELFTEST")
    if "PASS-16" not in adapter_output:
        raise InstallError("ADAPTER_SELFTEST_CAPSULE")
    print("[CHECK] Independent conformance self-test: subprocess isolation, hashes and manifest safety.")
    gate_output = run_child([sys.executable, str(conformance), "--self-test"], "CONFORMANCE_SELFTEST")
    if "PASS-20" not in gate_output:
        raise InstallError("CONFORMANCE_SELFTEST_CAPSULE")
    print("U1H1ST-N3-T54-X0-C0")


def copy_payload(prepared: Path, u001_root: Path) -> tuple[int, Path | None]:
    archive_root: Path | None = None
    copied = 0
    for relative_text in sorted(EXPECTED_FILES):
        relative_path = Path(relative_text)
        source = prepared / relative_path
        target = u001_root / relative_path
        if target.exists():
            if archive_root is None:
                archive_root = u001_root / "A-Archive" / f"U1H1_REPLACED_{timestamp()}"
            archive_path(target, archive_root, relative_path)
        target.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(source, target)
        copied += 1
    return copied, archive_root


def execute(u001_root: Path) -> None:
    self_test(u001_root)
    prepared = transfer_dir(u001_root) / ACTIVE_NAME / PREPARED_NAME
    copied, archive_root = copy_payload(prepared, u001_root)

    registry = u001_root / "01_PROGRAMS" / "core" / "u001_input_adapter_registry_v0_1.py"
    conformance = u001_root / "01_PROGRAMS" / "core" / "u001_input_adapter_conformance_v0_1.py"
    adapters_root = u001_root / "01_PROGRAMS" / "adapters"
    registry_output = u001_root / "02_CONTROL" / "registry" / "u001_input_adapter_registry_snapshot_v0_1.json"
    registry_output.parent.mkdir(parents=True, exist_ok=True)
    scan_stdout = run_child(
        [
            sys.executable,
            str(registry),
            "--scan",
            "--adapter-root",
            str(adapters_root),
            "--output",
            str(registry_output),
        ],
        "INSTALLED_REGISTRY_SCAN",
    )
    if "C0" not in scan_stdout.replace("-", ""):
        raise InstallError("INSTALLED_REGISTRY_CAPSULE")

    acceptance_root = u001_root / "05_TESTS" / f"U1H1_ACCEPTANCE_{timestamp()}"
    adapter_dir = adapters_root / "Foundation.Test.SyntheticInputAdapter" / "0.1.0"
    qualify_stdout = run_child(
        [
            sys.executable,
            str(conformance),
            "--qualify",
            "--adapter-dir",
            str(adapter_dir),
            "--output",
            str(acceptance_root),
        ],
        "INSTALLED_ADAPTER_QUALIFICATION",
    )
    if "C0" not in qualify_stdout.replace("-", ""):
        raise InstallError("INSTALLED_QUALIFICATION_CAPSULE")

    result = {
        "Node": "U1H1",
        "InstalledFileCount": copied,
        "AdapterCount": json.loads(registry_output.read_text(encoding="utf-8"))["AdapterCount"],
        "AdapterFamilyCount": json.loads(registry_output.read_text(encoding="utf-8"))["AdapterFamilyCount"],
        "QualificationReceipt": acceptance_root.name,
        "ReplacedFilesArchived": archive_root is not None,
        "CompletionCode": 0,
    }
    result_path = transfer_dir(u001_root) / "U1H1_EXECUTION_RESULT.json"
    result_path.write_text(json.dumps(result, indent=2) + "\n", encoding="utf-8")
    print(
        "U1H1-"
        f"N{result['AdapterCount']}-"
        f"F{result['AdapterFamilyCount']}-"
        "Q1-"
        f"A{copied}-"
        "X0-C0"
    )


def parser() -> argparse.ArgumentParser:
    result = argparse.ArgumentParser(description="Install U001 H1 Input Adapter Foundation")
    action = result.add_mutually_exclusive_group(required=True)
    action.add_argument("--prepare", action="store_true")
    action.add_argument("--self-test", action="store_true")
    action.add_argument("--execute", action="store_true")
    result.add_argument("--u001-root", type=Path)
    return result


def main(argv: list[str] | None = None) -> int:
    args = parser().parse_args(argv)
    try:
        root = resolve_u001_root(args.u001_root)
        if args.prepare:
            prepare(root)
        elif args.self_test:
            self_test(root)
        else:
            execute(root)
        return 0
    except (InstallError, OSError, ValueError, json.JSONDecodeError) as exc:
        print(f"U1H1E-K{type(exc).__name__.upper()}-D{str(exc)}-X1-C7", file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
