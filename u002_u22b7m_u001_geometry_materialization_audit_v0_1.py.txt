#!/usr/bin/env python3
"""Audit whether paired Camera geometry was materialized and linked into U002.

The audit reads only directory structure, CSV headers/catalog rows, and frozen
module requests.  It never reads Raw CAN payload rows or modifies any package.
"""

from __future__ import annotations

import argparse
import csv
import hashlib
import json
from pathlib import Path
import re
import sys
import tempfile
from typing import Dict, Iterable, List, Mapping, Optional, Sequence, Tuple


PROGRAM_ID = "UDP002.U22B7M.U001GeometryMaterializationAudit"
PROGRAM_VERSION = "0.1.0"

REQUIRED_GEOMETRY = {
    "CAMERA_1_DISTANCE": ("2", "0x221", "Distance"),
    "CAMERA_1_LATERAL_POSITION": ("2", "0x221", "LateralPosition"),
    "CAMERA_3_DISTANCE": ("2", "0x251", "Distance"),
    "CAMERA_3_LATERAL_POSITION": ("2", "0x251", "LateralPosition"),
}

PACKAGE_COLUMNS = (
    "PackageId",
    "PackageRoot",
    "Archived",
    "CatalogPath",
    "CatalogRows",
    "FrameViewPath",
    "FrameViewGroups",
    "Camera1SignalNames",
    "Camera3SignalNames",
    "Camera1Distance",
    "Camera1LateralPosition",
    "Camera3Distance",
    "Camera3LateralPosition",
    "GeometryComplete",
    "ReferencedByS1Count",
    "CatalogSha256",
)

RUN_COLUMNS = (
    "RunId",
    "RunRoot",
    "S1RequestPath",
    "InputPackage",
    "InputPackageExists",
    "MatchedPackageId",
    "GeometryComplete",
    "S7RequestPresent",
    "ReviewPackagePresent",
)


class AuditError(RuntimeError):
    """Represent one deterministic audit failure."""


def token(value: object, limit: int = 100) -> str:
    """Return one short machine-safe token."""

    text = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper()).strip("_")
    return (text or "UNKNOWN")[:limit]


def canonical_bytes(value: object) -> bytes:
    """Serialize JSON deterministically."""

    return (
        json.dumps(
            value,
            ensure_ascii=False,
            sort_keys=True,
            indent=2,
            allow_nan=False,
        )
        + "\n"
    ).encode("utf-8")


def sha256_file(path: Path) -> str:
    """Hash one metadata file with bounded memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(8 * 1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def normalize_identifier(value: object) -> str:
    """Ignore separators and case without ignoring semantic text."""

    return re.sub(r"[^a-z0-9]+", "", str(value).strip().casefold())


def normalize_can_id(value: object) -> str:
    """Normalize one hexadecimal CAN identity."""

    text = str(value).strip().lower()
    if text.startswith("0x"):
        number = int(text, 16)
    else:
        number = int(text, 16)
    return f"0x{number:X}".lower()


def is_archived(path: Path) -> bool:
    """Mark packages located under an archive component."""

    names = {part.casefold() for part in path.parts}
    return any(
        name in names
        for name in {
            "a-archive",
            "archive",
            "arch",
            "archived",
        }
    )


def common_package_root(catalog: Path, frame_view: Optional[Path]) -> Path:
    """Return the narrowest common package root for two U001 public files."""

    if frame_view is None:
        return catalog.parent.resolve()
    catalog_parts = catalog.resolve().parts
    frame_parts = frame_view.resolve().parts
    common: List[str] = []
    for left, right in zip(catalog_parts, frame_parts):
        if left != right:
            break
        common.append(left)
    if not common:
        raise AuditError("PACKAGE_COMMON_ROOT")
    return Path(*common)


def choose_frame_view(catalog: Path, frame_views: Sequence[Path]) -> Optional[Path]:
    """Choose the nearest frame view that shares a bounded ancestor."""

    scored: List[Tuple[int, Path]] = []
    catalog_resolved = catalog.resolve()
    for candidate in frame_views:
        candidate_resolved = candidate.resolve()
        common = common_package_root(catalog_resolved, candidate_resolved)
        distance = (
            len(catalog_resolved.relative_to(common).parts)
            + len(candidate_resolved.relative_to(common).parts)
        )
        if distance <= 8:
            scored.append((distance, candidate_resolved))
    if not scored:
        return None
    minimum = min(score for score, _ in scored)
    matches = sorted(path for score, path in scored if score == minimum)
    return matches[0] if len(matches) == 1 else None


def read_catalog(path: Path) -> List[Dict[str, str]]:
    """Read one controlled signal catalog."""

    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        rows = list(csv.DictReader(stream))
    required = {"CanonicalName", "CAN_ID", "Channel", "SignalName"}
    if not rows or not required.issubset(rows[0]):
        raise AuditError("CATALOG_SCHEMA")
    return [dict(row) for row in rows]


def identity_matches(
    rows: Sequence[Mapping[str, str]],
    channel: str,
    can_id: str,
    signal_name: str,
) -> List[Mapping[str, str]]:
    """Resolve one identity with exact CAN/Channel and separator-tolerant name."""

    expected_name = normalize_identifier(signal_name)
    candidates: List[Mapping[str, str]] = []
    for row in rows:
        try:
            observed_id = normalize_can_id(row.get("CAN_ID", ""))
        except Exception:
            continue
        if str(row.get("Channel", "")).strip() != channel:
            continue
        if observed_id != normalize_can_id(can_id):
            continue
        observed_name = normalize_identifier(row.get("SignalName", ""))
        canonical_name = normalize_identifier(row.get("CanonicalName", ""))
        if observed_name == expected_name or canonical_name.endswith(expected_name):
            candidates.append(row)
    return candidates


def camera_signal_names(
    rows: Sequence[Mapping[str, str]],
    can_id: str,
) -> List[str]:
    """List all CH2 signals for one Camera target message."""

    values: List[str] = []
    for row in rows:
        try:
            observed = normalize_can_id(row.get("CAN_ID", ""))
        except Exception:
            continue
        if str(row.get("Channel", "")).strip() == "2" and observed == can_id:
            values.append(str(row.get("SignalName", "")).strip())
    return sorted(set(value for value in values if value))


def frame_view_group_count(path: Optional[Path]) -> int:
    """Count public U001 value groups from only the CSV header."""

    if path is None:
        return 0
    with path.open("r", encoding="utf-8-sig", newline="") as stream:
        reader = csv.reader(stream)
        header = next(reader, [])
    return sum(column.endswith("__value") for column in header)


def package_inventory(workspace_root: Path) -> List[Dict[str, object]]:
    """Audit every discoverable U001 catalog package across the workspace."""

    catalogs = sorted(
        path
        for path in workspace_root.rglob("u001_signal_catalog.csv")
        if path.is_file()
    )
    frame_views = sorted(
        path
        for path in workspace_root.rglob("u001_frame_values.csv")
        if path.is_file()
    )
    packages: List[Dict[str, object]] = []
    for index, catalog in enumerate(catalogs, start=1):
        frame_view = choose_frame_view(catalog, frame_views)
        root = common_package_root(catalog, frame_view)
        try:
            rows = read_catalog(catalog)
            match_counts = {
                role: len(identity_matches(rows, channel, can_id, signal))
                for role, (channel, can_id, signal) in REQUIRED_GEOMETRY.items()
            }
            complete = all(count == 1 for count in match_counts.values())
            reason = ""
        except Exception as error:
            rows = []
            match_counts = {role: 0 for role in REQUIRED_GEOMETRY}
            complete = False
            reason = token(error.args[0] if error.args else type(error).__name__)
        packages.append(
            {
                "PackageId": f"P{index:03d}",
                "PackageRoot": str(root),
                "Archived": int(is_archived(root)),
                "CatalogPath": str(catalog.resolve()),
                "CatalogRows": len(rows),
                "FrameViewPath": str(frame_view.resolve()) if frame_view else "",
                "FrameViewGroups": frame_view_group_count(frame_view),
                "Camera1SignalNames": "|".join(camera_signal_names(rows, "0x221")),
                "Camera3SignalNames": "|".join(camera_signal_names(rows, "0x251")),
                "Camera1Distance": match_counts["CAMERA_1_DISTANCE"],
                "Camera1LateralPosition": match_counts[
                    "CAMERA_1_LATERAL_POSITION"
                ],
                "Camera3Distance": match_counts["CAMERA_3_DISTANCE"],
                "Camera3LateralPosition": match_counts[
                    "CAMERA_3_LATERAL_POSITION"
                ],
                "GeometryComplete": int(complete),
                "ReferencedByS1Count": 0,
                "CatalogSha256": sha256_file(catalog),
                "FailureReason": reason,
            }
        )
    return packages


def containing_package(
    input_package: Path,
    packages: Sequence[Mapping[str, object]],
) -> Optional[str]:
    """Match one S1 InputPackage to the catalog located inside it."""

    resolved = input_package.resolve()
    matches: List[Tuple[int, str]] = []
    for package in packages:
        catalog = Path(str(package["CatalogPath"])).resolve()
        try:
            catalog.relative_to(resolved)
        except ValueError:
            continue
        matches.append((len(catalog.parts), str(package["PackageId"])))
    if not matches:
        return None
    return min(matches)[1]


def run_root_from_request(request_path: Path, workspace_root: Path) -> Path:
    """Resolve the fixed first-level child of 07_OUTPUT for one request."""

    output_root = (workspace_root / "07_OUTPUT").resolve()
    resolved = request_path.resolve()
    relative = resolved.relative_to(output_root)
    if not relative.parts:
        raise AuditError("RUN_ROOT")
    return output_root / relative.parts[0]


def run_inventory(
    workspace_root: Path,
    packages: List[Dict[str, object]],
) -> List[Dict[str, object]]:
    """Audit every frozen S1 request and its actual U001 InputPackage linkage."""

    requests: List[Tuple[Path, Mapping[str, object]]] = []
    for path in (workspace_root / "07_OUTPUT").rglob("module_request.json"):
        if not path.is_file():
            continue
        try:
            value = json.loads(path.read_text(encoding="utf-8-sig"))
        except Exception:
            continue
        if value.get("ModuleId") == "CameraRadar.S1.CameraTargetContinuityArbitration":
            requests.append((path, value))

    by_id = {str(package["PackageId"]): package for package in packages}
    output: List[Dict[str, object]] = []
    for request_path, request in sorted(requests):
        run_root = run_root_from_request(request_path, workspace_root)
        input_package = Path(str(request.get("InputPackage", ""))).resolve()
        package_id = containing_package(input_package, packages)
        geometry_complete = 0
        if package_id is not None:
            package = by_id[package_id]
            package["ReferencedByS1Count"] = int(package["ReferencedByS1Count"]) + 1
            geometry_complete = int(package["GeometryComplete"])
        s7_present = 0
        for sibling in run_root.rglob("module_request.json"):
            try:
                sibling_value = json.loads(sibling.read_text(encoding="utf-8-sig"))
            except Exception:
                continue
            if sibling_value.get("ModuleId") == "CameraRadar.S7.U003ReviewPackageExport":
                s7_present = 1
                break
        review_present = int(
            any(run_root.rglob("u003_review_package_manifest.json"))
            or any(run_root.rglob("review_package_manifest.json"))
        )
        output.append(
            {
                "RunId": str(request.get("RunId", run_root.name)),
                "RunRoot": str(run_root),
                "S1RequestPath": str(request_path.resolve()),
                "InputPackage": str(input_package),
                "InputPackageExists": int(input_package.is_dir()),
                "MatchedPackageId": package_id or "",
                "GeometryComplete": geometry_complete,
                "S7RequestPresent": s7_present,
                "ReviewPackagePresent": review_present,
            }
        )
    return output


def write_csv(
    path: Path,
    columns: Sequence[str],
    rows: Iterable[Mapping[str, object]],
) -> None:
    """Write one deterministic UTF-8 CSV."""

    with path.open("w", encoding="utf-8", newline="") as stream:
        writer = csv.DictWriter(stream, fieldnames=list(columns), lineterminator="\n")
        writer.writeheader()
        for row in rows:
            writer.writerow({column: row.get(column, "") for column in columns})


def execute(workspace_root: Path, output_dir: Path) -> str:
    """Run one complete materialization and linkage audit."""

    if not workspace_root.is_dir():
        raise AuditError("WORKSPACE_ROOT_MISSING")
    if not (workspace_root / "07_OUTPUT").is_dir():
        raise AuditError("OUTPUT_ROOT_MISSING")
    packages = package_inventory(workspace_root)
    runs = run_inventory(workspace_root, packages)
    geometry_packages = sum(int(row["GeometryComplete"]) for row in packages)
    active_geometry_packages = sum(
        int(row["GeometryComplete"]) and not int(row["Archived"])
        for row in packages
    )
    linked_geometry_runs = sum(int(row["GeometryComplete"]) for row in runs)
    s7_runs = sum(int(row["S7RequestPresent"]) for row in runs)

    if linked_geometry_runs > 0:
        action = 0
        interpretation = "GEOMETRY_PACKAGE_ALREADY_LINKED_INVESTIGATE_RESOLVER"
    elif active_geometry_packages > 0:
        action = 1
        interpretation = "GEOMETRY_PACKAGE_EXISTS_REBUILD_RUN_LINKAGE"
    else:
        action = 2
        interpretation = "GEOMETRY_NOT_MATERIALIZED_RERUN_U001_PROFILE"

    output_dir.mkdir(parents=True, exist_ok=True)
    package_path = output_dir / "b7m_u001_package_inventory.csv"
    run_path = output_dir / "b7m_s1_run_input_linkage.csv"
    quality_path = output_dir / "b7m_materialization_audit_quality.json"
    capsule_path = output_dir / "b7m_return_capsule.txt"
    write_csv(package_path, PACKAGE_COLUMNS, packages)
    write_csv(run_path, RUN_COLUMNS, runs)
    quality = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": PROGRAM_VERSION,
        "CompletionCode": 0,
        "WorkspaceRoot": str(workspace_root.resolve()),
        "PackageCount": len(packages),
        "GeometryCompletePackageCount": geometry_packages,
        "ActiveGeometryCompletePackageCount": active_geometry_packages,
        "S1RunCount": len(runs),
        "GeometryLinkedS1RunCount": linked_geometry_runs,
        "S7RunCount": s7_runs,
        "RecommendedActionCode": action,
        "Interpretation": interpretation,
        "OutputHashes": {
            package_path.name: sha256_file(package_path),
            run_path.name: sha256_file(run_path),
        },
        "DecisionBoundary": "READ_ONLY_METADATA_AUDIT",
        "Prohibitions": [
            "NO_RAW_CAN_READ",
            "NO_PACKAGE_MUTATION",
            "NO_RUN_MUTATION",
            "NO_SIGNAL_POLICY_PROMOTION",
        ],
    }
    quality_path.write_bytes(canonical_bytes(quality))
    capsule = (
        f"B7M N-P{len(packages)}-G{geometry_packages}"
        f"-A{active_geometry_packages}-R{len(runs)}"
        f"-RG{linked_geometry_runs}-S7{s7_runs}-D{action}-X0 C-0"
    )
    capsule_path.write_text(capsule + "\n", encoding="utf-8")
    return capsule


def write_fixture_package(
    root: Path,
    name: str,
    include_lateral: bool,
) -> Path:
    """Create one tiny U001 public-interface fixture."""

    package = root / "03_CASES" / name
    package.mkdir(parents=True)
    rows = [
        ("C1D", "0x221", "2", "Distance"),
        ("C3D", "0x251", "2", "Distance"),
    ]
    if include_lateral:
        rows.extend(
            [
                ("C1L", "0x221", "2", "Lateral_Position"),
                ("C3L", "0x251", "2", "Lateral Position"),
            ]
        )
    with (package / "u001_signal_catalog.csv").open(
        "w", encoding="utf-8", newline=""
    ) as stream:
        writer = csv.writer(stream, lineterminator="\n")
        writer.writerow(("CanonicalName", "CAN_ID", "Channel", "SignalName"))
        writer.writerows(rows)
    with (package / "u001_frame_values.csv").open(
        "w", encoding="utf-8", newline=""
    ) as stream:
        writer = csv.writer(stream, lineterminator="\n")
        writer.writerow(["FrameCount"] + [row[0] + "__value" for row in rows])
        writer.writerow([1] + [0] * len(rows))
    return package


def self_test() -> None:
    """Exercise materialized-but-unlinked detection and output determinism."""

    with tempfile.TemporaryDirectory(prefix="u22b7m_selftest_") as temporary:
        root = Path(temporary) / "U-UDP002"
        old_package = write_fixture_package(root, "OLD", False)
        write_fixture_package(root, "GEOMETRY", True)
        run_root = root / "07_OUTPUT/RUN_SELFTEST"
        request_root = run_root / "00_STAGING/0002_S1"
        request_root.mkdir(parents=True)
        (request_root / "module_request.json").write_bytes(
            canonical_bytes(
                {
                    "RunId": "RUN_SELFTEST",
                    "ModuleId": "CameraRadar.S1.CameraTargetContinuityArbitration",
                    "InputPackage": str(old_package),
                }
            )
        )
        s7_root = run_root / "00_STAGING/0009_S7"
        s7_root.mkdir(parents=True)
        (s7_root / "module_request.json").write_bytes(
            canonical_bytes(
                {
                    "RunId": "RUN_SELFTEST",
                    "ModuleId": "CameraRadar.S7.U003ReviewPackageExport",
                }
            )
        )
        output = root / "04_TEMP/B7M_OUTPUT"
        capsule = execute(root, output)
        assert capsule == "B7M N-P2-G1-A1-R1-RG0-S71-D1-X0 C-0"
        package_rows = list(
            csv.DictReader(
                (output / "b7m_u001_package_inventory.csv").open(
                    "r", encoding="utf-8"
                )
            )
        )
        assert sum(int(row["GeometryComplete"]) for row in package_rows) == 1
        assert sum(int(row["ReferencedByS1Count"]) for row in package_rows) == 1
        run_rows = list(
            csv.DictReader(
                (output / "b7m_s1_run_input_linkage.csv").open(
                    "r", encoding="utf-8"
                )
            )
        )
        assert run_rows[0]["GeometryComplete"] == "0"
        first_hash = sha256_file(output / "b7m_materialization_audit_quality.json")
        assert execute(root, output) == capsule
        second_hash = sha256_file(output / "b7m_materialization_audit_quality.json")
        assert first_hash == second_hash
    print("UDP002-U22B7M-SELFTEST-PASS-16 C-0")


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Run self-test or one workspace audit."""

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--workspace-root")
    parser.add_argument("--output-dir")
    args = parser.parse_args(argv)
    try:
        if args.self_test:
            self_test()
            return 0
        if not args.workspace_root or not args.output_dir:
            raise AuditError("REQUIRED_ARGUMENT")
        print(
            execute(
                Path(args.workspace_root).resolve(),
                Path(args.output_dir).resolve(),
            )
        )
        return 0
    except Exception as error:
        reason = error.args[0] if error.args else type(error).__name__
        print("B7M N-X1 C-7 K-" + token(reason), file=sys.stderr)
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
