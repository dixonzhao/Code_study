#!/usr/bin/env python3
"""Read-only dependency classification for unexpected U001 active programs."""

from __future__ import annotations

import argparse
import ast
import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path


def classify(name: str, expected_names: set[str]) -> str:
    lower = name.lower()
    if any(token in lower for token in ("package", "portable", "launcher", "runtime_host")):
        return "PACKAGING"
    if any(token in lower for token in ("comparator", "validator", "replay", "probe", "diagnostic")):
        return "VALIDATION"
    family = lower.rsplit("_v", 1)[0]
    if any(item.lower().rsplit("_v", 1)[0] == family for item in expected_names):
        return "DEPRECATED_VERSION"
    return "OTHER"


def imported_modules(source: str) -> set[str]:
    result = set()
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return result
    for node in ast.walk(tree):
        if isinstance(node, ast.Import):
            result.update(alias.name.split(".")[0] for alias in node.names)
        elif isinstance(node, ast.ImportFrom) and node.module:
            result.add(node.module.split(".")[0])
    return result


def short_hash(text: str) -> str:
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:6].upper()


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--audit-report")
    parser.add_argument("--udp001-root")
    parser.add_argument("--out")
    args = parser.parse_args()
    if args.self_test:
        assert classify("u001_probe_v0_1.py", set()) == "VALIDATION"
        assert classify("u001_portable_host_v0_1.py", set()) == "PACKAGING"
        assert classify("u001_gui_v0_1.py", {"u001_gui_v0_2.py"}) == "DEPRECATED_VERSION"
        print("U001-G10D1D1-SELFTEST-PASS-6")
        return 0
    if not args.audit_report or not args.udp001_root or not args.out:
        parser.error("requires --audit-report --udp001-root --out")

    audit_path = Path(args.audit_report).resolve()
    root = Path(args.udp001_root).resolve()
    out = Path(args.out).resolve()
    if not audit_path.is_file() or not (root / "01_PROGRAMS").is_dir():
        print("G10D1D1-ERROR:INPUT C-7")
        return 7
    if out.exists():
        print("G10D1D1-ERROR:OUT_EXISTS C-7")
        return 7

    audit = json.loads(audit_path.read_text(encoding="utf-8-sig"))
    if audit.get("Schema") != "UDP001_G10D1_RELEASE_AUDIT_V0_1":
        print("G10D1D1-ERROR:AUDIT_SCHEMA C-7")
        return 7
    unexpected = list(audit.get("UnexpectedActivePythonFiles", []))
    expected = list(audit.get("ProgramOrder", []))
    if not unexpected:
        print("G10D1D1-ERROR:NO_UNEXPECTED_INPUT C-7")
        return 7

    program_root = root / "01_PROGRAMS"
    expected_source = {}
    for name in expected:
        path = program_root / name
        if not path.is_file():
            print("G10D1D1-ERROR:EXPECTED_MISSING C-7")
            return 7
        source = path.read_text(encoding="utf-8-sig", errors="replace")
        expected_source[name] = {
            "Text": source,
            "Imports": imported_modules(source),
        }

    rows = []
    reference_count = 0
    import_count = 0
    categories = {"VALIDATION": 0, "PACKAGING": 0, "DEPRECATED_VERSION": 0, "OTHER": 0}
    expected_set = set(expected)
    for index, name in enumerate(unexpected, 1):
        path = program_root / name
        if not path.is_file():
            print("G10D1D1-ERROR:UNEXPECTED_MISSING C-7")
            return 7
        stem = path.stem
        refs = []
        imports = []
        for owner, details in expected_source.items():
            if name in details["Text"] or stem in details["Text"]:
                refs.append(owner)
            if stem in details["Imports"]:
                imports.append(owner)
        category = classify(name, expected_set)
        categories[category] += 1
        reference_count += int(bool(refs))
        import_count += int(bool(imports))
        rows.append({
            "Index": index,
            "Name": name,
            "NameHash6": short_hash(name),
            "Category": category,
            "ReferencedByExpectedPrograms": refs,
            "ImportedByExpectedPrograms": imports,
            "Bytes": path.stat().st_size,
        })

    blockers = reference_count + import_count
    out.mkdir(parents=True, exist_ok=False)
    report = {
        "Schema": "UDP001_G10D1D1_UNEXPECTED_DEPENDENCY_V0_1",
        "CreatedUTC": datetime.now(timezone.utc).isoformat(),
        "ReadOnlyProbe": True,
        "SourceAudit": str(audit_path),
        "UnexpectedPrograms": rows,
        "UnexpectedCount": len(rows),
        "ReferencedCount": reference_count,
        "ImportedCount": import_count,
        "CategoryCounts": categories,
        "BlockingCount": blockers,
    }
    (out / "u001_g10d1d1_unexpected_program_dependency.json").write_text(
        json.dumps(report, indent=2, ensure_ascii=False), encoding="utf-8"
    )
    hashes = ".".join(row["NameHash6"] for row in rows)
    line1 = (
        f"G10D1D1 N-E{len(rows)}-R{reference_count}-I{import_count}-"
        f"V{categories['VALIDATION']}-P{categories['PACKAGING']}-"
        f"D{categories['DEPRECATED_VERSION']}-L{categories['OTHER']}-"
        f"X{blockers} C-{0 if blockers == 0 else 7}"
    )
    line2 = f"EH-{hashes}"
    (out / "u001_g10d1d1_return_capsule.txt").write_text(line1 + "\n" + line2 + "\n", encoding="utf-8")
    print(line1)
    print(line2)
    return 0 if blockers == 0 else 7


if __name__ == "__main__":
    raise SystemExit(main())
