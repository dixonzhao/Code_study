class HandoffError(RuntimeError):
    """Represent one deterministic real-handoff failure."""


def regular_file(path: Path) -> bool:
    """Accept ordinary files only."""

    return path.is_file() and not path.is_symlink()


def stable(value: object) -> str:
    """Convert a diagnostic value into a compact return token."""

    text = str(value).upper().replace(" ", "_").replace("-", "_")
    return "".join(char for char in text if char.isalnum() or char == "_")[:80]


def canonical_text(value: object) -> str:
    """Serialize one control document deterministically."""

    return json.dumps(
        value,
        ensure_ascii=False,
        sort_keys=True,
        separators=(",", ":"),
    ) + "\n"


def sha256_file(path: Path) -> str:
    """Hash a file without loading it into memory."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        while True:
            block = stream.read(1024 * 1024)
            if not block:
                break
            digest.update(block)
    return digest.hexdigest()


def package_files(root: Path) -> Dict[str, Path]:
    """Return the exact six public files from one package directory."""

    if not root.is_dir() or root.is_symlink():
        raise HandoffError("PACKAGE_DIRECTORY")
    result = {name: root / name for name in PACKAGE_NAMES}
    if any(not regular_file(path) for path in result.values()):
        raise HandoffError("PACKAGE_INCOMPLETE")
    return result


def package_snapshot(root: Path) -> Dict[str, Tuple[int, int, str]]:
    """Capture immutable size, modified time and content identity."""

    result: Dict[str, Tuple[int, int, str]] = {}
    for name, path in package_files(root).items():
        stat = path.stat()
        result[name] = (stat.st_size, stat.st_mtime_ns, sha256_file(path))
    return result


def package_fingerprint(snapshot: Mapping[str, Tuple[int, int, str]]) -> str:
    """Bind the six exact U001 public files into one fingerprint."""

    digest = hashlib.sha256()
    for name in sorted(PACKAGE_NAMES):
        size, _, file_hash = snapshot[name]
        digest.update(name.encode("utf-8"))
        digest.update(file_hash.encode("ascii"))
        digest.update(str(size).encode("ascii"))
    return digest.hexdigest()


def ignored(path: Path, search_root: Path) -> bool:
    """Ignore archives and transport folders during active-package discovery."""

    try:
        relative_parts = path.relative_to(search_root).parts
    except ValueError:
        return True
    return any(part.lower() in IGNORED_DIRECTORY_NAMES for part in relative_parts)


def discover_packages(search_root: Path) -> List[Path]:
    """Discover complete active six-file packages deterministically."""

    if not search_root.is_dir() or search_root.is_symlink():
        raise HandoffError("U001_SEARCH_ROOT")
    candidates: List[Path] = []
    for manifest in search_root.rglob("u001_run_manifest.json"):
        root = manifest.parent
        if ignored(root, search_root):
            continue
        try:
            package_files(root)
        except HandoffError:
            continue
        candidates.append(root.resolve())
    return sorted(set(candidates), key=lambda path: str(path).lower())


def choose_largest_package(candidates: Sequence[Path]) -> Path:
    """Choose the most substantial package, then newest, then path."""

    if not candidates:
        raise HandoffError("NO_COMPLETE_U001_PACKAGE")

    def rank(root: Path) -> Tuple[int, int, str]:
        frame_path = root / "u001_frame_values.csv"
        manifest_path = root / "u001_run_manifest.json"
        return (
            frame_path.stat().st_size,
            manifest_path.stat().st_mtime_ns,
            str(root).lower(),
        )

    return max(candidates, key=rank)


def run(command: Sequence[str], timeout: int = 7200) -> subprocess.CompletedProcess:
    """Run one controlled child and retain diagnostic streams."""

    return subprocess.run(
        list(command),
        capture_output=True,
        text=True,
        timeout=timeout,
        check=False,
    )


def require_foundation(root: Path) -> None:
    """Require the installed B1A module and frozen execution components."""

    required = (
        Path(MODULE_PATH) / "module_manifest.json",
        Path(MODULE_PATH) / "implementation.py",
        Path(MODULE_PATH) / "selftest.py",
        Path(PREFLIGHT),
        Path(EXECUTOR),
        Path(ARTIFACT_STORE),
        Path(FREEZE_AUDITOR),
    )
    for relative in required:
        if not regular_file(root / relative):
            raise HandoffError("MISSING_" + relative.name)

    input_package: Path,
    fingerprint: str,

    """Create one explicit real S0-only handoff Run Configuration."""

    return {

        "FrameStart": None,

        "InputPackage": str(input_package.resolve()),
        "SignalProfile": "U001RealControlledProfile@" + fingerprint[:12],
        "ModulePlan": [
            {
                "Sequence": 1,
                "Stage": "S0",
                "ModuleId": MODULE_ID,
                "ModuleVersion": MODULE_VERSION,
                "Configuration": {},
            }
        ],
        "PrimaryOutcome": {
            "Name": "ProvisionalCameraRadarDiscrepancy",
            "Formula": "RadarDistance-CameraDistance",
            "Sign": "REFERENCE-TARGET",
            "Unit": "m",
            "Purpose": "EXPLORATORY",
        },
        "CandidatePolicy": "NotAppliedInS0@0.1",
        "OutputSelection": ["U001_INTAKE_ACCEPTED"],
    }


def prepare(root: Path, configuration: Mapping[str, object]) -> subprocess.CompletedProcess:
    """Prepare the real S0 Run through frozen A5."""

    config_path = root / "02_CONFIG/run_configs" / (
        str(configuration["RunId"]) + ".json"
    )
    config_path.parent.mkdir(parents=True, exist_ok=True)
    config_path.write_text(canonical_text(configuration), encoding="utf-8")
    return run(
        [
            sys.executable,
            str(root / PREFLIGHT),
            "--prepare-run",
            str(config_path),
            "--workspace-root",
            str(root),
        ]
    )


def execute(root: Path, run_id: str) -> subprocess.CompletedProcess:
    """Execute the real S0 Run through frozen A6."""

    return run(
        [
            sys.executable,
            str(root / EXECUTOR),
            "--execute-run",
            str(root / "07_OUTPUT" / run_id),
            "--workspace-root",
            str(root),
            "--module-timeout-seconds",
            "7200",
        ]
    )


def audit(root: Path, run_id: str) -> subprocess.CompletedProcess:
    """Audit the produced real Run through frozen A3."""

    return run(
        [
            sys.executable,
            str(root / ARTIFACT_STORE),
            "--audit-run",
            "--run-root",
            str(root / "07_OUTPUT" / run_id),
            "--workspace-root",
            str(root),
        ]
    )


def freeze_audit(root: Path) -> subprocess.CompletedProcess:
    """Confirm the frozen Phase A foundation remains unchanged."""

    return run(
        [
            sys.executable,
            str(root / FREEZE_AUDITOR),
            "--audit-freeze",
            "--workspace-root",
            str(root),
        ],
        timeout=60,
    )


def inspect_summary(root: Path, run_id: str) -> Dict[str, object]:
    """Read the governed U001 intake summary through the artifact index."""

    run_root = root / "07_OUTPUT" / run_id
    index_path = run_root / "run_artifact_index_v0_1.json"
    if not regular_file(index_path):
        raise HandoffError("ARTIFACT_INDEX")
    index = json.loads(index_path.read_text(encoding="utf-8"))
    artifacts = index.get("Artifacts")
    if not isinstance(artifacts, list) or len(artifacts) != 1:
        raise HandoffError("ARTIFACT_COUNT")
    manifest_path = run_root / artifacts[0]["ManifestRelativePath"]
    summary_path = manifest_path.parent / "u001_intake_summary.json"
    if not regular_file(summary_path):
        raise HandoffError("SUMMARY_MISSING")
    summary = json.loads(summary_path.read_text(encoding="utf-8"))
    if summary.get("Status") != "INTAKE_ACCEPTED":
        raise HandoffError("SUMMARY_STATUS")
    return summary


def require_success(result: subprocess.CompletedProcess, label: str) -> None:
    """Convert a failed child process into one compact diagnostic."""

    if result.returncode != 0:
        detail = result.stderr.strip() or result.stdout.strip() or "NO_DETAIL"
        raise HandoffError(label + "_" + stable(detail))


def write_acceptance_receipt(
    root: Path,
    run_id: str,
    package: Path,
    candidate_count: int,
    snapshot: Mapping[str, Tuple[int, int, str]],
    summary: Mapping[str, object],
) -> None:
    """Persist a compact metadata-only real handoff receipt."""

    receipt = {
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": VERSION,
        "RunId": run_id,
        "SelectedPackage": str(package.resolve()),
        "DiscoveredCompletePackages": candidate_count,
        "PackageFingerprintSha256": package_fingerprint(snapshot),
        "InputFiles": [
            {
                "Basename": name,
                "Bytes": snapshot[name][0],
                "Sha256": snapshot[name][2],
            }
            for name in PACKAGE_NAMES
        ],
        "FrameDomain": summary["FrameDomain"],
        "SignalCount": summary["SignalCount"],
        "QualityGate": summary["QualityGate"],
        "Status": "REAL_U001_HANDOFF_ACCEPTED",
        "Prohibitions": summary["Prohibitions"],
    }
    path = root / "07_OUTPUT" / run_id / "u22b1ar_real_handoff_receipt.json"
    path.write_text(canonical_text(receipt), encoding="utf-8")


def self_test() -> int:
    """Test discovery, ranking, hashing and mutation detection utilities."""

    checks: List[bool] = []

    def check(condition: bool) -> None:
        if not condition:
            raise HandoffError("SELFTEST")
        checks.append(True)

    with tempfile.TemporaryDirectory(prefix="u22b1ar_selftest_") as temp:
        root = Path(temp)
        small = root / "small"
        large = root / "large"
        archived = root / "A-Archive" / "old"
        for package, size in ((small, 2), (large, 20), (archived, 100)):
            package.mkdir(parents=True)
            for name in PACKAGE_NAMES:
                payload = "{}\n" if name.endswith(".json") else "x\n"
                if name == "u001_frame_values.csv":
                    payload = "x" * size
                (package / name).write_text(payload, encoding="utf-8")
        candidates = discover_packages(root)
        check(candidates == sorted([small.resolve(), large.resolve()], key=lambda p: str(p).lower()))
        check(choose_largest_package(candidates) == large.resolve())
        first = package_snapshot(large)
        check(len(first) == 6)
        fingerprint = package_fingerprint(first)
        check(len(fingerprint) == 64)
        check(first == package_snapshot(large))
        (large / "u001_quality_report.csv").write_text("changed\n", encoding="utf-8")
        check(first != package_snapshot(large))
        check(ignored(archived, root))
        check(not ignored(small, root))
    print(f"U22B1AR-ST N-T{len(checks)}-X0 C-0")
    return 0


def main(argv: Sequence[str] = None) -> int:
    """Select a real U001 package and execute the governed S0 handoff."""

    parser = argparse.ArgumentParser()
    parser.add_argument("--self-test", action="store_true")
    parser.add_argument("--u002-workspace-root")
    parser.add_argument("--u001-search-root")
    parser.add_argument("--u001-package")
    args = parser.parse_args(sys.argv[1:] if argv is None else argv)
    if args.self_test:
        try:
            return self_test()
        except Exception as exc:
            print("U22B1AR-ST N-T0-X1 C-7 R-" + stable(exc))
            return 7
    try:
        if not args.u002_workspace_root:
            raise HandoffError("U002_ROOT_REQUIRED")
        root = Path(args.u002_workspace_root).resolve()
        require_foundation(root)
        if args.u001_package:
            package = Path(args.u001_package).resolve()
            package_files(package)
            candidates = [package]
        else:
            if not args.u001_search_root:
                raise HandoffError("U001_SEARCH_ROOT_REQUIRED")
            candidates = discover_packages(Path(args.u001_search_root).resolve())
            package = choose_largest_package(candidates)
        before = package_snapshot(package)
        fingerprint = package_fingerprint(before)
        run_id = "U22B1AR_" + fingerprint[:12] + "_" + str(int(time.time()))
        freeze_before = freeze_audit(root)
        require_success(freeze_before, "FREEZE_BEFORE")
        configuration = run_configuration(run_id, package, fingerprint)
        prepared = prepare(root, configuration)
        require_success(prepared, "PREFLIGHT")
        executed = execute(root, run_id)
        require_success(executed, "EXECUTOR")
        audited = audit(root, run_id)
        require_success(audited, "ARTIFACT_AUDIT")
        summary = inspect_summary(root, run_id)
        after = package_snapshot(package)
        if before != after:
            raise HandoffError("U001_INPUT_CHANGED")
        if summary.get("PackageFingerprintSha256") != fingerprint:
            raise HandoffError("FINGERPRINT_MISMATCH")
        freeze_after = freeze_audit(root)
        require_success(freeze_after, "FREEZE_AFTER")
        write_acceptance_receipt(
            root,
            run_id,
            package,
            len(candidates),
            before,
            summary,
        )
        frame = summary["FrameDomain"]
        print(
            "U22B1AR N-C"
            + str(len(candidates))
            + "-F"
            + str(frame["FrameRows"])
            + "-K"
            + str(summary["SignalCount"])
            + "-G"
            + str(frame["MissingFrameGapCount"])
            + "-Q1-R1-I1-H1-X0 C-0"
        )
        return 0
    except subprocess.TimeoutExpired:
        print("U22B1AR N-C0-F0-K0-G0-Q0-R0-I0-H0-X1 C-7 R-TIMEOUT")
        return 7
    except Exception as exc:
        print(
            "U22B1AR N-C0-F0-K0-G0-Q0-R0-I0-H0-X1 C-7 R-"
            + stable(exc)
        )
        return 7


if __name__ == "__main__":
    raise SystemExit(main())
