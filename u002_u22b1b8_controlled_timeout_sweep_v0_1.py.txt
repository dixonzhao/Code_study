class SweepError(RuntimeError):
    """Represent one deterministic fail-closed sweep failure."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = stable(reason, 56)
        self.detail = stable(detail, 96)
        super().__init__(self.reason)


def stable(value: object, limit: int = 80) -> str:
    """Convert diagnostic text into one compact capsule token."""

    normalized = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper())
    return normalized.strip("_")[:limit] or "NA"


def sha256_file(path: Path) -> str:
    """Hash one file by bounded blocks."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_base(program_directory: Path) -> ModuleType:
    """Load the exact B1B7 v0.2 dependency after byte-identity verification."""

    path = program_directory / BASE_FILENAME
    if not path.is_file() or path.is_symlink():
        raise SweepError("B1B7_V02_MISSING", path.name)
    observed = sha256_file(path)
    if observed != BASE_SHA256:
        raise SweepError("B1B7_V02_HASH", observed[:12])
    loader = importlib.machinery.SourceFileLoader("u22b1b8_base", str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None:
        raise SweepError("B1B7_IMPORT_SPEC")
    module = importlib.util.module_from_spec(spec)
    try:
        loader.exec_module(module)
    except Exception as exc:
        raise SweepError("B1B7_IMPORT", type(exc).__name__) from exc
    return module


def reconstruct_with_timeout(
    records: Sequence[Mapping[str, object]],
    configuration: Mapping[str, object],
    timeout_frames: int,
    module: ModuleType,
) -> List[Dict[str, object]]:
    """Apply the accepted v0.1 state machine plus one controlled timeout branch."""

    if timeout_frames < 1:
        raise SweepError("TIMEOUT_NOT_POSITIVE", timeout_frames)
    role_1 = module.ROLE_1
    role_3 = module.ROLE_3
    allowed_roles = module.ALLOWED_ROLES
    scale = float(configuration["ContinuityScaleMeters"])
    max_cost = float(configuration["MaxContinuityCost"])
    acquire_frames = int(configuration["AcquireConfirmFrames"])
    switch_frames = int(configuration["SwitchConfirmFrames"])
    switch_margin = float(configuration["SwitchMargin"])
    reset_frames = int(configuration["ResetAfterBothInvalidFrames"])
    seed_frame = configuration["SeedFrame"]
    seed_role = configuration["SeedRole"]

    selected_role: Optional[str] = None
    last_confident_value: Optional[float] = None
    pending_role: Optional[str] = None
    pending_count = 0
    pending_last_value: Optional[float] = None
    reacquire_hint_role: Optional[str] = None
    invalid_count = 0
    no_continuity_count = 0
    segment_id = 0
    previous_frame: Optional[int] = None
    outputs: List[Dict[str, object]] = []

    def clear_pending() -> None:
        nonlocal pending_role, pending_count, pending_last_value
        pending_role = None
        pending_count = 0
        pending_last_value = None

    for record in records:
        frame = int(record["FrameCount"])
        gap = previous_frame is not None and frame != previous_frame + 1
        if previous_frame is None or gap:
            reacquire_hint_role = selected_role
            segment_id += 1
            selected_role = None
            last_confident_value = None
            invalid_count = 0
            no_continuity_count = 0
            clear_pending()
        previous_frame = frame

        camera_1 = record[role_1]
        camera_3 = record[role_3]
        source = {role_1: camera_1, role_3: camera_3}
        cost_1 = module.cost(camera_1["Value"], last_confident_value, scale)
        cost_3 = module.cost(camera_3["Value"], last_confident_value, scale)
        costs = {role_1: cost_1, role_3: cost_3}

        if seed_frame == frame and seed_role in allowed_roles:
            no_continuity_count = 0
            candidate = source[str(seed_role)]
            if not candidate["Eligible"]:
                outputs.append(module.output_row(
                    record,
                    candidate["Value"],
                    None,
                    False,
                    None,
                    "AMBIGUOUS",
                    "CONFIGURED_SEED_INELIGIBLE",
                    False,
                    cost_1,
                    cost_3,
                    segment_id,
                ))
                continue
            selected_role = str(seed_role)
            last_confident_value = float(candidate["Value"])
            clear_pending()
            outputs.append(module.output_row(
                record,
                candidate["Value"],
                candidate["Value"],
                True,
                selected_role,
                "SEEDED",
                "CONFIGURED_CASE_SEED",
                False,
                cost_1,
                cost_3,
                segment_id,
            ))
            continue

        eligible_roles = [
            role for role in allowed_roles if source[role]["Eligible"]
        ]
        if not eligible_roles:
            no_continuity_count = 0
            invalid_count += 1
            clear_pending()
            status = "INVALID"
            reason = "BOTH_SOURCES_INELIGIBLE"
            if invalid_count >= reset_frames:
                if invalid_count == reset_frames:
                    reacquire_hint_role = selected_role
                    segment_id += 1
                selected_role = None
                last_confident_value = None
                if invalid_count == reset_frames:
                    status = "SEGMENT_RESET"
                    reason = "INVALID_RUN_RESET"
                else:
                    reason = "RESET_STATE_BOTH_SOURCES_INELIGIBLE"
            outputs.append(module.output_row(
                record,
                None,
                None,
                False,
                selected_role,
                status,
                reason,
                False,
                cost_1,
                cost_3,
                segment_id,
            ))
            continue
        invalid_count = 0

        if selected_role is None:
            no_continuity_count = 0
            candidate_role: Optional[str] = None
            if len(eligible_roles) == 1:
                candidate_role = eligible_roles[0]
            elif reacquire_hint_role in eligible_roles:
                candidate_role = reacquire_hint_role
            elif pending_role in eligible_roles:
                candidate_role = pending_role
            if candidate_role is None:
                clear_pending()
                outputs.append(module.output_row(
                    record,
                    None,
                    None,
                    False,
                    None,
                    "AMBIGUOUS",
                    "MULTIPLE_UNSEEDED_CANDIDATES",
                    False,
                    cost_1,
                    cost_3,
                    segment_id,
                ))
                continue
            candidate = source[candidate_role]
            internal_cost = module.cost(
                candidate["Value"],
                pending_last_value,
                scale,
            )
            if pending_role != candidate_role:
                pending_role = candidate_role
                pending_count = 1
            elif internal_cost is None or internal_cost <= max_cost:
                pending_count += 1
            else:
                pending_count = 1
            pending_last_value = float(candidate["Value"])
            if pending_count < acquire_frames:
                outputs.append(module.output_row(
                    record,
                    candidate["Value"],
                    None,
                    False,
                    None,
                    "ACQUIRING",
                    "SINGLE_SOURCE_CONFIRMATION_PENDING",
                    False,
                    cost_1,
                    cost_3,
                    segment_id,
                ))
                continue
            selected_role = candidate_role
            last_confident_value = float(candidate["Value"])
            reacquire_hint_role = None
            clear_pending()
            outputs.append(module.output_row(
                record,
                candidate["Value"],
                candidate["Value"],
                True,
                selected_role,
                "CONFIDENT",
                "SINGLE_SOURCE_ACQUIRED",
                False,
                cost_1,
                cost_3,
                segment_id,
            ))
            continue

        current_role = selected_role
        alternative_role = role_3 if current_role == role_1 else role_1
        current = source[current_role]
        alternative = source[alternative_role]
        current_cost = costs[current_role]
        alternative_cost = costs[alternative_role]
        current_plausible = bool(
            current["Eligible"]
            and current_cost is not None
            and current_cost <= max_cost
        )
        alternative_plausible = bool(
            alternative["Eligible"]
            and alternative_cost is not None
            and alternative_cost <= max_cost
        )
        strong_alternative = bool(
            alternative_plausible
            and (
                not current_plausible
                or float(alternative_cost) + switch_margin < float(current_cost)
            )
        )

        if strong_alternative:
            no_continuity_count = 0
            if pending_role == alternative_role:
                pending_count += 1
            else:
                pending_role = alternative_role
                pending_count = 1
            pending_last_value = float(alternative["Value"])
            if pending_count < switch_frames:
                outputs.append(module.output_row(
                    record,
                    alternative["Value"],
                    None,
                    False,
                    current_role,
                    "SWITCH_PENDING",
                    "ALTERNATIVE_CONFIRMATION_PENDING",
                    False,
                    cost_1,
                    cost_3,
                    segment_id,
                ))
                continue
            selected_role = alternative_role
            last_confident_value = float(alternative["Value"])
            clear_pending()
            outputs.append(module.output_row(
                record,
                alternative["Value"],
                alternative["Value"],
                True,
                selected_role,
                "CONFIDENT",
                "SWITCH_CONFIRMED",
                True,
                cost_1,
                cost_3,
                segment_id,
            ))
            continue

        clear_pending()
        if current_plausible:
            no_continuity_count = 0
            last_confident_value = float(current["Value"])
            outputs.append(module.output_row(
                record,
                current["Value"],
                current["Value"],
                True,
                current_role,
                "CONFIDENT",
                "CURRENT_SOURCE_PLAUSIBLE",
                False,
                cost_1,
                cost_3,
                segment_id,
            ))
            continue

        no_continuity_count += 1
        if no_continuity_count >= timeout_frames:
            reacquire_hint_role = current_role
            selected_role = None
            last_confident_value = None
            invalid_count = 0
            no_continuity_count = 0
            segment_id += 1
            clear_pending()
            outputs.append(module.output_row(
                record,
                None,
                None,
                False,
                None,
                "SEGMENT_RESET",
                "CONTINUITY_REJECTION_TIMEOUT_RESET",
                False,
                cost_1,
                cost_3,
                segment_id,
            ))
            continue
        outputs.append(module.output_row(
            record,
            None,
            None,
            False,
            current_role,
            "AMBIGUOUS",
            "NO_CONTINUITY_SUPPORTED_PATH",
            False,
            cost_1,
            cost_3,
            segment_id,
        ))
    return outputs


def locate_review_windows(run_root: Path) -> List[Dict[str, object]]:
    """Read the unique B1B4 review queue and preserve declared Event bounds."""

    paths = sorted((run_root / "04_REVIEW_PACKAGE").glob("*.csv"))
    accepted: List[Tuple[Path, List[Dict[str, object]]]] = []
    for path in paths:
        try:
            with path.open("r", encoding="utf-8-sig", newline="") as stream:
                reader = csv.DictReader(stream)
                header = reader.fieldnames or []
                if any(name not in header for name in REVIEW_REQUIRED):
                    continue
                rows: List[Dict[str, object]] = []
                for row in reader:
                    start = int(row["StartFrameCount"])
                    peak = int(row["PeakFrameCount"])
                    end = int(row["EndFrameCount"])
                    if not start <= peak <= end:
                        raise SweepError("REVIEW_WINDOW_ORDER", row["EventID"])
                    rows.append({
                        "EventID": row["EventID"],
                        "StartFrame": start,
                        "PeakFrame": peak,
                        "EndFrame": end,
                    })
                if rows:
                    accepted.append((path, rows))
        except SweepError:
            raise
        except Exception:
            continue
    if len(accepted) != 1:
        raise SweepError("REVIEW_QUEUE_COUNT", len(accepted))
    return accepted[0][1]


def select_q6_and_protected(
    windows: Sequence[Mapping[str, object]],
    loss_frame: int,
) -> Tuple[Mapping[str, object], List[Mapping[str, object]]]:
    """Separate the Q6 failure window from previously reviewed windows."""

    q6_matches = [
        item
        for item in windows
        if int(item["StartFrame"]) <= loss_frame <= int(item["EndFrame"])
    ]
    if len(q6_matches) != 1:
        raise SweepError("Q6_REVIEW_WINDOW_COUNT", len(q6_matches))
    q6 = q6_matches[0]
    protected = [item for item in windows if item is not q6]
    if not protected:
        raise SweepError("PROTECTED_WINDOWS_EMPTY")
    return q6, protected


def rows_in_bounds(
    rows: Sequence[Mapping[str, object]],
    start: int,
    end: int,
) -> Dict[int, Mapping[str, object]]:
    """Index one inclusive Frame range."""

    return {
        int(row["FrameCount"]): row
        for row in rows
        if start <= int(row["FrameCount"]) <= end
    }


def protected_differences(
    original: Sequence[Mapping[str, object]],
    candidate: Sequence[Mapping[str, object]],
    windows: Sequence[Mapping[str, object]],
    base: ModuleType,
) -> Tuple[int, int]:
    """Count lost and changed previously valid rows in protected windows."""

    original_by_frame = {int(row["FrameCount"]): row for row in original}
    candidate_by_frame = {int(row["FrameCount"]): row for row in candidate}
    lost = 0
    changed = 0
    counted = set()
    for window in windows:
        start = int(window["StartFrame"])
        end = int(window["EndFrame"])
        for frame in range(start, end + 1):
            if frame in counted or frame not in original_by_frame:
                continue
            counted.add(frame)
            old = original_by_frame[frame]
            new = candidate_by_frame.get(frame)
            if new is None:
                raise SweepError("CANDIDATE_FRAME_MISSING", frame)
            if str(old["CanonicalCameraValid"]) != "1":
                continue
            if str(new["CanonicalCameraValid"]) != "1":
                lost += 1
                continue
            same_role = (
                str(old["SelectedCameraRole"])
                == str(new["SelectedCameraRole"])
            )
            same_value = base.numeric_text_equal(
                old["CanonicalCameraDistance"],
                new["CanonicalCameraDistance"],
            )
            if not same_role or not same_value:
                changed += 1
    return lost, changed


def evaluate_threshold(
    timeout: int,
    original: Sequence[Mapping[str, object]],
    candidate: Sequence[Mapping[str, object]],
    protected: Sequence[Mapping[str, object]],
    loss_frame: int,
    acquire_frames: int,
    base: ModuleType,
) -> Dict[str, object]:
    """Evaluate Q6 recovery and protected-window regression for one threshold."""

    replay_window = base.window_rows(candidate)
    episode = base.counterfactual_episode_at(replay_window, loss_frame)
    recovery_value = episode["RecoveryFrame"]
    recovery = int(recovery_value) if recovery_value is not None else None
    delay = (
        recovery - loss_frame
        if recovery is not None
        else int(episode["InvalidRows"])
    )
    lost, changed = protected_differences(
        original,
        candidate,
        protected,
        base,
    )
    resets = sum(
        1
        for row in candidate
        if row["SelectionReason"] == "CONTINUITY_REJECTION_TIMEOUT_RESET"
    )
    newly_valid = sum(
        1
        for old, new in zip(original, candidate)
        if str(old["CanonicalCameraValid"]) == "0"
        and str(new["CanonicalCameraValid"]) == "1"
    )
    expected_horizon = timeout + acquire_frames
    passed = bool(
        recovery is not None
        and delay <= expected_horizon
        and lost == 0
        and changed == 0
        and resets > 0
    )
    return {
        "TimeoutFrames": timeout,
        "Q6RecoveryFrame": recovery,
        "Q6RecoveryDelayFrames": delay,
        "ExpectedMaximumDelayFrames": expected_horizon,
        "TimeoutResetCountFullRange": resets,
        "NewlyValidRowsFullRange": newly_valid,
        "LostOriginalValidRowsProtectedWindows": lost,
        "ChangedOriginalValidRowsProtectedWindows": changed,
        "MachineAcceptancePass": passed,
    }


def compact_metric(metric: Mapping[str, object]) -> str:
    """Encode one threshold result for low-bandwidth manual return."""

    recovery = metric["Q6RecoveryFrame"]
    return (
        "T"
        + str(metric["TimeoutFrames"])
        + "-R"
        + (str(recovery) if recovery is not None else "0")
        + "-D"
        + str(metric["Q6RecoveryDelayFrames"])
        + "-Z"
        + str(metric["TimeoutResetCountFullRange"])
        + "-L"
        + str(metric["LostOriginalValidRowsProtectedWindows"])
        + "-C"
        + str(metric["ChangedOriginalValidRowsProtectedWindows"])
        + "-P"
        + ("1" if metric["MachineAcceptancePass"] else "0")
    )


def trace_bytes(
    records: Sequence[Mapping[str, object]],
    results: Mapping[int, Sequence[Mapping[str, object]]],
    base: ModuleType,
) -> bytes:
    """Serialize only the Q6 analysis window for all four candidates."""

    import io

    records_by_frame = {int(row["FrameCount"]): row for row in records}
    buffer = io.StringIO(newline="")
    writer = csv.DictWriter(buffer, fieldnames=TRACE_FIELDS, lineterminator="\n")
    writer.writeheader()
    for timeout in TIMEOUT_VALUES:
        for row in base.window_rows(results[timeout]):
            frame = int(row["FrameCount"])
            record = records_by_frame[frame]
            writer.writerow({
                "TimeoutFrames": timeout,
                "FrameCount": frame,
                "Camera1Distance": record[base.ROLE_1]["Value"],
                "Camera1Eligible": int(bool(record[base.ROLE_1]["Eligible"])),
                "Camera3Distance": record[base.ROLE_3]["Value"],
                "Camera3Eligible": int(bool(record[base.ROLE_3]["Eligible"])),
                "CanonicalCameraDistance": row["CanonicalCameraDistance"],
                "CanonicalCameraValid": row["CanonicalCameraValid"],
                "SelectedCameraRole": row["SelectedCameraRole"],
                "SelectionStatus": row["SelectionStatus"],
                "SelectionReason": row["SelectionReason"],
                "SegmentId": row["SegmentId"],
            })
    return buffer.getvalue().encode("utf-8")


def run(root: Path, program_directory: Path) -> Tuple[str, str, str]:
    """Execute the exact-baseline guard and four-value timeout sweep."""

    base = load_base(program_directory)
    workspace = base.require_workspace(root)
    run_root, receipt = base.discover_b1b4(workspace)
    table = base.locate_s1_table(run_root)
    table_hash = base.sha256_file(table)
    configuration = base.load_s1_configuration(run_root)
    accepted, implementation, implementation_hash = base.load_accepted_s1(workspace)
    source_rows = base.read_source_rows(table)
    original_records, _ = base.normalized_records(source_rows, mask_sentinel=False)
    masked_records, mask_counts = base.normalized_records(
        source_rows,
        mask_sentinel=True,
    )
    accepted_replay = accepted.reconstruct(original_records, configuration)
    base.verify_original_replay(source_rows, accepted_replay)
    disabled_replay = reconstruct_with_timeout(
        original_records,
        configuration,
        DISABLED_TIMEOUT,
        accepted,
    )
    base.verify_original_replay(source_rows, disabled_replay)
    baseline_equivalent = True

    old_episode = base.longest_completed_episode(base.window_rows(source_rows))
    loss_frame = int(old_episode["StartFrame"])
    windows = locate_review_windows(run_root)
    q6_window, protected = select_q6_and_protected(windows, loss_frame)
    candidate_outputs: Dict[int, Sequence[Mapping[str, object]]] = {}
    metrics: List[Dict[str, object]] = []
    for timeout in TIMEOUT_VALUES:
        output = reconstruct_with_timeout(
            masked_records,
            configuration,
            timeout,
            accepted,
        )
        if len(output) != len(source_rows):
            raise SweepError("CANDIDATE_ROW_COUNT", timeout)
        candidate_outputs[timeout] = output
        metrics.append(evaluate_threshold(
            timeout,
            source_rows,
            output,
            protected,
            loss_frame,
            int(configuration["AcquireConfirmFrames"]),
            base,
        ))

    summary = {
        "Contract": "UDP002_U22B1B8_CONTROLLED_TIMEOUT_SWEEP_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": VERSION,
        "RunId": receipt.get("RunId"),
        "SourceTableSha256": table_hash,
        "AcceptedImplementationRelativePath": str(implementation.relative_to(workspace)),
        "AcceptedImplementationSha256": implementation_hash,
        "B1B7DependencySha256": BASE_SHA256,
        "BaselineDisabledTimeoutEquivalent": baseline_equivalent,
        "OriginalQ6Episode": old_episode,
        "Q6ReviewWindow": dict(q6_window),
        "ProtectedReviewWindows": [dict(item) for item in protected],
        "MaskCounts": mask_counts,
        "TimeoutValues": list(TIMEOUT_VALUES),
        "Metrics": metrics,
        "DecisionStatus": "EXPERIMENT_ONLY_THRESHOLD_NOT_APPROVED",
        "InputMutationCount": 0,
    }
    export_root = run_root / "05_EXPORTS"
    base.write_exact(export_root / SUMMARY_NAME, base.canonical_bytes(summary))
    base.write_exact(
        export_root / TRACE_NAME,
        trace_bytes(masked_records, candidate_outputs, base),
    )
    if base.sha256_file(table) != table_hash:
        raise SweepError("SOURCE_TABLE_CHANGED")
    main = (
        "U22B1B8 N-W"
        + str(len(windows))
        + "-P"
        + str(len(protected))
        + "-T"
        + str(len(TIMEOUT_VALUES))
        + "-E1-X0 C-0"
    )
    detail_1 = "U22B1B8D1 " + " ".join(
        compact_metric(metric) for metric in metrics[:2]
    ) + " C-0"
    detail_2 = "U22B1B8D2 " + " ".join(
        compact_metric(metric) for metric in metrics[2:]
    ) + " C-0"
    return main, detail_1, detail_2


def synthetic_module() -> ModuleType:
    """Build a minimal adapter for standalone state-machine self-test."""

    role_1 = "CAMERA_1_DISTANCE"
    role_3 = "CAMERA_3_DISTANCE"

    def cost(value, reference, scale):
        if value is None or reference is None:
            return None
        return abs(float(value) - float(reference)) / float(scale)

    def output_row(
        record,
        provisional,
        canonical,
        canonical_valid,
        selected_role,
        status,
        reason,
        changed,
        cost_1,
        cost_3,
        segment_id,
    ):
        return {
            "FrameCount": record["FrameCount"],
            "CanonicalCameraDistance": "" if canonical is None else canonical,
            "CanonicalCameraValid": int(canonical_valid),
            "SelectedCameraRole": selected_role or "NONE",
            "SelectionStatus": status,
            "SelectionReason": reason,
            "SelectionChanged": int(changed),
            "ContinuityCostCamera1": "" if cost_1 is None else cost_1,
            "ContinuityCostCamera3": "" if cost_3 is None else cost_3,
            "CostMargin": "",
            "SegmentId": segment_id,
        }

    return SimpleNamespace(
        ROLE_1=role_1,
        ROLE_3=role_3,
        ALLOWED_ROLES=(role_1, role_3),
        cost=cost,
        output_row=output_row,
    )


def self_test(program_directory: Path) -> str:
    """Exercise dependency identity and the added timeout transition."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise SweepError("SELFTEST")
        checks.append(True)

    base = load_base(program_directory)

def write_failure(root: Optional[Path], error: SweepError) -> None:
    """Persist full failure context while returning one short capsule."""

    if root is None or not root.is_dir():
        return
    payload = {
        "Contract": "UDP002_U22B1B8_FAILURE_DIAGNOSTIC_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": VERSION,
        "Reason": error.reason,
        "Detail": error.detail,
        "NextAction": "Return the one-line C-7 capsule and retain this JSON.",
    }
    try:
        path = root / DIAGNOSTIC_RELATIVE
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(
            json.dumps(payload, ensure_ascii=False, sort_keys=True, indent=2) + "\n",
            encoding="utf-8",
        )
    except Exception:
        pass


def parser() -> argparse.ArgumentParser:
    """Build the CLI."""

    result = argparse.ArgumentParser(description=PROGRAM_ID)
    result.add_argument("--workspace-root", type=Path)
    result.add_argument("--self-test", action="store_true")
    return result


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Execute one bounded mode."""

    arguments = parser().parse_args(argv)
    program_directory = Path(__file__).resolve().parent
    root: Optional[Path] = None
    try:
        if arguments.self_test:
            print(self_test(program_directory))
            return 0
        if arguments.workspace_root is None:
            raise SweepError("WORKSPACE_ARGUMENT")
        root = arguments.workspace_root.resolve()
        capsules = run(root, program_directory)
        for capsule in capsules:
            print(capsule)
        return 0
    except SweepError as exc:
        write_failure(root, exc)
        print("U22B1B8 N-X1 C-7 K-" + exc.reason + " D-" + exc.detail)
        return 7
    except Exception as exc:
        wrapped = SweepError("UNEXPECTED", type(exc).__name__)
        write_failure(root, wrapped)
        print("U22B1B8 N-X1 C-7 K-UNEXPECTED D-" + stable(type(exc).__name__))
        return 7


if __name__ == "__main__":
    sys.exit(main())
