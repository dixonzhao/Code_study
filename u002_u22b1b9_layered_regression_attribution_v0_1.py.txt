PROGRAM_ID = "U002-U22B1B9-LAYERED-REGRESSION-ATTRIBUTION"
VERSION = "0.1"
B1B8_FILENAME = "u002_u22b1b8_controlled_timeout_sweep_v0_1.py.txt"
B1B8_SHA256 = "c44640c39a5aeecbc122d5e65d54ea41fb88c4b3c2581ca38be71cd2f576001e"
SUMMARY_NAME = "u22b1b9_layered_regression_attribution_summary_v0_1.json"
TRACE_NAME = "u22b1b9_protected_window_change_trace_v0_1.csv"
DIAGNOSTIC_RELATIVE = Path(

    "Layer",
    "TimeoutFrames",
    "EventID",
    "FrameCount",
    "ChangeType",
    "ReferenceValid",
    "ReferenceRole",
    "ReferenceValue",
    "ReferenceStatus",

)


class AttributionError(RuntimeError):
    """Represent one deterministic fail-closed attribution failure."""

    def __init__(self, reason: str, detail: object = "NA"):
        self.reason = stable(reason, 56)
        self.detail = stable(detail, 96)
        super().__init__(self.reason)


def stable(value: object, limit: int = 80) -> str:
    """Convert diagnostic text to one compact machine token."""

    normalized = re.sub(r"[^A-Z0-9_]+", "_", str(value).upper())
    return normalized.strip("_")[:limit] or "NA"


def sha256_file(path: Path) -> str:
    """Hash one regular file in bounded blocks."""

    digest = hashlib.sha256()
    with path.open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def load_program(name: str, path: Path) -> ModuleType:
    """Load one verified Python text program as a module."""

    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    if spec is None:
        raise AttributionError("IMPORT_SPEC", path.name)
    module = importlib.util.module_from_spec(spec)
    try:
        loader.exec_module(module)
    except Exception as exc:
        raise AttributionError("IMPORT_EXEC", type(exc).__name__) from exc
    return module


def load_b1b8(program_directory: Path) -> ModuleType:
    """Load the exact B1B8 dependency after byte-identity verification."""

    path = program_directory / B1B8_FILENAME
    if not path.is_file() or path.is_symlink():
        raise AttributionError("B1B8_V01_MISSING", path.name)
    observed = sha256_file(path)
    if observed != B1B8_SHA256:
        raise AttributionError("B1B8_V01_HASH", observed[:12])
    return load_program("u22b1b9_b1b8", path)


def valid(row: Mapping[str, object]) -> bool:
    """Read the public canonical-valid flag consistently."""

    return str(row.get("CanonicalCameraValid", "")).strip() == "1"


def same_decision(
    reference: Mapping[str, object],
    candidate: Mapping[str, object],
    base: ModuleType,
) -> bool:
    """Compare only the accepted public target decision semantics."""

    return bool(
        str(reference.get("SelectedCameraRole", ""))
        == str(candidate.get("SelectedCameraRole", ""))
        and base.numeric_text_equal(
            reference.get("CanonicalCameraDistance", ""),
            candidate.get("CanonicalCameraDistance", ""),
        )
    )


def indexed_rows(
    rows: Sequence[Mapping[str, object]],
) -> Dict[int, Mapping[str, object]]:
    """Index a strictly unique result table by FrameCount."""

    result: Dict[int, Mapping[str, object]] = {}
    for row in rows:
        frame = int(row["FrameCount"])
        if frame in result:
            raise AttributionError("DUPLICATE_FRAME", frame)
        result[frame] = row
    return result


def protected_frames(
    windows: Sequence[Mapping[str, object]],
) -> Dict[int, str]:
    """Expand protected review windows to a unique Frame-to-Event map."""

    result: Dict[int, str] = {}
    for window in windows:
        event_id = str(window["EventID"])
        start = int(window["StartFrame"])
        end = int(window["EndFrame"])
        for frame in range(start, end + 1):
            if frame in result and result[frame] != event_id:
                raise AttributionError("PROTECTED_WINDOW_OVERLAP", frame)
            result[frame] = event_id
    return result


def compare_layer(
    reference: Sequence[Mapping[str, object]],
    candidate: Sequence[Mapping[str, object]],
    windows: Sequence[Mapping[str, object]],
    base: ModuleType,
    layer: str,
    timeout: Optional[int],
) -> Tuple[Dict[str, int], List[Dict[str, object]]]:
    """Compare one candidate against one explicit reference layer."""

    reference_by_frame = indexed_rows(reference)
    candidate_by_frame = indexed_rows(candidate)
    frames = protected_frames(windows)
    metrics = {
        "ReferenceValidRows": 0,
        "LostReferenceValidRows": 0,
        "ChangedReferenceValidRows": 0,
        "GainedCandidateValidRows": 0,
    }
    trace: List[Dict[str, object]] = []

    for frame, event_id in sorted(frames.items()):
        if frame not in reference_by_frame:
            continue
        if frame not in candidate_by_frame:
            raise AttributionError("CANDIDATE_FRAME_MISSING", frame)
        before = reference_by_frame[frame]
        after = candidate_by_frame[frame]
        change_type: Optional[str] = None

        if valid(before):
            metrics["ReferenceValidRows"] += 1
            if not valid(after):
                metrics["LostReferenceValidRows"] += 1
                change_type = "LOST_REFERENCE_VALID"
            elif not same_decision(before, after, base):
                metrics["ChangedReferenceValidRows"] += 1
                change_type = "CHANGED_REFERENCE_DECISION"
        elif valid(after):
            metrics["GainedCandidateValidRows"] += 1
            change_type = "GAINED_CANDIDATE_VALID"

        if change_type is not None:
            trace.append({
                "Layer": layer,
                "TimeoutFrames": "" if timeout is None else timeout,
                "EventID": event_id,
                "FrameCount": frame,
                "ChangeType": change_type,
                "ReferenceValid": int(valid(before)),
                "ReferenceRole": before.get("SelectedCameraRole", ""),
                "ReferenceValue": before.get("CanonicalCameraDistance", ""),
                "ReferenceStatus": before.get("SelectionStatus", ""),
                "CandidateValid": int(valid(after)),
                "CandidateRole": after.get("SelectedCameraRole", ""),
                "CandidateValue": after.get("CanonicalCameraDistance", ""),
                "CandidateStatus": after.get("SelectionStatus", ""),
            })
    return metrics, trace


def q6_recovery_metric(
    candidate: Sequence[Mapping[str, object]],
    loss_frame: int,
    timeout: int,
    acquire_frames: int,
    base: ModuleType,
) -> Dict[str, object]:
    """Measure recovery against the reviewed original Q6 episode."""

    episode = base.counterfactual_episode_at(
        base.window_rows(candidate),
        loss_frame,
    )
    recovery_value = episode["RecoveryFrame"]
    recovery = int(recovery_value) if recovery_value is not None else None
    delay = (
        recovery - loss_frame
        if recovery is not None
        else int(episode["InvalidRows"])
    )
    return {
        "RecoveryFrame": recovery,
        "RecoveryDelayFrames": delay,
        "ExpectedMaximumDelayFrames": timeout + acquire_frames,
        "RecoveryWithinExpectedHorizon": bool(
            recovery is not None and delay <= timeout + acquire_frames
        ),
    }


def layer_token(prefix: str, metrics: Mapping[str, int]) -> str:
    """Encode lost, changed and gained counts compactly."""

    return (
        prefix
        + str(metrics["LostReferenceValidRows"])
        + "."
        + str(metrics["ChangedReferenceValidRows"])
        + "."
        + str(metrics["GainedCandidateValidRows"])
    )


def threshold_token(result: Mapping[str, object]) -> str:
    """Encode one timeout's incremental and total attribution."""

    incremental = result["TimeoutVersusMaskOnly"]
    total = result["TimeoutVersusAccepted"]
    recovery = result["Q6Recovery"]
    passed = bool(result["AttributionPass"])
    return (
        "T"
        + str(result["TimeoutFrames"])
        + "-I"
        + str(incremental["LostReferenceValidRows"])
        + "."
        + str(incremental["ChangedReferenceValidRows"])
        + "."
        + str(incremental["GainedCandidateValidRows"])
        + "-A"
        + str(total["LostReferenceValidRows"])
        + "."
        + str(total["ChangedReferenceValidRows"])
        + "-D"
        + str(recovery["RecoveryDelayFrames"])
        + "-P"
        + ("1" if passed else "0")
    )


def trace_bytes(rows: Sequence[Mapping[str, object]]) -> bytes:
    """Serialize only changed protected-window rows."""

    buffer = io.StringIO(newline="")
    writer = csv.DictWriter(
        buffer,
        fieldnames=CHANGE_TRACE_FIELDS,
        lineterminator="\n",
    )
    writer.writeheader()
    for row in rows:
        writer.writerow({name: row.get(name, "") for name in CHANGE_TRACE_FIELDS})
    return buffer.getvalue().encode("utf-8")


def run(root: Path, program_directory: Path) -> Tuple[str, str, str]:
    """Execute accepted, mask-only and mask-plus-timeout comparison layers."""

    b1b8 = load_b1b8(program_directory)
    base = b1b8.load_base(program_directory)
    workspace = base.require_workspace(root)
    run_root, receipt = base.discover_b1b4(workspace)
    table = base.locate_s1_table(run_root)
    table_hash = base.sha256_file(table)
    configuration = base.load_s1_configuration(run_root)
    accepted, implementation, implementation_hash = base.load_accepted_s1(
        workspace
    )
    source_rows = base.read_source_rows(table)
    original_records, _ = base.normalized_records(
        source_rows,
        mask_sentinel=False,
    )
    masked_records, mask_counts = base.normalized_records(
        source_rows,
        mask_sentinel=True,
    )

    accepted_output = accepted.reconstruct(original_records, configuration)
    base.verify_original_replay(source_rows, accepted_output)
    disabled_output = b1b8.reconstruct_with_timeout(
        original_records,
        configuration,
        b1b8.DISABLED_TIMEOUT,
        accepted,
    )
    base.verify_original_replay(source_rows, disabled_output)

    mask_only_output = accepted.reconstruct(masked_records, configuration)
    if len(mask_only_output) != len(source_rows):
        raise AttributionError("MASK_ONLY_ROW_COUNT")

    original_episode = base.longest_completed_episode(
        base.window_rows(source_rows)
    )
    loss_frame = int(original_episode["StartFrame"])
    windows = b1b8.locate_review_windows(run_root)
    _, protected = b1b8.select_q6_and_protected(windows, loss_frame)

    mask_metrics, mask_trace = compare_layer(
        source_rows,
        mask_only_output,
        protected,
        base,
        "ACCEPTED_TO_MASK_ONLY",
        None,
    )
    all_trace = list(mask_trace)
    results: List[Dict[str, object]] = []
    acquire_frames = int(configuration["AcquireConfirmFrames"])

    for timeout in b1b8.TIMEOUT_VALUES:
        candidate = b1b8.reconstruct_with_timeout(
            masked_records,
            configuration,
            timeout,
            accepted,
        )
        if len(candidate) != len(source_rows):
            raise AttributionError("TIMEOUT_ROW_COUNT", timeout)
        incremental, incremental_trace = compare_layer(
            mask_only_output,
            candidate,
            protected,
            base,
            "MASK_ONLY_TO_TIMEOUT",
            timeout,
        )
        total, total_trace = compare_layer(
            source_rows,
            candidate,
            protected,
            base,
            "ACCEPTED_TO_TIMEOUT",
            timeout,
        )
        recovery = q6_recovery_metric(
            candidate,
            loss_frame,
            timeout,
            acquire_frames,
            base,
        )
        incremental_safe = bool(
            incremental["LostReferenceValidRows"] == 0
            and incremental["ChangedReferenceValidRows"] == 0
        )
        attribution_pass = bool(
            incremental_safe
            and recovery["RecoveryWithinExpectedHorizon"]
        )
        results.append({
            "TimeoutFrames": timeout,
            "TimeoutVersusMaskOnly": incremental,
            "TimeoutVersusAccepted": total,
            "Q6Recovery": recovery,
            "AttributionPass": attribution_pass,
            "PromotionAuthorized": False,
        })
        all_trace.extend(incremental_trace)
        all_trace.extend(total_trace)

    mask_changed = bool(
        mask_metrics["LostReferenceValidRows"]
        or mask_metrics["ChangedReferenceValidRows"]
    )
    timeout_changed = any(
        item["TimeoutVersusMaskOnly"]["LostReferenceValidRows"]
        or item["TimeoutVersusMaskOnly"]["ChangedReferenceValidRows"]
        for item in results
    )
    if mask_changed and timeout_changed:
        attribution_class = "MIXED"
    elif mask_changed:
        attribution_class = "MASK_ONLY_PROTECTED_CHANGE"
    elif timeout_changed:
        attribution_class = "TIMEOUT_INCREMENTAL_PROTECTED_CHANGE"
    else:
        attribution_class = "NO_PROTECTED_REGRESSION"

    summary = {
        "Contract": "UDP002_U22B1B9_LAYERED_REGRESSION_ATTRIBUTION_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": VERSION,
        "RunId": receipt.get("RunId"),
        "SourceTableSha256": table_hash,
        "AcceptedImplementationRelativePath": str(
            implementation.relative_to(workspace)
        ),
        "AcceptedImplementationSha256": implementation_hash,
        "B1B8DependencySha256": B1B8_SHA256,
        "B1B7DependencySha256": b1b8.BASE_SHA256,
        "BaselineDisabledTimeoutEquivalent": True,
        "ReviewWindowCount": len(windows),
        "ProtectedWindowCount": len(protected),
        "OriginalQ6Episode": original_episode,
        "MaskCounts": mask_counts,
        "MaskOnlyVersusAccepted": mask_metrics,
        "TimeoutResults": results,
        "AttributionClass": attribution_class,
        "DecisionStatus": "ATTRIBUTION_ONLY_NO_PROMOTION",
        "InputMutationCount": 0,
    }
    export_root = run_root / "05_EXPORTS"
    base.write_exact(export_root / SUMMARY_NAME, base.canonical_bytes(summary))
    base.write_exact(export_root / TRACE_NAME, trace_bytes(all_trace))
    if base.sha256_file(table) != table_hash:
        raise AttributionError("SOURCE_TABLE_CHANGED")

    main = (
        "U22B1B9 N-W"
        + str(len(windows))
        + "-P"
        + str(len(protected))
        + "-"
        + layer_token("M", mask_metrics)
        + "-T"
        + str(len(results))
        + "-E1-X0 C-0"
    )
    detail_1 = "U22B1B9D1 " + " ".join(
        threshold_token(item) for item in results[:2]
    ) + " C-0"
    detail_2 = "U22B1B9D2 " + " ".join(
        threshold_token(item) for item in results[2:]
    ) + " C-0"
    return main, detail_1, detail_2


def self_test(program_directory: Path) -> str:
    """Exercise dependency identity and all layer-comparison outcomes."""

    checks: List[bool] = []

    def check(value: bool) -> None:
        if not value:
            raise AttributionError("SELFTEST")
        checks.append(True)

    b1b8 = load_b1b8(program_directory)
    base = b1b8.load_base(program_directory)
    check(b1b8.VERSION == "0.1")
    check(base.VERSION == "0.2")
    windows = [{"EventID": "Q001", "StartFrame": 1, "EndFrame": 3}]
    accepted = [
        {
            "FrameCount": 1,
            "CanonicalCameraValid": 1,
            "SelectedCameraRole": "CAMERA_1_DISTANCE",
            "CanonicalCameraDistance": "10.0",
            "SelectionStatus": "CONFIDENT",
        },
        {
            "FrameCount": 2,
            "CanonicalCameraValid": 1,
            "SelectedCameraRole": "CAMERA_1_DISTANCE",
            "CanonicalCameraDistance": "11.0",
            "SelectionStatus": "CONFIDENT",
        },
        {
            "FrameCount": 3,
            "CanonicalCameraValid": 0,
            "SelectedCameraRole": "NONE",
            "CanonicalCameraDistance": "",
            "SelectionStatus": "INVALID",
        },
    ]
    mask_only = [
        {
            **accepted[0],
            "CanonicalCameraValid": 0,
            "CanonicalCameraDistance": "",
            "SelectionStatus": "INVALID",
        },
        {
            **accepted[1],
            "SelectedCameraRole": "CAMERA_3_DISTANCE",
        },
        dict(accepted[2]),
    ]
    timeout = [
        dict(mask_only[0]),
        dict(mask_only[1]),
        {
            **accepted[2],
            "CanonicalCameraValid": 1,
            "SelectedCameraRole": "CAMERA_3_DISTANCE",
            "CanonicalCameraDistance": "12.0",
            "SelectionStatus": "CONFIDENT",
        },
    ]
    mask_metrics, mask_trace = compare_layer(
        accepted,
        mask_only,
        windows,
        base,
        "A_B",
        None,
    )
    check(mask_metrics["ReferenceValidRows"] == 2)
    check(mask_metrics["LostReferenceValidRows"] == 1)
    check(mask_metrics["ChangedReferenceValidRows"] == 1)
    check(mask_metrics["GainedCandidateValidRows"] == 0)
    check(len(mask_trace) == 2)
    incremental, incremental_trace = compare_layer(
        mask_only,
        timeout,
        windows,
        base,
        "B_C",
        5,
    )
    check(incremental["ReferenceValidRows"] == 1)
    check(incremental["LostReferenceValidRows"] == 0)
    check(incremental["ChangedReferenceValidRows"] == 0)
    check(incremental["GainedCandidateValidRows"] == 1)
    check(len(incremental_trace) == 1)
    total, total_trace = compare_layer(
        accepted,
        timeout,
        windows,
        base,
        "A_C",
        5,
    )
    check(total["LostReferenceValidRows"] == 1)
    check(total["ChangedReferenceValidRows"] == 1)
    check(total["GainedCandidateValidRows"] == 1)
    check(len(total_trace) == 3)
    check(layer_token("M", mask_metrics) == "M1.1.0")
    token = threshold_token({
        "TimeoutFrames": 5,
        "TimeoutVersusMaskOnly": incremental,
        "TimeoutVersusAccepted": total,
        "Q6Recovery": {"RecoveryDelayFrames": 9},
        "AttributionPass": True,
    })
    check(token == "T5-I0.0.1-A1.1-D9-P1")
    check(len(trace_bytes(mask_trace)) > 100)
    check(stable("a/b c") == "A_B_C")
    check(len(checks) == 20)
    return "U22B1B9-ST N-T21-X0 C-0"


def write_failure(root: Optional[Path], error: AttributionError) -> None:
    """Persist detailed local evidence while returning one short capsule."""

    if root is None or not root.is_dir():
        return
    payload = {
        "Contract": "UDP002_U22B1B9_FAILURE_DIAGNOSTIC_V0_1",
        "ProgramId": PROGRAM_ID,
        "ProgramVersion": VERSION,
        "Reason": error.reason,
        "Detail": error.detail,
        "NextAction": "Return only the bounded C-7 capsule.",
    }
    try:
        path = root / DIAGNOSTIC_RELATIVE
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_text(
            json.dumps(payload, ensure_ascii=False, sort_keys=True, indent=2)
            + "\n",
            encoding="utf-8",
        )
    except Exception:
        pass


def parser() -> argparse.ArgumentParser:
    """Build the bounded command-line interface."""

    result = argparse.ArgumentParser(description=PROGRAM_ID)
    result.add_argument("--workspace-root", type=Path)
    result.add_argument("--self-test", action="store_true")
    return result


def main(argv: Optional[Sequence[str]] = None) -> int:
    """Execute one controlled mode and print at most three lines."""

    arguments = parser().parse_args(argv)
    program_directory = Path(__file__).resolve().parent
    root: Optional[Path] = None
    try:
        if arguments.self_test:
            print(self_test(program_directory))
            return 0
        if arguments.workspace_root is None:
            raise AttributionError("WORKSPACE_ARGUMENT")
        root = arguments.workspace_root.resolve()
        for capsule in run(root, program_directory):
            print(capsule)
        return 0
    except AttributionError as exc:
        write_failure(root, exc)
        print("U22B1B9 N-X1 C-7 K-" + exc.reason + " D-" + exc.detail)
        return 7
    except Exception as exc:
        wrapped = AttributionError("INTERNAL", type(exc).__name__)
        write_failure(root, wrapped)
        print("U22B1B9 N-X1 C-7 K-INTERNAL D-" + stable(type(exc).__name__))
        return 7


if __name__ == "__main__":
    sys.exit(main())
